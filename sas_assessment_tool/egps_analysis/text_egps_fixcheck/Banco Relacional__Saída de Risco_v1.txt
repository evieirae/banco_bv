PROC SQL;

  CONNECT TO bigquery AS perm (project='bv-sase-dtm' credfile='/projetos/acessos/bq/e/sase-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);

  CREATE TABLE work.GCP_base_saida_risco AS
    SELECT
      CdDimContaCartao,
      Class,
      CampParcFat,
      SafraRef
    FROM connection to perm
  (
    with Base AS (
      select
        cast(cd_dim_conta_cartao as int64) as CdDimContaCartao,
        ds_class_reneg_adim           as Class,
        vr_nova_tx_juros              as CampParcFat,
        ts_inclusao_analitico         as SafraRef
      from `bv-pdbd-prd.dvry_corp_credito.visaocliente_saidaderisco_dime`
      where ds_produto = 'CARTOES'
        and vr_nova_tx_juros in (6.99,10.99,9.99,9.49,8.99,13.99,7.99,12.99)
    ),

    ContaRowsSafraRef AS (
      select
        SafraRef,
        count(*) as QTDRows
      from Base
      group by SafraRef
      having QTDRows >= 80000
    ),

    TargetSafraRef AS (
      select max(SafraRef) as SafraRef
      from ContaRowsSafraRef
    )
    select
      CdDimContaCartao,
      Class,
      CampParcFat,
      SafraRef
    from Base
    where SafraRef = (select SafraRef from TargetSafraRef)
  );

  DISCONNECT FROM perm;
QUIT;

  length TaxaBAU $6;

PROC SQL;
CREATE TABLE work.base_saida_risco_1 AS
SELECT
a.CdDimContaCartao
,a.Class
,b.Campanha    							AS CampParcFat
FROM work.base_fmt a
INNER JOIN work.depara_clean b
    on a.CampParcFat_char = b.TaxaBAU;
QUIT;

PROC SQL;
  create table work._counts_camp as
  select CampParcFat, count(*) as cnt
  from work.base_saida_risco_1
  group by CampParcFat;
QUIT;

PROC SORT data=work.base_saida_risco_1; by CampParcFat; run;
DATA work._with_seq;
  merge work.base_saida_risco_1(in=a) work._counts_camp(in=b);
  by CampParcFat;
  if a;
  retain seq;
  if first.CampParcFat then seq = 0;
  seq + 1;
  frac_key = seq / cnt;            
  neg_cnt  = -cnt;                
RUN;

PROC SORT data=work._with_seq;
  by frac_key neg_cnt CampParcFat CdDimContaCartao;
RUN;

DATA work.base_saida_risco;
  set work._with_seq;
  drop seq frac_key neg_cnt cnt;
RUN;

PROC FREQ data=work.base_saida_risco;
  tables CampParcFat / nocum nopercent;
RUN;

PROC SQL;
CREATE TABLE work.nao_mapeados AS

a.*
FROM work.base_fmt a

    select 1
    from work.depara_clean as b
    where a.CampParcFat_char = b.TaxaBAU
