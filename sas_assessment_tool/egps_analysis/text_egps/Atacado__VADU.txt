	CREATE TABLE TBCARTEIRAGRUPO AS SELECT DISTINCT

		CdGrupoEconomico,

PROC SQL;

   CREATE TABLE QUERY_FOR_TBBRUTA AS SELECT DISTINCT

			t1.CdGpEconomico,
	     	t1.NmGpEconomico,
	      	t1.NmPessoa,
	       	t1.NuCpfCnpjRaiz,
	        t1.NuCpfCnpj,
	        t1.CdPessoa,
			t2.NmAnalistaCredito,
			t2.NmGerenteCredito,
	        CATX('',t2.NmAnalistaCredito,'@BV.COM.BR') AS email_analista,
	        CATX('',t2.NmGerenteCredito,'@BV.COM.BR') AS email_gerente,
	        CATX('',t2.NmGerenteExecutivo,'@BV.COM.BR') AS email_gerenteexec,
	        t1.NmOfficer,
			t1.VrFaturamentoP,
			t1.VrFaturamentoGP

      FROM _INSIGHT.TBBRUTA t1
	  LEFT JOIN TBCARTEIRAGRUPO t2 ON(t1.CdGpEconomico = t2.CdGrupoEconomico)

      WHERE t1.TpLancamento = 'Pessoa' AND
			t1.TpPessoa = 'J' AND

			t1.CdGpEconomico NOT = . AND
           	t1.NuCpfCnpjRaiz NOT = '00000000' AND
			t1.NuCpfCnpj NOT = '00000000000000' AND
			t2.NmAnalistaCredito NOT IN ('S/INF' 'MASSIFICADA' '')

;QUIT;
