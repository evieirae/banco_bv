/* =================== Extracao PTA  ====================== */
/* ======================================================== */

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
CREATE TABLE extracao_liquidacao AS
SELECT * FROM connection to dbcon
(

			SELECT
			 year(Contrato.DtLiquidacaoEfetiva)*100+month(Contrato.DtLiquidacaoEfetiva) AS NuAnoMes
			,Contrato.DtLiquidacaoEfetiva
			,ContratoFinanceiro.NuContrato
			,ContratoFinanceiro.SqContratoFinanceiro
			,Contrato.DtContrato
			,Contrato.CdParceiroComercial
			,Proposta.CdPessoa CdParceiroComercialCoban
			,Contrato.CdProduto*1000000 + Contrato.CdModalidadeProduto*1000 + coalesce(ObjGarantiaBem.CdTipoBem,0) CdGrupoSubProduto
			,Contrato.CdOperador CdColaborador
			,case
				when Contrato.CdProduto = 11 then coalesce(Colaborador.CdFilial,Contrato.CdFilial)
			 else Contrato.CdFilial end as CdFilial
			,case
				when ParcelaRecebimento.CdAgenteRecebedor=195 then 'Portabilidade'
				when ParcelaRecebimento.CdAgenteRecebedor in (239,105,120) then 'Repasse'
			else 'Boleto' end as TpLiquidacao
			,Contrato.NuCpfCnpj
			,ContratoFinanceiro.QtParcelas
			,ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada VrProducao
			,sum(case when datediff(month,Contrato.DtLiquidacaoEfetiva,ParcelaRecebimento.DtRecebimento) = 0 THEN 1 ELSE 0 END) AS QtParcelaAntecipada
			,sum(case when datediff(month,Contrato.DtLiquidacaoEfetiva,ParcelaRecebimento.DtRecebimento) = 0 THEN ParcelaRecebimento.VrRecebido ELSE 0 END) AS VrRecebidoAntecipado

		FROM DBA.VwTbContrato Contrato

			LEFT JOIN CDCCG.TbObjGarantiaBemVeiculo ObjGarantiaBemVeiculo
				ON  Contrato.NuContrato = ObjGarantiaBemVeiculo.NuContrato
				AND	ObjGarantiaBemVeiculo.SqObjGarantia = 1

			LEFT JOIN CDCCG.TbObjGarantiaBem ObjGarantiaBem
				ON	Contrato.NuContrato =	ObjGarantiaBem.NuContrato
				AND	ObjGarantiaBem.SqObjGarantia =	1

			LEFT JOIN DBA.VwTbContratoFinanceiro ContratoFinanceiro
				ON  Contrato.NuContrato = ContratoFinanceiro.NuContrato
				AND Contrato.SqContratoFinanceiroAtivo = ContratoFinanceiro.SqContratoFinanceiro

				LEFT JOIN DBA.TbColaborador Colaborador
					ON  Contrato.CdOperador = Colaborador.CdColaborador
					AND Colaborador.CdColaborador NOT IN (503,9777,11046,11210,18512,57936)
					AND Colaborador.CdOperador <> 0

			LEFT JOIN DBA.vwTbParcela Parcela
				ON  Contrato.NuContrato = Parcela.NuContrato
				AND Contrato.SqContratoFinanceiroAtivo = Parcela.SqContratoFinanceiro

				LEFT JOIN DBA.vwTbParcelaRecebimento ParcelaRecebimento
					ON  Parcela.NuContrato = ParcelaRecebimento.NuContrato
					AND Parcela.SqContratoFinanceiro = ParcelaRecebimento.SqContratoFinanceiro
					AND Parcela.NuParcela = ParcelaRecebimento.NuParcela

			LEFT JOIN CPC.TbContratoRefinanciado ContratoRefinanciado
				ON Contrato.NuContrato = ContratoRefinanciado.NuContratoRefinanciado

			LEFT JOIN DBA.TbProposta Proposta
				ON  Contrato.NuContrato = Proposta.NuContratoGestao

				INNER JOIN (SELECT NuContratoGestao,max(NuProposta) NuProposta FROM DBA.TbProposta GROUP BY NuContratoGestao) PropostaMax
					ON  Proposta.NuProposta = PropostaMax.NuProposta

		WHERE Parcela.DtVencimento > ParcelaRecebimento.DtRecebimento
		  AND ParcelaRecebimento.DtRecebimento >= today()-5
		  AND ParcelaRecebimento.DtRecebimento  < today()
		  AND ContratoRefinanciado.NuContratoRefinanciado IS NULL
		  AND ParcelaRecebimento.CdAgenteRecebedor IN (104,105,120,109,125,147,155,195,239,241)
		  AND datediff(month,Contrato.DtLiquidacaoEfetiva,ParcelaRecebimento.DtRecebimento) = 0

		GROUP BY

			 year(Contrato.DtLiquidacaoEfetiva)*100+month(Contrato.DtLiquidacaoEfetiva)
			,Contrato.DtLiquidacaoEfetiva
			,ContratoFinanceiro.NuContrato
			,ContratoFinanceiro.SqContratoFinanceiro
			,Contrato.DtContrato
			,Contrato.CdParceiroComercial
			,Proposta.CdPessoa
			,Contrato.CdProduto*1000000 + Contrato.CdModalidadeProduto*1000 + coalesce(ObjGarantiaBem.CdTipoBem,0)
			,Contrato.CdOperador
			,case
				when Contrato.CdProduto = 11 then coalesce(Colaborador.CdFilial,Contrato.CdFilial)
		 	 else Contrato.CdFilial end
			,case
				when ParcelaRecebimento.CdAgenteRecebedor=195 then 'Portabilidade'
				when ParcelaRecebimento.CdAgenteRecebedor in (239,105,120) then 'Repasse'
			 else 'Boleto' end
			,Contrato.NuCpfCnpj
			,ContratoFinanceiro.QtParcelas
			,ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada

		ORDER BY
			 ContratoFinanceiro.NuContrato
			,ContratoFinanceiro.SqContratoFinanceiro

)
;RUN;

/* ======================================================== */
/* ============== TbDimHierarquiaComercial ================ */
/* ======================================================== */

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
create table TbFilial as
select * from connection to dbcon
(

	SELECT
		 CdFilial
		,DsFilial

	FROM DBA.TbFilial
)
;run;

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbDimHierarquiaComercial as
select * from connection to dbcon
(
	with TbDeParaFilial as
	(

		SELECT
			 DeParaFilial.CdAreaEstruturaComercial
			,DeParaFilial.CdFilial
			,Filial.NmFilial
			,CASE
				WHEN Filial.NmFilial LIKE 'CP SOL%' THEN 5
				WHEN Filial.NmFilial LIKE 'CP CONS%' THEN 4
				WHEN DeParaFilial.CdFilial IN (275,276) THEN 3
				WHEN Filial.NmFilial LIKE 'CP %' THEN 2
				WHEN Filial.NmFilial NOT LIKE 'CP ' THEN 1
			 ELSE 0 END AS CdCanalEstruturaComercial

		FROM DWBI_DTWH.TbDeParaFilial DeParaFilial

			LEFT JOIN DWBI_DTWH.TbFilial Filial
				ON  DeParaFilial.CdFilial = Filial.CdFilialSistemaOrigem
				AND Filial.CdSistemaOrigem = 209

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial Area
				ON  DeParaFilial.CdAreaEstruturaComercial = Area.CdAreaEstruturaCmrclStmaOrgm

	)

	SELECT
		 CAST (Pessoa.CdPessoaOrigem AS INTEGER) CdOperador
		,TempoData.DtTempo
		,DeParaFilial.CdFilial

	FROM DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoa

		LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
			ON  PessoaRelacaoPessoa.CdPessoa = Pessoa.CdPessoa

		INNER JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
			ON  PessoaRelacaoPessoa.CdPessoaRelacao = HierarquiaComercial.CdColaborador

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
				ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

			LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
				ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

				INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
					ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
					AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador
					AND DeParaHierarquia.CdCanalEstruturaComercial in (2,4)

				INNER JOIN TbDeParaFilial DeParaFilial
					ON  DeParaHierarquia.CdAreaEstruturaComercial = DeParaFilial.CdAreaEstruturaComercial
					AND DeParaHierarquia.CdCanalEstruturaComercial = DeParaFilial.CdCanalEstruturaComercial

			INNER JOIN DWBI_DIME.TbDimTempoData TempoData
				ON  TempoData.DtTempo BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
				AND TempoData.DtTempo > '20000101'
				AND TempoData.DtTempo < today()

	WHERE PessoaRelacaoPessoa.CdTipoRelacaoPessoa = 7
	  AND today() BETWEEN PessoaRelacaoPessoa.DtInicioVigencia AND PessoaRelacaoPessoa.DtFimVigencia

)
;run;

proc sql;
create table TbGrVirtual as
select
	 Colaborador.CdColaborador as CdOperador
	,TempoData.DtTempo
	,GrVirtualVigencia.CdFilial

from bases.TbColaborador Colaborador

	left join bases.TbTempoData TempoData
		on 1=1

	inner join basegrc.TbGrVirtualVigencia GrVirtualVigencia
		on  Colaborador.CdColaborador = GrVirtualVigencia.CdColaborador
		and TempoData.DtTempo between GrVirtualVigencia.DtInicioVigencia and GrVirtualVigencia.DtFimVigencia

order by
	 Colaborador.CdColaborador
	,TempoData.DtTempo
;run;

data TbDimHierarquiaComercial;
set
	TbDimHierarquiaComercial
	TbGrVirtual
;run;

proc sql;
create table TbDimHierarquiaComercial as
select
	 DimHierarquiaComercial.CdOperador
	,DimHierarquiaComercial.DtTempo
	,DimHierarquiaComercial.CdFilial
	,Filial.DsFilial
from TbDimHierarquiaComercial DimHierarquiaComercial

	left join TbFilial Filial
		on  DimHierarquiaComercial.CdFilial = Filial.CdFilial

order by 1,2
;run;

proc sql;
create table TbExtLiquidacao as
select
	 ExtLiquidacao.NuAnoMes
	,ExtLiquidacao.DtLiquidacaoEfetiva
	,ExtLiquidacao.NuContrato
	,ExtLiquidacao.SqContratoFinanceiro
	,ExtLiquidacao.DtContrato
	,ExtLiquidacao.CdParceiroComercial
	,ExtLiquidacao.CdParceiroComercialCoban
	,ExtLiquidacao.CdGrupoSubProduto
	,ExtLiquidacao.CdColaborador
	,coalesce(HierarquiaComercial.CdFilial,ExtLiquidacao.CdFilial) as CdFilial
	,ExtLiquidacao.TpLiquidacao
	,ExtLiquidacao.VrProducao
	,ExtLiquidacao.NuCpfCnpj
	,ExtLiquidacao.QtParcelas
	,ExtLiquidacao.QtParcelas*(ExtLiquidacao.VrProducao) as VpPrazo
	,ExtLiquidacao.QtParcelaAntecipada
	,ExtLiquidacao.QtParcelaAntecipada*(ExtLiquidacao.VrProducao) as VpPrazoAntecipado
	,ExtLiquidacao.VrRecebidoAntecipado
from extracao_liquidacao ExtLiquidacao

	left join TbDimHierarquiaComercial HierarquiaComercial
		on  datepart(ExtLiquidacao.DtContrato) = HierarquiaComercial.DtTempo
		and ExtLiquidacao.CdColaborador = HierarquiaComercial.CdOperador
order by
	 ExtLiquidacao.NuContrato
	,ExtLiquidacao.SqContratoFinanceiro
;run;

data bases.TbExtLiquidacao;
	update
		bases.TbExtLiquidacao
		TbExtLiquidacao
	;by NuContrato SqContratoFinanceiro;
run;
