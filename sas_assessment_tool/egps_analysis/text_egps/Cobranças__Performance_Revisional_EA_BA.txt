PROC SQL;
CREATE TABLE BASE_REVI AS
SELECT
	A.SAFRA,
	'REVISIONAL' AS FASE,
	A.ASSESSORIA_FINAL AS ASSESSORIA,
	'RECUPERADO' AS TIPO,
	SUM(A.VLR_RECEB) AS VALOR

FROM REVISIONAL AS A
JOIN (SELECT DISTINCT
				SAFRA,
				MAX(DT_REF) AS DT_REF FORMAT DDMMYY10.
				FROM REVISIONAL
				GROUP BY
					SAFRA
				) AS B
ON A.SAFRA = B.SAFRA
AND A.DT_REF = B.DT_REF
GROUP BY
	A.SAFRA,
	A.ASSESSORIA_FINAL
UNION
SELECT
	SAFRA,
	'REVISIONAL' AS FASE,
	ASSESSORIA,
	'META' AS TIPO,
	META
FROM meta_revI
WHERE SAFRA >= 202402
AND ASSESSORIA <> 'BV'
;RUN;
