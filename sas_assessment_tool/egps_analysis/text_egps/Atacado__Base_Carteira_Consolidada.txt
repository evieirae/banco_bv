PROC SQL;
   CREATE TABLE VENCIMENTARIO AS
   SELECT
          'Carteira' as TIPO,
		  'Vencimentário' as View,
		  t1.DsTipoCarteira,
          t1.AnoMes_Referencia,
          t1.CdClienteRisco,
          t1.DsClienteRisco,
          t1.CdGrupo_Economico,
          t1.DsGrupoEconomicoRisco,
	      CASE
	            WHEN t1.DsGrupoProduto IN (
	                'Empréstimos', 'Trade Finance', 'External Trade Finance', 'BNDES',
	                'External Loans', 'Crédito Rural', 'Leasing', 'Créditos do Agronegócio'
	            ) THEN 'Crédito'
	            WHEN t1.DsGrupoProduto = 'Operações de Garantias' THEN 'Fiança'
	            WHEN t1.DsGrupoProduto = 'TVM' THEN 'TVM'
	            WHEN t1.DsGrupoProduto IN ('Mercado Bancário', 'Hedge') THEN 'Derivativos'
	            ELSE 'NA'
	       END AS DsGrupoProduto,
          t1.CdModalidadeProduto,
          t1.DsProduto,
          t1.DtInicioOperacao,
          t1.DtVencimentoParcela,
          t1.NumDiasVencidos,
          t1.VrSaldo,
          t1.PDEstagio,
          t1.PcLGD,
          t1.VrGarantiaLGD,
          t1.PcLgdElbe,
          t1.DtAtivoProblematico,
          t1.Pd12Concessao,
          t1.Pd12Atual,
          t1.PDLifeTime,
          t1.PerdaIncorrida,
          t1.PerdaEsperada,
          t1.VrProvisao,
	        CASE
	            WHEN (
	                DsTipoCarteira IN ("REGULAR", "OPAC", "COMISSAO_FIANÇA")
	                OR (
	                    DsTipoCarteira = "VOTORANTIM" AND CdModalidadeProduto NOT IN (368, 205, 268, 269, 273, 277, 358, 208)
	                )
	                OR (
	                    CdModalidadeProduto IN (208, 210) AND DsGrupoEconomicoRisco NOT IN ("VOTORANTIM FINANCAS", "VOTORANTIM INDUSTRIAL")
	                )
	            )
	            AND CdModalidadeProduto NE 80
	            AND DsGrupoEconomicoRisco NOT LIKE "%DEB SEC%"
	            AND DsGrupoEconomicoRisco NE "FIDC TRADEMASTER"
	            AND calculated DsGrupoProduto NE "Derivativos"
	            AND NuEstagioFinal IS NOT NULL
	            THEN 1
	            ELSE 0
	        END AS flag_pdd

      FROM WORK.BASE_VENCIMENTARIO t1
	  left join INSIGHT.DE_PARA_MACRO_SETOR T2 ON(T1.DsSetorClienteRisco_novo = T2.Setor_Credito)

	  ORDER BY  T1.AnoMes_Referencia DESC

;QUIT;

PROC SQL;
   CREATE TABLE RESUMO AS
   SELECT

          'Carteira' as TIPO,
		  t1.DsTipoCarteira,
          t1.AnoMes_Referencia,
          t1.CdClienteRisco,
          t1.DsClienteRisco,
          t1.CdGrupo_Economico,
          t1.DsGrupoEconomicoRisco,
	      CASE
	            WHEN t1.DsGrupoProduto IN (
	                'Empréstimos', 'Trade Finance', 'External Trade Finance', 'BNDES',
	                'External Loans', 'Crédito Rural', 'Leasing', 'Créditos do Agronegócio'
	            ) THEN 'Crédito'
	            WHEN t1.DsGrupoProduto = 'Operações de Garantias' THEN 'Fiança'
	            WHEN t1.DsGrupoProduto = 'TVM' THEN 'TVM'
	            WHEN t1.DsGrupoProduto IN ('Mercado Bancário', 'Hedge') THEN 'Derivativos'
	            ELSE 'NA'
	       END AS DsGrupoProduto,
          t1.CdModalidadeProduto,
          t1.DsProduto,
          SUM(t1.VrSaldo) AS VrSaldo,
          t1.PDEstagio,
          t1.PcLGD,
          sum(t1.VrGarantiaLGD) as VrGarantiaLGD,
          t1.DtAtivoProblematico,
          t1.Pd12Concessao,
          t1.Pd12Atual,
          t1.PDLifeTime,
          SUM(t1.PerdaIncorrida) AS PerdaIncorrida,
          SUM(t1.PerdaEsperada) AS PerdaEsperada,
          SUM(t1.VrProvisao) AS VrProvisao,
	        CASE
	            WHEN (
	                DsTipoCarteira IN ("REGULAR", "OPAC", "COMISSAO_FIANÇA")
	                OR (
	                    DsTipoCarteira = "VOTORANTIM" AND CdModalidadeProduto NOT IN (368, 205, 268, 269, 273, 277, 358, 208)
	                )
	                OR (
	                    CdModalidadeProduto IN (208, 210) AND DsGrupoEconomicoRisco NOT IN ("VOTORANTIM FINANCAS", "VOTORANTIM INDUSTRIAL")
	                )
	            )
	            AND CdModalidadeProduto NE 80
	            AND DsGrupoEconomicoRisco NOT LIKE "%DEB SEC%"
	            AND DsGrupoEconomicoRisco NE "FIDC TRADEMASTER"
	            AND calculated DsGrupoProduto NE "Derivativos"
	            AND NuEstagioFinal IS NOT NULL
	            THEN 1
	            ELSE 0
	      END AS flag_pdd

      FROM WORK.BASE_RESUMO t1

	  GROUP BY  TIPO,
				t1.DsTipoCarteira,
				t1.AnoMes_Referencia,
				t1.CdClienteRisco,
				t1.DsClienteRisco,
				t1.CdGrupo_Economico,
				t1.DsGrupoEconomicoRisco,
				calculated DsGrupoProduto,
				t1.CdModalidadeProduto,
				t1.DsProduto,
				t1.PDEstagio,
				t1.PcLGD,
				t1.DtAtivoProblematico,
				t1.Pd12Concessao,
				t1.Pd12Atual,
				t1.PDLifeTime,
				calculated flag_pdd

	  ORDER BY  T1.AnoMes_Referencia DESC

;QUIT;

PROC SQL;
   CREATE TABLE DIMENSAO_BRUTA AS SELECT DISTINCT

		t1.AnoMes_Referencia,
        t1.CdPessoa as CdClienteRisco,
        t1.CdGpEconomico as CdGrupo_Economico,


        case when t1.FlSuspensoGrupo eq 1 then 1 else
			case when t1.FlSuspensoP eq 1 then 1 else 0 end end as Flag_suspenso,
		t1.FlLRCliente,
		t1.FlPLCA,
        t1.VrFaturamentoP FORMAT=BESTX18. as VrFaturamentoP,
		t1.VrFaturamentoGP FORMAT=BESTX18. as VrFaturamentoGP,
		CASE WHEN ((t1.DtVencimentoGLCliente >= T1.DtProcessamento - 180 AND T1.VrTotalAprovadoGLCliente > 0) OR T1.VrTotalRiscoGLCliente > 0) AND B.NmGerenteCredito not in ('RODRIGO.ACASTRO' 'WENDEL.CRUZ') THEN 1 ELSE 0 END AS FlClienteCIB,
		CASE WHEN ((t1.DtVencimentoGLCliente >= T1.DtProcessamento - 180 AND T1.VrTotalAprovadoGLCliente > 0) OR T1.VrTotalRiscoGLCliente > 0) AND B.NmGerenteCredito = 'RODRIGO.ACASTRO' THEN 1 ELSE 0 END AS FlClienteUPPM,
		CASE WHEN CALCULATED FlClienteCIB = 1 OR CALCULATED FlClienteUPPM = 1 OR (t1.FlPLCA EQ 1 and ((t1.DtVencLimiteGrupo > t1.DtProcessamento or t1.DtVencLimiteCliente > t1.DtProcessamento) and (t1.VrTotalAprovadoGrupo > 0 or t1.VrTotalAprovadoCliente >0))) THEN 1 ELSE 0 END AS FlAtivo

	  FROM TBBRUTA AS t1
	  LEFT JOIN TBCARTEIRAGRUPOHIST AS B ON( T1.CdGpEconomico = B.CdGrupoEconomico AND T1.AnoMes_Referencia = B.AnoMes_Referencia)
	  WHERE B.AnoMes_Referencia NOT IS NULL OR T1.FlPLCA EQ 1
      ORDER BY T1.CdGpEconomico, T1.AnoMes_Referencia

;QUIT;

	proc sort data=DIMENSAO_BRUTA;
    by AnoMes_Referencia CdClienteRisco;
	run;

	data DIMENSAO_BRUTA;
    set DIMENSAO_BRUTA;
    by AnoMes_Referencia CdClienteRisco;
    if first.CdClienteRisco;
	run;

	case when a.CdGrupo_Economico eq . then a.VrTotalRiscoCliente else max(a.VrTotalRiscoGrupo) / count(a.TIPO) end as Risco_total_ajust

	WHERE NOT (A.TIPO EQ 'Limite' AND B.FlAtivo IN (0 .))

				A.TIPO,
				A.AnoMes_Referencia,

        ds_versao_modelo_aplicado,
        case
            when cd_rating_final = 'AAA'                         then 1
            when cd_rating_final = 'AA+'                         then 2
            when cd_rating_final = 'AA'                          then 3
            when cd_rating_final = 'AA-'                         then 4
            when cd_rating_final = 'A+'                          then 5
            when cd_rating_final = 'A'                           then 6
            when cd_rating_final = 'A-'                          then 7
            when cd_rating_final = 'BBB+'                        then 8
            when cd_rating_final = 'BBB'                         then 9
            when cd_rating_final = 'BBB-'                        then 10
            when cd_rating_final = 'BB+'                         then 11
            when cd_rating_final = 'BB'                          then 12
            when cd_rating_final = 'BB-'                         then 13
            when cd_rating_final = 'B+'                          then 14
            when cd_rating_final = 'B'                           then 15
            when cd_rating_final = 'B-'                          then 16
            when cd_rating_final = 'CCC+'                        then 17
            when cd_rating_final = 'CCC'                         then 18
            when cd_rating_final = 'CCC-'                        then 19
            when cd_rating_final = 'CC+'                         then 20
            when cd_rating_final = 'CC'                          then 21
            when cd_rating_final = 'CC-'                         then 22
            when cd_rating_final = 'C+'                          then 23
            when cd_rating_final = 'C'                           then 24
            when cd_rating_final = 'C-'                          then 25
            when cd_rating_final = 'D'                           then 26
            when cd_rating_final in ('D1','DC1','Dc1')           then 27
            when cd_rating_final in ('D2','DC2','Dc2')           then 28
            when cd_rating_final in ('D3','DC3','Dc3')           then 29
            when cd_rating_final = 'D4'                          then 30
            when cd_rating_final in ('E+','E','E-','F','G','H')  then 31
            when cd_rating_final = 'DF'                          then 32
            when cd_rating_final in ('WL','WL1')                 then 33
            else 0
        end as NuRatingP,
        year(calculated DtInicioRatingP)*100 + month(calculated DtInicioRatingP) as ANOMES_Inicio,
        year(calculated DtFimRatingP)*100 + month(calculated DtFimRatingP)       as ANOMES_Fim
    from rating_gcp_hist
    group by cd_cliente, calculated ANOMES_Inicio
    having calculated DtInicioRatingP >= '01JAN2022'd
       and max(ts_inicio_vigencia) = ts_inicio_vigencia
    order by cd_cliente, ts_inicio_vigencia
    ;

data rating_gcp_hist_AJUST_full_ffill;
    merge rating_gcp_hist_AJUST_full(in=a)
          _max_anomes_por_cliente(in=b);
    by cd_cliente;

    
    retain last_cd_calculo_rating last_cd_grupo_economico last_NuRatingP
           last_DsRatingP last_ds_versao_modelo_aplicado
           last_DtInicioRatingP last_DtFimRatingP
           have_value;

    
    length last_DsRatingP $110
           last_ds_versao_modelo_aplicado $110;

    
    format DtInicioRatingP DtFimRatingP ddmmyy10.;

    
    if first.cd_cliente then do;
        call missing(of last_:);
        have_value = 0;
    end;

    
    if flag_linha_nula = 0 then do;
        if not missing(cd_calculo_rating)         then last_cd_calculo_rating         = cd_calculo_rating;
        if not missing(cd_grupo_economico)        then last_cd_grupo_economico        = cd_grupo_economico;
        if not missing(NuRatingP)            then last_NuRatingP            = NuRatingP;

        if not missing(DsRatingP)                 then last_DsRatingP                 = DsRatingP;
        if not missing(ds_versao_modelo_aplicado) then last_ds_versao_modelo_aplicado = ds_versao_modelo_aplicado;

        if not missing(DtInicioRatingP)           then last_DtInicioRatingP           = DtInicioRatingP;
        if not missing(DtFimRatingP)              then last_DtFimRatingP              = DtFimRatingP;

        have_value = 1;
    end;

    
    length flag_ffill 3;
    flag_ffill = 0;

    
    if flag_linha_nula = 1 and have_value = 1 and ANOMES > ANOMES_MAX then do;
        if missing(cd_calculo_rating)         then cd_calculo_rating         = last_cd_calculo_rating;
        if missing(cd_grupo_economico)        then cd_grupo_economico        = last_cd_grupo_economico;
        if missing(NuRatingP)            then NuRatingP            = last_NuRatingP;

        if missing(DsRatingP)                 then DsRatingP                 = last_DsRatingP;
        if missing(ds_versao_modelo_aplicado) then ds_versao_modelo_aplicado = last_ds_versao_modelo_aplicado;

        if missing(DtInicioRatingP)           then DtInicioRatingP           = last_DtInicioRatingP;
        if missing(DtFimRatingP)              then DtFimRatingP              = last_DtFimRatingP;

        flag_ffill = 1;
    end;

    
run;

PROC SQL;
	CREATE TABLE Rating_ESG1 AS SELECT DISTINCT
		YEAR(a.DtInicioVigenciaRating)*100 + MONTH(a.DtInicioVigenciaRating) AS ANOMES_Inicio,
		YEAR(a.DtFimVigenciaRating)*100 + MONTH(a.DtFimVigenciaRating) AS ANOMES_Fim,
		a.*,
		b.CdPessoa,
		b.CdGpEconomico,
		b.NmGpEconomico

	FROM Rating_ESG as a
	LEFT JOIN INSIGHT.TBPESSOA as b ON( A.NuCnpj = B.NuCpfCnpj )
	WHERE a.'Rating ESG'N IS NOT NULL and b.CdPessoa IS NOT NULL and DtInicioVigenciaRating IS NOT NULL
;QUIT;



	data Rating_ESG2;
    set Rating_ESG1;

    
    dt_ini = input(cats(ANOMES_Inicio,'01'), yymmdd8.);
    dt_fim = input(cats(ANOMES_Fim   ,'01'), yymmdd8.);

    
    

    
    dt = dt_ini;
    do while (dt <= dt_fim);
        
        ANOMES = input(put(dt, yymmn6.), 6.);

        

        output;

        
        dt = intnx('month', dt, 1, 's');
    end;

    drop dt_ini dt_fim dt;

run;

	proc sort data= Rating_ESG2;
    by descending CdPessoa anomes descending DtFimVigenciaRating descending DtInicioVigenciaRating ;
	run;

	data INSIGHT.Rating_ESG;
    set Rating_ESG2;
    by descending CdPessoa anomes descending DtFimVigenciaRating descending DtInicioVigenciaRating ;
    if first.ANOMES;
	run;
