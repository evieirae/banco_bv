PROC SQL;
    CREATE TABLE emer.CT_PROP_ANALITICO(compress=yes) AS
        SELECT

			, upcase(T3.Publico) as PUBLICO
			, upcase(T3.EPS) as EPS
			, upcase(T3.Perfil) as PERFIL
            , T1.MEIO_VENDA
            , CASE
                WHEN T1.MEIO_VENDA = 'INTERNET CLIENTE' THEN
                    CASE
                        WHEN T2.VARIAVEL_ORIGEM = 'PRE_APROVADO' THEN 'PRE APROVADO'
                        ELSE T2.VARIAVEL_ORIGEM
                    END
                ELSE T1.ORIGEM_CLEARSALE
            END AS ORIGEM_CLEARSALE
            , CASE
                WHEN T1.MEIO_VENDA = 'INTERNET CLIENTE' THEN
                    CASE
                        WHEN T2.VARIAVEL_ORIGEM = 'PRE_APROVADO' THEN 'INTERNET PREA'
                        ELSE T1.DECISAO_VENDA
                    END
                ELSE T1.DECISAO_VENDA
            END AS DECISAO_VENDA
            , CASE
                WHEN T1.STATUS_PROPOSTA = 'N' AND T2.STATUS_CLEARSALE IN ('FRD', 'SUS', 'RPM', 'RPA') THEN 'FRAUDES'
                WHEN T1.STATUS_PROPOSTA = 'N' AND T2.STATUS_CLEARSALE IN ('APA', 'APM') THEN 'OUTROS'
				WHEN T1.STATUS_PROPOSTA = 'N' AND T2.STATUS_CLEARSALE IN ('CAN') THEN 'CANCELADO'
                WHEN T1.STATUS_PROPOSTA = 'N' AND T1.MOTIVO_RECUSA IS MISSING AND T2.STATUS_CLEARSALE NOT IN ('FRD', 'SUS', 'RPM', 'RPA', 'APA', 'APM') THEN 'OUTROS'
                WHEN T1.STATUS_PROPOSTA = 'A' THEN 'APROVACAO'
                ELSE T1.MOTIVO_RECUSA
            END AS MOTIVO_RECUSA
            , CASE
                WHEN T1.STATUS_PROPOSTA = 'N' AND T2.STATUS_CLEARSALE IN ('FRD', 'SUS', 'RPM', 'RPA', 'CAN') THEN 'RECUSA'
                WHEN T1.STATUS_PROPOSTA = 'N' AND T2.STATUS_CLEARSALE NOT IN ('FRD', 'SUS', 'RPM', 'RPA', 'APA', 'APM', 'CAN') THEN 'N/A'
                WHEN T1.STATUS_PROPOSTA = 'P' AND T2.STATUS_CLEARSALE IN ('AMA') THEN 'ANALISE MANUAL'
                ELSE T1.MOTIVO_CLEARSALE
            END AS MOTIVO_CLEARSALE
            , CASE
                WHEN T2.STATUS_CLEARSALE IN ('FRD', 'SUS', 'RPM', 'APM', 'CAN', 'AMA') THEN 'DERIVA'

                WHEN T2.STATUS_CLEARSALE IN ('APA') THEN 'MODELO - APROVA'
                WHEN T2.STATUS_CLEARSALE IN ('RPA') THEN 'MODELO - REPROVA'
                ELSE T1.RESULTADO_MODELO
            END AS RESULTADO_MODELO
            , CASE
                WHEN T1.SCORE_CLEAR IS MISSING THEN T2.SCORE_CLEARSALE
                ELSE T1.SCORE_CLEAR
            END AS SCORE_CLEAR
            , CASE
                WHEN T1.STS_RETORNO_CLEARSALE IS MISSING OR T1.STS_RETORNO_CLEARSALE = 'N/A' THEN T2.STATUS_CLEARSALE
                ELSE T1.STS_RETORNO_CLEARSALE
            END AS STS_RETORNO_CLEARSALE
			, T2.COMUNIC_RESP
			, T2.LEITURA_RESP
			, T2.FINALIZACAO
			, UPCASE(T2.CANAL) AS CANAL_DISPARO
            , CASE
                WHEN T1.DETALHE_RETORNO_FRAUDE IS MISSING OR T1.DETALHE_RETORNO_FRAUDE = 'N/A' THEN
                    CASE
                        WHEN T2.STATUS_CLEARSALE = 'APA' THEN 'APROVACAO AUTOMATICA'
                        WHEN T2.STATUS_CLEARSALE = 'APM' THEN 'APROVACAO MANUAL'

                        WHEN T2.STATUS_CLEARSALE = 'FRD' THEN 'FRAUDE CONFIRMADA'
                        WHEN T2.STATUS_CLEARSALE = 'RPA' THEN 'REPROVACAO AUTOMATICA'
                        WHEN T2.STATUS_CLEARSALE = 'RPM' THEN 'REPROVACAO MANUAL'
                        WHEN T2.STATUS_CLEARSALE = 'SUS' THEN 'SUSPEITA DE FRAUDE'
						WHEN T2.STATUS_CLEARSALE = 'AMA' THEN 'ANALISE MANUAL'
                        ELSE T1.DETALHE_RETORNO_FRAUDE
                    END
                ELSE T1.DETALHE_RETORNO_FRAUDE
            END AS DETALHE_RETORNO_FRAUDE
            , CASE

                        WHEN T2.STATUS_CLEARSALE = 'SUS' THEN 'RECUSADO POR SUSPEITA DE FRAUD'
                        ELSE T1.MOTIVO_RECUSA_DETALHE
                    END
                ELSE T1.MOTIVO_RECUSA_DETALHE
            END AS MOTIVO_RECUSA_DETALHE
			, CASE
				WHEN T2.STATUS_CLEARSALE = 'APA' AND DETALHE_RESPOSTA_ANALISE IS MISSING THEN 'Aprovado automaticamente modelo'
				WHEN T2.STATUS_CLEARSALE = 'RPA' AND DETALHE_RESPOSTA_ANALISE IS MISSING THEN 'Reprovado automaticamente modelo'

			, T4.GC
            , T1.NOME
            , T1.DT_NASCIMENTO
            , T1.DDD_TELEFONE

            , T1.ENTREGA_LOGRADOURO
            , T1.ENTREGA_NUMERO
            , T1.ENTREGA_COMPLEMENTO
            , T1.ENTREGA_BAIRRO
            , T1.ENTREGA_CEP
