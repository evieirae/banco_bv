SELECT

CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('Leves', 'Motos', 'Pesados') Then VrMetaNIN1 else 0 end as VrMetaNIN1_MC_Rodas,

CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('Leves', 'Motos', 'Pesados') and StVeiculo <> 'Usados' Then VrMedida else 0 end as VrMedida_MC_Rodas,
CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('Leves', 'Motos', 'Pesados') and StVeiculo <> 'Novos' Then VrMedidaNI else 0 end as VrMedidaNI_MC_Rodas,
CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('Leves', 'Motos', 'Pesados') Then VrMargemDescontos else 0 end as VrMargemDescontos_MC_Rodas,

/*MC EGV*/
CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('CVG Leves', 'CVG Pesados') Then VrMetaN1 else 0 end as VrMetaN1_MC_EGV,

CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('CVG Leves', 'CVG Pesados') Then VrMedida else 0 end as VrMedida_MC_EGV,
CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('CVG Leves', 'CVG Pesados') Then VrMedidaNI else 0 end as VrMedidaNI_MC_EGV,
CASE WHEN NmCanalEstruturaComercial in ('Canal Veiculos') and TpValor = 'VrMargemContribuicao' AND NmGrupoProdutoMeta IN ('CVG Leves', 'CVG Pesados') Then VrMargemDescontos else 0 end as VrMargemDescontos_MC_EGV,

/*META INAD15*/
CASE
	WHEN TpValor = 'Meta (Inad15)' and NUANOMES >= 202412 AND NUANOMES <= 202508 THEN 0.035
	WHEN TpValor = 'Meta (Inad15)' and NUANOMES <= 202509 THEN 0.027
	WHEN TpValor = 'Meta (Inad15)' and nuanomes >= 202510 and NmCanalEstruturaComercial in ('Canal Veiculos') and NmGrupoProdutoMeta IN ('Leves', 'Motos', 'Pesados') then 0 /*META NI DE INAD A PARTIR DE OUTUBRO-25 PARA CANAL VEICULOS*/
	ELSE 0
END AS VrMetaInad15,

/*META CONTA DIGITAL*/
CASE
	WHEN TpValor = 'ContaDigital' AND NUANOMES >= 202412 and NUANOMES <= 202501 and NmCanalEstruturaComercial in ('Canal Veiculos') and NmSubEstruturaComercialA NOT IN ('Canal Motos', 'Canal Pesados') THEN 0.23
	WHEN TpValor = 'ContaDigital' AND NUANOMES >= 202502 AND NUANOMES <= 202512 and NmCanalEstruturaComercial in ('Canal Veiculos') and NmSubEstruturaComercialA NOT IN ('Canal Motos', 'Canal Pesados') THEN (VrMetaN1 / VrMetaN1Denominador)
	WHEN TpValor = 'ContaDigital' AND NUANOMES >= 202601 and NmCanalEstruturaComercial in ('Canal Veiculos') THEN (VrMetaN1 / VrMetaN1Denominador)
END AS VrMetaContaDigital,

/*META PenPDV - Exclusivos Pesados*/
CASE
	WHEN TpValor = 'PenPdv' AND NUANOMES IN (202601, 202602) and NmCanalEstruturaComercial in ('Canal Veiculos') and NmSubEstruturaComercialA IN ('Canal Pesados') THEN 0.40
	WHEN TpValor = 'PenPdv' AND NUANOMES >= 202603 and NmCanalEstruturaComercial in ('Canal Veiculos') and NmSubEstruturaComercialA IN ('Canal Pesados') THEN 0.50
	ELSE 0
END AS VrMetaPenPDV

FROM BASE_META_PRODUCAO_ANALITICA A

		WHEN TpValor in ('VrSPF_AP') and NuAnoMes <= 202508 THEN (VrMetaN1 + VrMetaNIN1)
		WHEN TpValor in ('VrSegurosAuto') and NuAnoMes <= 202508 THEN (VrMetaN1 + VrMetaNIN1)

		WHEN TpValor in ('VrSPF_AP') and NuAnoMes >= 202509 THEN (VrMetaN1 + VrMetaNIN1)

		WHEN TpValor in ('VrSPF_AP') and NuAnoMes <= 202508 THEN (VrMetaProducaoN1)

PROC SQL;
CREATE TABLE AUX_UPDATE AS
SELECT

NuAnoMes,
CdColaborador,
NmGrupoCargoColaborador,
DsLoginN1,
'ExtraSeguros' AS TpValor,
SUM(Pc_Ating_Seguros_Total) AS Pc_Ating_Seguros_Total,

GROUP BY 1,2,3,4,5

WHERE Pc_Ating_ExtraSeguros <> 1;
;RUN;

update work.BASE_META_PRODUCAO_ANALITICA_03

where
	exists (select Pc_Ating_ExtraSeguros from AUX_UPDATE as t2
					where BASE_META_PRODUCAO_ANALITICA_03.NuAnoMes = t2.NuAnoMes
						and BASE_META_PRODUCAO_ANALITICA_03.CdColaborador = t2.CdColaborador

PROC SQL;
CREATE TABLE BASE_META_PRODUCAO_ANALITICA_07 AS
SELECT
A.NuAnoMes,

E.NmCanalEstruturaComercial,
E.NmEstruturaComercialA,
E.NmSubEstruturaComercialA,
E.NmAreaEstruturaComercial,
E.LstSubstituicao,
E.LstColaboradorFocal,
E.DsLoginN3A,
CASE WHEN E.DsLoginN2A = '' THEN 'cadeira.vazia' ELSE E.DsLoginN2A END AS DsLoginN2A,

A.CdColaborador,
A.NmGrupoCargoColaborador,
A.DsLoginN1,
A.TpValor,
A.VrAtingimento format percent9.2,

C.Peso AS PesoMeta format percent9.2,
CASE WHEN A.VrAtingimento >= 1 THEN D.VrComissaoMeta ELSE 0 END AS VrComissaoGR,
D.VrComissaoMeta as VrComissaoMaximaKpi,
B.VrComissaoMetaTotal,

CASE
	WHEN A.NuAnoMes <= 202512
		THEN (C.Peso * (CASE WHEN A.VrAtingimento >= 2 THEN 2 ELSE A.VrAtingimento END))

	WHEN A.NuAnoMes >= 202601 AND A.TpValor NOT IN ('Meta (Inad15)') /*DEMAIS TPVALOR*/
		THEN (C.Peso * (CASE WHEN A.VrAtingimento >= 2 THEN 2 ELSE A.VrAtingimento END))

	WHEN A.NuAnoMes >= 202601 AND A.TpValor IN ('Meta (Inad15)') /*ESPECÍFICO PARA A NOVA REGRA DE INAD15*/
		THEN (C.Peso * (CASE WHEN A.VrAtingimento >= 1 THEN 1 ELSE 0 END))

ELSE 0 END AS VrAtingimentoPonderado format percent9.2

FROM WORK.BASE_META_PRODUCAO_ANALITICA_06 A

LEFT JOIN BASEGRC.TBINCENTIVOALVOFINAL B ON B.NUANOMES = A.NUANOMES AND B.DsLoginN1 = A.DsLoginN1
LEFT JOIN BASEGRC.TBGRINCENTIVOPESOFINAL C ON C.NUANOMES = A.NUANOMES AND C.DsLoginN1 = A.DsLoginN1 AND C.Indicadores = A.TpValor
LEFT JOIN BASEGRC.TBGRINCENTIVOCARGACOMISSAOMETA D ON D.NUANOMES = A.NUANOMES AND D.DsLogin = A.DsLoginN1 AND D.Indicadores = A.TpValor
LEFT JOIN BASEGRC.TBGRINCENTIVO E ON E.DsLoginN1 = A.DsLoginN1 AND E.NUANOMES = A.NUANOMES

;RUN;

	set work.BASE_META_PRODUCAO_ANALITICA_07;

	array num_array _numeric_;
	do over num_array;
		if missing(num_array) then num_array = 0;
	end;
run;
