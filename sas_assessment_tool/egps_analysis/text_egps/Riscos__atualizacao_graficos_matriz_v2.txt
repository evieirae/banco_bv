proc sql;
create table matriz_resiliencia as
select
(case when a.Avaliação_Setorial="Fortemente Positivo" then 1 else
(case when a.Avaliação_Setorial="Levemente Positivo" then 2 else
(case when a.Avaliação_Setorial="Neutro" then 3 else
(case when a.Avaliação_Setorial="Levemente Negativo" then 4 else
(case when a.Avaliação_Setorial="Fortemente Negativo" then 5 else 0 end) end) end) end) end) as n,
a.Avaliação_Setorial,
b.Saldo as AtivosEspeciais,
c.Saldo as Critico,
d.Saldo as Preocupante,
e.Saldo as Atencao,
f.Saldo as Normal

from (select distinct Avaliação_Setorial from BASE_UNICA_ATACADO_corrigida) as a
left join (select distinct Avaliação_Setorial, sum(VrSaldo) as Saldo from BASE_UNICA_ATACADO_corrigida where Avaliação_Grupo="Ativos Especiais" group by 1) as b
on a.Avaliação_Setorial=b.Avaliação_Setorial
left join (select distinct Avaliação_Setorial, sum(VrSaldo) as Saldo from BASE_UNICA_ATACADO_corrigida where Avaliação_Grupo="Critico" group by 1) as c
on a.Avaliação_Setorial=c.Avaliação_Setorial
left join (select distinct Avaliação_Setorial, sum(VrSaldo) as Saldo from BASE_UNICA_ATACADO_corrigida where Avaliação_Grupo="Preocupante" group by 1) as d
on a.Avaliação_Setorial=d.Avaliação_Setorial
left join (select distinct Avaliação_Setorial, sum(VrSaldo) as Saldo from BASE_UNICA_ATACADO_corrigida where Avaliação_Grupo="Atenção" group by 1) as e
on a.Avaliação_Setorial=e.Avaliação_Setorial
left join (select distinct Avaliação_Setorial, sum(VrSaldo) as Saldo from BASE_UNICA_ATACADO_corrigida where Avaliação_Grupo="Normal" group by 1) as f
on a.Avaliação_Setorial=f.Avaliação_Setorial

;
quit;
