PROC SQL;
create table tempo.ValidaParcela as
/*create table ValidaParcela1 as */
select
VrFinanciamento FORMAT num18.2
,QtParcelas
,VrPrestacao FORMAT num18.2
,PcCoeficiente
,(VrPrestacao/PcCoeficiente)-VrFinanciamento as DifVrFinanciamentoRecalc FORMAT num18.2
,NuProposta
,DtEntradaProposta
,nmgrupoprodutometa
,StProposta
,NmCanalEstruturaComercial
,NmAreaEstruturaComercial
,*
from tmp.tboriginacao
where nmgrupoprodutometa in ('Leves','Motos','CVG Leves','Pesados','CVG Pesados')
and dtentradaproposta>= &DtCorte.
and case when ((VrPrestacao/PcCoeficiente)-VrFinanciamento) < -10 then 'Erro'
when ((VrPrestacao/PcCoeficiente)-VrFinanciamento) > 10 then 'Erro'
else 'Certo' end <>'Certo'
/* and cdparceirocomercial = 384666*/
/*and NmAreaEstruturaComercial in('Canal Londrina','Canal Criciúma','Canal Cascavel')*/
/*and StProposta='AP'*/
;run;

proc sql;
create table validaparcela as
select parc.VrFinanciamento
,parc.QtParcelas
,parc.VrPrestacao
,parc.PcCoeficiente
,parc.VrPrestacao / PcCoeficiente as VrReal FORMAT=numx18.2
,Calculated VrReal - parc.VrFinanciamento as Diferença FORMAT=numx18.2
,Case when Calculated Diferença between -200 and  200 then 'Arredondamento'
	  when Calculated Diferença < -200  			  then 'Pagou a menos'
	  when Calculated Diferença > 200  			  	  then 'Pagou a mais' end as Status
,Ori.VrEntrada as ValorEntrada_Proposta FORMAT=numx18.2
,Apro.VrEntrada as ValorEntrada_Simulação FORMAT=numx18.2
,coalesce(Apro.VrEntrada,0) - coalesce(Ori.VrEntrada,0) as Dif_EntreEntradas
/*,Apro.PcEntradaAjustada as EntradaAjustadaSimu*/
,Prod.VrProducao as VrProposta
,Calculated VrReal / parc.VrFinanciamento as VariaçãoDaDiferença
,Round(FLOOR(ABS(Calculated Dif_EntreEntradas)) - FLOOR(ABS(Calculated Diferença))) as ValidaçãoRefEntrada
,Case when SegCom.FlAutoCompleto = 1 then 'AutoCompleto' else '' end as LevantamentoAutocompleto

,Case 	when FLOOR(ABS(Calculated Diferença)) = 899  Then 'Seg Auto Parcial 594'
		when FLOOR(ABS(Calculated Diferença)) = 982  Then 'Seg Auto Parcial 547'
		when FLOOR(ABS(Calculated Diferença)) = 1601 Then 'Seg Auto Parcial 540'
		when FLOOR(ABS(Calculated Diferença)) = 2423 Then 'Seg Auto Parcial 547 e 608'
		when FLOOR(ABS(Calculated Diferença)) = 2882 Then 'Seg Auto Parcial 557'
		when FLOOR(ABS(Calculated Diferença)) = 2808 Then 'Seg Auto Parcial 577'
		when FLOOR(ABS(Calculated Diferença)) = 3332 Then 'Seg Auto Parcial 547 e 608 e 594'
		when FLOOR(ABS(Calculated Diferença)) = 5106 Then 'Seg Auto Parcial 566'
		when FLOOR(ABS(Calculated Diferença)) = 7915 Then 'Seg Auto Parcial 5106 e 577'
		when FLOOR(ABS(Calculated Diferença)) = 3322 Then 'Seg Auto Parcial 547 e 608 e 594'
		End as LevantamentoSeguroAutoParcial
 ,Seg.VrAuto as Seguros

,Case 	when Calculated Status = 'Pagou a menos' and Calculated ValidaçãoRefEntrada IN(0,.,-1,1) 			then 'Pagou a menos - entrada ajustada'
		when Calculated Status = 'Arredondamento' and Calculated LevantamentoAutocompleto = 'AutoCompleto' 	then 'Arred - Auto Completo'
		when Calculated Status = 'Arredondamento' and Calculated LevantamentoAutocompleto = '' 				then 'Arred - SEM Auto Completo'
		when Calculated Status = 'Pagou a menos' and FLOOR(ABS(Calculated ValidaçãoRefEntrada)) IN(1298,1299,1300)		then 'Pagou a menos - TC'
		when Calculated Status = 'Pagou a menos' and FLOOR(ABS(Calculated ValidaçãoRefEntrada)) IN(788,789,790)		then 'Pagou a menos - TAB Pesados'
		when Calculated VariaçãoDaDiferença between 1.05 and 1.20 and parc.nmgrupoprodutometa ='CVG Leves' 	then 'EGV - 120%'
		when Calculated LevantamentoSeguroAutoParcial <> '' 												then Calculated LevantamentoSeguroAutoParcial
		when Calculated Status = 'Pagou a mais' and FLOOR(ABS(Calculated ValidaçãoRefEntrada)) IN(788,789,790)		then 'Pagou a mais - TAB Pesados'
		when Calculated Status = 'Pagou a mais' and FLOOR(ABS(Calculated ValidaçãoRefEntrada)) IN(443,444,445)		then 'Pagou a mais - TAB Leves'
		when Calculated Status = 'Pagou a menos' and FLOOR(ABS(Calculated ValidaçãoRefEntrada)) IN(443,444,445)		then 'Pagou a mais - TAB Leves'

				when parc.NmGrupoProdutoMeta = 'Motos' and FLOOR(ABS(Seg.VrAuto)) IN(614)		then 'seguro auto parcial de motos (código 569)'
		when parc.NmGrupoProdutoMeta = 'Motos' and FLOOR(ABS(Seg.VrAuto)) IN(899)		then 'seguro auto parcial de motos (código 594)'

ENd as SubCenario
/*,parc.DifVrFinanciamentoRecalc NAO ENTRA*/
,parc.NuProposta
,parc.DtEntradaProposta
,parc.nmgrupoprodutometa
,parc.StProposta
,parc.NmCanalEstruturaComercial
,parc.NmAreaEstruturaComercial
,conv.nuversaomotor
,conv.sistemainicial
,conv.sistemafinal
,parc.DsLoginN1C
,parc.CdParceiroComercial
,parc.NmParceiroComercial
,prod.VrTarifaCadastro
,prod.VrTarifaAvaliacaoBem
,prod.QtSeguros
,prod.FlBVMais
,prod.FlCap
,prod.FlDemais
,prod.FlDemaisSemCompleto
,prod.FlCompleto
,prod.FlGarantiaMecanica
,prod.FlRCF
,prod.FlCasco
,prod.FlGap
,prod.FlDFI
,prod.VrSeguros
,prod.VrBVMais
,prod.VrCap
,prod.VrDemais
,prod.VrCompleto
,prod.VrGarantiaMecanica
,prod.VrRCF
,prod.VrCasco
,prod.VrGap
,prod.VrDFI
from tempo.ValidaParcela parc
left join convsimu.tbsimulacoes conv on conv.nuproposta= parc.nuproposta
left join tmp.tbproducao prod on prod.nuproposta = parc.nuproposta
left join tboriginacao Ori on Ori.nuproposta = parc.nuproposta
left join auxl.TBSEGUROSPROPOSTAS as Seg on Seg.nuproposta = parc.nuproposta
left join SegCompleto as SegCom on SegCom.nuproposta = parc.nuproposta
left join Aprosimu as Apro on Apro.CdSimulacaoInicial = conv.CdSimulacaoInicial
/*where parc.nuproposta in (702257993)*/
/*where parc.NmAreaEstruturaComercial in('Canal Londrina','Canal Criciúma','Canal Cascavel')*/
;run;
