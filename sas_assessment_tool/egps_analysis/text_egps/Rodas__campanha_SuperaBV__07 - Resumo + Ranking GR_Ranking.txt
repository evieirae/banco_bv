%include '/infogrc/utils/all-includes.sas';
LIBNAME CSBV '/infogrc/Campanha_SuperaBV';
LIBNAME CSBV_BKP '/infogrc/Incentivo_Comercial/Bases_BKP';

PROC SQL;
CREATE TABLE BASE_SUPERABV_RESUMO AS
SELECT

today() as Data format ddmmyy10.,
/*A.NmCanalEstruturaComercial,*/
/*A.NmSubEstruturaComercialA,*/
/*A.NmAreaEstruturaComercial,*/
/*'' AS LstSubstituicao,*/
/*'' AS LstColaboradorFocal,*/
/*A.DsLoginN3A,*/
/*A.DsLoginN2A,*/
/*A.CdColaborador,*/
/*A.NmGrupoCargoColaborador,*/
A.DsLoginN1,
/*'' AS NmEstadoColaborador,*/

sum(VrPontosTotalRanking) as VrPontosTotalRanking

FROM CSBV.TBSUPERABV_RESUMO A

/*where a.NUANOMES = &Base_Mes.*/
WHERE NmEstadoColaborador in ('Em Atividade Normal', 'Gozando Férias')

GROUP BY 1,2

;RUN;

DATA ESTRUTURA_GR;
SET CSBV.ESTRUTURA_COMERCIAL_01;
WHERE NmEstadoColaborador in ('Em Atividade Normal', 'Gozando Férias')
;RUN;

PROC SORT DATA=WORK.ESTRUTURA_GR OUT=WORK.ESTRUTURA_GR_01;BY DsLoginN1 DESCENDING NuAnoMes;RUN;
PROC SORT DATA=WORK.ESTRUTURA_GR_01 OUT=WORK.ESTRUTURA_GR_02 NODUPKEY;BY DsLoginN1;RUN;

PROC SQL;
CREATE TABLE BASE_SUPERABV_RESUMO_NOVA AS
SELECT

A.Data,

B.NmCanalEstruturaComercial,
B.NmSubEstruturaComercialA,
B.NmAreaEstruturaComercial,
B.DsLoginN2A,

A.DsLoginN1,
A.VrPontosTotalRanking

FROM BASE_SUPERABV_RESUMO A

LEFT JOIN WORK.ESTRUTURA_GR_02 B ON B.DSLOGINN1 = A.DSLOGINN1

;RUN;

PROC SORT DATA=WORK.BASE_SUPERABV_RESUMO_NOVA OUT=WORK.BASE_SUPERABV_RESUMO_01;
BY NmCanalEstruturaComercial NmSubEstruturaComercialA NmAreaEstruturaComercial DESCENDING VrPontosTotalRanking;
;RUN;

proc rank data=WORK.BASE_SUPERABV_RESUMO_01 descending out=WORK.BASE_SUPERABV_RESUMO_02 ties=dense;
    var VrPontosTotalRanking;
    ranks Ranking_VrPontosTotal;
    by NmCanalEstruturaComercial NmSubEstruturaComercialA NmAreaEstruturaComercial;
run;

/*********************************************/
/*TABELA COM O PRIMEIRO DO RANKING POR FILIAL*/
/*********************************************/

PROC SQL;
CREATE TABLE PRIMEIRO_RANKING AS
SELECT

/*NUANOMES,*/
NmCanalEstruturaComercial,
NmSubEstruturaComercialA,
NmAreaEstruturaComercial,
DsLoginN2A,
DsLoginN1,
VrPontosTotalRanking

FROM BASE_SUPERABV_RESUMO_02 A

WHERE Ranking_VrPontosTotal = 1 AND NmCanalEstruturaComercial <> ''

ORDER BY 1,2,3,4

;RUN;

PROC SQL;
CREATE TABLE BASE_SUPERABV_RESUMO_03 AS
SELECT

A.Data,
A.NmCanalEstruturaComercial,
A.NmSubEstruturaComercialA,
A.NmAreaEstruturaComercial,
A.DsLoginN2A,
A.DsLoginN1,
a.Ranking_VrPontosTotal,
A.VrPontosTotalRanking format commax20.2,

B.VrPontosTotalRanking AS VrPontosPrimeiroRankingFilial format commax20.2,
A.VrPontosTotalRanking - B.VrPontosTotalRanking as VrGAP_PontosPrimRankingFilial format commax20.2

FROM WORK.BASE_SUPERABV_RESUMO_02 A

LEFT JOIN PRIMEIRO_RANKING B ON

/*B.NUANOMES = A.NUANOMES*/
/*and */B.NmCanalEstruturaComercial = A.NmCanalEstruturaComercial
and B.NmSubEstruturaComercialA = A.NmSubEstruturaComercialA
and B.NmAreaEstruturaComercial = A.NmAreaEstruturaComercial
and B.DsLoginN2A = A.DsLoginN2A

where a.NmCanalEstruturaComercial <> '' /*and a.NUANOMES = &Base_Mes.*/

;RUN;

DATA CSBV.TbSuperaBV_Resumo_GR_Tri;
SET work.BASE_SUPERABV_RESUMO_03;
;RUN;
