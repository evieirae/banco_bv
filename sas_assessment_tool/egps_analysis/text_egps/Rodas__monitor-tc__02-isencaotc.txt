PROC SQL;
CREATE TABLE DEMAIS_CTOS AS SELECT
A.NUANOMES
,A.NUCONTRATO
,A.NUCPFCNPJ
,A.NMGRUPOPRODUTOMETA
,A.DTCONTRATO
,A.DtLiquidacaoEfetiva
,A.DtCancelamento
,B.DTCONTRATO AS ULTIMO_CTO
,CASE WHEN A.DtCancelamento=. AND A.DtLiquidacaoEfetiva=. THEN 'S'
	   WHEN A.DtLiquidacaoEfetiva>B.DTCONTRATO THEN 'S'
	   WHEN A.DtCancelamento>B.DTCONTRATO THEN 'S'

ON A.NUCPFCNPJ = B.NUCPFCNPJ
AND A.NUCONTRATO NE B.NUCONTRATO

WHERE A.DTCONTRATO <= B.DTCONTRATO
ORDER BY A.NUCPFCNPJ,A.DTCONTRATO DESC

;RUN;

proc sql;
create table contratos_clientebv as
select
	a.nucontrato,
	a.nucpfcnpj,
	a.nuproposta,
	a.DtEntradaProposta,
	a.DtSimulacaoInicial,
	a.Produto as ProdutoSimu,
	b.dtiniciocontacontrato,
	b.dtfimcontacontrato,
	b.nmtipoproduto,
	b.nmproduto,
	b.nmmodalidadeproduto,
	/*	possui contrato ativo na data do contrato*/
 	case	when b.dtiniciocontacontrato ne . and a.DtSimulacaoInicial > datepart(b.dtiniciocontacontrato)
												and a.DtSimulacaoInicial <= coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd) then 1 else . end as fl_contrato_ativo,
	/*	possui qualquer contrato encerrado na data do contrato*/
	case	when b.dtiniciocontacontrato ne . and a.DtSimulacaoInicial > coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd) then 1 else . end as fl_contrato_encerrado,
	/*	possui conta digital ativa na data do contrato*/
	case	when b.dtiniciocontacontrato ne . and b.NmTipoProduto EQ 'Conta Digital' and a.DtSimulacaoInicial > datepart(b.dtiniciocontacontrato)
												and a.DtSimulacaoInicial <= coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd) then 1 else . end as fl_conta_ativa,
	/*	possui cartão na data do contrato*/
	case	when b.dtiniciocontacontrato ne . and b.NmTipoProduto EQ 'Cartão' and a.DtSimulacaoInicial > datepart(b.dtiniciocontacontrato)
												and a.DtSimulacaoInicial <= coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd) then 1 else . end as fl_cartao_ativo,
	/*	possui contrato diferente de conta e cartao na data do contrato*/
	case	when b.dtiniciocontacontrato ne . and b.NmTipoProduto NOT IN ('Conta Digital' ,'Cartão') and a.DtSimulacaoInicial > datepart(b.dtiniciocontacontrato)
												and a.DtSimulacaoInicial <= coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd) then 1 else . end as fl_ctt_dif_conta_cartao_ativo,
	/*	conta digital foi contratado entre a data de entrada da proposta e a data do contrato*/
	case 	when b.dtiniciocontacontrato ne . and b.NmTipoProduto EQ 'Conta Digital' and a.DtSimulacaoInicial > datepart(b.dtiniciocontacontrato)
												and a.DtSimulacaoInicial <= coalesce(datepart(b.dtfimcontacontrato), '31dec2099'd)
												and datepart(b.dtiniciocontacontrato) BETWEEN a.dtentradaproposta and a.DtSimulacaoInicial then 1 else . end as fl_conta_integrada
from TbDtRefSim a
left join clientebvBQ b
	on a.nucpfcnpj= b.nucpfcnpj
	and datepart(b.dtiniciocontacontrato) <= a.DtSimulacaoInicial /* somente relacionamentos até a data da simulacao*/
	and b.nmproduto NE 'Seguro' /* seguro deveria estar sempre atrelado a outro produto */
;run;

proc sql;

	/*dtcontrato,*/
	dtentradaproposta,
	max(fl_contrato_ativo) as fl_contrato_ativo,
	max(fl_contrato_encerrado) as fl_contrato_encerrado,
	max(fl_conta_ativa) as fl_conta_ativa,
	max(fl_cartao_ativo) as fl_cartao_ativo ,
	max(fl_ctt_dif_conta_cartao_ativo) as fl_ctt_dif_conta_cartao_ativo,
	max(fl_conta_integrada) as fl_conta_integrada
from contratos_clientebv
group by 1,2,3,4,5,6
;run;

PROC SQL;
CREATE TABLE ALERTA_PAGAMENTO AS SELECT
*
,CASE WHEN Fl_ativo ='S' AND STATUS_TC NE 'ISENTO' THEN 'CLIENTE_ATIVO_N_ISENTO'
      WHEN (FLAG_CARTAO=1 AND FLAG_CONTA=1) AND STATUS_TC NE 'ISENTO' THEN 'CLIENTE_CARTAO_N_ISENTO'
	  WHEN Fl_ativo ='N' AND (FLAG_CARTAO=0 AND FLAG_CONTA=0) AND STATUS_TC NE 'N_ISENTO' THEN 'NAO_CLIENTE_ISENTO'
	  ELSE 'ERROR' END AS STATUS_ANALISE_ISENCAO

FROM PROPOSTAS_ISENCAO_TC_FINAL

WHERE CASE WHEN Fl_ativo ='S' AND STATUS_TC NE 'ISENTO' THEN 'CLIENTE_ATIVO_N_ISENTO'
      WHEN FLAG_CARTAO =1 AND STATUS_TC NE 'ISENTO' THEN 'CLIENTE_CARTAO_N_ISENTO'
	  WHEN Fl_ativo ='N' AND FLAG_CARTAO=0 AND STATUS_TC NE 'N_ISENTO' THEN 'NAO_CLIENTE_ISENTO'
	  ELSE 'ERROR' END NE 'ERROR'

;RUN;
