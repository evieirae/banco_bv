proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage);
create table TBTIPOMODELO_PROPOSTA as
select * from connection to perm(
select

cast(NUPROPOSTA as string) as nuproposta
,cast(cdtipomodeloproposta as int) as cdtipomodeloproposta
,cast(DTINCLUSAO as datetime) as dtinclusao
,cast(dataescolhaassinatura as datetime) as dataescolhaassinatura
,cast(datafinalizacaoassinatura as datetime) as datafinalizacaoassinatura
,cast(CDMOTIVOCANCELAMENTOASSINATURA as string) as cdmotivocancelamentoassintura
,cast(CDTIPOMODELO as string) as cdtipomodelo
,cast(SGCANALOPCAOASSINATURA as string) AS dscanalassinatura

from bv-pdbd-prd.trusted_dbcntf.tbtipomodeloproposta
/*from bv-pdbd-prd.dvry_negc_conta.tbtipomodeloproposta*/
where cast(DTINCLUSAO as date) >= '2024-01-01'
);
disconnect from perm;
quit;

PROC SQL;
CREATE TABLE LOG_MODELO_ASSINATURA AS
SELECT
	 input(NUPROPOSTA, 14.) as nuproposta
	,cdtipomodeloproposta
	,datepart(DTINCLUSAO) format date9. AS DTINCLUSAO
	,dataescolhaassinatura as DTESCOLHAASSINATURA
	,datafinalizacaoassinatura as DTFINALIZACAOASSINATURA
	,cdmotivocancelamentoassintura
	,input(CDTIPOMODELO, 2.) AS CDTIPOMODELO
	,CASE WHEN CDTIPOMODELO = '1' THEN 'Assinatura Digital'
			WHEN CDTIPOMODELO = '10' THEN 'Assinatura Digital - App Banco Digital'
				WHEN CDTIPOMODELO in ('14','15') THEN 'Captura de biometria para fluxo físico'
		ELSE 'ND' END AS DSTIPOMODELO
/*	,DSTIPOMODELOEXTERNO_DICT AS DSTIPOMODELO*/
	,dscanalassinatura
/*FROM DVRYBD.TBTIPOMODELO_PROPOSTA */
from TBTIPOMODELO_PROPOSTA
/*WHERE DTINCLUSAO >= "&dt_proposta_sas."d*/
/*WHERE DATEPART(DTINCLUSAO) BETWEEN '01JAN2019'D AND '31DEC2019'D*/
;RUN;

proc sql;
create table tpassinatura_novo as
select
	a.nuproposta,
	b.dtinclusao,
	b.dtescolhaassinatura as dataescolhaassinatura_ini,
	b.dstipomodelo as dstipomodelo_ini,
	c.dtescolhaassinatura as dataescolhaassinatura_fim,
	c.dstipomodelo as modelo_assinatura_final,
	case when d.dtfinalizacaoassinatura > c.dtescolhaassinatura then 1 else . end as fl_assinou
from(
	select
		nuproposta,
		min(cdtipomodeloproposta) as cdtipomodeloproposta_min,
		max(cdtipomodeloproposta) as cdtipomodeloproposta_max
	from log_modelo_assinatura
/*	where nuproposta = 511779112*/
	group by 1
	) a
left join log_modelo_assinatura b on a.cdtipomodeloproposta_min = b.cdtipomodeloproposta
left join log_modelo_assinatura c on a.cdtipomodeloproposta_max = c.cdtipomodeloproposta
left join (
	select
		nuproposta,
		max(dtfinalizacaoassinatura) as dtfinalizacaoassinatura
	from log_modelo_assinatura
	where dtfinalizacaoassinatura is not null
	and cdmotivocancelamentoassintura is null
/*	and nuproposta = 511779112*/
	group by 1
	) d on a.nuproposta = d.nuproposta
;run;
