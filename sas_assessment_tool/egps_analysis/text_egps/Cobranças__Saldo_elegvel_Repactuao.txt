PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'

CREATE TABLE RENEG_MENOR90 AS
SELECT
	*,
	B.SALDO_M0,
	CASE WHEN C.NUCONTRATOGESTAO IS NOT NULL THEN 1 ELSE 0 END AS FL_ADESAO

FROM empilha_reneg_final_&safra. AS A

LEFT JOIN CONNECTION TO PERM(
	SELECT DISTINCT
		A.NuContrato,
		MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro,
		CASE
			WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
			ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
		END AS SALDO_M0

	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

	LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
	ON A.NUCONTRATO = J.NUCONTRATO
	AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

	WHERE DATE(A.DtProcessamento) = &dt_fim_m0.
	GROUP BY A.nucontrato, J.FlRestruturacaoContrato
) AS B
ON A.NUCONTRATO = INPUT(B.NUCONTRATO,BEST32.)

AND A.SqContratoFinanceiro = B.sqcontratofinanceiro

LEFT JOIN CONNECTION TO PERM(

SELECT DISTINCT
	nucontratogestao,
	dt_baixa,
	st_reneg

FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
WHERE DATE(dt_baixa) > &dt_fim_m0. AND DATE(dt_baixa) <= &dt_fim_m1.
	AND (dt_cancelamento IS NULL OR dt_reativacao IS NOT NULL)
	AND st_reneg = 3
) AS C
ON A.NUCONTRATO = INPUT(C.nucontratogestao,BEST32.)

;QUIT;
