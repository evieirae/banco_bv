CREATE TABLE IPF.ANALITICO_&DATA. AS
SELECT
    A.*
FROM
    CONNECTION TO DBCON (
SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	C.RAZAO_NOME,
CASE
	WHEN C.RAZAO_NOME LIKE '%REGUA%'      THEN 'ADM_REGUA_DIGITAL'
	WHEN C.RAZAO_NOME LIKE '%BELLINATI%'  THEN 'BELLINATI'
	WHEN C.RAZAO_NOME LIKE '%TOLEDO%'     THEN 'TOLEDO'
	WHEN C.RAZAO_NOME LIKE '%RN%'         THEN 'RN_ADV'
	WHEN C.RAZAO_NOME LIKE '%MONACO%'     THEN 'LM_ANGELIS'
	WHEN C.RAZAO_NOME LIKE '%SCHULZE%'    THEN 'SCHULZE'
	WHEN C.RAZAO_NOME LIKE '%LM_ANGELIS%' THEN 'LM_ANGELIS'
	WHEN C.RAZAO_NOME LIKE '%MENDES%'     THEN 'MENDES_CUNHA'
	WHEN C.RAZAO_NOME LIKE '%PASQUALI%'   THEN 'PASQUALI'
	WHEN C.RAZAO_NOME LIKE '%MANDALITI%'  THEN 'MANDALITI'
	ELSE 'OUTROS'
 END AS ASSESSORIA,
   A.DT_ASSES,
   A.DT_RET_ASSES,
 CASE
	WHEN A.DT_ASSES < &DATA_INI_MES. THEN 'ESTOQUE'
	ELSE 'FLUXO'
	END AS CARTEIRA,
 CASE
   WHEN CARTEIRA = 'ESTOQUE' THEN DATEDIFF(DAY,D.DTVENCIMENTO,&DATA_INI_MES.)
   WHEN CARTEIRA = 'FLUXO'   THEN DATEDIFF(DAY,D.DTVENCIMENTO, DATE(A.DT_ASSES))
 END AS ATRASO_INICIO_MES,
 CASE
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 30  THEN '016_030'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 60  THEN '031_060'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 90  THEN '061_090'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 180 THEN '091_180'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 360 THEN '181_360'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 30  THEN '016_030'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 45  THEN '031_045'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 70  THEN '046_070'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 90  THEN '071_090'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 120 THEN '091_120'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 180 THEN '121_180'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 360 THEN '181_360'
 END AS FX_ATRASO,
 CASE
	WHEN A.COD_ACAO = 39                                        THEN 'ADMINISTRATIVA'
	WHEN A.COD_ACAO IN (1,3) AND C.RAZAO_NOME LIKE '%PRE CONT%' THEN 'ADMINISTRATIVA'
	WHEN A.COD_ACAO IN (1,3)                                    THEN 'CONTENCIOSA'
 END AS FASE,
 CASE
	WHEN C.RAZAO_NOME LIKE '%REGUA%'      THEN 'ADM_REGUA_DIGITAL'
	WHEN C.RAZAO_NOME LIKE '%RENEG%'      THEN 'RENEG'
	WHEN C.RAZAO_NOME LIKE '%PAGADOR%'    THEN 'BOM_PAGADOR'
	WHEN C.RAZAO_NOME LIKE '%REVISIONAL%' THEN 'REVISIONAL'
	WHEN C.RAZAO_NOME LIKE '%VISAO%'      THEN 'VISAO_UNICA'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%'   THEN 'PRE_CONTENCISO'
	ELSE 'NORMAL'
 END AS GRUPO,
    A.COD_ACAO,
    B.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
 CASE
	WHEN A.COD_ACAO = 39 AND ATRASO_INICIO_MES >= 71                                         THEN 1
	WHEN A.COD_ACAO IN ( 1,3 ) AND COD_ASSES NOT IN (2054,2055) AND ATRASO_INICIO_MES <= 70  THEN 1  
	WHEN GRUPO = 'VISAO_UNICA'                                                               THEN 1
	WHEN GRUPO = 'REVISIONAL'                                                                THEN 1
	WHEN GRUPO = 'ADM_REGUA_DIGITAL' AND ATRASO_INICIO_MES >= 31                             THEN 1
	WHEN GRUPO = 'BOM_PAGADOR'       AND ATRASO_INICIO_MES >= 31                             THEN 1
	WHEN DATE(DT_ASSES) = DATE(DT_RET_ASSES)                                                 THEN 1
	WHEN A.COD_ASSES IN ( 2065, 2063, 2064, 2062, 2066, 2067)                                THEN 1 
	ELSE 0
  END AS FLAG_RETIRAR
INTO
	#01
FROM
	DBA.SAJ_PROCESSO A LEFT JOIN DBA.SAJ_TIPO_ACAO B
	ON
	A.COD_ACAO = B.COD_ACAO

	LEFT JOIN  SCC_FIN.SAJ_AGENTE C
	ON
	A.COD_ASSES = C.COD_AGENTE

	LEFT JOIN  DBA.VWTBPARCELA D
	ON
	A.NUCONTRATOGESTAO = D.NUCONTRATO
	AND
	A.SQCONTRATOFINANCEIRO = D.SQCONTRATOFINANCEIRO
	AND
	A.PARC_ENVIADA = D.NUPARCELA
WHERE
	A.COD_ACAO IN (1,3,39)
	AND
	ATIVA_PRINC = 'E'
	AND
	(DATE(A.DT_ASSES) <= &DATA_FIM_MES. AND (A.DT_RET_ASSES IS NULL OR DATE(A.DT_RET_ASSES) >= &DATA_INI_MES.))
	AND
	ATRASO_INICIO_MES <= 360
	AND
	FLAG_RETIRAR = 0

	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,
    A.ATRASO_INICIO_MES,
    A.FX_ATRASO,
    A.FASE,
    A.GRUPO,
    A.COD_ACAO,
    A.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	B.NUPARCELA,
	B.VRPARCELA,
	B.STPARCELA,
	B.DTVENCIMENTO,
	B.DTULTIMORECEBIMENTO,
	B.VRULTRECEBIMENTO,
	B.DTPROCESSAMENTOULTRECEBIMENTO,
	DATEDIFF(DAY,B.DTVENCIMENTO,B.DTULTIMORECEBIMENTO) AS ATRASO_PAGTO,
	C.CDAGENTERECEBEDOR
INTO #02
FROM
	#01 A LEFT JOIN DBA.VWTBPARCELA B
	ON
	A.NUCONTRATOGESTAO = B.NUCONTRATO
	AND
	A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
	AND
	B.STPARCELA = 'Q'
	AND
	DATE(B.DTULTIMORECEBIMENTO) >= &DATA_INI_MES. AND DATE(B.DTULTIMORECEBIMENTO) <= &DATA_FIM_MES.
	AND
	DATE(B.DTULTIMORECEBIMENTO) >= DATE(A.DT_ASSES) AND (DATE(B.DTULTIMORECEBIMENTO) <= DATE(A.DT_RET_ASSES) OR DATE(A.DT_RET_ASSES) IS NULL)

	LEFT JOIN DBA.VWTBPARCELARECEBIMENTO C
	ON
	B.NUCONTRATO = C.NUCONTRATO
	AND
	B.SQCONTRATOFINANCEIRO = C.SQCONTRATOFINANCEIRO
	AND
	B.NUPARCELA = C.NUPARCELA

GROUP BY
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,
    A.ATRASO_INICIO_MES,
    A.FX_ATRASO,
    A.FASE,
    A.GRUPO,
    A.COD_ACAO,
    A.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	B.NUPARCELA,
	B.VRPARCELA,
	B.STPARCELA,
	B.DTVENCIMENTO,
	B.DTULTIMORECEBIMENTO,
	B.VRULTRECEBIMENTO,
	B.DTPROCESSAMENTOULTRECEBIMENTO,
	ATRASO_PAGTO,
	C.CDAGENTERECEBEDOR

SELECT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	MIN(B.DTVENCIMENTO) AS VENCIMENTO_ABERTO
INTO
	#PROX_VENCIMENTO
FROM
	#02 A INNER JOIN DBA.VWTBPARCELA B
	ON

	AND

	OR
	(B.STPARCELA = 'Q' AND DATE(B.DTULTIMORECEBIMENTO) > &DATA_FIM_MES.))
GROUP BY
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO

SELECT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,
    A.ATRASO_INICIO_MES,
    A.FX_ATRASO,

    A.GRUPO,
    A.COD_ACAO,
    A.NOM_ACAO,

    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	A.NUPARCELA,
	A.VRPARCELA,
	A.STPARCELA,
	A.DTVENCIMENTO,
	A.DTULTIMORECEBIMENTO,
	A.VRULTRECEBIMENTO,

	A.CDAGENTERECEBEDOR,
	B.VENCIMENTO_ABERTO,
	DATEDIFF(DAY,DATE(B.VENCIMENTO_ABERTO), &DATA_FIM_MES.) AS ATRASO_ATUAL
INTO
	#03
FROM

	ON

	AND

GROUP BY
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,

    A.FLAG_RETIRAR,
	A.NUPARCELA,

	A.DTVENCIMENTO,

	ATRASO_ATUAL



SELECT

SELECT
    NUCONTRATO,
    DTLIQUIDACAOEFETIVA,
    DTCANCELAMENTO,
    CDSITUACAOCONTRATO
INTO
	#QUITADO
FROM

WHERE
	DATE(DTLIQUIDACAOEFETIVA) >= &DATA_INI_MES. AND DATE(DTLIQUIDACAOEFETIVA) <= &DATA_FIM_MES.
	OR
	DATE(DTCANCELAMENTO) >= &DATA_INI_MES. AND DATE(DTCANCELAMENTO) <= &DATA_FIM_MES.

SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,

    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	A.NUPARCELA,
	A.VRPARCELA,
	A.STPARCELA,
	A.DTVENCIMENTO,
	A.DTULTIMORECEBIMENTO,
	A.VRULTRECEBIMENTO,
	A.DTPROCESSAMENTOULTRECEBIMENTO,
	A.ATRASO_PAGTO,
	A.CDAGENTERECEBEDOR,
	A.VENCIMENTO_ABERTO,
	A.ATRASO_ATUAL,
CASE
    WHEN DATE(B.DTPROCESSAMENTO) >= DATE(A.DT_ASSES) AND DATE(B.DTPROCESSAMENTO) <= DATE(A.DT_RET_ASSES) THEN 1
    ELSE 0
END AS FLAG_RENEG,
CASE
    WHEN DATE(C.DTENTRADA) >= DATE(A.DT_ASSES) AND (DATE(C.DTENTRADA) <= DATE(A.DT_RET_ASSES) OR DATE(A.DT_RET_ASSES) IS NULL) THEN 1
    ELSE 0
END AS FLAG_BNDU,
CASE
    WHEN DATE(D.DTLIQUIDACAOEFETIVA) >= DATE(A.DT_ASSES) AND DATE(D.DTLIQUIDACAOEFETIVA) <= DATE(A.DT_RET_ASSES)
    OR
	DATE(D.DTCANCELAMENTO) >= DATE(A.DT_ASSES) AND DATE(D.DTCANCELAMENTO) <= DATE(A.DT_RET_ASSES) THEN 1
ELSE 0
END AS FLAG_QUITACAO,
CASE
    WHEN FLAG_RENEG = 1                                          	   THEN B.DTPROCESSAMENTO
    WHEN FLAG_BNDU = 1                                           	   THEN C.DTENTRADA
    WHEN FLAG_QUITACAO = 1 AND DATE(D.DTLIQUIDACAOEFETIVA) IS NOT NULL THEN D.DTLIQUIDACAOEFETIVA
    WHEN FLAG_QUITACAO = 1 AND DATE(D.DTCANCELAMENTO) IS NOT NULL 	   THEN D.DTCANCELAMENTO
    ELSE A.DTULTIMORECEBIMENTO
END AS DT_PONTUACAO,
CASE
    WHEN DATE(DT_PONTUACAO) >= DATE(A.DT_ASSES) AND (DATE(DT_PONTUACAO) <= DATE(A.DT_RET_ASSES) OR DATE(DT_RET_ASSES) IS NULL)
	AND

PROC SQL;
CREATE TABLE BASE_CART AS
SELECT DISTINCT
	&DATA_EXEC. AS DATA_EXEC FORMAT DDMMYY10.,
	A.FASE,
	A.GRUPO,
	A.FLAG_PILOTO,
	A.ASSESSORIA,
	SUM(A.'Fluxo'n) AS 'Fluxo'n,
	SUM(A.'16-30'n) AS '16-30'n,
	SUM(A.'31-45'n) AS '31-45'n,
	SUM(A.'46-70'n) AS '46-70'n,
	SUM(A.'61-90'n) AS '61-90'n,
	SUM(A.'Fluxo_C'n) AS 'Fluxo_C'n,
	SUM(A.'71-90'n) AS '71-90'n,
	SUM(A.'91-120'n) AS '91-120'n,
	SUM(A.'121-180'n) AS '121-180'n,
	SUM(A.'181-360'n) AS '181-360'n
FROM
	BASE AS A
GROUP BY
	A.FASE,
	A.GRUPO,
	A.FLAG_PILOTO,
	A.ASSESSORIA
ORDER BY
	A.FASE,
	A.GRUPO,
	A.FLAG_PILOTO,
	A.ASSESSORIA;
QUIT;
RUN;

PROC SQL;
CONNECT TO SYBASEIQ AS DBCON
     (HOST='PTA-IQPRD01'
      SERVER=PTAIQPRD
      DB=PTAIQCONT
	  USER=&USER.
	  PASSWORD=&PASS.);
CREATE TABLE IPF.ANALITICO_&DATA. AS
SELECT
    A.*
FROM
    CONNECTION TO DBCON (
SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	C.RAZAO_NOME,
CASE
	WHEN C.RAZAO_NOME LIKE '%REGUA%'      THEN 'ADM_REGUA_DIGITAL'
	WHEN C.RAZAO_NOME LIKE '%BELLINATI%'  THEN 'BELLINATI'
	WHEN C.RAZAO_NOME LIKE '%TOLEDO%'     THEN 'TOLEDO'
	WHEN C.RAZAO_NOME LIKE '%RN%'         THEN 'RN_ADV'
	WHEN C.RAZAO_NOME LIKE '%MONACO%'     THEN 'LM_ANGELIS'
	WHEN C.RAZAO_NOME LIKE '%SCHULZE%'    THEN 'SCHULZE'
	WHEN C.RAZAO_NOME LIKE '%LM_ANGELIS%' THEN 'LM_ANGELIS'
	WHEN C.RAZAO_NOME LIKE '%MENDES%'     THEN 'MENDES_CUNHA'
	WHEN C.RAZAO_NOME LIKE '%PASQUALI%'   THEN 'PASQUALI'
	WHEN C.RAZAO_NOME LIKE '%MANDALITI%'  THEN 'MANDALITI'
	ELSE 'OUTROS'
 END AS ASSESSORIA,
   A.DT_ASSES,
   A.DT_RET_ASSES,
 CASE
	WHEN A.DT_ASSES < &DATA_INI_MES. THEN 'ESTOQUE'
	ELSE 'FLUXO'
	END AS CARTEIRA,
 CASE
   WHEN CARTEIRA = 'ESTOQUE' THEN DATEDIFF(DAY,D.DTVENCIMENTO,&DATA_INI_MES.)
   WHEN CARTEIRA = 'FLUXO'   THEN DATEDIFF(DAY,D.DTVENCIMENTO, DATE(A.DT_ASSES))
 END AS ATRASO_INICIO_MES,
 CASE
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 30  THEN '016_030'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 60  THEN '031_060'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 90  THEN '061_090'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 180 THEN '091_180'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 360 THEN '181_360'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 30  THEN '016_030'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 45  THEN '031_045'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 70  THEN '046_070'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 90  THEN '071_090'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 120 THEN '091_120'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 180 THEN '121_180'
	WHEN C.RAZAO_NOME NOT LIKE '%PRE CONT%' AND ATRASO_INICIO_MES <= 360 THEN '181_360'
 END AS FX_ATRASO,
 CASE
	WHEN A.COD_ACAO = 39                                        THEN 'ADMINISTRATIVA'
	WHEN A.COD_ACAO IN (1,3) AND C.RAZAO_NOME LIKE '%PRE CONT%' THEN 'ADMINISTRATIVA'
	WHEN A.COD_ACAO IN (1,3)                                    THEN 'CONTENCIOSA'
 END AS FASE,
 CASE
	WHEN C.RAZAO_NOME LIKE '%REGUA%'      THEN 'ADM_REGUA_DIGITAL'
	WHEN C.RAZAO_NOME LIKE '%RENEG%'      THEN 'RENEG'
	WHEN C.RAZAO_NOME LIKE '%PAGADOR%'    THEN 'BOM_PAGADOR'
	WHEN C.RAZAO_NOME LIKE '%REVISIONAL%' THEN 'REVISIONAL'
	WHEN C.RAZAO_NOME LIKE '%VISAO%'      THEN 'VISAO_UNICA'
	WHEN C.RAZAO_NOME LIKE '%PRE CONT%'   THEN 'PRE_CONTENCISO'
	ELSE 'NORMAL'
 END AS GRUPO,
    A.COD_ACAO,
    B.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
 CASE
	WHEN A.COD_ACAO = 39 AND ATRASO_INICIO_MES >= 71                                         THEN 1
	WHEN A.COD_ACAO IN ( 1,3 ) AND COD_ASSES NOT IN (2054,2055) AND ATRASO_INICIO_MES <= 70  THEN 1  
	WHEN GRUPO = 'VISAO_UNICA'                                                               THEN 1
	WHEN GRUPO = 'REVISIONAL'                                                                THEN 1
	WHEN GRUPO = 'ADM_REGUA_DIGITAL' AND ATRASO_INICIO_MES >= 31                             THEN 1
	WHEN GRUPO = 'BOM_PAGADOR'       AND ATRASO_INICIO_MES >= 31                             THEN 1
	WHEN DATE(DT_ASSES) = DATE(DT_RET_ASSES)                                                 THEN 1
	WHEN A.COD_ASSES IN ( 2065, 2063, 2064, 2062, 2066, 2067)                                THEN 1 
	ELSE 0
  END AS FLAG_RETIRAR
INTO
	#01
FROM
	DBA.SAJ_PROCESSO A LEFT JOIN DBA.SAJ_TIPO_ACAO B
	ON
	A.COD_ACAO = B.COD_ACAO

	LEFT JOIN  SCC_FIN.SAJ_AGENTE C
	ON
	A.COD_ASSES = C.COD_AGENTE

	LEFT JOIN  DBA.VWTBPARCELA D
	ON
	A.NUCONTRATOGESTAO = D.NUCONTRATO
	AND
	A.SQCONTRATOFINANCEIRO = D.SQCONTRATOFINANCEIRO
	AND
	A.PARC_ENVIADA = D.NUPARCELA
WHERE
	A.COD_ACAO IN (1,3,39)
	AND
	ATIVA_PRINC = 'E'
	AND
	(DATE(A.DT_ASSES) <= &DATA_FIM_MES. AND (A.DT_RET_ASSES IS NULL OR DATE(A.DT_RET_ASSES) >= &DATA_INI_MES.))
	AND
	ATRASO_INICIO_MES <= 360
	AND
	FLAG_RETIRAR = 0



SELECT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,
    A.ATRASO_INICIO_MES,
    A.FX_ATRASO,
    A.FASE,
    A.GRUPO,
    A.COD_ACAO,
    A.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	B.NUPARCELA,
	B.VRPARCELA,
	B.STPARCELA,
	B.DTVENCIMENTO,
	B.DTULTIMORECEBIMENTO,
	B.VRULTRECEBIMENTO,
	B.DTPROCESSAMENTOULTRECEBIMENTO,
	DATEDIFF(DAY,B.DTVENCIMENTO,B.DTULTIMORECEBIMENTO) AS ATRASO_PAGTO,
	C.CDAGENTERECEBEDOR
INTO #02
FROM
	#01 A LEFT JOIN DBA.VWTBPARCELA B
	ON
	A.NUCONTRATOGESTAO = B.NUCONTRATO
	AND
	A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
	AND
	B.STPARCELA = 'Q'
	AND
	DATE(B.DTULTIMORECEBIMENTO) >= &DATA_INI_MES. AND DATE(B.DTULTIMORECEBIMENTO) <= &DATA_FIM_MES.
	AND
	DATE(B.DTULTIMORECEBIMENTO) >= DATE(A.DT_ASSES) AND (DATE(B.DTULTIMORECEBIMENTO) <= DATE(A.DT_RET_ASSES) OR DATE(A.DT_RET_ASSES) IS NULL)

	LEFT JOIN DBA.VWTBPARCELARECEBIMENTO C
	ON
	B.NUCONTRATO = C.NUCONTRATO
	AND
	B.SQCONTRATOFINANCEIRO = C.SQCONTRATOFINANCEIRO
	AND
	B.NUPARCELA = C.NUPARCELA

GROUP BY
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	A.NUM_PROCESSO,
	A.ATIVA_PRINC,
	A.COD_ASSES,
	A.RAZAO_NOME,
    A.ASSESSORIA,
    A.DT_ASSES,
    A.DT_RET_ASSES,
    A.CARTEIRA,
    A.ATRASO_INICIO_MES,
    A.FX_ATRASO,
    A.FASE,
    A.GRUPO,
    A.COD_ACAO,
    A.NOM_ACAO,
    A.VL_CAUSA,
    A.COD_EVENTO,
    A.PARC_ENVIADA,
    A.FLAG_RETIRAR,
	B.NUPARCELA,
	B.VRPARCELA,
	B.STPARCELA,
	B.DTVENCIMENTO,
	B.DTULTIMORECEBIMENTO,
	B.VRULTRECEBIMENTO,
	B.DTPROCESSAMENTOULTRECEBIMENTO,
	ATRASO_PAGTO,
	C.CDAGENTERECEBEDOR

SELECT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	MIN(B.DTVENCIMENTO) AS VENCIMENTO_ABERTO
INTO
	#PROX_VENCIMENTO
FROM

	SUM(A.'121-180'n) AS '121-180'n,

	BASE AS A
GROUP BY
	A.FASE,
	A.GRUPO,
	A.FLAG_PILOTO,
	A.ASSESSORIA

	A.FASE,
	A.GRUPO,
	A.FLAG_PILOTO,

RUN;
