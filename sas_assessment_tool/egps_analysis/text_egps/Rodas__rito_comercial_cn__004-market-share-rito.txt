PROC SQL;
CREATE TABLE PotencialRodas AS
  SELECT
    POT.NuAnomes
    ,POT.CdParceiroComercial
    ,POT.FxIdade AS FxIdade
	,SUM(POT.QtContrato) AS Part_ContratosBV
    ,SUM(POT.QtGravamesMercado) AS Part_Mercado
	,coalesce(sum(POT.VrGravamesMercado),0) AS VrGravamesMercado
	,coalesce(sum(POT.VrGravamesMercado / TotalDiaUtilPrd * B.TotalDiaUtilMes),0) AS VrGravamesMercadoProj
  FROM TMP.TbPotencialRodas POT
		LEFT JOIN TbDiasUteis02 B
		ON POT.NuAnomes = b.NuAnomes
  WHERE 1=1
    AND POT.NuAnoMes >= &safraM6.
    AND POT.FlAtualizacaoMercado = 1
	And CASE WHEN POT.FxIdade IS NULL THEN ''
         WHEN POT.FxIdade = 'ZeroKM' THEN 'Novos'
         ELSE 'Usados'
         END = 'Usados'
	and Pot.NmGrupoProduto = 'Leves'
	and POT.NmCanalEstruturaComercialC = 'Canal Negocios'
	and POT.FlAtivo = 'S'
	and POT.NmTipoParceiroComercial in ('Concessionaria','Loja')
GROUP BY 1,2,3
;run;
