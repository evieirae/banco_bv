PROC SQL;

PROC SQL;
CREATE TABLE APLICACAO_NASSAU_2 AS
	SELECT Counterpart format $76.,
		   'Actual USS Principal'n AS Saldo
	FROM TESOURARIA
	WHERE 'Product Name'n = "OVER" AND Saldo <> 0 AND Counterpart <> 'BANCO VOTORANTIM S/A';

PROC SQL;
	CREATE TABLE APLICACAO_NASSAU_3 AS
	SELECT SUM(A.Saldo) AS NASSAU_BRANCH,
