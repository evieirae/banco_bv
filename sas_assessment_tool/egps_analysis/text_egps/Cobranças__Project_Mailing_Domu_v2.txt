CREATE TABLE WORK.Hist_Internalizacao AS
SELECT DISTINCT *
FROM CONNECTION TO PERM	(SELECT DISTINCT
							  dt_acionamento AS Dt
							, CONCAT(dt_acionamento,'T',hr_inicio_acionamento) AS Dth
							, tp_resultado_acionamento
							, CASE WHEN tp_resultado_acionamento in ('2','3','4','5','6') THEN 1 ELSE 0 END AS Flag_Alo
							, CASE
									WHEN tp_resultado_acionamento = '5' THEN 1
									WHEN tp_resultado_acionamento = '6' THEN 1
									ELSE 0 END AS Flag_CPC
							, cd_cpf_cnpj as IdPessoa
							, cd_contrato as NuContrato
							, ds_ocorrencia_chamada
							, tp_ocorrencia_chamada
							, cd_telefone_contato
						FROM `bv-pdbd-prd.dvry_corp_cobranca.assessoria_acionamentos_cobranca_fato`
						WHERE dt_acionamento >= DATE_ADD(CURRENT_DATE, INTERVAL -30 DAY)
						AND tp_resultado_acionamento in ('4','5','6')
						AND cd_cpf_cnpj IS NOT NULL
						AND cd_contrato IS NOT NULL)
;disconnect from perm
;QUIT;

PROC SQL;
CREATE TABLE WORK.AUX_CPC_INTENALIZ AS
SELECT Dt Format = yymmdd10. AS Dt
, Input(Dth,e8601dt19.) Format = e8601dt19. Informat = e8601dt19. As Dth
, Input(tp_resultado_acionamento,best2.)  Format = best2. Informat = best2. AS TpResultado
, CASE WHEN Length(COMPRESS(IdPessoa)) <=11 THEN PUT(Input(IdPessoa,best14.),z11.) Else PUT(Input(IdPessoa,best14.),z14.) End As IdPessoa
, Input(Substr(NuContrato,1,14),best14.) Format = best14. Informat = best14. As NuContrato
, ds_ocorrencia_chamada As DsIdOcorrencia
, Input(tp_ocorrencia_chamada,best3.) Format = Best3. Informat = Best3. As IdOcorrencia
, SUBSTR(COMPRESS(cd_telefone_contato),1,2) AS DDD
, SUBSTR(COMPRESS(cd_telefone_contato),3,Length(COMPRESS(cd_telefone_contato))) AS Numero
, CASE WHEN SUBSTR(COMPRESS(cd_telefone_contato),3,1) = '9' THEN 'CELULAR' ELSE 'FIXO' END AS Tipo_Numero
, Flag_Alo
, Flag_CPC
, CATS('55',COMPRESS(cd_telefone_contato)) AS Telefone_Completo
FROM WORK.Hist_Internalizacao
;QUIT;

PROC SQL;
CREATE TABLE WORK.BASE_TEL_00 AS
SELECT DISTINCT A.*, B.Flag_CPC, CASE WHEN B.Flag_CPC = 1 THEN 0 ELSE A.Ranking END AS Rank
FROM WORK.BASE_TEL AS A
LEFT JOIN (SELECT DISTINCT B.IDPESSOA, B.Telefone_COmpleto, MAX(B.Flag_CPC) AS Flag_CPC FROM WORK.AUX_CPC_INTENALIZ AS B GROUP BY B.IDPESSOA, B.Telefone_COmpleto) AS B ON A.Telefone_Completo = B.Telefone_Completo AND A.IdPessoa = B.IdPessoa
ORDER BY IdPessoa ASC, Rank ASC
;QUIT;
