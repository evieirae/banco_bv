PROC SQL;
connect to bigquery as perm (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
CREATE TABLE AUX AS
SELECT	*
from	connection to perm
		(SELECT C.nom_acao AS Acao,
				CASE WHEN D.CD_CONTRATO IS NOT NULL THEN 'Sim' ELSE 'Não Localizado' END AS Existe,
				'' AS FL_BNDU,
				'' AS FL_SUSPENSO,
				CAST(A.NuContratoGestao AS INT64) AS NuContratoGestao,
				A.SqContratoFinanceiro,
				A.num_processo,
				A.cod_asses,
				D.CD_CPF_CNPJ AS NUCPFCNPJ,
				CASE WHEN D.CD_SITUACAO_CONTRATO = '1' THEN 'Aberto' ELSE 'Fechado' END SituacaoContrato,
				A.cod_acao,
				DATE(A.dt_asses) AS dt_asses,
				A.parc_enviada,
				B.razao_nome,
				CAST(A.vl_causa AS FLOAT64) AS vl_causa

		FROM	bv-pdbd-prd.trusted_scc_fin.saj_processo AS A
				LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_agente AS B ON	A.cod_asses = B.cod_agente
				LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_tipo_acao AS C ON		A.cod_acao = C.cod_acao
				LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato  AS D ON		A.NuContratoGestao = D.CD_CONTRATO AND
														A.SqContratoFinanceiro = D.SQ_CONTRATO_FINANCEIRO_ATIVO
		WHERE	A.cod_acao IN (1,3,39,59,60) AND
				DATE(A.dt_asses) <= &DIA_REF.  AND
				(A.dt_ret_asses IS NULL OR
				DATE(A.dt_ret_asses) > &DIA_REF.))
;
disconnect from perm;
QUIT;

BY NuContratoGestao SqContratoFinanceiro DESCENDING dt_asses;

PROC SQL;
CREATE TABLE GRADATIVO_1_5 AS
SELECT DISTINCT
A.*
,G.CLIENTE_50K
,H.CART_2











,CASE WHEN A.COLLECTION_AUX = 'SEM MODELO' THEN 'BOM'
	  WHEN A.POLITICA_FINAL_ELEITA = 'TESTE_SEM_DESCONTO' THEN 'BOM'
	  ELSE A.COLLECTION_AUX END AS COLLECTION

,CASE WHEN A.COLLECTION_AUX = 'BOM' AND G.CLIENTE_50K = 1 THEN 1 ELSE 0 END AS FLAG_GRADATIVO_BOM

,CASE WHEN A.COLLECTION_AUX = 'BOM' AND G.CLIENTE_50K = 1 THEN 1
      WHEN A.COLLECTION_AUX = 'BOM' AND H.CART_2 IN ('01. RENEG_EMERG','02. RENEG_N_EMERG','03. BAD BANK') THEN 1 ELSE 0 END AS FLAG_GRADATIVO_BOM_VF_CEN3

,CASE WHEN A.COLLECTION_AUX = 'BOM' AND G.CLIENTE_50K = 1 THEN 1
      WHEN A.COLLECTION_AUX = 'BOM' AND H.CART_2 IN ('01. RENEG_EMERG','02. RENEG_N_EMERG') THEN 1 ELSE 0 END AS FLAG_GRADATIVO_BOM_VF_CEN2

,CASE WHEN A.COLLECTION_AUX = 'SEM MODELO' THEN 'BOM'
	  WHEN A.POLITICA_FINAL_ELEITA = 'TESTE_SEM_DESCONTO' THEN 'BOM'
	  WHEN CALCULATED FLAG_GRADATIVO_BOM_VF_CEN2 = 1 THEN 'REGULAR'
	  ELSE A.COLLECTION_AUX END AS COLLECTION_PROPOSTO_CEN2

,CASE WHEN A.COLLECTION_AUX = 'SEM MODELO' THEN 'BOM'
	  WHEN A.POLITICA_FINAL_ELEITA = 'TESTE_SEM_DESCONTO' THEN 'BOM'
	  WHEN CALCULATED FLAG_GRADATIVO_BOM_VF_CEN3 = 1 THEN 'REGULAR'
	  ELSE A.COLLECTION_AUX END AS COLLECTION_PROPOSTO_CEN3

,CASE WHEN CALCULATED FLAG_GRADATIVO_BOM = 1 THEN
      (CASE WHEN A.RDG_NUM >= 30 THEN 'TESTE_BOM_50K' ELSE 'CONTROLE_BOM_50K' END) END AS PUBLICO_GRADATIVO_BOM

,CASE WHEN A.SCORE_FINAL in (523, 524, 525, 526, 527, 528, 529, 530, 531, 532, 533, 534) THEN 'ELASTICIDADE' END AS FLAG_ELASTICIDADE

,D.PUBLICO_FLEXIBILIZACAO_MANUAL
,D.QUARTIL
,CASE WHEN D.NUCONTRATO IS NOT MISSING THEN (CASE WHEN D.QUARTIL <= 2 THEN 1 ELSE 0 END) ELSE 0 END AS FLAG_FLEXIBILIZACAO_MANUAL

,E.GRUPO_INICIATIVA


,F.TESTE_CONTROLE AS TESTE_CONTROLE_CONTRA
,CASE WHEN F5.CD_CONTRATO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_CONTRA_PESADOS

,CASE
	WHEN F5.CD_CONTRATO IS NOT MISSING THEN F5.PCT_CONTRA2
	WHEN F.NUCONTRATO IS NOT MISSING THEN F.DESCONTO
	END AS DESCONTO_CONTRA
,CASE WHEN A.RDG_NUM >= 20 THEN 'TESTE' ELSE 'CONTROLE' END AS FLAG_TESTE_CONTROLE
,CASE WHEN F1.NUCONTRATO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_BOLHA
,CASE WHEN F2.GRUPO_TEST = 'MIGRACAO_WO' THEN 1 ELSE 0 END AS FLAG_MIGRACAO
,CASE WHEN F2.GRUPO_TEST = 'LIMPEZA_CARTEIRA' THEN 1 ELSE 0 END AS FLAG_LIMPEZA

,CASE WHEN F1.NUCONTRATO IS NOT MISSING THEN (
      CASE WHEN A.RDG_NUM >= 50 THEN 'TESTE' ELSE 'CONTROLE' END)

      ELSE (CASE WHEN A.RDG_NUM >= 20 THEN 'TESTE' ELSE 'CONTROLE' END) END AS FLAG_TESTE_CONTROLE

,F.TESTE_CONTROLE AS TESTE_CONTROLE_CONTRA_V2
,F.DESCONTO AS DESCONTO_CONTRA_V2

LEFT JOIN LDUARTE.BASE_ALAVANCAGEM_DESC_202503 F2 ON
A.NUCONTRATO = F2.NUCONTRATO


LEFT JOIN DESCONTO F5 ON
A.NUCONTRATO = INPUT(F5.CD_CONTRATO, 14.)

GROUP BY
	A.NUCONTRATO

	F.DTPROCESSAMENTO = MAX(F.DTPROCESSAMENTO)
