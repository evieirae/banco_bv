create table bases.TbTipoCanalProducaoProposta as
select distinct
	Proposta.NuProposta
    ,case
        when Proposta.CdColaborador =  2826 then 'CONCILIACAO'
        when Proposta.CdColaborador in (56380,11308)
			 OR ParceiroComercial.NmParceiroComercial LIKE '%CP GARAN TMLKT%'
			 OR ParceiroComercial.CdParceiroComercial IN(466807,418607,468271,472272) then 'TELEMARKETING'
        when Proposta.CdColaborador = 1000000 then 'CANAL DIGITAL'
		when Proposta.CdParceiroComercial IN (378215,451012,419715,448756,460298,467954,468906,449109,452671,455884,454428,462320,460903,462486,468957,469033,469213,468126,

											  465690,465799) then 'CANAL DIGITAL'
		when Proposta.CdParceiroComercial IN (465399,424584,452470,414643) then 'CANAL DIRETO'
		when ParceiroComercial.CdParceiroComercial = 365352	then 'PRAVALER'
        when Proposta.CdFilial = 1 then 'BV FINANCEIRA - OPERACAO ESPECIAL'
        when Proposta.CdFilial in (275,276) then 'NEGOCIOS CENTRALIZADOS'
		when Filial.DsFilial like 'CP CONS%' then 'CANAL PRIVADO'
		when Filial.DsFilial like 'CP SOLAR%' then 'CANAL SOLAR'
	    when ParceiroComercial.CdTipoParceiroComercial = 12 then 'LOJA CANAL NEGOCIOS'
		when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and upper(ParceiroComercialCoban.NmPromotora) like 'BV %' then 'LOJA CANAL NEGOCIOS'
        when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and upper(ParceiroComercialCoban.NmPromotora) not like 'BV %' then 'PROMOTORA'
        when Filial.DsFilial like 'CP %' and ParceiroComercialCoban.NmPromotora is null then 'LOJA CANAL NEGOCIOS'
        when Filial.DsFilial like 'CP %' and upper(ParceiroComercialCoban.NmPromotora) like 'BV %' then 'LOJA CANAL NEGOCIOS'
        when Filial.DsFilial like 'CP %' and upper(ParceiroComercialCoban.NmPromotora) not like 'BV %' then 'PROMOTORA'
        when Filial.DsFilial not like 'CP %' and ParceiroComercial.CdTipoParceiroComercial = 2 then 'CONCESSIONARIAS'
        when Filial.DsFilial not like 'CP %' and ParceiroComercial.CdTipoParceiroComercial = 5 then 'MULTIMARCAS'
     end as DsTipoCanalProducao
	 ,case
        when Proposta.CdColaborador =  2826 then 'CONCILIACAO'
        when Proposta.CdColaborador in (56380,11308)
			 OR ParceiroComercial.NmParceiroComercial LIKE '%CP GARAN TMLKT%'
			 OR ParceiroComercial.CdParceiroComercial IN(466807,418607,468271,472272) then 'TELEMARKETING'
        when Proposta.CdColaborador = 1000000 then 'CANAL DIGITAL'
		when Proposta.CdParceiroComercial IN (378215,451012,419715,448756,460298,467954,468906,449109,452671,455884,454428,462320,460903,462486,468957,469033,469213,468126,
											  412970,412971,412972,413732,414401,414420,433211,443941,444528,448742,448744,450421,453649,461517,461648,461879,462438,463084,
											  465690,465799) then 'CANAL DIGITAL'
		when Proposta.CdParceiroComercial IN (465399,424584,452470,414643) then 'CANAL DIRETO'
		when ParceiroComercial.CdParceiroComercial = 365352	then 'PRAVALER'
        when Proposta.CdFilial = 1 then 'BV FINANCEIRA - OPERACAO ESPECIAL'
        when Proposta.CdFilial in (275,276) then 'NEGOCIOS CENTRALIZADOS'
		when upper(ParceiroComercial.NmParceiroComercial) like '%GR CONSULTOR%' then 'GR CONSULTOR'
		when Proposta.CdParceiroComercial = 418254 then 'MEU FINANCIAMENTO SOLAR'
		when Proposta.CdParceiroComercial = 378215 then 'PORTAL SOLAR'
		when Filial.DsFilial like 'CP SOLAR%' then 'CANAL SOLAR'
	    when ParceiroComercial.CdTipoParceiroComercial = 12 then 'LOJA CANAL NEGOCIOS'
		when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and upper(ParceiroComercialCoban.NmPromotora) like 'PARLA %' then 'PARLA'
		when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and Proposta.FlPropostaBiPartida = 1 then 'BIPARTIDO'
		when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and upper(ParceiroComercialCoban.NmPromotora) like 'BV %' then 'LOJA CANAL NEGOCIOS'
        when ParceiroComercial.CdTipoParceiroComercial in (3,4,6) and upper(ParceiroComercialCoban.NmPromotora) not like 'BV %' then 'PROMOTORA'
        when Filial.DsFilial like 'CP %' and ParceiroComercialCoban.NmPromotora is null then 'LOJA CANAL NEGOCIOS'
        when Filial.DsFilial like 'CP %' and upper(ParceiroComercialCoban.NmPromotora) like 'BV %' then 'LOJA CANAL NEGOCIOS'
        when Filial.DsFilial like 'CP %' and upper(ParceiroComercialCoban.NmPromotora) not like 'BV %' then 'PROMOTORA'
        when Filial.DsFilial not like 'CP %' and ParceiroComercial.CdTipoParceiroComercial = 2 then 'CONCESSIONARIAS'
        when Filial.DsFilial not like 'CP %' and ParceiroComercial.CdTipoParceiroComercial = 5 then 'MULTIMARCAS'
     end as DsTipoSubCanalProducao

        on  Proposta.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

	);run;
