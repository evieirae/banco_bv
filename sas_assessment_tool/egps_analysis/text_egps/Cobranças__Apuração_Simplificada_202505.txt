CREATE TABLE RECEBIMENTOS AS
SELECT DISTINCT
A.*,
A.DtProcessamento AS DT_PROC_PAG

FROM CONNECTION TO PERM(

		SELECT
		CAST(A.NUCONTRATO AS INT64) AS NUCONTRATO,
		A.SQCONTRATOFINANCEIRO,
		DATE(A.DtProcessamento) AS DtProcessamento,
		'FLEX' AS SISTEMA,
		Count(A.NuParcela) AS QtdParcelas,
		SUM(CAST(A.VrRecebido AS FLOAT64)) AS TotalRecebido,
		SUM(CAST(A.VrPrincipalDevido AS FLOAT64)) AS TotalDevido,
		SUM(CAST(A.VrPrincipalBaixado AS FLOAT64)) AS TotalBaixado,
		FROM bv-pdbd-prd.trusted_stgtxt_flex.tbparcelarecebimento AS A

		LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
			ON A.NUCONTRATO = B.CD_CONTRATO
			AND A.SQCONTRATOFINANCEIRO = B.SQ_CONTRATO_FINANCEIRO
			AND A.NuParcela = B.Nu_Parcela

		LEFT JOIN bv-pdbd-prd.trusted_stgtxt_flex.tbcontrato AS C
			ON A.NUCONTRATO = C.NUCONTRATO

		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS D
			ON A.NUCONTRATO = D.NUCONTRATOGESTAO

		WHERE A.DTRECEBIMENTO >= '2025-03-20' AND  A.FlEstornado = 'N' AND
		((C.CDPRODUTO = 13 AND D.CDSUBPRODUTO = 70) OR (C.CDPRODUTO = 13 AND C.CDMODALIDADEPRODUTO = 55 AND D.CDSUBPRODUTO IS NULL))
		GROUP BY A.NUCONTRATO,
		A.SQCONTRATOFINANCEIRO,
		A.DtProcessamento

		UNION ALL

		SELECT
		CAST(A.CD_CONTRATO AS INT64) AS NUCONTRATO,
		A.SQ_CONTRATO_FINANCEIRO AS SQCONTRATOFINANCEIRO,
		A.DT_PROCESSAMENTO AS DtProcessamento,
		'GCON' AS SISTEMA,
		Count(A.Nu_Parcela) AS QtdParcelas,
		SUM(A.Vr_Recebido) AS TotalRecebido,
		SUM(A.Vr_Principal_Devido) AS TotalDevido,
		SUM(A.Vr_Principal_Baixado) AS TotalBaixado
		FROM bv-pdbd-prd.dvry_core.tbparcelarecebimento AS A

		LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
			ON A.CD_CONTRATO = B.CD_CONTRATO
			AND A.SQ_CONTRATO_FINANCEIRO = B.SQ_CONTRATO_FINANCEIRO
			AND A.Nu_Parcela = B.Nu_Parcela

		LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato  AS C
			ON A.CD_CONTRATO = C.CD_CONTRATO

		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS D
			ON A.CD_CONTRATO = D.NUCONTRATOGESTAO

	 WHEN CALCULATED ATRASO_PAG <= 45 THEN '0016 A 0045'

	 WHEN A.ELEG_ENCARGOS LIKE ('%Controle%') AND RDG_ENC >= 90  THEN 'CONTROLE'

	ON A.NUCONTRATO = B.NUCONTRATO
	AND A.DT_INICIO <= B.DT_PROC_PAG
