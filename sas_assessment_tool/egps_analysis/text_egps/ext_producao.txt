PROC SQL;
CREATE TABLE TbRastreador AS
  SELECT distinct
    NuCpfCnpj,
	Nuproposta,
	input(put(DtSafra,yymmn6.),6.) as NuAnoMes,
	DtSafra,
	Rastreador
  FROM
    AQUISIC.PREANALISE_SIMU_PROP
  WHERE 1=1
	and Visao = 'Simulacao'
    and RASTREADOR in('DERIVADO','APROVADO')
	and FlCPFUnicoMes = 1
	and Nuproposta <> .
/*	and NuCpfCnpj = 0000936375*/
	order by 5,1 desc
;RUN;

proc sql;
	create table TMP.TbProducao as
		select distinct
			Contrato.NuAnoMes
			,Contrato.NuProposta
			,datepart(Contrato.DtEntradaProposta) as DtEntradaProposta format date9.
			,Contrato.NuContrato
			,Contrato.SqContratoFinanceiro
			,Contrato.NuContratoLegado
			,datepart(Contrato.DtContrato) as DtContrato format date9.
			,datepart(Contrato.DtLiquidacaoEfetiva) as DtLiquidacaoEfetiva format date9.
			,datepart(Contrato.DtCancelamento) as DtCancelamento format date9.
			,datepart(Contrato.DtInicioVigencia) as DtInicioVigencia format date9.
			,datepart(Contrato.DtFimVigencia) as DtFimVigencia format date9.
			,datepart(Contrato.DtPrimeiroVencimento) as DtPrimeiroVencimento format date9.
			,datepart(Contrato.DtUltimoVencimento) as DtUltimoVencimento format date9.
			,Contrato.CdColaboradorPromotor
			,
			case
				when Contrato.CdColaborador <> coalesce(Contrato.CdColaboradorPromotor,Contrato.CdColaborador) then
					case
						when ColaboradorPromotor.CdGrupoCargoColaborador in (5,15) and GrupoProdutoMeta.NmGrupoProdutoMeta = 'Privado'
							then 1
						when ColaboradorPromotor.CdGrupoCargoColaborador = 2 and GrupoProdutoMeta.NmGrupoProdutoMeta = 'CIG'
							then 1
						else 0
					end
				else 0
			end
			as FlProducaoHibrida
			,TipoCanalProducao.DsTipoCanalProducao
			,TipoCanalProducao.DsTipoSubCanalProducao
			,
			case
				when Contrato.nuanomes >= 202301 and TipoCanalProducao.DsTipoCanalProducao in ('CONCESSIONARIAS','MULTIMARCAS')
					then 'CANAL VEICULOS'
				else TipoCanalProducao.DsTipoCanalProducao
			end as DsTipoCanalProducaoMeta
			,EstruturaComercial.NmCanalEstruturaComercial
			,Contrato.CdColaborador
			,Colaborador.NmColaborador
			,EstruturaComercialN1.NmGrupoCargo as NmGrupoCargoColaborador
			,Colaborador.DsLogin as DsLoginN1

			,case
				when ParceiroComercial.CdParceiroComercial = 414921 then Colaborador.DsLogin
				when ParceiroComercial.NmParceiroComercial LIKE '%BV BALCAO%' then Colaborador.DsLogin

			/*Ajuste para pegar os parceiros do MFS - Sinergia para Produção Não Isonomica*/
				when Contrato.CdParceiroComercial = 418254 and EstruturaComercial.NmCanalEstruturaComercial = 'Canal Solar' and Cartni_mfs.NuCnpjLoja NE '' Then Cartni_mfs.DsLoginN1
			/*CONTRATOS DE LEVES USADOS CV QUE ESTÃO SEM MARCAÇÃO NO CAMPO DSLOGINN1CARTPDV*/
			/*PDVS MIGRADOS PARA OUTROS CANAIS, PORÉM COM PRODUÇÃO ATRELADO AO GR*/
				when Contrato.NuContrato = 12357000002241 THEN 'william.lima'

				else Cartni.DsLoginN1
			end AS DsLoginN1CartPDV

			,coalesce(PropostaSinergia.CdColaboradorC, ParceiroComercial.CdColaboradorC) as CdColaboradorC
			,coalesce(PropostaSinergia.NmColaboradorC, ParceiroComercial.NmColaboradorC) as NmColaboradorC
			,coalesce(PropostaSinergia.DsLoginN1C, ParceiroComercial.DsLoginN1C) as DsLoginN1C
			,Contrato.CdFilial
			,Contrato.DsFilial
			,EstruturaComercial.CdAreaEstruturaComercial
			,EstruturaComercial.NmAreaEstruturaComercial
			,EstruturaComercial.DsLoginN2A
			,EstruturaComercial.DsLoginN2H
			,coalesce(PropostaSinergia.CdFilialC, ParceiroComercial.CdFilialC) as CdFilialC
			,coalesce(PropostaSinergia.NmFilialC, ParceiroComercial.NmFilialC) as NmFilialC
			,coalesce(PropostaSinergia.CdAreaEstruturaComercialC, ParceiroComercial.CdAreaEstruturaComercialC) as CdAreaEstruturaComercialC
			,coalesce(PropostaSinergia.NmAreaEstruturaComercialC, ParceiroComercial.NmAreaEstruturaComercialC) as NmAreaEstruturaComercialC
			,coalesce(PropostaSinergia.DsLoginN2C, ParceiroComercial.DsLoginN2C) as DsLoginN2C
			,EstruturaComercial.NmSubEstruturaComercialA
			,EstruturaComercial.NmSubEstruturaComercialH
			,EstruturaComercial.DsLoginN3A
			,EstruturaComercial.DsLoginN3H
			,coalesce(PropostaSinergia.NmSubEstruturaComercialC, ParceiroComercial.NmSubEstruturaComercialC) as NmSubEstruturaComercialC
			,coalesce(PropostaSinergia.DsLoginN3C, ParceiroComercial.DsLoginN3C) as DsLoginN3C
			,EstruturaComercial.LstSubstituicao
			,ParceiroComercial.LstSubstituicaoC
			,
		case
			when Tbpropostamultibanco.nuproposta is not null then cats(GrupoProdutoMeta.NmGrupoProdutoMeta,Tbpropostamultibanco.Tipo)
			when BVO.CdParceiroComercial<>. and GrupoProdutoMeta.NmGrupoProdutoMeta = 'Leves' and contrato.fxidade ='ZeroKM' then 'Leves-BVO'
			when Contrato.CdParceiroComercial in (477503) then 'Credito Auto-BV'
			else GrupoProdutoMeta.NmGrupoProdutoMeta
		end
		as NmGrupoProdutoMeta
		,
		case
			when GrupoProdutoMeta.NmGrupoProdutoMeta = 'Solar'
				then coalesce(ParceiroSinergia.NuCnpjLoja,PropostaSinergia.NuCnpjLoja,ParceiroComercial.NuCnpjLoja)
		end
		as NuCnpjLojaFavorecido
		,
		case
			when GrupoProdutoMeta.NmGrupoProdutoMeta <> 'Solar'
				then ''
			when Contrato.CdParceiroComercial in (418254, 378215) and ParceiroSinergia.NmParceiroComercial is null
				then 'Parceiro não cadastrado'
			when Contrato.CdParceiroComercial Not in (418254, 378215)
				then ParceiroComercial.NmParceiroComercial
			else ParceiroSinergia.NmParceiroComercial
		end
		as NmParceiroComercialFavorecido

	,case
		when GrupoProdutoMeta.NmGrupoProdutoMeta <> 'Solar' then .
		when Contrato.CdParceiroComercial in (418254, 378215) and ParceiroSinergia.NmParceiroComercial is not null then 1
		when Contrato.CdParceiroComercialCoban in (418254, 378215) Then 1
	 	else 0
	end as FlParceiroSinergia

	,case
		when GrupoProdutoMeta.NmGrupoProdutoMeta <> 'Solar' then .
		when Contrato.CdParceiroComercialCoban in (418254, 378215) and Contrato.CdColaborador in (., 1000000) Then 2
		when Contrato.CdParceiroComercialCoban in (418254, 378215) and Contrato.CdColaborador Not in (., 1000000) Then 1
		when PropostaSinergia.NuProposta is not null then 1
		else 0
	end as FlPropostaSinergia

	,case
		when GrupoProdutoMeta.NmGrupoProdutoMeta <> 'Solar' Then ''
		when GrupoProdutoMeta.NmGrupoProdutoMeta = 'Solar' and TipoCanalProducao.DsTipoCanalProducao = 'CANAL DIGITAL' Then 'Canal MFS'
		when Calculated FlPropostaSinergia = 2 Then 'Substabelecido'
		when Calculated FlPropostaSinergia = 1 and Calculated FlParceiroSinergia = 1 Then 'Sinergia'
		when Calculated FlPropostaSinergia = 1 and Calculated FlParceiroSinergia = 0 Then 'Canal MFS'
		when Calculated FlPropostaSinergia = 0 and Calculated FlParceiroSinergia = 0 and Contrato.CdParceiroComercial in (418254, 378215) Then 'Canal MFS'
		when Calculated FlPropostaSinergia = 0 and Calculated FlParceiroSinergia = 0 Then 'Canal Solar'
		else 'Canal Solar'
	end as CanalProducaoSolar

		,Base_Logins.DsLogin_Vendedor as DsLoginOriginador
		,ParceiroComercial.NuCnpjLoja
		,Contrato.CdParceiroComercial

		,ParceiroComercial.NmParceiroComercial
		,ParceiroComercial.NmTipoParceiroComercial
		,ParceiroComercial.FlIntegracaoParceiroFeI
		,Contrato.CdParceiroComercialCoban
		,ParceiroComercialCoban.NmParceiroComercial as NmParceiroComercialCoban
		,ParceiroComercialCoban.NmPromotora
		,ParceiroComercialCoban.DsTipoCanalPromotora
		,Contrato.CdPessoa
/*		,Contrato.NmPessoa*/
		,compress(Contrato.NmPessoa,'0A0D'x) as NmPessoa
		,Contrato.TpPessoa
		,Contrato.NuCpfCnpj
		,Contrato.CdUfPessoa
		,Contrato.NmCidadePessoa
		,Contrato.QtAnosIdadePessoa
		,Contrato.CdTabelaFinanciamento
		,TabelasFinanciamento.NmTipoTabelaFinanciamento
		,Contrato.CdTipoCobranca
		,Contrato.NmTipoCobranca
		,Contrato.QtAnosIdade
		,Contrato.FxIdade
		,Contrato.QtParcelas
		,Contrato.FxParcela
		,Contrato.PcEntradaMolicar
		,Contrato.FxEntrada
		,Contrato.NuScoreFim
		,'SCORE '||
		case
			when Contrato.NuScoreFim >= 9
				then '9-10'
			when Contrato.NuScoreFim >= 8
				then '8-9'
			when Contrato.NuScoreFim >= 7
				then '7-8'
			when Contrato.NuScoreFim >= 6
				then '6-7'
			when Contrato.NuScoreFim >= 5
				then '5-6'
			else '< 5'
		end
		as FxScoreFim
		,Contrato.NuRating
		,Contrato.FlTxAlterada
		,Contrato.FlRefinConsignado
		,Contrato.FlCompraDivida
		,Contrato.FlContratoExcecao
		,Contrato.CdDocumentoAlienacao
		,Contrato.StDocumentoAlienacao
		,
		case
			when BaseDNE.NuContrato <> .
				then 1
			else 0
		end
		as FlDNE
		,Contrato.VrCliente
		,Contrato.VrRefin
		,Contrato.VrCompra
		,Contrato.VrOutroDesdobro
		,Contrato.VrProducaoLiquida
		,Contrato.VrProducao
		,Contrato.VrContrato
		,Contrato.VrEntrada
		,ResultadoGerencial.VrMargemContribuicao
		,ResultadoGerencial.VrMargemContribuicaoFinal
		,Case
			When Contrato.NuAnomes <= 202410 Then coalesce(ResultadoGerencial.VrMargemContribuicaoFinal,ResultadoGerencial.VrMargemContribuicao)
			Else ResultadoGerencial.VrMargemContribuicao
		End as VrMargem
		,Contrato.VrObjGarantia
		,Contrato.VrComissaoTotal
		,Contrato.VrComissaoParcelada
		,ResultadoGerencial.VrAcordoComercial
		,ResultadoGerencial.VrDurationVp as VpDuration
		,Contrato.VrFinanciamento
		,ResultadoGerencial.VrPerdaEstimada
		,Contrato.VrTarifaCadastro
		,Contrato.VrTarifaAvaliacaoBem
		,Contrato.VpTxGerencial
		,
		case
			when Contrato.VrRetencao > 1
				then (Finance('RATE', Contrato.QtParcelas, Contrato.VrPrestacao, -((Contrato.VrContrato-Contrato.VrRetencao)+Contrato.VrIof+coalesce(Contrato.VrTarifaCadastro,0)))*Contrato.VrProducao)
			else Contrato.VpTxGerencial
		end
		as VpTxGerencialRebate
		,Contrato.VpTxJuros
		,ResultadoGerencial.PcRoe
		,ResultadoGerencial.VpRoe
		,
		(case
			when GrupoProdutoMeta.NmGrupoProdutoMeta = 'Solar' then TaxaMercadoSolar.TxTirDivulgada
			else TaxaMercado.TxTirDivulgada
		end)
		as TxTirDivulgada
		,(calculated TxTirDivulgada/100) * (Contrato.VrProducao) as VpTirDivulgada
		,
		(case
			when GrupoProdutoMeta.NmGrupoProdutoMeta = 'Solar'
				then TaxaMercadoSolar.TxTirContabil
			else TaxaMercado.TxTirContabil
		end)
		as TxTirContabil
		,(calculated TxTirContabil/100) * (Contrato.VrProducao) as VpTirContabil
		,((1+(Contrato.VpTxGerencial/Contrato.VrProducao)) / (1+(((calculated TxTirDivulgada/100) * (Contrato.VrProducao))/Contrato.VrProducao)) -1 ) * Contrato.VrProducao AS VpSpread
		,((1+(Contrato.VpTxGerencial/Contrato.VrProducao)) / (1+(((calculated TxTirContabil/100) * (Contrato.VrProducao))/Contrato.VrProducao)) -1 ) * Contrato.VrProducao AS VpSpreadContabil
		,
		((1+(
			(case
				when Contrato.VrRetencao > 1
					then (Finance('RATE', Contrato.QtParcelas, Contrato.VrPrestacao, -((Contrato.VrContrato-Contrato.VrRetencao)+Contrato.VrIof+coalesce(Contrato.VrTarifaCadastro,0)))*Contrato.VrProducao)
				else Contrato.VpTxGerencial
			end)
		/Contrato.VrProducao)) / (1+(((calculated TxTirDivulgada/100) * (Contrato.VrProducao))/Contrato.VrProducao)) -1 ) * Contrato.VrProducao AS VpSpreadRebate
		,((1+(

			when ParceiroComercial.DsGrupoParceiroLocadoras <> 'LOCALIZA' and GrupoProdutoMeta.FlAutoCompleto = 1 and Contrato.QtAnosIdade between 0 and 7 and Contrato.VrProducao <= 200000 and Contrato.TpPessoa = 'F' then 1
			else 0
		end as FlCompletoEleg

		,case
			when Rastreador.Rastreador =  'APROVADO'
				then 2
			when Rastreador.Rastreador =  'DERIVADO'
				then 1
				else 0
		end as FlRastreador

		,case
			when Tbpropostamultibanco.tipo = '-Multibanco'
				then 1
			else 0
		end as FlMultibanco
		,Atualizacao.DtAtualizacao
		,Atualizacao.TsProcessamento

		from TbExtContrato as Contrato

		left join basegrc.TbGrupoProdutoMeta GrupoProdutoMeta
		on  Contrato.CdGrupoSubProduto = GrupoProdutoMeta.CdGrupoSubProduto
		left join bases.TbTipoCanalProducaoContrato TipoCanalProducao
		on Contrato.NuContrato = TipoCanalProducao.NuContrato
		left join bases.TbEstruturaComercial EstruturaComercial
		on  Contrato.CdFilial = EstruturaComercial.CdFilial
		and Contrato.NuAnoMes  = EstruturaComercial.NuAnoMes
		and EstruturaComercial.CdFilial is not null
		and EstruturaComercial.NmNivel = 'N2'
		left join bases.TbColaborador Colaborador
		on  Contrato.CdColaborador = Colaborador.CdColaborador
		left join bases.TbEstruturaComercial EstruturaComercialN1
		on  Colaborador.CdColaboradorBI = EstruturaComercialN1.CdColaboradorBI
		and Contrato.NuAnoMes  = EstruturaComercialN1.NuAnoMes
		and EstruturaComercialN1.CdColaboradorBI is not null
		and EstruturaComercialN1.NmNivel = 'N1'
		left join bases.TbColaborador ColaboradorPromotor
		on  Contrato.CdColaboradorPromotor = ColaboradorPromotor.CdColaborador
		left join tmp.TbParceiroComercial ParceiroComercial
		on  Contrato.CdParceiroComercial = ParceiroComercial.CdParceiroComercial
		left join tmp.TbParceiroComercial ParceiroComercialCoban
		on  Contrato.CdParceiroComercialCoban = ParceiroComercialCoban.CdParceiroComercial
		left join bases.TbDimTaxaMercado TaxaMercado
		on  datepart(Contrato.DtContratoFinanceiro) = TaxaMercado.DtReferencia
		and Contrato.QtParcelas = TaxaMercado.QtParcelas
		and Contrato.CdTipoMercado = TaxaMercado.CdTipoTaxaMercadoOrigem
		left join bases.TbDimTaxaMercado TaxaMercadoSolar
		on  datepart(Contrato.DtEntradaProposta) = TaxaMercadoSolar.DtReferencia
		and Contrato.QtParcelas = TaxaMercadoSolar.QtParcelas
		and Contrato.CdTipoMercado = TaxaMercadoSolar.CdTipoTaxaMercadoOrigem
		left join bases.TbResultadoGerencial ResultadoGerencial
		on  Contrato.NuContrato = ResultadoGerencial.NuContrato
		and Contrato.SqContratoFinanceiro = ResultadoGerencial.SqContratoFinanceiro
		left join bases.TbExtSeguros Seguro
		on 	Contrato.NuContrato = Seguro.NuContrato
		and Contrato.SqContratoFinanceiro = Seguro.SqContratoFinanceiro
		left join basegrc.TbSeguroAcaoTaNaArea SeguroAcaoTaNaArea
		on  datepart(Contrato.DtEntradaProposta) >= SeguroAcaoTaNaArea.DtInicioVigencia
		and datepart(Contrato.DtEntradaProposta) <= SeguroAcaoTaNaArea.DtFimVigencia
		and Contrato.CdParceiroComercial = SeguroAcaoTaNaArea.CdParceiroComercial
		left join basegrc.TbTabelasFinanciamento TabelasFinanciamento
		on Contrato.CdTabelaFinanciamento = TabelasFinanciamento.CdTabelaFinanciamento
		left join bases.TbTempoData TempoData
		on  datepart(Contrato.DtContrato) = TempoData.DtTempo
		left join bases.TbTempoData TempoDataProposta
		on  datepart(Contrato.DtEntradaProposta) = TempoDataProposta.DtTempo

		left join bases.TbPropostaComplementoParceiro PropostaSinergia
		on  Contrato.NuProposta = PropostaSinergia.NuProposta

		left join TbPropostaComplementoParceiro ParceiroSinergia
		on  coalesce(PropostaSinergia.NuCnpjLoja, ParceiroComercial.NuCnpjLoja) = ParceiroSinergia.NuCnpjLoja

		left join TbAtualizacao Atualizacao
		on 1=1
		left join basegrc.TbRastreador  Rastreador
		on Contrato.nuproposta = Rastreador.nuproposta
	/*		and Contrato.nuanomes = Rastreador.nuanomes*/
		left join tbencarteiramentoni cartni
			on Contrato.NuAnoMes = Cartni.NuAnoMes
			and Contrato.CdParceiroComercial = Cartni.CdParceiroComercial
			and GrupoProdutoMeta.NmGrupoProdutoMeta = cartni.Produto

	/********Pegar lojas do Solar quando vier pelo MFS*********/
		left join tbencarteiramentoni cartni_mfs
			 on Contrato.NuAnoMes = Cartni_mfs.NuAnoMes
			and Cartni_mfs.Produto = 'Solar'
			and EstruturaComercial.NmCanalEstruturaComercial = 'Canal Solar'
			and (Case when GrupoProdutoMeta.NmGrupoProdutoMeta = 'Solar'
					  then coalesce(ParceiroSinergia.NuCnpjLoja,PropostaSinergia.NuCnpjLoja,ParceiroComercial.NuCnpjLoja)
				  end) = Cartni_mfs.NuCnpjLoja

		LEFT JOIN TbEncarteiramentoNIDryve cartniDryve
			ON 	Contrato.NuAnoMes = cartniDryve.NuAnoMes
			and Contrato.CdParceiroComercial = cartniDryve.CdParceiroComercial
			and GrupoProdutoMeta.NmGrupoProdutoMeta = cartniDryve.Produto
			and Contrato.CdColaborador = cartniDryve.CdColaborador

		left join basegrc.Tbpropostamultibanco  Tbpropostamultibanco
			on Contrato.nuproposta = Tbpropostamultibanco.nuproposta

		left join basegrc.TbSpreadTeorico TbSpreadTeorico
			on Contrato.nuanomes = TbSpreadTeorico.nuanomes
			and EstruturaComercial.NmAreaEstruturaComercial = TbSpreadTeorico.Filial
			and GrupoProdutoMeta.NmGrupoProdutoMeta = 'Leves'
			and EstruturaComercial.NmCanalEstruturaComercial <> 'Canal Negócios'

		left join Bases.Base_DsLogin_Originador AS Base_Logins
	        ON Contrato.NuProposta = Base_Logins.NuProposta

		left join TbDneCarteiraPend  BaseDNE
			on Contrato.NuContrato = BaseDNE.NuContrato

		LEFT JOIN BASEGRC.TbLevesNovosBVO BVO
    		ON BVO.CdParceiroComercial = Contrato.CdParceiroComercial
			and BVO.nuanomes = Contrato.nuanomes
		;
run;

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbDimHierarquiaComercial as
select * from connection to dbcon
(
	with TbDeParaFilial as
	(

		SELECT
			 DeParaFilial.CdAreaEstruturaComercial
			,DeParaFilial.CdFilial
			,Filial.NmFilial
			,CASE
				WHEN Filial.NmFilial LIKE 'CP SOL%' THEN 5
				WHEN Filial.NmFilial LIKE 'CP CONS%' THEN 4
				WHEN DeParaFilial.CdFilial IN (275,276) THEN 3
				WHEN Filial.NmFilial LIKE 'CP %' THEN 2
				WHEN Filial.NmFilial NOT LIKE 'CP ' THEN 1
			 ELSE 0 END AS CdCanalEstruturaComercial

		FROM DWBI_DTWH.TbDeParaFilial DeParaFilial

			LEFT JOIN DWBI_DTWH.TbFilial Filial
				ON  DeParaFilial.CdFilial = Filial.CdFilialSistemaOrigem
				AND Filial.CdSistemaOrigem = 209

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial Area
				ON  DeParaFilial.CdAreaEstruturaComercial = Area.CdAreaEstruturaCmrclStmaOrgm
	)

	SELECT
		 CAST (Pessoa.CdPessoaOrigem AS INTEGER) CdOperador
		,TempoData.DtTempo
		,DeParaFilial.CdFilial

	FROM DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoa

		LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
			ON  PessoaRelacaoPessoa.CdPessoa = Pessoa.CdPessoa

		INNER JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
			ON  PessoaRelacaoPessoa.CdPessoaRelacao = HierarquiaComercial.CdColaborador

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
				ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

			LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
				ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

			INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
				ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
				AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador
				AND DeParaHierarquia.CdCanalEstruturaComercial IN (2,4)

			INNER JOIN TbDeParaFilial DeParaFilial
				ON  DeParaHierarquia.CdAreaEstruturaComercial = DeParaFilial.CdAreaEstruturaComercial
				AND DeParaHierarquia.CdCanalEstruturaComercial = DeParaFilial.CdCanalEstruturaComercial

			INNER JOIN DWBI_DIME.TbDimTempoData TempoData
				ON  TempoData.DtTempo BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
				AND TempoData.DtTempo > '20000101'
				AND TempoData.DtTempo < today()

	WHERE PessoaRelacaoPessoa.CdTipoRelacaoPessoa = 7
	  AND today() BETWEEN PessoaRelacaoPessoa.DtInicioVigencia AND PessoaRelacaoPessoa.DtFimVigencia
)
;run;

	,TempoData.DtTempo
