proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
   schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table Bases.TagVeicular as
   select Distinct * from connection to perm
	(
	  select * from `bv-pdbd-prd.dvry_negc_conta.contadigital_tagbv_fato`
	)
;Quit;

%VerificaAtualizacao(/projetos/MISV/data/bases/tmp/tbproducao.sas7bdat,60,150);

Proc Sql;
Create Table TagVeicular as
	  select
	  	Prod.NuAnoMes
		,Prod.NuProposta
		,Prod.NuContrato
		,Prod.DtContrato
		,Prod.NmSubEstruturaComercialA
		,Prod.NmCanalEstruturaComercial
		,Prod.NmAreaEstruturaComercial
		,Prod.NmColaborador
		,Prod.DsLoginN1
		,Prod.NmAreaEstruturaComercialC
		,Prod.NmColaboradorC
		,Prod.DsLoginN1C
		,Prod.DsLoginN1CartPDV
		,Prod.DsLoginN2C
		,Prod.NmGrupoProdutoMeta
		,Prod.DsLoginOriginador
		,Prod.NuCnpjLoja
		,Prod.CdParceiroComercial
		,Prod.NmParceiroComercial
		,Prod.NmPessoa
		,Prod.TpPessoa
		,Prod.NuCpfCnpj
		,Prod.VrProducao
		,Prod.DtAtualizacao

		,Conta.NuConta
		,Conta.DtAbertuaCC
		,Conta.DtAtivacao
		,Conta.FlContaAberta
		,Conta.FlDebitoAutomaticoEfetivado

		,Tag.ds_ultm_evento as Status_TAG
		,Tag.fl_solicitacao_app
		,Tag.dt_solicitacao
		,Tag.dt_ativacao
		,Tag.fl_tag_ativada
		,Tag.dt_primeiro_uso
		,Tag.fl_tag_utilizada
		,Tag.dt_cancelamento
		,Tag.fl_cancelada

		from tmp.TbProducao Prod
			Left join BaseGrc.TBABERTURACONTADARODAS Conta
				 On Prod.NuCpfCnpj = Conta.NuCpfCnpj

				 Left join Bases.TagVeicular as Tag
				 		On InPut(Conta.NuConta, 8.) = InPut(Tag.cd_conta_corrente, 8.)

WHERE Prod.NuAnomes >= 202501 and TpPessoa = 'F'
;Quit;
