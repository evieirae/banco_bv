PROC SQL;
CREATE TABLE BASE_ATIVA AS
	SELECT DISTINCT
		A.DT_BASE,
		A.NR_CONTRATO AS NUCONTRATO,
		A.SQCONTAT AS SQCONTRATOFINANCEIRO,
		A.NR_CPFCNPJ,
		A.NR_DIAS_ATRASO AS ATRASO,
		A.CD_ASSESSORIA,
		A.VL_DIVIDA AS RISCO_ENVIO,
		A.DTDISPONIBILIZACAO,
		A.DTNOVADISPONIBILIZACAO,
		A.TPCOB,
		A.DT_ENVIO,
		A.DT_ENVIO_NOVO,
		A.DT_VCT,
		A.PRORROG,
	    CASE WHEN B.CDMODALIDADEPRODUTO IN (1,2,3,6,8,27,38,41,45,56) THEN 'LEVES'
	         WHEN B.CDMODALIDADEPRODUTO IN (5) THEN 'MOTOS'
	         WHEN B.CDMODALIDADEPRODUTO IN (7) THEN 'VANS'
	         WHEN B.CDMODALIDADEPRODUTO IN (9,28) THEN 'PESADOS'
	         WHEN B.CDMODALIDADEPRODUTO IN (16,33) THEN 'INSS'

FROM CONNECTION TO PERM (

WITH BASE_GERACAO AS (

			 SELECT

					B.NUCONTRATOGESTAO,
					B.SQCONTRATOFINANCEIRO,
					B.DTGERACAO

			FROM bv-pdbd-prd.trusted_dbcontencioso.tbnegociacaocontrato B
			WHERE
					CAST(DTGERACAO AS DATE)>='2025-11-30' AND CAST(DTGERACAO AS DATE)<='2025-12-31' 

		),
BASE_PAGAMENTO AS (
			SELECT

				b.TPENTRADA,
				cast(b.NUPROCESSO AS NUMERIC) AS NUPROCESSO,
				CAST(A.NUCONTRATOGESTAO AS NUMERIC) AS NUCONTRATOGESTAO,
				A.SQCONTRATOFINANCEIRO,
				A.DTGERACAO,
				B.DTPAGAMENTO,
				B.DTVENCIMENTO

			FROM BASE_GERACAO A LEFT JOIN bv-pdbd-prd.trusted_dbcontencioso.tbnegociacaocontrato B
			ON
				A.NUCONTRATOGESTAO=B.NUCONTRATOGESTAO
			AND
				A.SQCONTRATOFINANCEIRO=b.SQCONTRATOFINANCEIRO
			AND
				A.DTGERACAO = B.DTGERACAO

			WHERE
				B.DTPAGAMENTO IS NOT NULL
			AND
				CAST(B.DtPagamento AS DATE) >= '2025-11-30' AND CAST(B.DtPagamento AS DATE) <= '2025-12-31'
			AND
				B.STNEGOCIACAO='PL')

			SELECT

				   A.*,
					CASE
						 WHEN TpEntrada='A' THEN 'ASSESSORIA'
						 WHEN TpEntrada='R' THEN 'ROBO'
						 WHEN TpEntrada='B' THEN 'CHATBOT'

						 WHEN TpEntrada='C' THEN 'SAC'
						 WHEN TpEntrada='U' THEN 'URA'
						 WHEN TpEntrada='H' THEN 'APP'
						 WHEN TpEntrada='F' THEN 'MKTPLACE SERASA'
						 WHEN TpEntrada='G' THEN 'MKTPLACE QUERO QUITAR'
						 WHEN TpEntrada='D' THEN 'MKTPLACE BLUE365'
						 WHEN TpEntrada='E' THEN 'MKTPLACE ACORDO CERTO'
						 WHEN TpEntrada='I' THEN 'ROBBU'
						 WHEN TpEntrada='J' THEN 'RECALIBRA'
						 WHEN TpEntrada='Q' THEN 'WHATSAPP BOT'
						 WHEN TpEntrada='M' THEN 'MANUAL ASSESSORIA'
						 WHEN TpEntrada='L' THEN 'WHATSAPP SF SAC'
						 WHEN TpEntrada='T' THEN 'WHATSAPP SF COB'
						 WHEN TpEntrada='V' THEN 'JABUTI'
						 WHEN TpEntrada='O' THEN 'GOOGLE'
						 ELSE 'OUTROS' END AS CANAL,
				   E.RAZAO_NOME,
				   B.COD_ACAO,
				   CAST(B.DT_ASSES AS DATE) AS DT_ENVIO,
				   CAST(B.DT_RET_ASSES AS DATE) AS DT_RETIRADA,
				   G.CDAGENTERECEBEDOR,
				   G.NMAGENTERECEBEDOR

				FROM BASE_PAGAMENTO A LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_processo B
				ON
				A.NUCONTRATOGESTAO= CAST(B.NUCONTRATOGESTAO AS NUMERIC)
				AND
				A.SQCONTRATOFINANCEIRO=B.SQCONTRATOFINANCEIRO
				AND
				CAST(A.NUPROCESSO AS INTEGER)=B.NUM_PROCESSO

				LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_agente E
				ON
				B.COD_ASSES=E.COD_AGENTE

				LEFT JOIN bv-pdbd-prd.dvry_core.tbparcelarecebimento F
				on A.NUCONTRATOGESTAO= CAST(F.CD_CONTRATO AS NUMERIC)
				AND
				A.SQCONTRATOFINANCEIRO=F.SQ_CONTRATO_FINANCEIRO AND CAST(A.DTPAGAMENTO AS DATE) = CAST(F.Dt_RECEBIMENTO AS DATE)

				LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbagenterecebedor G

PROC SQL;
CREATE TABLE ATIVA.BASE_ATIVA_IA_NOVO_&DT_BASE. AS SELECT
	A.*,
	UPCASE(B.COLLECTION) AS COLLECTION,

	CASE
		WHEN RISCO_ENVIO <= 5000 THEN 'a. ate 5mil'
		WHEN RISCO_ENVIO > 5000 AND RISCO_ENVIO <=10000  THEN 'b. 5mil a 10mil'
		WHEN RISCO_ENVIO > 10000 AND RISCO_ENVIO <=15000  THEN 'c. 10mil a 15mil'
		WHEN RISCO_ENVIO > 15000 AND RISCO_ENVIO <=20000  THEN 'd. 15mil a 20mil'
		WHEN RISCO_ENVIO > 20000 AND RISCO_ENVIO <=25000  THEN 'd. 20mil a 25mil'
		WHEN RISCO_ENVIO > 25000 AND RISCO_ENVIO <=30000  THEN 'e. 25mil a 30mil'
		WHEN RISCO_ENVIO > 30000 AND RISCO_ENVIO <=35000  THEN 'e. 30mil a 35mil'
		WHEN RISCO_ENVIO > 35000 						  THEN 'e. acima 35mil'
		ELSE 'verificar' END AS FX_RISCO,

	C.TPENTRADA AS TIPO_CANAL

FROM  ATIVA.BASE_ATIVA_IA_NOVO_&DT_BASE. A LEFT JOIN COLLECTION_5_30 B
ON
	A.NUCONTRATO = B.NUCONTRATO
	AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

ON
	A.NUCONTRATO = C.NUCONTRATOGESTAO

	A.SqContratoFinanceiro = C.SqContratoFinanceiro

CREATE TABLE LISTA_DATA AS

				A.DT_BASE,
				B.DU,
				A.DT_ENVIO_FINAL,
				A.TPCOB,
				A.ASSESSORIA,
				A.DSGRUPO,
				A.SEGMENTACAO,
				A.DSGRUPO_2,
				A.TP_CLASSE,
				A.COLLECTION,
				A.FX_RISCO,
				A.TIPO_CANAL,
				A.TP_CARTEIRA,
				A.FLAG_PARCELA,
				DTREBEB_FINAL,
				FL_PAGO,
				A.INCONSISTENCIA,
				A.FLAG_PILOTO;
		QUIT;

		RUN;

	%IF &J=1 %THEN
		%DO;

	%ELSE

	%END;

%END;
