PROC SQL;
   CREATE TABLE WORK.ATIVOS_RECORRENTES AS
   SELECT 
            (YEAR(t1.ANOMES)) AS ANO,
          
            (month(t1.ANOMES)) AS MES,
          t1.'DIVISAO COMERCIAL'n,
          t1.SEGMENTO,
          t1.'AREA PRODUTO'n,
          t1.PRODUTO,
          t1.VL_PRODUCAO_YTD,
          t1.Receita,
          t1.ESTOQUE_PRODUCAO,
          t1.'Tipo Res'n
      FROM EXTRATO.EXTRATO_&ANOMES_EXTRATO. t1;
QUIT;

DATA WORK.ATIVOS_RECORRENTES (COMPRESS=YES);
	SET WORK.ATIVOS_RECORRENTES;
FORMAT AREA_PRODUTO_AJ $CHAR100.;

IF PRODUTO IN ('CESSÃO ANTECIPA BV - ISENTO',
'CESSÃO ANTECIPA BV - sem IOF',
'CESSÃO DE REC S/ COOB MAND COB',
'Cessão  Antecipa BV VNP IPCA',
'Cessão Antecipa BV',
'Cessão Mand Cob TM',
'Cessão Dir. Credit. S/ Coobr. Mandatário Cobrança',
'Cessão Risco BV Emissor',
'Cessão de Direitos Creditórios sem Coobrigação',
'Cessão de Rec S/ Coob IF',
'Cessão de Rec S/ Coob Mand Cob',
'Cessão de Recebíveis S/ Coobrigação',
"CESSÃO ANTECIPA BV - SEM IOF",
"CESSÃO DE DIREITOS CREDITÓRIOS SEM COOBRIGAÇÃO",
"CESSÃO DIR. CREDIT. S/ COOBR. MANDATÁRIO COBRANÇA",
"CESSÃO DE RECEBÍVEIS S/ COOBRIGAÇÃO")
THEN AREA_PRODUTO_AJ = "CESSÃO";

IF 'AREA PRODUTO'n IN ("CESSAO ANTECIPA BV","CESSAO FI") THEN AREA_PRODUTO_AJ = "CESSÃO";

RUN;
DATA WORK.ATIVOS_RECORRENTES (COMPRESS=YES);
	SET WORK.ATIVOS_RECORRENTES;

IF 'AREA PRODUTO'n IN ("AML")
	AND AREA_PRODUTO_AJ ^= "CESSÃO"
		AND INDEX(UPCASE(PRODUTO),"CESSAO")=0
			AND INDEX(UPCASE(PRODUTO),"CESSÃO")=0
					THEN AREA_PRODUTO_AJ = "AML";

IF 'AREA PRODUTO'n IN ("AME")
	AND AREA_PRODUTO_AJ ^= "CESSÃO"
		AND INDEX(UPCASE(PRODUTO),"CESSAO")=0
			AND INDEX(UPCASE(PRODUTO),"CESSÃO")=0
					THEN AREA_PRODUTO_AJ = "AME";

RUN;
