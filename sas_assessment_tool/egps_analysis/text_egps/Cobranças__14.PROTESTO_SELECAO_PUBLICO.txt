proc sql;
connect to bigquery as perm (
  project='bv-sadi-dtm'
  credfile='/projetos/acessos/bq/di/sadi-dtm.json'
  schema='bv-pdbd-prd'
  READ_MODE=storage
);

create table BASE_ATRASO as
select * from connection to perm (

  SELECT
    
    t.cd_contrato,
    k.SqContratoFinanceiro,
    k.NUCONTRATO,

    
    k.Dias_atraso,
    k.Cdmotivocontratofinanceiro,
    k.saldo,
    k.cdsituacaocontabil,

    
    c.cd_cpf_cnpj AS NuCpfCnpj,
    SAFE_CAST(c.Cd_Produto AS NUMERIC)           AS CdProduto,
    SAFE_CAST(c.Cd_Modalidade_Produto AS NUMERIC) AS CdModalidadeProduto,
    SAFE_CAST(c.Cd_Pessoa AS NUMERIC)             AS CdPessoa,
    a.CdSubProduto,
    h.CDGRUPOSUBPRODUTO,

    
    MIN(CAST(t.nu_parcela AS NUMERIC)) AS parcela,
    MIN(t.Dt_Vencimento)               AS dt_venc_min,
    COUNT(t.Dt_Vencimento)             AS qtde_parcelas,
    SUM(t.Vr_Parcela)                  AS risco_contrato,

    
    SUM(CASE WHEN s.cod_desp IN (14, 80)
             THEN SAFE_CAST(s.valor_desp AS NUMERIC)
             ELSE 0 END)              AS VRAJUIZ,

    
    CASE WHEN SAFE_CAST(b.NuContratoGestao AS NUMERIC) IS NOT NULL THEN 1 ELSE 0 END AS BNDU,

    
    
    MAX(SAFE_CAST(seg.VRPREMIOSEGURO       AS NUMERIC)) AS VRPREMIOSEGURO,
    MAX(SAFE_CAST(seg.VRCOMISSAOCALCULADA  AS NUMERIC)) AS VRCOMISSAOCALCULADA,
    
    MAX(CAST(seg2.FL_SEGURO_CONTRATADO AS INT64))       AS FLSEGUROCONTRATADO,

    
    CASE
      WHEN MAX(CAST(seg2.FL_SEGURO_CONTRATADO AS INT64)) = 1
        OR COUNT(seg.NUCONTRATO) > 0
      THEN 1 ELSE 0
    END AS FLAG_SEGURO

  FROM bv-pdbd-prd.dvry_core.tbparcela AS t

  
  JOIN (
    SELECT
      CAST(Z.NUCONTRATO AS NUMERIC) AS NUCONTRATO,
      Z.SqContratoFinanceiro,
      MAX(Z.nudiasatraso) AS Dias_atraso,
      Z.Cdmotivocontratofinanceiro,
      CAST(Z.vrtaxacontrato AS NUMERIC) + CAST(Z.vrpermanenciaapropriado AS NUMERIC) AS saldo,
      Z.cdsituacaocontabil
    FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS Z
    WHERE Z.dtprocessamento = &diaontem.
      AND Z.nudiasatraso BETWEEN &atraso1. AND &atraso2.
    GROUP BY
      Z.NUCONTRATO, Z.SqContratoFinanceiro,
      Z.Cdmotivocontratofinanceiro,
      Z.vrtaxacontrato, Z.vrpermanenciaapropriado,
      Z.cdsituacaocontabil
  ) AS k
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = k.NUCONTRATO
   AND t.sq_contrato_financeiro = k.SqContratoFinanceiro

  
  LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS c
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(c.cd_contrato AS NUMERIC)

  LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS a
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(a.nucontrato AS NUMERIC)

  LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_desp_contrato AS s
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(s.NUCONTRATOGESTAO AS NUMERIC)
   AND t.sq_contrato_financeiro = s.SQCONTRATOFINANCEIRO

  LEFT JOIN bv-pdbd-prd.trusted_dbpatio.tbentrada AS b
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(b.NuContratoGestao AS NUMERIC)

  LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS h
    ON a.CdSubProduto = h.CdSubProduto

  
  LEFT JOIN bv-pdbd-prd.trusted_dbcontratoseguro.tbsegurocontratofinanceiro AS seg
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(seg.NUCONTRATO AS NUMERIC)
   AND t.sq_contrato_financeiro = seg.SQCONTRATOFINANCEIRO
   AND seg.CDPRODUTOSEGURO IN (12, 14)

  LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS seg2
    ON SAFE_CAST(t.cd_contrato AS NUMERIC) = SAFE_CAST(seg2.cd_contrato AS NUMERIC)
   AND t.sq_contrato_financeiro = seg2.SQ_CONTRATO_FINANCEIRO
   AND seg2.FL_SEGURO_CONTRATADO = TRUE

  WHERE t.st_parcela = 'A'
    AND SAFE_CAST(c.Cd_Produto AS NUMERIC) IN (12, 14)

  GROUP BY
    t.cd_contrato,
    k.SqContratoFinanceiro,
    k.NUCONTRATO,
    k.Dias_atraso,
    k.Cdmotivocontratofinanceiro,
    k.saldo,
    c.cd_cpf_cnpj,
    c.Cd_Produto,
    c.Cd_Modalidade_Produto,
    c.Cd_Pessoa,
    a.CdSubProduto,
    b.NuContratoGestao,
    h.CDGRUPOSUBPRODUTO,
    k.cdsituacaocontabil
);

disconnect from perm;
quit;
