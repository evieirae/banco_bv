PROC SQL;
CREATE TABLE BASE_COMISSAO AS
SELECT

A.NuAnoMes,
NmGrupoCargo as NmGrupoCargoColaborador,
A.DsLoginN1,

SUM(CASE WHEN A.TpValor = 'Seguros' THEN A.PercComissaoFinal ELSE 0 END) AS VrComissaoSeguros FORMAT COMMAX20.2,
SUM(CASE WHEN A.TpValor = 'NaPista' THEN A.PercComissaoFinal ELSE 0 END) AS VrComissaoNaPista FORMAT COMMAX20.2,
SUM(CASE WHEN A.TpValor = 'Leves BVO' THEN A.PercComissaoFinal ELSE 0 END) AS VrComissaoLevesBVO FORMAT COMMAX20.2,
SUM(A.PercComissaoFinal) AS VrComissaoProduto FORMAT COMMAX20.2

FROM basegrc.TbGrComissaoProduto A

group by 1,2,3

;RUN;

data work.BASE_COMISSAO_PRODUTO;
	set work.BASE_COMISSAO;

	array num_array _numeric_;
	do over num_array;
		if missing(num_array) then num_array = 0;
	end;
run;
