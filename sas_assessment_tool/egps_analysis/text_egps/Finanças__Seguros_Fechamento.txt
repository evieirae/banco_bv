proc sql;
	Create table Work.Base_Fechamento_Rateio  as

		case when a.ProdutoRateio = 'EP' then coalesce(m.Trabalhistas * a.Perc_LACanc,0) when a.ProdutoRateio = 'Privado' then coalesce(m.Trabalhistas * a.Perc_LACanc2,0) else coalesce(m.Trabalhistas * a.Perc_PB,0) end as Trabalhistas,

		0  as RepasseCartoes,
		(coalesce(a.ReceitaBruta,0) + coalesce(a.CxCompetencia,0) + coalesce(a.LA,0) + coalesce(a.Cancelado,0)) * -0.1125 as Tributos,

data work.premios_unpivot;
set work.premios;
length SeguroVCS $20 VrPremioSeguro 8;

array seguros[6] Prestamista AP Equipamento Auto_Completo Auto_Parcial Imobiliario;
array nomes[6] $20 _temporary_('Prestamista','AP','Equipamento','Auto Completo','Auto Parcial','Imobiliario');

do i=1 to dim(seguros);
    if seguros[i] > 0 then do;
        SeguroVCS=nomes[i];
        VrPremioSeguro=seguros[i];
        output;
    end;
end;

keep DtContrato NuContrato Produto Canal SeguroVCS VrPremioSeguro;
run;

data work.premios_iof;
set work.premios_unpivot;
length IOF 8;

select(SeguroVCS);
    when('Prestamista') IOF = VrPremioSeguro* -0.0038;
    when('AP') IOF = VrPremioSeguro* -0.0038;
    when('Equipamento') IOF = VrPremioSeguro* -0.0738;
    when('Auto Completo') IOF = VrPremioSeguro* -0.0738;
    when('Auto Parcial') IOF = VrPremioSeguro* -0.0738;
    otherwise IOF = 0;
end;
PremioLiq = VrPremioSeguro + IOF;
run;

data work.tb_ReceitaBruta;
set work.premios_iof;
length ReceitaLiq ComissaoTotal 8;

select(SeguroVCS);
    when('Prestamista') do;
        ComissaoTotal = 0.723;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    when('AP') do;
        ComissaoTotal = 0.55;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    when('Equipamento') do;
        ComissaoTotal = 0.312;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    when('Auto Completo') do;
        ComissaoTotal = 0.26;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    when('Auto Parcial') do;
        ComissaoTotal = 0.58;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    when('Imobiliario') do;
        ComissaoTotal = 0.20;
        ReceitaLiq = PremioLiq * ComissaoTotal;
    end;
    otherwise do;
        ComissaoTotal = .;
        ReceitaLiq = .;
    end;
end;
run;
