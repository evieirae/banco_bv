proc sql;
  create table BASE_ICM as
  select
    NuAnoMes,
    NmAreaEstruturaComercial as Filial,
    DsLoginN1 as GR,
    sum(VrComissaoTotal) as VrComissaoTotal format=numx12.2,
    sum(ICM)              as ICM              format=numx12.4
  from ICOM.TBGRCOMISSAOICM
  group by 1,2,3
  ;
quit;

/* Data do 1º dia do mês de referência */
data BASE_ICM_DT;
  set BASE_ICM;
  length _yyyymm $6;
  _yyyymm = put(NuAnoMes, z6.);              /* ex: 202501 */
  DtMes   = input(cats(_yyyymm,'01'), yymmdd8.);
  format DtMes yymmn6.;
  drop _yyyymm;
run;

/* (Opcional) Índices */
proc datasets library=work nolist;
  modify BASE_ICM_DT;
  index create GR;
  index create Filial;
  index create NuAnoMes;
  index create DtMes;
quit;

/* ROSTER (mês, filial, GR) */
proc sql;
  create table ROSTER as
  select distinct
         NuAnoMes,
         DtMes,
         Filial,
         GR
  from BASE_ICM_DT
  ;
quit;

/* ===================================================================== */
/* ========================= MÉTRICAS DE 3 MESES ======================== */
/* ===================================================================== */

proc sql;
  create table MEDIA3_GR as
  select
    r.NuAnoMes,
    r.GR,
    avg(case when h.VrComissaoTotal > 0 then h.VrComissaoTotal end)  as Media3_GR format=numx12.2,
    count(case when h.VrComissaoTotal > 0 then 1 end)                as QtdeMesesTrab3_GR
  from ROSTER r
  left join BASE_ICM_DT h
    on  h.GR   = r.GR
    and h.DtMes between intnx('month', r.DtMes, -3, 'b')
                   and     intnx('month', r.DtMes, -1, 'e')
  group by r.NuAnoMes, r.GR
  ;
quit;

proc sql;
  create table MEDIA3_ROSTER as
  select
    r.NuAnoMes,
    r.Filial,
    r.GR,
    m.Media3_GR
  from ROSTER r
  left join MEDIA3_GR m
    on  m.NuAnoMes = r.NuAnoMes
    and m.GR       = r.GR
  ;
quit;

proc sort data=MEDIA3_ROSTER out=MEDIA3_SORT;
  by NuAnoMes Filial descending Media3_GR;
run;

data TOP3_3M_FILIAL (keep=NuAnoMes Filial Media3_Top3Filial QtdeGRs_Considerados_3m);
  set MEDIA3_SORT;
  by NuAnoMes Filial;
  retain soma cnt;
  if first.Filial then do; soma=0; cnt=0; end;

  if not missing(Media3_GR) and cnt < 3 then do;
     soma + Media3_GR;
     cnt  + 1;
  end;

  if last.Filial then do;
     Media3_Top3Filial       = ifn(cnt>0, soma/cnt, .);
     QtdeGRs_Considerados_3m = cnt;
     output;
  end;
run;

  from ROSTER r
  left join BASE_ICM_DT h

    and h.DtMes between intnx('month', r.DtMes, -12, 'b')
                   and     intnx('month', r.DtMes,  -1, 'e')
  group by r.NuAnoMes, r.GR
  ;
quit;

/* ===================== ICM - Média móvel de 12 meses ================== */

proc sql;
  create table MEDIA12_ICM_GR as
  select
    r.NuAnoMes,
    r.GR,
    avg(case when h.ICM > 0 then h.ICM end)  as Media12_ICM format=numx12.4,
    count(case when h.ICM > 0 then 1 end)    as QtdeMesesTrab_ICM_12m
  from ROSTER r
  left join BASE_ICM_DT h
    on  h.GR   = r.GR
    and h.DtMes between intnx('month', r.DtMes, -12, 'b')
                   and     intnx('month', r.DtMes,  -1, 'e')
  group by r.NuAnoMes, r.GR
  ;
quit;

proc sql;
  create table tbremuneracaogr as
  select
    b.NuAnoMes,
    b.Filial,
    b.GR,
    b.VrComissaoTotal          as VrComissao_Mes  format=numx12.2,
    m12icm.Media12_ICM         as ICM_Media12     format=numx12.4,
    m12.Media12_GR                              format=numx12.2,
    m3.Media3_GR                                format=numx12.2,
    t12.Media12_Top3Filial                      format=numx12.2,
    t3.Media3_Top3Filial                        format=numx12.2
  from BASE_ICM_DT           b
  left join MEDIA12_GR       m12    on m12.NuAnoMes=b.NuAnoMes  and m12.GR=b.GR
  left join MEDIA3_GR        m3     on m3.NuAnoMes=b.NuAnoMes   and m3.GR=b.GR
  left join TOP3_FILIAL      t12    on t12.NuAnoMes=b.NuAnoMes  and t12.Filial=b.Filial
  left join TOP3_3M_FILIAL   t3     on t3.NuAnoMes=b.NuAnoMes   and t3.Filial=b.Filial
  left join MEDIA12_ICM_GR   m12icm on m12icm.NuAnoMes=b.NuAnoMes and m12icm.GR=b.GR
  ;
quit;

PROC SQL;
   CREATE TABLE TBVISITAFINAL_ AS
   SELECT t1.DtRef,
   		  t1.DtAtualizacao,
          t1.NmSubEstruturaComercialC,
          t1.NmAreaEstruturaComercialC,
          t1.Colaborador,
		  t1.CdNmParceiro,
          t1.nmparceirocomercial,
          t1.dsloginn1,
          t1.dsloginn2,
          t1.dsloginn3,
		  t1.DsLoginN1C,
		  t1.DsLoginN2C,
		  t1.DsLoginN3C,
		  t1.FiltroRelatorioGRFoco,
		  t1.FiltroRelatorioPDVFoco,
          /* PDV_Foco */
            (SUM(t1.PDV_Foco)) AS PDV_Foco,
          /* GR_Foco */
            (SUM(t1.GR_Foco)) AS GR_Foco,
          /* Media Progresso */
            (AVG(t1.progresso)) AS 'Progresso'n,
          /* Compromisso */
            (SUM(t1.Compromisso)) AS Compromisso,
          /* Feedback2 */
            (SUM(t1.feedback2)) AS Feedback2,
          /* Feedback3 */
            (SUM(t1.feedback3)) AS Feedback3,
          /* Feedback4 */
            (SUM(t1.feedback4)) AS Feedback4,
          /* Campanha */
            (SUM(t1.Campanha)) AS Campanha,
          /* FlQualidade */
            (SUM(t1.FlQualid)) AS FlQualidade,
          /* POntos_Gr */
            (MAX(t1.POntos_Gr)) AS POntos_Gr FORMAT=COMMAX10.2,
          /* Pontos_Filial */
            (MAX(t1.Pontos_Filial)) AS Pontos_Filial FORMAT=COMMAX10.2,
          /* Pontos_Regional */
            (MAX(t1.Pontos_Regional)) AS Pontos_Regional FORMAT=COMMAX10.2,
          /* Part_SemCard */
            (SUM(t1.Part_SemCard)) AS Part_SemCard,
          /* Part_APNP */
            (SUM(t1.Part_APNP)) AS Part_APNP,
          /* Part_Cancelada */
            (SUM(t1.Part_Cancelada)) AS Part_Cancelada,
          /* Part_Card */
            (SUM(t1.Part_Card)) AS Part_Card,
          /* Part_ContratosBV */
            (SUM(t1.Part_ContratosBV)) AS Part_ContratosBV,
          /* Part_Negada */
            (SUM(t1.Part_Negada)) AS Part_Negada,
          /* Part_Mercado */
            (SUM(t1.Part_Mercado)) AS Part_Mercado,
          /* Perf_PartJogo3M */
            (SUM(t1.Perf_PartJogo3M)) AS Perf_PartJogo3M FORMAT=COMMAX10.2,
          /* Part_CardRec */
            (SUM(t1.Part_CardRec)) AS Part_CardRec,
          /* QtdProducaoRG */
            (SUM(t1.QtdProducaoRG)) AS QtdProducaoRG,
          /* VrProducao */
            (SUM(t1.VrProducao)) AS VrProducao FORMAT=COMMAX10.2,
          /* VrProducaoProjetado */
            (SUM(t1.VrProducaoProjetado)) AS VrProducaoProjetado FORMAT=COMMAX10.2,
			(sum(MetaVrProducao)) as MetaVrProducao FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(VrComissao) else 0 end) as VrComissao FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(Media12_GR) else 0 end) as Media12_GR FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(Media12_Top3Filial) else 0 end) as Media12_Top3Filial FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(Media3_GR) else 0 end) as Media3_GR FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(Media3_Top3Filial) else 0 end) as Media3_Top3Filial FORMAT=COMMAX10.2,
			(case when GR_UNIQ = 1 then sum(ICM_Media12) else 0 end) as ICM_Media12 FORMAT=COMMAX10.4

      FROM WORK.TBVISITA_ t1
      GROUP BY t1.DtRef,
	           t1.DtAtualizacao,
               t1.NmSubEstruturaComercialC,
               t1.NmAreaEstruturaComercialC,
               t1.Colaborador,

               t1.nmparceirocomercial,
               t1.dsloginn1,
               t1.dsloginn2,
               t1.dsloginn3,
			   t1.DsLoginN1C,
			   t1.DsLoginN2C,
			   t1.DsLoginN3C,
			   t1.FiltroRelatorioGRFoco,
			   t1.FiltroRelatorioPDVFoco;
