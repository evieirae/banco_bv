PROC SQL;
%CONECTADB(server=CTR);

  CREATE TABLE tempw.Privado_Leadtime AS
	SELECT *
	FROM CONNECTION TO DBCON
				(

	SELECT
			'TbLogStatusProposta' AS tabela,
			stproposta.nuproposta,
			cast(stproposta.dtstatusproposta as date) as dtstatusproposta_ddmmyyy,
			stproposta.dtstatusproposta,
			stproposta.stproposta
		FROM DBCRED..TbLogStatusProposta stproposta

			WHERE
				year(stproposta.dtstatusproposta ) >= YEAR(getdate())-1
				AND stproposta.stproposta IN ('D','AV','AP')

				UNION

				SELECT
				'TbSituacaoAverbacaoProposta' AS tabela,
				tbstproposta.nuproposta,
				cast(tbstproposta.dtsituacaoaverbacao as date) as dtstatusproposta_ddmmyyy,
				tbstproposta.dtsituacaoaverbacao AS dtstatusproposta,
				'AV' AS stproposta
			FROM DBCPC..TbSituacaoAverbacaoProposta tbstproposta

				WHERE tbstproposta.StAverbacao = 'A' and
					year(tbstproposta.dtsituacaoaverbacao) >= YEAR(getdate())-1

					UNION

				SELECT
					'TbControleStatusAverbacaoPrvdo' AS tabela,
					tbstproposta.nuproposta,
					cast(tbstproposta.dtevento as date) as dtstatusproposta_ddmmyyy,
					tbstproposta.dtevento AS dtstatusproposta,
					'AV' AS stproposta
				FROM DBCPC..TbControleStatusAverbacaoPrvdo tbstproposta

					WHERE tbstproposta.StAverbacaoPrivado  = 'A' and
					YEAR(tbstproposta.dtevento) >= YEAR(getdate())-1

);
RUN;
