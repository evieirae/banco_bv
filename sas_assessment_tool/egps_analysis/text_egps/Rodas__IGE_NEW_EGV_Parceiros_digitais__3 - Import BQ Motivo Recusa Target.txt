proc sql print;
	select (year(max(dt_entrada_proposta))*100)+month(max(dt_entrada_proposta))-1 as MXDATAPROP INTO :MXDATAPROP from tribo.base_MotivoRecusaTargetEgv;
QUIT;

proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table base_MotivoRecusaTargetEgv_0 as
   select * from connection to perm
     (

SELECT
  *
  ,  CASE
		WHEN ds_motivo_recusa_minimo LIKE ('%LIMITE MÝNIMO%') THEN 1
		WHEN ds_motivo_recusa_minimo LIKE ('%PRAZO FORA DA POLÝTICA%') THEN 1
		WHEN ds_motivo_recusa_minimo LIKE ('%Recusa LTV%') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('%COMPROMETIMENTO%') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('%Valor financiado fora da polít%') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('TABELA: PRAZO NÒO PERMITIDO') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('CADASTRO: DATA NASCIMENTO DIVERGENTE') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('RECUSA NOME OU DATA DE NASCIMENTO DIVERGENTE') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('FORA NORMA : VALOR PROPOSTA') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('PRAZO FORA DA POLÝTICA') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('VALOR FINANCIADO FORA DA POLÝTICA') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('LIMITE MÝNIMO DE FINANCIAMENTO') THEN 1
		WHEN UPPER(ds_motivo_recusa_minimo) LIKE ('TABELA: % COMPR. RENDA NÒO PERMITIDO') THEN 1
		          ELSE 0 END AS Motivo_selecao
 FROM bv-pdbd-prd.dvry_corp_credito.motivorecusa_proposta_fato

  	) A
WHERE 1=1
AND a.ds_produto in ('egv_leves','egv_pesados')
AND (year(a.dt_entrada_proposta)*100)+month(a.dt_entrada_proposta) >= &MXDATAPROP.;
   disconnect from perm;
quit;

PROC SQL;
   CREATE TABLE WORK.base_MotivoRecusaTargetEgv_1 AS
   SELECT t1.nu_proposta,
          t1.dt_entrada_proposta,
          t1.ds_produto,
          t1.ds_subproduto,
          t1.st_proposta,
			CASE
				WHEN UPPER(t1.ds_motivo_recusa_minimo)	= 'RESTRIÞÒO POLITICA: COMPROMETIMENTO RENDA'
					THEN 'RESTRIÇÃO POLITICA: COMPROMETIMENTO RENDA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo)	= 'PRAZO FORA DA POLÝTICA'
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo)	= 'LIMITE MÝNIMO DE FINANCIAMENTO'
					THEN 'LIMITE MÍNIMO DE FINANCIAMENTO'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('%Valor financiado fora da polít%')
					THEN 'VALOR FINANCIADO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('TABELA: PRAZO NÒO PERMITIDO')
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('PRAZO FORA DA POLÝTICA')
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('VALOR FINANCIADO FORA DA POLÝTICA')
					THEN 'VALOR FINANCIADO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('LIMITE MÝNIMO DE FINANCIAMENTO')
					THEN 'LIMITE MÍNIMO DESC FINANCIAMENTO'
				WHEN UPPER(t1.ds_motivo_recusa_minimo) LIKE ('TABELA: % COMPR. RENDA NÒO PERMITIDO')
					THEN 'COMPROMETIMENTO DE RENDA'
		  				ELSE UPPER(t1.ds_motivo_recusa_minimo)
		  	END AS ds_motivo_recusa_minimo,
			CASE
				WHEN t1.ds_motivo_recusa_maximo = 'RESTRIÞÒO POLITICA: COMPROMETIMENTO RENDA'
					THEN 'RESTRIÇÃO POLITICA: COMPROMETIMENTO RENDA'
				WHEN t1.ds_motivo_recusa_maximo =  'PRAZO FORA DA POLÝTICA'
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN t1.ds_motivo_recusa_maximo =  'TABELA: ANO DO BEM NÒO PERMITIDO'
					THEN 'TABELA: ANO DO BEM NÃO  PERMITIDO'
				WHEN t1.ds_motivo_recusa_maximo =  'LIMITE MÝNIMO DE FINANCIAMENTO'
					THEN 'LIMITE MÍNIMO DE FINANCIAMENTO'
					WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('%Valor financiado fora da polít%')
					THEN 'VALOR FINANCIADO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('TABELA: PRAZO NÒO PERMITIDO')
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('PRAZO FORA DA POLÝTICA')
					THEN 'PRAZO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('VALOR FINANCIADO FORA DA POLÝTICA')
					THEN 'VALOR FINANCIADO FORA DA POLÍTICA'
				WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('LIMITE MÝNIMO DE FINANCIAMENTO')
					THEN 'LIMITE MÍNIMO DESC FINANCIAMENTO'
				WHEN UPPER(t1.ds_motivo_recusa_maximo) LIKE ('TABELA: % COMPR. RENDA NÒO PERMITIDO')
					THEN 'COMPROMETIMENTO DE RENDA'

						ELSE t1.ds_motivo_recusa_maximo
				END AS  ds_motivo_recusa_maximo,
			t1.fl_motivo_score_psicometrico,
			t1.ds_motivo_recusa_1,
			t1.ds_motivo_recusa_2,
			t1.ds_motivo_recusa_3,
			t1.ds_motivo_recusa_4,
			t1.ds_motivo_recusa_5,
			t1.ds_motivo_recusa_6,
			t1.ds_motivo_recusa_7,
			t1.ds_motivo_recusa_8,
			t1.ds_motivo_recusa_9,
			t1.qt_motivo_recusa,
			t1.ts_inclusao_analitico,
			t1.Motivo_selecao
      FROM WORK.base_MotivoRecusaTargetEgv_0 t1
      WHERE t1.Motivo_selecao = 1;
QUIT;

PROC SQL;
   CREATE TABLE base_MotivoRecusaTargetEgv AS
   SELECT t1.nu_proposta,
          t1.dt_entrada_proposta,
          t1.ds_produto,
          t1.ds_subproduto,
          t1.st_proposta,
          COALESCE(t1.ds_motivo_recusa_minimo,t1.ds_motivo_recusa_maximo,'N/D') AS ds_motivo_recusa,
          t1.fl_motivo_score_psicometrico,
          t1.ds_motivo_recusa_1,
          t1.ds_motivo_recusa_2,
          t1.ds_motivo_recusa_3,
          t1.ds_motivo_recusa_4,
          t1.ds_motivo_recusa_5,
          t1.ds_motivo_recusa_6,
          t1.ds_motivo_recusa_7,
          t1.ds_motivo_recusa_8,
          t1.ds_motivo_recusa_9,
          t1.qt_motivo_recusa,
          t1.ts_inclusao_analitico,
          t1.Motivo_selecao
      FROM WORK.base_MotivoRecusaTargetEgv_1 t1;
QUIT;

PROC SQL;
	DELETE * FROM TRIBO.base_MotivoRecusaTargetEgv WHERE (year(dt_entrada_proposta)*100)+month(dt_entrada_proposta) >= &MXDATAPROP.;
QUIT;

data TRIBO.base_MotivoRecusaTargetEgv (compress=yes);
	set
		TRIBO.base_MotivoRecusaTargetEgv
		base_MotivoRecusaTargetEgv;
run;
