	CREATE TABLE WORK.BASE_CYBER_VEIC_MKTPLACES_TRAT AS

		SELECT DISTINCT
			  *

				WHEN VEIC_PLACA IS MISSING THEN ''

PROC SQL;
	CREATE TABLE WORK.AUX_BASE_CYBER_1 AS

		SELECT DISTINCT
			  A.*
			, C.FLAG_CONTROLE_100D
			, CASE
				WHEN A.Flag_Acess_Site = 1 THEN 1
				WHEN A.Flag_Acess_App = 1 THEN 1
				ELSE 0
			  END AS Flag_Acess_Digital
			, INPUT(SUBSTR(A.NUCPFCNPJ,6,1),Z1.) AS DIG_CPF
			, INPUT(SUBSTR(PUT(A.NUCONTRATOGESTAO,14.),12,1),Z1.) AS DIG_CONTR

;QUIT; 



options reuse=yes compress=binary;

PROC SQL;
	CREATE TABLE WORK.AUX_BASE_CYBER_2 AS

		SELECT
		  *
		, CASE
			WHEN RANDON BETWEEN  0 AND 19 THEN 'QUERO_QUITAR'
			WHEN RANDON BETWEEN 20 AND 39 THEN 'ACORDO_CERTO'
			WHEN RANDON BETWEEN 40 AND 59 THEN 'BLU365'
			WHEN RANDON BETWEEN 60 AND 79 THEN 'SERASA'
			WHEN RANDON BETWEEN 80 AND 99 THEN 'CONTROLE'
			ELSE 'DEMAIS'
		  END AS PUBLICO_OLD
		, CASE
		
			WHEN RANDON BETWEEN 90 AND 99 THEN 'CONTROLE'
			WHEN ATRASO BETWEEN  5 AND  15   THEN 'ATRASO5A15'
			WHEN ATRASO < 5   THEN 'ATRASOMENOR5'
			WHEN FLAG_CONTROLE_100D = 'S' THEN 'CONTR100D'

		
			WHEN ATRASO BETWEEN  16   AND   70 AND RANDON BETWEEN 00 AND 29 THEN 'QQ_16_70'
			WHEN ATRASO BETWEEN  71   AND  180 AND RANDON BETWEEN 00 AND 29 THEN 'QQ_71_180'
			WHEN ATRASO BETWEEN  181  AND  360 AND RANDON BETWEEN 00 AND 29 THEN 'QQ_181_360'
			WHEN ATRASO 				 > 360 AND RANDON BETWEEN 00 AND 29 THEN 'QQ_361_9999'

		
			WHEN ATRASO BETWEEN  16   AND   70 AND RANDON BETWEEN 30 AND 89 THEN 'SERASA_16_70'

			WHEN ATRASO BETWEEN  181  AND  360 AND RANDON BETWEEN 30 AND 89 THEN 'SERASA_181_360'
