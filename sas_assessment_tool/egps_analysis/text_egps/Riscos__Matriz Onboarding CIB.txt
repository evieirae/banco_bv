PROC SQL;
   CREATE TABLE BASE_ONBOARDING_INI_FLAG AS
   SELECT DISTINCT
        T1.*,
		(CASE WHEN INPUT(T1.NuConsultado,BEST14.)IN (0,1,.,12,26) THEN 0
              WHEN T2.DOCUMENTO IS NOT MISSING THEN 0
			  WHEN INPUT(T1.NuConsultado,BEST14.) IS MISSING THEN 0
			  WHEN T3.DOCUMENTO IS NOT MISSING  THEN 0
			  WHEN T1.DOC_NUM_QUADRO IN (.,0) THEN 0
			  ELSE 1 END) AS FLAG

   FROM BASE_ONBOARDING_INI t1
   LEFT JOIN BS_PROCESSOS_CIB_PJ T2 ON (INPUT(T1.NuConsultado,BEST14.) = T2.DOCUMENTO and t2.Nome NOT = '')
    LEFT JOIN BS_PROCESSOS_CIB T3 ON (T1.DOC_NUM_QUADRO = T3.DOCUMENTO)

;QUIT;

PROC SQL;
   CREATE TABLE BS_APTOS AS
   SELECT DISTINCT
          t1.NOME,
          t1.NuConsultado,
          
            (SUM(t1.FLAG)) AS SUM_of_FLAG
      FROM WORK.BASE_ONBOARDING_INI_FLAG t1
      GROUP BY t1.NOME,
               t1.NuConsultado;
QUIT;

PROC SQL;
   CREATE TABLE BASE_ONBOARDING_INI_FLAG AS
   SELECT DISTINCT
        T1.*

   FROM BASE_ONBOARDING_INI_FLAG t1
   LEFT JOIN BS_APTOS T2 ON (T1.NOME = T2.NOME AND t1.NuConsultado = T2.NuConsultado)
   WHERE t2.SUM_of_FLAG = 0

;QUIT;

PROC SQL;
   CREATE TABLE BASE_ONBOARDING_INI AS
   SELECT DISTINCT
        T1.*,
		COALESCE(T3.ENDERECO,T4.ENDERECO) FORMAT $200. as ENDERECO,
		COALESCE(T3.BAIRRO,T4.BAIRRO) FORMAT $100. as BAIRRO,
		COALESCE(T3.CIDADE,T4.CIDADE) FORMAT $50. as CIDADE,
		COALESCE(T3.UF,T4.UF) FORMAT $10. as UF,
		COALESCE(T3.CEP,T4.CEP) as CEP,
		input(t2.RENDA, 8.) AS RENDA,
		(CASE WHEN COALESCE(T5.ds_situacao_cadastral,T6.ds_situacao_cadastral) IS MISSING
                   AND T1.DOC_NUM_QUADRO = 0 THEN 'ESTRANGEIRO'
              WHEN COALESCE(T5.ds_situacao_cadastral,T6.ds_situacao_cadastral) IS MISSING THEN 'NAO ENCONTRADO'
              ELSE COALESCE(T5.ds_situacao_cadastral,T6.ds_situacao_cadastral) END) FORMAT $100. as situacao_cadastral

   FROM BASE_ONBOARDING_INI t1
   LEFT JOIN BS_RENDA_PF T2 ON (T1.DOC_NUM_QUADRO = INPUT(T2.CPF,BEST14.))
   LEFT JOIN BS_ENDERECO_PF T3 ON (T1.DOC_NUM_QUADRO = INPUT(T3.CD_CPF,BEST14.))
   LEFT JOIN BS_ENDERECO_PJ T4 ON (T1.DOC_NUM_QUADRO = INPUT(T4.cnpj,BEST14.))
   LEFT JOIN BS_SIT_CADASTRAL_PJ T5 ON (T1.DOC_NUM_QUADRO = INPUT(T5.cd_cpf_cnpj,BEST14.))

PROC SQL;
   CREATE TABLE LISTA_BLOQUEADOS AS
   		SELECT DISTINCT
		  	t1.DOC_NUM,
		   	t1.DATA_ATUAL,
		  	SUM((CASE WHEN T1.LISTA = 'CEIS' THEN 1 ELSE 0 END)) AS CEIS,
			SUM((CASE WHEN T1.LISTA = 'TRAB_ESCRAVO' THEN 1 ELSE 0 END)) AS TRAB_ESCRAVO,
			SUM((CASE WHEN T1.LISTA IN ('Proprietário preso por envol','Mídia Negativa') THEN 1 ELSE 0 END)) AS MIDIA,
			SUM((CASE WHEN T1.LISTA = 'CNEP' THEN 1 ELSE 0 END)) AS CNEP,
			SUM((CASE WHEN T1.LISTA = 'LENIENCIA' THEN 1 ELSE 0 END)) AS LENIENCIA,
			SUM((CASE WHEN T1.LISTA = 'CEPIM' THEN 1 ELSE 0 END)) AS CEPIM,
			SUM((CASE WHEN T1.LISTA IN ('RECUSADO','Recusados','Recusados Matriz','RECUSADO BD',
                                        'RECUSADO MATRIZ','RECUSADO MELIUZ','RECUSADO ONEPLD','RECUSADO SALES',
                                        'Renovação Contínua','RENOVACAO') THEN 1 ELSE 0 END)) AS RECUSADOS,
			SUM((CASE WHEN T1.LISTA IN ('Monitoramento de PLD','CONTA PONTE','PLD2','MONITORAMENTO TRANSACIONAL',
                                        'MONITORAMENTO_MELIUZ','Monitoramento de PLD / BAAS','Device') THEN 1 ELSE 0 END)) AS MONITORAMENTO_PLD,
			SUM((CASE WHEN T1.LISTA IN ('DENUNCIA','Denúncia') THEN 1 ELSE 0 END)) AS DENUNCIA,
			SUM((CASE WHEN T1.LISTA IN ('Reportados Coaf','Sancionados') THEN 1 ELSE 0 END)) AS OUTROS,
			SUM((CASE WHEN T1.LISTA = 'FRAUDES' THEN 1 ELSE 0 END)) AS LISTA_FRAUDES,
			SUM((CASE WHEN T1.LISTA = 'Favorecido Bloqueado' THEN 1 ELSE 0 END)) AS Fav_Bloqueado,
			SUM((CASE WHEN T1.LISTA IS MISSING THEN 1
                      WHEN T1.LISTA IN ('BLOQUEADO ONEPLD','BLOQUEADOS ONEPLD') THEN 1 ELSE 0 END)) AS BLOQUEADOS,
			SUM((CASE WHEN T1.LISTA = 'EX-COLABORADOR' THEN 1 ELSE 0 END)) AS EX_COLABORADOR,
			1 AS TOTAL

   FROM LISTA_BLOQUEADOS t1
  		WHERE T1.DOC_NUM NOT=. AND T1.DOC_NUM >0 AND t1.TPLISTA NOT = 'RETIRAR'
		GROUP BY T1.DOC_NUM
   ;
QUIT;

PROC SQL;
   CREATE TABLE BASE_ONBOARDING_INI AS
   		SELECT DISTINCT
		  	t1.*,
		  	COALESCE(T2.CEIS,0) AS LISTA_CEIS,
			COALESCE(T2.TRAB_ESCRAVO,0) AS LISTA_TRAB_ESCRAVO,
			COALESCE(T2.MIDIA,0) AS LISTA_MIDIA,
			COALESCE(T2.CNEP,0) AS LISTA_CNEP,
			COALESCE(T2.LENIENCIA,0) AS LISTA_LENIENCIA,
			COALESCE(T2.CEPIM,0) AS LISTA_CEPIM,
			(CASE WHEN COALESCE(T2.RECUSADOS,0) > 0 THEN 1 ELSE 0 END) AS LISTA_RECUSA,
			COALESCE(T2.MONITORAMENTO_PLD,0) AS LISTA_MONITORA_PLD,
			COALESCE(T2.DENUNCIA,0) AS LISTA_DENUNCIA,
			COALESCE(T2.OUTROS,0) AS LISTA_OUTROS,
			COALESCE(T2.LISTA_FRAUDES,0) AS LISTA_FRAUDES,
			COALESCE(T2.Fav_Bloqueado,0) AS LISTA_FAV_BLOQUEADO,
			COALESCE(T2.BLOQUEADOS,0) AS LISTA_BLOQUEADOS,
			COALESCE(T2.EX_COLABORADOR,0) AS LISTA_EX_COLABORADOR,
			(CASE WHEN SUM((CASE WHEN T3.LISTA = 'PEP_INTERNACIONAL' THEN 1 ELSE 0 END))>0 THEN 1 ELSE 0 END) AS LISTA_PEP_INTERNACIONAL,
			(CASE WHEN SUM((CASE WHEN T4.LISTA = 'SANCIONADO' THEN 1 ELSE 0 END))>0 THEN 1 ELSE 0 END) AS LISTA_SANCIONADO_INTER,
			t4.Motivo_Sancoes

   FROM BASE_ONBOARDING_INI t1
   LEFT JOIN LISTA_BLOQUEADOS T2 ON (T1.DOC_NUM_QUADRO = T2.DOC_NUM)
   LEFT JOIN PEP_INTER T3 ON (T1.NOME_QUADRO = T3.NOME)
   LEFT JOIN SANCIONADOS_BS T4 ON (T1.NOME_QUADRO = T4.NOME)
   GROUP BY T1.NOME;

QUIT;

PROC SQL;
	CREATE TABLE BASE_ONBOARDING_INI AS
		SELECT DISTINCT
			A.*,
		    (CASE WHEN A.DOC_NUM_QUADRO IS MISSING THEN 'ESTRANGEIRO'
			      WHEN A.DOC_NUM_QUADRO = 0 THEN 'ESTRANGEIRO'
                  WHEN B.DOCUMENTO IS NOT MISSING THEN 'REPROVADO'
				  ELSE 'APROVADO' END)
                  FORMAT $100. AS STATUS_PROCESSO,
          B.Motivo_da_Reprovacao

FROM BASE_ONBOARDING_INI A
LEFT JOIN BS_PROCESSOS B ON (A.DOC_NUM_QUADRO = B.DOCUMENTO)
LEFT JOIN BS_PROCESSOS_SUM_CIB C ON (A.DOC_NUM_QUADRO = C.DOCUMENTO);

QUIT;

PROC FREQ DATA=BASE_ONBOARDING_INI ; TABLES STATUS_PROCESSO; RUN ;



PROC SQL;
	CREATE TABLE BASE_ONBOARDING_FILTRO AS
		SELECT DISTINCT
		        CODIGO,
                NuConsultado,
				NOME,
				(CASE WHEN SUM(SCORE_PEP) > 0 THEN 1 ELSE 0 END) AS PEP,
                (CASE WHEN (SUM(SCORE_LISTA) - SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM)) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS INTERNAS'N,
				(CASE WHEN SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS EXTERNAS'N,
				(CASE WHEN SUM(SCORE_SANCIONADOS) > 0 THEN 1 ELSE 0 END)  AS 'SANÇÕES'N,
                (CASE WHEN SUM(MIDIA) > 0 THEN 1 ELSE 0 END) AS MIDIA,
				(CASE WHEN SUM(SCORE_PRODUTO) > 0 THEN 1 ELSE 0 END) AS 'Opera câmbio?'n,
                
				(CASE WHEN SUM(CASE WHEN STATUS_PROCESSO IN ('REPROVADO') THEN 1 ELSE 0 END)> 0 THEN 1 ELSE 0 END) AS Processos

	FROM BASE_ONBOARDING_INI
	GROUP BY NuConsultado,NOME
;
QUIT;

DATA BASE_ONBOARDING_FILTRO;
SET BASE_ONBOARDING_FILTRO;
FORMAT 'ANALISTA'N 'VALIDAÇÃO BENEFICIÁRIO FINAL'N STATUS_FINAL VISITA 'DATA DA VISITA'n 'RADAR(S/N)'n 'PARECER DO ANALISTA'n 'CONCLUSÃO FINAL'n $150.;
STATUS_FINAL = 'Sem classificacao';



IF 'SANÇÕES'N = 1 and 'Opera câmbio?'n = 1 AND (PEP + 'LISTAS INTERNAS'N + Processos + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(sanções) + câmbio';
IF 'SANÇÕES'N = 1 and  (PEP + 'LISTAS INTERNAS'N + Processos + MIDIA + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(sanções)';
IF 'SANÇÕES'N = 1 AND 'LISTAS INTERNAS'N = 1 AND Processos = 1 and  (PEP + MIDIA + 'Opera câmbio?'n + 'LISTAS EXTERNAS'n) = 0 THEN STATUS_FINAL = 'Com risco(sanções + lista interna + processo)';


IF Processos = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(processo)';
IF Processos = 1 AND MIDIA = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(processo) + midia';
IF Processos = 1 AND MIDIA = 1 AND PEP = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(processo + pep) + midia';
IF Processos = 1 AND MIDIA = 1 AND 'SANÇÕES'N  = 1 AND (PEP + 'LISTAS INTERNAS'N + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(processo + sanções) + midia';
IF Processos = 1 AND MIDIA = 1 AND 'Opera câmbio?'n = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N) = 0 THEN STATUS_FINAL = 'Com risco(processo) + midia + câmbio';
IF Processos = 1 AND 'Opera câmbio?'n = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(processo) + câmbio';
IF Processos = 1 AND 'LISTAS INTERNAS'N = 1 AND (PEP +'Opera câmbio?'n  + 'SANÇÕES'N + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(processo + lista interna)';
IF Processos = 1 AND 'LISTAS INTERNAS'N = 1 and 'Opera câmbio?'n = 1 AND (PEP  + 'SANÇÕES'N + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(processo + lista interna) + câmbio';
IF Processos = 1 AND 'LISTAS INTERNAS'N = 1 and 'Opera câmbio?'n = 1 and MIDIA = 1 AND (PEP  + 'SANÇÕES'N  ) = 0 THEN STATUS_FINAL = 'Com risco(processo + lista interna) + midia + câmbio';
IF Processos = 1 AND 'LISTAS INTERNAS'N = 1 and 'Opera câmbio?'n = 1 and MIDIA = 1 AND (PEP ) = 1 THEN STATUS_FINAL = 'Com risco(processo + lista interna) + midia + câmbio';


IF 'LISTAS INTERNAS'N  = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA  + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(lista interna)';
IF 'LISTAS INTERNAS'N  = 1 AND 'Opera câmbio?'n = 1 and (PEP + Processos + 'SANÇÕES'N + MIDIA  ) = 0 THEN STATUS_FINAL = 'Com risco(lista interna) + câmbio';
IF 'LISTAS INTERNAS'N  = 1 AND 'Opera câmbio?'n = 1 and MIDIA = 1 AND (PEP + Processos + 'SANÇÕES'N ) = 0 THEN STATUS_FINAL = 'Com risco(lista interna) + câmbio + midia';
IF 'LISTAS INTERNAS'N  = 1 AND PEP= 1 and MIDIA = 1 AND (Processos + 'SANÇÕES'N ) = 0 THEN STATUS_FINAL = 'Com risco(lista interna + pep) + midia';
IF 'LISTAS INTERNAS'N  = 1 and MIDIA = 1 AND (Processos + 'SANÇÕES'N + PEP) = 0 THEN STATUS_FINAL = 'Com risco(lista interna) + midia';

IF 'Opera câmbio?'n = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA  + 'LISTAS INTERNAS'N  ) = 0 THEN STATUS_FINAL = 'Sem risco + câmbio';


IF  MIDIA = 1 AND (PEP + Processos + 'SANÇÕES'N + 'Opera câmbio?'n + 'LISTAS INTERNAS'N  ) = 0 THEN STATUS_FINAL = 'Sem risco + midia';
IF  MIDIA = 1 AND 'Opera câmbio?'n = 1 AND (PEP + Processos + 'SANÇÕES'N + 'LISTAS INTERNAS'N  ) = 0 THEN STATUS_FINAL = 'Sem risco + midia e câmbio';

IF PEP = 1 and (Processos  + 'SANÇÕES'N + 'LISTAS INTERNAS'N +'Opera câmbio?'n + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(PEP)';
IF PEP = 1 and 'Opera câmbio?'n = 1 and MIDIA = 1 AND (Processos  + 'SANÇÕES'N + 'LISTAS INTERNAS'N ) = 0 THEN STATUS_FINAL = 'Com risco(PEP) + midia + câmbio';
IF PEP = 1 and 'Opera câmbio?'n = 1  AND (Processos  + 'SANÇÕES'N + 'LISTAS INTERNAS'N + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(PEP) + câmbio';
IF PEP = 1 and MIDIA = 1 AND (Processos  + 'SANÇÕES'N + 'LISTAS INTERNAS'N +'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(PEP) + midia';
IF PEP = 1 and MIDIA = 1 AND 'Opera câmbio?'n = 1 AND (Processos  + 'SANÇÕES'N + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(PEP) + midia + câmbio';
IF PEP = 1 and 'Opera câmbio?'n = 1 and MIDIA = 1 AND Processos = 1 AND ('SANÇÕES'N + 'LISTAS INTERNAS'N ) = 0 THEN STATUS_FINAL = 'Com risco(PEP + processos) + midia + câmbio';
IF PEP = 1 AND Processos = 1 AND ('SANÇÕES'N + 'LISTAS INTERNAS'N +'Opera câmbio?'n + MIDIA ) = 0 THEN STATUS_FINAL = 'Com risco(PEP + processos)';
IF PEP = 1 and 'Opera câmbio?'n = 1 and 'LISTAS INTERNAS'N = 1 AND ('SANÇÕES'N + MIDIA + Processos) = 0 THEN STATUS_FINAL = 'Com risco(PEP + listas internas) + câmbio';
IF PEP = 1 and 'Opera câmbio?'n = 1 and Processos = 1 AND ('SANÇÕES'N + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(PEP + processos) + câmbio';
IF PEP = 1 and  MIDIA = 1 and Processos = 1 AND 'LISTAS INTERNAS'N = 1 and ('SANÇÕES'N + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Com risco(PEP + processos + listas internas) + midia';


IF (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + Processos  + 'Opera câmbio?'n) = 0 THEN STATUS_FINAL = 'Sem risco';
