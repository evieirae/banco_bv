	A.CdRubricaTransacaoCartao,
	A.DsTransacao,
	A.DtReferencia,
	A.FlParcelado,
	A.VrTransacao,
	A.MaiorParcelaPaga,
	COALESCE(B.VrContratoAdesao,C.VALOR_PRINC) AS VrContrato,
	COALESCE(B.DtAceleracao,C.DtAceleracao) AS DtAceleracao FORMAT DATE9.,
	COALESCE(B.VrParcelaAdesao, C.VALOR_PMT, A.VrTransacao) AS VrPMT,
	COALESCE(B.TxRateAdesao, C.TAXA)/100 AS Taxa
FROM PARCELADOS_ACEL_ACORDO A
LEFT JOIN CARNEIRO.TbContratoParcelamentoFatura B
	ON A.CdDimContaCartao=B.CdContaCartao AND A.QtTotalParcelaCompra=B.NuParcelaAdesao AND A.DtPostagemTransacao=B.DtAdesao AND A.CdRubricaTransacaoCartao IN (6061, 6212, 6078)
LEFT JOIN ACORDOS_CA_UNICO C
	ON A.CdDimContaCartao=(C.PRODUTO*1e9+C.NUCONTA) AND A.DtPostagemTransacao=C.DATA_EFET AND A.QtTotalParcelaCompra=C.QTDE_PARCELAS AND A.CdRubricaTransacaoCartao IN (6053)
;QUIT;

DATA PARCELADOS_ACEL_ACORDO_2;
	SET PARCELADOS_ACEL_ACORDO_TAXAS;
	ParcelasRestantes = QtTotalParcelaCompra-MaiorParcelaPaga;
	ValorRestante = VrPMT*ParcelasRestantes;
	IF DtAceleracao >= DtAcordo AND NOT MISSING(Taxa) THEN VrAceleracoes=(VrPMT*(1-(1+Taxa)**(MaiorParcelaPaga-QtTotalParcelaCompra)))/(Taxa);
	ELSE IF MISSING(Taxa) THEN VrAceleracoes=ValorRestante;
	ELSE VrAceleracoes=0;
	IF DtReferencia >= DtAcordo THEN VrLancamentos = VrTransacao+VrAceleracoes;
	ELSE VrLancamentos=VrAceleracoes;
RUN;
