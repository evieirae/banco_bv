	CREATE TABLE analitico_seguros AS
	SELECT * FROM connection to dbcon
	(
		select distinct
			 Seguro.NuContrato
			,Seguro.SqContratoFinanceiro
		    ,Seguro.DtInicioVigencia
		    ,Seguro.DtFimVigencia
			/*,ProdutoSeguro.SgProdutoSeguradora*/
			,ProdutoSeguro.CdTipoSeguro
			,ProdutoSeguro.CdProdutoSeguro
			,ProdutoSeguro.NmProdutoSeguro
			,Seguro.VrPremioSeguro

		from DBCONTRATOSEGURO.TbSeguroContratoFinanceiro Seguro

			left join DBCONTRATOSEGURO.TbProdutoSeguro ProdutoSeguro
				on Seguro.CdProdutoSeguro = produtoseguro.CdProdutoSeguro

		where Seguro.NuContrato > 0
		  and Seguro.DtProcessamento < today()
		  and ProdutoSeguro.CdProdutoSeguro <> 525
		order by
			 Seguro.NuContrato
			,Seguro.SqContratoFinanceiro
		)
;QUIT;

proc sql;
%CONECTADB(server=PTA);
	CREATE TABLE analitico_fraudados AS
	SELECT * FROM connection to dbcon
	(
		/*Base de contratos Gestão de Contratos*/
		select distinct
			a.NuContrato as NuContratoGestao
			,CONVERT(VARCHAR, a.dtProcessamento - DAY(a.dtProcessamento) + 1, 103) AS dtProcessamento
			,LEFT(CONVERT(VARCHAR, a.dtProcessamento,112),6) AS NuAnoMes
			,a.CdAgenteRecebedor
			,b.NuContaDebito
			,CASE WHEN A.CdAgenteRecebedor = 198 THEN 'FRAUDE CONFIRMADA'
				WHEN A.CdAgenteRecebedor = 218 THEN 'FRAUDE ERRO OPERACIONAL'
		 		WHEN A.CdAgenteRecebedor = 219 THEN 'FRAUDE CONDENATORIA'
		 		WHEN A.CdAgenteRecebedor = 248 THEN 'FRAUDE ELETRONICA'
			END AS DsTipoFraude
			,sum(a.VrLancamentoContabil) AS VrLancamentoContabil
			,CASE WHEN C.CdProduto = 10 THEN 'CRÉDITO PESSOAL'
				WHEN C.CdProduto = 11 THEN 'CONSIGNADO'
		 		WHEN C.CdProduto = 12 THEN 'CDC - Crédito Direto'
		 		WHEN C.CdProduto = 13 THEN 'CDC - Crédito Direto'
		 		WHEN C.CdProduto = 14 THEN 'CDC - Crédito Direto'
			END AS Produto
		from
			gestaocontabil.TbLancamentoContabil AS a
    		LEFT JOIN gestaocontabil.TbRoteiroContabil AS b ON a.CdRoteiroContabil = b.CdRoteiroContabil
			LEFT JOIN DBA.VwTbContrato AS c on a.Nucontrato = c.Nucontrato
			LEFT JOIN DBA.TbProposta AS e ON a.Nucontrato = e.Nucontratogestao and e.Stproposta = 'AP'
		where
			a.CdAgenteRecebedor IN (198,219,218)
			and not a.Nucontrato in (12035000320728,12040000154450,12053000231655,12065000181678,12078000141055,12085000107639
				,12116000006624,12258000049153,12272000029540,12272000040389,12272000054659,12272000066425)
    		and b.NuContaDebito IN (81999008405008,81999008400243,81999008400251,81999008400260,81999008403005,81952100100009)
			and LEFT(CONVERT(VARCHAR, a.dtProcessamento,112),6) = &mes_apuracao_char.
			/*Alterei between  &mes_cancelados_char. and &mes_apuracao_char.*/
		group by
			a.nucontrato,
			a.dtProcessamento,
			A.CdAgenteRecebedor,
			b.NuContaDebito,
			c.DtContrato,
			c.CdProduto

		union

		/*Base de contratos FLEX*/
		select distinct
			a.NuContrato as NuContratoGestao
			,CONVERT(VARCHAR, a.DtMovimento - DAY(a.DtMovimento) + 1, 103) AS dtProcessamento
			,LEFT(CONVERT(VARCHAR, a.DtMovimento,112),6) AS NuAnoMes
			,a.NuContaContabil AS CdAgenteRecebedor
			,a.NuContaContabilFlexCube AS NuContaDebito
			,CASE WHEN A.NuContaContabil = 198 THEN 'FRAUDE CONFIRMADA'
		 		WHEN A.NuContaContabil = 218 THEN 'FRAUDE ERRO OPERACIONAL'
		 		WHEN A.NuContaContabil = 219 THEN 'FRAUDE CONDENATORIA'
			END AS DsTipoFraude
			,sum(a.VrLancamentoContabil) AS VrLancamentoContabil
			,CASE WHEN A.CdProduto = 10 THEN 'CRÉDITO PESSOAL'
		 		WHEN A.CdProduto = 11 THEN 'CONSIGNADO'
		 		WHEN A.CdProduto = 12 THEN 'CDC - Crédito Direto'
		 		WHEN A.CdProduto = 13 THEN 'CDC - Crédito Direto'
		 		WHEN A.CdProduto = 14 THEN 'CDC - Crédito Direto'
			END AS Produto
		from
    		STGTXT_FLEX.TbLancamentoContabilFlexCube AS a
    	where
     		a.NuContaContabil IN (198,219,218)
     		AND a.NuContaContabilFlexCube IN (81999008405008,81999008400243,81999008400251,81999008400260,81999008403005,81952100100009)
			AND a.DtMovimento =
				(
					SELECT MAX(aa.DtMovimento)
    				FROM STGTXT_FLEX.TbLancamentoContabilFlexCube AS aa
    				WHERE aa.Nucontrato = a.Nucontrato AND aa.DtMovimento = a.DtMovimento
				)
			and LEFT(CONVERT(VARCHAR, a.DtMovimento,112),6) = &mes_apuracao_char.
			/*Alterei between  &mes_cancelados_char. and &mes_apuracao_char.*/
		group by
			a.nucontrato,
			a.DtMovimento,
			a.NuContaContabil,
			a.NuContaContabilFlexCube,
			a.CdProduto
		order by 2
	)
;QUIT;

PROC SORT DATA=analitico_fraudados NODUPKEY;
BY NuContratoGestao;
RUN;

/**ATENCAO: TODA PRODUCAO PJ É EXPURGADA, POIS VALOR TOTAL SÓ ENTRA NO DENOMINADOR**/
proc sql;
create table prod_total as
select
	Prod.NuAnoMes as Ano_Base
	,Prod.CdColaborador as Cod_Operador
	,Colab.idmatricula as ID_RH
	,sum(case when Prod.TpPessoa = 'F'
				and Prod.NmGrupoProdutoMeta<>'Leves-BVO'
				and Prod.NmGrupoProdutoMeta<>'Leves-Multibanco'
				then Prod.VrProducao else 0 end) as VrProducaoTotal
from
	TMP.TbProducao as Prod
	inner join BASES.TbEstruturaComercial Colab
	on Prod.CdColaborador = Colab.CdColaboradorAtivo
	and Prod.NuAnoMes = Colab.NuAnoMes
where
	/*Contratos do mês de apuracao ou provenientes de cancelamento do mês anterior*/
	Prod.NuAnoMes = &mes_apuracao. and Prod.DtCancelamento is null
	or
	Prod.NuAnoMes = &mes_apuracao. and put(prod.DtCancelamento,yymmn6.) >= &mes_apuracao_char.
	/*Excluindo os GRs de veículos devido ao Bateu Levou NmGrupoCargo = 'Gerente Relacionamento Veiculos' */
	and Colab.NmGrupoCargo not in ('Gerente Relacionamento Veiculos','Gerente Relacionamento Solar')
group by
	Prod.NuAnoMes
	,Prod.CdColaborador
	,Colab.idmatricula
;run;

	when Prod.NmCanalEstruturaComercial = 'Canal Solar' and Seguro.CdTipoSeguro = 2 then 10546
	else 0
	end as Produto

	,case when Seguro.CdTipoSeguro = 2 and Seguro.CdProdutoSeguro = 136 then 'BV Mais Pesados'
	when Seguro.CdTipoSeguro = 2 and NmGrupoProdutoMeta = 'Leves' then 'BV Mais Leves'
	when Seguro.CdTipoSeguro = 2 and NmGrupoProdutoMeta = 'Motos' then 'BV Mais Motos'
	when Seguro.CdTipoSeguro = 2 and NmGrupoProdutoMeta = 'Pesados' then 'BV Mais Pesados'
	/*CVG do BV MAIS junta-se com Leves e Pesados*/
	when Seguro.CdTipoSeguro = 2 and NmGrupoProdutoMeta = 'CVG Leves' then 'BV Mais Leves'
	when Seguro.CdTipoSeguro = 2 and NmGrupoProdutoMeta = 'CVG Pesados' then 'BV Mais Pesados'
	/*---*/
	when Seguro.CdTipoSeguro = 5 AND CdGrupoProdutoSeguro IN(42,43,44,45,46,47)  then 'Prêmio AP - FGTS'
	when Seguro.CdTipoSeguro = 5 then 'Prêmio AP Premiável'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and Seguro.CdProdutoSeguro = 512 then 'Seguros RCF Pesados'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and NmGrupoProdutoMeta = 'Leves' then 'Seguros RCF Leves'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and NmGrupoProdutoMeta = 'Motos' then 'Seguros RCF Motos'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and NmGrupoProdutoMeta = 'Pesados' then 'Seguros RCF Pesados'
	/*CVG do RCF junta-se com Leves e Pesados*/
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and NmGrupoProdutoMeta = 'CVG Leves' then 'Seguros RCF Leves'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 13 and NmGrupoProdutoMeta = 'CVG Pesados' then 'Seguros RCF Pesados'
	/*---*/
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 11 then 'Seguros Completo Leves'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 14 then 'Seguros Casco Leves'
	when Seguro.CdTipoSeguro = 1 and CdGrupoProdutoSeguro = 12 then 'Seguros GE'

    /*----Privado ----*/
	when Prod.NmCanalEstruturaComercial = 'Canal Privado e FGTS' and Seguro.CdTipoSeguro = 2 then 'BV Mais Privado'
	/*----Solar ----*/
	when Prod.NmCanalEstruturaComercial = 'Canal Solar' and Seguro.CdTipoSeguro = 2 then 'Prêmio SPF Solar'
	else 'Desconhecido'
	end as Descricao
	,Case When fraude.NuContratoGestao = Prod.NuContrato Then 'S' Else 'N' End as Fraude
	,Sum(Seguro.VrPremioSeguro) as Valor
from
	TMP.TbProducao Prod
 inner join BASES.TbEstruturaComercial Colab
		on Prod.CdColaborador = Colab.CdColaboradorAtivo
		and Colab.NuAnoMes = &mes_apuracao.
 left join analitico_seguros Seguro
     on (Prod.NuContrato = Seguro.NuContrato
           and Prod.SqContratoFinanceiro = Seguro.SqContratoFinanceiro)
 left join Automati.TbSeg_ProdutoSeguro GrupoProdutoSeguro
		on  Seguro.CdProdutoSeguro = GrupoProdutoSeguro.CdProdutoSeguro
 left join analitico_fraudados as fraude on prod.Nucontrato = fraude.NuContratoGestao
where
	Prod.NuAnoMes = &mes_apuracao.
	and Prod.NmGrupoProdutoMeta<>'Leves-BVO'
	and Prod.NmGrupoProdutoMeta<>'Leves-Multibanco'
	/*Retira fraudados*/
/*	and Prod.NuContrato not in (select distinct NuContratoGestao from analitico_fraudados)*/
	/*Retirado os seguros não identificados ou gerados por cartesiano e excluído operado Canal Digital
	e tbm colaboradores de Negócios Centralizados*/
	and Seguro.CdTipoSeguro is not null
	and Seguro.CdProdutoSeguro is not null
	/*Retira Operador Canal Digital Especial*/
	and Prod.CdColaborador <> 1000000
	/*Retira Contratos de Negócios Centralizados*/
	and Prod.DsTipoCanalProducao <> 'NEGOCIOS CENTRALIZADOS'
/*Redundância no  código excluindo os GRs de veículos devido ao Bateu Levou NmGrupoCargo = 'Gerente Relacionamento Veiculos' */
	and Colab.NmGrupoCargo not in ('Gerente Relacionamento Veiculos','Gerente Relacionamento Solar')

group by 1,2,3,4,5,6,7

;run;
