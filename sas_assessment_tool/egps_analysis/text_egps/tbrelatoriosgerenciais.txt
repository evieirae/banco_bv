PROC SQL;
CREATE TABLE TbMeta AS
SELECT distinct
	 Meta.NuAnoMes
	,Meta.NmGrupoProdutoMeta
	,Meta.DsTipoCanalProducao
	,CASE
		WHEN Meta.TpValor like '%SPF%' THEN 'SPF'
		WHEN Meta.TpValor like '%Cap%' THEN 'Cap'
		WHEN Meta.TpValor like '%Demais%' THEN 'Demais'
        WHEN Meta.TpValor = 'VrAutoTotal' THEN  'Completo' /* altered on feb 24 2021 */
		END AS TpSeguro
									/*META*/

	,sum(case when Meta.Tipo = 'Meta'	and Meta.TpValor = 'VrProducao' then Meta.VrMedida else 0 end ) as VrProducaoMeta
	,sum(case when Meta.Tipo = 'Meta'	and Meta.TpValor = 'VrMargemContribuicao' then Meta.VrMedida else 0 end ) as VrMargemContribuicaoMeta
	,sum(case when Meta.Tipo = 'Meta'	and Meta.TpValor = 'VrFaturamento' then Meta.VrMedida else 0 end ) as VrFaturamentoMeta
	,sum(case when Meta.Tipo = 'Meta'	and Meta.TpValor = 'VpTxFinanciamento' then Meta.VrMedida else 0 end ) as VpTxFinanciamentoMeta

	,sum(case when Meta.Tipo = 'Meta'	and Meta.TpValor in ('VrDemais','VrCap','VrSPF','VrAutoTotal') then Meta.VrMedida else 0 end ) as VrPremioMeta
	,max(case when Meta.Tipo = 'Meta'	and Meta.TpValor in ('PcSPF','PcCap','PcDemais','PcAutoTotal') then Meta.VrMedida else 0 end ) as PcSegurosMeta

									/*ORÇAMENTO*/

	,sum(case when Meta.Tipo = 'Orçamento'  and Meta.TpValor = 'VrProducao' then Meta.VrMedida else 0 end ) as VrProducaoOrcado
	,sum(case when Meta.Tipo = 'Orçamento'	and Meta.TpValor = 'VrMargemContribuicao' then Meta.VrMedida else 0 end ) as VrMargemContribuicaoOrcado
	,sum(case when Meta.Tipo = 'Orçamento'	and Meta.TpValor = 'VpTxSpread' then Meta.VrMedida else 0 end ) as VpTxSpreadOrcado
	,sum(case when Meta.Tipo = 'Orçamento'	and Meta.TpValor = 'VpTxFinanciamento' then Meta.VrMedida else 0 end ) as VpTxFinanciamentoOrcado

	,sum(case when Meta.Tipo = 'Orçamento'	and Meta.TpValor in ('VrDemais','VrCap','VrSPF','VrAutoTotal') then Meta.VrMedida else 0 end ) as VrPremioOrcado
	,sum(case when Meta.Tipo = 'Orçamento' 	and Meta.TpValor in ('PcSPF','PcCap','PcDemais','PcAutoTotal') then Meta.QtNumerador else 0 end ) as QtSegurosNumerador
	,sum(case when Meta.Tipo = 'Orçamento' 	and Meta.TpValor in ('PcSPF','PcCap','PcDemais','PcAutoTotal') then Meta.QtDenominador else 0 end ) as QtSegurosQtDenominador

									/*BEST NUMBER*/

	,sum(case when Meta.Tipo = 'BN' 		and Meta.TpValor = 'VrProducao' then Meta.VrMedida else 0 end ) as VrProducaoBN

FROM bases.TbMeta Meta
WHERE Meta.NuAnoMes >= 201901

group by 1,2,3,4
;run;

PROC SQL;
CREATE TABLE TbProducao AS
SELECT
									/*Producao*/
	producao.NuAnoMes
	,producao.NmGrupoProdutoMeta
	,producao.DsTipoCanalProducao
	,producao.FxIdade
	,sum(producao.VrProducao) as VrProducao
	,sum(producao.VrProducaoLiquida) as VrProducaoLiquida
	,sum(producao.VrMargem) as VrMargemContribuicao
	,sum(case when producao.NuContrato is not null then 1 else 0 end) as QtContrato
	,sum(producao.VrProducaoProjetado) as VrProducaoProjetado
	,sum(producao.VpTirContabil) as VpTirContabil
	,sum(producao.VpTxGerencial) as VpTxGerencial
	,sum(producao.VrTarifaCadastro) as VrTarifaCadastro
	,sum(producao.VrTarifaAvaliacaoBem) as VrTarifaAvaliacaoBem
	,sum(CASE WHEN Producao.VrTarifaCadastro > 0 THEN 1 ELSE 0 END) AS QtTarifaCadastro
	,sum(CASE WHEN Producao.VrTarifaAvaliacaoBem > 0 THEN 1 ELSE 0 END) AS QtTarifaAvaliacaoBem
	,sum(case when producao.StData = 1 then producao.VrProducao else 0 end) as VrProducaoDia
	,sum(producao.VrProducaoDiaUtil) as VrProducaoDiaUtil
	,max(producao.QtTotalDiaUtilMes) as QtTotalDiaUtilMes
	,max(producao.QtCorridosDiaUtilMes) as QtCorridosDiaUtilMes
	,sum(producao.VrComissaoTotal) as VrComissaoTotal
	,sum(producao.VrComissaoParcelada) as VrComissaoParcelada
    ,sum(producao.VrFinanciamento) as VrFinanciamento
	,sum(producao.VrPerdaEstimada) as VrPerdaEstimada
    ,sum(producao.VpSpreadRebate) as VpSpreadRebate
	,sum(producao.VpSpread) as VpSpread

FROM tmp.TbProducao producao

WHERE producao.NuAnoMes >= 201901
group by 1,2,3,4
;run;

PROC SQL;
CREATE TABLE TbSeguros AS
SELECT
										/*Seguros*/
	producao.NuAnoMes
	,producao.NmGrupoProdutoMeta
	,producao.DsTipoCanalProducao
	,sum(producao.VrBVMais) as VrSPF
	,sum(producao.VrCap) as VrCap
	,sum(producao.VrDemais) as VrDemais
	,sum(producao.VrBVMaisProjetado) as VrSPFProjetado
	,sum(producao.VrCapProjetado) as VrCapProjetado
	,sum(producao.VrDemaisProjetado) as VrDemaisProjetado
	,sum(producao.VrDemaisSemCompletoMaxProjetado) AS VrDemaisSCProjetado /* add on feb 24 2021 */
	,sum(producao.VrCompletoProjetado) AS VrCompletoProjetado /* add on feb 24 2021 */
	,sum(case when producao.StData = 1 then producao.VrBVMais else 0 end) as VrSPFDia
	,sum(case when producao.StData = 1 then producao.VrCap else 0 end) as VrCapDia
	,sum(case when producao.StData = 1 then producao.VrDemais else 0 end) as VrDemaisDia
	,sum(producao.FlBVMais) as QtSPF
	,sum(producao.FlCap) as QtCap
	,sum(producao.FlDemais) as QtDemais
	,sum(producao.FlDemaisSemCompleto) as QtDemaisSC
	,sum(producao.FlCompleto) as QtCompleto
	,sum(CASE WHEN producao.FlBVMais = 1 OR /* add on feb 24 2021 */
                   producao.FlCap = 1 OR /* add on feb 24 2021 */
                   producao.FlDemais = 1 OR /* add on feb 24 2021 */
                   producao.FlDemaisSemCompleto = 1 OR /* add on feb 24 2021 */
                   producao.FlCompleto = 1 /* add on feb 24 2021 */
              THEN 1 ELSE 0 END) as QtContratosSeguros /* add on feb 24 2021 */
	,sum(producao.FlBVMaisEleg) as QtSPFEleg
	,sum(producao.FlCapEleg) as QtCapEleg
	,sum(producao.FlDemaisEleg) as QtDemaisEleg
	,sum(producao.FlDemaisEleg) as QtCompletoEleg
	,sum(producao.VrComissaoTotal) as VrComissaoTotal
	,sum(producao.VrComissaoParcelada) as VrComissaoParcelada
    ,sum(producao.VrFinanciamento) as VrFinanciamento
	,sum(producao.VrPerdaEstimada) as VrPerdaEstimada
	,sum(producao.VpSpreadRebate) as VpSpreadRebate
	,sum(producao.VpSpread) as VpSpread

FROM tmp.TbProducao producao

WHERE producao.NuAnoMes >= 201901
group by 1,2,3
;run;

proc sql;
create table TbSegurosCancelados as
select
	 SegurosCancelados.NuAnoMes
	,SegurosCancelados.NmGrupoProdutoMeta
	,SegurosCancelados.DsTipoCanalProducao

	,sum(SegurosCancelados.VrBVMais) as VrSPFCancelados
	,sum(SegurosCancelados.VrCap) as VrCapCancelados
	,sum(SegurosCancelados.VrDemais) as VrDemaisCancelados

	,sum(SegurosCancelados.FlBVMais) as QtSPFCancelados
	,sum(SegurosCancelados.FlCap) as QtCapCancelados
	,sum(SegurosCancelados.FlDemais) as QtDemaisCancelados

from bases.TbSegurosProducaoCancelados SegurosCancelados
Where SegurosCancelados.NuAnoMes < year(today())*100+month(today())

data TbConsolidadoSeguros;

		TbSegurosCancelados
	;run;

proc sort data=TbConsolidadoSeguros;
by NuAnoMes NmGrupoProdutoMeta DsTipoCanalProducao
;run;

proc transpose data=TbConsolidadoSeguros out=TbConsolidadoSegurosUnpivot
(drop=_LABEL_ rename=_NAME_=TpValor rename=COL1=VrMedida rename=COL2=VrMedidaCancelado where=(VrMedida<>0));
by NuAnoMes NmGrupoProdutoMeta DsTipoCanalProducao
;run;

PROC SQL;
CREATE TABLE TbSegurosProducao as
SELECT
	 SegurosUnpivot.NuAnoMes
	,SegurosUnpivot.NmGrupoProdutoMeta
	,SegurosUnpivot.DsTipoCanalProducao
	,CASE
		WHEN SegurosUnpivot.TpValor like '%SPF%' THEN 'SPF'
		WHEN SegurosUnpivot.TpValor like '%Cap%' THEN 'Cap'
		WHEN SegurosUnpivot.TpValor like '%Demais%' THEN 'Demais'
		WHEN SegurosUnpivot.TpValor like '%Completo%' THEN 'Completo'
		END AS TpSeguro

	,SUM(CASE WHEN TpValor in ('VrSPF','VrCap','VrDemais') then VrMedida else 0 end) as VrPremio
	,SUM(CASE WHEN TpValor in ('VrSPFProjetado','VrCapProjetado','VrDemaisSCProjetado','VrCompletoProjetado') then VrMedida else 0 end) as VrPremioProjetado /* altered on feb 24 2021 */
	,SUM(CASE WHEN TpValor in ('VrSPFDia','VrCapDia ','VrDemaisDia') then VrMedida else 0 end) as VrPremioDia

	,SUM(CASE WHEN TpValor in ('QtSPF','QtCap','QtDemaisSC','QtCompleto') then VrMedida else 0 end) as QtSeguros /* altered on feb 24 2021 */
	,SUM(CASE WHEN TpValor in ('QtSPFEleg','QtCapEleg','QtDemaisEleg','QtCompletoEleg') then VrMedida else 0 end) as QtSegurosElegiveis /* altered on feb 24 2021 */

	,SUM(CASE WHEN TpValor in ('QtSPFCancelados','QtCapCancelados','QtDemaisCancelados') then VrMedidaCancelado else 0 end) as QtSegurosCancelados
	,SUM(CASE WHEN TpValor in ('VrSPFCancelados','VrCapCancelados','VrDemaisCancelados') then VrMedidaCancelado else 0 end) as VrPremioCancelados

	,SUM(CASE WHEN TpValor in ('QtContratosSeguros') then VrMedida else 0 end) as QtContratosSeguros
FROM TbConsolidadoSegurosUnpivot SegurosUnpivot
GROUP BY 1,2,3,4
;RUN;

PROC SQL;
	CREATE TABLE TbDiasUteis AS
		SELECT
			NuAnoMes
			,MAX(QtCorridosDiaUtilMes) AS  QtCorridosDiaUtilMes

		FROM bases.TbTempoData
		WHERE DtTempo <  today()

		GROUP BY 1

PROC SQL;
	CREATE TABLE  TbProducaoMCN  as
	SELECT
	 McnProducao.NuAnoMes
	,McnProducao.NmGrupoProdutoMeta
	,McnProducao.DsTipoCanalProducao
	,DiasUteis.QtTotalDiaUtilMes
	,DiasUteis.QtCorridosDiaUtilMes
	,SUM(CASE WHEN McnProducao.TpValor = 'VrMcn' THEN McnProducao.vrMedida END) AS VrProducao
	,SUM(CASE WHEN McnProducao.TpValor = 'VrMcn' THEN McnProducao.vrMedida END)/DiasUteis.QtCorridosDiaUtilMes as VrProducaoDiaUtil

	FROM bases.TbMetaMCN McnProducao
			LEFT JOIN TbDiasUteis DiasUteis
				ON McnProducao.nuAnoMes =  DiasUteis.nuAnoMes

	WHERE McnProducao.nuAnoMes >=201901

	GROUP BY 1,2,3,4,5
