CREATE TABLE work.tblProcessoObjetoPrognostico AS

SELECT * FROM CONNECTION TO postgres(

select
case
 	when p.serial_import like ('%INQ%') then (left(p.SERIAL_IMPORT,10))
    when p.serial_import like ('%BVT%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%CON%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%VAC%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%CTO%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%TRI%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%TRB%') then (left(p.SERIAL_IMPORT,10))
		else p.serial_import end        as Cd_Causa
,p.pasta
,p.SERIAL_IMPORT 						as SERIAL_IMPORT
,p.ID_NATUREZA						    as IdNatureza
,n.NATUREZA								as NmNatureza
,p.NATUREZA_UNIDADE_CUSTOM				as IdNaturezaUnidadeCustom
,NC.NATUREZA_UNIDADE_CUSTOM				as NmNaturezaUnidadeCustom
,null::numeric 							as CdCausaUnidade
,'p' 									as CdSistema
,CASE
	WHEN p.ID_NATUREZA = 28 AND p.NATUREZA_UNIDADE_CUSTOM = 2 THEN 'TRI'
	WHEN p.ID_NATUREZA = 28 AND p.NATUREZA_UNIDADE_CUSTOM = 1 THEN 'TBR'

	WHEN p.ID_NATUREZA = 21 AND p.NATUREZA_UNIDADE_CUSTOM = 12 THEN 'CON'

	WHEN p.ID_NATUREZA = 27 AND p.NATUREZA_UNIDADE_CUSTOM = 5 THEN 'BVT'
	WHEN p.ID_NATUREZA = 32 AND p.NATUREZA_UNIDADE_CUSTOM IN (7, 11 , 12) THEN 'INQ'
	ELSE NULL END as CdUnidadeExyon

,p.id_carteira_custom					as Id_Carteira_Custom

,CAST(O.DATA_CADASTRO AS date)			AS DtCadastroObjeto
,O.ID_METODO_ATUALIZACAO				AS IdMetodoAtulizacao
,MET.METODO_ATUALIZACAO					AS MetodoAtualizacao
,O.ID_OBJETO							AS IdObjeto
,POP.MULTA_PEDIDO						AS MultaVlrPedido
,POP.MULTA_OFICIO_PEDIDO    			AS MultaOficio
,POP.MULTA_ISOLADA_PEDIDO				AS MultaIsolada
,OBJ.OBJETO								AS NmObjeto
,PC.VALOR_PERDA_PROVAVEL				AS VlProvisaoProvavel
,POP.VALOR_PERDA_PROVAVEL_ATUALIZAD 	AS VlProvisaoProvavelAtualizado
,( PC.VALOR_PEDIDO - (PC.VALOR_PERDA_POSSIVEL + pc.VALOR_PERDA_PROVAVEL) ) AS VlProvisaoRemoto
,( POP.VALOR_PEDIDO_ATUALIZADO - (POP.VALOR_PERDA_POSSIVEL_ATUALIZAD + POP.VALOR_PERDA_PROVAVEL_ATUALIZAD) ) AS VlProvisaoRemotoAtualizado
,PC.VALOR_PERDA_POSSIVEL				AS VlProvisaoPossivel
,POP.VALOR_PERDA_POSSIVEL_ATUALIZAD		AS VlProvisaoPossivelAtualizado
,PC.VALOR_PEDIDO						AS VlProvisaoPedido
,POP.VALOR_PEDIDO_ATUALIZADO			AS VlProvisaoPedidoAtualizado
,tese.TESE_CUSTOM 						AS Ds_Tese_Custom
,PP.DATA_CALCULO 						AS DtProvisaoProcessoCalculo
,P.PROVISIONAR							AS FlagProvisao
,P.ID_ASSUNTO_CUSTOM					AS IdAssuntoJudicial
,ASS.ASSUNTO_CUSTOM						AS NmAssuntoJudicial
,POP.TAXA_PEDIDO						AS Sucumbencia
,POP.JUROS_PEDIDO						AS Juros_Pedido
,POP.CORRECAO_MONETARIA_PEDIDO			AS CorrecaoMonetariaPedido
,parte.PRINCIPAL						AS Principal
,parte.CLIENTE 				 			AS Cliente
,CASE WHEN nc.NATUREZA_UNIDADE_CUSTOM LIKE ('%Recuperação%') THEN 'Recuperacao' ELSE 'Demais' END AS Acesso
,CASE WHEN p.Data_Encerramento IS NULL THEN 'EM CURSO' ELSE 'ENCERRADO' END AS StatusCausa
,(SELECT NOME FROM public.LT_CAUSA_RAIZ WHERE ID_CAUSA_RAIZ = p.ID_CAUSA_RAIZ)                AS Motivo
,(SELECT NOME FROM public.LT_CAUSA_RAIZ_DOIS WHERE ID_CAUSA_RAIZ_DOIS = p.ID_CAUSA_RAIZ_DOIS) AS SubMotivo
,o.ULTIMA_MODIFICACAO                   AS UltimaDataCalculo
,o.JUROS_INICIAL                        AS JurosInicial

,pc.VALOR_PEDIDO                                                        AS ValorPedido
,pc.VALOR_PERDA_PROVAVEL                                                AS ValorPerdaProvavel
,pc.VALOR_PERDA_POSSIVEL                                                AS ValorPerdaPossivel
,(pc.VALOR_PEDIDO -(pc.VALOR_PERDA_POSSIVEL + pc.VALOR_PERDA_PROVAVEL)) AS ValorPerdaRemoto
,pc.VALOR_INSS                                                          AS ValorINSS
,pc.VALOR_RISCO                                                         AS ValorRisco
,pop.VALOR_PEDIDO                                                        									AS ValorPedidoProvisao

,pop.VALOR_PERDA_PROVAVEL                                                									AS ValorProvavelProvisao
,pop.VALOR_PERDA_PROVAVEL_ATUALIZAD                                      									AS ValorProvavelProvisaoAtualizado
,(pop.VALOR_PEDIDO-(pop.VALOR_PERDA_POSSIVEL + pop.VALOR_PERDA_PROVAVEL))  									AS ValorRemotoProvisao
,(pop.VALOR_PEDIDO_ATUALIZADO-(pop.VALOR_PERDA_POSSIVEL_ATUALIZAD + pop.VALOR_PERDA_PROVAVEL_ATUALIZAD)) 	AS ValorPerdaRemotoAtualizado
,CASE WHEN p.DATA_ENCERRAMENTO IS NULL THEN 'SIM' ELSE 'NÃO' END AS FlNmCausaAtiva
,CASE WHEN p.DATA_ENCERRAMENTO IS NULL THEN 1 ELSE 0 END AS FlCausaAtiva
,case when
( (case when O.ID_PROCESSO is null then 0 else 1 end) +
 (case when PC.ID_PROCESSO_OBJETO is null then 0 else 1 end) +
 (case when OBJ.ID_OBJETO is null then 0 else 1 end) ) = 3 then 1 else 0 end as FlObjetoProv

from public.M_PROCESSO as P
left join public.D_PROCESSO_OBJETO 				as O 	 on O.ID_PROCESSO 					= P.ID_PROCESSO
left join public.D_PROCESSO_OBJETO_PARCELA 		as PC 	 on PC.ID_PROCESSO_OBJETO 			= O.ID_PROCESSO_OBJETO
left join public.D_PROVISAO_PROCESSO 			as PP 	 on PP.ID_PROCESSO 		   			= P.ID_PROCESSO
		and PP.DATA_CALCULO 				= (SELECT MAX(DATA_CALCULO) from public.D_PROVISAO_PROCESSO x where x.ID_PROCESSO = P.ID_PROCESSO)
left join public.D_PROVISAO_OBJETO_PROCESSO 	as POP 	 on POP.ID_PROVISAO_PROCESSO 		= PP.ID_PROVISAO_PROCESSO
														and POP.ID_PROCESSO_OBJETO 			= O.ID_PROCESSO_OBJETO
left join public.D_PROCESSO_PARTE 				as parte on parte.ID_PROCESSO 				= P.ID_PROCESSO
left join public.LT_NATUREZA 					as n	 on N.ID_NATUREZA 					= p.ID_NATUREZA
left join public.LT_NATUREZA_UNIDADE_CUSTOM 	as NC	 on NC.id_NATUREZA_UNIDADE_CUSTOM 	= P.NATUREZA_UNIDADE_CUSTOM
left join public.LT_OBJETO 						as OBJ 	 on OBJ.ID_OBJETO 					= O.ID_OBJETO
left join public.LT_TESE_CUSTOM 				as tese  on O.TESE_CUSTOM = tese.ID_TESE_CUSTOM
left join public.LT_METODO_ATUALIZACAO 			as MET	 on O.ID_METODO_ATUALIZACAO = MET.ID_METODO_ATUALIZACAO
left join public.LT_ASSUNTO_CUSTOM 				as ASS	 on P.ID_ASSUNTO_CUSTOM = ASS.ID_ASSUNTO_CUSTOM

);DISCONNECT FROM postgres;
