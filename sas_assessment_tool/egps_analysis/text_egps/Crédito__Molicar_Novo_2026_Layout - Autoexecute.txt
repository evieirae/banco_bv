PROC SQL;
    CREATE TABLE Validacao_Inc_Exc_Molicar AS
        SELECT DISTINCT
            a.*,
            b.CODIGO AS cod_atual,
            b.MARCA AS marca_atual,
            c.CODIGO AS cod_anterior,
            c.MARCA AS marca_anterior,

        case when c.CODIGO NE '' and b.CODIGO =  '' and a.tipo = 'exclusao' then 'Sim'
             when c.CODIGO =  '' and b.CODIGO NE '' and a.tipo = 'inclusao' then 'Sim'

        FROM INC_EXC as a LEFT JOIN BV_REG as b
                            ON a.codigo = b.CODIGO
                          LEFT JOIN BV_REG4 as c

    OKM 7. V2026 7. V2025 7. V2024 7. V2023 7. V2022 7. V2021 7. V2020 7. V2019 7. V2018 7. V2017 7. V2016 7. V2015 7. V2014 7. V2013 7. V2012 7. V2011 7.

    MOTOR2=MOTOR;
    MOTOR3 = COMPRESS (MOTOR2);
