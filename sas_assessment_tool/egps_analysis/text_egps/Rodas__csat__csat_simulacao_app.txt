%include '/infogrc/utils/all-includes.sas';

proc sql; select max(dt_evento) format date9. into:dt_max from leadsdig.csat_simulacao_app; run;
proc sql; select max(dt_evento) format date9. into:dt_maxc from leadsdig.csat_contratacao_app; run;

data _null_;
dt_merge = intnx('day',"&dt_max."d,-5);
dt_merge_bq  = "'"||put(year(dt_merge),4.)||"-"||put(month(dt_merge),z2.)||"-"||put(day(dt_merge),z2.)||"'";

call symputx('dt_merge_bq',dt_merge_bq);
call symputx('dt_merge',put(dt_merge, date9.));
;run;

/*%let dt_merge = '01may2025'd;*/
/*%let dt_merge_bq = '2025-05-01';*/

%put &dt_merge;
%put &dt_merge_bq;

proc sql;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage);
create table csat as
select * from connection to perm(
SELECT
  * EXCEPT(ROW)
FROM (
  SELECT
    *,
    ROW_NUMBER() OVER(PARTITION BY dt_evento, cd_bv, simulacao ORDER BY comentario DESC, pesquisa DESC) ROW
  FROM (
    SELECT
      dt_evento,
      dm_evento,
      cd_pseudo_id,
      cd_bv,
      COALESCE(simulacao, simulacao2) AS simulacao,
      COALESCE(pesquisa, pesquisa2) AS pesquisa,
      COALESCE(comentario, comentario2) AS comentario,
      origem
    FROM (
      SELECT
        dt_evento,
        dm_evento,
        cd_pseudo_id,
        cd_bv,
        simulacao,
        simulacao2,
        CASE
          WHEN pesquisa = "Muito insatisfeito" THEN "Péssima"
          WHEN pesquisa = "Insatisfeito" THEN "Ruim"
          WHEN pesquisa = "Neutro" THEN "Regular"
          WHEN pesquisa = "Satisfeito" THEN "Boa"
          WHEN pesquisa = "Muito satisfeito" THEN "Ótima"
          ELSE pesquisa
      END
        AS pesquisa,
        CASE
          WHEN pesquisa2 = "Muito insatisfeito" THEN "Péssima"
          WHEN pesquisa2 = "Insatisfeito" THEN "Ruim"
          WHEN pesquisa2 = "Neutro" THEN "Regular"
          WHEN pesquisa2 = "Satisfeito" THEN "Boa"
          WHEN pesquisa2 = "Muito satisfeito" THEN "Ótima"
          ELSE pesquisa2
      END
        AS pesquisa2,
        comentario,
        comentario2,
        origem
      FROM (
        SELECT
          DISTINCT dt_evento,
          dm_evento,
          cd_pseudo_id,
          cd_bv,
          nm_tela,
          CASE
            WHEN nm_tela LIKE "%bd|financiamentoVeiculo|contratacao|%" THEN 'contratacao'
            ELSE 'simulacao'
        END
          origem,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "simulationId") AS simulacao,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "simulation_id") AS simulacao2,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "pesquisa_satisfacao") AS pesquisa,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "survey_answer") AS pesquisa2,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "textComments") AS comentario,
          (
          SELECT
            te_parametro_evento
          FROM
            UNNEST(ar_parametros_evento)
          WHERE
            nm_parametro_evento = "text_comment") AS comentario2
        FROM
          `bv-pdbd-prd.dvry_clie_expdigital.navegacao_appweb_fato`
        WHERE
          1=1
          AND dt_particao >= &dt_merge_bq
          AND tp_plataforma = 'APP'
          AND nm_evento = 'buttonClick'
          AND (nm_tela LIKE '%bd|financiamentoVeiculo|contratacao|%'
            OR nm_tela LIKE '%bd|financiamentoVeiculos|simulador|%') )) ))
WHERE
  ROW = 1
  AND pesquisa IS NOT null
);
   disconnect from perm;
quit;

data contratacao; set csat (where=(origem="contratacao" and dt_evento ne . and (pesquisa ne '' or comentario ne ''))); drop origem;run;

data simulacao; set csat (where=(origem="simulacao" and dt_evento ne . and (pesquisa ne '' or comentario ne ''))); drop origem;run;

/*data leadsdig.csat_contratacao_app; set contratacao; run;*/
/*data leadsdig.csat_simulacao_app; set simulacao; run;*/

proc sql; delete from leadsdig.csat_simulacao_app where dt_evento >= "&dt_merge."d; run;
proc sql; insert into leadsdig.csat_simulacao_app select * from simulacao; run;

proc sql; delete from leadsdig.csat_contratacao_app where dt_evento >= "&dt_merge."d; run;
proc sql; insert into leadsdig.csat_contratacao_app select * from contratacao; run;

	FILE="/riscosbvf/BA/processos/output/csat_simulacao_app.csv" REPLACE

RUN;
