%macro teste_sazonalidade(variavel);

PROC SQL;
SELECT MIN(&variavel.) INTO :MIN
FROM BASE_OVER90_A;
RUN;

proc sql;
create table BASE_OVER90_A as
select *,
(case when &min.<=0 then (&variavel.+((-1)*(&min.)+0.1)) else &variavel. end) as &variavel.2
from BASE_OVER90_A;
quit;

ods output D8A#1=SalesD8A_1;
ods output D8A#2=SalesD8A_2;
ods output D8A#3=SalesD8A_3;
ods output D8A#4=SalesD8A_4;

proc x13 data=BASE_OVER90_A date=data;
   var &variavel.2;
   transform power=0;
   arima model=( (0,1,1)(0,1,1) );
   estimate;
   x11;
run;

title 'Stable Seasonality Test';
proc print data=SalesD8A_1 LABEL;
run;
title 'Nonparametric Stable Seasonality Test';
proc print data=SalesD8A_2 LABEL;
run;
title 'Moving Seasonality Test';
proc print data=SalesD8A_3 LABEL;
run;
title 'Combined Seasonality Test';
proc print data=SalesD8A_4 LABEL NOOBS;
   var _NAME_ Name1 Label1 cValue1;
run;

proc x12 data=BASE_OVER90_A date=data noprint;
var &variavel.2;
x11;
output out=out a1 d11;
run;
title 'Serie DSZonalizada';
proc sgplot data=out;
series x=data y=&variavel.2_A1 / name = "A1" markers
markerattrs=(color=red symbol='asterisk')
lineattrs=(color=red);
series x=data y=&variavel.2_D11 / name= "D11" markers
markerattrs=(symbol='circle')
lineattrs=(color=blue);
yaxis label='Original and Seasonally Adjusted Time Series';
run;

proc sql;
create table BASE_OVER90_A as
select a.*,
(case when &min.<=0 then b.&variavel.2_D11+&min.-0.1 else b.&variavel.2_D11 end) as &variavel._DSZ
from BASE_OVER90_A as a
left join out as b
on a.data=b.data;
quit;

%mend;

%macro jbtest(data,var);
   ods listing close;
   proc univariate data = &data ;
       var &var;
      ods output moments = _1;
   run;

   data _2;
      set _1;
      label = label1; value = nValue1; output;
      label = label2; value = nValue2; output;
      drop cvalue: label1 label2 nvalue:;
   run;

   proc transpose data = _2 out = _3;
      by varname notsorted;
      id label;
      var value;
   run;

   data JB_final;
      set _3;
      jb = (skewness**2 * n)/6 + ((kurtosis)**2 *n)/24;
      p = 1 - probchi(jb, 2);
      lable jb = 'JB Statistic' p = 'P-value'
         kurtosis = 'Excess Kurtosis';
   run;
   ods listing;
   proc print data = JB_final label;
      var varname n skewness kurtosis jb p;
   run;

   proc datasets nolist;
      delete _:;
   quit;
%mend;
%jbtest(OUTRES, RES);

				  else "não" end) as normalidade_residuos
from JB_final;

VAR &vari.;
HISTOGRAM &vari. / normal;
PROBPLOT &vari.;
output kurtosis=kur skewness=ske;
TITLE "Tests for Normality of Variable: &vari..";
QUIT;
%mend;

%macro graph_biv(base, resp, vari);

define statgraph dualseries;
   begingraph;
   entrytitle "Distribuição séries";
   layout overlay / xaxisopts=(label="Safra");

	seriesplot x=data y=&resp. / name="Y" legendlabel="Resposta: &resp..";
	seriesplot x=data y=&vari. / yaxis=y2 lineattrs=(color=red) name="X" legendlabel="Explicativa: &vari..";

      discretelegend "Y" "X" / valign=bottom title="Análise descritiva das séries";
   endlayout;
   endgraph;
end;
run;

run;
