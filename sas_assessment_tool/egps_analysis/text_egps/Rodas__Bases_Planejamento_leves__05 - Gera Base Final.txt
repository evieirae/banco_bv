proc sql;
create table BasePotencialFinal as
select
A.NuCnpjLoja,
/*a.ChaveCdParceiro_FxIdade,*/
a.nuanomes,
a.cdparceirocomercial,
catt(a.Chave1,a.NmSubEstruturaComercialC) as Chave2,
b.Share as ShareIdeal2,
a.qtgravamesplayers_ProjSab,
b.Share * a.qtgravamesplayers_ProjSab as ContratosIdeias2,
C.TicketFinal as TicketFinal1,

Case when c.Ticket_BV_ProjSab <=0 then e.Ticketbv else c.TicketFinal end as TicketRegional,

c.TicketFinal / CALCULATED TicketRegional AS Multiplo FORMAT COMMAX20.2,

/*case when CALCULATED TicketRegional <=0 then CALCULATED TicketRegional else C.TicketFinal end as tss*/
case when
	case when	CALCULATED Multiplo >=3 then CALCULATED TicketRegional else C.TicketFinal end =0
	then case when CALCULATED TicketRegional <=0 then CALCULATED TicketRegional else C.TicketFinal end
else
case when	CALCULATED Multiplo >=3 then CALCULATED TicketRegional else C.TicketFinal end
end
as TicketFinal,

CALCULATED TicketFinal * CALCULATED ContratosIdeias2 as ProdIdeal2

from POTENCIAL_RODAS_04 as A

	left join ShareRegional as B
	on catt(a.Chave1,a.NmSubEstruturaComercialC) = b.Chave1

	left join BASE_TICKET_0 as C
	on a.ChaveCdParceiro_FxIdade = c.ChaveCdParceiro_FxIdade

	Left join TicketCdParceiroFxIdade as D
	on d.ChaveCdParceiro_FxIdade = a.ChaveCdParceiro_FxIdade

	Left join TicketRegional as e
	on e.CHAVE = CATT(a.NmAreaEstruturaComercialC,a.FXIDADE)

where  1=1
/*and a.NuCnpjLoja = '07796847000173'*/
/*and a.CdParceiroComercial = 774*/
/*and a.cdparceirocomercial = 1086*/
/*and catt(a.Chave1,a.NmSubEstruturaComercialC) = '1 ou 2ConcessionariaPequenoRegião Sul3 a 7 anosCanal Rodas e EGV II - Regional VI'*/
;run;
