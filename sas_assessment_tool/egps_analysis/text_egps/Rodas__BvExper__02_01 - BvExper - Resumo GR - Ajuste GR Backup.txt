/**********************************************************************/
/*SEPARA OS GR QUE FORAM BACKUPS EM ALGUM MÊS DE ACORDO COM A PLANILHA*/
/**********************************************************************/

PROC SQL;
CREATE TABLE BASE_GR_BACKUP_APPEND AS
SELECT

A.*,
b.NuAnoMes_Pausa_BvExper

FROM WORK.TBBVEXPER_RESUMO_GR A

INNER JOIN BASEGRC.TbGrBackup B ON A.NUANOMES = B.NuAnoMes_Congelamento_BvExper AND A.DSLOGINN1 = B.DsLogin_GR

where
A.NmCanalEstruturaComercial not in ('Canal Negocios', 'Canal Solar') /*30/06/2025 - DENIS RH INFORMOU QUE NÃO CONGELAMOS A PONTUAÇÃO DE CANAL DE NEGÓCIOS*/

;RUN;

/****************************************************************/
/*ALTERA A DATA DE CONGELAMENTO DO MÊS ANTERIOR PARA O MES ATUAL*/
/****************************************************************/

DATA BASE_GR_BACKUP_APPEND_1;
SET WORK.BASE_GR_BACKUP_APPEND;

NuAnoMes_Orig = NuAnoMes;
NuAnoMes = NuAnoMes_Pausa_BvExper;

DROP NuAnoMes_Pausa_BvExper;

;RUN;

/*******************************************/
/*CRIA BASE DE EXCLUSAO DE GR BACKUP DO MÊS*/
/*******************************************/

PROC SORT DATA=WORK.BASE_GR_BACKUP_APPEND_1 OUT=BASE_EXCLUSAO(KEEP=NuAnoMes DsLoginN1) NODUPKEY;BY NUANOMES DsLoginN1;RUN;

/**************************************************/
/*ORDENA BASE PRINCIPAL PARA O "MERGE" DE EXCLUSÃO*/
/**************************************************/

PROC SORT DATA=WORK.TbBvExper_Resumo_GR OUT=WORK.TBBVEXPER_RESUMO_GR_1;BY NUANOMES DSLOGINN1;RUN;

/**************************************************/
/*DELETA DA BASE QUE VAI PARA O DASH OS GRS BACKUP*/
/**************************************************/

DATA TBBVEXPER_RESUMO_GR_SEM_GR_BKP;
MERGE 	WORK.TBBVEXPER_RESUMO_GR_1(IN=A)
		WORK.BASE_EXCLUSAO (IN=B);
BY NUANOMES DsLoginN1;
IF A AND NOT B;
;RUN;

/*APENDA BASE DO GR BACKUP COM OS DADOS DO MêS ANTERIOR NO MÊS ATUAL*/

DATA WORK.TBBVEXPER_RESUMO_GR_FINAL;
SET WORK.TBBVEXPER_RESUMO_GR_SEM_GR_BKP
	WORK.BASE_GR_BACKUP_APPEND_1;
;RUN;

/*******************/
/*AJUSTES DE CAMPOS*/
/*******************/

DATA TBBVEXPER_RESUMO_GR_FINAL;
SET WORK.TBBVEXPER_RESUMO_GR_FINAL;
IF NuAnoMes_Orig NE . THEN FlGrBkp = 1;ELSE FlGrBkp = 0;
RENAME NuAnoMes_Orig = NuAnoMes_Original;
;RUN;

PROC SORT DATA=WORK.TBBVEXPER_RESUMO_GR_FINAL;BY NmCanalEstruturaComercial NmSubEstruturaComercialA NuAnoMes DsLoginN1 TpBase;RUN;

/********************************************************************************************************************/
/*06/06/2024 - CRAVAMOS A THATIANA COMO GF POIS NA HIERARQUIA NAO ESTA VINDO ESTA MARCAÇÃO PARA ALGUNS COLABORADORES*/
/********************************************************************************************************************/

proc sql;
update work.TBBVEXPER_RESUMO_GR_FINAL
set DsLoginN2A = 'thatiana.souza'
where
	NuAnoMes >= 202405
  and NmAreaEstruturaComercial = 'Canal Privado e FGTS'
  and NmCanalEstruturaComercial = 'Canal Privado e FGTS'
;RUN;

/**********************/
/*BKP DA BASE ANTERIOR*/
/**********************/

DATA BKP_BU.TbBaseUnica_BvExper_GR_ResumoBKP;
SET EXP_BU.TbBaseUnica_BvExper_GR_Resumo;
;RUN;

/*******************/
/*SALVA A BASE NOVA*/
/*******************/

DATA EXP_BU.TbBaseUnica_BvExper_GR_Resumo;
SET WORK.TBBVEXPER_RESUMO_GR_FINAL;
;RUN;

proc export

delimiter=';';
run;
