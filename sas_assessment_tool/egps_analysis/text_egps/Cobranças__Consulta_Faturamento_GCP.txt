proc sql;
connect to bigquery as perm (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS = must MAX_CHAR_LEN = 30);

create table FATURAS_GCP as
	select distinct
	100*YEAR(DtVencimentoFatura)+MONTH(DtVencimentoFatura) as SAFRA
	,count(*) as Quantidade
	,sum(vrsaldofatura) AS VrSaldoFatura

	from(
    select distinct
	input(cddimcontacartao,best22.) as cddimcontacartao
	,input(cdfatura,best32.) 		as cdfatura format 21.
	,input(vrsaldofatura,best12.) 	as vrsaldofatura
	,dtvencimentofatura

    from connection to perm (
	select distinct
	cddimcontacartao
	,cdfatura
	,vrsaldofatura
	,dtvencimentofatura

	from bv-pdbd-prd.trusted_dwbi_dmoc.tbfatofatura
	where dtvencimentofatura >= '2024-03-01'
	))
group by 1

;

disconnect from perm;
quit;
