CREATE TABLE CARTEIRA_&REF. AS
SELECT A.*
,&DTCARTEIRA. AS DTCARTEIRA FORMAT DATE9.
,CASE WHEN PRODUTO LIKE '%CP_%' THEN 'EP'
	  WHEN PRODUTO LIKE '%CDC_SAUDE%' THEN 'SAUDE'
	  WHEN PRODUTO LIKE '%SOLAR%' THEN 'SOLAR'
	  WHEN PRODUTO LIKE '%PRIVADO%' THEN 'PRIVADO'
	  ELSE 'OUTROS' END AS PRODUTO_AGRUP
,&REF. AS SAFRA
,SALDO AS SALDO_2 FORMAT COMMAX17.2
,&SAFRA_COLL. AS SAFRA_COLL
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT DISTINCT
    CAST(a.nucontrato AS INT64) AS NUCONTRATO,
	c.cd_cpf_cnpj AS nucpfcnpj,
    a.sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,
	a.cdproduto,

	'GCON' AS SISTEMA,

				WHEN K.CD_MOTIVO_CONTRATO_FINANCEIRO IS NULL THEN 'OUTROS'
				ELSE 'OUTROS'
			END AS FL_RENEG,
			CASE
				WHEN CAST(C.CD_PRODUTO AS INT64) = 11 AND K.CD_TIPO_COBRANCA = '1' THEN 'RENEG CONSIG'
				ELSE 'OUTROS'
			END AS FL_RENEG_CONSIG,
	CASE
	     WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO IS NULL                           THEN 'CP_BV'

	     WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO = 56                              THEN 'CP_PRE_APROVADO'
	     WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO = 67                              THEN 'CP_GUIA_BOLSO'
	     WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO = 71                              THEN 'CP_BOM_PRA_CREDITO'
	     WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO = 72                              THEN 'CP_SERASA_ECRED'

    format
        Data DtCompetencia DtFimCompetencia DtReferenciaQuinzena PrxDataUtil date9.;
    do Data = '01JAN2026'd to '31DEC2026'd;

        Ano   = year(Data);
        CdMes = month(Data);
        Dia   = day(Data);

        Safra = Ano*100 + CdMes;

        DtCompetencia    = intnx('month', Data, 0, 'b');
        DtFimCompetencia = intnx('month', Data, 0, 'e');

        CdDiaSemana = weekday(Data);
        DsDiaSemana = put(CdDiaSemana, DOWPT.);

        FlFDS = (CdDiaSemana in (1,7));

        if Dia <= 15 then do;
            CdPeriodo = 1;
            Periodo = '1ª Quinzena';
            DtReferenciaQuinzena = DtCompetencia;
            DiaReferenciaQuinzena = 1;
        end;
        else do;
            CdPeriodo = 2;
            Periodo = '2ª Quinzena';
            DtReferenciaQuinzena = intnx('day', DtCompetencia, 15);
            DiaReferenciaQuinzena = 16;
        end;

        PrxDataUtil = intnx('weekday', Data, 1);

        output;
    end;
run;

SELECT A.*
,SALDO AS SALDO_2 FORMAT COMMAX17.2
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT DISTINCT
    CAST(a.nucontrato AS INT64) AS NUCONTRATO,
	c.cd_cpf_cnpj AS nucpfcnpj,
    a.sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,

    a.dtprocessamento = '2025-06-30'

    AND a.cdproduto IN (13)
    AND (CAST(c.cd_modalidade_produto AS INT64) = 55 OR d.cdsubproduto = 70)

        WHEN x.nudiasatraso BETWEEN 91 AND 150 THEN 'a. 91 a 150'
        WHEN x.nudiasatraso BETWEEN 151 AND 180 THEN 'b. 151 a 180'
        WHEN x.nudiasatraso BETWEEN 181 AND 210 THEN 'c. 181 a 210'
        WHEN x.nudiasatraso BETWEEN 211 AND 270 THEN 'd. 211 a 270'
		ELSE 'e. 271 a 540'

    (CAST(x.vrpermanenciaapropriado AS FLOAT64) + CAST(x.vrtaxacontrato AS FLOAT64)) AS SALDO

    x.dtprocessamento = '2025-06-30'

PROC SQL;
CREATE TABLE BASE_UNICA_LEFT AS
SELECT DISTINCT A.NUCONTRATO
      ,A.SQCONTRATOFINANCEIRO
      ,A.DATA_PAGAMENTO
      ,B.DTULTIMORECEBIMENTO
      ,B.DTVENCIMENTO

      ,CASE WHEN B.DTVENCIMENTO < A.DATA_PAGAMENTO AND (B.DTULTIMORECEBIMENTO >= A.DATA_PAGAMENTO OR
														B.DTULTIMORECEBIMENTO IS NULL) THEN 1 ELSE 0 END AS FL_ATRASO

      ,CASE WHEN B.DTULTIMORECEBIMENTO = A.DATA_PAGAMENTO THEN 1 ELSE 0 END AS FL_PAGO_NO_DIA

FROM RECEBIMENTOS_202507 A
LEFT JOIN TB_PARCELA_FULL B ON
   A.NUCONTRATO = INPUT(B.NUCONTRATO,14.)
   AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

;QUIT;



PROC SQL;
CREATE TABLE TIPO_DE_PAGAMENTO AS
SELECT  NUCONTRATO,
		SQCONTRATOFINANCEIRO,
		DATA_PAGAMENTO,
		SUM(FL_ATRASO) AS QTD_PARC_ATRASO,
		SUM(FL_PAGO_NO_DIA) AS QTD_PARC_PAGAS,
		COUNT(NUCONTRATO) AS QTD_PARCELAS_ABERTO,
		CASE WHEN DATA_PAGAMENTO IS NULL THEN 'SEM PG'
			 WHEN CALCULATED QTD_PARC_PAGAS < CALCULATED QTD_PARC_ATRASO THEN 'PMT'
			 WHEN CALCULATED QTD_PARC_PAGAS = CALCULATED QTD_PARCELAS_ABERTO THEN 'QUITACAO'
			 ELSE 'ATUALIZACAO' END AS TIPO_PAGAMENTO

FROM BASE_UNICA_LEFT
WHERE DTULTIMORECEBIMENTO >= DATA_PAGAMENTO OR DTULTIMORECEBIMENTO IS NULL
GROUP BY NUCONTRATO, SQCONTRATOFINANCEIRO, DATA_PAGAMENTO

;QUIT;
