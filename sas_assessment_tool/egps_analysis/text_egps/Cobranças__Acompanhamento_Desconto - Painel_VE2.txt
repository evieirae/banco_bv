CREATE TABLE CONSULTA_RECEB_FLEX AS
SELECT DISTINCT
B.*,
DATEPART(B.DTRECEBIMENTO) FORMAT DATE9. AS DTRECEBIMENTO2,
DATEPART(B.DTPROCESSAMENTO) FORMAT DATE9. AS DTPROCESSAMENTO2,
DATEPART(B.DTVENCIMENTO) FORMAT DATE9. AS DTVENCIMENTO2,
'FLEX' AS SISTEMA

FROM CONNECTION TO dbcon(
    SELECT DISTINCT
        A.NUCONTRATO,
        A.SQCONTRATOFINANCEIRO,
        CONVERT(DATE,A.DTRECEBIMENTO) AS DTRECEBIMENTO,
        CONVERT(DATE,A.DTPROCESSAMENTO) AS DTPROCESSAMENTO,
        B.DTVENCIMENTO,
        A.NUPARCELA,
        A.VRRECEBIDO,
		A.VRPRINCIPALDEVIDO,
        B.VRPARCELA,
        E.TxJuros,
        E.TxInternaRetorno,
        E.TxComissaoPermanenciaMes,
		A.VRPRINCIPALDEVIDO,
		A.VRDESCONTOJUROSFUTUROS,
		A.VRPRINCIPALBAIXADO,
		A.VRCOMISSAOPERMANENCIADEVIDA,
		A.VRMULTADEVIDA,
		A.VRCOMISSAOPERMANENCIABAIXADA,
		A.VRMULTABAIXADA,
        E.TxMulta
    FROM STGTXT_FLEX.TBPARCELARECEBIMENTO AS A
    LEFT JOIN STGTXT_FLEX.TbParcela AS B
        ON A.NUCONTRATO = B.NUCONTRATO
        AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
        AND A.NUPARCELA = B.NUPARCELA
    LEFT JOIN STGTXT_FLEX.TbContratoFinanceiro AS E
        ON A.NUCONTRATO = E.NUCONTRATO
        AND A.SQCONTRATOFINANCEIRO = E.SQCONTRATOFINANCEIRO
    WHERE A.DTPROCESSAMENTO >= '20240701'
) B;
QUIT;

PROC SORT DATA=CONSULTA_RECEB_FLEX OUT=CONSULTA_RECEB_FLEX_2 NODUPKEY;
BY NUCONTRATO NUPARCELA;
RUN;

DATA CONSULTA_RECEB_FLEX_3;
SET CONSULTA_RECEB_FLEX_2;

	TxJuros2 = TxJuros/100;
	TxInternaRetorno2 = TxInternaRetorno/100;

	TxMoraMes = TxComissaoPermanenciaMes/100;
	TxMulta2 = TxMulta/100;

	TxInternaRetorno3 = ((1+TxInternaRetorno2)**(1/360)) - 1;

	Atraso = intck('days', DtVencimento2, DtRecebimento);

	if Atraso > 0 then do;

		MoraEstimada = ROUND(TxMoraMes*VrParcela/30,0.01) * Atraso;
		MultaEstimada = ROUND(TxMulta2*VrParcela, 0.01);
		ParcelaEstimada = VrParcela;
		VlrDevidoTotal = MoraEstimada + MultaEstimada + ParcelaEstimada;
		LiquidacaoAntecipada = 0;

	end; else if Atraso <= 0 then do;

		MoraEstimada = 0;
		MultaEstimada = 0;
		ParcelaEstimada = ROUND(VrParcela/((1+TxInternaRetorno3)**(-Atraso)),0.01);
		VlrDevidoTotal = MoraEstimada + MultaEstimada + ParcelaEstimada;
		LiquidacaoAntecipada = VrParcela - ParcelaEstimada;

	end;

	If VrRecebido > ParcelaEstimada then do;
			DescParcelaEstimado = 0;
			ReceitaEncargosEstimado = VrRecebido - ParcelaEstimada;
	end; else do;

			DescParcelaEstimado = ROUND(VrRecebido - ParcelaEstimada, 0.01);
			ReceitaEncargosEstimado = 0;
	end;

RUN;
