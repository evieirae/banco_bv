PROC SQL;
CREATE TABLE TbNaPistaMetaGr AS
SELECT
	     NuAnoMes
		,CdColaborador
		,CdFilial
		,SUM(VrProducao) AS Meta
FROM basegrc.TbMetaNaoIsonomicaNaPista
WHERE DsLoginN1 IS NOT NULL
AND NuAnoMes >= &safraM0.
GROUP BY  1,2,3
;RUN;

PROC SQL;
CREATE TABLE TbNaPistaMetaGF AS
SELECT
	     NaPista.NuAnoMes
		,Colaborador.CdColaboradorAtivoN2A AS CdColaborador
		,NaPista.CdFilial
		,SUM(NaPista.VrProducao) as Meta
FROM basegrc.TbMetaNaoIsonomicaNaPista NaPista
	LEFT JOIN bases.TbEstruturaComercial Colaborador
		ON NaPista.DsLoginN2A = Colaborador.DsLoginN2A
		AND NaPista.NuAnoMes = Colaborador.NuAnoMes
		AND Colaborador.NmCanalEstruturaComercial = 'Canal Veiculos'
		AND Colaborador.NmGrupoCargo = 'Gerente Filial Veiculos'
		AND Colaborador.NmNivel = 'N2'
WHERE NaPista.DsLoginN1 IS NOT NULL
AND NaPista.NuAnoMes >= &safraM0.
GROUP BY 1,2,3
;RUN;

data basegrc.carteira_meta_bvia;
set TbNaPistaMetaGF
	TbNaPistaMetaGr;
;run;

PROC SQL;
CREATE TABLE ParceiroComercialPorte AS
SELECT
		TbParceiroComercial.CdParceiroComercial
		,TbParceiroComercial.NmParceiroComercial
		,TbParceiroComercial.CdFilialC
		,TbParceiroComercial.NmFilialC
		,TbParceiroComercial.CdAreaEstruturaComercialC
		,TbParceiroComercial.NmAreaEstruturaComercialC
		,TbParceiroComercial.NmSubEstruturaComercialC
		,TbParceiroComercial.NmCanalEstruturaComercialC
		,CASE WHEN TbParceiroComercial.NmAreaEstruturaComercialC = 'Canal Motos' then PortePDV.FlPorteNovoMotos
		else PortePDV.FlPorteA end as FlPorteA
		,TbParceiroComercial.DsLoginN2C
		,TbParceiroComercial.DsLoginN3C
		,TbParceiroComercial.LstSubstituicaoC
		,TbParceiroComercial.PredominanciaProduto
FROM tmp.TbParceiroComercial TbParceiroComercial
	LEFT JOIN basegrc.TbPortepdv PortePDV
		ON TbParceiroComercial.NuCnpjLoja = PortePDV.NuCnpjLoja
		AND PortePDV.NuAno = year(today())
WHERE FlAtivo = 'S'
;run;

proc sql ;
CREATE TABLE parceiroPorteClassificacao AS
SELECT
				CdParceiroComercial
				,coalesce(FlPorteA,'Sem porte informado') as FlPorteA
		                ,PredominanciaProduto

				,"" AS DsClassificacaoA
FROM ParceiroComercialPorte
;run;

proc sql;
CREATE TABLE TbEncartComercial1 AS
SELECT
		  ni.NuAnoMes
		, ni.Produto                  as Produto
		, ''                          as StVeiculo
		, ni.CdColaborador            as CdColaborador
		, ni.CdFilial                 as CdFilial
		, ni.NmAreaEstruturaComercial as NmAreaEstruturaComercial
		, ni.DsLoginN1                as DsLoginN1
		, ni.DsLoginN2A               as DsLoginN2A
		, ni.DsLoginN3A               as DsLoginN3A
		, ni.LstSubstituicao          as LstSubstituicao
		, ni.CdParceiroComercial      as CdParceiroComercial
		, pc.NuCnpjLoja               as NuCnpjLoja

FROM basegrc.TbMetaNaoIsonomicaNaPista ni
	LEFT JOIN bases.tbparceirocomercial pc
		ON pc.CdParceiroComercial = ni.CdParceiroComercial
WHERE ni.nuanomes >= &safraM0.
AND UPCASE(ni.Produto) ='NAPISTA'
ORDER BY ni.NuAnoMes, ni.CdColaborador
;run;

proc sql;
CREATE TABLE TbEncartComercial2 as
SELECT
		  ni.NuAnoMes                 as NuAnoMes
		, ''                          as Produto
		, ''                          as StVeiculo
		, ni.CdColaborador            as CdColaborador
		, ni.CdFilial                 as CdFilial
		, ni.NmAreaEstruturaComercial as NmAreaEstruturaComercial
		, ni.DsLoginN1                as DsLoginN1
		, ni.DsLoginN2A               as DsLoginN2A
		, ni.DsLoginN3A               as DsLoginN3A
		, ni.LstSubstituicao          as LstSubstituicao
		, ni.CdParceiroComercial      as CdParceiroComercial
		, pc.NuCnpjLoja               as NuCnpjLoja
FROM basegrc.CarteiraNaPistaProdutos ni
	LEFT JOIN bases.tbparceirocomercial pc
		ON pc.CdParceiroComercial = ni.CdParceiroComercial
WHERE ni.nuanomes >= &safraM0.
AND ni.CdColaborador is not null
ORDER BY ni.NuAnoMes, ni.CdColaborador
;RUN;
