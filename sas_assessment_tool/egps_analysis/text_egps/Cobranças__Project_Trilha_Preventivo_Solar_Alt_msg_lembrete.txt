PROC SQL;
CONNECT TO SYBASE AS DBCON

(SERVER   = PTASYBCTR
 DATABASE = DBCOR
 USER     = &User_PTASYBCTR.
 PASSWORD = &pw_PTASYBCTR.   );
CREATE TABLE WORK.BASE_ACORDO_GCON AS
SELECT DISTINCT A.*
FROM CONNECTION TO DBCON(
SELECT DISTINCT
  Year(A.DtGeracao)*100+Month(A.DtGeracao) AS SAFRA
, A.DtGeracao
, A.NuContratoGestao
, A.SqContratoFinanceiro
, A.TpEntrada
, A.DtVencimento
, A.StNegociacao
, CASE
				When A.StNegociacao = 'AN' Then 'A Negociar'
				When A.StNegociacao = 'N'  Then 'Negociado'
				When A.StNegociacao = 'PA' Then 'Pendente De Aprovacao'
				When A.StNegociacao = 'A'  Then 'Aprovado Pela Bv'
				When A.StNegociacao = 'R'  Then 'Recusado Pela Bv'
				When A.StNegociacao = 'NA' Then 'Negociacao Em Andamento'
				When A.StNegociacao = 'C'  Then 'Negociacao Cancelada'
				When A.StNegociacao = 'LP' Then 'Liberado Pagamento'
				When A.StNegociacao = 'PG' Then 'Pago'
				When A.StNegociacao = 'PL' Then 'Pagamento Realizado'
				When A.StNegociacao = 'DA' Then 'Devolvido Assessoria'
				When A.StNegociacao = 'CD' Then 'Cheque Devolvido'
				When A.StNegociacao = 'AC' Then 'Cheque Aguardando Compensacao'
             ELSE '99.Verificar' END Status_Boleto
, CASE WHEN A.DtGeracao IS NOT NULL THEN 1 ELSE 0 END AS Flag_Boleto_Gerado
, CASE WHEN A.StNegociacao IN ('PL','LP') THEN 1 ELSE 0 END AS Flag_Boleto_Pago
, A.TpAcordoNegociacao
, A.VrTotParcela
, A.VrTotMora
, A.VrTotMulta
, A.VrTotHonorario
, A.VrTotDespesa
, A.VrBoleto
, A.DtCancelamento
, A.DtStNegociacao
FROM dbcontencioso..TbNegociacaoContrato A
WHERE A.DtGeracao  >= DATEADD(DD, -20, GETDATE())
AND A.DtVencimento <= DATEADD(DD,  20, GETDATE())

AND A.StNegociacao NOT IN ('C','PL','LP','R')
AND A.DtVencimento > DATEADD(DD, -1, GETDATE())
ORDER BY A.DtGeracao ASC, A.NuContratoGestao) AS A
;QUIT;

PROC SQL;
CREATE TABLE WORK.BASE_GERAL_PREVSOL_ AS
SELECT DISTINCT
    A.*
, CASE WHEN A.NUCONTRATO = SMS.NUCONTRATO 		THEN 1 ELSE 0 END AS FLAG_SMS
, 0 AS FLAG_FLASH

, 0 AS FLAG_EMAIL

, CASE WHEN A.NUCONTRATO = PUSH.NUCONTRATO 		THEN 1 ELSE 0 END AS FLAG_PUSH
, CASE WHEN A.NUCONTRATO = WHATS.NUCONTRATO 	THEN 1 ELSE 0 END AS FLAG_WHATS
, CASE
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.mil.br%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.jus.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.gov.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.sj.%")    	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.tj.br%")  	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.procon.%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%;%") 		THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') 									THEN 1
	ELSE 0
	END AS FLAG_EMAIL_VALIDO
FROM WORK.BASE_FINAL					AS A
LEFT JOIN WORK.TEMP_SMS_PREVSOL_02 		AS SMS   	ON A.NUCONTRATO = SMS.NUCONTRATO


LEFT JOIN WORK.TEMP_PUSH_PREVSOL_02   	AS PUSH  	ON A.NUCONTRATO = PUSH.NUCONTRATO
LEFT JOIN WORK.TEMP_WHATS_PREVSOL_02 	AS WHATS  	ON A.NUCONTRATO = WHATS.NUCONTRATO
;QUIT;
