/* Chamar macros principais */
%CRIAVARDB;
%SAFRAESG();

options compress=yes;

/* INFORMAR QTD MESES PARA O DASH */
%let qtMeses = 13;

DATA _NULL_;
  /* Define parâmetros com base no dia atual */
  hoje = TODAY();
  hojeXM = INTNX('MONTH', hoje, ((&qtMeses. - 1) * -1), 'BEGINNING');
  anoAnt = INTNX('YEAR', hoje, -1, 'BEGINNING');

  /* Calcula variáveis */
  safraXM = (YEAR(hojeXM) * 100) + MONTH(hojeXM);
  anoXM = YEAR(hojeXM);
  anoAtual = YEAR(hoje);
  dtIniPta = (safraXM * 100) + 1;

  /* Declarar macro variáveis */
  CALL SYMPUT('safraXM', safraXM);
  CALL SYMPUT('anoXM', PUT(anoXM,4.));
  CALL SYMPUT('anoAtual', PUT(anoAtual,4.));
  CALL SYMPUT('dtIniPta', "'"||PUT(dtIniPta,8.)||"'");
RUN;
