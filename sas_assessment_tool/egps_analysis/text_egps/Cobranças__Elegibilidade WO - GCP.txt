proc sql;
CONNECT TO sybaseiq AS dbcon
(host='pta-iqprd01' server=ptaiqprd db=ptaiqcont user=&user. password=&passwd.);
CREATE TABLE acao_contra AS
SELECT DISTINCT
    a.*,
	case
		when c.NuContratoGestao is not Null THEN 1
		ELSE 0
	end AS Fl_Acao_Contra

FROM base_contabil a
LEFT JOIN CONNECTION TO dbcon(
	SELECT DISTINCT
		b.NuContratoGestao,
		b.SqContratoFinanceiro

	FROM DBA.SAJ_PROCESSO b
	WHERE DT_RET_ASSES IS NULL AND b.ATIVA_PRINC='C'
	GROUP BY b.NuContratoGestao, b.SqContratoFinanceiro, B.ATIVA_PRINC
	) c
ON a.NuContrato= c.NuContratoGestao AND a.SqContratoFinanceiro = c.SqContratoFinanceiro
;
QUIT;

PROC SQL;
CONNECT TO sybaseiq AS dbcon
(host='pta-iqprd01' server=ptaiqprd db=ptaiqcont user=&user. password=&passwd.);
CREATE TABLE Marca_reneg AS
SELECT
     a.*,
     c.fl_reneg

FROM acao_contra a
LEFT JOIN CONNECTION TO DBCON(
	SELECT DISTINCT
    	NuContrato,
    	count(NuContrato) AS fl_reneg

	FROM dBA.VwTbContratoFinanceiro where CdMotivoContratoFinanceiro = 4
    GROUP BY NuContrato
	) c
ON a.NuContrato = c.NuContrato;
QUIT;

PROC SQL;
CONNECT TO sybaseiq AS dbcon
(host='pta-iqprd01' server=ptaiqprd db=ptaiqcont user=&user. password=&passwd.);
CREATE TABLE marca_recompra AS
SELECT
     a.*,
     c.fl_recompra
FROM Marca_reneg a
LEFT JOIN CONNECTION TO DBCON(
	SELECT DISTINCT
    	NuContrato,
    	count(NuContrato) AS fl_recompra

	FROM CDCCG.TbContratofinanceiro where CdMotivoContratoFinanceiro = 8
    GROUP BY NuContrato
	) c
ON a.NuContrato = c.NuContrato;
QUIT;

PROC SQL;
CONNECT TO SYBASEIQ (host='pta-iqprd01' server=ptaiqprd db=PTAIQCONT user=&user. password=&passwd.);
CREATE TABLE marca_parcelas AS
SELECT
     a.*,
	 c.Parcelas_pagas,
	 c.Parcelas_abertas,
	 c.Parcelas_Vencidas,
	 c.Qtde_Parc,
	 c.Vrparcela

FROM (SELECT * FROM marca_recompra) a
LEFT JOIN CONNECTION TO SYBASEIQ(
	SELECT DISTINCT
		b.NuContrato,
		b.SqContratoFinanceiro,
	    sum(case when stparcela='Q' then 1 else 0 end) AS Parcelas_pagas,
	    sum(case when stparcela='A' or dtultimorecebimento >= getdate()  then 1 else 0 end) AS Parcelas_abertas,
	    sum(case when stparcela='A' and DtVencimento <= getdate() then 1 else 0 end) AS Parcelas_Vencidas,
		count(nuparcela) as Qtde_Parc,
		max(Vrparcela) as Vrparcela

	FROM DBA.Vwtbparcela b
	GROUP BY NuContrato, SqContratoFinanceiro
	) c
ON a.NuContrato= c.NuContrato AND a.SqContratoFinanceiro = c.SqContratoFinanceiro;
QUIT;

%macro acumular_reneg(dataini,datafim);

PROC SQL;
CONNECT TO SYBASE (SERVER=PTASYBCTR DATABASE=DBCOR user=&user. password=&password.);

CREATE TABLE WORK.RENEG_SIMULADOR AS
SELECT *

FROM CONNECTION TO SYBASE(

     SELECT
          NuContratoGestao,
          SqContratoFinanceiro,
          dt_baixa,
		  ST_RENEG,
          COUNT(SqContratoFinanceiro) as Qtde

          INTO #BASE1
          FROM dbreneg..tb_reneg_simulador

          WHERE dt_baixa >= &dataini. and dt_baixa <= &datafim. and dt_cancelamento is null
          GROUP BY NuContratoGestao, SqContratoFinanceiro, dt_baixa

          SELECT     A.NuContratoGestao,
                     A.SqContratoFinanceiro,
                     A.dt_baixa,
					 A.ST_RENEG,
					 B.VrParcela,
                     DATEDIFF(DAY, MIN(CASE WHEN     B.StParcela <> 'Q' OR
                                                          (B.StParcela = 'Q' AND
                                                          B.DtUltimoRecebimento >= dt_baixa)  THEN DtVencimento ELSE NULL END), dt_baixa) AS Atraso,
                     SUM(CASE WHEN   B.StParcela <> 'Q' OR
                                          (B.StParcela = 'Q' AND
                                          B.DtUltimoRecebimento >= dt_baixa)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
                     COUNT(CASE WHEN B.StParcela <> 'Q' OR
                                          (B.StParcela = 'Q' AND
                                          B.DtUltimoRecebimento >= dt_baixa)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
                     SUM(CASE WHEN   (B.StParcela <> 'Q' OR
                                          (B.StParcela = 'Q' AND
                                          (B.DtUltimoRecebimento) >= dt_baixa)) AND
                                          (B.DtVencimento) < A.dt_baixa THEN VrParcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
                     COUNT(CASE WHEN (B.StParcela <> 'Q' OR
                                          (B.StParcela = 'Q' AND
                                          (B.DtUltimoRecebimento) >= dt_baixa)) AND
                                          (B.DtVencimento) < A.dt_baixa THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
                     MIN(CASE WHEN   B.StParcela <> 'Q' OR
                                                          (B.StParcela = 'Q' AND
                                                          B.DtUltimoRecebimento >= dt_baixa)  THEN DtVencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,
                     MIN(CASE WHEN   B.StParcela <> 'Q' OR
                                                          (B.StParcela = 'Q' AND
                                                          B.DtUltimoRecebimento >= dt_baixa)  THEN Nuparcela ELSE NULL END) as PARCELA

          FROM #BASE1 AS A
                     LEFT JOIN DBGESTAOCDCCG..TbParcela AS B ON     A.NuContratoGestao = B.NuContrato AND
                                                                    A.SqContratoFinanceiro = B.SqContratoFinanceiro
          GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.dt_baixa, B.VrParcela, A.ST_RENEG
     );
QUIT;

DATA ACUMULADO_RENEG;
SET ACUMULADO_RENEG RENEG_SIMULADOR;
RUN;

%mend;
