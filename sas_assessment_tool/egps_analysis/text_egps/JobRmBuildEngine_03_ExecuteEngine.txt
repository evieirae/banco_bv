%include "/projetos/riscosmercado/rmsas/bin/sas/engine/JobRmBuildEngine_00_InicializaMacros.sas";

%let prll_atual=0;
%let _NmJOBSAS=;
%let _CdOrdemAssunto=;
%let _CdParalelo=;
%let _CdMultiData=;
%let _waitfor=;

%macro GeraJobAutomation;
	%if &seq = 1 %then %do;
		%let _NmJOBSAS=&NmJOBSAS;

		%sysExec echo "/*" >> &mainjob;
		%sysExec echo "&dtjob..&hrjob. - JOB AUTOMATION ENGINE - CONTROLE AUTOMÁTICO DE MALHA */" >> &mainjob;
		%sysExec echo "*/" >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo "" >> &mainjob;

		%sysExec echo '%let sufix=&sysparm.;' >> &mainjob;
		%sysExec echo '%let AAAAMMDD=&sufix.;' >> &mainjob;
		%sysExec echo '%let AAAAMM=%sysfunc(substr(&sufix.,1,6));' >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo '%macro ExecutaProcs;' >> &mainjob;

		%let _waitfor=waitfor _all_ ;
	%end;

	%if &CdOrdemAssunto ne &assunto_atual or &CdParalelo ne &prll_atual or &CdMultiData ne &sequencia_atual %then %do;
		%sysExec echo "	&_waitfor.;" >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo "	systask command "\""&cmd_shell. &assunto. &NmJOBSAS. &DTREF. &VlSysparm."\"" taskname=task&seq. status=RC&seq.;" >> &mainjob;

		%let _waitfor=waitfor _all_ task&seq.;
		%let assunto_atual=&CdOrdemAssunto;
		%let prll_atual=&CdParalelo;
		%let sequencia_atual=&CdMultiData;
		%let _NmJOBSAS=&NmJOBSAS;
	%end;
	%else %do;
		%sysExec echo "	systask command "\""&cmd_shell. &assunto. &NmJOBSAS. &DTREF. &VlSysparm."\"" taskname=task&seq. status=RC&seq.;" >> &mainjob;

		%let _waitfor=&_waitfor task&seq ;
	%end;

	%if &seq = &QTD_REG %then %do;
		%sysExec echo "	&_waitfor.;" >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo '%mend ExecutaProcs;' >> &mainjob;
		%sysExec echo "" >> &mainjob;
		%sysExec echo '%ExecutaProcs;' >> &mainjob;
	%end;

%mend GeraJobAutomation;

%macro ExecutaProcs;
	%let bypass_ctrl=&dir_engine./bypass.ctrl;

	%if %sysfunc(fileexist("&bypass_ctrl.")) = 1 %then
		%goto exit_proc;
	%else
		%sysExec echo "JOB &mainjob. EM EXECUÇÃO" >> &bypass_ctrl;

	%if %sysfunc(fileexist("&mainjob.")) = 1 %then %do;
		x "rm &mainjob.";
	%end;

	proc sort data=_RMCAD.LISTA_JOBSAS out=WORK.LISTA_JOBSAS;
		by CdOrdemAssunto CdParalelo CdMultiData;
		WHERE StExecutaJOB="S";
	run;

	proc sql noprint;
		SELECT COUNT(*) AS QTD
		INTO :QTD_REG
		FROM WORK.LISTA_JOBSAS;
	run;

	data _null_;
		set WORK.LISTA_JOBSAS;

		if _N_ eq 1 then do;
			call symputx('prll_atual',CdParalelo,'G');
			call symputx('assunto_atual',CdOrdemAssunto,'G');
			call symputx('sequencia_atual',CdMultiData,'G');
		end;

		call symputx('seq',_N_,'G');
		call symputx('NmAssunto',compress(NmAssunto),'G');
		call symputx('CdOrdemAssunto',CdOrdemAssunto,'G');
		call symputx('CdParalelo',CdParalelo,'G');
		call symputx('CdMultiData',CdMultiData,'G');
		call symputx('VlSysparm',VlSysparm,'G');
		call symputx('NmJOBSAS',NmJOBSAS,'G');
		call symputx('DTREF',DTREF,'G');

		call execute('%GeraJobAutomation;');

	run;

	%if %sysfunc(fileexist("&mainjob.")) = 1 %then %do;
		x "&cmd_shell. engine &nome_job. &AAAAMMDD.";
	%end;

	x "rm &bypass_ctrl.";

%exit_proc:
	%put >>>>> JOB FINALIZADO SEM PROCESSAR.;
	%put >>>>> AGUARDANDO TÉRMINO DE PROCESSAMENTO ANTERIOR...;

%mend ExecutaProcs;

%let periodo_retencao=5;

%macro PurgeOldJobs;
	%let dtpurge=%sysfunc(putn(%sysfunc(datepart(%sysfunc(datetime()))),date9.));

	data _null_;
		call symputx('dtpurge',put(intnx('day',"&dtpurge."d, -&periodo_retencao., 'same'),yymmddn.));
	run;

	%let CHAVE=_99_Automation_;

	%PUT CHAVE=&CHAVE.;
	%PUT dtpurge=&dtpurge.;

	x "rm &dir_cad./lista_oldjobs.lst";
	x "ls -dtr &dir_engine./*&CHAVE.*.sas > &dir_cad./lista_oldjobs.lst";

	data lista_01;
		infile "&dir_cad./lista_oldjobs.lst"
		lrecl = 1000
		firstobs = 1;
		attrib nm_job length = $1000;
		input nm_job;
	run;
	data lista_02;
		set lista_01;
		dtjob=scan(scan(nm_job,-1,"_"),1,".");
		if dtjob < &dtpurge. then output;
	run;
	data _null_;
		set lista_02;

		x_command = "x 'rm " || compress(nm_job) || ";'";

		call symputx('x_command',x_command,'G');
		call execute('%ExecutaXCommand;');
	run;
	x "rm &dir_cad./lista_oldjobs.lst";

%mend PurgeOldJobs;

%ExecutaProcs;
%PurgeOldJobs;
