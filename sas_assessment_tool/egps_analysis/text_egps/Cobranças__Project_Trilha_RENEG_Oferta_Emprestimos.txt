PROC SQL;
CREATE TABLE WORK.BASE_GERAL_OFERTARENEG_ AS
SELECT
    A.*
, CASE WHEN A.NUCONTRATO = SMS.NUCONTRATO 		THEN 1 ELSE 0 END AS FLAG_SMS
, CASE WHEN A.NUCONTRATO = FLASH.NUCONTRATO		THEN 1 ELSE 0 END AS FLAG_FLASH
, CASE WHEN A.NUCONTRATO = EMAIL.NUCONTRATO 	THEN 1 ELSE 0 END AS FLAG_EMAIL
, 0 AS FLAG_PUSH

, CASE
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.mil.br%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.jus.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.gov.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.sj.%")    	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.tj.br%")  	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.procon.%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%;%") 		THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') 									THEN 1
	ELSE 0
	END AS FLAG_EMAIL_VALIDO
FROM WORK.BASE_FINAL				AS A
LEFT JOIN WORK.TEMP_SMS_OFRENEG_02 AS SMS   ON A.NUCONTRATO = SMS.NUCONTRATO
LEFT JOIN WORK.TEMP_FLASH_OFRENEG_02  AS FLASH ON A.NUCONTRATO = FLASH.NUCONTRATO
LEFT JOIN WORK.TEMP_EMAIL_OFRENEG_02  AS EMAIL ON A.NUCONTRATO = EMAIL.NUCONTRATO

;QUIT;


PROC EXPORT
  DATA=WORK.BASE_GERAL_OFERTARENEG_
  OUTFILE="/riscosbv/cobranca/brsouza/reguas/bases_dia/bases_gerais/BASE_GERAL_OFERTARENEG_&DT_ARQ..csv"
  DBMS=csv
  REPLACE;
  PUTNAMES=YES;
 DELIMITER=';';
RUN;

PROC SQL;
CREATE TABLE WORK.RESUMO_OFERTRENEG_ AS
SELECT
  TODAY() Format yymmdd10. as DT_REF
, PUBLICO
, COUNT(DISTINCT NUCONTRATO) AS QTD_CONTR
, SUM(CASE WHEN FLAG_SMS = 1 OR FLAG_FLASH = 1 OR FLAG_EMAIL = 1 AND FLAG_EMAIL_VALIDO = 1 THEN 1 ELSE 0 END) AS QTD_CONTR_ACAO
, SUM(FLAG_SMS) AS SMS
, SUM(FLAG_FLASH) AS FLASH
, SUM( CASE WHEN FLAG_EMAIL = 1 AND FLAG_EMAIL_VALIDO = 1 THEN FLAG_EMAIL ELSE 0 END) AS EMAIL
, SUM(FLAG_PUSH) AS PUSH
FROM WORK.BASE_GERAL_OFERTARENEG_
GROUP BY DT_REF, PUBLICO
;QUIT;
