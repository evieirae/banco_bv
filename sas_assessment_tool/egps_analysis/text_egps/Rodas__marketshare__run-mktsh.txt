PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbDiasUteis AS SELECT * FROM connection to dbcon
(
WITH
TbTempoData AS (
  SELECT
    NuAnoMes,
    CASE WHEN DtTempo BETWEEN '20211227' AND '20211230' THEN NuDiaUtilMes - 1
         WHEN DtTempo IN ('20211224','20211231') THEN -3
         ELSE NuDiaUtilMes END AS NuDiaUtilMes,
    NuSemanaAno,
    DtTempo,
	DdData,
	NuDiaSemana
  FROM DWBI_DIME.TbDimTempoData
),
QtdMes AS (
  SELECT
    NuAnoMes,
    MAX(NuDiaUtilMes) AS NuDiaUtilMes
  FROM TbTempoData
  WHERE
    NuAnoMes >= 201801
    AND DtTempo <= &LstGrav.
  GROUP BY NuAnoMes
),
SemanaInicial AS (
  SELECT DISTINCT
    NuAnoMes,
    MIN(NuSemanaAno) AS PrimeiraSemanaMes,
	MAX(CASE WHEN DdData = 1 AND NuDiaSemana = 1 THEN 1 ELSE 0 END) AS InicioDomingo
  FROM TbTempoData
  WHERE
    NuAnoMes >= 201801
    AND DtTempo <= &LstGrav.
  GROUP BY NuAnoMes
),
CalculoSemana AS (
  SELECT
    (((CASE WHEN DATEPART(weekday, A.DtTempo) <> 1 THEN A.NuSemanaAno ELSE A.NuSemanaAno - 1 END) - B.PrimeiraSemanaMes) +  1) + (CASE WHEN InicioDomingo = 1 THEN 1 ELSE 0 END) AS NuSemana,
    DtTempo
  FROM TbTempoData A
  LEFT JOIN SemanaInicial B
    ON A.NuAnoMes = B.NuAnoMes
  WHERE
    A.NuAnoMes >= 201801
    AND A.DtTempo <= &LstGrav.
),
QtdSemana AS(
  SELECT
    A.NuAnoMes,
    B.NuSemana,
    COUNT(DISTINCT CASE WHEN A.NuDiaUtilMes > 0 THEN A.NuDiaUtilMes END) AS NuDiaUtilSemana
  FROM TbTempoData A
  LEFT JOIN CalculoSemana B
    ON A.DtTempo = B.DtTempo
  WHERE
    A.NuAnoMes >= 201801
    AND A.DtTempo <= &LstGrav.
  GROUP BY A.NuAnoMes,B.NuSemana
)
SELECT DISTINCT
  A.NuAnoMes,
  A.NuDiaUtilMes,
  's' + TRIM(STR(B.NuSemana)) AS NuSemanaMes,
  B.NuDiaUtilSemana
  FROM QtdMes A
  LEFT JOIN QtdSemana B ON
    A.NuAnoMes = B.NuAnoMes
UNION
SELECT DISTINCT
  A.NuAnoMes,
  A.NuDiaUtilMes,
  'mf' AS NuSemanaMes,
  0 AS NuDiaUtilSemana
  FROM QtdMes A
);
RUN;
