/*****REGIONAL*****/
proc sql;
create table BvExper_Regional as
select distinct
TpNivel,
NuAnoMes,
NmSubEstruturaComercialA as Nome,
DsLoginN3A as dslogin,
NmElegAcelerador as Apto,
FAIXA_PONTOS_TURBO as Cluster,
PONTOS_TURBO as Pontos
from exp_bu.BVEXPER_BASEUNICA_RESUMO_TOTAL
where 1=1
and TpBase = 'Pontos'
/*and NmSubEstruturaComercialA like '%Canal Rodas%'*/
and TpNivel = 'Re'
and NuAnoMes between 202510 and 202512
and NmSubEstruturaComercialA not in ('', 'Canal Solar e Privado')
/*and NmCanalEstruturaComercial = 'Canal Veiculos';*/
;run;
