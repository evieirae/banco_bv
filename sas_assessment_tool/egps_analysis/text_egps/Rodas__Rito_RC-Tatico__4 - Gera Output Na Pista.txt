%include '/infogrc/utils/all-includes.sas';

/* ============ NaPista Rito Comercial Leves ======================= */
/* ================================================================= */

data base_NaPistaRito;
    set tmp.TbNaPistaRitoComercial (where=(nuanomes <= &mindata.));

   	data = input(put((nuAnoMes*100)+1, 8.), yymmdd8.);
    format data date9.;
    ano = year(data);
    trimestre = qtr(data);
    mes = month(data);
run;

proc sql print;
    select distinct cats(ano, trimestre)
    into :trimestres SEPARATED BY ' '
    from base_NaPistaRito
    order by ano desc, trimestre desc;
quit;

%let trim1 = %scan(&trimestres, 1);
%let trim2 = %scan(&trimestres, 2);
%let trim3 = %scan(&trimestres, 3);
%let trim4= %scan(&trimestres, 4);
%let trim_atual = %scan(&trimestres, 1);
%let trim_anterior = %scan(&trimestres, 2);

proc sql;
    create table resumo_trimestre_NaPistaRito as
    select
		"BASE_TRI" AS TP_BASE,
        ano,
		. AS NuAnomes,
		ano*100+trimestre as NuAnoTri,
          t1.NmColaboradorC,
		  t1.DsLoginN1C as DsLoginN1,
          t1.DsLoginN2C as DsLoginN2,
          t1.DsLoginN3C as DsLoginN3,
          t1.LstSubstituicaoC as LstSubstituicao,
          t1.NmCanalEstruturaComercialC,
          t1.NmSubEstruturaComercialC,
          t1.NmAreaEstruturaComercialC,
          t1.NmParceiroComercial,
		  	max(DtAtualizacao) as DtAtualizacao Format ddmmyy10.,
            (SUM(t1.estoque_zero))AS estoque_zero,
            (SUM(t1.lojas_ativas))AS lojas_ativas,
			(SUM(t1.lojas_ativas))AS Qtlojas_ativas,
            (SUM(t1.QtLojas_Estoque_0_Digital))AS QtLojas_Estoque_0_Digital,
            (SUM(t1.QtLojas_Ativas_Digital))AS QtLojas_Ativas_Digital,
            (SUM(t1.QtLojas_Class_Digital))AS QtLojas_Class_Digital,
			(SUM(t1.qtd_classificados))AS Qtd_classificados

      FROM WORK.base_NaPistaRito t1
	   where cats(t1.ano, t1.trimestre) in ("&trim1", "&trim2", "&trim3")
	  group by t1.ano,
				t1.ano*100+t1.trimestre,
				/*		put(ano*100+trimestre,best.)||"t",*/
				t1.trimestre,
				t1.NmColaboradorC,
			    t1.DsLoginN1C,
				t1.DsLoginN2C,
				t1.DsLoginN3C,
				t1.LstSubstituicaoC,
				t1.NmCanalEstruturaComercialC,
				t1.NmSubEstruturaComercialC,
				t1.NmAreaEstruturaComercialC,
				t1.NmParceiroComercial;
QUIT;

proc sql;
    create table resumo_mes_NaPistaRito as
	select
		"BASE_MES" AS TP_BASE,
        t1.ano,
		t1.NuAnomes,
		t1.ano*100+t1.trimestre as NuAnoTri,
	    t1.trimestre,
		t1.NmColaboradorC,
		 t1.DsLoginN1C as DsLoginN1,
          t1.DsLoginN2C as DsLoginN2,
          t1.DsLoginN3C as DsLoginN3,
          t1.LstSubstituicaoC as LstSubstituicao,
          t1.NmCanalEstruturaComercialC,
          t1.NmSubEstruturaComercialC,
          t1.NmAreaEstruturaComercialC,
          t1.NmParceiroComercial,
		  max(DtAtualizacao) as DtAtualizacao Format ddmmyy10.,
            (SUM(t1.estoque_zero))AS estoque_zero,
            (SUM(t1.lojas_ativas))AS lojas_ativas,
			(SUM(t1.lojas_ativas))AS Qtlojas_ativas,
            (SUM(t1.QtLojas_Estoque_0_Digital))AS QtLojas_Estoque_0_Digital,
            (SUM(t1.QtLojas_Ativas_Digital))AS QtLojas_Ativas_Digital,
            (SUM(t1.QtLojas_Class_Digital))AS QtLojas_Class_Digital,
			(SUM(t1.qtd_classificados))AS Qtd_classificados
      FROM WORK.base_NaPistaRito t1
	  	 where cats(ano, trimestre) in ("&trim1", "&trim2", "&trim3")
	  group by
			t1.ano,
			t1.NuAnomes,
			t1.ano*100+t1.trimestre,
			t1.trimestre,
			t1.NmColaboradorC,
			t1.DsLoginN1C,
			t1.DsLoginN2C,
			t1.DsLoginN3C,
			t1.LstSubstituicaoC,
			t1.NmCanalEstruturaComercialC,
			t1.NmSubEstruturaComercialC,
			t1.NmAreaEstruturaComercialC,
			t1.NmParceiroComercial;
QUIT;

data base_final_RC_Tatico_NaPistaRito;
    set resumo_trimestre_NaPistaRito resumo_mes_NaPistaRito;
run;

%let dir_csv = /riscosbvf/BA/processos/output;

PROC EXPORT

		outfile="&dir_csv./BASE_FINAL_RC_TATICO_NAPISTARITO.csv" REPLACE
   		dbms=dlm
    		replace;
    		delimiter=';';
    		putnames=yes;
run;
