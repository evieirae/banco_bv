		CASE
			WHEN t1.C349EBITDAResFin > 20.0 OR t1.C349EBITDAResFin < 1.0 THEN 15
			WHEN (t1.C349EBITDAResFin > 16.0 AND t1.C349EBITDAResFin <= 20.0) OR (t1.C349EBITDAResFin >= 1.0 AND t1.C349EBITDAResFin <= 1.1) THEN 6
			ELSE 1
		END AS PVAR031_EBITDAResultadoFin,

		CASE
			WHEN t1.CCFCLDespFin > 20.0 OR t1.CCFCLDespFin < 1.0 THEN 15
			WHEN (t1.CCFCLDespFin > 16.0 AND t1.CCFCLDespFin <= 20.0) OR (t1.CCFCLDespFin >= 1.0 AND t1.CCFCLDespFin <= 1.1) THEN 6
			ELSE 1
		END AS PVAR032_FCLResultadoFin,

		CASE
			WHEN t1.CCFCODespFin > 20.0 OR t1.CCFCODespFin < 1.2 THEN 15
			WHEN (t1.CCFCODespFin > 16.0 AND t1.CCFCODespFin <= 20.0) OR (t1.CCFCODespFin >= 1.2 AND t1.CCFCODespFin <= 1.4) THEN 6
			ELSE 1
		END AS PVAR033_FCOResultadoFin,

		CASE
			WHEN t1.VAR034_DefasagemBalanco > 210 AND t1.CCFCODespFin < 390 THEN 11
			WHEN t1.VAR034_DefasagemBalanco > 390 THEN 15
			ELSE 1
		END AS PVAR034_DefasagemBalanco

	FROM DEMONSTRATIVOS_FULL_FINAL AS t1
	ORDER BY t1.CodCliente;
