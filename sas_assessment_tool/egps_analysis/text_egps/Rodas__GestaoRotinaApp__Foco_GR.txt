%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbColaboradoresAtivos as
select *
from connection to dbcon
(SELECT HierarquiaComercial.NmEstruturaComercial
,HierarquiaComercial.NmSubEstruturaComercial
,HierarquiaComercial.NmAreaEstruturaComercial
,Colaborador.CdColaboradorRH as IdMatricula
,HierarquiaComercial.CdColaborador as CdColaboradorBI
,HierarquiaComercial.NmColaborador
,HierarquiaComercial.NmCargoColaborador
,HierarquiaComercial.CdEstadoColaborador
,HierarquiaComercial.NmEstadoColaborador
,HierarquiaComercial.DtNascimento
,HierarquiaComercial.DsLogin as DsLogin
,HierarquiaComercial.DsEmail as DsEmail
,HierarquiaComercial.NmColaboradorImediato
,HierarquiaComercial.DsLoginColaboradorImediato
,HierarquiaComercial.DtAdmissaoColaborador
,year(HierarquiaComercial.DtAdmissaoColaborador)*100 + month(HierarquiaComercial.DtAdmissaoColaborador) as NuAnoMesAdmissao
FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
LEFT JOIN DWBI_DTWH.TbColaborador Colaborador
ON  HierarquiaComercial.CdColaborador = Colaborador.CdPessoa
AND Colaborador.CdSistemaOrigem = 202
AND today() BETWEEN Colaborador.DtInicioVigencia and Colaborador.DtFimVigencia
WHERE today() BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
AND YEAR(HierarquiaComercial.DtInicioVigencia) BETWEEN YEAR(today())-1 AND YEAR(today())
AND HierarquiaComercial.CdEstadoColaborador  in (5)
AND HierarquiaComercial.CdEstruturaComercial = 8
order by HierarquiaComercial.NmSubEstruturaComercial,
HierarquiaComercial.NmAreaEstruturaComercial,
HierarquiaComercial.NmColaborador
)
;
run;
quit;

proc sql;
create table TbFocoGr_V01 as
select distinct

 &SafraM0. as Safra
,dashM1.DsLoginN1
,dashM1.NmAreaEstruturaComercial
,sum(dashM1.VrProducao) as Meta
,sum(dashM1.VrProducaoRealizado) as Producao
,sum(dashM1.VrProducaoRealizado)/sum(dashM1.VrProducao) as Atingmento
,sum(dashM1.VrProducao)-coalesce(sum(dashM1.VrProducaoRealizado),0) as Saldo

from bases.tbmetani_atingmetas_Dash dashM1

	inner join TbColaboradoresAtivos tbcolab on tbcolab.DsLogin = dashM1.DsLoginN1

where 1=1
/*and dashM0.FlPorteA in ('M','G','GG')*/
and dashM1.NuAnoMes =&SafraM1.
and dashM1.produto ='Leves'
and NuAnoMesAdmissao <= &SafraM6.
/*and dashM0.dsloginn1 = 'bruno.dias'*/
/*and NmParceiroComercial='103 - SAVOL VEICULOS'*/
group by 1,2,3
;run;

proc sort data=TbFocoGr_V01  ;by descending saldo ;run;
proc sort data=TbFocoGr_V01  ;by NmAreaEstruturaComercial ;run;

data TbFocoGr_V02;
    set TbFocoGr_V01 ;
    by  NmAreaEstruturaComercial  ;
    retain Top 0;
    if first.NmAreaEstruturaComercial  then Top = 0;
    Top + 1;
   if Top <=5
;run;

proc sql;
create table TbGr_FocoFinal as
select distinct

a.NuAnomes ,
a.DsloginN1,
a.DsloginN2A,
a.DsloginN3A,
case when a.DsLoginn1 = b.DsLoginn1 then 1 else 0 end as Indicado,
case when a.DsLoginn1 = b.DsLoginn1 then 1 else 0 end as Confirmado,
'' as Just
from basegrc.TBMETANAOISONOMICA as a
	left join TbFocoGr_V02 as b
	on a.DsLoginn1 = b.DsLoginn1
where a.nuanomes=&SafraM0.
and a.Produto='Leves'
and a.DsloginN1 <> ''
;run;

proc sql;
	insert into Basegrc.TbFoco_GR
	select * from TbGr_FocoFinal;
quit;

proc sort data=Basegrc.TbFoco_GR  nodupkey; by NuAnomes DsloginN1 ;run;

proc export
data=TbGr_FocoFinal
file="/riscosbvf/BA/processos/output/tbvisitagrfocoinput.csv" replace
dbms=dlm;
delimiter=';';
run;
