dtmesant = intnx('month',today(),-1);

base = year(intnx('month',today(),-1))*100+month(intnx('month',today(),-1));

	when (month(dtmesant) = 1) call symput('mes','Janeiro');

	when (month(dtmesant) = 3) call symput('mes','Março');
	when (month(dtmesant) = 4) call symput('mes','Abril');
	when (month(dtmesant) = 5) call symput('mes','Maio');
	when (month(dtmesant) = 6) call symput('mes','Junho');
	when (month(dtmesant) = 7) call symput('mes','Julho');
	when (month(dtmesant) = 8) call symput('mes','Agosto');
	when (month(dtmesant) = 9) call symput('mes','Setembro');
	when (month(dtmesant) = 10) call symput('mes','Outubro');
	when (month(dtmesant) = 11) call symput('mes','Novembro');

PROC SQL;
CREATE TABLE TbContratosOperadorFeirao AS
SELECT Producao.NuContrato
	,Contratos.NuContratoLegado
	,Producao.DtContrato
	,Producao.DsTipoCanalProducao
	,Producao.NmColaborador
	,Producao.VrProducao format commax18.0
	,Case when Producao.FlBVMais ge 1 then 'Sim' else 'Não' end as FlBvMais
	,Case when Producao.FlCap ge 1 then 'Sim' else 'Não' end as FlCap
	,Case when Producao.FlDemais ge 1 then 'Sim' else 'Não' end as FlDemais

FROM BASES.TbProducao Producao
	LEFT JOIN bases.TbExtContrato Contratos
		ON Producao.NuContrato = Contratos.NuContrato
WHERE Producao.CdFilial = 276
	AND Producao.NuAnoMes = &base.
ORDER BY DtContrato
;RUN;
