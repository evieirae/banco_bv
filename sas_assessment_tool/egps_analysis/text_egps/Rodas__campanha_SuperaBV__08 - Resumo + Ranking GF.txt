PROC SQL;
CREATE TABLE BASE_SUPERABV_RESUMO_GF_CV AS
SELECT

today() as Data format ddmmyy10.,
/*A.NUANOMES,*/
A.NmCanalEstruturaComercial,
A.NmSubEstruturaComercialA,
A.NmAreaEstruturaComercial,
'' AS LstSubstituicao,
'' AS LstColaboradorFocal,
A.DsLoginN3A,
A.DsLoginN2A,

/*SUM(CASE WHEN TpValor = 'VrSPF_VrAP_VrAuto' THEN VrAtingimento ELSE 0 END) AS Perc_Ating_SegurosTotal,*/
/*SUM(CASE WHEN TpValor = 'VrMargemContribuicao' THEN VrAtingimento ELSE 0 END) AS Perc_Ating_McTotal,*/
/*SUM(CASE WHEN TpValor = 'ContaDigital' THEN VrAtingimento ELSE 0 END) AS Perc_Ating_ContaDigital,*/

SUM(A.VrPontosAtingimento) AS VrPontosAtingimento,
SUM(A.VrPontosExtra) AS VrPontosExtra,
CALCULATED VrPontosAtingimento + CALCULATED VrPontosExtra AS VrPontosTotal,
count( distinct A.DsLoginN1) AS Qtd_GR,
B.Med_Qtd_DsLoginN1,

CALCULATED VrPontosTotal / B.Med_Qtd_DsLoginN1 AS VrMedTrim_GF

FROM CSBV.TbSuperaBV A

INNER JOIN WORK.TbEncarteiramentoNI_02 B ON B.DsLoginN2A = A.DsLoginN2A /*JOIN PARA CONSIDERAR APENAS OS GRS COM CARTEIRA DE METAS NI*/

where NmEstadoColaborador in ('Em Atividade Normal', 'Gozando Férias') AND NmCanalEstruturaComercial = 'Canal Veiculos'

GROUP BY 1,2,3,4,5,6,7,8,13

;RUN;

	WHEN A.NmAreaEstruturaComercial = 'Canal de Negócios III' THEN 'renata.fuzzato'
	WHEN A.NmAreaEstruturaComercial = 'Canal de Negócios IV' THEN 'cyntia.pita'

PROC SORT DATA=WORK.BASE_SUPERABV_RESUMO_GF OUT=WORK.BASE_SUPERABV_RESUMO_GF_01;

proc rank data=WORK.BASE_SUPERABV_RESUMO_GF_01 descending out=WORK.BASE_SUPERABV_RESUMO_GF_02 ties=dense;
    var VrPontosTotal;
    ranks Ranking_VrPontosTotal;
    by NmCanalEstruturaComercial NmSubEstruturaComercialA;
