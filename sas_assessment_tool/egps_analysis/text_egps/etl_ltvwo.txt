/*Gera a Base de Vetados*/
proc sql noprint;
	connect to SYBASEIQ (&conex_syb.);
	CREATE TABLE Leasing AS
		SELECT * FROM connection to SYBASEIQ
			(
			Declare @DtCalculo Varchar(8)
			Set @DtCalculo=&ult_dia.
			SELECT DISTINCT
				(Nucontrato),
				FlagTroca=1
			INTO #ER
				FROM lsg.tbobjgarantia
					WHERE DtSubstituicao is not null
						SELECT
							a.*,
							c.NuContrato,
							c.SqContratoFinanceiroAtivo,
							Produto = Case
								when c.CdModalidadeProduto in (1,2,3,7,8,25,26,27,38,41,46,48)     		then 'Leves'
								when c.CdModalidadeProduto in (5,49)   	                            	then 'Motos'
								when c.CdModalidadeProduto in (9,28,47)   		                        then 'Pesados'
								when c.CdModalidadeProduto in (17)		                           		then 'Privado'
								when c.CdModalidadeProduto in (14,18,32,34,37,39,42) 	                then 'Publico'
								when c.CdModalidadeProduto in (16,33,43)                             	then 'INSS'
								when c.CdModalidadeProduto in (4,11,12,13,14,15,50,51,52,53,54,55)    	then 'MC'
								when c.CdModalidadeProduto in (10,19,20,21,22,23,24,29,30,31,44,57)    	then 'CP'
								when c.CdModalidadeProduto in (6,45,56,78,79) and f.cdtipobem in (9,10) 		then 'Leves'
								when c.CdModalidadeProduto in (6,45,56,78,79) and f.cdtipobem in (5,6,7)		then 'Motos'
								when c.CdModalidadeProduto in (6,45,56,78,79) and f.cdtipobem in (1,2,4,8)	then 'Pesados'
							end,
							Origem = 'Leasing'
						FROM dbcont.lsg_risco AS a
							INNER JOIN lsg.tbcontrato AS c
							ON a.contrato=c.NuContratoLegado

							LEFT JOIN lsg.TbObjGarantiaBemVeiculo AS b
							ON c.nucontrato=b.nucontrato

							LEFT JOIN lsg.tbobjgarantia AS e
							ON e.nucontrato=b.nucontrato AND e.SqObjGarantia=b.SqObjGarantia

							LEFT JOIN lsg.tbobjgarantiabem AS f
							ON f.nucontrato=b.nucontrato
							and f.SqObjGarantia=b.SqObjGarantia

							LEFT JOIN #ER AS y
							ON b.NuContrato=y.NuContrato

							LEFT JOIN dba.tbproposta AS l
							ON b.Nucontrato=l.Nucontratogestao

							LEFT JOIN dba.tbpropgarantiamaqequip AS k
							ON l.Nuproposta=k.Nuproposta

						WHERE a.dt_calculo = @DtCalculo

						AND a.faixa in ('HH')
						AND a.carteira IN ('00000001', '00000006', '00000007', '00000008')
						AND a.moeda = 'R'
			)
	;
QUIT;

PROC SORT DATA=Leasing noduprecs;
	BY NuContrato;
RUN;

	create table valor_parc as
		select * from connection to SYBASEIQ
		(

SELECT
			NuContrato,
			SqContratoFinanceiro,
			max(Vrparcela) as Vrparcela,
			count(NuParcela) as Parc_aberto
		 into #ER
			FROM cdccg.tbparcela
				WHERE StParcela = 'A'
					GROUP BY NuContrato, SqContratoFinanceiro

					Insert into #ER
						SELECT
							NuContrato,
							SqContratoFinanceiro,
							max(Vrparcela) as Vrparcela,
							count(NuParcela) as Parc_aberto
						FROM lsg.tbparcela
							WHERE StParcela = 'A'
								GROUP BY NuContrato, SqContratoFinanceiro

								SELECT * FROM #ER

)
	;
QUIT;

/* PEGA A DATA E VALOR DA VENDA*/
proc sql noprint;
	connect to SYBASEIQ (&conex_syb.);
	create table VENDA as
		select * from connection to SYBASEIQ
		(SELECT * FROM DBA.TBVENDA WHERE FlRejeitado <> 'S')
	;
quit;

DATA VENDA2;
	SET VENDA;
	FORMAT DT_VENDA DATE9.;
	WHERE DtEstornoContabil=.;
	DT_VENDA=DATEPART(DTVENDA);
RUN;

PROC SORT DATA=VENDA2;
	BY NuContratoGestao DESCENDING DT_VENDA;
RUN;

PROC SORT DATA=VENDA2 nodupkey out=VENDA3;
	BY NuContratoGestao;
RUN;

proc sql noprint;
	CREATE TABLE VENDA4 AS
		SELECT
			NUCONTRATOGESTAO,
			MAX(DT_VENDA)AS DTVENDA,
			SUM(VRVENDA) AS VRVENDA2
		FROM VENDA3
			GROUP BY NUCONTRATOGESTAO
	;
QUIT;

/*INSERE DATA E VALOR DA VENDA NA BASE*/
proc sql noprint;
	CREATE TABLE LTV_WO_&mes_proces. AS
		SELECT
			a.*,
			b.DTVENDA,
			b.VRVENDA2
		FROM ltv_wo.LTV_WO_&mes_proces. AS a
			LEFT JOIN VENDA4 AS b
			ON a.Nucontrato = b.NucontratoGestao;
QUIT;
