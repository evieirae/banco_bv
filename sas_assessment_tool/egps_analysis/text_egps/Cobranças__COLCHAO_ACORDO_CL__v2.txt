	CY.DTPRIMEIROATRASO AS DT_PRI_ATRASO,

	CY.NUACORDO,
	CY.DT_CRIACAO,
	CY.DT_CLICLOCRIACAO,
	CY.PLANO,
	CASE
		WHEN CY.PLANO	<=4		THEN 'ATE 4 PARCELAS'
		WHEN CY.PLANO	<=7		THEN 'DE 5 A 7 PARCELAS'
		WHEN CY.PLANO	<=12	THEN 'DE 8 A 12 PARCELAS'
		WHEN CY.PLANO	<=16	THEN 'DE 13 A 16 PARCELAS'
		WHEN CY.PLANO	>16		THEN 'ACIMA DE 16 PARCELAS'
	END AS FAIXA_PLANO,
	CY.VLR_ACORDO,
	CY.DTVENCPRIMEIRAPMT,
	CY.DTPRIMEIROATRASO,
	CY.PARCELA_ATU,

		WHEN CY.PARCELA_ATU	<=12	THEN 'DE 8 A 12 PARCELAS'
		WHEN CY.PARCELA_ATU	<=16	THEN 'DE 13 A 16 PARCELAS'
		WHEN CY.PARCELA_ATU	>16		THEN 'ACIMA DE 16 PARCELAS'
	END AS FAIXA_ATUAL,
	CY.DT_VENCIMENTO,
	CASE
		WHEN FSEM.DT_ENVIO_REAL IS NULL THEN CY.DT_VENCIMENTO
		ELSE FSEM.DT_ENVIO_REAL
	END AS DT_VENCIMENTO_UTIL FORMAT DDMMYY10.,
	CASE
		WHEN FSEM.DT_ENVIO_REAL IS NULL AND CY.DT_VENCIMENTO IS NULL THEN 'S'
		WHEN (CASE WHEN FSEM.DT_ENVIO_REAL IS NULL THEN CY.DT_VENCIMENTO ELSE FSEM.DT_ENVIO_REAL END) > DATEPART(&DT_FIM_MES.) THEN 'S'
		ELSE 'N'
	END AS VENC_NO_MES,
	CY.VLR_PARCELA,

	CY.DT_QUEBRAACORDO,
	CASE
		WHEN FSEM1.DT_ENVIO_REAL IS NULL THEN CY.DT_QUEBRAACORDO
		ELSE FSEM1.DT_ENVIO_REAL
	END AS DT_QUEBRAACORDO_UTIL FORMAT DDMMYY10.,
	CY.PMTQUEBRA,
	CY.DTVENCQUEBRA,
	CASE
		WHEN CY.PMTQUEBRA > 0 AND CY.FLQUEBRAMANUAL = '1' 	THEN 'MANUAL'
		WHEN CY.PMTQUEBRA > 0								THEN 'AUTOMATICO'
	END AS FL_QUEBRA,
	CY.DT_QUITACAO,
	CY.DT_PROMESSA
FROM CYBER_ACORDO_3 AS CY LEFT JOIN CARTEIRA_1 AS CART
	ON CART.NUCONTA = CY.NUCONTA
	AND CART.PRODUTO = CY.PRODUTO
	AND CART.ASSESSORIA = CY.NMAGENTE

		ON CY.DT_VENCIMENTO = FSEM.FIM_DE_SEMANA

		ON CY.DT_QUEBRAACORDO = FSEM1.FIM_DE_SEMANA

WHERE CY.DT_CRIACAO < &DT_INICIO_MES. AND CY.CDAGENTE NE 1015

CREATE TABLE BASE_COLCHAO AS
SELECT DISTINCT

	CART.DATA_BASE,
	'ESTOQUE' AS TIPO,
	CART.DTENVIO,
	CART.CHAVE,
	CART.NUCONTA,

	CART.FX_ATRASO,
	CART.CL_STATUS,
	CART.VLR_SALDO_ATUAL,
	CART.DT_PRI_ATRASO,

	CART.NUACORDO,
	CART.DT_CRIACAO,
	CART.DT_CLICLOCRIACAO,
	CART.PLANO,
	DAY(CART.DT_VENCIMENTO) AS DIA_VENC,
	CART.FAIXA_PLANO,
	CART.VLR_ACORDO,
	CART.DTVENCPRIMEIRAPMT,
	CART.DTPRIMEIROATRASO,

	CART.VLR_EM_ABERTO,
	CASE
		WHEN REC.DT_RECEB IS NOT NULL
			AND CART.DT_QUEBRAACORDO IS NOT NULL 	THEN 'QUEBRA_COM_REPACTUACAO'
		WHEN REC.DT_RECEB IS NOT NULL 				THEN 'RECEBIMENTO'
		WHEN CART.DT_QUEBRAACORDO IS NOT NULL 		THEN 'QUEBRA'
		WHEN REC1.DT_RECEB IS NOT NULL 				THEN 'RECEBIMENTO_ANTECIPADO'
		WHEN REC2.DT_RECEB IS NOT NULL 				THEN 'RECEBIMENTO_ANTECIPADO'
		ELSE 'EM ABERTO'
	END AS SITUACAO_ACORDO,

	CART.STATUS_ACORDO,
	CART.PARCELA_ATU,
	CART.FAIXA_ATUAL,
	CART.DT_VENCIMENTO,
	CART.DT_VENCIMENTO_UTIL,
	CART.DU_DT_VENCIMENTO,
	CART.VENC_NO_MES,
	CART.VLR_PARCELA,

	CART.DT_QUEBRAACORDO_UTIL,
	CART.DU_DT_QUEBRAACORDO,
	CART.FL_QUEBRA,
	CART.PMTQUEBRA,
	CART.DTVENCQUEBRA,
	CART.DT_PROMESSA,

	REC.DT_RECEB,
	REC.DU_DT_RECEB,
	REC.QTD_RECEB,
	CASE
		WHEN REC.MAX_PARC_RECEB = CART.PLANO THEN 'S'
		ELSE ''
	END AS QUITADO,
	CASE
		WHEN REC.QTD_RECEB > 1 	THEN 'MAIS DE 1 PARCELA'
		WHEN REC.QTD_RECEB = 1	THEN '1 PARCELA'
	END AS QTD_RECEB_MES,
	CART.DT_QUITACAO
FROM INF_ACORDO_1 AS CART

LEFT JOIN RECEBIMENTO_M0 AS REC
ON CART.NUCONTA = REC.NUCONTA
AND CART.PRODUTO = REC.PRODUTO

ON CART.NUCONTA = REC1.NUCONTA
AND CART.PRODUTO = REC1.PRODUTO

LEFT JOIN RECEBIMENTO_M_2 AS REC2
ON CART.NUCONTA = REC2.NUCONTA
AND CART.PRODUTO = REC2.PRODUTO
ORDER BY
	CART.DATA_BASE
;RUN;

PROC SQL;
CREATE TABLE BASE AS
SELECT
	A.DATA_BASE,
	A.NUCONTA,
	A.PRODUTO,
	A.TIPO,
	A.ASSESSORIA,
	A.CL_STATUS,
	A.FAIXA_AGRUPAMENTO,
	A.DU_DT_VENCIMENTO,
	A.DU_DT_QUEBRAACORDO,
	A.VLR_SALDO_ATUAL,
	A.VLR_PARCELA,
	A.SITUACAO_ACORDO,
	B.DT_RECEB,
	B.DU_DT_RECEB,
	CASE
		WHEN B.DU_DT_RECEB = . THEN .
		WHEN B.DU_DT_RECEB <> A.DU_DT_VENCIMENTO THEN A.DU_DT_VENCIMENTO
		ELSE B.DU_DT_RECEB
	END AS DU_DT_RECEB_AJUSTADO,
	B.VLR_RECEB,
	CASE
		WHEN A.SITUACAO_ACORDO IN ('QUEBRA','QUEBRA_COM_REPACTUACAO') THEN A.VLR_PARCELA
	END AS VLR_QUEBRA
FROM PERF_CL.COLCHAO_CL AS A LEFT JOIN PERF_CL.COLCHAO_CL (WHERE=(TIPO = 'RECEBIMENTO')) AS B
	ON A.DATA_BASE = B.DATA_BASE
	AND A.NUCONTA = B.NUCONTA
	AND A.PRODUTO = B.PRODUTO

WHERE A.DATA_BASE = &SAFRA.
	AND A.TIPO = 'ESTOQUE'
	AND A.ASSESSORIA NOT IN ('SERVICES','VILELA IBANEZ');
QUIT;

TABLES DT_RECEB;

ELSE DT_RECEB = DT_RECEB;

RUN;

	DT_RECEB

PROC SQL;
CREATE TABLE BASE_1 AS
SELECT
	A.DATA_BASE,
	&DT_REF. AS DT_REF FORMAT DDMMYY10.,
	B.DU AS DIA_UTIL,
	'BV' AS ASSESSORIA,
	A.FAIXA_AGRUPAMENTO,
	COUNT(A.NUCONTA) 			AS QTD_ACORDOS_ATIVOS,
	SUM(A.VLR_SALDO_ATUAL) 		AS VLR_TOTAL_EM_ABERTO,
	SUM(A.VLR_PARCELA)			AS VLR_A_RECEBER,
	SUM(C.VLR_RECEB)			AS VLR_RECEBIMENTO,
	SUM(A.VLR_QUEBRA)			AS VLR_QUEBRA
FROM BASE AS A
LEFT JOIN b_geral.DIAS_UTEIS B
				ON
				&DT_REF. = B.DATA
LEFT JOIN BASE AS C
ON A.NUCONTA = C.NUCONTA
AND A.PRODUTO = C.PRODUTO
AND (C.DT_RECEB <= &DT_REF. OR C.DT_RECEB = . )
GROUP BY
	A.DATA_BASE,
	B.DU,
	A.FAIXA_AGRUPAMENTO
UNION
SELECT
	A.DATA_BASE,
	&DT_REF. AS DT_REF FORMAT DDMMYY10.,
	B.DU AS DIA_UTIL,
	A.ASSESSORIA,
	A.FAIXA_AGRUPAMENTO,
	COUNT(A.NUCONTA) 			AS QTD_ACORDOS_ATIVOS,
	SUM(A.VLR_SALDO_ATUAL) 		AS VLR_TOTAL_EM_ABERTO,
	SUM(A.VLR_PARCELA)			AS VLR_A_RECEBER,
	SUM(C.VLR_RECEB)			AS VLR_RECEBIMENTO,
	SUM(A.VLR_QUEBRA)			AS VLR_QUEBRA
FROM BASE AS A
LEFT JOIN B_GERAL.DIAS_UTEIS B
				ON
				&DT_REF. = B.DATA
LEFT JOIN BASE AS C
ON A.NUCONTA = C.NUCONTA
AND A.PRODUTO = C.PRODUTO
AND (C.DT_RECEB <= &DT_REF. OR C.DT_RECEB = . )
GROUP BY
	A.DATA_BASE,
	B.DU,
	A.ASSESSORIA,
	A.FAIXA_AGRUPAMENTO

;RUN;

	%IF &J=1 %THEN
		%DO;

			DATA BASE_CONSOLIDADA;
				SET BASE_1;
			RUN;

		%END;
	%ELSE
		%DO;

			DATA BASE_CONSOLIDADA;
				SET
					BASE_CONSOLIDADA
					BASE_1;
			RUN;
	%END;
	PROC DELETE DATA= BASE_1;RUN;
%END;
%MEND FILEREAD;
