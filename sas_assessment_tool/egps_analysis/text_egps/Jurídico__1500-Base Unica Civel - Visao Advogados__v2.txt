PROC SQL;
CREATE TABLE work.Tbl1500ENCERRAMENTOREABERTURA (COMPRESS=YES) AS

SELECT

	a.Cd_Causa 				AS CdCausa
,	a.SERIAL_IMPORT
,	a.pasta
,	a.NmNatureza
,	a.NmNaturezaUnidadeCustom
,	intnx('day', a.DtCadastradoEvento, 21916) 			as DtCadastradoEvento2C
,	a.NmResponsavelEvento
,	intnx('day', a.DtEvento , 21916) 					as DtEvento2C
,	a.TpEncerramento
,	CASE WHEN NmEvento = 'Encerrar'
				THEN upper(a.NmMotivoEncerramento)
				ELSE upper(r.MotivoReabertura) END AS NmMotivo
,	CASE WHEN NmEvento = 'Encerrar'
				THEN 'ENCERRAMENTO'
				ELSE 'REABERTURA' END AS NmEvento
, year(a.DtCadastradoEvento) * 100 + month(a.DtCadastradoEvento) as DtCadastradoEventoYYMM
,'ReabEnc' 															 as Fonte
,case when a.CdSistema = 'e' then 'Exyon' else 'Projuris' end as NmSistema
,a.CdSistema
,a.cdunidadeexyon

FROM DBEP.TBLUSUARIOENCERRAMENTO as a

	left join DBEP.TBLPROCESSOREABERTURA as R
	ON R.Pasta = A.Pasta
	and R.cdunidadeexyon in ('VAC','CTO','CON','INQ','TRI','COB')
    and R.cdsistema = 'p'

WHERE a.CdUnidadeExyon in ('VAC','CTO','CON','INQ','TRI','COB')
and a.cdsistema = 'p'

;RUN;
