PROC SQL;
	CREATE TABLE tempw.Producao AS

		SELECT distinct
			t2.Dtcontrato,
			t2.NuProposta,
			t2.NmColaboradorC as NmGr,
		    t2.NmGrupoProdutoMeta as Produto,
			t2.DsLoginN1C as GR,
			t2.DsLoginN2C as GF,
			t2.NmFilialC,
			t2.NmAreaEstruturaComercial,
			t2.VrMargem format commax10.2,
			t2.QtSeguros,
			t2.VrSeguros format commax10.2,
			t2.VrBVMais format commax10.2 as VrSeguro_SPF,
			t2.VrCap format commax10.2 as VrSeguro_AP

			FROM MISV.TBPRODUCAO t2
				WHERE year(t2.DtEntradaProposta) >= year(today())-1
					/*AND t2.PcMargemLiquida < 0*/
	AND t2.NmGrupoProdutoMeta = 'Privado';RUN;
