%include '/infogrc/utils/all-includes.sas';

PROC SQL;
  SELECT MAX(DtAtualizacaoCetip) INTO :DtAt FROM TMP.TBPOTENCIALRODAS;
RUN;

PROC SQL;
CREATE TABLE ParticipacaoNoJogo AS
  SELECT
    Safra,
    /* SUBSTR(PUT("&M3."d,ddmmyy10.),4,7)||' a '||SUBSTR(PUT("&M1."d,ddmmyy10.),4,7) AS Safra, */
    CdParceiroComercial,
    Produto,
    MIN(SUM(CASE WHEN Status <> 'NaoPassouBV' THEN QuantidadePgtoLoja END) / SUM(QuantidadePgtoLoja),1) AS ParticipacaoJogo
  FROM
    SOLITMP.TBVISIBILIDADEPOTENCIAL
  WHERE
    Safra = (CASE WHEN DAY(TODAY()) >= 10 THEN &safraM0. ELSE &safraM1. END)
    /* Safra BETWEEN &safraM3. AND &safraM1. */
    AND CdParceiroComercial <> .
  GROUP BY
    1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbProducao AS
  SELECT
    NuAnoMes AS Safra,
    /* SUBSTR(PUT("&M3."d,ddmmyy10.),4,7)||' a '||SUBSTR(PUT("&M1."d,ddmmyy10.),4,7) AS Safra, */
    CdParceiroComercial,
    NmGrupoProdutoMeta AS Produto,
    SUM(FlTxAlterada) / COUNT(DISTINCT NuContrato) AS PA,
    SUM(VpSpread) / SUM(VrProducao) AS Spread,
    SUM(FlCompleto) / SUM(FlCompletoEleg) AS Completo
  FROM
    TMP.TbProducao
  WHERE
    NuAnoMes = (CASE WHEN DAY(TODAY()) >= 10 THEN &safraM0. ELSE &safraM1. END)
    /* NuAnoMes BETWEEN &safraM3. AND &safraM1. */
    AND NmGrupoProdutoMeta IN ('Leves','Motos','Pesados')
   AND DtContrato < &DtAt.
  GROUP BY
    1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbParceiroComercial AS
  SELECT
    NuAnoMes AS Safra,
    /* SUBSTR(PUT("&M3."d,ddmmyy10.),4,7)||' a '||SUBSTR(PUT("&M1."d,ddmmyy10.),4,7) AS Safra, */
    CdParceiroComercial,
    NmGrupoProduto AS Produto,
    SUM(QtContrato) / SUM(QtGravamesMercado) AS ShareBV,
    SUM(QtBvMais) / SUM(QtBVMaisEleg) AS SPF,
    SUM(QtCap) / SUM(QtCapEleg) AS CAP,
    SUM(QtDemais) / SUM(QtDemaisEleg) AS Demais
  FROM
    TMP.TbPotencialRodas
  WHERE
    NuAnoMes = (CASE WHEN DAY(TODAY()) >= 10 THEN &safraM0. ELSE &safraM1. END)
   /* NuAnoMes BETWEEN &safraM3. AND &safraM1. */
    AND CdParceiroComercial <> .
    AND FlAtualizacaoMercado = 1
  GROUP BY
    1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbInfoParceiroComercial AS
  SELECT
    /*PUT(&DtAt.,ddmmyy10.) AS DtAtualizacao,*/
    A.Safra,
    A.CdParceiroComercial,
    A.Produto,
    COALESCE(B.PA,0) AS PA ,
    PUT(COALESCE(B.Spread,0), 10.8) AS Spread,
    PUT(COALESCE(C.ParticipacaoJogo,0), 10.8) AS ParticipacaoJogo,
    PUT(COALESCE(A.ShareBV,0), 10.8) AS ShareBV,
    COALESCE(A.SPF,0) AS SPF ,
    COALESCE(A.CAP,0) AS CAP ,
    COALESCE(B.Completo,0) AS Completo ,
    COALESCE(A.Demais,0) AS Demais
  FROM
    TbParceiroComercial A
  LEFT JOIN
    TbProducao B
    ON A.CdParceiroComercial = B.CdParceiroComercial
    AND A.Produto = B.Produto
  LEFT JOIN
    ParticipacaoNoJogo C
    ON A.CdParceiroComercial = C.CdParceiroComercial
    AND A.Produto = C.Produto;
RUN;
