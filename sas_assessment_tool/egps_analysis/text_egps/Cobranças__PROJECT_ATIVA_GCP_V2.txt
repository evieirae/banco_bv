CREATE TABLE MARCADOS_VULNERAVEL AS

		Y.SqContratoFinanceiro,

	WITH BASE1 AS

				NuContratoGestao,

			SELECT

				A.NuContratoGestao,
				A.SqContratoFinanceiro,

	        ON
				A.NuContratoGestao=B.CD_Contrato

				A.SqContratoFinanceiro=B.SQ_CONTRATO_FINANCEIRO_ATIVO

CREATE TABLE BASE_ATIVA AS
	SELECT DISTINCT
		A.DT_BASE,
		A.NR_CONTRATO AS NUCONTRATO,
		A.SQCONTAT AS SQCONTRATOFINANCEIRO,
		A.NR_CPFCNPJ,
		A.NR_DIAS_ATRASO AS ATRASO,
		A.CD_ASSESSORIA,
		A.VL_DIVIDA AS RISCO_ENVIO,

		A.TPCOB,

		A.PRORROG,
	    CASE WHEN B.CDMODALIDADEPRODUTO IN (1,2,3,6,8,27,38,41,45,56) THEN 'LEVES'
	         WHEN B.CDMODALIDADEPRODUTO IN (5) THEN 'MOTOS'

	         WHEN B.CDMODALIDADEPRODUTO IN (9,28) THEN 'PESADOS'
	         WHEN B.CDMODALIDADEPRODUTO IN (16,33) THEN 'INSS'
	         WHEN B.CDMODALIDADEPRODUTO IN (17) THEN 'PRIVADO'
	         WHEN B.CDMODALIDADEPRODUTO IN (18,32,34,37,39,42,43) THEN 'PUBLICO'
	         WHEN B.CDMODALIDADEPRODUTO IN (19,20,21,22,23,24,29,30,31,44,57) THEN 'CP'
	         WHEN B.CDMODALIDADEPRODUTO IN (4,10,11,12,13,14,15,50,51,52,53,54,55) THEN 'CDC'
	         WHEN B.CDMODALIDADEPRODUTO IN (25,26,46,47,48,49) THEN 'LEASING'
	     ELSE 'OUTROS' END AS DSGRUPO,
		 A.DMQUE
	FROM
		BASE_ATIVA_CYBER A LEFT JOIN BASE_CONTRATOS_1 B
		ON
			A.NR_CONTRATO = B.NR_CONTRATO;
