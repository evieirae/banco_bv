proc sql ;
connect to bigquery as perm (project='bv-safn-dtm' credfile='/projetos/acessos/bq/fn/safn-dtm.json'
schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
Create table POSICAO_CONTABIL as
select * from connection to perm
(

  SELECT
	Nuobjetonegociacao
	,coalesce(cast(pr.CdModalidadeProduto as INT64),-1) as CdModalidadeCatalogoProduto
	,ger.cdbook
	,pc.DtReferencia
	,sum(coalesce(cast(VrSaldoContabilSMTM as FLOAT64),0)) as VrSaldoContabilSMTM

	from bv-pdbd-prd.trusted_posicaoconsolidadavarejo.tbposicaocontabil pc
	left outer join  bv-pdbd-prd.trusted_posicaoconsolidada.tbestruturaproduto as pr
    on ifnull(Pc.CdCnsldrEstruturaProduto,-1) = pr.CdCnsldrEstruturaProduto
	left outer join bv-pdbd-prd.trusted_posicaoconsolidada.tbestruturagerencial as ger
    on pc.CdCnsldrEstruturaGerencial = ger.CdCnsldrEstruturaGerencial
	where pc.DtReferencia = &DtReferencia. and  pc.CdCarteira <> 13

	group by
	Nuobjetonegociacao
	,pr.CdModalidadeProduto
	,cdbook
	,pc.DtReferencia

);
disconnect from perm
;quit;

proc sql;
create table POSICAO_CONTABIL as select
*
,CASE WHEN CdModalidadeCatalogoProduto = -1 OR CdModalidadeCatalogoProduto = .
		OR cdbook = -1 OR cdbook = .
		THEN 'NAO ERA CONSIDERADO' ELSE 'JA ERA CONSIDERADO'
		END AS FLAG_BASE

from POSICAO_CONTABIL
WHERE COALESCE(VRSALDOCONTABILSMTM,0) <> 0
;quit;

PROC SQL;
CREATE TABLE VERIFICA_TOTAL AS SELECT
DTREFERENCIA
,SUM(COALESCE(VRSALDOCONTABILSMTM,0)) format= commax22.2 AS SALDO_CARTEIRA
,SUM(COALESCE(VRSALDOCONTABILSMTM,0)) * ((&MEDIA_CDI. ** &DU.)-1) format= commax22.2 AS CUSTO_CAP
FROM POSICAO_CONTABIL
WHERE FLAG_BASE = 'NAO ERA CONSIDERADO'
GROUP BY
DTREFERENCIA
;QUIT;

PROC SQL;
CREATE TABLE VALOR_CONTRATRO AS SELECT
NUOBJETONEGOCIACAO
,CDMODALIDADECATALOGOPRODUTO
,CDBOOK
,DTREFERENCIA
,COALESCE(VRSALDOCONTABILSMTM,0) AS VRSALDOCONTABILSMTM
,COALESCE(VRSALDOCONTABILSMTM,0) * ((&MEDIA_CDI. ** &DU.)-1) AS CUSTO_CAP
FROM POSICAO_CONTABIL
WHERE FLAG_BASE = 'NAO ERA CONSIDERADO'
GROUP BY
NUOBJETONEGOCIACAO
,CDMODALIDADECATALOGOPRODUTO
,CDBOOK
,DTREFERENCIA
;QUIT;
