PROC SQL;
CONNECT TO SYBASE AS DBCON
(SERVER="PTASYBCTR"
DATABASE="DBCONTENCIOSO"
USER= &User_PTASYBCTR.
PASSWORD= &pw_PTASYBCTR.);
CREATE TABLE WORK.BASE_ACORDO AS
SELECT DISTINCT A.*
FROM CONNECTION TO DBCON(
SELECT DISTINCT
  Year(A.DtGeracao)*100+Month(A.DtGeracao) AS SAFRA
, A.DtGeracao
, A.NuContratoGestao
, A.SqContratoFinanceiro
, A.TpEntrada
, A.DtVencimento
, A.StNegociacao
, CASE
				When A.StNegociacao = 'AN' Then 'A Negociar'
				When A.StNegociacao = 'N'  Then 'Negociado'
				When A.StNegociacao = 'PA' Then 'Pendente De Aprovacao'
				When A.StNegociacao = 'A'  Then 'Aprovado Pela Bv'
				When A.StNegociacao = 'R'  Then 'Recusado Pela Bv'
				When A.StNegociacao = 'NA' Then 'Negociacao Em Andamento'
				When A.StNegociacao = 'C'  Then 'Negociacao Cancelada'
				When A.StNegociacao = 'LP' Then 'Liberado Pagamento'
				When A.StNegociacao = 'PG' Then 'Pago'
				When A.StNegociacao = 'PL' Then 'Pagamento Realizado'
				When A.StNegociacao = 'DA' Then 'Devolvido Assessoria'
				When A.StNegociacao = 'CD' Then 'Cheque Devolvido'
				When A.StNegociacao = 'AC' Then 'Cheque Aguardando Compensacao'
             ELSE '99.Verificar' END Status_Boleto
, A.TpAcordoNegociacao
, A.VrTotParcela
, A.VrTotMora
, A.VrTotMulta
, A.VrTotHonorario
, A.VrTotDespesa
, A.VrBoleto
, A.DtCancelamento
, A.DtStNegociacao
FROM dbcontencioso..TbNegociacaoContrato A
WHERE A.DtGeracao  > DATEADD(DD, -1, GETDATE())
AND A.DtVencimento <= DATEADD(DD,  20, GETDATE())
AND A.DtVencimento > DATEADD(DD, -1, GETDATE())
AND CAST(A.DtGeracao AS DATE) = CAST(Getdate() AS DATE)
AND A.StNegociacao NOT IN ('C','PL','LP','R')) AS A
;QUIT;

PROC SQL;
CREATE TABLE WORK.BASE_ACORDO_TRAT AS
SELECT DISTINCT
  TODAY() Format = yymmdd10. AS DT_REF
, A.SAFRA
, A.DtGeracao
, A.NuContratoGestao
, A.SqContratoFinanceiro
, A.TpEntrada
, C.Canal As Canal_Boleto
, C.Canal_Macro
, A.DtVencimento
, A.StNegociacao
, A.Status_Boleto
, A.TpAcordoNegociacao
, A.VrTotParcela
, A.VrTotMora
, A.VrTotMulta
, A.VrTotHonorario
, A.VrTotDespesa
, A.VrBoleto
, A.DtCancelamento
, A.DtStNegociacao
, intck('DAY', DATEPART(A.DtGeracao), DATEPART(A.DtVencimento), 'C')  AS Intervalo_GerVenc_Bol
, DATEPART(A.DtGeracao) Format yymmdd10. AS Dt_Geracao_Boleto
, DATEPART(A.DtVencimento) Format yymmdd10. AS Dt_Vencimento_Boleto
, HOUR(A.DtGeracao) AS Hr_Geracao
, B.NuCPFCNPJ
, B.ATRASO AS ATRASO_ATUAL
, B.PRI_NOME
, Case
	WHEN C.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) <=  7 THEN 1
	WHEN C.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) >= 20 THEN 1
	ELSE 0 END AS Flag_ForaExped
, CASE WHEN A.NuContratoGestao = B.NuContratoGestao THEN 1 ELSE 0 END AS Flag_Base_Cob
FROM WORK.BASE_ACORDO AS A
LEFT JOIN CONTATOS.AUX_BASE_CYBER AS B ON A.NuContratoGestao = B.NuContratoGestao
LEFT JOIN TCONFIG.CONFIG_TPENTRADA AS C ON A.TpEntrada = C.TpEntrada
