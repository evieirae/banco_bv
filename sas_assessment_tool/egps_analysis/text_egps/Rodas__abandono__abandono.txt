	set TbAbandonoGCPNew;
	anomes = compress(put(year(dt_simulacao_proposta),4.)||put(month(dt_simulacao_proposta),z2.));

	length Class $50.;

	/* Vendedor e Desistência do Cliente */
	if nm_contato_observacao = 'Vendedor' and ds_classificacao = 'Desistência do cliente' then do;
		if ds_motivo_classificacao = 'Quebra de crédito' then Class = 'DC-QC';
		else if ds_motivo_classificacao = 'Valor da parcela' then Class = 'DC-VP';
	end;

	/* Vendedor e Fechou na concorrência */
	if nm_contato_observacao = 'Vendedor' and ds_classificacao = 'Fechou com a concorrência' then do;
		if ds_motivo_classificacao = 'Quebra de crédito' then Class = 'FC-QC';
		else if ds_motivo_classificacao = 'Valor da parcela' then Class = 'FC-VP';
	end;

run;

data leads.abandono;
	set abandono_parcial;
run;

	  a.cd_simulacao_proposta								as _CD_SIMULACAO_PROPOSTA
