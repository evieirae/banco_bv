CREATE TABLE ENTRADA AS
SELECT
	A.CDBEM,
	A.CDCONTRATO,
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	DATEPART(A.DTINCLUSAO) FORMAT DDMMYY10. AS DTINCLUSAO,
	A.SITUACAO,
	A.CDAGENTE,
	B.PATIO,
	A.NUPROCESSO,
	A.CDMODALIDADEPRODUTO,
	A.PRODUTO,
    A.NUPLACA,
    A.NUCHASSI,
	A.MARCA,
	A.MODELO,
	INPUT(A.ANO_FABRICACAO,4.) AS ANO_FABRICACAO,
	INPUT(A.ANO_MODELO,4.) AS ANO_MODELO,
	CASE
		WHEN (&ANO. - (INPUT(A.ANO_MODELO,4.))) < 3  	THEN 'ATÉ 3 ANOS DE USO'
		WHEN (&ANO. - (INPUT(A.ANO_MODELO,4.))) < 6  	THEN 'DE 4 A 6 ANOS DE USO'
		WHEN (&ANO. - (INPUT(A.ANO_MODELO,4.))) < 10  	THEN 'DE 7 A 10 ANOS DE USO'
		WHEN (&ANO. - (INPUT(A.ANO_MODELO,4.))) >= 10  	THEN 'ACIMA DE 10 ANOS DE USO'
		ELSE 'VERIFICAR'
	END AS FAIXA_ANO_USO,
	A.EMPRESA,
	A.VRBNDU
FROM CONNECTION TO SYBASE
					(
					SELECT DISTINCT
						ENT.CDBEM,
						ENT.CDAGENTE,
						ENT.DTINCLUSAO,
						ENT.NUPROCESSO,
						ENT.CDCONTRATO,
						ENT.NUCONTRATOGESTAO,
						ENT.SQCONTRATOFINANCEIRO,
						CTT.CDMODALIDADEPRODUTO,
						CASE
					   		WHEN CTT.CDMODALIDADEPRODUTO IN (1,3,6,7,8,27,25,38,41,45,48,56,26,46,79)  	THEN 'LEVES'
							WHEN CTT.CDMODALIDADEPRODUTO IN (5,49)             							THEN 'MOTOS'
							WHEN CTT.CDMODALIDADEPRODUTO IN (9,28,47)         		   					THEN 'PESADOS'
							WHEN LSG.CDMODALIDADEPRODUTO IN (1,3,6,7,8,27,25,38,41,45,48,56,26,46,79)  	THEN 'LEVES'
							WHEN LSG.CDMODALIDADEPRODUTO IN (5,49)             							THEN 'MOTOS'
							WHEN LSG.CDMODALIDADEPRODUTO IN (9,28,47)         		   					THEN 'PESADOS'

						ELSE 'OUTROS'
					  	END AS PRODUTO,
					  	CASE
							WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))) = 1 THEN '200' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))
						 	WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))) = 2 AND LEFT(CONVERT(VARCHAR,BEM.AAANOMODELO),1) IN ('5','6','7','8','9') THEN '19' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))
						 	WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))) = 2 AND LEFT(CONVERT(VARCHAR,BEM.AAANOMODELO),1) NOT IN ('5','6','7','8','9') THEN '20' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AAANOMODELO)))
							ELSE CONVERT(VARCHAR,BEM.AAANOMODELO)
						END AS ANO_MODELO,
					    BEM.NUPLACA,
					    BEM.NUCHASSI,
						BEM.dsMARCA as MARCA,
						BEM.dsMODELO AS MODELO,
					  	CASE
							WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))) = 1 THEN '200' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))
						 	WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))) = 2 AND LEFT(CONVERT(VARCHAR,BEM.AaAnoFabricacao),1) IN ('5','6','7','8','9') THEN '19' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))
						 	WHEN LEN(LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))) = 2 AND LEFT(CONVERT(VARCHAR,BEM.AaAnoFabricacao),1) NOT IN ('5','6','7','8','9') THEN '20' + LTRIM(RTRIM(CONVERT(VARCHAR,BEM.AaAnoFabricacao)))
							ELSE CONVERT(VARCHAR,BEM.AaAnoFabricacao)
						END AS ANO_FABRICACAO,
						BEM.CDSITUACAO,
						SIT.DSSITUACAO AS SITUACAO,
						CASE
							WHEN EMP.NMRAZAOSOCIAL IS NULL THEN EMP1.NMRAZAOSOCIAL
							ELSE EMP.NMRAZAOSOCIAL
						END AS EMPRESA,
						BEM.VRBNDU
					FROM DBPATIO..TBENTRADA AS ENT
					LEFT JOIN DBGESTAOCDCCG..TbContrato AS CTT
					ON ENT.NUCONTRATOGESTAO = CTT.NUCONTRATO
					LEFT JOIN DBPATIO..TBBEM AS BEM
					ON BEM.CDBEM = ENT.CDBEM
					LEFT JOIN DBPATIO..TBSITUACAO AS SIT
					ON BEM.CDSITUACAO = SIT.CDSITUACAO
					LEFT JOIN DBGESTAOLSG..TbContrato LSG
					ON ENT.NUCONTRATOGESTAO = LSG.NuContrato
					LEFT JOIN DBGESTAOCDCCG..TbContratoFinanceiro AS CTF
					ON ENT.NUCONTRATOGESTAO = CTF.NuContrato
					AND ENT.SQCONTRATOFINANCEIRO = CTF.SQCONTRATOFINANCEIRO
					LEFT JOIN DBCOR..TBVEICULOLEGAL AS EMP
					ON CTF.CDVEICULOLEGAL = EMP.CDVEICULOLEGAL
					LEFT JOIN DBCOR..TBVEICULOLEGAL AS EMP1
					ON LSG.CDVEICULOLEGAL = EMP1.CDVEICULOLEGAL
					WHERE ENT.DTINCLUSAO >= &DT_INICIAL. 	AND ENT.DTINCLUSAO  < &DT_FINAL.
					) A
LEFT JOIN EST_BNDU.AGENTE_PATIO AS B
ON A.CDAGENTE = B.CDAGENTE
;
RUN;

CREATE TABLE ENTRADA AS
SELECT
	&SAFRA. AS SAFRA,
	A.*,
	B.ENTRADA_LIQUIDA,
	B.TPAPREENSAO,
	CASE
		WHEN B.TPAPREENSAO IN (1,2) THEN 'BUSCA'
		WHEN B.TPAPREENSAO = 3 		THEN 'EA_C_QUITACAO'
		WHEN B.TPAPREENSAO = 4 		THEN 'EA_S_QUITACAO'
		WHEN B.TPAPREENSAO = 5 		THEN 'EA_CONVER'
		WHEN B.TPAPREENSAO = 6 		THEN 'ENTREGA_DIRETA'
		WHEN B.TPAPREENSAO = 7 		THEN 'CUMPRIMENTO_DE_OBRIGACAO'
		ELSE 'ENTRADA_MANUAL'
	END AS TIPO_ENTRADA,
	CASE
		WHEN B.TPAPREENSAO IN (1,2) 		THEN 'BUSCA'
		WHEN B.TPAPREENSAO IN (3,4,5,6,7) 	THEN 'ENTREGA'
		ELSE 'MANUAL'
	END AS GRUPO,
	B.RAZAO_NOME,
	CASE
		WHEN B.RAZAO_NOME LIKE '%MEDINA%'                                             	THEN 'MEDINA'
		WHEN B.RAZAO_NOME LIKE '%REALE%'           AND B.RAZAO_NOME LIKE '%PREDITIVO%'  THEN 'PREDITIVO_REALE'
		WHEN B.RAZAO_NOME LIKE '%REALE%'           AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_REALE'
		WHEN B.RAZAO_NOME LIKE '%REALE%'                                       			THEN 'REALE'
		WHEN B.RAZAO_NOME LIKE '%CARVALHO%' 							         		THEN 'CARVALHO'
		WHEN B.RAZAO_NOME LIKE '%SCHULZE%'         AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_SCHULZE'
		WHEN B.RAZAO_NOME LIKE '%SCHULZE%'                                            	THEN 'SCHULZE'
		WHEN B.RAZAO_NOME LIKE '%TOLEDO%'          AND B.RAZAO_NOME LIKE '%PREDITIVO%'  THEN 'PREDITIVO_TOLEDO'
		WHEN B.RAZAO_NOME LIKE '%TERESINA - TOLEDO PIZA ADVOGADOS ASSOCIADOS%' 			THEN 'TOLEDO'
		WHEN B.RAZAO_NOME LIKE '%TOLEDO%'          AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_TOLEDO'
		WHEN B.RAZAO_NOME LIKE '%TOLEDO%' 									 			THEN 'TOLEDO'
		WHEN B.RAZAO_NOME LIKE '%CONT_TOLEDO VISAO UNICA%' 									 			THEN 'TOLEDO'
		WHEN B.RAZAO_NOME LIKE '%PASQUALI%'        AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_PASQUALI'
		WHEN B.RAZAO_NOME LIKE '%PASQUALI%' 									 		THEN 'PASQUALI'
		WHEN B.RAZAO_NOME LIKE '%LONDRINA - ADVOCACIA BELLINATI PEREZ%'        			THEN 'BELLINATI'
		WHEN B.RAZAO_NOME LIKE '%BELLINATI%'       AND B.RAZAO_NOME LIKE '%NA - %'     	THEN 'NA_BELLINATI'
		WHEN B.RAZAO_NOME LIKE '%BELLINATI%' 									 		THEN 'BELLINATI'
		WHEN B.RAZAO_NOME LIKE '%MARCONDES%'       AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_MARCONDES'
		WHEN B.RAZAO_NOME LIKE '%MARCONDES%' 									 		THEN 'MARCONDES'
		WHEN B.RAZAO_NOME LIKE '%RN_ADV%'          AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_RN_ADV'
		WHEN B.RAZAO_NOME LIKE '%RN_ADV%' 									 			THEN 'RN_ADV'
		WHEN B.RAZAO_NOME LIKE '%CONT_RN%' 									 			THEN 'RN_ADV'
		WHEN B.RAZAO_NOME LIKE '%MELHADO%' 									 			THEN 'MELHADO'
		WHEN B.RAZAO_NOME LIKE '%PORTINHO%'        AND B.RAZAO_NOME LIKE '%PREDITIVO%'  THEN 'PREDITIVO_PORTINHO'
		WHEN B.RAZAO_NOME LIKE '%PORTINHO%'        AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_PORTINHO'
		WHEN B.RAZAO_NOME LIKE '%PORTINHO%' 									        THEN 'PORTINHO'
		WHEN B.RAZAO_NOME LIKE '%HERMES%' 									        	THEN 'HERMES'
		WHEN B.RAZAO_NOME LIKE '%LM_ANGELIS%' 								        	THEN 'LM_ANGELIS'
		WHEN B.RAZAO_NOME LIKE '%ANASTASSOPOULOS%' AND B.RAZAO_NOME LIKE '%NA - %'      THEN 'NA_A COSTA'
		WHEN B.RAZAO_NOME LIKE '%ANASTASSOPOULOS%'                                    	THEN 'ACOSTA'
		WHEN B.RAZAO_NOME LIKE '%HOEPERS%'         AND B.RAZAO_NOME LIKE '%NA - %' 	    THEN 'NA_HOEPERS'
		WHEN B.RAZAO_NOME LIKE '%HOEPERS%' 									   			THEN 'HOEPERS'
		WHEN B.RAZAO_NOME LIKE '%TEIXEIRA%'                                     		THEN 'TEIXEIRA'
		WHEN B.RAZAO_NOME LIKE '%MENDES E CUNHA%'                                     	THEN 'MENDES_E_CUNHA'
		WHEN B.RAZAO_NOME LIKE '%MENDES_E_CUNHA%'                                     	THEN 'MENDES_E_CUNHA'
		WHEN B.RAZAO_NOME LIKE '%MANDALITI%' OR B.RAZAO_NOME LIKE '%CONCILIG%'          THEN 'CONCILIG'
		ELSE 'OUTROS'
	END AS ASSESSORIA,
	B.ATRASO,
	CASE
		WHEN B.ATRASO <= 360 THEN '000_0360'
		ELSE '361_999'
	END AS FX_ATRASO,
	CASE
		WHEN B.ATRASO <= 360 THEN 'IPF'
		ELSE 'WO'
	END AS FASE

								SELECT
									ENT.CDBEM,
									ENT.NUPROCESSO,
									ENT.NUCONTRATOGESTAO,
									ENT.SQCONTRATOFINANCEIRO,
									ENT.DTINCLUSAO,
									PRO.TPAPREENSAO,
									ENV.COD_ASSES,
									NAS.RAZAO_NOME,
									MIN(PRC.NUPARCELA) AS NUPARCELA_CDC,
									MIN(LSG.NUPARCELA) AS NUPARCELA_LSG
								INTO #BASE
								FROM DBPATIO..TBENTRADA AS ENT
								LEFT JOIN DBCONTENCIOSO..TBPROCESSOJURIDICO PRO
								ON ENT.NUCONTRATOGESTAO = PRO.NUCONTRATO
								AND ENT.NUPROCESSO = PRO.NUPROCESSO
								LEFT JOIN SCC_FIN..SAJ_PROCESSO ENV
								ON ENV.NUCONTRATOGESTAO = PRO.NUCONTRATO
								AND ENV.NUM_PROCESSO = PRO.NUPROCESSO
									LEFT JOIN SCC_FIN..SAJ_AGENTE NAS
								ON ENV.COD_ASSES = NAS.COD_AGENTE
								LEFT JOIN DBGESTAOCDCCG..TbParcela AS PRC
								ON ENT.NUCONTRATOGESTAO = PRC.NUCONTRATO
								AND ENT.SQCONTRATOFINANCEIRO = PRC.SQCONTRATOFINANCEIRO
								AND (PRC.STPARCELA <> 'Q' OR PRC.STPARCELA = 'Q' AND PRC.DtUltimoRecebimento > ENT.DTINCLUSAO)
								LEFT JOIN DBGESTAOLSG..TbParcela AS LSG
								ON ENT.NUCONTRATOGESTAO = LSG.NUCONTRATO
								AND ENT.SQCONTRATOFINANCEIRO = LSG.SQCONTRATOFINANCEIRO
								AND (LSG.STPARCELA <> 'Q' OR LSG.STPARCELA = 'Q' AND LSG.DtUltimoRecebimento > ENT.DTINCLUSAO)
								WHERE ENT.DTINCLUSAO >= &DT_INICIAL.
								AND ENT.DTINCLUSAO  < &DT_FINAL.
								GROUP BY
									ENT.CDBEM,
									ENT.NUPROCESSO,
									ENT.NUCONTRATOGESTAO,
									ENT.SQCONTRATOFINANCEIRO,
									ENT.DTINCLUSAO,
									PRO.TPAPREENSAO,
									ENV.COD_ASSES,
									NAS.RAZAO_NOME

								SELECT DISTINCT
									A.CDBEM,
									A.NUPROCESSO,
									A.NUCONTRATOGESTAO,
									A.SQCONTRATOFINANCEIRO,
									CONVERT(DATE,A.DTINCLUSAO) AS DT_INCLUSAO,
									A.TPAPREENSAO,
									A.COD_ASSES,
									A.RAZAO_NOME,
									CASE
										WHEN A.NUPARCELA_CDC IS NOT NULL THEN A.NUPARCELA_CDC
										WHEN A.NUPARCELA_LSG IS NOT NULL THEN A.NUPARCELA_LSG
										ELSE NULL
									END AS NUPARCELA,
									CASE
										WHEN PRC.DtVencimento IS NOT NULL THEN PRC.DtVencimento
										WHEN LSG.DtVencimento IS NOT NULL THEN LSG.DtVencimento
										ELSE NULL
									END AS DTVENCIMENTO,
									CASE
										WHEN SAI.CDBEM IS NULL THEN 'S'
										ELSE 'N'
									END AS ENTRADA_LIQUIDA
								INTO #BASE1
								FROM #BASE A
								LEFT JOIN DBGESTAOCDCCG..TbParcela AS PRC
								ON A.NUCONTRATOGESTAO = PRC.NUCONTRATO
								AND A.SQCONTRATOFINANCEIRO = PRC.SQCONTRATOFINANCEIRO
								AND A.NUPARCELA_CDC = PRC.NUPARCELA
								LEFT JOIN DBGESTAOLSG..TbParcela AS LSG
								ON A.NUCONTRATOGESTAO = LSG.NUCONTRATO
								AND A.SQCONTRATOFINANCEIRO = LSG.SQCONTRATOFINANCEIRO
								AND A.NUPARCELA_LSG = LSG.NUPARCELA
								LEFT JOIN DBPATIO..TBSAIDA AS SAI
								ON A.CDBEM = SAI.CDBEM
								AND SAI.DTSAIDA BETWEEN &DT_INICIAL. AND &DT_FINAL.
								AND SAI.CDMOTIVO IN (2,20,21,22)

								SELECT
									*,
									ATRASO =  DATEDIFF(DAY,DTVENCIMENTO,DT_INCLUSAO)
								FROM #BASE1
																)B
ON A.CDBEM = B.CDBEM
AND A.NUPROCESSO = B.NUPROCESSO
AND A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO;
RUN;
