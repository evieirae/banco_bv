proc sql;
create table tbAtingConta as
select
NmSubEstruturaComercialA as Regional
,NmAreaEstruturaComercial as Filial
,NmColaborador as GR
,DsLoginN1 as DsLoginN1C

		when nuanomes = &SafraM5. then 'M5'
		when nuanomes = &SafraM4. then 'M4'
		when nuanomes = &SafraM3. then 'M3'
		when nuanomes = &SafraM2. then 'M2'
		when nuanomes = &SafraM1. then 'M1'
		when nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,((SUM(coalesce(QtNumerador,0)) / SUM(coalesce(QtDenominador,0))) / (sum(VrMetaN1)/sum(VrMetaN1Denominador))) *100  AS AtingMetaAtivaConta
from bases.TbMetaProducaoFinal_Dash
where NuAnoMes >=&SafraM7. AND NUANOMES >= &SafraM7. AND TpValor='ContaDigital'
GROUP BY 1,2,3,4,5
having ((SUM(coalesce(QtNumerador,0)) / SUM(coalesce(QtDenominador,0))) / (sum(VrMetaN1)/sum(VrMetaN1Denominador))) *100 <>.
;run;

    id nuanomes;
    var AtingMetaAtivaConta ;

  case  	when nuanomes = &SafraM7. then 'M7'
		when nuanomes = &SafraM6. then 'M6'
		when nuanomes = &SafraM5. then 'M5'
		when nuanomes = &SafraM4. then 'M4'
		when nuanomes = &SafraM3. then 'M3'
		when nuanomes = &SafraM2. then 'M2'
		when nuanomes = &SafraM1. then 'M1'
		when nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,NmSubEstruturaComercialC as Regional
,NmAreaEstruturaComercialC	as  Filial
/*,NmParceiroComercial as PDV*/
,NmColaboradorC as GR
/*,FxEntradaMinima	*/
,Banco
,Sum(Qtd_APNP_Conco_MesmaLoja) + Sum(Qtd_Sem_Prop_MesmaLoja)  as TotalConcorrencia
from basegrc.ResumoSimulacoesRito
where 1=1
and banco not in ('N/A','BV')
/*and CdParceiroComercial  IN( 7665 ,469165 )*/
/*and FxEntradaMinima = '01 - até 10%'*/
/*and NuAnoMes = 202502*/
group by 1,2,3,4,5
ORDER by 1,2,3,4 ,6 desc
;quit;

PROC SORT DATA=BancosRankingGR ;
BY NuAnoMes GR ;
RUN;

data BancosRanking_TopGR;
    set BancosRankingGR (Where=(TotalConcorrencia > 0));
    by  NuAnoMes GR ;
    retain Top 0;
    if first.GR then Top = 0;
    Top + 1;
	BancoQtd = strip(Banco)||'('||strip(TotalConcorrencia)||')'; /* Nova coluna concatenada */
	TopBanco = 'Banco'||strip(Top)||'_';
run;

PROC SORT DATA=BancosRanking_TopGR ;
BY Regional Filial GR ;
RUN;

/* Transformação dos dados */
proc transpose data=BancosRanking_TopGR out=BancosRankingTranspose_TopGR (drop=_NAME_);
    by  Regional Filial GR ;
    id TopBanco NuAnoMes ;
    var Banco ;
	Where Top <=3;
run;

proc sql;
create table TbPorte_AgentGR as
select distinct
 NmSubEstruturaComercialC as Regional
,NmAreaEstruturaComercialC as Filial
,NmColaboradorC 		as GR
,NmParceiroComercial	as PDV
,case when porte = 'Sem Porte' then 'Sem_Porte' else porte end as Porte
from basegrc.TbRitoFinal
where  1=1
/*and CdParceiroComercial IN( 7665 ,469165 )*/
/*and nuanomes = 202411*/
;run;

proc sql;
create table QtdPDV_AgentGR as
select distinct
 GR
,Regional
,Filial
,Count(distinct PDV) as Qtdporte
from TbPorte_AgentGR
where  1=1
and GR <> ''
/*and CdParceiroComercial IN( 7665 ,469165 )*/
/*and nuanomes = 202411*/
group by 1,2,3
;run;

proc sql;
create table TbPorteQtd_AgentGR as
select distinct
 a.GR
,a.Regional
,a.Filial
,a.Porte
,Count(distinct a.PDV)  / avg(b.Qtdporte)  as PercPort
from TbPorte_AgentGR as a
	left join QtdPDV_AgentGR as b
	on a.GR = b.GR
where  1=1
and a.GR <> ''
/*and CdParceiroComercial IN( 7665 ,469165 )*/
/*and nuanomes = 202411*/
group by 1,2,3,4
;run;

PROC SORT DATA=TbPorteQtd_AgentGR ;
BY  GR Regional  Filial PercPort ;
RUN;

proc transpose data=TbPorteQtd_AgentGR out=TbPorteQtd_AgentGRTransposedGR (drop=_NAME_) prefix=Porte_ ;
    by  GR Regional  Filial;
    id Porte;
    var PercPort ;
run;

PROC SORT DATA=TbPorteQtd_AgentGRTransposedGR ;
BY  Regional Filial GR ;
RUN;
