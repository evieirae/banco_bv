PROC SQL;
	%CONECTADB(server=BI);

	CREATE TABLE simulacoes_00 AS
		SELECT *
		FROM CONNECTION TO DBCON
		(
			SELECT A.cdsimulacaoProposta				AS Cod_Simulacao
					, A.SgSistemaOrigem						AS Sistema_Origem
					, A.DtSimulacaoproposta					AS Dt_Simulacao
					, A.CdgrupoCategoriaVeiculo				AS Cod_Grupo_Categoria_Veiculo
					, A.nucpfcnpj							AS CPF_CNPJ
					, H.FlIsentoTarifaAberturaCrdo
					, H.PcEntradaAjustada
					, H.VrEntrada
					, H.VrProposta
					, D.Nuproposta 							as NuProposta
					, B.VrVeiculoSimulacao
					, B.AaModelo
					, B.FlZeroKM
					, B.CdMolicar
			  		, C.qtparcela
			  		, C.VrPrestacao
			  		, C.VrTotalSeguro
			  		, C.VrTotalCustoOperacao
					, C.PcVariacaoTabelaFnnco
					, C.TxOriginalFinanciamento
				, c.CdSimulacaoPrazofnnco
				, c.TxInicioFinanciamento
				, a.CdParceiroComercial

				FROM STGORA_APRO_simu.TbSimulacaoProposta 				AS A
				LEFT JOIN STGORA_APRO_simu.TbSimulacaoPropostaGarantia	AS B
					ON A.CdSimulacaoProposta = B.CdSimulacaoProposta
				LEFT JOIN
						(
							SELECT * FROM STGORA_APRO_simu.TbSimulacaoPrazoFnnco
								WHERE CdSimulacaoPrazoFnnco IN
									(SELECT MAX(CdSimulacaoPrazoFnnco)
										FROM STGORA_APRO_simu.TbSimulacaoPrazoFnnco
										GROUP BY CdSimulacaoDadofnnco)
						)
		AS C
					ON B.CdSimulacaoDadofnnco	= C.CdSimulacaoDadofnnco
				LEFT JOIN STGORA_APRO_SIMU.TbSimulacaoDadoFnnco AS H
					ON B.CdSimulacaoDadoFnnco = H.CdSimulacaoDadoFnnco
				LEFT JOIN STGORA_APRO_simu.TbSimulacaoPrzoFnncoPrpsa	AS D
					ON C.CdSimulacaoPrazofnnco	= D.CdSimulacaoPrazofnnco

/*		WHERE A.DtSimulacaoproposta >= &dt_simu_bi. and (A.SgSistemaOrigem in ('SMICSTVE', 'MOBIAUTO', 'SIMUMLVL', 'SIMUMCVL', 'SIMUVEIC') */
/*		or (A.SgSistemaOrigem = 'CRSMINTLV' and a.CdParceiroComercial in (400732, 400733, 400734, 434334, 434335)))*/
		WHERE A.DtSimulacaoproposta >= '20240101' and
/*			  A.SgSistemaOrigem in ('SMICSTVE', 'MOBIAUTO', 'SIMUMLVL', 'SIMUMCVL', 'SIMUVEIC','CRSMINTLV','SIMUSCVL','SIMUSCMT','SIMUKVVL','SIMUKVMT','SIMUDVVL','SIMUAPPBV','SIMUBVIVL','SIMUBVIMT','SIMUBVIPD','APPBV') and*/
/*								      ICARROS    MOBIAUTO      MELI	       MCN       SITE_BV    USADOSBR   SÓ_CARRÃO   SÓ_CARRÃO  KARVI Lev  KARVI Motos  DRYVE     APBV         NAPISTA                             APPBV*/
			  A.CdgrupoCategoriaVeiculo in (2) and
D.Nuproposta is not null

		)

;QUIT;

PROC SORT DATA=simulacoes_00
	OUT=simulacoes_01;
	BY DESCENDING NuProposta Dt_Simulacao ;
RUN;

data work.simulacoes_02;
	set work.simulacoes_01;
	BY DESCENDING NuProposta Dt_Simulacao ;

	if first.NuProposta then
		n=1;
	else n+1;
run;

PROC SQL;

CREATE TABLE PropostaComEdicao as

Select *
from simulacoes_02
where n > 1;quit;

PROC SQL;

CREATE TABLE PropostaSemEdicao as

Select *
from simulacoes_02
where NuProposta not in (select distinct NuProposta from PropostaComEdicao);quit;

PROC SQL;

CREATE TABLE PropostaComEdicaoFull as

Select *
from simulacoes_02
where NuProposta not in (select distinct NuProposta from PropostaSemEdicao);quit;
PROC SQL;

CREATE TABLE PropostaComEdicaoFullPriPerna as

Select *
from PropostaComEdicaoFull
where n = 1;

PROC SQL;

CREATE TABLE PropostaComEdicaoFullUltPerna as
Select A.NuProposta,A.FlZeroKM, A.N AS MAXN
from PropostaComEdicaoFull A
JOIN
(Select NuProposta, max(n) as MaxNB
from PropostaComEdicaoFull
group by NuProposta) B
ON A.NuProposta = B.NuProposta
AND A.N = B.MaxNB ;

PROC SQL;

CREATE TABLE PropostaComEdicaoFullUnion as

Select a.*,b.NuProposta as NuProposta_b, b.FlZeroKM as FlZeroKM_b, MaxN as MaxN_b
from PropostaComEdicaoFullPriPerna a
join PropostaComEdicaoFullUltPerna b
on a.NuProposta = b.NuProposta;quit;

proc sql;
Create table TbPropostaEditadaSum as

select
NuProposta,
count(NuProposta) as TtlPropEdit,
sum(MaxN_b) as TtlEdicoes
from PropostaComEdicaoFullUnion
group by NuProposta;quit;
