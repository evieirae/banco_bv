PROC SQL;
CREATE TABLE Convsimu AS
  SELECT * FROM CONVSIMU.CONVERSAO_SIMULACAO_V2
 WHERE
    Canal IN ('CONCESSIONARIAS','MULTIMARCAS')
    AND FlSimuValidaEfet = 1
    AND TpPessoa = 'PF'
    AND SistemaInicial IN ('ACOM','APAR','PPAR','PPAR2')
    AND FlConversao = 1
    AND FlConcorrente = 1
    AND Produto = 'Leves'
	or Canal IN ('NEGOCIOS CENTRALIZADOS')
    AND FlSimuValidaEfet = 1
    AND TpPessoa = 'PF'
    AND FlConversao = 1
    AND FlConcorrente = 1
    AND Produto = 'Leves';

RUN;

/* Busca dados de GF e GR */
PROC SQL;
CREATE TABLE TbColaboradorPDV AS
  SELECT DISTINCT
    CdColaboradorC AS CdColaborador,
    NmColaboradorC,
    CdParceiroComercial,
    NmParceiroComercial,
    LstSubstituicaoC
  FROM BASES.TbParceiroComercial
  WHERE
    CdParceiroComercial <> .
    AND CdColaboradorC <> .;
RUN;

PROC SQL;
CREATE TABLE TbColaborador AS
  SELECT
    A.DsLoginN1 AS CdGR,
    A.DsLoginN2A AS CdGF,
    B.LstSubstituicaoC AS LstFocais,
    B.CdParceiroComercial
  FROM
    BASES.TbEstruturaComercial A
  INNER JOIN
    TbColaboradorPDV B
    ON A.CdColaboradorAtivo = B.CdColaborador
  WHERE
    TODAY() BETWEEN A.DtInicioVigencia AND A.DtFimVigencia
    AND A.DsLoginN1 IS NOT NULL
    AND A.DsLoginN2A IS NOT NULL;
RUN;

PROC SQL;
CREATE TABLE Convsimu002 AS
  SELECT
    A.NuAnoMes,
    A.Banco,
    A.CdParceiroComercial,
    A.NmParceiroComercial,
    A.Regional,
    A.Filial,
    B.CdGF,
    B.CdGR,
    B.LstFocais,
    A.CdSimulacaoInicial,
    CASE WHEN A.NuTelefone <> '()' THEN A.NuTelefone END AS NuTelefone,
    A.StatusSimulacao,
    A.Mensageria,
    A.NuCpfCnpj,
    A.NuProposta,
    A.DtRetornoCetip,
    A.FxIdade,
    A.StPagoLoja
  FROM
    Convsimu A
  LEFT JOIN
    TbColaborador B
    ON A.CdParceiroComercial = B.CdParceiroComercial;
RUN;

DATA Originacao (KEEP=NuProposta FlTxAlterada);
SET TMP.TbOriginacao&a1. - TMP.TbOriginacao&a0.;
  WHERE NuAnoMes >= &safraM13.
    AND NmGrupoProdutoMeta = 'Leves'
    AND DsTipoCanalProducao IN ('CONCESSIONARIAS','MULTIMARCAS','NEGOCIOS CENTRALIZADOS');
RUN;

PROC SQL;
SELECT MAX(DtSimulacaoInicial) INTO :DtAt FROM CONVSIMU.CONVERSAO_SIMULACAO_V2;
RUN;

PROC SQL;
CREATE TABLE ConversaoC6_DadosCliente AS
  SELECT
    A.NuAnoMes,
    A.Banco,
    A.CdParceiroComercial as CdParceiroComercial,
    A.NmParceiroComercial,
    A.Regional,
    A.Filial,
    A.CdGF,
    A.CdGR,
    A.LstFocais,
    A.CdSimulacaoInicial,
    A.StatusSimulacao,
    A.Mensageria,
    A.NuCpfCnpj,
    COALESCE(B.NmPessoa, 'DESCONHECIDO') AS NmPessoa,
    B.NuTelefone1,
    B.NuTelefone2,
    B.NuTelefone3,
    B.DsEmail,
    A.NuProposta,
    C.FlTxAlterada AS FlPA,
    A.DtRetornoCetip as DtInclusaoGravame,
    A.StPagoLoja,
    A.FxIdade,
    PUT(&DtAt.,DATE9.) AS DtAtualizacao
  FROM
    Convsimu002 A
  LEFT JOIN
    bases.TbPessoa B
    ON A.NuCpfCnpj = B.NuCpfCnpj
  LEFT JOIN
    Originacao C
    ON A.NuProposta = C.NuProposta;
RUN;
