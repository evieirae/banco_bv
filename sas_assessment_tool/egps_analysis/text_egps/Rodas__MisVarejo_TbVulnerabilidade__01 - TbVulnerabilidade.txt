PROC SQL;
CREATE TABLE TbVulnerabilidade as
SELECT DISTINCT
	 safra
	,nucpfcnpj
	,risco_cliente
	,risco_instituicao
	,priorizacao
	,Prob1
FROM DVRYATND.tbVulnerabilidade as Vulnerabilidade
WHERE Safra >= &safraM3.
;QUIT;

proc sort data=TbVulnerabilidade nodupkey;by nucpfcnpj;run;

PROC SQL;
CREATE TABLE TbProducao AS
SELECT
	 TbProducao.NuAnoMes
	,TbProducao.NuContrato
	,TbProducao.NmCanalEstruturaComercial
	,TbProducao.NmAreaEstruturaComercial
	,TbProducao.NmSubEstruturaComercialA
	,INPUT(nucpfcnpj,NUM14.) AS nucpfcnpj
	,TbProducao.nucpfcnpj
	,TbProducao.CdParceiroComercial
	,TbProducao.NmParceiroComercial
	,TbProducao.TpPessoa
	,TbProducao.NmGrupoProdutoMeta
	,TbProducao.CdColaborador
	,TbProducao.NmColaborador
	,TbProducao.DsLoginN1
	,TbProducao.DsLoginN2A
	,TbProducao.DsLoginN3A
FROM Bases.TbProducao
;run;

PROC SQL;
CREATE TABLE TbReclamacaoOuvidoria AS
SELECT
	input(substr('Número de contrato'n,1,14),14.) AS NuContrato format 18.
	,SubMotivo
FROM ouv.PBI_BASEOUVIDORIA_
WHERE Motivo IN ('Seguro','Seguros','Lojista/Correspondente')

AND SubMotivo IN ('Venda Casada','Cancelamento','Não contratado','Ausência de pagamento áo vendedor/proprietário do veículo',
				 'Demora na entrega de documentos','Demora na entrega do bem ao Financiado','Entrega de produto diverso do contratado',
				 'Equipamentos entregues, mas não instalados','Má conduta do parceiro','Postura','Recusa Cancelamento do contrato',
				 'Vício na documentação','Vício no produto')

AND calculated NuContrato is not null
AND 'Motivo do Encerramento'N IN ('PROCEDENTE ATENDIDO','PROCEDENTE NÃO ATENDIDO')
AND Canal IN('0800','BACEN','Consumidor.GOV','PROCON AUD','PROCON CIP','Procon Fone','Reclame Aqui')
;RUN;

PROC SQL;
CREATE TABLE TbVulnerabilidadeFinal AS
SELECT
	'Sem Reclamação' As DsBaseReclamacao
	,Vulnerabilidade.safra
	,Put(Vulnerabilidade.nucpfcnpj,z14.) As nucpfcnpj
	,Vulnerabilidade.risco_cliente
	,Vulnerabilidade.risco_instituicao
	,Vulnerabilidade.priorizacao
	,Vulnerabilidade.Prob1
	,TbProducao.NuAnoMes
	,TbProducao.NuContrato
	,TbProducao.NmCanalEstruturaComercial
	,TbProducao.NmAreaEstruturaComercial
	,TbProducao.NmSubEstruturaComercialA
	,Put(TbProducao.nucpfcnpj,z14.) As PNuCpfCnpj
	,TbProducao.CdParceiroComercial
	,TbProducao.NmParceiroComercial
	,TbProducao.TpPessoa
	,TbProducao.NmGrupoProdutoMeta
	,TbProducao.CdColaborador
	,TbProducao.NmColaborador
	,TbProducao.DsLoginN1
	,TbProducao.DsLoginN2A
	,TbProducao.DsLoginN3A

FROM TbVulnerabilidade Vulnerabilidade
	LEFT JOIN TbProducao TbProducao
		ON Vulnerabilidade.nucpfcnpj = TbProducao.nucpfcnpj

	WHERE NuContrato <> .
	AND NmSubEstruturaComercialA <> ''
;RUN;

PROC SQL;
CREATE TABLE TbVulnerabilidadeReclamacao AS
SELECT
	'Reclamação' As DsBaseReclamacao
	,Vulnerabilidade.safra
	,Put(Vulnerabilidade.nucpfcnpj,z14.) As nucpfcnpj
	,Vulnerabilidade.risco_cliente
	,Vulnerabilidade.risco_instituicao
	,Vulnerabilidade.priorizacao
	,Vulnerabilidade.Prob1
	,TbProducao.NuAnoMes
	,TbProducao.NuContrato
	,TbProducao.NmCanalEstruturaComercial
	,TbProducao.NmAreaEstruturaComercial
	,TbProducao.NmSubEstruturaComercialA
	,Put(TbProducao.nucpfcnpj,z14.) As PNuCpfCnpj
	,TbProducao.CdParceiroComercial
	,TbProducao.NmParceiroComercial
	,TbProducao.TpPessoa
	,TbProducao.NmGrupoProdutoMeta
	,TbProducao.CdColaborador
	,TbProducao.NmColaborador
	,TbProducao.DsLoginN1
	,TbProducao.DsLoginN2A
	,TbProducao.DsLoginN3A
	,SubMotivo

FROM TbVulnerabilidade Vulnerabilidade
	LEFT JOIN TbProducao TbProducao
		ON Vulnerabilidade.nucpfcnpj = TbProducao.nucpfcnpj

	LEFT JOIN TbReclamacaoOuvidoria TbReclamacaoOuvidoria
		ON TbProducao.NuContrato = TbReclamacaoOuvidoria.NuContrato

	WHERE TbProducao.NuContrato <> .
	AND TbProducao.NmSubEstruturaComercialA <> ''
;RUN;

data TbVulnerabilidade;
set
	TbVulnerabilidadeFinal
	TbVulnerabilidadeReclamacao
;run;

SELECT * FROM TbVulnerabilidade
