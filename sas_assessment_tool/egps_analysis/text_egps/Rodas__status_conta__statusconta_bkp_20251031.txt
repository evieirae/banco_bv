PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE TbProposta AS
SELECT * FROM connection to dbcon
(
WITH

TbColaborador AS
(
	SELECT DISTINCT
		 Colaborador.CdOperador
		,first_value(Colaborador.CdColaborador) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdColaborador
		,first_value(Colaborador.CdFilial) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdFilial

	FROM DBA.TbColaborador Colaborador
	WHERE Colaborador.CdColaborador > 0
	AND Colaborador.CdOperador > 0
)

,TbProposta AS
(
			SELECT
				 year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) NuAnoMes
				,DATE(Proposta.DtEntradaProposta) AS DtEntradaProposta
				,DATE(Proposta.DtStatusProposta) AS DtStatusProposta
				,Proposta.StProposta
                ,SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0) CdGrupoSubProduto
				,Colaborador.CdColaborador
				,Proposta.NuProposta
				,Proposta.NuContratoGestao
				,CASE WHEN SubProduto.CdProduto = 11 THEN Colaborador.CdFilial ELSE Regiao.CdFilial END AS CdFilial
				,Proposta.CdLoja CdParceiroComercial
				,PessoaSobAnalise.TpPessoa
				,Proposta.CdSubProduto
				,PessoaSobAnalise.NuCpfCnpj

			FROM DBA.TbProposta Proposta

				LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
					ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
					AND PessoaSobAnalise.CdPapel = 1

				LEFT JOIN DBA.TbSubProduto SubProduto
					ON Proposta.CdSubProduto = SubProduto.CdSubProduto

				LEFT JOIN DBA.TbLoja Loja
					ON Proposta.CdLoja = Loja.CdLoja

				LEFT JOIN DBA.TbRegiao Regiao
					ON  Loja.CdRegiao = Regiao.CdRegiao

				LEFT JOIN DBA.TbParceiroComercial ParceiroComercial
					ON Proposta.CdLoja = ParceiroComercial.CdParceiroComercial

				LEFT JOIN TbColaborador Colaborador
					ON  Proposta.CdOperador = Colaborador.CdOperador

				WHERE Proposta.TpProposta <> '3'
					AND DATE(Proposta.DtEntradaProposta) >=  DATEADD(DAY, -30, TODAY())
					AND Proposta.NuProposta <> 0
					AND Proposta.StProposta IN ('A','AP')
)
SELECT
*
FROM TbProposta
);QUIT;

proc sql;
create table TbPropostaDetalhe as
select
	 Proposta.NuProposta
	,Proposta.DtEntradaProposta
	,Proposta.StProposta
	,Proposta.NuCpfCnpj
	,Proposta.TpPessoa
	,CATS(Proposta.CdParceiroComercial,' - ',COMPRESS(ParceiroComercial.NmParceiroComercial,'0D0A'x)) as NmParceiroComercial
	,Colaborador.NmColaborador
	,EstruturaComercial.NmAreaEstruturaComercial
	,EstruturaComercial.NmSubEstruturaComercialA
	,EstruturaComercial.NmCanalEstruturaComercial
	,Colaborador.DsLogin as DsLoginN1
	,EstruturaComercial.DsLoginN2A as DsLoginN2
	,EstruturaComercial.DsLoginN3A as DsLoginN3
	,ColaboradorFocal.LstColaboradorFocal

	,GrupoProdutoMeta.nmgrupoprodutometa
from TbProposta Proposta
		LEFT JOIN tmp.TbParceiroComercial ParceiroComercial
			ON  Proposta.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

		LEFT JOIN bases.TbEstruturaComercial EstruturaComercial
			ON  Proposta.CdFilial = EstruturaComercial.CdFilial
			AND Proposta.NuAnoMes  = EstruturaComercial.NuAnoMes
			AND EstruturaComercial.NmNivel = 'N2'

		LEFT JOIN bases.TbColaborador Colaborador
			ON  Proposta.CdColaborador = Colaborador.CdColaborador

		inner JOIN basegrc.TbGrupoProdutoMeta GrupoProdutoMeta
			ON Proposta.CdGrupoSubProduto = GrupoProdutoMeta.CdGrupoSubProduto
			and GrupoProdutoMeta.NmGrupoProdutoMeta IN ('CVG Pesados', 'CVG Leves', 'Leves', 'Motos','Pesados')

		LEFT JOIN basegrc.TbColaboradorFocal ColaboradorFocal
			on  Proposta.CdFilial = ColaboradorFocal.CdFilial
			and GrupoProdutoMeta.NmGrupoProdutoMeta = ColaboradorFocal.NmGrupoProdutoMeta

;run;

proc sql;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage);
create table StatusConta as
select * from connection to perm(
WITH
  tb1 AS (
  SELECT
    controle.cdcontrolestatuscadastro,
    controle.dtcontrolestatuscadastro,
    controle.dtalteracao,
    controle.cdstatuspreenchimentocadastro,
    status.dsstatuspreenchimentocadastro,
    pessoa.nucpfcnpj,
    pessoa.nmpessoa,
    pessoa.tppessoa,
    ROW_NUMBER() OVER(PARTITION BY pessoa.nucpfcnpj ORDER BY controle.cdcontrolestatuscadastro DESC) ROW
  FROM
    `bv-pdbd-prd.trusted_dbglobal.tbcontrolestatuscadastro` controle
  INNER JOIN
    `bv-pdbd-prd.trusted_dbglobal.tbpessoa` pessoa
  ON
    controle.cdpessoa = pessoa.cdpessoa
  LEFT JOIN `bv-pdbd-prd.trusted_dbglobal.tbstatuspreenchimentocadastro` status
  ON status.cdstatuspreenchimentocadastro = controle.cdstatuspreenchimentocadastro
  WHERE
    TIMESTAMP_TRUNC(dtcontrolestatuscadastro, MONTH) >= TIMESTAMP_TRUNC(TIMESTAMP_SUB(CURRENT_TIMESTAMP(),INTERVAL 30 DAY),MONTH) )
SELECT
  a.cdcontrolestatuscadastro,
  a.dtcontrolestatuscadastro,
  a.dtalteracao dtalteracaofim,
  a.cdstatuspreenchimentocadastro,
  a.dsstatuspreenchimentocadastro,
  a.nucpfcnpj,
  a.nmpessoa,
  a.tppessoa,
  b.dtcontrolestatuscadastro dtcontrolestatuscadastroini
FROM
  tb1 a
LEFT JOIN (
  SELECT
    nucpfcnpj,
    MIN(dtcontrolestatuscadastro) dtcontrolestatuscadastro
  FROM
    tb1
  GROUP BY
    1) b
ON
  a.nucpfcnpj = b.nucpfcnpj
WHERE
  a.ROW = 1
);
disconnect from perm;
run;
