create table COBRANCA_SQCONTRATOFINANCEIRO as
select a.*,
	   c.atraso_Contabil,
	   c.Risco

from CARTEIRA_RODAS_EMP_202505 as a

left join connection to perm (

		SELECT	 NuContrato,
				 max(SqContratoFinanceiro) as SqContratoFinanceiro,
				 max(nudiasatraso) as atraso_Contabil,
				 sum(SAFE_CAST(VrFuturo AS FLOAT64)) as Risco

		FROM `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil`
		where dtprocessamento = '2025-05-31'
		GROUP BY NuContrato) as C

ON a.nucontrato = INPUT(C.NUCONTRATO, 14.)

;
QUIT;
