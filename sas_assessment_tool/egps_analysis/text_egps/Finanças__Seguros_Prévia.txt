proc sql;
create table work.premios as
select
    DtContrato format=DDMMYY10.,
    NmGrupoProdutoMeta as Produto,

    case
        when NmGrupoProdutoMeta='Solar' and NmParceiroComercial='MEU FINANCIAMENTO SOLAR LTDA' then 'MFS'
        when NmGrupoProdutoMeta='Solar' then 'CS'
        else 'NA_Canal'
    end as Canal,

    VrSeguros as PremioTotal,
    VrBVMais as Prestamista,
    VrCap as AP,
    VrPlacaSolar as Equipamento,
    VrCompleto as Auto_Completo,
    sum(VrGarantiaMecanica, VrRCF, VrCasco) as Auto_Parcial,
    VrDFI as Imobiliario,
	VrProducao as Producao
from bases.tbproducao
where DtContrato between &dataSAS_i. and &dataSAS_f.
and NmGrupoProdutoMeta not in ('CP', 'EP')
;quit;

data work.premios_unpivot;
set work.premios;
length SeguroVCS $20 VrPremioSeguro 8;

array seguros[6] Prestamista AP Equipamento Auto_Completo Auto_Parcial Imobiliario;
array nomes[6] $20 _temporary_('Prestamista','AP','Equipamento','Auto Completo','Auto Parcial','Imobiliario');

do i=1 to dim(seguros);
    if seguros[i] > 0 then do;
        SeguroVCS=nomes[i];
        VrPremioSeguro=seguros[i];
        output;
    end;
end;

keep DtContrato Produto Canal SeguroVCS VrPremioSeguro;
run;

data work.premios_iof;
set work.premios_unpivot;
length IOF 8;

select(SeguroVCS);
    when('Prestamista') IOF = VrPremioSeguro* -0.0038;
    when('AP') IOF = VrPremioSeguro* -0.0038;
    when('Equipamento') IOF = VrPremioSeguro* -0.0738;
    when('Auto Completo') IOF = VrPremioSeguro* -0.0738;
    when('Auto Parcial') IOF = VrPremioSeguro* -0.0738;
    otherwise IOF = 0;
end;
PremioLiq = VrPremioSeguro + IOF;
run;

data work.ReceitaBruta;
set work.premios_iof;
length ReceitaBruta ComissaoTotal 8;

select(SeguroVCS);
    when('Prestamista') do;
        ComissaoTotal = 0.723;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    when('AP') do;
        ComissaoTotal = 0.55;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    when('Equipamento') do;
        ComissaoTotal = 0.312;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    when('Auto Completo') do;
        ComissaoTotal = 0.26;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    when('Auto Parcial') do;
        ComissaoTotal = 0.58;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    when('Imobiliario') do;
        ComissaoTotal = 0.20;
        ReceitaBruta = PremioLiq * ComissaoTotal;
    end;
    otherwise do;
        ComissaoTotal = .;
        ReceitaBruta = .;
    end;
end;
run;
