/*
   STEP 04.01
   Conexão ao PTAIQPRD/CONT para buscar dados de Propostas
*/
PROC SQL;
  %CONECTADB(server=PTA);
    CREATE TABLE INFO_SEGUROS AS SELECT * FROM connection TO dbcon (
      SELECT
        a.NuProposta,
        SUM(a.VrTotalSeguro) AS VrTotalSeguro
      FROM (
        SELECT
          NuProposta,
          AVG(VrSeguro) AS VrTotalSeguro
        FROM
          DBCRED.TbPropostaSeguro
        GROUP BY
          NuProposta,
          cdtiposeguro
      ) a
      GROUP BY
        a.NuProposta
    )
;
RUN;
