proc sql;
create table TbPdv_Grav_s_Cont_Meta as
select distinct
/*,input(scan(pdv,1,'-'),10.) as cdpdv*/

/*,compbl(pdv||'Com Meta de ') || put(TbMetaPDV,2.) ||'%' as PDVMeta*/

proc sql;
create table TbRecadoPDV as
select
GR
,sum(case when GravamesMercado_M0 >=1 and GravamesBV_M0 <=0 then GravamesMercado_M0 else 0 end) as GravamesGR_SemContrato
,sum(case when GravamesMercado_M0 >=1 and GravamesBV_M0 <=0 then GravamesMercado_M0 else 0 end)*35000 as VrGravameMercado
,Sum(GravamesMercado_M0) as QtdGravamesCarteira
,sum(GravamesBV_M0) as QtdContratosCarteira
,sum(case when GravamesMercado_M0 >=1 and GravamesBV_M0 <=0 then 1 else 0 end) as QTDPdvsGR_SemContrato
,count(distinct PDV) as QtdPDVsCarteira
,(calculated QTDPdvsGR_SemContrato / calculated QtdPDVsCarteira)*100  as PDVsSemProduçãoBV
,(calculated GravamesGR_SemContrato / calculated QtdGravamesCarteira)*100 as PDVsSemRepresentaTotal
from auxl.BaseAgenteSpacePDV_GCP
where GR<>''
and GR not in (select GR from TbPdv_Grav_s_Cont)
group by GR
having calculated PDVsSemProduçãoBV>=30
;run;
