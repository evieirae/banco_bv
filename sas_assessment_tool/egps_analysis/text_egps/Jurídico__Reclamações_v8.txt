proc sql;
create table WORK.BASE_DEFINE_LIMITES_FINANCEIRA AS
			select DISTINCT
				SINTOMA,
				MOTIVO,
				case when SUBMOTIVO = 'Aumento/Redução - Massivo' then 'Massivo ? Redução' else SUBMOTIVO end as SUBMOTIVO,
				PRODUTO,
				MDY(INPUT(SUBSTR(DATA_ABERTURA,5,2),8.),INPUT(SUBSTR(DATA_ABERTURA,7,2),8.),INPUT(SUBSTR(DATA_ABERTURA,1,4),8.)) AS DATA,
				count(*) as QTDE
			from alertas  
			where weekday(MDY(INPUT(SUBSTR(DATA_ABERTURA,5,2),8.),INPUT(SUBSTR(DATA_ABERTURA,7,2),8.),INPUT(SUBSTR(DATA_ABERTURA,1,4),8.))) = weekday(&DiaBase.)
				and MDY(INPUT(SUBSTR(DATA_ABERTURA,5,2),8.),INPUT(SUBSTR(DATA_ABERTURA,7,2),8.),INPUT(SUBSTR(DATA_ABERTURA,1,4),8.)) between INTNX('DAY', MDY(MONTH(today()),DAY(today()),YEAR(today())),-90) and &DIABASE.
				and sintoma is not null
GROUP BY SINTOMA,case when SUBMOTIVO = 'Aumento/Redução - Massivo' then 'Massivo ? Redução' else SUBMOTIVO end,PRODUTO, MDY(INPUT(SUBSTR(DATA_ABERTURA,5,2),8.),INPUT(SUBSTR(DATA_ABERTURA,7,2),8.),INPUT(SUBSTR(DATA_ABERTURA,1,4),8.));
quit;
proc sql;

set SUBMOTIVO = 'Massivo ? Reducao'
where SUBMOTIVO = 'Aumento/Redução - Massivo';
quit;

proc sql;
create table WORK.LIMITES_ALERTAS as
	select * ,
			CASE WHEN DESVIOPADRAO < 10
				THEN MEDIA + (DESVIOPADRAO*2)
				ELSE MEDIA + (MEDIA * 0.3)
			END AS TRIGGER
	from (
		select	SINTOMA, MOTIVO, SUBMOTIVO, PRODUTO,
				AVG(QTDE) AS MEDIA,
				STD(QTDE) AS DESVIOPADRAO
		from (select * from
			  WORK.BASE_DEFINE_LIMITES_FINANCEIRA
			  where Produto in ('cartão de crédito'
			  				   ,'app banco bv'
							   ,'conta digital'
							   ,'Cartões'
							   ,'Cartões de Crédito'
							   ,'Conta Digital / Cartão Múltiplo'
							   ,'Banco Digital'
							   ,'Cartões de Crédito'
							   ,'Cartões'
							   ,'Conta Digital'
							   ,'cartão de crédito'
							   ,'Banco Digital'))
		GROUP BY SINTOMA, MOTIVO, SUBMOTIVO, PRODUTO) as b
	order by TRIGGER DESC;
quit;
PROC SQL;
CREATE TABLE WORK.DISPARO_ALERTA AS SELECT
A.SINTOMA,
A.MOTIVO,
A.SUBMOTIVO,
A.PRODUTO,
A.DATA,
put(&DIABASE.,date9.) as DataBase,
b.MEDIA,
A.QTDE_DIA,
CASE WHEN B.TRIGGER = . THEN 0 ELSE B.TRIGGER END AS TRIGGER
FROM (
	SELECT
		SINTOMA,
		MOTIVO,
		SUBMOTIVO,
		PRODUTO,
		DATA,
		COUNT(*) AS QTDE_DIA
			FROM (
					SELECT 	*,
							MDY(INPUT(SUBSTR(DATA_ABERTURA,5,2),8.),INPUT(SUBSTR(DATA_ABERTURA,7,2),8.),INPUT(SUBSTR(DATA_ABERTURA,1,4),8.)) AS DATA
					FROM alertas) AS A 
							WHERE DAY(DATA) = DAY(&DIABASE.) AND
							MONTH(DATA) = MONTH(&DIABASE.) AND
							YEAR(DATA) = YEAR(&DIABASE.)
							GROUP BY SINTOMA,MOTIVO,SUBMOTIVO,PRODUTO,A.DATA) AS A
INNER JOIN WORK.LIMITES_ALERTAS B
ON
A.SINTOMA = B.SINTOMA AND
A.MOTIVO = B.MOTIVO AND
A.SUBMOTIVO = B.SUBMOTIVO AND
A.PRODUTO = B.PRODUTO
WHERE A.QTDE_DIA >B.TRIGGER
AND B.TRIGGER > 0;QUIT;

create table WORK.LIMITES_ALERTAS as
	select * ,
			CASE WHEN DESVIOPADRAO < 10
				THEN MEDIA + (DESVIOPADRAO*2)
				ELSE MEDIA + (MEDIA * 0.3)
			END AS TRIGGER
	from (
		select	SINTOMA, MOTIVO, SUBMOTIVO, PRODUTO,
				AVG(QTDE) AS MEDIA,
				STD(QTDE) AS DESVIOPADRAO
		from (select * from
			  WORK.BASE_DEFINE_LIMITES_FINANCEIRA
			  where Produto not in ('cartão de crédito'
			  				   ,'app banco bv'
							   ,'conta digital'

							   ,'Conta Digital / Cartão Múltiplo'
							   ,'Banco Digital'
							   ,'Cartões de Crédito'
							   ,'Cartões'
							   ,'Conta Digital'
							   ,'cartão de crédito'
							   ,'Banco Digital'))
		GROUP BY SINTOMA, MOTIVO, SUBMOTIVO, PRODUTO) as b
	order by TRIGGER DESC;
quit;

CREATE TABLE WORK.CRUZAMENTO_JUR AS
SELECT
	A.*
   ,B.*
FROM OUV.PASSAGEM_PREVIA AS A
INNER JOIN WORK.JURIDICO_TRATADO AS B
ON A.CPF_CNPJ = B.CPF_CNPJ
WHERE B.DT_CADASTRAMENTO BETWEEN
INTNX('DAY',A.DATA_ABERTURA_OV,+1)
AND INTNX('DAY',A.DATA_ABERTURA_OV,+60)
AND A.CPF_CNPJ >0
;QUIT;

PROC SQL;
connect to bigquery as perm (project='bv-saou-dtm' credfile='/projetos/acessos/bq/ou/saou-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=50);
Create table casos as
select	* from connection to perm(
select distinct
    	casenumber as numero_do_caso,
		bvd_case_owner__c,
		regexp_extract(atd_log_atendimento__c, r'Usuário</td><td[^>]*>([^<]+)</td>') AS primeiro_usuario,
		regexp_extract(nomeclientegrupocomercial__c, r'>([^<]+)<') AS nome_cliente
FROM bv-pdbd-prd.trusted_salesforce.case as a
where cast(casenumber as bigint) >= 24513067);
disconnect from perm;
QUIT;

PROC SQL;
CREATE INDEX numero_do_caso on casos(numero_do_caso)
;QUIT;

PROC SQL;
UPDATE ouv.RECLAMACOES_Mar26_NEW2 as A
	set USUARIO =
	  (select max(primeiro_usuario)
	  from WORK.casos B
	  where A.ID_RECLAMACAO = B.numero_do_caso)
	where exists (
	  select 1
	  from WORK.casos B
	  where A.ID_RECLAMACAO = B.numero_do_caso)
and substr(data_abertura,1,4) in ('2025','2026')
and USUARIO is null
;QUIT;

PROC SQL;
connect to bigquery as perm (project='bv-saou-dtm' credfile='/projetos/acessos/bq/ou/saou-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=50);
Create table casos as
select	* from connection to perm(
select distinct
    	casenumber as numero_do_caso,
		bvd_case_owner__c,
		regexp_extract(atd_log_atendimento__c, r'Usuário</td><td[^>]*>([^<]+)</td>') AS primeiro_usuario,
		regexp_extract(nomeclientegrupocomercial__c, r'>([^<]+)<') AS nome_cliente
FROM bv-pdbd-prd.trusted_salesforce.case as a
where cast(casenumber as bigint) >= 24513067);
disconnect from perm;
QUIT;

PROC SQL;
CREATE INDEX numero_do_caso on casos(numero_do_caso)
;QUIT;

PROC SQL;
UPDATE ouv.RECLAMACOES_Mar26_NEW2 as A
	set USUARIO =
	  (select max(primeiro_usuario)
	  from WORK.casos B
	  where A.ID_RECLAMACAO = B.numero_do_caso)
	where exists (
	  select 1
	  from WORK.casos B
	  where A.ID_RECLAMACAO = B.numero_do_caso)
and substr(data_abertura,1,4) in ('2025','2026')
and USUARIO is null
;QUIT;

PROC SQL;

	  select 1
	  from WORK.casos B
	  where A.ID_RECLAMACAO = B.numero_do_caso)

and CLIENTE is null
;QUIT;
