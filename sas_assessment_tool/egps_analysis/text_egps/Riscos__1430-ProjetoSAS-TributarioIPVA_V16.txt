    FROM public.d_provisao_processo

SELECT
    p.pasta                                AS pasta,
    p.serial_import                        AS serial_import,
    a.assunto_custom                       AS assunto,
    n.natureza                             AS natureza,
    nu.natureza_unidade_custom             AS natureza_unidade_custom,

    o.id_processo_objeto,
    o.exercicio,
    CASE
        WHEN p.data_cadastro IS NULL AND p.data_pre_cadastro IS NOT NULL AND p.data_encerramento IS NULL THEN 'PRE CADASTRO'
        WHEN p.data_encerramento IS NOT NULL THEN 'ENCERRADO'
        WHEN p.status_baixa::text = 'B'::text AND p.data_encerramento IS NULL THEN 'BAIXA_PROVISORIA'
        ELSE 'ATIVA'
    END                                     AS status_processo,
    CASE WHEN p.provisionar = 'T'::bpchar THEN 'SIM' ELSE 'NÃO' END AS ds_provisionar,
    p.provisionar,
    lma.metodo_atualizacao                 AS metodo_atualizacao,
    CAST(pc.data_base_atualizacao AS date)       AS data_base_atualizacao,
    CAST(pc.data_base_atualizacao_juros AS date) AS data_base_atualizacao_juros,

    o.data_cadastro                        AS data_cadastro,

    o.juros_inicial                        AS juros_inicial,
    pc.percentual_provavel,
    pc.percentual_possivel,
    100::numeric - (pc.percentual_provavel + pc.percentual_possivel) AS percentual_remoto,
    pc.valor_pedido,
    pc.valor_perda_provavel,
    pc.valor_perda_possivel,
    pc.valor_pedido - (pc.valor_perda_possivel + pc.valor_perda_provavel) AS valor_perda_remoto,
    pc.valor_inss,

    pop.valor_pedido                       AS valor_pedido_provisao,
    pop.valor_pedido_atualizado,
    pop.valor_perda_possivel               AS valor_possivel_provisao,
    pop.valor_perda_possivel_atualizad,
    pop.valor_perda_provavel               AS valor_provavel_provisao,
    pop.valor_perda_provavel_atualizad,
    pop.valor_pedido - (pop.valor_perda_possivel + pop.valor_perda_provavel) AS valor_remoto_provisao,
    pop.valor_pedido_atualizado - (pop.valor_perda_possivel_atualizad + pop.valor_perda_provavel_atualizad) as valor_perda_remoto_atualizad
FROM m_processo p
JOIN d_processo_objeto o
  ON o.id_processo = p.id_processo
JOIN lt_objeto lo
  ON lo.id_objeto = o.id_objeto
JOIN lt_natureza n
  ON n.id_natureza = p.id_natureza
LEFT JOIN lt_assunto_custom a
  ON a.id_assunto_custom = p.id_assunto_custom
LEFT JOIN lt_metodo_atualizacao lma
  ON lma.id_metodo_atualizacao = o.id_metodo_atualizacao

 AND pop.id_processo_objeto = o.id_processo_objeto
JOIN d_processo_objeto_parcela pc
  ON pc.id_processo_objeto = o.id_processo_objeto
LEFT JOIN lt_natureza_unidade_custom nu
  ON nu.id_natureza_unidade_custom = p.natureza_unidade_custom

      where p.data_encerramento IS NULL
      AND n.NATUREZA in ('Tributária')
      AND lo.objeto LIKE '%IPVA%'

);
