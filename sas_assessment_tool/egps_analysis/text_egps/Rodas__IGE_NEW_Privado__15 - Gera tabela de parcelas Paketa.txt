PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.Tb_Parcelas_Paketa AS
	SELECT *
	FROM CONNECTION TO DBCON

			(

		SELECT
		year(parc.dtvencimento)*100+month(parc.dtvencimento) AS anomes_vencimento,
         Pessoa.NmPessoa,
         Parceirocom.NmFantasia,
         parc.NuContrato,
         parc.SqContratoFinanceiro,
         parc.NuParcela,
         parc.VrPrincipal,
         parc.VrJuros,
         parc.VrParcela,
         parc.StParcela,
         cast(parc.DtVencimento as date) as DtVencimento,
         parc.VrParcelaIndice,
         parc.VrTarifaCobranca,
         cast(parc.DtUltimoRecebimento as date) as DtUltimoRecebimento,
         parc.VrUltRecebimento,
         parc.VrResidual,
         parc.VrResidualIndice,
         parc.VrIss,
         parc.VrPis,
         cast(parc.DtProcessamentoUltRecebimento as date) as DtProcessamentoUltRecebimento,
         cast(parc.DtInclusao as date) as DtInclusao,
         cast(parc.DtAlteracao as date) as DtAlteracao,
         parc.CdColaborador,
         parc.TsReplicacao,
         parc.VrJuroPagoParcela,
         parc.VrJuroNaoPagoParcela,
		 CTTFIN.CdTipoCobranca

		FROM CPC.TbParcela parc
/**************RETIRADO MAX SQ CONTRATOFINANCEIRO A PEDIDO DA CLAUDIA LAGE****************/

/*			INNER JOIN */
/*				(select nucontrato, MAX(SqContratoFinanceiro) as mxsqcontrato from CPC.TbParcela GROUP BY nucontrato) parc2*/
/*				on parc.nucontrato = parc2.nucontrato */
/*				and parc.SqContratoFinanceiro = parc2.mxsqcontrato*/

/*******************************************************************************************************************/
			INNER JOIN DBA.TbProposta Proposta ON parc.nucontrato = Proposta.nucontratogestao
/*			INNER JOIN CPC.TbContrato ON Proposta.nucontratogestao = TbContrato.nucontrato AND TbContrato.CdSituacaoContrato = 1*/
				LEFT JOIN DBA.TbParceiroComercial Parceiropessoa ON Parceiropessoa.CdParceiroComercial = Proposta.CdPessoa
				LEFT JOIN DBA.TbParceiroComercial Parceirocom ON Parceirocom.CdParceiroComercial = Proposta.cdloja
				LEFT JOIN DBA.TbPessoa Pessoa ON Pessoa.CdPessoa = Parceiropessoa.CdPessoa
				LEFT JOIN CPC.TbContratoFinanceiro CTTFIN
				ON CTTFIN.NUCONTRATO = Proposta.NUCONTRATOGESTAO AND CTTFIN.SqContratoFinanceiro = parc.SqContratoFinanceiro
					WHERE Proposta.cdsubproduto = 12 AND
						proposta.cdpessoa in (466420,465612) AND
						/*month(parc.dtvencimento) = 5 AND */
						year(parc.dtvencimento) >= 2024
);
RUN;

proc sql;
create table tempw.Tb_Parcelas_Paketa_2 as
Select
year(Parcelas.dtvencimento)*100+month(Parcelas.dtvencimento) AS anomes_vencimento,
         Parcelas.NmPessoa,
         Parcelas.NmFantasia,
         Parcelas.NuContrato,
         Parcelas.SqContratoFinanceiro,
		 Parcelas.CdTipoCobranca,
         Parcelas.NuParcela,
         Parcelas.VrPrincipal format=commax12.2,
         Parcelas.VrJuros format=commax12.2,
         Parcelas.VrParcela format=commax12.2,
         Parcelas.StParcela,
         Parcelas.DtVencimento,
         Parcelas.VrParcelaIndice format=commax12.2,
         Parcelas.VrTarifaCobranca format=commax12.2,
         Parcelas.DtUltimoRecebimento,
         Parcelas.VrUltRecebimento format=commax12.2,
         Parcelas.VrResidual format=commax12.2,
         Parcelas.VrResidualIndice format=commax12.2,
         Parcelas.VrIss format=commax12.2,
         Parcelas.VrPis format=commax12.2,
         Parcelas.DtProcessamentoUltRecebimento,
         Parcelas.DtInclusao,
         Parcelas.DtAlteracao,
         Parcelas.CdColaborador,
         Parcelas.TsReplicacao,
         Parcelas.VrJuroPagoParcela format=commax12.2,
         Parcelas.VrJuroNaoPagoParcela format=commax12.2,
		 Case When Cancel.Contrato_Cancelado = 1 Then 'Sim' Else 'Não' End as Contrato_Cancelado

from tempw.Tb_Parcelas_Paketa as Parcelas
	 Left Join tempw.Ctrs_Cancelados as Cancel On Parcelas.NuContrato = Cancel.nucontratogestao

;RUN;

RUN;
