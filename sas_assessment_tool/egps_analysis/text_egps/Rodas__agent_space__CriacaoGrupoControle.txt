PROC SQL;
CREATE TABLE PotencialRodasShare AS
  SELECT
     POT.CdParceiroComercial
	,SUM(POT.QtContrato) AS Part_ContratosBV
    ,SUM(POT.QtGravamesMercado) AS Part_Mercado
	,SUM(POT.QtContrato) / SUM(POT.QtGravamesMercado) as Share
  FROM   TMP.TbPotencialRodas POT

  WHERE 1=1
    AND POT.NuAnoMes >= &safraM6.
    AND POT.FlAtualizacaoMercado = 1
	And CASE WHEN POT.FxIdade IS NULL THEN ''
         WHEN POT.FxIdade = 'ZeroKM' THEN 'Novos'
         ELSE 'Usados'
         END = 'Usados'
	and Pot.NmGrupoProduto = 'Leves'
	and POT.NmCanalEstruturaComercialC = 'Canal Veiculos'
	and POT.FlAtivo = 'S'
	and POT.NmTipoParceiroComercial in ('Concessionaria','Loja')
GROUP BY   1
;run;

proc sql;
create table CriaGrupoControle as
select
a.*,
b.Share
from Inpo.TbInsights as a
	left join PotencialRodasShare as b
	on a.pdv = b.CdParceiroComercial
where 1=1
and data >= today()
and Agrupamento <> ''
and Agrupamento not in('Mais de 40% de Pdvs C\Gravame e S\Contrato BV','Mais de 30% de Pdvs C\Gravame e S\Contrato BV','Mais de 30% de Pdvs C\Gravame e S\Contrato BV no mes anterior') /*TbInsight5*/
/*Or data >= '02JUN2025'd and Agrupamento_Controle <> ''*/
;run;
