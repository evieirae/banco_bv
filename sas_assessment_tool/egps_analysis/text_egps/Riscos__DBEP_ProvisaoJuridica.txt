PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.tblProcessoProvisaoJuridica  AS
select

 Cd_Causa 						as Cd_Causa
,SERIAL_IMPORT 					as SERIAL_IMPORT
,Pasta 							as Pasta
,IdNatureza 					as IdNatureza
,NmNatureza 					as NmNatureza
,IdNaturezaUnidadeCustom 		as IdNaturezaUnidadeCustom
,NmNaturezaUnidadeCustom 		as NmNaturezaUnidadeCustom
,CdCausaUnidade 				as CdCausaUnidade
,CdSistema 						as CdSistema
,CdUnidadeExyon 				as CdUnidadeExyon
,ID_CARTEIRA_CUSTOM 			as ID_CARTEIRA_CUSTOM
,DtCalculo 						as DtCalculo
,VlProvisaoProvavelAtualizado as VlProvisaoProvavelAtualizado
,VlProvisaoRemotoAtualizado as VlProvisaoRemotoAtualizado
,VlProvisaoPossivelAtualizado as VlProvisaoPossivelAtualizado
,VlProvisaoPedidoAtualizado as VlProvisaoPedidoAtualizado
,FgProvisaoAtual as FgProvisaoAtual

FROM CONNECTION TO postgres(

select
case
        when p.serial_import like ('%INQ%') then (left(p.SERIAL_IMPORT,10))
        when p.serial_import like ('%BVT%') then (left(p.SERIAL_IMPORT,10))
        when p.serial_import like ('%CON%') then (left(p.SERIAL_IMPORT,10))
        when p.serial_import like ('%VAC%') then (left(p.SERIAL_IMPORT,10))
		when p.serial_import like ('%CTO%') then (left(p.SERIAL_IMPORT,10))
	    when p.serial_import like ('%TRI%') then (left(p.SERIAL_IMPORT,10))
	    when p.serial_import like ('%TRB%') then (left(p.SERIAL_IMPORT,10))
                 else p.serial_import end                       as Cd_Causa
,p.SERIAL_IMPORT 						as SERIAL_IMPORT
,p.pasta                                                        as Pasta
,p.ID_NATUREZA                                                  as IdNatureza
,n.NATUREZA							as NmNatureza
,p.NATUREZA_UNIDADE_CUSTOM				        as IdNaturezaUnidadeCustom
,NC.NATUREZA_UNIDADE_CUSTOM				        as NmNaturezaUnidadeCustom
,null 								as CdCausaUnidade
,'p' 								as CdSistema
,case
 	when p.ID_NATUREZA = 28 and p.NATUREZA_UNIDADE_CUSTOM = 2             then 'TRI'
 	when p.ID_NATUREZA = 28 and p.NATUREZA_UNIDADE_CUSTOM = 1             then 'TBR'
	when p.ID_NATUREZA = 21 and p.NATUREZA_UNIDADE_CUSTOM = 4             then 'CTO'
 	when p.ID_NATUREZA = 21 and p.NATUREZA_UNIDADE_CUSTOM = 12            then 'CON'
 	when p.ID_NATUREZA = 21 and p.NATUREZA_UNIDADE_CUSTOM = 13            then 'VAC'
 	when p.ID_NATUREZA = 21 and p.NATUREZA_UNIDADE_CUSTOM in (1 ,7, 11)   then 'INQ'
 	when p.ID_NATUREZA = 27 and p.NATUREZA_UNIDADE_CUSTOM = 5             then 'BVT'
 	when p.ID_NATUREZA = 32 and p.NATUREZA_UNIDADE_CUSTOM in (7, 11 , 12) then 'INQ'
 		else null end 					as CdUnidadeExyon

,P.ID_CARTEIRA_CUSTOM
,cast(pp.DATA_CALCULO as date)			                as DtCalculo
,POP.VALOR_PERDA_PROVAVEL_ATUALIZAD 	AS VlProvisaoProvavelAtualizado
,( POP.VALOR_PEDIDO_ATUALIZADO - (POP.VALOR_PERDA_POSSIVEL_ATUALIZAD + POP.VALOR_PERDA_PROVAVEL_ATUALIZAD)) as VlProvisaoRemotoAtualizado
,POP.VALOR_PERDA_POSSIVEL_ATUALIZAD		as VlProvisaoPossivelAtualizado
,POP.VALOR_PEDIDO_ATUALIZADO			as VlProvisaoPedidoAtualizado
,case when dt.MaxDate = pp.ULTIMA_MODIFICACAO then 1 else 0 end  as FgProvisaoAtual

from public.M_PROCESSO P
	inner join public.LT_NATUREZA 				as n
		on N.ID_NATUREZA 					= p.ID_NATUREZA
	inner join public.LT_NATUREZA_UNIDADE_CUSTOM		as NC
		on NC.id_NATUREZA_UNIDADE_CUSTOM 	= P.NATUREZA_UNIDADE_CUSTOM
	left join public.D_PROVISAO_PROCESSO 			as PP
		on PP.ID_PROCESSO 		   			= P.ID_PROCESSO
	left join (
                             select
                                     P.ID_PROVISAO_PROCESSO
                                        ,cast(P.DATA_CADASTRO as date)             as DATA_CADASTRO
                                        ,sum(P.VALOR_PERDA_PROVAVEL_ATUALIZAD)     as VALOR_PERDA_PROVAVEL_ATUALIZAD
                                        ,sum(P.VALOR_PEDIDO_ATUALIZADO)		   	as VALOR_PEDIDO_ATUALIZADO
                                        ,sum(P.VALOR_PERDA_POSSIVEL_ATUALIZAD)     as VALOR_PERDA_POSSIVEL_ATUALIZAD
                                from public.D_PROVISAO_OBJETO_PROCESSO             as P
                                group by
                                        P.ID_PROVISAO_PROCESSO
                                        ,cast(P.DATA_CADASTRO as date)
                                                                                                                  ) as POP
                 on POP.ID_PROVISAO_PROCESSO = PP.ID_PROVISAO_PROCESSO
	left join (
                               select
					ID_PROCESSO
					,max(ULTIMA_MODIFICACAO)                    as MaxDate
				from public.D_PROVISAO_PROCESSO
				group by
                                      ID_PROCESSO
                                                                                                                  )  as DT
                on DT.ID_PROCESSO = P.ID_PROCESSO
where P.DATA_ENCERRAMENTO is null
and n.NATUREZA in ('Cível','Tributária')
and NC.NATUREZA_UNIDADE_CUSTOM  not in ('Inspetoria','Ofícios','Recuperação de Crédito')
AND CAST(pp.ultima_modificacao AS date) >= cast(now() as date)-10
AND pp.ID_NATUREZA IN (21, 28)


);DISCONNECT FROM postgres;
RUN;
