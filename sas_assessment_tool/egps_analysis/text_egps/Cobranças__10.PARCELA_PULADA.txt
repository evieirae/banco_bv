SELECT
    A.DATA_BASE
  , A.NuCpfCnpj
  , A.NUCONTRATO
  , A.SQCONTRATOFINANCEIRO
  , A.CdSItuacaoContabil
  , A.CDMOTIVOCONTRATOFINANCEIRO
  , A.CDPRODUTO
  , A.CDSUBPRODUTO
  , A.CDMODALIDADEPRODUTO
  , A.NMMODALIDADEPRODUTO
  , A.NMGRUPOSUBPRODUTO
  , A.PRODUTO
  , A.LETRA_VALIDA AS LETRA_ATUAL
  , A.SALDO
  , A.QTATRASO AS ATRASO
  , B.nudiasatraso AS ATRASO_A1
  , C.nudiasatraso AS ATRASO_A2
  , D.nudiasatraso AS ATRASO_A3
  , E.nudiasatraso AS ATRASO_A4
  , F.nudiasatraso AS ATRASO_A5
  , G.nudiasatraso AS ATRASO_A6
  , FECH.nudiasatraso AS ATRASO_FECH

  
  , CASE WHEN NOT MISSING(A.QTATRASO) AND NOT MISSING(FECH.nudiasatraso)
  			  AND FECH.nudiasatraso < A.QTATRASO THEN 1 ELSE 0 END AS FL_PAGAMENTO

			  AND FECH.nudiasatraso > 0 THEN 1 ELSE 0 END AS FL_PARCIAL

              AND ABS(A.QTATRASO - B.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c01
  , (CASE WHEN NOT MISSING(B.nudiasatraso) AND NOT MISSING(C.nudiasatraso)
              AND ABS(B.nudiasatraso - C.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c02
  , (CASE WHEN NOT MISSING(C.nudiasatraso) AND NOT MISSING(D.nudiasatraso)
              AND ABS(C.nudiasatraso - D.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c03
  , (CASE WHEN NOT MISSING(D.nudiasatraso) AND NOT MISSING(E.nudiasatraso)
              AND ABS(D.nudiasatraso - E.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c04
  , (CASE WHEN NOT MISSING(E.nudiasatraso) AND NOT MISSING(F.nudiasatraso)
              AND ABS(E.nudiasatraso - F.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c05
  , (CASE WHEN NOT MISSING(F.nudiasatraso) AND NOT MISSING(G.nudiasatraso)
              AND ABS(F.nudiasatraso - G.nudiasatraso) < 5 THEN 1 ELSE 0 END) AS c06

  , (CALCULATED c01 + CALCULATED c02 + CALCULATED c03
     + CALCULATED c04 + CALCULATED c05 + CALCULATED c06) AS QT_OSCILACOES_MENOS_20

  , (CALCULATED c01 * CALCULATED c02
     + CALCULATED c02 * CALCULATED c03
     + CALCULATED c03 * CALCULATED c04
     + CALCULATED c04 * CALCULATED c05
     + CALCULATED c05 * CALCULATED c06) AS QT_PARES_CONSECUTIVOS

  
  , ( CALCULATED c01 * (CASE WHEN CALCULATED c02 = 1 THEN 1 ELSE 0 END)
    + CALCULATED c02 * (CASE WHEN CALCULATED c01 = 1 OR CALCULATED c03 = 1 THEN 1 ELSE 0 END)
    + CALCULATED c03 * (CASE WHEN CALCULATED c02 = 1 OR CALCULATED c04 = 1 THEN 1 ELSE 0 END)
    + CALCULATED c04 * (CASE WHEN CALCULATED c03 = 1 OR CALCULATED c05 = 1 THEN 1 ELSE 0 END)
    + CALCULATED c05 * (CASE WHEN CALCULATED c04 = 1 OR CALCULATED c06 = 1 THEN 1 ELSE 0 END)
    + CALCULATED c06 * (CASE WHEN CALCULATED c05 = 1 THEN 1 ELSE 0 END)
    ) AS QT_VEZES_CONSECUTIVAS

  
  , ( CALCULATED c01 * CALCULATED c02 * CALCULATED c03 * CALCULATED c04 * CALCULATED c05 *  CALCULATED c06) AS FL_SEIS_CONSECUTIVAS
  , ( CALCULATED c01 * CALCULATED c02 * CALCULATED c03 * CALCULATED c04 * CALCULATED c05) AS FL_CINCO_CONSECUTIVAS
  , ( CALCULATED c01 * CALCULATED c02 * CALCULATED c03 * CALCULATED c04) AS FL_QUATRO_CONSECUTIVAS
  , ( CALCULATED c01 * CALCULATED c02 * CALCULATED c03) AS FL_TRES_CONSECUTIVAS
  , ( CALCULATED c01 * CALCULATED c02) AS FL_DUAS_CONSECUTIVAS
  , ( CALCULATED c01) AS FL_UMA_CONSECUTIVAS

  , H.vr_provisao AS vr_provisao
  , I.vr_provisao AS vr_provisao_A1
  , J.vr_provisao AS vr_provisao_A2
  , K.vr_provisao AS vr_provisao_A3
  , L.vr_provisao AS vr_provisao_A4
  , M.vr_provisao AS vr_provisao_A5
  , N.vr_provisao AS vr_provisao_A6

  , H.cd_estagio AS cd_estagio
  , I.cd_estagio AS cd_estagio_A1
  , J.cd_estagio AS cd_estagio_A2
  , K.cd_estagio AS cd_estagio_A3
  , L.cd_estagio AS cd_estagio_A4
  , M.cd_estagio AS cd_estagio_A5
  , N.cd_estagio AS cd_estagio_A6

  , H.fl_ativo_problematico AS fl_ativo_problematico
  , I.fl_ativo_problematico AS fl_ativo_problematico_A1
  , J.fl_ativo_problematico AS fl_ativo_problematico_A2

  , M.fl_ativo_problematico AS fl_ativo_problematico_A5
  , N.fl_ativo_problematico AS fl_ativo_problematico_A6

    LEFT JOIN ATRASO_&SAFRAA4. E

    LEFT JOIN ATRASO_&SAFRAA5. F

    LEFT JOIN ATRASO_&SAFRAA6. G

proc sql;
  %CONNECT_GCP;

  create table PARCELAS_&SAFRA. as
  select
      SAFRA,
      dtprocessamento,
	  DTVENCIMENTO_ATRASO,
      nucontrato,
      sqcontratofinanceiro,
      NUPARCELA,
      VRPRINCIPAL,

      VRPARCELA,
      STPARCELA,
      DTVENCIMENTO,
      DT_ULTIMO_RECEBIMENTO,
      DT_PROCESSAMENTO_ULT_RECEBIMENTO,
      FL_PARCELA_PULADA,
      FL_PAGO_POS_PULADA,
      max(nudiasatraso) as nudiasatraso
  from connection to PERM
  (
    with parc as (
      select
        &SAFRA. as SAFRA,
        A.dtprocessamento,
        A.nucontrato,
        A.sqcontratofinanceiro,
        A.nudiasatraso,
        B.NU_PARCELA                      as NUPARCELA,
        B.VR_PRINCIPAL                    as VRPRINCIPAL,
        B.VR_JUROS                        as VRJUROS,
        B.VR_PARCELA                      as VRPARCELA,
        B.ST_PARCELA                      as STPARCELA,
        B.DT_VENCIMENTO                   as DTVENCIMENTO,
        B.DT_ULTIMO_RECEBIMENTO,
        B.DT_PROCESSAMENTO_ULT_RECEBIMENTO
      from  `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` A
      left join `bv-pdbd-prd.dvry_core.tbparcela` B
        on A.nucontrato = B.CD_CONTRATO
       and A.sqcontratofinanceiro = B.SQ_CONTRATO_FINANCEIRO
      where A.dtprocessamento = &dt_process.

    ),

    
    marcacao as (
      select
        p.*,

        
        date_sub(date(p.dtprocessamento), interval p.nudiasatraso day) as DTVENCIMENTO_ATRASO,

        
        case
          when p.STPARCELA = 'Q'
           and p.DT_ULTIMO_RECEBIMENTO is not null
           and date(p.DT_ULTIMO_RECEBIMENTO) <= date(p.dtprocessamento)
          then 1 else 0
        end as IS_PAGO_NO_SNAPSHOT
      from parc p
    ),

    
    marcacao2 as (
      select
        m.*,
        max(
          case
            when m.IS_PAGO_NO_SNAPSHOT = 1
             and date(m.DTVENCIMENTO) > m.DTVENCIMENTO_ATRASO
            then 1 else 0
          end
        ) over (
          partition by m.nucontrato, m.sqcontratofinanceiro, m.dtprocessamento
          rows between unbounded preceding and unbounded following
        ) as HAS_PAGO_POS_ANCHOR
      from marcacao m
    ),

    
    marcacao_flag as (
      select
        m2.*,
        case
          when m2.STPARCELA = 'A'
           and date(m2.DTVENCIMENTO) <= m2.DTVENCIMENTO_ATRASO
           and coalesce(m2.HAS_PAGO_POS_ANCHOR, 0) = 1
          then 1 else 0
        end as FL_PARCELA_PULADA
      from marcacao2 m2
    ),

    
    marcacao_flag2 as (
      select
        mf.*,
        max(case when mf.FL_PARCELA_PULADA = 1 then 1 else 0 end)
          over (partition by mf.nucontrato, mf.sqcontratofinanceiro, mf.dtprocessamento) as HAS_PULADA
      from marcacao_flag mf
    )

    
    select
      f.SAFRA,
      f.dtprocessamento,
	  f.DTVENCIMENTO_ATRASO,
      f.nucontrato,
      f.sqcontratofinanceiro,
      f.NUPARCELA,
      f.VRPRINCIPAL,
      f.VRJUROS,
      f.VRPARCELA,
      f.STPARCELA,
      f.DTVENCIMENTO,
      f.DT_ULTIMO_RECEBIMENTO,
      f.DT_PROCESSAMENTO_ULT_RECEBIMENTO,
      f.FL_PARCELA_PULADA,

      
      case
        when f.IS_PAGO_NO_SNAPSHOT = 1
         and date(f.DTVENCIMENTO) > f.DTVENCIMENTO_ATRASO
         and coalesce(f.HAS_PULADA, 0) = 1
        then 1 else 0
      end as FL_PAGO_POS_PULADA,

      f.nudiasatraso
    from marcacao_flag2 f
  )
  group by
      SAFRA,
      dtprocessamento,
	  DTVENCIMENTO_ATRASO,
      nucontrato,
      sqcontratofinanceiro,
      NUPARCELA,
      VRPRINCIPAL,
      VRJUROS,
      VRPARCELA,
      STPARCELA,
      DTVENCIMENTO,
      DT_ULTIMO_RECEBIMENTO,
      DT_PROCESSAMENTO_ULT_RECEBIMENTO,
      FL_PARCELA_PULADA,
      FL_PAGO_POS_PULADA
  ;
quit;

PROC SQL;
CREATE TABLE PARCELAS_&SAFRA._CRUZADA_1
AS SELECT
A.*
, CASE WHEN DATEPART(A.dtprocessamento) <= A.DTVENCIMENTO THEN 1 ELSE 0 END AS FL_PARCELAS_FUTURAS
, CASE WHEN DATEPART(A.dtprocessamento) > A.DTVENCIMENTO THEN 1 ELSE 0 END AS FL_PARCELAS_PASSADAS
FROM PARCELAS_&SAFRA._CRUZADA A
;QUIT;

PROC SQL;
CREATE TABLE PARCELAS_&SAFRA._CRUZADA_2 AS
SELECT
    A.SAFRA
  , A.dtprocessamento
  , A.DTVENCIMENTO_ATRASO
  , A.nucontrato
  , A.sqcontratofinanceiro
  , A.nudiasatraso

  
  , SUM(CASE WHEN NUPARCELA <> . THEN 1 ELSE 0 END)                                               AS QTD_PAR
  , SUM(CASE WHEN FL_PARCELAS_PASSADAS = 1 THEN 1 ELSE 0 END)                                     AS QTD_PAR_PAS
  , SUM(CASE WHEN FL_PARCELAS_FUTURAS  = 1 THEN 1 ELSE 0 END)                                     AS QTD_PAR_FUT

  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 THEN 1 ELSE 0 END)                                        AS QTD_PAR_PUL
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN 1 ELSE 0 END)           AS QTD_PAR_PUL_PAS
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN 1 ELSE 0 END)           AS QTD_PAR_PUL_FUT

  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 THEN 1 ELSE 0 END)                                       AS QTD_PAR_PAG_POSPUL
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN 1 ELSE 0 END)          AS QTD_PAR_PAG_POSPUL_PAS
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN 1 ELSE 0 END)          AS QTD_PAR_PAG_POSPUL_FUT

  , SUM(CASE WHEN STPARCELA IN ('q','Q') THEN 1 ELSE 0 END)                                       AS QTD_PAR_PAG
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_PASSADAS = 1 THEN 1 ELSE 0 END)          AS QTD_PAR_PAG_PAS
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_FUTURAS  = 1 THEN 1 ELSE 0 END)          AS QTD_PAR_PAG_FUT

  , SUM(CASE WHEN STPARCELA IN ('a','A') AND FL_PARCELAS_PASSADAS = 1 THEN 1 ELSE 0 END)          AS QTD_PAR_ATR_PAS
  , SUM(CASE WHEN STPARCELA IN ('a','A') THEN 1 ELSE 0 END)                                       AS QTD_PAR_ABE

  
  , SUM(CASE WHEN NUPARCELA <> . THEN VRPARCELA ELSE 0 END)                                       AS VR_PAR
  , SUM(CASE WHEN FL_PARCELAS_PASSADAS = 1 THEN VRPARCELA ELSE 0 END)                             AS VR_PAR_PAS
  , SUM(CASE WHEN FL_PARCELAS_FUTURAS  = 1 THEN VRPARCELA ELSE 0 END)                             AS VR_PAR_FUT

  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 THEN VRPARCELA ELSE 0 END)                                AS VR_PAR_PUL
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN VRPARCELA ELSE 0 END)   AS VR_PAR_PUL_PAS
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN VRPARCELA ELSE 0 END)   AS VR_PAR_PUL_FUT

  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 THEN VRPARCELA ELSE 0 END)                               AS VR_PAR_PAG_POSPUL
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN VRPARCELA ELSE 0 END)  AS VR_PAR_PAG_POSPUL_PAS
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN VRPARCELA ELSE 0 END)  AS VR_PAR_PAG_POSPUL_FUT

  , SUM(CASE WHEN STPARCELA IN ('q','Q') THEN VRPARCELA ELSE 0 END)                               AS VR_PAR_PAG
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_PASSADAS = 1 THEN VRPARCELA ELSE 0 END)  AS VR_PAR_PAG_PAS
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_FUTURAS  = 1 THEN VRPARCELA ELSE 0 END)  AS VR_PAR_PAG_FUT

  , SUM(CASE WHEN STPARCELA IN ('a','A') AND FL_PARCELAS_PASSADAS = 1 THEN VRPARCELA ELSE 0 END)  AS VR_PAR_ATR_PAS
  , SUM(CASE WHEN STPARCELA IN ('a','A') THEN VRPARCELA ELSE 0 END)                               AS VR_PAR_ABE

  
  , SUM(CASE WHEN NUPARCELA <> . THEN VRPRINCIPAL ELSE 0 END)                                      AS PRC_PAR
  , SUM(CASE WHEN FL_PARCELAS_PASSADAS = 1 THEN VRPRINCIPAL ELSE 0 END)                            AS PRC_PAR_PAS
  , SUM(CASE WHEN FL_PARCELAS_FUTURAS  = 1 THEN VRPRINCIPAL ELSE 0 END)                            AS PRC_PAR_FUT

  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 THEN VRPRINCIPAL ELSE 0 END)                               AS PRC_PAR_PUL
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN VRPRINCIPAL ELSE 0 END)  AS PRC_PAR_PUL_PAS
  , SUM(CASE WHEN FL_PARCELA_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN VRPRINCIPAL ELSE 0 END)  AS PRC_PAR_PUL_FUT

  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 THEN VRPRINCIPAL ELSE 0 END)                              AS PRC_PAR_PAG_POSPUL
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_PASSADAS = 1 THEN VRPRINCIPAL ELSE 0 END) AS PRC_PAR_PAG_POSPUL_PAS
  , SUM(CASE WHEN FL_PAGO_POS_PULADA = 1 AND FL_PARCELAS_FUTURAS  = 1 THEN VRPRINCIPAL ELSE 0 END) AS PRC_PAR_PAG_POSPUL_FUT

  , SUM(CASE WHEN STPARCELA IN ('q','Q') THEN VRPRINCIPAL ELSE 0 END)                              AS PRC_PAR_PAG
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_PASSADAS = 1 THEN VRPRINCIPAL ELSE 0 END) AS PRC_PAR_PAG_PAS
  , SUM(CASE WHEN STPARCELA IN ('q','Q') AND FL_PARCELAS_FUTURAS  = 1 THEN VRPRINCIPAL ELSE 0 END) AS PRC_PAR_PAG_FUT

  , SUM(CASE WHEN STPARCELA IN ('a','A') THEN VRPRINCIPAL ELSE 0 END)                              AS PRC_PAR_ABE

FROM PARCELAS_&SAFRA._CRUZADA_1 A
GROUP BY
    A.SAFRA
  , A.dtprocessamento
  , A.DTVENCIMENTO_ATRASO
  , A.nucontrato
  , A.sqcontratofinanceiro
  , A.nudiasatraso
