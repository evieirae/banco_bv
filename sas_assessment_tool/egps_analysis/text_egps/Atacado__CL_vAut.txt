   CREATE TABLE TbIntCL AS SELECT

   		  t1.CdPropostaLimiteCredito,

          t2.NmEstadoProposta,
		  t4.DtConclusaoAtividade as DtDimConclusao,
		  t5.NmModeloProposta,
		  t6.NmTipoPropostaLimite,
		  CASE WHEN t4.DtConclusaoAtividade is not null THEN datepart(t4.DtConclusaoAtividade) - datepart(t1.DtCriacaoProposta) ELSE . END AS SLA

      FROM CL_TBPROPOSTALIMITECREDITO AS t1

	  LEFT JOIN CL_TBESTADOPROPOSTA AS t2 ON t1.CdEstadoProposta = t2.CdEstadoProposta

	  LEFT JOIN CONCLUSAO_PROPOSTA AS t4 ON t1.CdPropostaLimiteCredito = t4.CdPropostaLimiteCredito

	  LEFT JOIN CL_TBMODELOPROPOSTA AS t5 ON t1.CdModeloProposta = t5.CdModeloProposta

	  LEFT JOIN CL_TBTIPOPROPOSTALIMITE AS t6 ON t1.CdTipoPropostaLimite = t6.CdTipoPropostaLimite;

QUIT;
