CREATE TABLE WORK.Isencao_Encargos AS
SELECT DISTINCT A.*
, INPUT(COMPRESS(A.cd_contrato_gestao),best14.) Format = best14. Informat = best14. AS NuContratoGestao
FROM CONNECTION TO PERM(
Select Distinct HUB.cd_contrato_gestao
, Case When HUB.cd_contrato_gestao = UDA2.u2acct AND (Safe_Cast(UDA2.U2SCORE16 AS INT64) IN (999,2001) OR (Safe_Cast(UDA2.U2SCORE16 AS INT64) BETWEEN 1010 AND 1090)) Then 1 Else 0 End As Flag_IsencaoEnc
From `bv-pdbd-prd.dvry_corp_cobranca.inadvarejo_hub_fato` AS HUB
Inner Join `bv-pdbd-prd.trusted_rcvry.uda2` AS UDA2 ON HUB.cd_contrato_gestao = UDA2.u2acct
Where Date(HUB.ts_inclusao) >= DATE_ADD(CURRENT_DATE, INTERVAL -1 DAY)
AND (Safe_Cast(UDA2.U2SCORE16 AS INT64) IN (999,2001) OR (Safe_Cast(UDA2.U2SCORE16 AS INT64) BETWEEN 1010 AND 1090))
) AS A
;disconnect from perm
