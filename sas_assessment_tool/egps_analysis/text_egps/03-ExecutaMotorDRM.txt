/*Imputar Datas manuais*/

%include "/&Modo_Exec./riscosmercado/rmsas/bin/sas/joblist/00-IncludeFatorVertice.sas";
%include "/&Modo_Exec./riscosmercado/rmsas/bin/sas/joblist/01-MotorDRM_AlocaVertices.sas";

%let DTAUX=%sysfunc(SCAN(&sysparm.,1,"/"));
/*Data de mudança no DRM - condições para reprocessamento retroativo*/
%LET DTMUD = 20180530;
%let dtCorteOne = 20220830;

%let origem=%sysfunc(SCAN(&sysparm.,-1,"/"));

%let DIA = %sysfunc(substr(&DTAUX.,7,2));
%let MES = %sysfunc(substr(&DTAUX.,5,2));
%let ANO = %sysfunc(substr(&DTAUX.,1,4));
%let AAAAMM  =&ANO.&MES.;
%let AAAAMMDD=&ANO.&MES.&DIA.;

%let dtproc = %sysfunc(putn(%sysfunc(mdy(&MES.,&DIA.,&ANO.)), date9.));

%put DTAUX=&DTAUX.;
%put DTREF=&DTREF.;
%put origem=&origem.;
%put dtproc=&dtproc.;

%macro ExecutaDRM;

	%GeraFatorVertice(False);

	%let flagAlgoOne = 0;

	%if &AAAAMMDD. < &dtCorteOne. %then %do;
		%let result_rwa=%sysfunc(exist(_INPUT.RWA_BACEN&origem._&AAAAMMDD.));
		%let result_pos=%sysfunc(exist(_INPUT.TBPOSICAO&origem._&AAAAMMDD.));
		%let flagAlgoOne = 0;
	%end;
	%else %do;
		%let result_rwa=%sysfunc(exist(_CADONE.RWA_BACENMOR_&AAAAMMDD.));
		%let result_pos=%sysfunc(exist(_POSONE.TBPOSICAOMOR_&AAAAMMDD.));
		%let flagAlgoOne = 1;
	%end;

	%if &result_rwa eq 1 and &result_pos eq 1 %then %do;
		%CriaFatoresDeRisco;

		%if &flagAlgoOne eq 1 %then %do;
			%CriaRWAPOSICAO_One(&origem.);
		%end;
		%else %do;
			%CriaRWAPOSICAO(&origem.);
		%end;

		%GeraBaseDRMContabil(&origem.);
		%GeraBaseDRMAtivFinan(&origem.);
/*		%GeraArquivoXML(&origem.,2040);*/
		%GeraArquivoXML(&origem.,2060);
	%end;
	%else %do;
		%put result_rwa=&result_rwa.;
		%put result_pos=&result_pos.;
	%end;
%mend ExecutaDRM;

%ExecutaDRM;
