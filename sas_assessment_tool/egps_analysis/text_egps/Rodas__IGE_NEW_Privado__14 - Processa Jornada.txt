proc sql;
	create table tempw.CONSIGNADO_PRIVADO2 as
		select CONSP.*,
			CAP.NmColaboradorC as NmGr,
			PRD.Produto,
			CAP.DsLoginN1C as GR,
			CAP.DsLoginN2C as GF,
			CAP.NmFilialC,
			CAP.NmAreaEstruturaComercialC as NmAreaEstruturaComercial,
			CAP.NmClassificacaoLimiteConvenio as Grupo_Economico,
			CAP.DsQuantidadeVidas as Porte_Convenio,
			0 as Margem_Negativa,
			PO.VrMargem,
			TBSEG.QtSeguros,
			TBSEG.VrSeguros FORMAT COMMAX12.2 ,
			TBSEG.VrSeguro_SPF FORMAT COMMAX12.2,
			TBSEG.VrSeguro_AP FORMAT COMMAX12.2,
			PRD.FlPropostaAprovada,
		case
			when PA.nuproposta is not null then 1
			else 0
		end
	as Proposta_Averbada,
		PVCTO.dtprivencimento format ddmmyy10.,
		PVCTO.DtDescFolha
	from tempw.CONSIGNADO_PRIVADO CONSP
		left join tempw.Propostas_Averbadas PA
			on CONSP.nuproposta = PA.nuproposta
		left join
			tempw.Propostas_Originacao PRD
			on CONSP.nuproposta = PRD.nuproposta
		left join
			tempw.Producao PO
			on CONSP.nuproposta = PO.nuproposta
			/*left join*/
	left join tempw.BASE_CONV_CADSTRADOS_PRIVADO CAP
		on CONSP.cdparceirocomercial = CAP.cdparceirocomercial
	left join tempw.TB_PRIMEIRO_VCTO PVCTO
		on CONSP.nucontratogestao = PVCTO.nucontrato
	left join
		tempw.PRIVADO_TBSEGUROS TBSEG
		on CONSP.nuproposta = TBSEG.nuproposta
	;
RUN;

PROC SQL;
	CREATE TABLE tempw.Privado_Leadtime_4
		AS

		select
			t1.nuproposta,
			t1.dtstatusproposta_ddmmyyy,
			t1.dtstatusproposta,
			t1.stproposta,
			t1.n,
			t2.nuproposta as nuproposta_2,
			t2.dtstatusproposta_ddmmyyy as dtstatusproposta_ddmmyyy_2,
			t2.dtstatusproposta as dtstatusproposta_2,
			t2.stproposta as stproposta_2,
			t2.n as n_2

		from
			(
		SELECT tabela,
			nuproposta,
			dtstatusproposta_ddmmyyy,
			dtstatusproposta,
			stproposta,
			n
		FROM tempw.PRIVADO_LEADTIME_3) t1

		LEFT JOIN
			(
		SELECT tabela,
			nuproposta,
			dtstatusproposta_ddmmyyy,
			dtstatusproposta,
			stproposta,
			n
		FROM tempw.PRIVADO_LEADTIME_3) t2
			ON
			t1.nuproposta = t2.nuproposta and
			t1.n = t2.n-1
		where t2.nuproposta not is null;
RUN;

proc sql;
	create table tempw.Privado_Leadtime_D_VA_AP
		as
			select tbleadtime.*,
				catx('-',tbleadtime.StProposta,tbleadtime.StProposta_2) as jornada,
				intck('second',tbleadtime.dtstatusproposta ,tbleadtime.dtstatusproposta_2 ) AS leadtime_seconds,
				intck('second',tbleadtime.dtstatusproposta ,tbleadtime.dtstatusproposta_2 )/3600 AS leadtime_horas

			from
				tempw.Privado_Leadtime_4 tbleadtime
			inner join
				tempw.CONSIGNADO_PRIVADO2 tbproposta
				on tbleadtime.nuproposta = tbproposta.nuproposta;
RUN;

proc sql;
	create table tempw.agrupa_jornada
		as
			SELECT A.NUPROPOSTA,
				CASE
					WHEN A.jornada = 'D-AV' THEN  A.leadtime_seconds
					ELSE 0
				END
			AS LEAD_D_AV,
				CASE
					WHEN A.jornada = 'D-AP' THEN  A.leadtime_seconds
					ELSE 0
				END
			AS LEAD_D_AP,
				CASE
					WHEN B.leadtime_seconds IS NULL THEN 0
					ELSE  B.leadtime_seconds
				END
			AS LEAD_AV_AP
				FROM
					(SELECT NUPROPOSTA, jornada, SUM(leadtime_seconds) AS leadtime_seconds FROM tempw.Privado_Leadtime_D_VA_AP WHERE jornada = 'D-AV' OR jornada = 'D-AP' GROUP BY NUPROPOSTA, jornada ) A
						LEFT JOIN
							(SELECT NUPROPOSTA, jornada, SUM(leadtime_seconds) AS leadtime_seconds FROM tempw.Privado_Leadtime_D_VA_AP WHERE jornada = 'AV-AP' GROUP BY NUPROPOSTA, jornada) B
								ON A.NUPROPOSTA = B.NUPROPOSTA;
RUN;

proc sql;
	create table tempw.agrupa_jornada2 as
		select
			proposta.NuCpfCnpj_SobAnalise AS CPF_JORNADA,
			JORNADA.nuproposta,
			JORNADA.LEAD_D_AV,
		CASE
			WHEN JORNADA.LEAD_D_AV <> 0 THEN proposta.NuCpfCnpj_SobAnalise
			ELSE ''
		END
	AS CPF_JORNADA_D_AV,
		JORNADA.LEAD_D_AP,
	CASE
		WHEN JORNADA.LEAD_D_AP <> 0 THEN proposta.NuCpfCnpj_SobAnalise
		ELSE ''
	END
AS CPF_JORNADA_D_AP,
	JORNADA.LEAD_AV_AP,
CASE
	WHEN JORNADA.LEAD_AV_AP <> 0 THEN proposta.NuCpfCnpj_SobAnalise
	ELSE ''
END
AS CPF_JORNADA_AV_AP,
JORNADA.LEAD_D_AV + JORNADA.LEAD_AV_AP + JORNADA.LEAD_D_AP as LEADTIME_TOTAL
from tempw.agrupa_jornada JORNADA
INNER JOIN
	tempw.CONSIGNADO_PRIVADO2 proposta
	ON JORNADA.NUPROPOSTA = proposta.NUPROPOSTA
	;

PROC SQL;
	create table tempw.BI_CONSIG_LEADTIME_PROPOSTAS
		as
			SELECT
				'PROPOSTAS' AS TIPO_PROPOSTA,
				proposta.*,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN 0
				ELSE jornada.LEAD_D_AV
			END
		AS LEADTIME_D_AV,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN ''
				ELSE jornada.CPF_JORNADA_D_AV
			END
		AS CPF_JORNADA_D_AV,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN 0
				ELSE jornada.LEAD_AV_AP
			END
		AS LEAD_AV_AP,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN ''
				ELSE jornada.CPF_JORNADA_AV_AP
			END
		AS CPF_JORNADA_AV_AP,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN 0
				ELSE jornada.LEAD_D_AP
			END
		AS LEAD_D_AP,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN ''
				ELSE jornada.CPF_JORNADA_D_AP
			END
		as CPF_JORNADA_D_AP,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN 0
				ELSE jornada.LEADTIME_TOTAL
			END
		AS LEADTIME_TOTAL,
			CASE
				WHEN Proposta.stproposta <> 'AP' THEN ''
				ELSE jornada.CPF_JORNADA
			END
		AS CPF_JORNADA
			from
				tempw.CONSIGNADO_PRIVADO2 proposta
			left join
				tempw.agrupa_jornada2 jornada
				on proposta.nuproposta = jornada.nuproposta;
RUN;
