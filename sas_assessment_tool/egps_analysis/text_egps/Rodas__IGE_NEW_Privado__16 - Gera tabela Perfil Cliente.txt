PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.BI_PERIFL_CLIENTE AS
	SELECT *
	FROM CONNECTION TO DBCON

(
SELECT
CONVERT(VARCHAR(4), DATEPART (year,Proposta.DtStatusProposta))+'/'+CONVERT(VARCHAR(4),DATEPART(month,Proposta.DtStatusProposta)) AS anomes,
	year(Proposta.DtStatusProposta) as ano,
	month(Proposta.DtStatusProposta) as mes,
	tbpessoafisica.nudocumento,
	proposta.nuproposta,
	Proposta.StProposta,
TbEstadoCivil.dsestadocivil,
tbpessoafisica.cdtiporesidencia,
tbpessoafisica.cdsexo,
tbpessoafisica.dsnaturalidade,
tbpessoafisica.dsufnaturalidade,
tbpessoafisica.qtmesesresidenciaatual/12 AS tempo_residencia_em_anos,
DATEDIFF(YY,tbpessoafisica.DTNASCIMENTO,Proposta.DtentradaProposta) AS idade_calc,
tbpessoafisica.pccompromentrenda,
tbpessoafisica.vrtotalreceitas,
tbpessoafisica.qtveiculos
FROM
DBA.TbProposta Proposta
INNER JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
ON Proposta.NuProposta = PessoaSobAnalise.NuProposta
INNER JOIN
DBA.TbPessoaFisicaSobAnalise tbpessoafisica
ON PessoaSobAnalise.CdPessoaSobAnalise = tbpessoafisica.CdPessoasobanalise
INNER JOIN
DBA.TbEstadoCivil TbEstadoCivil
ON tbpessoafisica.cdestadocivil = TbEstadoCivil.cdestadocivil
WHERE Proposta.CdSubProduto = 12
AND year(Proposta.DtStatusProposta) >= year(today())-1

);
RUN;

%let dir_csv = /riscosbvf/BA/processos/output;

PROC EXPORT
	DATA= tempw.BI_PERIFL_CLIENTE
	FILE="&dir_csv./Privado_BI_PERIFL_CLIENTE.csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;
