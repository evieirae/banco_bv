   CREATE TABLE WORK.QUERY_FOR_TBBRUTA AS SELECT DISTINCT
		t1.CdGpEconomico,

		t1.NmPessoa,
		case when T1.DsClassificacao ne '' then T1.DsClassificacao else T1.DsClassificacaoSME end as NmClassificacao,
		t1.DtVencimentoGLCliente,
		t1.VrTotalAprovadoGLCliente,

	A.*,

	LEFT JOIN QUERY_FOR_TBBRUTA AS B ON(A.CNPJ_RAIZ_SACADO = B.NuCpfCnpjRaiz)
