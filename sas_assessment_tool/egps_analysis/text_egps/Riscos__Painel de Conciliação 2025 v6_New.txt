            'EVENTO-ENVIAR BACEN?'N
        from TEMP.TB_RPRO5050_COMPILADO
        where ENTIDADE IN ('EVENTO','EVENTO+CONTABILIZAÇÃO')
    ) as base
    inner join (
        select
            CATX('-', 'EVENTO-NUMERO DE REGISTRO'N, 'EVENTO-SISTEMA'N, 'EVENTO-GRUPO PERDA'N) as IC_CHAVE,
            min('EVENTO-DATA OCORRÊNCIA'N) as MIN_OCORRENCIA,
            min('EVENTO-DATA RECONHECIMENTO'N) as MIN_RECONHECIMENTO
        from TEMP.TB_RPRO5050_COMPILADO
        where ENTIDADE IN ('EVENTO','EVENTO+CONTABILIZAÇÃO')
        group by CATX('-', 'EVENTO-NUMERO DE REGISTRO'N, 'EVENTO-SISTEMA'N, 'EVENTO-GRUPO PERDA'N)

    on base.IC_CHAVE = filtro.IC_CHAVE

        T2.NmEmpresaCustomDbpo,

            when T1.Grupo_perda = 'Despesas fraudes' then 'Fraudes'
