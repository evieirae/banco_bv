proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table aprosimu_bvo_bb_0 as

   select * from connection to perm

(
SELECT DISTINCT

cast(a.cd_simulacao_proposta AS NUMERIC) 		AS cd_simulacao_proposta
,d.CD_SIMULACAO_PRAZO_FNNCO                     AS Simu_Parc
,d.cd_simulacao_dado_fnnco                      AS cd_simu_dado
,e.DS_MOTIVO_DECISAO_PDEC                       AS motivo_decisao
,a.st_pre_analise_bvo                        	AS Decisao_preAnalise_Bvo
,e.PC_PROBABILIDADE_APROVACAO                   AS ProbabilidadeAprovacao
,cast(g.nu_proposta AS NUMERIC) 			    AS nu_proposta
,h.cd_parceiro_externo 							AS cd_parceiro_bb
,g.NU_PROPOSTA 									as nuProposta

FROM bv-pdbd-prd.trusted_apro_simu.tb_simulacao_proposta a
LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_simulacao_proposta_garantia b
			ON a.cd_simulacao_proposta = b.cd_simulacao_proposta
LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_simulacao_dado_fnnco c
			ON b.cd_simulacao_dado_fnnco = c.cd_simulacao_dado_fnnco
LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_simulacao_prazo_fnnco d
			ON c.cd_simulacao_dado_fnnco = d.cd_simulacao_dado_fnnco
LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_simulacao_prazo_faixa_aprvo e
		ON d.cd_simulacao_prazo_fnnco = e.cd_simulacao_prazo_fnnco
/*LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_pz_faixa_motivo_recusa f*/
/*	   ON e.cd_simulacao_prazo_faixa_aprvo = f.cd_simulacao_prazo_faixa_aprvo*/
LEFT JOIN bv-pdbd-prd.trusted_apro_simu.tb_simulacao_przo_fnnco_prpsa g
			ON d.cd_simulacao_prazo_fnnco = g.cd_simulacao_prazo_fnnco
			AND e.DS_MOTIVO_DECISAO_PDEC  = 'OK'
INNER JOIN bv-pdbd-prd.trusted_apro_simu.tb_proposta_parceria h
			ON a.cd_simulacao_proposta = h.cd_simulacao_proposta
/* 			AND coalesce(h.FL_ATIVO, 'S') = 'S'*/
WHERE  a.dt_simulacao_proposta >= '2024-11-01'
/*AND tspfa.PC_PROBABILIDADE_APROVACAO = 100*/
and h.cd_parceiro_externo = '41'
);disconnect from perm;
quit;

PROC SQL;
CREATE TABLE aprosimu_bvo_bb_1 AS

SELECT * FROM aprosimu_bvo_bb_0

;quit;
PROC SORT DATA=aprosimu_bvo_bb_1
	OUT=aprosimu_bvo_bb_1_1;
	BY DESCENDING cd_simulacao_proposta DESCENDING cd_simu_dado  DESCENDING Simu_Parc;
RUN;

data work.aprosimu_bvo_bb_1_2;
	set work.aprosimu_bvo_bb_1_1;
	BY DESCENDING cd_simulacao_proposta DESCENDING cd_simu_dado  DESCENDING Simu_Parc;

	if first.cd_simulacao_proposta then
		n=1;
	else n+1;
run;
proc sql;
create table work.aprosimu_bvo_bb_1_2_1 as
	select
		cd_simulacao_proposta,
		avg(input(ProbabilidadeAprovacao,3.)) format commax12.2 as PcProbAproAjustada
	from aprosimu_bvo_bb_1_2
group by cd_simulacao_proposta;quit;

proc sql;
create table aprosimu_bvo_bb_1_3 as
Select *
from aprosimu_bvo_bb_1_2
where n = 1;quit;

proc sql;

create table aprosimu_bvo_bb_3 as

	Select
		a.cd_simulacao_proposta
		,a.Simu_Parc
		,a.cd_simu_dado
		,a.Decisao_preAnalise_Bvo
		,a.motivo_decisao
		,a.ProbabilidadeAprovacao
		,a.nu_proposta
		,a.cd_parceiro_bb
		,a.nuProposta
		,a.n
		,b.PcProbAproAjustada
		,case when b.PcProbAproAjustada > 0 and a.motivo_decisao = 'OK' then 1 else 0 end as AprovacaoAjustada
	from aprosimu_bvo_bb_1_3 a
		left join work.aprosimu_bvo_bb_1_2_1 b
			on a.cd_simulacao_proposta = b.cd_simulacao_proposta

;quit;

proc sql;

create table aprosimu_bvo_bb as

select * from aprosimu_bvo_bb_3;quit;
