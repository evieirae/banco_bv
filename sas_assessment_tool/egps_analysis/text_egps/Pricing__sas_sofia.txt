PROC SQL;
CREATE TABLE COMISSAO_CONTRATO AS
SELECT
  A.NuProposta,
  INTNX('quarter', A.DtContrato, '0') FORMAT YYQ. AS Trimestre,
  INPUT( cats(year(A.DtContrato), put(month(A.DtContrato), z2.)), 8.) AS Mes,
  A.DtContrato,
  A.CdParceiroComercial,
  A.Regional,
  A.Filial,
  A.Canal,
  
  CASE
    WHEN A.CdTaxaRetorno='2' THEN 1.2
    WHEN A.CdTaxaRetorno='3' THEN 2.4
    
    ELSE 0
  END AS ComissaoPadrao,
  
  COALESCE(P.PercentualComissao,
           CASE
             WHEN A.CdTaxaRetorno='2' THEN 1.2
             WHEN A.CdTaxaRetorno='3' THEN 2.4
             
             ELSE 0
           END) AS ComissaoFinal,
  A.PcTaxaRetorno AS ComissaoTotal,
  SUM(A.VrProducao) AS VrProducao
FROM INFCVS.TBSIMULACOES AS A
LEFT JOIN WORK.POLITICA_COMISSAO AS P
  ON P.Canal = A.Canal
 AND P.Produto = A.Produto
 AND (P.CdParceiroComercial IS NULL OR P.CdParceiroComercial = A.CdParceiroComercial)
 AND A.DtContrato BETWEEN P.VigenciaDe AND P.VigenciaAte
