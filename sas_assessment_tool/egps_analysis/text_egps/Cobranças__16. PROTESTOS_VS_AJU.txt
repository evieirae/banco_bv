PROC SQL;
  CREATE TABLE por_dia AS
  SELECT
      nucontrato,
      sqcontratofinanceiro,
      safra,
	  PRODUTO,
      COALESCE(DATEPART(dtprocessamento), dtprocessamento) AS dtproc FORMAT=date9.,
      MAX(Dias_atraso) AS Dias_atraso_dia
  FROM ATRASOS_HIST
  WHERE safra BETWEEN 202501 AND 202602
  GROUP BY nucontrato, sqcontratofinanceiro, safra, calculated dtproc, PRODUTO
  ;
QUIT;
PROC SQL;
  CREATE TABLE ATRASOS_ABERT_FECH AS
  SELECT
      nucontrato,
      sqcontratofinanceiro,
      safra,
	  PRODUTO,
      MAX(CASE WHEN DAY(dtproc) = 1 THEN Dias_atraso_dia END)    AS Dias_atraso_abertura,
      MAX(CASE WHEN DAY(dtproc) = 1 THEN dtproc END) FORMAT=date9. AS Dt_abertura,
      MAX(CASE WHEN dtproc = INTNX('month', dtproc, 0, 'end')
               THEN Dias_atraso_dia END) AS Dias_atraso_fechamento,
      MAX(CASE WHEN dtproc = INTNX('month', dtproc, 0, 'end')
               THEN dtproc END) FORMAT=date9. AS Dt_fechamento
  FROM por_dia
  GROUP BY nucontrato, sqcontratofinanceiro, safra, PRODUTO
  ORDER BY nucontrato, safra
;
QUIT;

CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_2 AS
SELECT
    NUCONTRATO,
    sqcontratofinanceiro,
    QTD_PARCELAS,
    QTD_PARCELAS_PAGAS,
    
    ROUND(QTD_PARCELAS_PAGAS / QTD_PARCELAS, 0.01) AS pc_amort_parc,

    
    CASE
      WHEN CALCULATED pc_amort_parc = 0                                  THEN 'a. FPD'
      WHEN CALCULATED pc_amort_parc  > 0  AND CALCULATED pc_amort_parc <= 0.10 THEN 'b. até 10%'
      WHEN CALCULATED pc_amort_parc  > 0.10 AND CALCULATED pc_amort_parc <= 0.30 THEN 'c. 10% a 30%'
      WHEN CALCULATED pc_amort_parc >= 0.31 AND CALCULATED pc_amort_parc <= 0.70 THEN 'd. 31% a 70%'
      WHEN CALCULATED pc_amort_parc  > 0.70                               THEN 'e. >70%'
      ELSE 'N/A'
    END AS fx_pc_amort_parc
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_1
;
