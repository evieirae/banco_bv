	CASE
		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN C.DtProcessamentoLiquidacao
		WHEN D.dt_baixa IS NOT NULL 					THEN D.dt_baixa

	END FORMAT=DDMMYY10. AS DT_PAGAMENTO,

		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN "Quitacao"

		WHEN F.dt_ultimo_recebimento IS NOT NULL 		THEN "Pgmt"
		ELSE "Sem Acao"
	END AS TP_PAGAMENTO

		SELECT DISTINCT
			A.Nucontrato,
			MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro_m0,
			MAX(A.nudiasatraso) as atraso_m0,
			CASE
				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO_M0

		GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato
) AS B
ON A.nucontrato = B.NUCONTRATO

		SELECT DISTINCT
			Nucontrato,
			MIN(Date(DtProcessamentoLiquidacao)) as DtProcessamentoLiquidacao

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) >= &inicio_mes_m1. and DATE(dtprocessamento) <= &fim_mes_m1.

		WHERE 	DATE(dt_baixa) > &fim_mes_m0. AND DATE(dt_baixa) <= &fim_mes_m1. and

				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) BETWEEN 1 AND 90 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_90d,

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G
ON A.NuContrato = G.cd_contrato AND A.SqContratoFinanceiro = G.sq_contrato_financeiro

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=15);
CREATE TABLE apuracao_recorrente AS
SELECT
	A.*,
	B.atraso_m0,
	B.sqcontratofinanceiro_m0,

	CASE
		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN C.DtProcessamentoLiquidacao
		WHEN D.dt_baixa IS NOT NULL 					THEN D.dt_baixa
		WHEN F.dt_ultimo_recebimento IS NOT NULL 		THEN F.dt_ultimo_recebimento
	END FORMAT=DDMMYY10. AS DT_PAGAMENTO,

	CASE
		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN "Quitacao"
		WHEN D.dt_baixa IS NOT NULL 					THEN "Reneg"
		WHEN F.dt_ultimo_recebimento IS NOT NULL 		THEN "Pgmt"
		ELSE "Sem Acao"
	END AS TP_PAGAMENTO

FROM BASE_ELEG_SEGMENTACAO AS A

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			A.Nucontrato,
			MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro_m0,
			MAX(A.nudiasatraso) as atraso_m0,
			CASE
				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE DATE(A.dtprocessamento) = &fim_mes_m0.
		GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato
) AS B
ON A.nucontrato = B.NUCONTRATO

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			Nucontrato,
			MIN(Date(DtProcessamentoLiquidacao)) as DtProcessamentoLiquidacao

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) >= &inicio_mes. and DATE(dtprocessamento) <= &fim_mes_m1.
			and DtProcessamentoLiquidacao IS NOT NULL
		group by nucontrato
) AS C

ON A.nucontrato = C.NUCONTRATO and A.Data_Execucao+1 < C.DtProcessamentoLiquidacao


LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			NuContratoGestao,
			MAX(SqContratoFinanceiro)      		AS SqContratoFinanceiro,
			NuContratoGestaoNovo,
			MAX(sqcontratofinanceironovo)     	AS sqcontratofinanceironovo,
			MAX(DATE(dt_baixa))         		AS dt_baixa,
			MAX(dt_reneg)         				AS dt_reneg,
			MAX(SAFE_CAST(vr_presente AS FLOAT64))   	AS vr_presente,
			MAX(SAFE_CAST(vr_financiado AS FLOAT64))  	AS vr_financiado,
			MAX(SAFE_CAST(Tx_Pactuada AS FLOAT64))   	AS Tx_Pactuada,
			MAX(qt_parcela)        				AS qt_parcela,
			MAX(vr_prestacao)        			AS vr_prestacao,
			MAX(FlRestruturacaoContrato)     	AS FlRestruturacaoContrato

		FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
		WHERE 	DATE(dt_baixa) > &fim_mes_m0. AND DATE(dt_baixa) <= &fim_mes_m1. and
				(dt_cancelamento is null or dt_reativacao is not null) and
				ST_RENEG = 3
		GROUP BY
			NuContratoGestao,
		    NuContratoGestaoNovo
) AS D
ON A.nucontrato = D.NuContratoGestao AND A.sqcontratofinanceiro = D.SqContratoFinanceiro and A.Data_Execucao+1 < D.dt_baixa

LEFT JOIN CONNECTION TO PERM (
	SELECT
		cd_Contrato,
		sq_contrato_financeiro,

		MAX(DATE(dt_ultimo_recebimento)) AS dt_ultimo_recebimento

	FROM bv-pdbd-prd.dvry_core.tbparcela
	WHERE st_parcela = 'Q' AND DATE(dt_ultimo_recebimento) > &fim_mes_m0. AND DATE(dt_ultimo_recebimento) <= &fim_mes_m1.
	GROUP BY 1,2
) as F
ON A.nucontrato = F.cd_Contrato and A.sqcontratofinanceiro = F.sq_contrato_financeiro and A.Data_Execucao+1 < F.dt_ultimo_recebimento

;QUIT;

PROC SORT DATA=apuracao_recorrente out=apuracao_recorrente_&anomes. NODUPKEY;
BY nucontrato sqcontratofinanceiro;
RUN;
