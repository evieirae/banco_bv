PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.tblProcessoGarantia_Projuris  AS
SELECT

 Cd_Causa as Cd_Causa
,SERIAL_IMPORT as SERIAL_IMPORT
,Pasta as Pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdCausaUnidade as CdCausaUnidade
,CdSistema as CdSistema
,CdUnidadeExyon as CdUnidadeExyon
,IdProcessoGarantia as IdProcessoGarantia
,DtCadastramento as DtCadastramento
,TipoGarantia as TipoGarantia
,DtCadastroGarantia as DtCadastroGarantia
,DtGarantia as DtGarantia
,VlLancamentoGarantia as VlLancamentoGarantia
,VlLancamentoGarantiaAtualizacao as VlLancamentoGarantiaAtualizacao
,NmGarantiaStatus as NmGarantiaStatus
,DtBaixaGarantia as DtBaixaGarantia
,TipoBaixaGarantia as TipoBaixaGarantia
,IdSeguroAvita as IdSeguroAvita
,FlSeguroAvita as FlSeguroAvita
,identificador_seguro 	as IdentificadorSeguro
,FlInconsitenciaSeguro 	as FlInconsitenciaSeguro

FROM CONNECTION TO postgres(

select
case
	when pr.serial_import like ('%INQ%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%BVT%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%CON%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%VAC%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%CTO%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRI%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRB%') then (left(pr.SERIAL_IMPORT,10))
		else pr.serial_import end       as Cd_Causa
,pr.SERIAL_IMPORT 						as SERIAL_IMPORT
,pr.pasta                               as Pasta
,pr.ID_NATUREZA							as IdNatureza
,pr.NATUREZA_UNIDADE_CUSTOM				as IdNaturezaUnidadeCustom
,null::numeric 							as CdCausaUnidade
,'p' 									as CdSistema

,case
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 2        then 'TRI'
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 1        then 'TBR'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 4        then 'CTO'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 12       then 'CON'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 13       then 'VAC'
	when pr.ID_NATUREZA = 27 and pr.NATUREZA_UNIDADE_CUSTOM = 5        then 'BVT'
	when pr.ID_NATUREZA = 32 and pr.NATUREZA_UNIDADE_CUSTOM in (7, 12) then 'INQ'
		else null end 					as CdUnidadeExyon

,pg.ID_PROCESSO_GARANTIA 				as IdProcessoGarantia
,cast(pr.DATA_CADASTRO as date)			as DtCadastramento
,upper(ltg.GARANTIA)					as TipoGarantia
,cast(pg.DATA_CADASTRO as date)			as DtCadastroGarantia
,cast(pg.DATA_INCLUSAO as date)			as DtGarantia
,pg.VALOR								as VlLancamentoGarantia
,pg.VALOR								as VlLancamentoGarantiaAtualizacao
,pg.identificador_seguro				as IdSeguroAvita
,case when pg.identificador_seguro is null
	  then 0
      else 1 end						as FlSeguroAvita
,pg.identificador_seguro  																							as identificador_seguro
,CASE WHEN pg.identificador_seguro ~ '[^0-9]' THEN 1 ELSE 0 END 													as FlInconsitenciaSeguro


	

,case when pg.REVERTIDA = 'T' then 'REVERTIDA'
	  when gc.status_garantia = 'NÃO LIBERADA' then 'NAO LIBERADA'
	  else gc.status_garantia end		as NmGarantiaStatus
,cast(gl2.DATA_LIBERACAO as date)		as DtBaixaGarantia
,upper(GL2.TIPO_BAIXA)					as TipoBaixaGarantia


from public.M_PROCESSO pr
inner join public.D_PROCESSO_GARANTIA   pg on pg.ID_PROCESSO = pr.ID_PROCESSO
left join public.LT_GARANTIA           ltg on ltg.ID_GARANTIA = pg.ID_GARANTIA
left join  public.VW_GARANTIAS_COMPLETA gc on gc.ID_PROCESSO_GARANTIA = pg.ID_PROCESSO_GARANTIA
left join (
	select
		ID_PROCESSO_GARANTIA
	    ,max(ID_PROCESSO_GARANTIA_LIBERACAO) as ID_PROCESSO_GARANTIA_LIBERACAO
	    from public.VW_GARANTIAS_LIBERACAO
	    group by ID_PROCESSO_GARANTIA) as GL on PG.ID_PROCESSO_GARANTIA = GL.ID_PROCESSO_GARANTIA
left join public.VW_GARANTIAS_LIBERACAO as GL2
	on PG.ID_PROCESSO_GARANTIA = GL2.ID_PROCESSO_GARANTIA
	and GL.ID_PROCESSO_GARANTIA_LIBERACAO = GL2.ID_PROCESSO_GARANTIA_LIBERACAO

;
);DISCONNECT FROM postgres;
RUN;
