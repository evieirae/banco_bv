PROC SQL;
CREATE TABLE BA_ELEGIBILIDADE AS SELECT
	A.CONTRATO,
	A.MUTUARIO AS NOME_CLIENTE,
	A.CPFCNPJ,
	A.IMOVEL_ESTADO AS UF,

	A.PRAZO,
	A.ATRASO_DIAS,
	A.VALOR_CONTRATO2,
	A.VALOR_ENCARGO2 as VALOR_ENCARGO,

	D.ATRASO_DATA_DE AS DT_VCTO,
	INTCK('DAY',D.ATRASO_DATA_DE,&dt_corte.) AS ATRASO_ATUAL_CALCULADO,

	CASE WHEN calculated ATRASO_ATUAL_CALCULADO=. THEN "1. em dia"
	ELSE PUT(calculated ATRASO_ATUAL_CALCULADO,FX_ATRASO.) end as FAIXA_ATRASO,

	CASE WHEN ATRASO_DIAS > 0 AND CALCULATED ATRASO_ATUAL_CALCULADO > 30 AND CALCULATED ATRASO_ATUAL_CALCULADO < 1790 THEN 1 ELSE 0 END AS FLAG_ELEG_SERASA,
	CASE WHEN ATRASO_DIAS > 0 AND CALCULATED ATRASO_ATUAL_CALCULADO > 45 AND CALCULATED ATRASO_ATUAL_CALCULADO < 1790 THEN 1 ELSE 0 END AS FLAG_ELEG_SCPC,

	CASE WHEN B.NUCONTRATO_CIG IS NOT NULL THEN 1 ELSE 0 END AS FLAG_NEGATIVADO_SCPC,
	CASE WHEN C.CNPJ_CONVENIO IS NOT NULL THEN 1 ELSE 0 END AS FLAG_NEGATIVADO_SERASA

	FROM CARTEIRA_CIG_TRATADA A LEFT JOIN ARQUIVO_CIG_SCPC B
ON A.CPFCNPJ = B.CPFCNPJ_CIG AND A.CONTRATO = B.NUCONTRATO_CIG
LEFT JOIN ARQUIVO_SERASA C
ON A.CPFCNPJ = C.CNPJ_CONVENIO 

LEFT JOIN MONTREAL D
ON A.CONTRATO=D.CONTRATO;

QUIT;

TABLE FLAG_ELEG_SCPC*FLAG_NEGATIVADO_SCPC
	FLAG_ELEG_SERASA * FLAG_NEGATIVADO_SERASA/ LIST MISSING;
RUN;



DATA BA_ELEGIBILIDADE;

FORMAT DT_REF DDMMYY10.;
DT_REF = TODAY();

		IF FLAG_ELEG_SERASA=1 AND FLAG_NEGATIVADO_SERASA=1  THEN DO;
				CRITERIO_NEG_SERASA='OK';
				ACAO_SERASA='OK';

			END;ELSE

		IF FLAG_ELEG_SERASA=1 AND FLAG_NEGATIVADO_SERASA=0 THEN DO;
			CRITERIO_NEG_SERASA='DIVERGENTE';
			ACAO_SERASA='NEGATIVAR SERASA';

			END;ELSE

		IF FLAG_ELEG_SERASA=0 AND FLAG_NEGATIVADO_SERASA=1 THEN DO;
			CRITERIO_NEG_SERASA='NEGATIVADO INDEV.';
			ACAO_SERASA='BAIXAR NEGATIVACAO';

			END;ELSE

		IF FLAG_ELEG_SERASA=0 AND FLAG_NEGATIVADO_SERASA=0 THEN DO;
			CRITERIO_NEG_SERASA='OK';
				ACAO_SERASA='OK';

			END;

	

		IF FLAG_ELEG_SCPC=1 AND FLAG_NEGATIVADO_SCPC=1 THEN DO;
			CRITERIO_NEG_SCPC='OK';
			ACAO_SCPC='OK';

			END;ELSE

		IF FLAG_ELEG_SCPC=1 AND FLAG_NEGATIVADO_SCPC=0  THEN DO;
			CRITERIO_NEG_SCPC='DIVERGENTE';
			ACAO_SCPC='NEGATIVAR SCPC';

			CRITERIO_NEG_SCPC='NEGATIVADO INDEV.';
			ACAO_SCPC='BAIXAR NEGATIVACAO';
