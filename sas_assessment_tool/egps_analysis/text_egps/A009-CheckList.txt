libname _CHECK "/projetos/riscosmercado/dados/rmone/checklist";
%let AAAAMMDD=%sysfunc(SCAN(&sysparm.,1,"/"));

%macro TrataBaseCheckList();

	PROC SQL;
		CREATE TABLE WORK.CHECKLIST_DATABASE AS
			SELECT
				*
			FROM
				_CHECK.CHECKLIST_&AAAAMMDD.
		;
	RUN;

	PROC SQL;
		CREATE TABLE WORK.CHECKLIST_DATA_HIST AS
			SELECT
				DtReferencia,
				Linhas,
				Tamanho,
				Diretorio,
				Arquivo
			FROM
				_CHECK.CHECKLIST_HISTORICO
			WHERE
				DtReferencia NOT IN (SELECT DISTINCT DtReferencia FROM WORK.CHECKLIST_DATABASE)
		;
	RUN;

	PROC SQL;
		CREATE TABLE _CHECK.CHECKLIST_HISTORICO AS
			SELECT * FROM WORK.CHECKLIST_DATABASE
			UNION ALL
			SELECT * FROM WORK.CHECKLIST_DATA_HIST
		;
	RUN;

	PROC SQL;

		CREATE TABLE _CHECK.CHECKLIST_HISTORICO AS
		SELECT
			t.DtReferencia,
			m.Tag,
			catx("/",t.Diretorio,t.Arquivo) AS ChaveDirArq,
/*			%SYSFUNC(CATX('',t.Diretorio,'/',t.Arquivo)) AS ChaveDir,*/
			t.Linhas,
			t.Tamanho,
			t.Diretorio,
			t.Arquivo
		FROM
			_CHECK.CHECKLIST_HISTORICO t
		LEFT JOIN
			_CHECK.DEPARA_MONITORAMENTO_ALGO_ONE m
			ON m.Diretorio = t.Diretorio
		WHERE
			t.Diretorio NE "" OR t.Arquivo NE ""
		ORDER BY
			t.DtReferencia DESC,
			m.Tag,
			t.Diretorio,
			t.Arquivo,
			t.Tamanho DESC
		;

	RUN;

%mend TrataBaseCheckList;

%TrataBaseCheckList();
