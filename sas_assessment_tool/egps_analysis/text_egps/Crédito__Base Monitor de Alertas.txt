proc sql;
    create table tabelas as
    select memname as tabela,
           input(substr(memname, 9, 8), yymmdd8.) as data format=date9.,
           cats('_INSIGHT.', memname) as origem 
    from dictionary.tables
    where libname = '_INSIGHT'
          and memname like 'TBBRUTA%'
    order by data desc;
quit;

proc sql;

create table tabelas_aux as select

case when tabela = 'TBBRUTA' then cats('TBBRUTA_', put(today(), yymmddn8.)) else tabela end as tabela,
case when tabela = 'TBBRUTA' then today() else data end as data format ddmmyy10.,
origem
from tabelas;
quit;

data tabelas1;
set tabelas_aux;
where data ne .;
safra=year(data)*100+month(data);
run;

proc sql;
    create table tabelas2 as
    select safra,
	max(data) as max_data format=date9.
    from tabelas1
	group by 1;
quit;

proc sql;
	create table tabelas3 as
	select a.*,
	case when a.data=b.max_data then 1 else 0 end as flag
	from tabelas1 as a
	left join tabelas2 as b
	on a.safra=b.safra;
quit;

data tabelas4;
set tabelas3;
where flag = 1 and safra >= (year(intnx('month',today(),-24))*100+month(intnx('month',today(),-24)));
