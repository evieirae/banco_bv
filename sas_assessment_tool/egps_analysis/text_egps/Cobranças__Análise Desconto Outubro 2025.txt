A.NUCONTRATO
,A.CDSITUACAOCONTABIL

,A.MAX_NUPARCELA

,CASE WHEN A.DTPROCESSAMENTOLIQUIDACAO >= A.DATA_INICIO
            AND A.DTPROCESSAMENTOLIQUIDACAO <=  A.DATA_FIM   THEN A.DTPROCESSAMENTOLIQUIDACAO

	  WHEN A.DTCESSAO_M1 >= A.DATA_INICIO
      AND A.DTCESSAO_M1 <= A.DATA_FIM                                 THEN A.DTCESSAO_M1

	  WHEN A.DTRENEG IS NOT MISSING                                   THEN A.DTRENEG
      WHEN A.QTATRASO <= 30 THEN (
      CASE WHEN (A.ATRASO_M1 <= A.DIF_DIAS AND A.DIF_DIAS_M1_M0 < A.DIF_DIAS) THEN A.DTRECEBIMENTOPROCESSAMENTO
           WHEN A.ATRASO_M1 > A.DIF_DIAS AND A.DIF_DIAS_M1_M0 < A.DIF_DIAS THEN A.DTRECEBIMENTOPROCESSAMENTO END)

           WHEN A.ATRASO_M1 > 30 AND A.DIF_DIAS_M1_M0 < A.DIF_DIAS THEN A.DTRECEBIMENTOPROCESSAMENTO END) END FORMAT DATE9. AS DATA_PAGAMENTO

	  WHEN CALCULATED DIA_DA_SEMANA = 5 THEN 'Quinta-feira'
	  WHEN CALCULATED DIA_DA_SEMANA = 6 THEN 'Sexta-feira'
	  WHEN CALCULATED DIA_DA_SEMANA = 7 THEN 'Sábado' ELSE 'Desconhecido' END AS NOME_DIA
