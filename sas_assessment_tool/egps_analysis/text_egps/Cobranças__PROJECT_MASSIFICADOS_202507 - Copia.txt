PROC SQL;
CREATE TABLE REFERENCIAS AS
SELECT
	A.*,
	B.DU,
	B.SAFRA,
	B.DT_ULT_DU
FROM
	REFERENCIAS AS A INNER JOIN (	SELECT
										A.DTPROCES,
										A.DU,
										A.SAFRA,
										MAX(B.DATA) AS DT_ULT_DU
									FROM
										DT_UTIL AS A INNER JOIN DIAS_UTEIS AS B
										ON A.SAFRA = B.SAFRA
									WHERE
										B.DU < A.DU
										AND B.FLFERIADO = 0
										AND B.FLFDS = 0
									GROUP BY
										A.DTPROCES,
										A.DU,
										A.SAFRA) AS B
	ON A.DTPROCES = B.DTPROCES
;QUIT;RUN;

	AND A.PARCELA = B.NUPARCELA

	A.NUCONTRATOGESTAO,
	A.SQFINANCEIRAVISAO,
	A.PARCELA,
	A.DTULTIMORECEBIMENTO,
	A.VRULTRECEBIMENTO ,
	A.MINPARCELAABERTA,
	A.CDAGENTERECEBEDOR, 
	B.DTVENCIMENTO,
	DATEDIFF(DAY,B.DTVENCIMENTO, A.DTULTIMORECEBIMENTO) AS ATRASO_MIN,
	CASE
		WHEN ATRASO_MIN > 0 THEN 1
		ELSE 2
		END AS TP_PGT

	AND A.SQFINANCEIRAVISAO = B.SQCONTRATOFINANCEIRO
	AND A.MINPARCELAABERTA = B.NUPARCELA

SELECT
	NUCONTRATOGESTAO,
	SQFINANCEIRAVISAO,
	DTULTIMORECEBIMENTO,
	CDAGENTERECEBEDOR, 
	SUM(VRULTRECEBIMENTO) AS VRULTRECEBIMENTO ,
	TP_PGT
INTO #RISCO_5_5
FROM #RISCO_5_4
GROUP BY
	NUCONTRATOGESTAO,
	SQFINANCEIRAVISAO,
	DTULTIMORECEBIMENTO,
	CDAGENTERECEBEDOR, 
	TP_PGT

SELECT DISTINCT
	A.*,
	CASE
		WHEN B.DTULTIMORECEBIMENTO >= DATE(A.DT_ASSES) AND (B.DTULTIMORECEBIMENTO <= DATE(A.DT_RET_ASSES) OR A.DT_RET_ASSES IS NULL ) THEN B.DTULTIMORECEBIMENTO
		ELSE NULL

		ELSE NULL
	END AS VRULTRECEBIMENTO,
	B.TP_PGT,
	B.CDAGENTERECEBEDOR, 
	CASE
		WHEN ATRASO >= 5   AND ATRASO <= 360 AND FASE = 'CP'               AND DTULTIMORECEBIMENTO IS NOT NULL AND TP_PGT = 2 THEN RISCO
		WHEN ATRASO >= 5   AND ATRASO <= 360 AND FASE = 'CP'               AND DTULTIMORECEBIMENTO IS NOT NULL AND TP_PGT = 1 THEN RISCO
		WHEN ATRASO >= 361                   AND FASE = 'CP'               AND DTULTIMORECEBIMENTO IS NOT NULL                THEN VRULTRECEBIMENTO

	    WHEN ATRASO >= 5   AND ATRASO <= 15  AND FASE = 'RENEG_CONSIGNADO' AND DTULTIMORECEBIMENTO IS NOT NULL                THEN RISCO
		WHEN ATRASO >= 16  AND ATRASO <= 180 AND FASE = 'RENEG_CONSIGNADO' AND DTULTIMORECEBIMENTO IS NOT NULL AND TP_PGT = 2 THEN RISCO
		WHEN ATRASO >= 16  AND ATRASO <= 180 AND FASE = 'RENEG_CONSIGNADO' AND DTULTIMORECEBIMENTO IS NOT NULL AND TP_PGT = 1 THEN RISCO
	    WHEN ATRASO >= 181                   AND FASE = 'RENEG_CONSIGNADO' AND DTULTIMORECEBIMENTO IS NOT NULL                THEN VRULTRECEBIMENTO
		END AS PGTO

	AND DATE(B.DTULTIMORECEBIMENTO) >= DATE(A.DT_ASSES) AND (DATE(B.DTULTIMORECEBIMENTO) <= DATE(A.DT_RET_ASSES) OR DATE(A.DT_RET_ASSES) IS NULL)

SELECT

	A.*,
	B.RISCO_CARNE,
	b.DT_PRI_VENC

	(SELECT
		A.NUCONTRATOGESTAO,

		A.VR_PRESTACAO * A.QT_PARCELA AS RISCO_CARNE

		DBRENEG..TB_RENEG_SIMULADOR A,
										(SELECT DISTINCT
												NUCONTRATOGESTAO,
												MAX(DT_RENEG) AS DT_RENEG
											FROM
												DBRENEG..TB_RENEG_SIMULADOR
											GROUP BY
												NUCONTRATOGESTAO) B
	WHERE
	A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
	AND A.DT_RENEG = B.DT_RENEG) B
ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
