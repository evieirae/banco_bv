PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=25);
CREATE TABLE GCON_&ANOMES. AS
SELECT *
	,YEAR(DTPROCESSAMENTO)*100 + MONTH(DTPROCESSAMENTO) AS SAFRA_ATRASO

FROM CONNECTION TO PERM(

WITH LISTA_CONTRATOS AS (

	SELECT DISTINCT
		NuContratoGestao
	FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador

	WHERE dt_baixa >= &dt_inicio. and dt_baixa < &dt_fim. and
	(dt_cancelamento is null or dt_reativacao is not null) and
	ST_RENEG = 3
)

	SELECT
		DATE(B.DTPROCESSAMENTO) 	AS DTPROCESSAMENTO,
		A.NuContratoGestao AS NUCONTRATO,
		MAX(B.NuDiasAtraso) as NuDiasAtraso,
		B.SQCONTRATOFINANCEIRO

	FROM LISTA_CONTRATOS AS A

	LEFT JOIN bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS B
	ON A.NuContratoGestao = B.NuContrato

	WHERE B.CDCARTEIRA IN (1,3,4,5,12)

	GROUP BY
		B.DTPROCESSAMENTO,
		A.NuContratoGestao,
		B.SQCONTRATOFINANCEIRO
)
;QUIT;

PROC SQL;
CREATE TABLE DT_BAIXA_&anomes. AS
SELECT
	NUCONTRATO,
	SQCONTRATOFINANCEIRO,
	MAX(DTPROCESSAMENTO) FORMAT=DDMMYY10. AS DT_ATRASO_RENEG
FROM GCON_&anomes.
GROUP BY 1,2
;QUIT;

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=25);

CREATE TABLE RENEG_SIMULADOR_ATRASO_&anomes. AS
SELECT
	A.*
	,GCON.NUDIASATRASO
	,GCON.DTPROCESSAMENTO
	,GCON.SAFRA_ATRASO
	,GCON2.Atraso_Baixa
	,CASE
		WHEN BCOB.CARTEIRA IS NOT NULL THEN BCOB.CARTEIRA
		ELSE BCOB2.CARTEIRA
	END AS SALDO_M0
	,CASE
		WHEN BCOB.NmProdutoRisco IS NOT NULL THEN BCOB.NmProdutoRisco
		ELSE BCOB2.NmProdutoRisco
	END AS PRODUTO
	,CASE
	    WHEN GCON.DTPROCESSAMENTO - A.dt_baixa <= 30 THEN 1
	    WHEN GCON.DTPROCESSAMENTO - A.dt_baixa <= 60 THEN 2
	    WHEN GCON.DTPROCESSAMENTO - A.dt_baixa <= 90 THEN 3
	    ELSE .
	END AS MOB

FROM RENEG_SIMULADOR_&anomes. AS A

LEFT JOIN BASE_COBRANCA as BCOB
ON A.NUCONTRATOGESTAO = BCOB.NuContrato and BCOB.SAFRA = &safra_m0.

LEFT JOIN BASE_COBRANCA as BCOB2
ON A.NUCONTRATOGESTAO = BCOB2.NuContrato and BCOB2.SAFRA = &anomes.

LEFT JOIN GCON_&anomes. AS GCON
ON A.NuContratoGestao = GCON.NUCONTRATO
AND A.SqContratoFinanceiroNovo = GCON.SQCONTRATOFINANCEIRO
AND GCON.DTPROCESSAMENTO > A.DT_BAIXA
AND GCON.DTPROCESSAMENTO <= A.DT_BAIXA + 90

LEFT JOIN (

	SELECT
		A.*,
		B.NuDiasAtraso AS Atraso_Baixa
	FROM DT_BAIXA_&anomes. 	AS A
	LEFT JOIN GCON_&anomes. AS B
	ON A.NUCONTRATO = B.NUCONTRATO AND A.DT_ATRASO_RENEG = B.DTPROCESSAMENTO
) AS GCON2
ON A.NuContratoGestao = GCON2.NUCONTRATO
AND A.SqContratoFinanceiro = GCON2.SQCONTRATOFINANCEIRO

WHERE GCON.NUDIASATRASO <> .

;QUIT;

PROC SQL;
CREATE TABLE RENEG_RUIM_&anomes. AS
SELECT
		NuContratoGestao,
		SqContratoFinanceiro,
		NuContratoGestaoNovo,
		SqContratoFinanceiroNovo,
		PRODUTO,
		CASE
			WHEN ATRASO_BAIXA <= 0    THEN 'EM DIA'
			WHEN ATRASO_BAIXA <= 4    THEN '0001 A 0004'
			WHEN ATRASO_BAIXA <= 15   THEN '0005 A 0015'
			WHEN ATRASO_BAIXA <= 30   THEN '0016 A 0030'
			WHEN ATRASO_BAIXA <= 60   THEN '0031 A 0060'
			WHEN ATRASO_BAIXA <= 90   THEN '0061 A 0090'
			WHEN ATRASO_BAIXA <= 120  THEN '0091 A 0120'
			WHEN ATRASO_BAIXA <= 150  THEN '0121 A 0150'
			WHEN ATRASO_BAIXA <= 180  THEN '0151 A 0180'
			WHEN ATRASO_BAIXA <= 360  THEN '0181 A 0360'
			WHEN ATRASO_BAIXA >  360  THEN '0360 A INF'
		ELSE 'EM DIA' END AS FX_ATRASO,
		ATRASO_BAIXA,
		dt_baixa,
		dt_reneg,
		FlRestruturacaoContrato,
		safra_reneg,
		MAX(safra_atraso) AS safra_atraso,
		MOB,
		CASE
			WHEN MAX(nudiasatraso) > 31 THEN 31
			ELSE MAX(nudiasatraso)
		END AS nudiasatraso,
		CASE
			WHEN MAX(NUDIASATRASO) > 30 AND ATRASO_BAIXA > 30  THEN 1
			ELSE 0
		END AS FL_RENEG_RUIM_AUX,
		SALDO_M0

FROM RENEG_SIMULADOR_ATRASO_&anomes.
WHERE PRODUTO IS NOT NULL

GROUP BY
		NuContratoGestao,
		SqContratoFinanceiro,
		NuContratoGestaoNovo,
		SqContratoFinanceiroNovo,
		PRODUTO,
		FX_ATRASO,
		ATRASO_BAIXA,
		dt_baixa,
		dt_reneg,
		FlRestruturacaoContrato,
		safra_reneg,
		MOB,
		SALDO_M0
;QUIT;

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=25);
CREATE TABLE RENEG_RUIM_RISCO_&ANOMES. AS
SELECT DISTINCT
	A.NuContratoGestao
	,A.SqContratoFinanceiro
	,A.NuContratoGestaoNovo
	,A.SqContratoFinanceiroNovo
	,A.PRODUTO
	,A.FX_ATRASO
	,A.ATRASO_BAIXA
	,A.dt_baixa
	,A.dt_reneg
	,A.FlRestruturacaoContrato
	,A.safra_reneg
	,A.safra_atraso
	,A.MOB
	,MAX(A.nudiasatraso) as nudiasatraso
	,A.SALDO_M0
	,MAX(C.FL_RENEG_RUIM) AS FL_RENEG_RUIM
	,GCON.DTPROCESSAMENTO
	,GCON.NUDIASATRASO AS ATRASO_DIA_ANTERIOR
	,CASE
		WHEN GCON.NUDIASATRASO BETWEEN 05 AND 09 AND ATRASO_BAIXA > 30  THEN 1
		WHEN GCON.NUDIASATRASO BETWEEN 10 AND 25 AND ATRASO_BAIXA > 30 	THEN 2
		WHEN GCON.NUDIASATRASO BETWEEN 26 AND 29 AND ATRASO_BAIXA > 30 	THEN 3
		ELSE 0
	END AS FL_RISCO_RENEG_RUIM
	,PUT((GCON.DTPROCESSAMENTO + (31-GCON.NUDIASATRASO)), DATE9.) AS DT_FUTURA_QUEBRA
	,MAX(CASE
			WHEN POL.Grupo_Final_Decisao IS NOT NULL THEN POL.Grupo_Final_Decisao
			WHEN POL2.Modelo_Final IS NOT NULL THEN POL2.Modelo_Final
		ELSE "N/A"
		END)
	AS POLITICA

FROM RENEG_RUIM_&anomes. AS A

LEFT JOIN CONNECTION TO PERM (
	SELECT
		NUCONTRATO,
		DATE(DTPROCESSAMENTO) AS DTPROCESSAMENTO,
		NUDIASATRASO
	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
	WHERE DTPROCESSAMENTO = (SELECT MAX(DTPROCESSAMENTO) FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil)
) AS GCON
ON A.NuContratoGestao = GCON.NUCONTRATO

LEFT JOIN (
	SELECT
		NuContratoGestao,
		SqContratoFinanceiro,
		FL_RENEG_RUIM_AUX as FL_RENEG_RUIM,
		MIN(MOB) AS MOB
	FROM RENEG_RUIM_&anomes.
	GROUP BY 1,2,3
) AS C
ON A.NuContratoGestao = C.NuContratoGestao AND C.MOB <= A.MOB

LEFT JOIN WORK.empilha_reneg_final_&anomes. AS POL
ON INPUT(A.NuContratoGestao, best32.) = POL.NUCONTRATO AND A.SqContratoFinanceiro = POL.SQCONTRATOFINANCEIRO


LEFT JOIN WORK.base_eleg_cont_&anomes. AS POL2
ON INPUT(A.NuContratoGestao, best32.) = POL2.NUCONTRATO AND A.SqContratoFinanceiro = POL2.SQCONTRATOFINANCEIRO


GROUP BY A.NuContratoGestao, A.MOB

;QUIT;

	A.PRODUTO,
	A.FX_ATRASO,
	A.ATRASO_BAIXA,
	A.dt_baixa,
	A.dt_reneg,
	A.FlRestruturacaoContrato,
	A.safra_reneg,
	MAX(A.FL_RENEG_RUIM) as FL_RENEG_RUIM,
	CASE
		WHEN MAX(A.FL_RENEG_RUIM) = 1 THEN MAX(B.safra_atraso)
		WHEN MAX(A.FL_RENEG_RUIM) = 0 THEN MAX(C.safra_atraso)
	END AS safra_atraso,
	CASE
		WHEN MAX(A.FL_RENEG_RUIM) = 1 THEN MAX(B.MOB)
		WHEN MAX(A.FL_RENEG_RUIM) = 0 THEN MAX(C.MOB)
	END AS MOB,

	A.SALDO_M0

LEFT JOIN (
	SELECT
		NuContratoGestao,
		SqContratoFinanceiro,
		FL_RENEG_RUIM_AUX,
		MIN(MOB) AS MOB,
		MIN(safra_atraso) AS safra_atraso
	FROM RENEG_RUIM_&anomes.
	WHERE FL_RENEG_RUIM_AUX = 1
	GROUP BY 1,2,3
) AS B
ON A.NuContratoGestao = B.NuContratoGestao

	SELECT
		NuContratoGestao,
		SqContratoFinanceiro,
		FL_RENEG_RUIM_AUX,

		MAX(safra_atraso) AS safra_atraso
	FROM RENEG_RUIM_&anomes.
	WHERE FL_RENEG_RUIM_AUX = 0
	GROUP BY 1,2,3
) AS C
ON A.NuContratoGestao = C.NuContratoGestao

GROUP BY
	A.NuContratoGestao,
	A.SqContratoFinanceiro,
	A.NuContratoGestaoNovo,
	A.SqContratoFinanceiroNovo,
	A.PRODUTO,

;QUIT;
