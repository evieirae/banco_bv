   CREATE TABLE WORK.CPF_CONTRATOS_UNICOS AS
   SELECT t1.Cd_Causa,
          t1.NuCPFCNPJ as CPFCNPJ,
            (MAX(t1.NumContrato)) AS NO_CONTRATO

proc  sql;
create table ACM.MASSIFICADO_ACORDO (compress=yes index=(NumRegSistContrGest/unique)) as
		select distinct
				con.NumRegSistContrGest,
				con.DtDetec,
				t2.CPFCNPJ,
				cto.NumContrato as NO_CONTRATO,
				t3.NuContrato,
				iq.VrPrestacao as VrPrestacaoContrato,
				iq.CdUF as UFContrato,
				T4.DtEncerramento
		from BDPO.BDPORTAL_CONSOLIDADA as con
		LEFT JOIN WORK.CPF_CONTRATOS_UNICOS t2 ON (CON.NumRegSistContrGest = t2.Cd_Causa)
		LEFT JOIN BDPORTAL.CONTRATOS_PROJURIS as cto on (con.NumRegSistContrGest=cto.Cd_Causa)
		LEFT JOIN yribeiro.tblcontratosiq iq ON (cto.NumContratoTx = iq.NuContratoText)
		LEFT JOIN BDPO.CONTRATOSFINCOMP t3 ON (cto.NumContratoTx = t3.NuContratoTx)
		LEFT JOIN ACM.MASSIFICADO T4 ON (con.NumRegSistContrGest = T4.NumRegSistContrGest)
where con.AreaReporte = 'Jurídico BV Financeira - Ações Contrárias' and
			  con.GestProc ne 'RELEVANTES' and
			  con.NaturezaAcao not in ('NÃO É AÇÃO PRINCIPAL NÃO DEVE CONSTAR') and
			  con.Motivo ne 'VAC INTIMAÇÃO' AND
			  CON.AcaoJud NOT IN ("PROCON","AGRAVO  RETIDO","GRAVO DE INSTRUMENTO", "AGRAVO DE PETIÇÃO","AGRAVO REGIMENTAL",
									"ALVARÁ","APELAÇÃO","AUTO DE INFRAÇÃO","CARTA CONVITE","EMBARGOS À EXECUÇÃO",

PROC SQL;
CREATE TABLE WORK.VARIAVEIS_PLEITO_PROCURADOR AS
SELECT
	T1.NumRegSistContrGest AS VAC_ACAO,
	T1.DtDetec AS DATA_DETECCAO_ACAO,
	T2.NumRegSistContrGest AS VAC_BASE_ENTRADAS,
	T2.ProcuradorSemAcento
FROM
	ACM.MASSIFICADO_ACORDO T1
LEFT JOIN
	(SELECT
		NumRegSistContrGest,
		CPFCNPJ,
		ProcuradorSemAcento
	FROM
		WORK.EntradasComProcuradorAjustado
	GROUP BY
		NumRegSistContrGest,
		CPFCNPJ,
		ProcuradorSemAcento) T2
ON
	T1.NumRegSistContrGest = T2.NumRegSistContrGest;
QUIT;

PROC SQL;
CREATE TABLE WORK.VARIAVEL_RECORRENCIA_PROCURADOR AS
SELECT
	VAC_ACAO,
	DATA_DETECCAO_ACAO,
	ProcuradorSemAcento,
	COUNT(DATA_MATCH) AS QtdeCasosAnteriores,
	SUM(FlagStatusPerdaZero) AS QtdeStatusPerdaZero,
	SUM(FlagStatusCondenacao) AS QtdeStatusCondenacao,
	SUM(FlagStatusAcordo) AS QtdeStatusAcordo,
	SUM(FlagStatusAdministrativo) AS QtdeStatusAdministrativo,
	SUM(FlagStatusAtiva) AS QtdeStatusAtiva
FROM
	(SELECT
		T1.VAC_ACAO,
		T1.DATA_DETECCAO_ACAO,
		T1.ProcuradorSemAcento,
		T2.ProcuradorSemAcento as PROCURADOR_MATCH,
		T2.DtDetec as DATA_MATCH,
		T2.DtEncerramento as Dt_Encerramento,
		T2.Desfecho_Status as Encerramento_Boletim,
		CASE
			WHEN T2.Desfecho_Status IS NOT NULL AND T2.Desfecho_Status = '1.Perda Zero' THEN 1
			ELSE 0
		END as FlagStatusPerdaZero,
		CASE
			WHEN T2.Desfecho_Status IS NOT NULL AND T2.Desfecho_Status = '3.Condenação' THEN 1
			ELSE 0
		END as FlagStatusCondenacao,
		CASE
			WHEN T2.Desfecho_Status IS NOT NULL AND T2.Desfecho_Status = '2.Acordo' THEN 1
			ELSE 0
		END as FlagStatusAcordo,
		CASE
			WHEN T2.Desfecho_Status IS NOT NULL AND T2.Desfecho_Status = 'ADMINISTRATIVO' THEN 1
			ELSE 0
		END as FlagStatusAdministrativo,
		CASE
			WHEN T2.Desfecho_Status IS NULL AND T2.Desfecho_Status = '' THEN 1
			ELSE 0
		END as FlagStatusAtiva
	FROM
		(SELECT
			VAC_ACAO,
			DATA_DETECCAO_ACAO,
			ProcuradorSemAcento
		FROM
			WORK.VARIAVEIS_PLEITO_PROCURADOR
		WHERE
			ProcuradorSemAcento IS NOT NULL) T1
	LEFT JOIN
		WORK.EntradasComProcuradorAjustado T2
	ON
		INTCK('day', T2.DtDetec, T1.DATA_DETECCAO_ACAO) > 0 AND
		INTCK('month', T2.DtDetec, T1.DATA_DETECCAO_ACAO) <= 12 AND
		T1.ProcuradorSemAcento = T2.ProcuradorSemAcento)
GROUP BY
	VAC_ACAO,
	DATA_DETECCAO_ACAO,
	ProcuradorSemAcento;
QUIT;

PROC SQL;
CREATE TABLE WORK.VARIAVEl_QTDE_BA_ENTRADA AS
SELECT
	T1.NumRegSistContrGest as VAC_ACAO,
	T1.NuContrato AS NU_CONTRATO_ACAO,
	T1.DtDetec,
	T2.NuContrato AS NU_CONTRATO_BASE_SAJ,
	COUNT(DISTINCT dt_asses) AS QtdeBuscaApreensaoAntesEntrada
FROM
	(SELECT DISTINCT
		NumRegSistContrGest,
		NuContrato,
		DtDetec
	FROM
		ACM.MASSIFICADO_ACORDO
	WHERE
		NuContrato IS NOT NULL) T1
LEFT JOIN
	(SELECT
		DISTINCT *
	FROM
		(SELECT
			NuContratoGestao as NuContrato format=best32.,
			dt_asses,
			cod_acao,
			nom_acao
		FROM
			YRIBEIRO.TBLSAJINICIAL
		WHERE
			cod_acao = 3)
		UNION ALL
			(SELECT
				NuContrato format=best32.,
				dt_asses,
				cod_acao,
				nom_acao
			FROM
				YRIBEIRO.TBLSAJ
			WHERE
				cod_acao = 3)) T2
ON
	T1.NuContrato = T2.NuContrato AND
	YEAR(T1.DtDetec)*10000 + MONTH(T1.DtDetec)*100 + DAY(T1.DtDetec) >
		YEAR(dt_asses)*10000 + MONTH(dt_asses)*100 + DAY(dt_asses)
GROUP BY
	T1.NumRegSistContrGest,
	T1.NuContrato,
	T1.DtDetec,
	T2.NuContrato;
QUIT;
