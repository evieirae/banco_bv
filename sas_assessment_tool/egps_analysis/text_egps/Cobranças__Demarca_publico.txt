    SELECT
         X.NUCONTRATO
        ,X.SQCONTRATOFINANCEIRO
        ,X.DATA_BASE
        ,X.QTATRASO AS ATRASO_MAXIMO
    FROM CARTEIRA_FECHAMENTO_EMP X

        SELECT
             NUCONTRATO
            ,SQCONTRATOFINANCEIRO
            ,MAX(QTATRASO) AS ATRASO_MAXIMO
        FROM CARTEIRA_FECHAMENTO_EMP
        GROUP BY
             NUCONTRATO
            ,SQCONTRATOFINANCEIRO
    ) Y
        ON X.NUCONTRATO = Y.NUCONTRATO
       AND X.SQCONTRATOFINANCEIRO = Y.SQCONTRATOFINANCEIRO
       AND X.QTATRASO = Y.ATRASO_MAXIMO
) C
    ON A.NuContrato = C.NUCONTRATO
   AND A.SqContratoFinanceiro = C.SQCONTRATOFINANCEIRO

	A.NuContrato

	C.DATA_BASE = MAX(C.DATA_BASE)
