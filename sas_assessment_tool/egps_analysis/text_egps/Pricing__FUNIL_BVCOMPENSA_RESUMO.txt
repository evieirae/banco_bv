PROC SQL;
  CREATE TABLE SHARE_PDV AS
  SELECT

'Share' as TpDados,
CASE
	 WHEN A.NuAnoMes in (202401,202402,202403) then '1T24'
	 WHEN A.NuAnoMes in (202404,202405,202406) then '2T24'
	 WHEN A.NuAnoMes in (202407,202408,202409) then '3T24'
	 WHEN A.NuAnoMes in (202410,202411,202412) then '4T24'

	 WHEN A.NuAnoMes in (202511) then '202511'
	 WHEN A.NuAnoMes in (202512) then '202512'
	 WHEN A.NuAnoMes in (202601) then '202601'
	 WHEN A.NuAnoMes in (202602) then '202602'

CASE WHEN B.NmSubEstruturaComercialC = 'Canal Concessionárias' THEN 'Canal Concessionárias'

     WHEN B.NmSubEstruturaComercialC = 'Canal Rodas e EGV - Regional II' THEN 'C1R2'

     WHEN B.NmSubEstruturaComercialC = 'Canal Rodas e EGV - Regional VI' THEN 'C2R6'
