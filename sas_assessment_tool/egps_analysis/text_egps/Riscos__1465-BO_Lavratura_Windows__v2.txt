PROC SQL;
CREATE TABLE work.tblBaseExp AS
SELECT
 u.*
,case
	when i.NuContratoText is null and o.ogbvNuContrato is not null then 'Contrato não localizado e garantia localizada'
	when i.NuContratoText is null and o.ogbvNuContrato is null then 'Contrato e garantia não localizados'
	when i.NuContratoText is not null and o.ogbvNuContrato is not null then 'Contrato e garantia localizados'
	when i.NuContratoText is not null and o.ogbvNuContrato is null then 'Contrato localizado e garantia não localizada'
end as DsLocalizado
,case when i.NuContratoText is not null then 'Sim' else 'Não' end as FlContratoIQLocalizado
,case when o.ogbvNuContrato is not null then 'Sim' else 'Não' end as FlGarantiaLocalizada
,today() as DtProcessamento format ddmmyy8.
,i.*
,o.*
,intnx('day', DATA_CONTRATO, 21916)        AS  DATA_CONTRATO2C
,intnx('day', DtContrato, 21916)           AS  DtContrato2C
,intnx('day', DtInicioVigencia, 21916)     AS  DtInicioVigencia2C
,intnx('day', DtCancelamento, 21916)       AS  DtCancelamento2C
,intnx('day', DtFimVigencia, 21916)        AS  DtFimVigencia2C
,intnx('day', DtProximoVencimento, 21916)  AS  DtProximoVencimento2C
,intnx('day', DtLiquidacaoEfetiva, 21916)  AS  DtLiquidacaoEfetiva2C
,intnx('day', DtUltimoVencimento, 21916)   AS  DtUltimoVencimento2C
,intnx('day', DtPrimeiroVencimento, 21916) AS  DtPrimeiroVencimento2C
,intnx('day', DtLiquidacao, 21916)         AS  DtLiquidacao2C
,intnx('day', today(), 21916)              AS  DtProcessamento2C

FROM 		WORK.TBLPlanUsuario 			as u
left JOIN	work.tblContratosIQ 			as i 	ON i.NuContratoText = u.NuContratoText
left JOIN	work.tblObjGarantiaBemVeiculo	as o 	ON o.ogbvNuContrato = i.NuContrato
