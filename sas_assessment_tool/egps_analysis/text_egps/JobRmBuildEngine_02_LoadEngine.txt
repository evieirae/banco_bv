				TABLE_OUT(in=c)
				LIB_OUT(in=d);

		by 		CODIGO;
		attrib LIB_IN_ANTE 		LENGTH=$1024 FORMAT=$1024. INFORMAT=$1024.;
		attrib TABLE_IN_ANTE 	LENGTH=$32 FORMAT=$32. INFORMAT=$32.;
		attrib LIB_OUT_ANTE 	LENGTH=$1024 FORMAT=$1024. INFORMAT=$1024.;
		attrib TABLE_OUT_ANTE 	LENGTH=$32 FORMAT=$32. INFORMAT=$32.;

		attrib TpInput 			LENGTH=$3 FORMAT=$3. INFORMAT=$3.;
		attrib TpOutput 		LENGTH=$3 FORMAT=$3. INFORMAT=$3.;
		attrib EXTENSAO_IN 		LENGTH=$20 FORMAT=$20. INFORMAT=$20.;
		attrib EXTENSAO_OUT 	LENGTH=$20 FORMAT=$20. INFORMAT=$20.;

		attrib DT_IN 			LENGTH=8 FORMAT=DATETIME20. INFORMAT=DATETIME20.;
		attrib DT_OUT 			LENGTH=8 FORMAT=DATETIME20. INFORMAT=DATETIME20.;

		if LIB_IN ne "" then LIB_IN_ANTE=LIB_IN;
		if LIB_IN eq "" then LIB_IN=LIB_IN_ANTE;

		if TABLE_IN ne "" then TABLE_IN_ANTE=TABLE_IN;
		if TABLE_IN eq "" then TABLE_IN=TABLE_IN_ANTE;

		if LIB_OUT ne "" then LIB_OUT_ANTE=LIB_OUT;
		if LIB_OUT eq "" then LIB_OUT=LIB_OUT_ANTE;

		if TABLE_OUT ne "" then TABLE_OUT_ANTE=TABLE_OUT;
		if TABLE_OUT eq "" then TABLE_OUT=TABLE_OUT_ANTE;

		CdOrdemAssunto="&CdOrdemAssunto.";
		CdParalelo="&CdParalelo.";
		NmJOBSAS="&NmJOBSAS.";
		VlSysparm="&VlSysparm.";

		TpOutput = "&TpOutput.";

		call symputx('_LIB_IN', compress(LIB_IN),'G');

		call symputx('_LIB_OUT', compress(LIB_OUT),'G');
		call symputx('_TABLE_OUT', compress(TABLE_OUT),'G');
		call symputx('_TpInput', compress(TpInput),'G');
		call symputx('_TpOutput', compress(TpOutput),'G');

		if TpInput eq "tab" then do;
			EXTENSAO_IN = ".sas7bdat";
			call execute('%DictionaryVerify2;');
		end;

			EXTENSAO_IN = ".csv";

		if TpOutput eq "tab" then do;
			EXTENSAO_OUT = ".sas7bdat";
		end;
		else if TpOutput eq "xml" then
			EXTENSAO_OUT = ".xml";
		else if TpOutput eq "csv" then
			EXTENSAO_OUT = ".csv";

		by TABLE_IN;
	run;

	data LISTA_&ordem;
		merge 	LISTA_&ordem(in=a)
				full_verify(in=b);
		by 		TABLE_IN;
		if a=1 then output;

	proc sql;
		create table ENTRADAS AS
		select DISTINCT CODIGO from LISTA_&ordem;

		create table ENTRADAS_REF AS
		select DTREF,COUNT(*) AS QTD
		from LISTA_&ordem
		group by DTREF
		having qtd >= &qtd_entradas.;

		CREATE TABLE ENGINE.LISTA_&ordem._consol AS
		SELECT 	DISTINCT
				A.*
		FROM 	LISTA_&ordem A INNER JOIN ENTRADAS_REF B
				ON (A.DTREF = B.DTREF);

		create table LISTA_STATUS_01 as

			CdOrdemAssunto,
			CdParalelo,
			NmAssunto,
			NmJOBSAS,
			DTREF,
			VlSysparm,
			"S" as StExecutaJOB
		from ENGINE.LISTA_&ordem._consol
		where StDatas = "S"
		order by
			CdOrdemAssunto,
			CdParalelo,
			DTREF;
