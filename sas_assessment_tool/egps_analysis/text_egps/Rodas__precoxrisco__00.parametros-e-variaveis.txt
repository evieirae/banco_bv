/* STEP 00.01 - Declarar variáveis globais */
%include '/infogrc/utils/pwd.sas';
%include '/infogrc/utils/IQ.sas';
%CRIAVARDB;
/* Períodos consultados */
%let inicio = 20250101;
%let final  = 20301231;

/* STEP 00.02 - Declarar libnames que serão usadas durante o processo */
%include '/infogrc/utils/libname.sas';
LIBNAME BA_HOML '/projetos/MISV/data/bases';
LIBNAME TMP '/projetos/MISV/data/bases/tmp';
LIBNAME BA_DADOS '/riscosbvf/BA/bases_gerenciais';
LIBNAME CRM_MIS '/riscosbvf/CRM/SAS/DM_MIS';
LIBNAME CRM_ADD '/riscosbvf/CRM/SAS/AddIn';
LIBNAME INFGER '/infogrc/precoxrisco';

/* STEP 00.03 - Formatar variáveis */
data _null_;
    anoAtual   = year(today());

    /* Seus formatos existentes */
    inicioIq   = "'" || put(&inicio., 8.) || "'";
    finalIq    = "'" || put(&final., 8.)  || "'";
    inicioCtr  = "'" || substr(put(&inicio., 8.), 1, 4) || "-" ||
                        substr(put(&inicio., 8.), 5, 2) || "-" ||
                        substr(put(&inicio., 8.), 7, 2) || "'";
    finalCtr   = "'" || substr(put(&final., 8.), 1, 4)  || "-" ||
                        substr(put(&final., 8.), 5, 2)  || "-" ||
                        substr(put(&final., 8.), 7, 2)  || "'";

    safraInicio = input(substr(put(&inicio., 8.), 1, 6), 6.);
    inicioDate  = input(put(&inicio., 8.), yymmdd8.);

    /* NOVA macro para uso no SPI (SQL pass-through):
       - Versão com aspas simples e formato AAAA-MM-DD
       - Versão sem aspas
    */
    InicioGCp_q  = quote( put( input("&inicio.", yymmdd8.), yymmdd10.) ); /* -> '2020-01-01' */
    InicioGCp_nq =       put( input("&inicio.", yymmdd8.), yymmdd10.  );  /* ->  2020-01-01  */

    /* Gravar macros */
    call symputx('anoAtual',        put(anoAtual, 4.));
    call symputx('inicioIq',        inicioIq);
    call symputx('finalIq',         finalIq);
    call symputx('inicioCtr',       inicioCtr);
    call symputx('finalCtr',        finalCtr);
    call symputx('safraInicio',     safraInicio);
    call symputx('inicioDate',      inicioDate);
    call symputx('InicioGCp',       InicioGCp_q);   /* principal: com aspas, pronta para SPI */
    call symputx('InicioGCp_nq',    InicioGCp_nq);  /* opcional: sem aspas */
run;
