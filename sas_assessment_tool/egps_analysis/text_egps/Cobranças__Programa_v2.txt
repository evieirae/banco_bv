CREATE TABLE
 PERF_ATUAL AS
SELECT DISTINCT
 A.*
 ,B.CDPESSOA
 ,C.SqEndereco
 ,C.CDUF
FROM
 PERF_BNDU_2 A
LEFT JOIN
 CONNECTION TO PERM(
 SELECT DISTINCT
       CAST(Z.CD_CONTRATO AS INT64) AS NUCONTRATO,
   Z.CD_PESSOA AS CDPESSOA,
   Z.DT_CONTRATO AS DTCONTRATO,
   Z.CD_CPF_CNPJ AS NUCPFCNPJ

FROM bv-pdbd-prd.dvry_core.tbcontrato AS Z
WHERE Z.CD_PRODUTO IN ('12', '14')) B
ON
 INPUT(A.NUCONTRATO,17.) = B.NUCONTRATO

LEFT JOIN
 CONNECTION TO PERM
 (SELECT DISTINCT
  Z.DSLOGRADOURO,
  Z.NULOGRADOURO,
  Z.DSCOMPLEMENTOLOGRADOURO,
  Z.CDCEP,
  Z.DSBAIRRO,
  Z.NMCIDADE,
  Z.CDUF,
  Z.CDPESSOA,
  Z.SqEndereco

FROM bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS Z
WHERE Z.CdUf <> ""

GROUP BY
  Z.DSLOGRADOURO,
  Z.NULOGRADOURO,
  Z.DSCOMPLEMENTOLOGRADOURO,
  Z.CDCEP,
  Z.DSBAIRRO,
  Z.NMCIDADE,
  Z.CDUF,
  Z.CDPESSOA,
  Z.SqEndereco
HAVING Z.SqEndereco = MAX(Z.SqEndereco)
) C
ON
 B.CDPESSOA = C.CDPESSOA

;quit;

	A.*

CREATE TABLE
 PERF_ATUAL AS
SELECT DISTINCT
 A.*
 ,B.CDPESSOA
 ,C.SqEndereco
 ,C.CDUF
FROM
 PERF_BNDU_2 A
LEFT JOIN
 CONNECTION TO PERM(
 SELECT DISTINCT
       CAST(Z.CD_CONTRATO AS INT64) AS NUCONTRATO,
   Z.CD_PESSOA AS CDPESSOA,
   Z.DT_CONTRATO AS DTCONTRATO,
   Z.CD_CPF_CNPJ AS NUCPFCNPJ

FROM bv-pdbd-prd.dvry_core.tbcontrato AS Z
WHERE Z.CD_PRODUTO IN ('12', '14')) B
ON
 INPUT(A.NUCONTRATO,17.) = B.NUCONTRATO

LEFT JOIN
 CONNECTION TO PERM
 (SELECT DISTINCT
  Z.DSLOGRADOURO,
  Z.NULOGRADOURO,
  Z.DSCOMPLEMENTOLOGRADOURO,
  Z.CDCEP,
  Z.DSBAIRRO,
  Z.NMCIDADE,
  Z.CDUF,
  Z.CDPESSOA,
  Z.SqEndereco

FROM bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS Z
WHERE Z.CdUf <> ""

GROUP BY
  Z.DSLOGRADOURO,
  Z.NULOGRADOURO,
  Z.DSCOMPLEMENTOLOGRADOURO,
  Z.CDCEP,
  Z.DSBAIRRO,
  Z.NMCIDADE,
  Z.CDUF,
  Z.CDPESSOA,
  Z.SqEndereco
HAVING Z.SqEndereco = MAX(Z.SqEndereco)
) C

 B.CDPESSOA = C.CDPESSOA

               CASE WHEN T1.NMPRODUTORISCO = "CVG_LEVES" THEN "CVG LEVES"
                    WHEN T1.NMPRODUTORISCO = "CVG_PESADOS" THEN "CVG PESADOS" ELSE T1.NMPRODUTORISCO END AS NMPRODUTORISCO,

			   T1.FX_ATRASO_NEW,

			   CASE WHEN B.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FLAG_PULADAS,
			   CASE WHEN C.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FLAG_VENDA,
			   FLAG_UFS_ACAO,

			    SUM(CASE WHEN T1.tipo_pag_v2 = 'Atualizacao' THEN 1 ELSE 0 END) AS QTD_ATUALIZACAO,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'Quitacao' THEN 1 ELSE 0 END) AS QTD_QUITACAO,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'pagamento parcial' THEN 1 ELSE 0 END) AS QTD_PMT,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'Reneg' THEN 1 ELSE 0 END) AS QTD_RENEG,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'sem acao' THEN 1 ELSE 0 END) AS QTD_SEM_ACAO,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'BNDU' THEN 1 ELSE 0 END) AS QTD_BNDU,

				SUM(CALCULATED QTD_ATUALIZACAO, CALCULATED QTD_QUITACAO, CALCULATED QTD_PMT, CALCULATED QTD_RENEG, CALCULATED QTD_BNDU) AS QTD_I1,
				SUM(CALCULATED QTD_ATUALIZACAO, CALCULATED QTD_QUITACAO) AS QTD_I2,

				SUM(CASE WHEN T1.tipo_pag_v2 = 'Atualizacao' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_ATUALIZACAO,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'Quitacao' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_QUITACAO,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'pagamento parcial' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_PMT,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'Reneg' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_RENEG,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'BNDU' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_BNDU,
				SUM(CASE WHEN T1.tipo_pag_v2 = 'sem acao' THEN T1.CARTEIRA ELSE 0 END) AS CARTEIRA_SEM_ACAO,

				SUM(CALCULATED CARTEIRA_ATUALIZACAO, CALCULATED CARTEIRA_QUITACAO, CALCULATED CARTEIRA_PMT, CALCULATED CARTEIRA_RENEG, CALCULATED CARTEIRA_BNDU) AS CARTEIRA_I1,
				SUM(CALCULATED CARTEIRA_ATUALIZACAO, CALCULATED CARTEIRA_QUITACAO) AS CARTEIRA_I2,
				COUNT(T1.NUCONTRATOGESTAO) AS QTD_TOTAL,
				SUM(T1.CARTEIRA) AS SUM_CARTEIRA_TOTAL
				,(CALCULATED SUM_CARTEIRA_TOTAL - CALCULATED CARTEIRA_SEM_ACAO) AS CARTEIRA_I1_2
				,'' AS Pivô

	T1.StOperacao <> 'P'

SAFRA_M1,
