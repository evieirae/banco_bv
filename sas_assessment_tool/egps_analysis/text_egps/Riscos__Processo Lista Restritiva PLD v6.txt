   CREATE TABLE BS_PROCESSOS_MATRIZ_ND AS
   SELECT t1.DOCUMENTO,
          t1.Nome,
          DATEPART('Data da criacao'n) FORMAT DATE9. AS DATA_NOVA,
		  t1.'Status da consulta'n,
		  'J' AS tppesoa

      FROM PLD.BS_PROCESSOS_MATRIZ_PJ_MOTIVOS t1

UNION ALL

   SELECT input(t1.DOCUMENTO,best14.) format best14. as DOCUMENTO,
          t1.Nome,
           DATEPART('Data da criacao'n) FORMAT DATE9. AS DATA_NOVA,
		  t1.'Status da consulta'n,
		  'F' AS tppesoa

      FROM PLD.BS_PROCESSOS_MATRIZ_MOTIVOS t1;
