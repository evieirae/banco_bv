P.PASTA,

G.GARANTIA,

INNER JOIN D_PROCESSO_GARANTIA_LIBERACAO PGL ON PGL.ID_PROCESSO_GARANTIA = PG.ID_PROCESSO_GARANTIA

select * from connection to postgres (

M.MUNICIPIO
