%include '/infogrc/utils/all-includes.sas';

LIBNAME BL '/infogrc/Base_Lojas';

%CRIAVARDB;
PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE BaseLojasAtivas AS
SELECT * FROM connection to dbcon
(

SELECT
	pc.CdParceiroComercial AS CdConvenio
	,pe.NuCpfCnpj
	,pc.NmFantasia AS NmConvenio
	,pcc.cdcolaborador AS CdColaboradorConvenio
	,Col.CdOperador AS CdOperadorConvenio
	,Col.NmColaborador AS NmOpConvenio
  	,F.CdGerenteRegional
	,Col1.NmColaborador AS NmRegional
	,F.cdGerenteComercial
	,Col2.NmColaborador AS NmGerenteComercial
	,f.CdFilial
	,DsFilial
	,PC.flativo
	,MAX(PC.DtUltimaAtualizacao) AS MaxAtu
	,pc.CdTipoParceiroComercial
	,TPC.NmTipoParceiroComercial
	,PC.NmFantasia
	,Pe.NmPessoa
	,PC.DtCadastro
	,pcc.TsReplicacao AS TsReplicacao_ParcComerColab

INTO #BVMCN
FROM dba.TbParceiroComercial PC

	INNER JOIN 	dba.TbPessoa 						 PE 		ON PE.CdPessoa 			   		=	 pc.CdPessoa
	LEFT JOIN 	dbcor.Tbparceirocomercialcolaborador pcc 		ON pcc.CdParceiroComercial 		= 	 pc.CdParceiroComercial
	LEFT JOIN 	dba.TBcolaborador 					 Col    	ON Col.Cdcolaborador 	   		=    pcc.cdcolaborador
	LEFT JOIN 	DBA.TbRegiao 						 R			ON Pc.CdRegiao			   		= 	 R.cdregiao
	LEFT JOIN	DBA.TbFilial						 F			ON F.CdFilial			   		= 	 R.CdFilial
	LEFT JOIN 	DBCOR.TBTIPOPARCEIROCOMERCIAL		 TPC 		ON TPC.CdTipoParceiroComercial 	= 	 PC.CdTipoParceiroComercial
	LEFT JOIN 	DBA.TbColaborador					 Col1   	ON Col1.CdColaborador      		= 	 F.CdGerenteRegional
	LEFT JOIN 	DBA.TbColaborador					 Col2   	ON Col2.CdColaborador      		= 	 F.cdGerenteComercial

WHERE
	TPC.CdTipoParceiroComercial IN (2, 5)
	AND F.DsFilial NOT LIKE 'CP_%'
	AND F.DsFilial NOT LIKE 'BV_%'
/*--	AND PC.cdparceirocomercial = 220639*/

GROUP BY
	pc.CdParceiroComercial
	,pe.NuCpfCnpj
	,pc.NmFantasia
	,pcc.cdcolaborador
	,Col.CdOperador
	,Col.NmColaborador
	,DsFilial
	,f.cdfilial
	,PC.flativo
	,pc.CdTipoParceiroComercial
	, TPC.NmTipoParceiroComercial
	,f.CdGerenteRegional
	,Col1.NmColaborador
	,F.cdGerenteComercial
	,Col2.NmColaborador
	,PC.NmFantasia
	,PC.DtCadastro
	,PE.NmPessoa
	,TsReplicacao_ParcComerColab

ORDER BY
DsFilial
,pc.CdParceiroComercial

SELECT
B.NmRegional,
B.DsFilial AS Filial,
B.NmGerenteComercial,
B.NmOpConvenio AS GerenteRelacionamento,
b.CdColaboradorConvenio,
B.CdOperadorConvenio AS URA,
B.CdFilial,
B.CdConvenio AS CodigoLoja,
B.NmPessoa AS NomeLoja,
B.NmFantasia,
B.DtCadastro,
B.flativo AS Ativo,
B.nucpfcnpj AS CNPJ,
B.NmTipoParceiroComercial AS Parceiro,
b.TsReplicacao_ParcComerColab,
today() as DtAtualizacaoBase

FROM #BVMCN B

WHERE B.flativo = 'S'

ORDER BY b.TsReplicacao_ParcComerColab DESC, B.DsFilial, b.CdColaboradorConvenio

)
;run;

DATA WORK.BASELOJASATIVAS;
SET WORK.BASELOJASATIVAS;
FORMAT TsReplicacao_ParcComerColab datetime19.;
;RUN;

/*DATA BL.BASELOJASATIVAS;
SET WORK.baselojasativas;
;RUN;*/

PROC SQL;
DELETE FROM BL.BASELOJASATIVAS_HIST
WHERE DtAtualizacaoBase = TODAY();
;RUN;

PROC APPEND BASE=BL.BASELOJASATIVAS_HIST DATA=WORK.BaseLojasAtivas FORCE;RUN;

PROC SQL;
CREATE TABLE VALIDA AS
SELECT
DtAtualizacaoBase,
MAX(TsReplicacao_ParcComerColab) AS Max_TsReplicacao_ParcComerColab FORMAT datetime19.,
count(DISTINCT CodigoLoja) as QtdLojas FORMAT COMMAX20.0

FROM BL.BASELOJASATIVAS_HIST

GROUP BY 1
;RUN;
