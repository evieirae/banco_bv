PROC SQL;
	CREATE TABLE BASE_PME AS
	SELECT
		&REF. AS mes,
		COMPRESS(TRANWRD(DsProduto, "CESSÃO DE PORTFÓLIO ", "")) AS Portfolio,
		SUM(VrSaldo)/1000000 AS SALDO_CARTEIRA FORMAT=COMMAX25.2

	WHERE DsTipoCarteira eq 'REGULAR' AND (UPCASE(DsProduto) CONTAINS ("PORTF") OR UPCASE(DsProduto) CONTAINS ("MAND COB TM"))

	ORDER BY CALCULATED SALDO_CARTEIRA DESC;

	CREATE TABLE BASE_PME_&REF. AS
	SELECT *, MONOTONIC() AS RANKING FROM BASE_PME WHERE CALCULATED RANKING LE 3

	UNION ALL SELECT mes, "TOTAL" AS Portolio, sum(SALDO_CARTEIRA) AS SALDO_CARTEIRA FORMAT=COMMAX25.2 FROM BASE_PME GROUP BY 1,2

	;

proc sql;

			else 'CORE' end as Segmento
		,case when tipo_carteira = 'SMES' or tipo_carteira_gerencial = 'SMES' then 'PM/UM'
			  when tipo_carteira in ('GPR/AE') then tipo_carteira_gerencial
			  when tipo_carteira in ('PROJECT FINANCE') then tipo_carteira_financas

		,case when calculated Atraso > 90 then 'Over90'
			  when calculated Atraso > 60 then 'Over60'
			  when calculated Atraso > 30 then 'Over30'
			  when calculated Atraso > 15 then 'Over15'
			  else 'Em dia' end as Fx_Atraso

      DsGrupoProduto not in ("Operações de Garantias" "Hedge" "Mercado Bancário" "TVM") and DsProduto ne ("Stand-by")

		AnoMes_Referencia
		,Segmento
		
		,sum(Saldo) as Saldo 									format commax26.2

		,sum(case when Atraso>30 then Saldo end) as Over30		format commax26.2
		,sum(case when Atraso>60 then Saldo end) as Over60		format commax26.2
		,sum(case when Atraso>90 then Saldo end) as Over90 		format commax26.2
		,calculated over15 / calculated saldo as percent_over15 format percent10.2
		,calculated over30 / calculated saldo as percent_over30 format percent10.2
		,calculated over60 / calculated saldo as percent_over60 format percent10.2
		,calculated over90 / calculated saldo as percent_over90 format percent10.2
from base1
group by 1,2;

		,sum(case when Atraso>15 then Saldo end) as Over15		format commax26.2

from base1

%macro calcular_over (anomes);
proc sql;
create table base1 as
select AnoMes_Referencia
		,case when tipo_carteira = 'SMES' or tipo_carteira_gerencial = 'SMES' then 'PM/UM'
			else 'CORE' end as Segmento
		,case when tipo_carteira = 'SMES' or tipo_carteira_gerencial = 'SMES' then 'PM/UM'
			  when tipo_carteira in ('GPR/AE') then tipo_carteira_gerencial
			  when tipo_carteira in ('PROJECT FINANCE') then tipo_carteira_financas
			  when tipo_carteira in ('CORRETORA') then 'CORPORATE'
			  else tipo_carteira end as SegmentoCliente
		,DsGrupoEconomicoRisco
		,NuObjetoNegociacao
		,sum(VrSaldo) as Saldo
		,max(NumDiasVencidos) as Atraso
		,case when calculated Atraso > 90 then 'Over90'
			  when calculated Atraso > 60 then 'Over60'
			  when calculated Atraso > 30 then 'Over30'
			  when calculated Atraso > 15 then 'Over15'
			  else 'Em dia' end as Fx_Atraso

from miscred.base_unica_atacado_&anomes.
where DsTipoCarteira = 'REGULAR' and
      DsGrupoProduto not in ("Operações de Garantias" "Hedge" "Mercado Bancário" "TVM") and DsProduto ne ("Stand-by")
group by 1,2,3,4,5;

create table OVER_&ANOMES. as
select
		AnoMes_Referencia
		,Segmento
		
		,sum(Saldo) as Saldo 									format commax26.2
		,sum(case when Atraso>15 then Saldo end) as Over15		format commax26.2
		,sum(case when Atraso>30 then Saldo end) as Over30		format commax26.2
		,sum(case when Atraso>60 then Saldo end) as Over60		format commax26.2
		,sum(case when Atraso>90 then Saldo end) as Over90 		format commax26.2
		,calculated over15 / calculated saldo as percent_over15 format percent10.2
		,calculated over30 / calculated saldo as percent_over30 format percent10.2
		,calculated over60 / calculated saldo as percent_over60 format percent10.2
		,calculated over90 / calculated saldo as percent_over90 format percent10.2
from base1
group by 1,2;

create table OVER_GRUPOS_&ANOMES. as
select
		AnoMes_Referencia
		,Segmento
		,DsGrupoEconomicoRisco
		,sum(case when Atraso>90 then Saldo end) as Over90 		format commax26.2
		,sum(case when Atraso>60 then Saldo end) as Over60		format commax26.2
		,sum(case when Atraso>30 then Saldo end) as Over30		format commax26.2
		,sum(case when Atraso>15 then Saldo end) as Over15		format commax26.2

from base1
group by 1,2,3;
quit;




proc sql;

    select
		anomes_referencia,
        segmento,
        sum(case when atraso > 15 then 1 else 0 end)/1000 as atraso_15,

    from
        base1
    group by
        anomes_referencia, segmento;

%MACRO Inads(dt_ref, dt_ref30, dt_ref60, dt_ref90);

data _null_;

anomes30 = input(substr(put(&dt_ref30.,8.), 1,6),8.);

proc sql;
	create table over_&dt_ref. as
	select  input(substr(put(&dt_ref.,8.), 1,6),8.) as Ref
			,(YEAR(a.dt_ven)*100+MONTH(a.dt_ven)) as Safra_Venc
			,a.Tp_Portfolio
			,a.ds_portfolio
			,a.Classificação
			,sum(a.vr_tit) as Saldo format commax22.

			,sum(case when b.atraso gt 30 then a.vr_tit else 0 end) as Inad30 format commax22.

			%if &dt_ref60. ne . %then %do;
			,sum(case when c.atraso gt 60 then a.vr_tit else 0 end) as Inad60 format commax22.
			%end;

			%if &dt_ref90. ne . %then %do;
			,sum(case when d.atraso gt 90 then a.vr_tit else 0 end) as Inad90 format commax22.
			%end;

from MONITRCA.BASE_PME_&dt_ref. a
left join MONITRCA.BASE_PME_&dt_ref30. b on a.nr_nos_nr = b.nr_nos_nr

%if &dt_ref60. ne . %then %do;
left join MONITRCA.BASE_PME_&dt_ref60. c on a.nr_nos_nr = c.nr_nos_nr
%end;

%if &dt_ref90. ne . %then %do;
left join MONITRCA.BASE_PME_&dt_ref90. d on a.nr_nos_nr = d.nr_nos_nr
%end;

where a.Classificação not in ('aux_resumo') and a.Classificação not contains ('PEAC') and
		(YEAR(a.dt_ven)*100+MONTH(a.dt_ven)) in (&anomes.)
group by 1,2,3,4,5;

create table inad_resumo_&dt_ref. as
select *
		,sum(Inad30)/sum(Saldo) as Percent_Inad30 format percent10.2

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60)/sum(Saldo) as Percent_Inad60 format percent10.2
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90)/sum(Saldo) as Percent_Inad90 format percent10.2
		%end;

from over_&dt_ref.
group by 1,2,3,4,5
union all
select Ref
		,Safra_Venc
		,'' as Tp_Portfolio
		,'' as ds_portfolio
		,'Total' as Classificação
		,sum(saldo) as Saldo format commax22.

		,sum(Inad30) as Inad30 format commax22.

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60) as Inad60 format commax22.
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90) as Inad90 format commax22.
		%end;

		,sum(Inad30)/sum(saldo) as Percent_Inad30 format percent10.2

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60)/sum(saldo) as Percent_Inad60 format percent10.2
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90)/sum(saldo) as Percent_Inad90 format percent10.2
		%end;

from over_&dt_ref.
group by 1,2,3,4,5;
QUIT;
%MEND;
