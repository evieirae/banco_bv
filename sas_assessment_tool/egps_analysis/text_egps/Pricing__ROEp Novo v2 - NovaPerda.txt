%macro BaseMediaInad(group_vars, NomeMes , DataMes, DeltaMes);
PROC SQL;
CREATE TABLE BaseMEDIA_Inad_&NomeMes AS
SELECT DISTINCT
&DataMes AS DataContrato FORMAT DDMMYY10.,
&group_vars.,
Produto,

SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30) AS Inad30Media,
SUM(VrParcelaEver30_Mob3)/SUM(VrParcelaElegivelEver30_Mob3) AS Ever30Medio,

CASE WHEN Produto = "Leves" THEN
((SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30)*1.3) * Beta) + Alpha
ELSE ((SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30)) * Beta) + Alpha END AS PerdaCalcINADMedia,

CASE WHEN Produto = "Leves" THEN
((SUM(VrParcelaEver30_Mob3)/SUM(VrParcelaElegivelEver30_Mob3)) * Beta) + Alpha
ELSE 0 END AS PerdaCalcEverMedio,

SUM(VrParcelaInad30) AS VrParcelaInad30Media,
SUM(VrParcelaElegivelInad30) AS VrParcelaElegivelInad30Media,
SUM(VrParcelaEver30_Mob3) AS VrParcelaEver30Medio,
SUM(VrParcelaElegivelEver30_Mob3) AS VrParcelaElegivelEver30Medio

FROM PRODUCAO  as A
WHERE DataContrato >= INTNX ('MONTH',&DataMes,&DeltaMes) AND DataContrato < &m3
GROUP BY 1,&group_vars.,Produto;RUN;
%mend BaseMediaInad;

%macro join_bases(group_vars3);
    %let n = %sysfunc(countw(&group_vars3, %str(,)));
    %let join_clause = ;
    %let select_clause = ;
    %let group_by_clause = ;

    %do i = 1 %to &n;
        %let var = %scan(&group_vars3, &i, %str(,));
        %if &i = 1 %then %do;
            %let join_clause = A.&var = B.&var;
            %let select_clause = A.&var;
            %let group_by_clause = A.&var;
        %end;
        %else %do;
            %let join_clause = &join_clause and A.&var = B.&var;
            %let select_clause = &select_clause, A.&var;
            %let group_by_clause = &group_by_clause, A.&var;
        %end;
    %end;

    CREATE TABLE PROD_ROE AS

    SUM(A.QntdContratos) AS QntdContratos,
