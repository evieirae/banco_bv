PROC SQL;

	CREATE TABLE INFOS_SACADOS_PF_V4 AS SELECT

		A.*
		,CASE WHEN A.'Restriçoes'n = 'Sem Restrições' THEN 100
			WHEN A.'Restriçoes'n = 'Com Restrições (não relevantes)' THEN 50
			ELSE 0 END AS Peso_Restricao

		,CASE WHEN A.'Comercial'n = 'Ruim' THEN 0
			ELSE 100 END AS Peso_Comercial

		,CASE WHEN A.RISCO_PRODUTIVO = "Muito Baixo" THEN 100
			WHEN A.RISCO_PRODUTIVO = "Baixo" THEN 75
			WHEN A.RISCO_PRODUTIVO = "Médio" THEN 50
			WHEN A.RISCO_PRODUTIVO = "Alto" THEN 25
			ELSE 0 END AS Peso_Risco_Produtivo

		,CASE WHEN A.'Proporção de Área Própria (%)'n < 50 THEN 0
			WHEN A.'Proporção de Área Própria (%)'n < 65 THEN 50
			WHEN A.'Proporção de Área Própria (%)'n < 80 THEN 75
			ELSE 100 END AS Peso_Area_Propria

		,CASE WHEN (&ANO. - A.'Ano Inicial do Atividade Agricol'n) <= 3 THEN 0
			WHEN (&ANO. - A.'Ano Inicial do Atividade Agricol'n) <= 5 THEN 50
			WHEN (&ANO. - A.'Ano Inicial do Atividade Agricol'n) <= 10 THEN 75
			ELSE 100 END AS Peso_Tempo_Cultura

		,CASE WHEN A.EVOLUCAO = 'Entre -15% e 15%' THEN 100
			WHEN A.EVOLUCAO = 'Crescimento superior a 15% justificado' THEN 90
			WHEN A.EVOLUCAO = 'Crescimento superior a 15% não Justificado' THEN 0
			WHEN A.EVOLUCAO = 'Redução superior a 15% justificado' THEN 80
			WHEN A.EVOLUCAO = 'Redução superior a 15% não justificado' THEN 0
			ELSE 0 END AS Peso_Evolucao_Area

		,CASE WHEN A.'Pontualidade'n = 'Pontual' THEN 100
			WHEN A.'Pontualidade'n = 'Novo no Crédito' THEN 100
			WHEN A.'Pontualidade'n = 'Atraso entre 01 e 14 dias' THEN 100
			WHEN A.'Pontualidade'n = 'Atraso entre 15 e 30 dias' THEN 75
			ELSE 0 END AS Peso_Historico_Credito

		,CASE WHEN A.'Tempo como Cliente'n >= 5 THEN 100
			WHEN A.'Tempo como Cliente'n >= 2 THEN 70
			ELSE 0 END AS Peso_Tempo_Cliente

		,CASE WHEN A.'Liquidez'n >= 1.43 THEN 100
			WHEN A.'Liquidez'n >= 1.19 THEN 75
			WHEN A.'Liquidez'n >= 1.00 THEN 50
			WHEN A.'Liquidez'n >= 0.85 THEN 25
			ELSE 0 END AS Peso_Liquidez

	FROM INFOS_SACADOS_PF_V3 AS A;

PROC SQL;

	CREATE TABLE INFOS_SACADOS_PF_V5 AS SELECT

		 A.*
		,(0.399*Peso_Restricao)
		+(0.019*Peso_Comercial)
		+(0.235*Peso_Risco_Produtivo)
		+(0.113*Peso_Area_Propria)

		+(0.027*Peso_Evolucao_Area)

		,CASE WHEN CALCULATED Score_BV >= 94.13 THEN "AA"
		WHEN CALCULATED Score_BV >= 88.24 THEN "A+"
		WHEN CALCULATED Score_BV >= 82.36 THEN "A"
		WHEN CALCULATED Score_BV >= 76.48 THEN "A-"
		WHEN CALCULATED Score_BV >= 70.60 THEN "B+"
		WHEN CALCULATED Score_BV >= 64.71 THEN "B"
		WHEN CALCULATED Score_BV >= 58.83 THEN "B-"
		WHEN CALCULATED Score_BV >= 52.95 THEN "C+"
		WHEN CALCULATED Score_BV >= 47.06 THEN "C"
		WHEN CALCULATED Score_BV >= 41.18 THEN "C-"
		WHEN CALCULATED Score_BV >= 35.30 THEN "D+"
		WHEN CALCULATED Score_BV >= 29.41 THEN "D"
		WHEN CALCULATED Score_BV >= 23.53 THEN "D-"
		WHEN CALCULATED Score_BV >= 17.65 THEN "E"

	DATA = BASE_COMPLETA_PF OUT = BASE_COMPLETA_PF_V2 (DROP = _NAME_);

	BY CPF11 NOTSORTED;

	VAR LIMITE_GLOBAL;

PROC SQL;

CREATE TABLE TABELA_INFOS_PF_FINAL AS SELECT

     A.*
		,(CASE WHEN F.LIMITE_GLOBAL >= 0 THEN F.LIMITE_GLOBAL
			ELSE 0 END) AS LIMITE_GLOBAL_CONVENIO
		,(CASE WHEN E.LIMITE_GLOBAL_BV >= 0 THEN E.LIMITE_GLOBAL_BV
			ELSE 0 END) AS LIMITE_GLOBAL_BV

	FROM INFOS_SACADOS_PF_V6 AS A

	LEFT JOIN BASE_COMPLETA_PF_V2 AS D ON A.CPF11 = D.CPF11

	LEFT JOIN BASE_COMPLETA_PF_LIMITE AS E ON A.CPF11 = E.CPF11

	LEFT JOIN BASE_COMPLETA_PF AS F ON A.CPF11 = F.CPF11 AND A.PORTFOLIO = F.PORTFOLIO_ORIGEM;

RUN;

		,CASE WHEN









			 FLAG_SERASA = 1 THEN "Restritivos"

			WHEN FLAG_TEMPO_CLIENTE = 1 THEN "Tempo como Cliente"

			WHEN RATING = "C+" THEN "Rating Agrometrika"
			WHEN RATING = "C" THEN "Rating Agrometrika"
			WHEN RATING = "C-" THEN "Rating Agrometrika"
			WHEN RATING = "D+" THEN "Rating Agrometrika"
			WHEN RATING = "D" THEN "Rating Agrometrika"
			WHEN RATING = "D-" THEN "Rating Agrometrika"
