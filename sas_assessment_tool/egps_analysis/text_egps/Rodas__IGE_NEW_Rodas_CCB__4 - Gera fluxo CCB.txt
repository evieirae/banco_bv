proc sql;
	connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
		schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
	Create table CCBCLICRM as

	select a.* from connection to perm

		(
	SELECT * FROM bv-pdbd-prd.dvry_clie_crm.tbdisparocanalemailhist
		where CAST(dtdisparo AS DATE) >= '2025-01-01'
		and dssfnome like "%TP-0000049158%"
			) a
		JOIN CCBENVIODOCSCPF b
			ON a.idpessoa = b.NuCpfCnpj_ori

	;
	disconnect from perm;
quit;

PROC SQL;
	CREATE TABLE CCBENVIODOCCRM AS
		SELECT C.*,D.*,intck('day',c.dtcontrato,datepart(d.dtdisparo)) as difdias
			FROM
				(
			SELECT DISTINCT A.*
				FROM CCBENVIODOCS2 A
					WHERE 1=1
/*						and A.DsStatusEnvioDocumento in ('Documento pendente','E-mail não preenchido','Falha no envio','Pendente Envio')*/
) C
					LEFT JOIN
						(SELECT DISTINCT idpessoa, dtdisparo, dsnvlinteracao,intnx('day',DATEPART(dtdisparo),15) FORMAT DDMMYY10. AS DISP15
							FROM CCBCLICRM) D
								ON C.NuCpfCnpj_ori = D.idpessoa
								AND c.dtcontrato <= datepart(D.dtdisparo)
								AND intck('day',c.dtcontrato,datepart(d.dtdisparo))<=15;
QUIT;

PROC SORT DATA=CCBENVIODOCCRM
	OUT=CCBENVIODOCCRM1;
	BY DESCENDING NuCpfCnpj_ori DESCENDING NUPROPOSTA DESCENDING dtdisparo;
RUN;

data work.CCBENVIODOCCRM1_2;
	set work.CCBENVIODOCCRM1;
	BY DESCENDING NuCpfCnpj_ori DESCENDING NUPROPOSTA DESCENDING dtdisparo;

	if first.NUPROPOSTA then
		n=1;
	else n+1;
run;

PROC SQL;
	CREATE TABLE MXDISPARO AS
		SELECT
		  a.DtEntradaProposta,
          a.DtContrato,
          a.NuCpfCnpj_ori,
          a.NuProposta,
          a.NuContrato,
          a.DsStatusEnvioDocumento,
          a.DtPriInclusaoDoc,
          a.DtPriAlteracaoDoc,
          a.DtUltInclusaoDoc,
          a.DtUltAlteracaoDoc,
          a.NmGrupoProdutoMeta,
          a.TtlInclusoesDocs,
          a.idpessoa,
          a.dtdisparo as DtPriDisparo,
          a.dsnvlinteracao as PriInteracao,
		CASE
			WHEN A.IDPESSOA IS NULL THEN 0
			ELSE A.n
		END
	AS TtlDisparosCRM
		FROM CCBENVIODOCCRM1_2 A
			JOIN
				(SELECT NUPROPOSTA, MAX(N) AS MXN FROM CCBENVIODOCCRM1_2 GROUP BY NUPROPOSTA) B
					ON A.NUPROPOSTA = B.NUPROPOSTA AND A.N = B.MXN;
QUIT;

PROC SQL;
CREATE TABLE CCBENVIODOCCRM2 AS
SELECT
/*INPUT('21/03/2025',DDMMYY10.) FORMAT DDMMYY10. AS SYSINPUT,*/
		TODAY() FORMAT DDMMYY10. AS SYSINPUT,
		 a.DtEntradaProposta,
          a.DtContrato,
          a.NuCpfCnpj_ori,
          a.NuProposta,
          a.NuContrato,
          a.DsStatusEnvioDocumento,
          a.DtPriInclusaoDoc,
          a.DtPriAlteracaoDoc,
          a.DtUltInclusaoDoc,
          a.DtUltAlteracaoDoc,
          a.NmGrupoProdutoMeta,
          a.TtlInclusoesDocs,
          a.idpessoa,
		  b.DtPriDisparo,
          a.dtdisparo AS DtUltDisparo,
		  b.PriInteracao,
          a.dsnvlinteracao as UltInteracao,
 		  b.TtlDisparosCRM,
		  case when b.TtlDisparosCRM <> 0 then INTCK('DAY',a.DtContrato,datepart(a.DtPriInclusaoDoc)) end AS LTimeCttIncluDoc,
		  case when b.TtlDisparosCRM <> 0 then INTCK('DAY',a.DtContrato,datepart(a.DtUltAlteracaoDoc)) end AS LTimeCttUltAlteracaoDoc,
		  case when b.TtlDisparosCRM <> 0 then INTCK('DAY',a.DtContrato,datepart(b.DtPriDisparo)) end AS LTimeCttPriDisparoCRM,
		  case when b.TtlDisparosCRM <> 0 then INTCK('DAY',a.DtContrato,datepart(a.dtdisparo)) end AS LTimeCttUltDisparoCRM

FROM CCBENVIODOCCRM1_2 A
JOIN MXDISPARO B
ON A.NuProposta = B.NuProposta

WHERE A.N = 1;quit;

proc sql NOPRINT;
select count(*) INTO: count_exist
from TRIBO.CCBENVIODOCCRMFINAL
where SYSINPUT in (select distinct SYSINPUT from CCBENVIODOCCRM2);
quit;

%if &count_exist = 0 %then %do;
    data TRIBO.CCBENVIODOCCRMFINAL (compress='yes');
        set
			TRIBO.CCBENVIODOCCRMFINAL
            CCBENVIODOCCRM2;
    run;
%end;

proc sql;

create table CCBENVIODOCCRMASSINATURA AS

SELECT A.*, B.MODELO_ASSINATURA_FINAL
FROM TRIBO.CCBENVIODOCCRMFINAL A
LEFT JOIN tpassinatura_novo B
ON A.NUPROPOSTA = B.NUPROPOSTA
WHERE YEAR(A.SYSINPUT)*100+MONTH(A.SYSINPUT) >= YEAR(TODAY())*100+MONTH(TODAY())-1
;QUIT;

		FILE="&dir_csv./CCBENVIODOCCRMFINAL.csv" REPLACE

	DELIMITER=';';
RUN;
