PROC SQL;
CREATE TABLE CART_TOTAL_2025_4 AS
SELECT DISTINCT
A.*
,B.Atraso AS AtrasoAnterior
,CASE WHEN B.fx_atraso_cont  IN ('') THEN '01 - EM DIA' ELSE B.fx_atraso_cont  END AS fx_atraso_anterior
,CASE WHEN B.fx_atraso_cont2 IN ('') THEN '01 - EM DIA' ELSE B.fx_atraso_cont2 END AS fx_atraso_anterior2

,CASE WHEN A.fx_atraso_cont =  B.fx_atraso_cont THEN 'Manutenção'
	  WHEN A.fx_atraso_cont <> B.fx_atraso_cont AND A.Atraso < B.Atraso THEN 'Retrocesso'
	  WHEN A.fx_atraso_cont <> B.fx_atraso_cont AND A.Atraso > B.Atraso THEN 'Rolagem Pura'
	  ELSE 'Rolagem Pura' END AS StatusRolagem

,CASE WHEN A.fx_atraso_cont2 =  B.fx_atraso_cont2 THEN 'Manutenção'
	  WHEN A.fx_atraso_cont2 <> B.fx_atraso_cont2 AND A.Atraso < B.Atraso THEN 'Retrocesso'
	  WHEN A.fx_atraso_cont2 <> B.fx_atraso_cont2 AND A.Atraso > B.Atraso THEN 'Rolagem Pura'
	  ELSE 'Rolagem Pura' END AS StatusRolagem2

FROM CART_TOTAL_2025_3 AS A
LEFT JOIN CART_TOTAL_2025_3 AS B ON A.Cd = B.Cd AND A.Safra = B.Safra_M1
;QUIT;
