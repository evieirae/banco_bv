proc sql;
connect to bigquery as perm (project='bv-safn-dtm' credfile='/projetos/acessos/bq/fn/safn-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table carteira_GCON as
   select * from connection to perm
     (SELECT

	 a.CdCarteira

	,a.DtProcessamento
	,a.NuContrato
    ,a.CdModalidadeProduto
    ,b.NmModalidadeProduto
    ,sum(coalesce(CAST (VrSaldo AS FLOAT64),0))*-1 as Carteira
	FROM bv-pdbd-prd.trusted_gestaocontabil.tbsaldocontabil a
	INNER JOIN bv-pdbd-prd.trusted_dba.tbmodalidadeproduto b ON (a.CdModalidadeProduto = b.CdModalidadeProduto)

WHERE a.DtProcessamento = '2025-01-31'
		AND a.CdCarteira NOT IN (13)
		AND (a.NuContaContabil || a.CdModalidadeProduto) <> '060'
		AND DsAgrupamentoContabil IN (
		'Carteira Propria - VrFuturo',
		'Carteira Propria - VrRAP',
		'Carteira Propria - Ganho Sobre Renegociacao',
		'Carteira Propria - Permanencia',
		'Carteira Propria - Restruturacao',
		'Carteira Cedida  - VrFuturo Ativo',
		'Carteira Cedida  - VrRAP Ativo',
		'Carteira Cedida - Permanencia',
	    'Diferimento Acordos Comerciais',
		'Diferimento Comissao',
		'Diferimento Comissao de GR',
		'Diferimento GRAVAME',
		'Diferimento Lead Digital',
		'Diferimento BV Compensa',
		'Diferimento Meu Financiamento Solar')

GROUP BY
 a.CdCarteira

,a.DtProcessamento
,a.NuContrato
,a.CdModalidadeProduto
,b.NmModalidadeProduto

);
   disconnect from perm;
quit;

CREATE TABLE DATAS_FLEX AS
SELECT * FROM CONNECTION TO perm (
SELECT  max(PARSE_DATE('%Y%m%d',DATABASE)) as  DT_FLEX
		FROM bv-pdbd-prd.trusted_stgtxt_flex.tbrelcarteira

);

PROC SQL;
CREATE TABLE CARTEIRA_FLEX_&anomes. AS
SELECT
		input(C.CONTRATO,comman15.) as NuContrato format best15.,
	   INPUT(C.DATABASE, YYMMDD10.) AS MesReferencia FORMAT DATE9.,
	   input(substr(C.PRODUCT_CODE, 3, 2), 10.) AS CdModalidadeProduto,
	   input(C.VL_PRESENTE, best32.) AS Carteira,
	   input(C.VL_MORA, best32.) AS Mora,

	proc sql;
    create table work.Carteira_Leves_Solar as
    select
        t1.*,
        t2.NmGrupoProdutoMeta,
        t2.TpPessoa as TpPessoa_t2,
        e.TpPessoa  as TpPessoa_t3,
        m.Max_Sq_Contrato as SqContratoFinanceiro_Max,
        case
            when kupcase(strip(e.DsSubProduto)) in (
                'CDC FINANCIAMENTO … LOJISTA LEVES',
                'CDC ÔNIBUS','CDC ONIBUS',
                'CDC VANS E MICRO-ÔNIBUS','CDC VANS E MICRO-ONIBUS',
                'CDC VEÍCULOS','CDC VEICULOS',
                'LEAS FINANCIAMENTO À LOJISTA LEVES','LEAS FINANCIAMENTO Ã  LOJISTA LEVES','LEAS FINANCIAMENTO A LOJISTA LEVES',
                'LEAS ÔNIBUS','LEAS ONIBUS',
                'LEAS VANS E MICRO-ÔNIBUS','LEAS VANS E MICRO-ONIBUS',
                'LEAS VEÍCULOS','LEAS VEICULOS',
                'PRÉ APROVADO CDC LEVES','PRE APROVADO CDC LEVES',
                'AUTO - FUNCIONÁRIOS VOTORANTIM FINANÇAS','AUTO - FUNCIONARIOS VOTORANTIM FINANCAS'
            ) then 'Leves'
            when kupcase(strip(e.DsSubProduto)) in (
                'EQUIPAMENTO FOTOVOLTAICO',
                'EQUIPAMENTOS - CARNE',
                'EQUIPAMENTOS - CHEQUE'
            ) then 'Solar'
            else 'Outros'
        end as DeParaSubProduto length=10
    from work.carteira_final_gcon_flex as t1
    left join bases.tbproducao as t2
        on t1.nucontrato = t2.nucontrato
    left join (
        select
            nucontrato,
            max(sqcontratofinanceiro) as Max_Sq_Contrato
        from bases.TBEXTCONTRATO
        group by nucontrato
    ) as m
        on t1.nucontrato = m.nucontrato
    left join bases.TBEXTCONTRATO as e
        on  e.nucontrato            = t1.nucontrato
        and e.sqcontratofinanceiro = m.Max_Sq_Contrato
    ;
quit;
