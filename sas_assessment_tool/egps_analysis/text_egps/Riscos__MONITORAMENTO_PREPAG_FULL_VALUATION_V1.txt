proc sql;
	create table CashFlow_vertice as
	select 	t1.*,
			t4.vertice1,
			t4.vertice2,
			(case	when t1.DU >= 7560 then 0
					else (t4.vertice2 - t1.DU)/(t4.vertice2 - t4.vertice1)
			end) as prop_vertice1,
			(case	when t1.DU >= 7560 then t1.DU/7560
					else (t1.DU - t4.vertice1)/(t4.vertice2 - t4.vertice1)
			end) as prop_vertice2

	from WORK.cashflow_taxa_du as t1
	left join vertices as t4
	on t1.DU >= t4.vertice1 and t1.DU < t4.vertice2
	having t1.DU > 0;
quit;


proc sql;
create table cashflow_taxa_du_choque as
select t1.*,
       t2.NmChoque,
	   t2.Choque as ChoqueVertice1,
	   t3.Choque as ChoqueVertice2,
	   t2.Choque * t1.prop_vertice1 + t3.Choque * t1.prop_vertice2 as Choque_interpolado
	from CashFlow_vertice as t1
	left join walid.choques_EVE as t2
	on t1.vertice1 = t2.DU
	left join walid.choques_EVE as t3
	on t1.vertice2 = t3.DU
	AND t2.NmChoque = t3.NmChoque
	ORDER BY t1.TIMESTEPS_KEEPSLASH,
			   t1.'Grupo de Conciliacao'n,
               t1.'Tipo de Negociacao'n,
               t1.Veiculo,
               t1.'Produto Gerencial'n,
               t1.'Mesa Operacao'n,
               t1.'Carteira Gerencial'n,
               t1.'Book Gerencial'n,
               t1.'Modalidade SIC'n,
               t1.'Tipo Taxa'n,
			   t1.VF,
               t1.SCENARIO_NAMES
;
quit;



proc sql;
	create table VF_MtM_EVE as
	select 	t1.*,
			t1.VF_Base/(1+Taxa/100)**(DU/252) as MtM_Base, 
			t1.VF/(1+Taxa/100)**(DU/252) as MtM_Choq_PREPAG, 
			t1.VF - VF_Base as Delta_VF_VFBase, 
			t1.VF/(1+Taxa/100+Choque_interpolado)**(DU/252) as MtM_Choq_EVE, 
			Calculated MtM_Choq_EVE - Calculated MtM_Base as DeltaEVE, 
			Calculated MtM_Choq_EVE - Calculated MtM_Choq_PREPAG as DeltaEVEChoq 

	from cashflow_taxa_du_choque
			(where = ('Book Gerencial'n not in ('37 - ACCRUAL PRE' , '564 - OVER 60'))) as t1
;
quit;
