Proc Sql;
Create Table Cobans_Solar as
Select Distinct
NmAreaEstruturaComercialC
,DsLoginN1C
,DsLoginN2C
,DsLoginN3C
,NmColaboradorC
,cdParceiroComercial
,NuCnpjLoja
,DatePart(DtCadastro) as DtCadastro Format DDMMYY10.
,(YEAR(DatePart(DtCadastro)) * 100) + MONTH(DatePart(DtCadastro)) as NuAnomes_Cadastro
,FlAtivo
,&DtAtualizacao. as DataBase Format DDMMYY10.
from tmp.tbparceirocomercial
where 1=1
and tpatividade = 7 /*and FlAtivo = 'S' and*/
and NmCanalEstruturaComercialC = 'Canal Solar' and
		NmAreaEstruturaComercialC in
		(
			'Canal Solar CO e NO',
			'Canal Solar MG, BA, AL e SE',
			'Canal Solar NE',
			'Canal Solar SP Interior',
			'Canal Solar SP, RJ e ES',
			'Canal Solar Sul'
		)
;
run;

Proc Sql;
Create Table Base_Producao_PDVs as
Select
Orig.NuAnoMes
,Orig.CdParceiroComercial
,Orig.NuCnpjLojaFavorecido
,Orig.QtCorridosDiaUtilMes

,Sum(Orig.FlPropostaOriginada) as FlPropostaOriginada
,Sum(Orig.FlPropostaAprovada) as FlPropostaAprovada
,Sum(Orig.FlPropostaEfetivada) as FlPropostaEfetivada

From Bases.TbOriginacao as Orig
Where Orig.NuAnomes >= &Base_m5. and Orig.NmGrupoProdutoMeta = 'Solar' and
		Orig.NmAreaEstruturaComercialC in
		(
			'Canal Solar CO e NO',
			'Canal Solar MG, BA, AL e SE',
			'Canal Solar NE',
			'Canal Solar SP Interior',
			'Canal Solar SP, RJ e ES',
			'Canal Solar Sul'
		)

;Quit;

Proc Sql;
Create Table Base_PDVs_Originando as
Select Distinct
Pdvs.NmAreaEstruturaComercialC
,PDVs.DsLoginN1C
,PDVs.DsLoginN2C
,PDVs.DsLoginN3C

,Pdvs.NmColaboradorC
,Pdvs.cdParceiroComercial
,Pdvs.NuCnpjLoja
,Pdvs.DtCadastro Format DDMMYY10.
,Pdvs.FlAtivo
,Max(Orig.QtCorridosDiaUtilMes) as QtCorridosDiaUtilMes

,Sum(Case When Orig.FlPropostaOriginada >0 Then 1 Else 0 End) as Qtd_PDVs_Originando
,Sum(Case When Orig.FlPropostaAprovada  >0 Then 1 Else 0 End) as Qtd_PDVs_Aprovando
,Sum(Case When Orig.FlPropostaEfetivada >0 Then 1 Else 0 End) as Qtd_PDVs_Efetivando

,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m0. Then 1 Else 0 End) as NVParc_&Base_m0.
,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m1. Then 1 Else 0 End) as NVParc_&Base_m1.
,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m2. Then 1 Else 0 End) as NVParc_&Base_m2.
,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m3. Then 1 Else 0 End) as NVParc_&Base_m3.
,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m4. Then 1 Else 0 End) as NVParc_&Base_m4.
,Sum(Case When Pdvs.NuAnomes_Cadastro = &Base_m5. Then 1 Else 0 End) as NVParc_&Base_m5.

,Sum(Case When Orig.NuAnomes Between &&Base_m3. and &Base_m1. Then Orig.FlPropostaOriginada Else 0 End) as QtPropostaOriginada_90
,Sum(Case When Orig.NuAnomes Between &&Base_m1. and &Base_m0. Then Orig.FlPropostaOriginada Else 0 End) as QtPropostaOriginada_30

,Sum(Case When Orig.NuAnomes Between &&Base_m3. and &Base_m1. Then Orig.FlPropostaAprovada Else 0 End) as QtPropostaAprovada_90
,Sum(Case When Orig.NuAnomes Between &&Base_m1. and &Base_m0. Then Orig.FlPropostaAprovada Else 0 End) as QtPropostaAprovada_30

,Sum(Case When Orig.NuAnomes Between &&Base_m3. and &Base_m1. Then Orig.FlPropostaEfetivada Else 0 End) as QtPropostaEfetivada_90
,Sum(Case When Orig.NuAnomes Between &&Base_m1. and &Base_m0. Then Orig.FlPropostaEfetivada Else 0 End) as QtPropostaEfetivada_30
,Max(PDVs.DataBase) as DataBase Format DDMMYY10.

From Cobans_Solar as PDVs
	Left Join Base_Producao_PDVs as Orig On Pdvs.NuCnpjLoja = Orig.NuCnpjLojaFavorecido and Pdvs.CdParceiroComercial = Orig.CdParceiroComercial
Where Orig.NuANomes >= &Base_m5.
Group by 1,2,3,4,5,6,7,8,9
;Quit;
