proc sql;
create table BaseFeirao as
select distinct
AnoMes as NuAnomes,
CdParceiroComercial,
NmModeloEvento as NmEvento,
'Campanha RC' as NmTipoEvento
from basegrc.TbFeiraoArquivos
where 1=1
and AnoMes >= &safraM3.
and CdParceiroComercial <> .
;run;

proc sql;
create table TBBVRELACIONAR as
select distinct
    NuAnoMes,
    CdParceiroComercial,
    NmCampanha as NmEvento,
	NmTipoEvento
from basegrc.TBBVRELACIONAR
where 1=1
and NmGrupo = 'Grupo Ação'
and NuAnoMes >= &safraM3.
and upcase(NmCampanha) not like '%MOTOS%'
and upcase(NmCampanha) not like '%TESTE%'
;quit;

Data BaseCampanhas_V01;
set TBBVRELACIONAR
	BaseFeirao
;run;

proc sql;
create table PdvSemCampanha as
select distinct
    CdParceiroComercial,
	'Sem Campanha' as NmTipoEvento
from tmp.TbParceiroComercial
where CdParceiroComercial not in (select distinct CdParceiroComercial from BaseCampanhas_V01)
;quit;

Data BaseCampanhasCN;
set BaseCampanhas_V01
	PdvSemCampanha
;run;

%ExportaCsv(BaseCampanhaRitoV2_CN,BaseCampanhasCN);

PROC SQL;
CREATE TABLE TbColaboradorPDV AS
  SELECT DISTINCT
    CdColaboradorC AS CdColaborador,
    NmColaboradorC,
    CdParceiroComercial,
    NmParceiroComercial,
    LstSubstituicaoC
  FROM BASES.TbParceiroComercial
  WHERE
    CdParceiroComercial <> .
    AND CdColaboradorC <> .;
RUN;

PROC SQL;
CREATE TABLE TbColaborador AS
  SELECT
    A.DsLoginN1 AS CdGR,
    A.DsLoginN2A AS CdGF,
    B.LstSubstituicaoC AS LstFocais,
    B.CdParceiroComercial
  FROM
    BASES.TbEstruturaComercial A
  INNER JOIN
    TbColaboradorPDV B
    ON A.CdColaboradorAtivo = B.CdColaborador
  WHERE
    TODAY() BETWEEN A.DtInicioVigencia AND A.DtFimVigencia
    AND A.DsLoginN1 IS NOT NULL
    AND A.DsLoginN2A IS NOT NULL;
RUN;

PROC SQL;
  SELECT MAX(DtAtualizacaoCetip) format date9. INTO :DtAtCetip2 FROM TMP.TBPOTENCIALRODAS;

PROC SQL;
CREATE TABLE Convsimu002 AS
  SELECT
    A.NuAnoMes,
    CASE WHEN TODAY() - A.DtInclusaoGravame < 8 AND FlConcorrente = 1 THEN 'Converteu' ELSE A.Banco END AS Banco,
    A.CdParceiroComercial,
    A.NmParceiroComercial,
    A.Regional,
    A.Filial,
    B.CdGF,
    B.CdGR,
    B.LstFocais,
    A.CdSimulacaoInicial,
    0 AS NuTelefone,
    A.StatusSimulacao,
    A.Mensageria,
    A.NuCpfCnpj,
    A.NuProposta,
    A.DtInclusaoGravame,
    A.FxIdade,
    A.StPagoLoja,
	A.DtSimulacaoInicial,
	Case when A.DtSimulacaoInicial < "&DtAtCetip2."d then 'Sim' else 'Não' End as Cetip,
	A.FxEntrada
  FROM
    Convsimu A
  LEFT JOIN
    TbColaborador B
    ON A.CdParceiroComercial = B.CdParceiroComercial;
RUN;

PROC SQL;
/*CREATE TABLE ConversaoC6_DadosCliente AS*/
CREATE TABLE ConversaoC6_DadosClienteTMP AS
  SELECT
    A.NuAnoMes,
    A.Banco,
    A.CdParceiroComercial,
    A.NmParceiroComercial,
    A.Regional,
    A.Filial,
    A.CdGF,
    A.CdGR,
    A.LstFocais,
    A.CdSimulacaoInicial,
    A.StatusSimulacao,
    A.Mensageria,
    A.NuCpfCnpj,
    COALESCE(B.NmPessoa, 'DESCONHECIDO') AS NmPessoa,
    B.NuTelefone1,
    B.NuTelefone2,
    B.NuTelefone3,
    B.DsEmail,
    A.NuProposta,
    C.FlTxAlterada AS FlPA,
    A.DtInclusaoGravame,
    A.StPagoLoja,
    A.FxIdade,
    PUT(&DtAt.,DATE9.) AS DtAtualizacao,
	A.Cetip,
	Case when D.NuCpfCnpj is null then 'N' else 'S' end as Efetivos,
	A.FxEntrada
  FROM
    Convsimu002 A
  LEFT JOIN
/*    AUXL.TbContatoPessoa B*/
    Bases.TbPessoa B
    ON A.NuCpfCnpj = B.NuCpfCnpj
  LEFT JOIN
    Originacao C
    ON A.NuProposta = C.NuProposta

  LEFT JOIN TbAbandonoSumarizadaMes02 D
	on  A.NuCpfCnpj = D.NuCpfCnpj
	and A.DtSimulacaoInicial = D.DT_SIMULACAO_PROPOSTA
	and A.CdParceiroComercial = D.CdParceiroComercial
;
RUN;

			on TMP.CdSimulacaoInicial = APROSIMU.CD_SIMULACAO_PROPOSTA
