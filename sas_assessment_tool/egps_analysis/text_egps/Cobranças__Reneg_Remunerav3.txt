CREATE TABLE PERFORMANCE_3 AS
SELECT DISTINCT
	*,
	CASE
		WHEN STPARCELA = 'Q' THEN 1
		ELSE 0
	END AS FLAG_PGTO,
	CASE
		WHEN ATRASO_NOVO >= 31 AND STPARCELA = 'Q' THEN 1
		ELSE 0
	END AS FLAG_INCONSITENCIA,
	CASE
		WHEN ATRASO_NOVO > 30 THEN 1
		ELSE 0
		END AS FLAG_INAD,
	CASE
		WHEN ATRASO_NOVO >= 0 OR STPARCELA = 'Q' THEN 1
		ELSE 0
		END AS FLAG_VALIDO,

		WHEN STPARCELA <> 'Q' AND ATRASO_NOVO <=30  AND (ATRASO_NOVO >= 0 OR STPARCELA = 'Q') THEN 1
		ELSE 0
		END AS FLAG_SEM_INAD,

			and DATEPART(DTVENCIMENTO_NOVO) - DTAPROVACAORENEG >= 30 THEN 2
		ELSE 1
	END AS FLAG_ACAO_RS,
	CASE
		WHEN ATRASO <= 0 THEN 'ADIMPLENTE'
		ELSE 'INADIMPLENTE'
	END AS ATRASO_CONTRATACAO,
	CASE
		WHEN ATRASO <= 0   THEN 'EM_DIA'
		WHEN ATRASO <= 30  THEN '001_030'
		WHEN ATRASO <= 60  THEN '031_060'
		WHEN ATRASO <= 90  THEN '061_090'
		WHEN ATRASO <= 180 THEN '091_180'
		WHEN ATRASO <= 360 THEN '181_360'
		WHEN ATRASO >= 361 THEN '361_999'
	END AS FX_ATRASO_CONTRATACAO
FROM PERFORMANCE_2;
QUIT;
RUN;

CREATE TABLE RENEG.RENEG_IMP_&SAFRA. AS
SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	DATEPART(A.DT_RENEG) AS DT_RENEG FORMAT DDMMYY10.,
	DATEPART(A.DT_CANCELAMENTO) AS DT_CANCELAMENTO FORMAT DDMMYY10.,
	DATEPART(A.DTAPROVACAORENEG) AS DTAPROVACAORENEG FORMAT DDMMYY10.,
	A.TP_ORIGEM
FROM CONNECTION TO SYBASE (
				SELECT DISTINCT
					NUCONTRATOGESTAO,
					SQCONTRATOFINANCEIRO,
					DTAPROVACAORENEG,
					TP_ORIGEM
				INTO
					#BASE_1
				FROM
					DBRENEG..TB_RENEG_SIMULADOR
				WHERE
					DTAPROVACAORENEG BETWEEN &DATAINI. AND &DATAFIM.
				ORDER BY
					NUCONTRATOGESTAO ASC

				SELECT DISTINCT
					A.NUCONTRATOGESTAO,
					A.SQCONTRATOFINANCEIRO,
					B.DT_RENEG,
					B.DT_CANCELAMENTO,
					A.DTAPROVACAORENEG,
					A.TP_ORIGEM
				FROM
					#BASE_1 A LEFT JOIN DBRENEG..TB_RENEG_SIMULADOR	B
					ON
					A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
					AND
					A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
					AND
					A.DTAPROVACAORENEG = B.DTAPROVACAORENEG) A;
QUIT;
RUN;

    t1.DtContratoContabil AS DTINICIOCONTRATO,

FROM
    GESTAOCONTABIL.VW_CONTRATOCONTABIL AS t1
WHERE
    t1.DtContratoContabil >= &DTini.
    AND t1.DtContratoContabil <= &dtfim.
    AND t1.cdcarteira IN (1, 3, 4, 5, 12)
GROUP BY
    t1.nucontrato,
    t1.sqcontratofinanceiro,
    t1.dtprocessamento,
    DTINICIOCONTRATO
UNION DISTINCT
SELECT DISTINCT
    t2.nucontrato AS NUCONTRATO,
    t2.sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,
    t2.dtprocessamento AS DTPROCESSAMENTO,
    t2.DtContratoContabil AS DTINICIOCONTRATO,

FROM
    STGTXT_FLEX.TBCONTRATOCONTABIL AS t2
WHERE
    	t2.DtContratoContabil >= &dtini.
    AND t2.DtContratoContabil <= &dtfim.
    AND t2.cdcarteira IN (1, 3, 4, 5, 12)
GROUP BY
    t2.nucontrato,
    t2.sqcontratofinanceiro,

%macro set_dates;
    %let start_year = 2024;
    %let start_month = 2;
    %let current_year = %sysfunc(year(%sysfunc(today())));
    %let current_month = %sysfunc(month(%sysfunc(today())));

    %do year = &start_year %to &current_year;
        %if &year = &current_year %then %let end_month = &current_month;
        %else %let end_month = 12;

        %if &year = &start_year %then %let start_loop_month = &start_month;
        %else %let start_loop_month = 1;

        %do month = &start_loop_month %to &end_month;
            %let month_name = %sysfunc(putn(%sysfunc(mdy(&month, 1, &year)), monname3.));
            %let DATAINI = %str(%'&month_name 01 &year 00:00:00%');
            %let last_day = %sysfunc(intnx(month, %sysfunc(mdy(&month, 1, &year)), 0, end));
            %let last_day_name = %sysfunc(putn(&last_day, monname3.));
            %let DATAFIM = %str(%'&last_day_name %sysfunc(day(&last_day)) &year 23:59:59%');
            %let SAFRA = %sysfunc(putn(&year, 4.))%sysfunc(putn(&month, z2.));

            %put DATAINI = &DATAINI;
            %put DATAFIM = &DATAFIM;
            %put SAFRA = &SAFRA;

PROC SQL;
CREATE TABLE PERFORMANCE_3 AS
SELECT DISTINCT
	*,
	CASE
		WHEN STPARCELA = 'Q' THEN 1
		ELSE 0
	END AS FLAG_PGTO,
	CASE
		WHEN ATRASO_NOVO >= 31 AND STPARCELA = 'Q' THEN 1
		ELSE 0
	END AS FLAG_INCONSITENCIA,
	CASE
		WHEN ATRASO_NOVO > 30 THEN 1
		ELSE 0
		END AS FLAG_INAD,
	CASE
		WHEN ATRASO_NOVO >= 0 OR STPARCELA = 'Q' THEN 1
		ELSE 0
		END AS FLAG_VALIDO,
	CASE
		WHEN STPARCELA <> 'Q' AND ATRASO_NOVO <=30  AND (ATRASO_NOVO >= 0 OR STPARCELA = 'Q') THEN 1
		ELSE 0
		END AS FLAG_SEM_INAD,
	CASE
		WHEN CDUF = 'RS' AND DTAPROVACAORENEG >= '10MAY2024'D AND DTAPROVACAORENEG <= '17JUN2024'D
			and DATEPART(DTVENCIMENTO_NOVO) - DTAPROVACAORENEG >= 30 THEN 2
		ELSE 1
	END AS FLAG_ACAO_RS,
	CASE
		WHEN ATRASO <= 0 THEN 'ADIMPLENTE'
		ELSE 'INADIMPLENTE'
	END AS ATRASO_CONTRATACAO,
	CASE
		WHEN ATRASO <= 0   THEN 'EM_DIA'
		WHEN ATRASO <= 30  THEN '001_030'
		WHEN ATRASO <= 60  THEN '031_060'
		WHEN ATRASO <= 90  THEN '061_090'
		WHEN ATRASO <= 180 THEN '091_180'
		WHEN ATRASO <= 360 THEN '181_360'
		WHEN ATRASO >= 361 THEN '361_999'
	END AS FX_ATRASO_CONTRATACAO
FROM PERFORMANCE_2;
QUIT;
RUN;
