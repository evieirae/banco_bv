PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.tblUsuarioEncerramento AS
SELECT

 Cd_Causa as Cd_Causa
,SERIAL_IMPORT as SERIAL_IMPORT
,Pasta as Pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdSistema as CdSistema
,DtCadastradoEvento as DtCadastradoEvento
,NmResponsavelEvento as NmResponsavelEvento
,DtEvento as DtEvento
,TpEncerramento as TpEncerramento
,NmMotivoEncerramento as NmMotivoEncerramento
,NmEvento as NmEvento

FROM CONNECTION TO postgres(

select
case
 	when p.serial_import like ('%INQ%') then (left(p.SERIAL_IMPORT,10))
    when p.serial_import like ('%BVT%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%CON%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%VAC%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%CTO%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%TRI%') then (left(p.SERIAL_IMPORT,10))
	when p.serial_import like ('%TRB%') then (left(p.SERIAL_IMPORT,10))
		else p.serial_import end        as Cd_Causa
, P.SERIAL_IMPORT
, P.PASTA								as Pasta
, P.ID_NATUREZA							as IdNatureza
, p.NATUREZA_UNIDADE_CUSTOM				as IdNaturezaUnidadeCustom
,'p' 									as CdSistema
, cast(L.DATA_CADASTRO as date)			as DtCadastradoEvento
, U.NOME 								as NmResponsavelEvento
, case when L.ID_EVENTO = 'Encerrar' then cast(P.DATA_ENCERRAMENTO as date)
	   when L.ID_EVENTO = 'Reabrir' then cast(P.DATA_REABERTURA as date)
	   else null end					as DtEvento

, case when L.ID_EVENTO = 'Encerrar'  then T.DESCRICAO
	   else null end					as TpEncerramento
, case when L.ID_EVENTO = 'Encerrar'  then E.ENCERRAMENTO
	   else null end 					as NmMotivoEncerramento
, L.ID_EVENTO							as NmEvento

from public.M_LOG L
inner join public.M_PROCESSO          P on P.ID_PROCESSO          = cast(L.ID_REGISTRO_MASTER as integer)
inner join public.M_USUARIO           U on U.ID_USUARIO           = L.ID_USUARIO_CADASTRO
left join public.LT_ENCERRAMENTO      E on E.ID_ENCERRAMENTO      = P.ID_ENCERRAMENTO
left join public.LT_TIPO_ENCERRAMENTO T on T.ID_TIPO_ENCERRAMENTO = P.ID_TIPO_ENCERRAMENTO
where
    P.ID_NATUREZA in (21, 27, 28, 31, 32)
and p.NATUREZA_UNIDADE_CUSTOM in (	1,	2,	3,	4,	5,	6,	7,	8,	11,	12,	13,	14,	15,	16)
and P.DATA_ENCERRAMENTO is not null
and L.ID_EVENTO in ('Encerrar'  , 'Reabrir' )

and cast(L.DATA_CADASTRO as date) >= cast(current_date - interval '7 days' as date)
and cast(L.DATA_CADASTRO as date) <> cast(current_date as date)



;

);DISCONNECT FROM postgres;

proc sql;
create table work.tblUsuarioEnc as
select DISTINCT

	a.Cd_Causa
,	a.SERIAL_IMPORT
,	a.Pasta
,	a.IdNatureza
,	b.NATUREZA as NmNatureza
,	a.IdNaturezaUnidadeCustom
,	b.NATUREZAUNIDADECUSTOM as NmNaturezaUnidadeCustom
,	a.CdSistema
,case
	when a.cdsistema = 'e' then substr(a.Cd_Causa, 1, 3)
	else b.causa end		as CdUnidadeExyon
,	a.DtCadastradoEvento
,	a.NmResponsavelEvento
,	a.DtEvento
,	a.TpEncerramento
,	a.NmMotivoEncerramento
,	a.NmEvento

from WORK.tblUsuarioEnc as a
left join yribeiro.tbldeparanatureza b
on  b.IdNatureza = a.IdNatureza
and b.IdNaturezaUnidadeCustom = a.IdNaturezaUnidadeCustom

where b.FLNATUREZAUNIDADECONSIDERAR <> 0

;run;
