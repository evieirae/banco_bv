PROC SQL;

PROC SQL;

    SELECT DtBase,

/* Acompanhamento do volume (Proporção) */
PROC SQL;
CREATE TABLE Monit_Prod_Rodas AS
    SELECT a.prod AS Analise,
           a.DtBase,
           a.Quantidade/b.Total AS Valor,
           'Produto - Proporção na safra' AS Medida,
		   '' AS SubMedida
    FROM Base_Monit_Prod_Rodas AS a
    LEFT JOIN Monit_Prod_Rodas_Total AS b on a.DtBase = b.DtBase;
QUIT;

DATA Monit_Prod_Rodas;
SET Monit_Prod_Rodas;
    FORMAT DtBase DDMMYY8.;
RUN;

/* Acompanhamento do volume (Quantidade) */
PROC SQL;
CREATE TABLE Monit_Prod_Rodas_Vol_Tt AS
    SELECT a.prod AS Produto,
           MEAN(a.Quantidade) AS MediaQuantidade
    FROM Base_Monit_Prod_Rodas AS a
    LEFT JOIN Monit_Prod_Rodas_Total AS b on a.DtBase = b.DtBase

    LEFT JOIN Monit_Prod_Rodas_Vol_Tt AS b on a.prod = b.Produto;

%MACRO MonitoramentoVariaveisRodas(Produto, Pop, Variavel);
/*macro*/ /* encontrar o MIX médio dos ultimos meses (sem contar a referência) */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE MIXMedio AS
/*macro*/     SELECT &Variavel.,
/*macro*/            COUNT(NuContrato) AS Contador
/*macro*/            FROM Bhs&Produto.Pop&Pop.
/*macro*/     WHERE DtBase >=&DataRef6. AND DtBase <=&DataRef2.
/*macro*/     GROUP BY &Variavel.;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Total do MIS médio */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE MIXMedio_Total AS
/*macro*/     SELECT COUNT(NuContrato) AS Total
/*macro*/     FROM Bhs&Produto.Pop&Pop.
/*macro*/     WHERE DtBase >= &DataRef6. AND DtBase <= &DataRef2.;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Encotrar o Percentual do MIX Médio */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE MIXMedio AS
/*macro*/     SELECT a.&Variavel.,
/*macro*/            a.Contador/b.Total AS MixMedio
/*macro*/     FROM MIXMedio AS a
/*macro*/     FULL JOIN MIXMedio_Total AS b on a.Contador;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Sumarizar por categoria/DtBase para calcular o MIX */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE &Variavel. AS
/*macro*/     SELECT DtBase,
/*macro*/            &Variavel.,
/*macro*/     COUNT(NuContrato) AS Contador
/*macro*/     FROM Bhs&Produto.Pop&Pop.
/*macro*/     GROUP BY DtBase, &Variavel.;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Total por DtBase para o MIX */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE &Variavel._Total AS
/*macro*/     SELECT DtBase,
/*macro*/            SUM(Contador) AS Total
/*macro*/     FROM &Variavel.
/*macro*/     GROUP BY DtBase;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Cálculo do MIX e do Mix Médio para IEP e volume */
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE &Variavel. AS
/*macro*/     SELECT DISTINCT
/*macro*/            a.DtBase,
/*macro*/            a.&Variavel.,
/*macro*/            a.Contador/b.Total AS Mix,
/*macro*/            c.MixMedio
/*macro*/     FROM &Variavel. AS a
/*macro*/     LEFT JOIN &Variavel._Total AS b on a.DtBase = b.DtBase
/*macro*/     LEFT JOIN MIXMedio AS c on a.&Variavel. = c.&Variavel.;
/*macro*/ QUIT;
/*macro*/
/*macro*/ /* Cálculos e ajuste para o calculo do IEP */
/*macro*/ DATA &Variavel.;
/*macro*/ SET &Variavel.;
/*macro*/     Mix2 = Mix;
/*macro*/     IF MixMedio = . THEN MixMedio = 2;
/*macro*/     IF Mix = 0 THEN Mix = 1;
/*macro*/
/*macro*/     /* Cálculos par o calculo do IEP */
/*macro*/     dif = MixMedio - Mix;
/*macro*/     LNdif = log(MixMedio/Mix);
/*macro*/     Produto = dif * LNdif;
/*macro*/ RUN;
/*macro*/
/*macro*/ /*Sumarizar o IEP por DtBase para colocar no acompanhamento*/
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE IEP_&Variavel. as
/*macro*/     SELECT PUT(CATS("&Variavel."), $32.) AS Analise,
/*macro*/            DtBase,
/*macro*/            SUM(Produto) AS Valor,
/*macro*/            PUT(CATS("IEP - &Produto.Pop&Pop."), $32.) AS Medida,
/*macro*/		     PUT(CATS(&Variavel.), $32.) AS SubMedida
/*macro*/     FROM &Variavel.
/*macro*/     WHERE DtBase = &DataRef1.
/*macro*/     GROUP BY SubMedida;
/*macro*/ QUIT;
/*macro*/
/*macro*/ PROC SQL;
/*macro*/ CREATE TABLE IEP_&Variavel._MIX as
/*macro*/     SELECT PUT(CATS("&Variavel."), $32.) AS Analise,
/*macro*/            DtBase,
/*macro*/            Mix2 AS Valor,
/*macro*/            PUT(CATS("MIX - &Produto.Pop&Pop."), $32.) AS Medida,
/*macro*/		     PUT(CATS(&Variavel.), $32.) AS SubMedida
/*macro*/     FROM &Variavel.;
/*macro*/ QUIT;
/*macro*/
/*macro*/ PROC DATASETS; APPEND base=analiseIEP data=IEP_&Variavel. FORCE;
/*macro*/ PROC DATASETS; APPEND base=analiseIEP_Mix data=IEP_&Variavel._MIX FORCE;
%MEND;
