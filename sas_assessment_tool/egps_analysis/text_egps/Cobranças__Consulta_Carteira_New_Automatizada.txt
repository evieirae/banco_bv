proc sql;
connect to bigquery as perm(project='bv-sadi-dtm' 
credfile='/projetos/acessos/bq/di/sadi-dtm.json' 
schema='bv-pdbd-prd' READ_MODE=storage);
Create table BASE_CARTEIRA_&data. as select * from connection to perm
(
SELECT
  CAST(A.CONTRATO_LEGADO AS INT64) AS CONTRATO_LEGADO ,
  CAST(A.DT_ENVIO_ASSESSORIA AS DATE) AS DT_ENVIO_ASSESSORIA,
  CAST(A.DATA_CONTRATO AS DATE) AS DATA_CONTRATO,
  A.ASSESSORIA,
  A.COMPLEMENTO_ASSESSORIA,
  A.PRODUTO,
  A.NOME_CLIENTE,
  SAFE_CAST(A.SALDO_ATUAL AS FLOAT64) AS SALDO_ATUAL,
  SAFE_CAST(A.DIAS_EM_ATRASO AS INT64) AS DIAS_EM_ATRASO,
  CAST(A.FASE AS INT64) AS FASE,
  A.PLANO,
  SAFE_CAST(A.QTD_PARCELAS_PAGAS AS INT64) AS QTD_PARCELAS_PAGAS,
  SAFE_CAST(A.PARCELAS_NAO_PAGAS AS INT64) AS PARCELAS_NAO_PAGAS,
  SAFE_CAST(A.PARCELAS_VINCENDAS AS INT64) AS PARCELAS_VINCENDAS,
  SAFE_CAST(A.VALOR_PARCELA AS FLOAT64) AS VALOR_PARCELA,
  CASE
    WHEN A.PRODUTO = 'Aviso' THEN 'IMPLANTACAO DE SALDO'
    WHEN A.PRODUTO = 'Saldo' AND SAFE_CAST(A.PARCELAS_VINCENDAS AS INT64) = 0 THEN 'APENAS QUITACAO'
    WHEN A.PRODUTO = 'Saldo' AND SAFE_CAST(A.PARCELAS_VINCENDAS AS INT64) > 0 THEN 'ATUALIZACAO OU QUITACAO'
  END AS MOD_NEGOCIACAO,
  CASE
    WHEN A.PRODUTO = 'Saldo' AND SAFE_CAST(A.PARCELAS_NAO_PAGAS AS INT64) BETWEEN 1 AND 3 THEN '1 A 3'
    WHEN A.PRODUTO = 'Saldo' AND SAFE_CAST(A.PARCELAS_NAO_PAGAS AS INT64) BETWEEN 4 AND 24 THEN '4 A 24'
    WHEN A.PRODUTO = 'Saldo' AND SAFE_CAST(A.PARCELAS_NAO_PAGAS AS INT64) = 0 AND SAFE_CAST(A.PARCELAS_VINCENDAS AS INT64) > 0 THEN 'Em dia'
  END AS MOD_PARCELAS
FROM (
  SELECT DISTINCT
    DQ.DMACCT AS CONTRATO_LEGADO,
    DQ.DMQCHGDT AS DATA_CONTRATO,
    DQ.DMHOSTUP AS DT_ENVIO_ASSESSORIA,
    CASE
      WHEN AG.AGFIRM = 'RR SALDO' THEN 'FLEX'
      WHEN AG.AGFIRM = 'SALDO REMANESC. - RN' THEN 'RN'
      WHEN AG.AGFIRM = 'TOLEDO SALDO' THEN 'TOLEDO'
      ELSE 'SUSPENSO'
    END AS ASSESSORIA,
    AG.AGFIRM AS COMPLEMENTO_ASSESSORIA,
    CASE
      WHEN DQ.DMLABEL2 = 'SD01' THEN 'Saldo'
      WHEN DQ.DMLABEL2 = 'AV01' THEN 'Aviso'
    END AS PRODUTO,
    DQ.DMNAME AS NOME_CLIENTE,
    U2.U2VLAVICLI AS SALDO_ATUAL,
    DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) AS DIAS_EM_ATRASO,
    CASE
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 6 AND 90 THEN '1'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 91 AND 180 THEN '2'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 181 AND 360 THEN '3'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 361 AND 540 THEN '4'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 541 AND 720 THEN '5'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 721 AND 1080 THEN '6'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) BETWEEN 1081 AND 1440 THEN '7'
      WHEN DATE_DIFF(CURRENT_DATE(), DATE(DQ.DMDLQDT), DAY) >= 1441 THEN '8'
    END AS FASE,
    U2.U2PLANO AS PLANO,
    U2.U2QTDPARC AS QTD_PARCELAS_PAGAS,
    U2.U2QTDPANPG AS PARCELAS_NAO_PAGAS,
    U2.U2QTDPAVINC AS PARCELAS_VINCENDAS,
    U2.U2VLTOTPARC AS VALOR_PARCELA
  FROM `bv-pdbd-prd.trusted_rcvry.delqmst` DQ
  INNER JOIN `bv-pdbd-prd.trusted_rcvry.uda2` U2
    ON DQ.DMACCT = U2.U2ACCT AND DQ.DMACCTG = U2.U2ACCTG
  INNER JOIN `bv-pdbd-prd.trusted_rcvry.agency` AG
    ON DQ.DMAGENCY = AG.AGAGENCY
  WHERE DQ.DMLABEL2 IN ('SD01','AV01')
    AND DQ.DMSTATUS IS NULL
) A);QUIT;
