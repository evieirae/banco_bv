PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.TB_PRIMEIRO_VCTO AS
	SELECT *
	FROM CONNECTION TO DBCON

			(

SELECT a.nucontrato, CAST(a.dtvencimento AS DATE) AS dtprivencimento,
c.DdDescontoFolha as DtDescFolha
FROM CPC.TbParcela a
INNER JOIN
DBA.tbproposta b
ON a.nucontrato = b.nucontratogestao
LEFT JOIN DBCOR.TbEmpresaConveniada c
on b.cdloja = c.cdloja

WHERE
a.sqcontratofinanceiro = 1 AND
a.nuparcela = 1
AND b.CdSubProduto = 12
	AND CAST(b.Dtentradaproposta AS DATE) between &database. and  DATEADD(DAY,-1,getdate())
	AND CAST(b.DtStatusProposta AS DATE) <= DATEADD(DAY,-1,getdate()));RUN;
