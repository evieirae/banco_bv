	WHEN A.QTDDIASATRASO <= 2520 THEN '26 - 2341 A 2520'
	WHEN A.QTDDIASATRASO <= 2700 THEN '27 - 2521 A 2700'
	WHEN A.QTDDIASATRASO <= 2880 THEN '28 - 2701 A 2880'
	WHEN A.QTDDIASATRASO <= 3600 THEN '29 - 2881 A 3600'
	WHEN A.QTDDIASATRASO  > 3600 THEN '30 - ACIMA 3600'
ELSE 'EM DIA' END AS FX_ATRASO,
	A.VRSALDO AS SALDO,
	A.VRPDD AS PDD_CONTABIL
FROM
	RAFA.CARTAO_3040_SEMARRASTO_&DATA. A LEFT JOIN (SELECT
	A.NUCPFCNPJ,
	A.NUPROPOSTA,
	B.DECISAO_VENDA
FROM
	BASEUNIQ.BASEUNICACARTOES B RIGHT JOIN (SELECT
	NUCPFCNPJ,
	MAX(NUPROPOSTA) AS NUPROPOSTA
FROM
	BASEUNIQ.BASEUNICACARTOES
WHERE
	FLATIVADA = 'S'
	AND
	DATEPART(DTENTRADAPROPOSTA) <= &DT_FIM_MES.
GROUP BY
	NUCPFCNPJ) A
	ON
	B.NUCPFCNPJ = A.NUCPFCNPJ
	AND
	B.NUPROPOSTA = A.NUPROPOSTA) B
	ON
	A.NUCPFCNPJ = B.NUCPFCNPJ
GROUP BY
	DATA_BASE,
	STATUS,
	BASE,
	PRODUTO,
	FAIXA_ATRASO,
	FX_ATRASO;
QUIT;
