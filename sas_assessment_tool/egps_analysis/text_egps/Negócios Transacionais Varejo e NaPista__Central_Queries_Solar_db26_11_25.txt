proc sql;

Create table base_solar_originacao as
Select distinct
t1.*,

Case
	When (t1.QtDiasCarencia) <= 30 then 30
	When (t1.QtDiasCarencia) between 31 and 60 then 60
	When (t1.QtDiasCarencia) between 61 and 90 then 90
	When (t1.QtDiasCarencia) between 91 and 120 then 120
End as DiasCarencia,

t1.QtDiasCarencia*(t1.VrPropostaOriginada) as VrCarenciaOrig,
t1.QtDiasCarencia*(t1.VrPropostaAprovada) as VrCarenciaAprov,
t1.QtDiasCarencia*(t1.VrPropostaEfetivada) as VrCarenciaEfet,

t1.DtStatusProposta-t1.DtEntradaProposta as DiasEsteira,

	

Case
    
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaOriginada > 0) then a.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaOriginada > 0) then b.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaOriginada > 0) then c.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaOriginada > 0) then d.TxRebate * t1.VrPropostaOriginada
	when (t1.FlPropostaOriginada > 0) then t1.VpTxGerencialRebate
    else .
End as VpTxNet_Orig,

Case
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaOriginada > 0) then a.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaOriginada > 0) then b.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaOriginada > 0) then c.TxRebate * t1.VrPropostaOriginada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaOriginada > 0) then d.TxRebate * t1.VrPropostaOriginada
    when (t1.FlPropostaOriginada > 0) then t1.VpTxPactuada
    else .
End as VpTxPactuada_Orig,

Case
    when (t1.FlPropostaOriginada > 0) then t1.VpTirContabil
    else .
End as VpTirContabil_Orig,



Case
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaAprovada > 0) then a.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaAprovada > 0) then b.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaAprovada > 0) then c.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaAprovada > 0) then d.TxRebate * t1.VrPropostaAprovada
    when (t1.FlPropostaAprovada > 0) then t1.VpTxGerencialRebate
    else .
End as VpTxNet_Aprov,

Case
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaAprovada > 0) then a.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaAprovada > 0) then b.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaAprovada > 0) then c.TxRebate * t1.VrPropostaAprovada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaAprovada > 0) then d.TxRebate * t1.VrPropostaAprovada
    when (t1.FlPropostaAprovada > 0) then t1.VpTxPactuada
    else .
End as VpTxPactuada_Aprov,

Case
    when (t1.FlPropostaAprovada > 0) then t1.VpTirContabil
    else .
End as VpTirContabil_Aprov,



Case
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaEfetivada > 0) then a.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaEfetivada > 0) then b.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaEfetivada > 0) then c.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaEfetivada > 0) then d.TxRebate * t1.VrPropostaEfetivada
    when (t1.FlPropostaEfetivada > 0) then t1.VpTxGerencialRebate
    else .
End as VpTxNet_Efet,

Case
    when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null and t1.FlPropostaEfetivada > 0) then a.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null and t1.FlPropostaEfetivada > 0) then b.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null and t1.FlPropostaEfetivada > 0) then c.TxRebate * t1.VrPropostaEfetivada
    when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null and t1.FlPropostaEfetivada > 0) then d.TxRebate * t1.VrPropostaEfetivada
    when (t1.FlPropostaEfetivada > 0) then t1.VpTxPactuada
    else .
End as VpTxPactuada_Efet,

Case
    when (t1.FlPropostaEfetivada > 0) then t1.VpTirContabil
    else .
End as VpTirContabil_Efet,


Case
	when t3.NmCampanhaCupom = "" THEN "-"
	else t3.NmCampanhaCupom
End as NmCampanhaCupom,

Case
	when t3.NmCampanhaCupom = "" then "N"
	else "S"
End as Flag_Cupom,

Case
	when t2.QtSeguros is null THEN 0
	else t2.QtSeguros
End as QtSeguros,

Case
	when t2.FlSPF is null THEN 0
	else t2.FlSPF
End as FlSPF,

Case
	when t2.VrSPF is null THEN 0
	else t2.VrSPF
End as VrSPF,

Case
	when t2.FlAP is null THEN 0
	else t2.FlAP
End as FlAP,

Case
	when t2.VrAP is null THEN 0
	else t2.VrAP
End as VrAP,

Case
	when t2.FlPlaca is null THEN 0
	else t2.FlPlaca
End as FlPlaca,

Case
	when t2.VrPlaca is null THEN 0
	else t2.VrPlaca
End as VrPlaca

from HOML.TBORIGINACAO as t1
left join DATASEG.TBSEGUROSCOMPLETA as t2 on (t2.Nuproposta=t1.Nuproposta)
left join AUXL.tbinfocupom as t3 on (t3.NuProposta = t1.NuProposta)

left join tuyra.solar_rebate_0424 as a on (t1.CdTabelaFinanciamento=a.TABELA and t1.QtParcelas=a.PRAZO)
left join tuyra.solar_rebate_0624 as b on (t1.CdTabelaFinanciamento=b.TABELA and t1.QtParcelas=b.PRAZO)
left join tuyra.solar_rebate_1024 as c on (t1.CdTabelaFinanciamento=c.TABELA and t1.QtParcelas=c.PRAZO)
left join tuyra.solar_rebate_&Data_Rebate as d on (t1.CdTabelaFinanciamento=d.TABELA and t1.QtParcelas=d.PRAZO)

where t1.NuAnoMes >= &Data_Ref and t1.nmgrupoprodutometa in ('Solar')

;quit;



PROC SORT DATA=base_solar_originacao NODUPKEY;
    BY NuProposta;
RUN;

  WHEN t1.Flag_Cupom = 'S' AND t1.NmCampanhaCupom LIKE '%_BV_%' THEN 'Combate 5'
  when t1.Flag_Cupom = 'S' and t1.NmCampanhaCupom like '%_PREMIUM_%' then 'Premium'
  WHEN t1.Flag_Cupom = 'S' AND t1.NmCampanhaCupom like '%BVTURBINADO_%' then 'BvTurbinado'
  WHEN t1.Flag_Cupom = 'S' AND
       t1.NmCampanhaCupom NOT LIKE '%_ENERGIA_%' AND
       t1.NmCampanhaCupom NOT LIKE '%_SOLAR_%' AND

       t1.NmCampanhaCupom NOT LIKE '%_RENOVAVEL_%' AND
       t1.NmCampanhaCupom NOT LIKE '%3_BELENUS_%' AND
       t1.NmCampanhaCupom NOT LIKE '%349_BELENUS3%' AND
       t1.NmCampanhaCupom NOT LIKE '%_BVSOLAR_%' AND
       t1.NmCampanhaCupom NOT LIKE '%4_BELENUS_%' AND
       t1.NmCampanhaCupom NOT LIKE '%349_BELENUS4_%' AND
       t1.NmCampanhaCupom NOT LIKE '%_BV_%' AND

	When (t1.DtPrimeiroVencimento - t1.DtContrato) <= 30 then 30
	When (t1.DtPrimeiroVencimento - t1.DtContrato) between 31 and 60 then 60
	When (t1.DtPrimeiroVencimento - t1.DtContrato) between 61 and 90 then 90
	When (t1.DtPrimeiroVencimento - t1.DtContrato) between 91 and 120 then 120

	Case
		when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null) then a.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null) then b.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null) then c.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null) then d.TxRebate*t1.VrProducao
		else t1.VpTxGerencialRebate
	End as VpTxNet,

		when (t1.DtEntradaProposta < '03jun2024'd and a.TABELA is not null) then a.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '03jun2024'd and b.TABELA is not null) then b.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '01jul2024'd and c.TABELA is not null) then c.TxRebate*t1.VrProducao
		when (t1.DtEntradaProposta >= '01nov2024'd and d.TABELA is not null) then d.TxRebate*t1.VrProducao
		else t1.VpTxJuros
	End as VpTxPactuada,

	when t3.NmCampanhaCupom = "" THEN "-"
	else t3.NmCampanhaCupom

%macro _eg_conditional_dropds /parmbuff;

   	%local num;
   	%local stepneeded;
   	%local stepstarted;
   	%local dsname;
	%local name;

   	%let num=1;
	
	
	%let stepneeded=0;
	%let stepstarted=0;
   	%let dsname= %qscan(&syspbuff,&num,',()');
	%do %while(&dsname ne);
		%let name = %sysfunc(left(&dsname));
		%if %qsysfunc(exist(&name)) %then %do;
			%let stepneeded=1;
			%if (&stepstarted eq 0) %then %do;
				proc sql;
				%let stepstarted=1;

			%end;
				drop table &name;
		%end;

			%let stepneeded=1;
			%if (&stepstarted eq 0) %then %do;
				proc sql;
				%let stepstarted=1;
			%end;
				drop view &name;
		%end;
		%let num=%eval(&num+1);
      	%let dsname=%qscan(&syspbuff,&num,',()');
	%end;
	%if &stepstarted %then %do;
		quit;
	%end;

	Case
	    when b.CdTabelaFinanciamento in ('BVXGK','BV7YG') then 'CS_Padrão'
	    when b.CdTabelaFinanciamento in ('BV13J','BV7YH') then 'CS_Top 1'

	    when b.CdTabelaFinanciamento in ('BVYB4') then 'MFS_Combate II'
	    when b.CdTabelaFinanciamento in ('BV29Y','BV7YJ') then 'MFS_Top'
	    when b.CdTabelaFinanciamento in ('BVXVD','BV7YW') then 'MFS_Contingencia'

	    when b.CdTabelaFinanciamento in ('BV60U', 'BV7DH', 'BV7LH') then 'Belenus'
	    when b.CdTabelaFinanciamento in ('BV7LX') then 'Fotus'
	    when b.CdTabelaFinanciamento in ('BV7U5') then 'ASC'

	    when b.CdTabelaFinanciamento in ('BV654') then 'AVT'

	    when t1.CdTabelaFinanciamento in ('BVXVD','BV7YW') then 'MFS_Contingencia'

	    when t1.CdTabelaFinanciamento in ('BV7YY','BVXGP') then 'Comissionamento 2'

	    when t1.CdTabelaFinanciamento in ('BV7II') then 'Feirao'
	    when t1.CdTabelaFinanciamento in ('BV7LV','BV7YX') then 'Guerrilha Padrão'

	    when t1.CdTabelaFinanciamento in ('BV60U', 'BV7DH', 'BV7LH') then 'Belenus'
	    when t1.CdTabelaFinanciamento in ('BV7LX') then 'Fotus'
	    when t1.CdTabelaFinanciamento in ('BV7U5') then 'ASC'
	    when t1.CdTabelaFinanciamento in ('BV7O3') then 'SolarPrime'
	    when t1.CdTabelaFinanciamento in ('BV654') then 'AVT'
	    when t1.CdTabelaFinanciamento in ('BV7FL') then 'EcoPower'
	    when t1.CdTabelaFinanciamento in ('BV7YB') then 'Guerrilha Sul'

	    when t1.CdTabelaFinanciamento in ('BV2B5','BV653','BV6L3') then 'Especiais'

	    else 'Outras'
	End as DsTipoTabela,

	COUNT (DISTINCT t1.NuCpfCNPJSimu) AS QtdSimu,
	COUNT (Distinct MSGPOS.NuCpfCNPJSimu) AS QtdSimuAprov,
	SUM (t1.FlPropostaOriginada) AS QtdOrig,
	SUM (t1.FlPropostaAprovada) AS QtdAprov,
	SUM (t1.FlPropostaEfetivada) AS QtdEfet,

	SUM (t1.VrPropostaOriginada) AS VrOrig,
	SUM (t1.VrPropostaAprovada) AS VrAprov,
	SUM (t1.VrPropostaEfetivada) AS VrEfet,



	(SUM (t1.FlPropostaEfetivada) / COUNT (DISTINCT t1.NuCpfCNPJSimu)) as ConvGeral format=percent8.2,



	
	(SUM(t4.VpTxNet_Orig)) AS Valor_Tx_Orig,
	(SUM(t4.VpTxPactuada_Orig)) AS Valor_Tx_Orig,
	(SUM(t4.VpTirContabil_Orig)) AS Valor_TIR_Orig,

	
	(SUM(t4.VpTxNet_Aprov)) AS Valor_Tx_Aprov,
	(SUM(t4.VpTxPactuada_Aprov)) AS Valor_Tx_Pactuada_Aprov,
	(SUM(t4.VpTirContabil_Aprov)) AS Valor_TIR_Aprov,

	
	(SUM(t4.VpTxNet_Efet)) AS Valor_Tx_Efet,
	(SUM(t4.VpTxPactuada_Efet)) AS Valor_Tx_Pactuada_Efet,
	(SUM(t4.VpTirContabil_Efet)) AS Valor_TIR_Efet

	FROM Work.TbFunilSimul_Filtros as t1

	Left Join

	    From Bases.TbFunilSimulacao_Solar
	    Where DtSimulacaoProposta >= &Data_Simul and FlMSGPositiva = 1

	On t1.CodSimu = MSGPOS.CodSimu

	Left Join WORK.base_solar_originacao as t4 on (t1.NuProposta = t4.NuProposta)

	GROUP BY 1,2,3
	order by 1
