%LET DATA_AJUSTE = '30SEP2025'D; 
%LET VEICULO_LEGAL = 640; 



PROC SQL;
   CREATE TABLE WORK.QUERY_FOR_CT2 AS
   SELECT t1.Filial,
          t1.'Data Lcto'n,
          t1.'Cta Debito'n,
          t1.'Cta Credito'n,
          t1.Valor,
          t1.'Hist Lanc'n,
          t1.'C Custo Deb'n,
          t1.'C Custo Cred'n,
          t1.'Item Conta D'n,
          t1.'Item Conta C'n,
          t1.'Clas Valor D'n,
          t1.'Clas Valor C'n
      FROM WORK.CT2 t1;
QUIT;



PROC SQL;
   CREATE TABLE WORK.QUERY_FOR_DEPARA_CONTA_CV_VL AS
   SELECT t1.'Conta Contábil'n
      FROM WORK.DEPARA_CONTA_CV_VL t1;
QUIT;



PROC SQL;
   CREATE TABLE EMPILHA_CONTA_DEBITO AS
   SELECT t1.'Cta Debito'n,
          t1.Valor,
          t1.'Hist Lanc'n,
          t1.'C Custo Deb'n,
          t1.'Clas Valor D'n
      FROM WORK.QUERY_FOR_CT2 t1
	  WHERE t1.'Cta Debito'n IN (SELECT 'Conta Contábil'n FROM WORK.QUERY_FOR_DEPARA_CONTA_CV_VL)
;
QUIT;



PROC SQL;
   CREATE TABLE EMPILHA_CONTA_CREDITO AS
   SELECT t1.'Cta Credito'n,
          (t1.Valor)*-1 AS Valor,
          t1.'Hist Lanc'n,
          t1.'C Custo Cred'n,
          t1.'Clas Valor C'n
      FROM WORK.QUERY_FOR_CT2 t1
	  WHERE t1.'Cta Credito'n IN (SELECT 'Conta Contábil'n FROM WORK.QUERY_FOR_DEPARA_CONTA_CV_VL)
;
QUIT;



PROC SQL;
   CREATE TABLE EMPILHA_CRED_DEB AS
   SELECT t1.'Cta Debito'n AS 'Conta Contábil'n,
          t1.Valor AS 'Valor do Ajuste'n,
          t1.'Hist Lanc'n AS 'Histórico'n,
          t1.'C Custo Deb'n AS 'Centro de Custo'n,
          t1.'Clas Valor D'n AS 'Classe de Valor'n
	FROM EMPILHA_CONTA_DEBITO t1

   UNION ALL
   SELECT t1.'Cta Credito'n AS 'Conta Contábil'n,
		  t1.Valor AS 'Valor do Ajuste'n,
          t1.'Hist Lanc'n AS 'Histórico'n,
		  t1.'C Custo Cred'n AS 'Centro de Custo'n,
          t1.'Clas Valor C'n AS 'Classe de Valor'n
      FROM EMPILHA_CONTA_CREDITO t1
	  ;QUIT;



PROC SQL;
   CREATE TABLE EMPILHA_TEMPLATE AS
   SELECT &DATA_AJUSTE. FORMAT DATE10. AS 'Data do Ajuste'n,
   		  &VEICULO_LEGAL. AS 'Código do Veículo'n,
		  t1.'Conta Contábil'n,
          t1.'Valor do Ajuste'n,
          t1.'Histórico'n,
		  t2.'Grupo Acionista'n,
          t2.'Grupo Conta'n,
		  'M' AS 'Tipo de Ajuste'n,
          t1.'Centro de Custo'n,
          t1.'Classe de Valor'n
      FROM WORK.EMPILHA_CRED_DEB t1

	LEFT JOIN WORK.DEPARA_CV t2 ON ( t1.'Classe de Valor'n = t2.'Classe de Valor'n )

;
QUIT;



PROC EXPORT
  DATA=EMPILHA_TEMPLATE
  DBMS=XLSX
  OUTFILE="/riscosbvf/financas/PE_DATA_VIZ/TEMPLATE_AJUSTE_CT2.xlsx"
  REPLACE;
  SHEET= EMPILHA_TEMPLATE;
RUN;
