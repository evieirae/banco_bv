/* Declarar libnames e strings de conexão com Sybase */
%include '/infogrc/utils/all-includes.sas';
LIBNAME CONTRATO '/infogrc/contratacao_veiculos/bases'; /*CONTRATOS C2C*/

/* Chamar macros principais */
%SAFRAESG();

options compress=yes;

/* INFORMAR QTD MESES PARA O DASH */
%let qtMeses = 6;
/*%let qtMeses = 1;*/
DATA _NULL_;
  /* Define parâmetros com base no dia atual */
  hoje = TODAY();
  hojeXM = INTNX('MONTH', hoje, ((&qtMeses. - 1) * -1), 'BEGINNING');
  hoje13M = INTNX('MONTH', hoje, -12, 'BEGINNING');
/*	hoje13M = INTNX('MONTH', hoje, -1, 'BEGINNING');*/

  /* Calcula variáveis */
  safraXM = (YEAR(hojeXM) * 100) + MONTH(hojeXM);
  anoAtual = YEAR(hoje);
  anoXM = YEAR(hojeXM);
  ano13M = YEAR(hoje13M);
  dtIniPta = (safraXM * 100) + 1;
  safra13M = (YEAR(hoje13M) * 100) + MONTH(hoje13M);

  /* Declarar macro variáveis */
  CALL SYMPUT('safraXM', safraXM);
  CALL SYMPUT('anoXM', PUT(anoXM,4.));
  CALL SYMPUT('ano13M', PUT(ano13M,4.));
  CALL SYMPUT('anoAtual', PUT(anoAtual,4.));
  CALL SYMPUT('dtIniPta', "'"||PUT(dtIniPta,8.)||"'");
  CALL SYMPUT('safra13M', safra13M);
  CALL SYMPUT('hojeXM', "'"||PUT(hojeXM,8.)||"'");
RUN;
%PUT &safraXM, &anoXM ano13M &anoAtual &dtIniPta &safra13M;
