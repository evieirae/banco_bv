    from tmp.TbProducao;

Proc Sql;
Create Table TbMetaProducao as
Select Distinct
NuAnoMes
,DsLoginN1 as DsLoginN1C
,DsLoginN2A as DsLoginN2C
,DsLoginN3A as DsLoginN3C
,NmAreaEstruturaComercial
,NmColaborador
,AtingProdN1Proj
,Case When Coalesce(AtingProdN1Proj,0) Between 0 and 0.600000000 Then 1 Else 0 End as Qt_Grs_Ate60
,Case When Coalesce(AtingProdN1Proj,0) Between 0.600000001 and 0.800000000 Then 1 Else 0 End as Qt_Grs_Ate80
,Case When Coalesce(AtingProdN1Proj,0) Between 0.800000001 and 1.000000000 Then 1 Else 0 End as Qt_Grs_Ate100
,Case When Coalesce(AtingProdN1Proj,0) > 1.000000000 Then 1 Else 0 End as Qt_Grs_Acima100
,Max(&DtAtualizacao.) as DataBase Format DDMMYY10.

From Bases.TbMetaProducaoFinal
Where NuAnomes >= &Base_m5. and NmGrupoProdutoMeta = 'Solar' and TpValor = 'VrProducao' and VrMetaNIN1 NE . and NmColaborador NE "";
Run;

Proc Sql;
Create Table Base_Pessoas_Canal_Total as
Select
'Gestao' as Base
,Prod.NuAnoMes
,'Canal' as Visao
,"" as DsLoginN1C
,"" as DsLoginN2C
,"" as DsLoginN3C
,'Canal Solar' as Dimensao1
,'Canal Solar' as Dimensao2
,'Canal Solar' as Dimensao3
,'Mês Fechado' as Semana
,0 as QtCorridosDiaUtilMes
,Count(distinct NmColaborador) as QtGrs
,Sum(Qt_Grs_Ate60) as Qt_Grs_Ate60
,Sum(Qt_Grs_Ate80) as Qt_Grs_Ate80
,Sum(Qt_Grs_Ate100) as Qt_Grs_Ate100
,Sum(Qt_Grs_Acima100) as Qt_Grs_Acima100
,Max(&DtAtualizacao.) as DataBase Format DDMMYY10.

From TbMetaProducao as Prod
Where NuANomes >= &Base_m5. and
		NmAreaEstruturaComercial in

			'Canal Solar CO e NO',
			'Canal Solar MG, BA, AL e SE',
			'Canal Solar NE',

			'Canal Solar SP, RJ e ES',
			'Canal Solar Sul'
		)
