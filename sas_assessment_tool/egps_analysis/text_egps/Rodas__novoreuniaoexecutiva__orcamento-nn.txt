LIBNAME RELEXC '/infogrc/relatorioexecutivo';

PROC SQL;
  CREATE TABLE orc_nn_01 AS
    SELECT
      NuAnoMes AS Safra,
	  NmGrupoProdutoMeta AS Produto,
	  COALESCE(DsTipoCanalProducao,'DESCONHECIDO') AS Canal,
      SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS OrcamentoVrProducao FORMAT=NUM18.2,
	  SUM(CASE WHEN Tipo = 'Meta' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS MetaVrProducao FORMAT=NUM18.2,
	  SUM(CASE WHEN Tipo = 'BN' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS BNVrProducao FORMAT=NUM18.2,
      SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = 'VpTxSpread' THEN VrMedida ELSE 0 END) AS OrcamentoSpreadVp FORMAT=NUM18.2
    FROM RELEXC.TbMeta
	WHERE NmGrupoProdutoMeta IN ('Estudantil','Saude','Solar','Turismo')
	  AND NmGrupoProdutoMeta IS NOT NULL
	  AND NuAnoMes >= 202007
    GROUP BY 1,2,3;
RUN;

PROC SQL;
  CREATE TABLE orc_nn AS
    SELECT
      *
    FROM orc_nn_01
	WHERE (OrcamentoVrProducao + MetaVrProducao + BNVrProducao + OrcamentoSpreadVp) <> 0;
RUN;

filename exprt '/infogrc/novoreuniaoexecutiva/bases/orcamento_nn.csv' encoding="utf-8";

proc export
data=orc_nn
outfile=exprt replace
dbms=dlm;
delimiter=';';
run;
