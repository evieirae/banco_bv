PROC SQL;
   CREATE TABLE WORK.OUVIDORIA_COM_CONTRATO AS
   SELECT DISTINCT t1.*,
		  t2.DtContrato,
          t2.NuContrato as NuContrato2,
          t2.NuCpfCnpj,
          t2.CdPessoa,
          t2.NmPessoa,
		  t2.CdColaborador,
t2.NmColaborador,
t2.NmGrupoCargoColaborador,
t2.NuCnpjLoja,
t2.CdParceiroComercial,
t2.NmParceiroComercial

      FROM WORK.OUVIDORIA_AJUSTADO t1
           LEFT JOIN WORK.TBPRODUCAO t2 ON (t1.CPF_CNPJ_2 = t2.NuCpfCnpj and DATA_ABERTURA_ >= DtContrato)
WHERE t1.CLASSIFICACAO in
('Reclamação',
'RECLAMAÇÃO',
'RECLAMACÃO ONLINE',
'Reclamação Online')
;
QUIT;

PROC SQL;
   CREATE TABLE WORK.QUERY_FOR_OUVIDORIA_AJUSTAD2O AS
   SELECT
t1.PRODUTO,
t1.SINTOMA,
t1.SUBMOTIVO,
t1.MOTIVO,
          
            (COUNT(t1.PRODUTO)) AS COUNT_of_PRODUTO
      FROM WORK.OUVIDORIA_AJUSTADO t1
	  WHERE
	  UPPER(t1.PRODUTO) LIKE '%CONSIGNADO%' AND
     (UPPER(t1.SINTOMA) LIKE '%CORRESPONDENTE%' or
      UPPER(t1.MOTIVO) LIKE '%CORRESPONDENTE%' or
      UPPER(t1.SUBMOTIVO) LIKE '%CORRESPONDENTE%' or
      UPPER(t1.SINTOMA) LIKE '%LOJISTA%' or
      UPPER(t1.MOTIVO) LIKE '%LOJISTA%' or
      UPPER(t1.SUBMOTIVO) LIKE '%LOJISTA%' or
	  UPPER(t1.SINTOMA) LIKE '%COBAN%' or
      UPPER(t1.MOTIVO) LIKE '%COBAN%' or
      UPPER(t1.SUBMOTIVO) LIKE '%COBAN%' or
      UPPER(t1.SINTOMA) LIKE '%PARCEIRO%' or
      UPPER(t1.MOTIVO) LIKE '%PARCEIRO%' or
      UPPER(t1.SUBMOTIVO) LIKE '%PARCEIRO%')
      GROUP BY 1,2,3,4
      ORDER BY COUNT_of_PRODUTO DESC;
QUIT;
