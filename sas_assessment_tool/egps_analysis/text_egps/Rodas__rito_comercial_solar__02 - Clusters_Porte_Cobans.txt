/*Gerando a base de Target sem o efeito da duplicidade devido migração da Sinergia para o CS*/
Proc Sql;
Create Table Base_Porte_Cobans as
Select Distinct
Target.NuAnomes
,Parc.DsLoginN1C
,Parc.DsLoginN2C
,Parc.DsLoginN3C
,Target.NuCnpjLojaFavorecido
,Target.CdParceiroComercial
,Target.Cluster_Final as Porte
,Target.TipoParceiro
,Parc.DtCadastro as DtCadastro Format DDMMYY10.
From ritosol.Classificacao_Parceiros_MIS as Target
Left join Bases.TbParceiroComercial as Parc On Target.CdParceiroComercial = Parc.CdParceiroComercial
;Quit;

Proc Sql;
Create Table Base_Porte_Cobans_CNPJ as
Select Distinct
Target.NuAnomes
,Target.NuCnpjLojaFavorecido
,Target.Cluster_Final as Porte
,Target.TipoParceiro

From ritosol.Classificacao_Parceiros_MIS as Target
;Quit;

Proc Sql;
Create Table Base_Cobans_Solar as
Select Distinct
TbParc.CdParceiroComercial
,TbParc.NmParceiroComercial
,TbParc.NmAreaEstruturaComercialC
,TbParc.NmColaboradorC
,TbParc.DsLoginN1C
,TbParc.DsLoginN2C
,TbParc.DsLoginN3C
,TbParc.NuCnpjLoja
,TbParc.DtCadastro
,TbParc.FlAtivo
,Target.Porte
,Target.TipoParceiro
,TbParc.DtCadastro as DtCadastro Format DDMMYY10.
From Bases.TbParceiroComercial as TbParc
	Left Join Base_Porte_Cobans_CNPJ as Target On TbParc.NuCnpjLoja = Target.NuCnpjLojaFavorecido
Where TbParc.TpAtividade = 7 and TbParc.NmCanalEstruturaComercialC = 'Canal Solar'
order by TbParc.NuCnpjLoja, TbParc.FlAtivo DESC, TbParc.DtCadastro DESC, Target.Porte Desc
;Quit;

proc Sort NoDupkey Data=Base_Cobans_Solar out=Base_Cobans_Solar_Saida; by NuCnpjLoja; Quit;

Data ritosol.Base_Cobans_Solar_Saida;
Set Base_Cobans_Solar_Saida;
Run;
