connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd'
READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100

create table BaseFraude as
select
*
from connection to perm(

WITH
    propostas AS (
        SELECT DISTINCT
            event_id as IdEvento,
            event_external_id as NuProposta,
            branch_id as CdParceiroComercial,
            product_type AS NmProduto

        FROM bv-pdbd-prd.trusted_corp_fraudes.tb_am_event_account_opening

        WHERE DATE(TIMESTAMP_MILLIS(event_received_at)) >= DATE_ADD(CURRENT_DATE(), INTERVAL -12 MONTH)
          AND outcome_decision = 'review'
		  AND product_type IN ('CDC COM GARANTIA', 'CDC SEM GARANTIA')
/*          AND event_external_id = '405405195'*/
    ),

      status_inicial as (
        SELECT
        event_id as IdEvento,
        timestamp_millis(timestamp) AS DtEntradaEvento

        FROM bv-pdbd-prd.trusted_corp_fraudes.tb_am_event_status es

        WHERE DATE(TIMESTAMP_MILLIS(timestamp)) >= DATE_ADD(CURRENT_DATE(), INTERVAL -12 MONTH)
        AND new_status_id = 'ready'
/*        and event_id in ('73ed009d-873a-4ea4-a452-2beedcdf7bef', '06c59403-e679-41b2-8edc-48fc58981616')*/
    ),

    status_final AS (
        SELECT DISTINCT
            es.event_id as IdEvento,
            TIMESTAMP_MILLIS(es.`TIMESTAMP`) AS DtFinalEvento,
			new_status_id as DsStatusEvento,
            REPLACE(pulse_user.name, '@bv.com.br', '') AS NmAnalista

        FROM bv-pdbd-prd.trusted_corp_fraudes.tb_am_event_status AS es
        INNER JOIN bv-pdbd-prd.trusted_corp_fraudes.tb_pulse_user AS pulse_user ON pulse_user.`id` = es.`USER_ID`

        WHERE pulse_user.`name` <> 'system'
        AND DATE(TIMESTAMP_MILLIS(es.timestamp)) >= DATE_ADD(CURRENT_DATE(), INTERVAL -12 MONTH)
/*        and event_id in ('73ed009d-873a-4ea4-a452-2beedcdf7bef', '06c59403-e679-41b2-8edc-48fc58981616')*/
    ),

    parceiro AS (
        SELECT
            parceiro.CdParceiroComercial,
            filial.dsfilial as DsFilial

        FROM bv-pdbd-prd.trusted_dba.tbparceirocomercial AS parceiro
        LEFT JOIN bv-pdbd-prd.trusted_dba.tbloja AS loja ON parceiro.CdParceiroComercial = loja.Cdloja
        LEFT JOIN bv-pdbd-prd.trusted_dba.tbregiao AS regiao ON loja.CdRegiao = regiao.CdRegiao
        LEFT JOIN bv-pdbd-prd.trusted_dba.tbfilial AS filial ON regiao.CdFilial = filial.CdFilial
    ),

    pending_hist AS (
        SELECT
            event_id as IdEvento,
            MIN(timestamp_millis(timestamp)) as DtPendenciamento

        FROM bv-pdbd-prd.trusted_corp_fraudes.tb_am_event_status

        WHERE DATE(TIMESTAMP_MILLIS(timestamp)) >= DATE_ADD(CURRENT_DATE(), INTERVAL -12 MONTH)
            AND new_status_id = 'pending'

        GROUP BY event_id
    ),

    tb_final AS (
        SELECT *
        FROM (
            SELECT
                propostas.NuProposta,
                propostas.IdEvento,
                inicial.DtEntradaEvento,
				pending_hist.DtPendenciamento,
                final.DtFinalEvento,
				propostas.NmProduto,
                final.NmAnalista,
                parceiro.DsFilial,
				CASE WHEN (final.DsStatusEvento = 'pending' OR final.DtFinalEvento IS NULL) THEN 'S' ELSE 'N' END AS FlEventoPendente,
                CASE WHEN pending_hist.IdEvento IS NOT NULL THEN 'S' ELSE 'N' END AS FlEventoPendenciado,
                ROW_NUMBER() OVER (PARTITION BY propostas.NuProposta, propostas.IdEvento ORDER BY final.DtFinalEvento DESC ) AS row_num
                FROM propostas
                LEFT JOIN status_inicial AS inicial ON propostas.IdEvento = inicial.IdEvento
                LEFT JOIN status_final AS final ON propostas.IdEvento = final.IdEvento
                LEFT JOIN parceiro ON CAST(propostas.CdParceirocomercial AS INT64) = CAST(parceiro.CdParceiroComercial AS INT64)
                LEFT JOIN pending_hist ON propostas.IdEvento = pending_hist.IdEvento
        )
      WHERE row_num = 1
    )

    SELECT
        cast (NuProposta as int64) as NuProposta,
        IdEvento,
        DtEntradaEvento,
		DtPendenciamento,
        DtFinalEvento,
		NmProduto,
        NmAnalista,
        DsFilial,
		FlEventoPendente,
        FlEventoPendenciado
    FROM tb_final

);
disconnect from perm;
quit;

PROC SQL;
CREATE TABLE ConversaoC6_DadosCliente AS
SELECT DISTINCT
	NuAnoMes
	,Banco
	,CdParceiroComercial
	,NmParceiroComercial
	,Produto
	,Regional
	,Filial
	,CdGF
	,CdGR
	,CdGREG
	,LstFocais
	,CdSimulacaoInicial
	,StatusSimulacao
	,Mensageria
	,NuCpfCnpj

	,NuTelefone1

	,NuRating
	,FlPA
	,DtInclusaoGravame
	,StPagoLoja
	,FxIdade
	,DtAtualizacao

,CASE
  WHEN FlAbandono = 'S'

    AND LOWER(Mensageria) NOT LIKE '%1. alta chance%'

    AND LOWER(Mensageria) NOT LIKE '%2. grande chance >70%'
    AND LOWER(Mensageria) NOT LIKE '%2. tem chance%'
    AND LOWER(Mensageria) NOT LIKE '%3. boa chance >50%'
  THEN ''
  ELSE Preenchimento
END AS Preenchimento

	,FlAbandono

	,CASE
  WHEN FlAbandono = 'S'
    AND LOWER(Mensageria) NOT LIKE '%0. aprovado%'
    AND LOWER(Mensageria) NOT LIKE '%0. aprovado 100%'
    AND LOWER(Mensageria) NOT LIKE '%1. alta chance%'
    AND LOWER(Mensageria) NOT LIKE '%1. aprovado com rastreador%'
    AND LOWER(Mensageria) NOT LIKE '%1. muita chance >85%'
    AND LOWER(Mensageria) NOT LIKE '%2. grande chance >70%'
    AND LOWER(Mensageria) NOT LIKE '%2. tem chance%'
    AND LOWER(Mensageria) NOT LIKE '%3. boa chance >50%'
  THEN ''
  ELSE BancoRetornoAbandono
END AS BancoRetornoAbandono
,FlFraude
