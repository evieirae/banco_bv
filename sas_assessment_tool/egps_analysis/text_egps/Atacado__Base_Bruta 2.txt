proc sql;
  create table TBCLASS_MONITORAMENTO_DIME as
  select a.cd_cnpj,
         a.cd_cnpj_raiz,
         a.nu_classificacao,
         a.ds_modelo,
         a.dt_classificacao,
         a.dt_movimento,
         input(a.cd_grupo_economico, best32.) as cd_grupo_economico,

         a.ds_monitoramento,

         a.ds_login,

         a.ts_alteracao_dvry
  from tbclass_monitoramento_dime_GCP as a
  order by input(a.cd_pessoa, best32.), dt_referencia desc
  ;
quit;

proc sql;
    create table TBCLASS_MONITORAMENTO_DIME as
    select datepart(max(ts_inclusao_dvry)) as DtProcessamento format=date9. as DtProcessamento, a.*
    from TBCLASS_MONITORAMENTO_DIME as a
    inner join (
        select cd_pessoa,
               ds_monitoramento,
               max(ts_inclusao_dvry) as max_ts_inclusao format=datetime27.6
        from TBCLASS_MONITORAMENTO_DIME
        group by cd_pessoa, ds_monitoramento
    ) as b
    on  a.cd_pessoa        = b.cd_pessoa
    and a.ds_monitoramento  = b.ds_monitoramento
    and a.ts_inclusao_dvry  = b.max_ts_inclusao
	order by cd_pessoa, dt_referencia desc

;quit;

proc sort data=TBCLASS_MONITORAMENTO_DIME;
    by cd_pessoa descending dt_classificacao;
run;

        CASE

            WHEN a.NmDivisaoOfficer IN ('') THEN ''
            WHEN a.NmDivisaoOfficer IN (

            ELSE a.DsClassificacaoSME

            WHEN b.DsStatus_Monitoramento IS NOT NULL AND
                 b.DsStatus_Monitoramento NE
                 CASE
                     WHEN CALCULATED Segmento_Bruta IN ('PME', 'SME') AND
                          UPCASE(a.DsClassificacao) LIKE 'PCMR%' THEN a.DsClassificacao
                     WHEN CALCULATED Segmento_Bruta = 'CIB' THEN a.DsClassificacao
                     WHEN CALCULATED status1 = '' AND a.FlEsteiraPME = 0 THEN

                             WHEN a.DsClassificacao = 'CMR' AND CALCULATED Segmento_Bruta ^= 'CIB' THEN ''
                             ELSE a.DsClassificacao

                     ELSE CALCULATED status1
                 END

            ELSE
                 CASE
                     WHEN CALCULATED Segmento_Bruta IN ('PME', 'SME') AND
                          UPCASE(a.DsClassificacao) LIKE 'PCMR%' THEN a.DsClassificacao
                     WHEN CALCULATED Segmento_Bruta = 'CIB' THEN a.DsClassificacao
                     WHEN CALCULATED status1 = '' AND a.FlEsteiraPME = 0 THEN
                         CASE

                     ELSE CALCULATED status1
