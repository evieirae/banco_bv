PROC SQL;
	CREATE TABLE Tb_Fatura_TpPgto
		AS SELECT DISTINCT
			a.*
			, ant2.cdfaturaanttexto
			, prox.cdfaturatexto as cdfaturaproxtexto
			, CASE
				WHEN CdRubricaTransacaoCartao = '5568' THEN 'Parc Fatura'
				WHEN CdRubricaTransacaoCartao = '5578' THEN 'Parc Acordo'
				WHEN CdRubricaTransacaoCartao = '5572' THEN 'Parc SaldoTotal'
				
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) = a.VrSaldoFatura THEN 'PagtoTotal'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) > a.VrSaldoFatura THEN 'PagtoAcimaTotal'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) = a.VrPagamentoMinimoFatura THEN 'Rotativo'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) between a.VrPagamentoMinimoFatura AND a.VrSaldoFatura  THEN 'Rotativo2'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) = 0 THEN 'SemPagto'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) < a.VrPagamentoMinimoFatura THEN 'PagtoAbaixoMinimo'
			ELSE 'Verificar' END AS TpF
			, c.DsTransacao
			, CdRubricaTransacaoCartao as rubrica_parc
			, c.VrTransacao
			, pf.nmtipoparcelado
			, pf.dtAdesao
			, pf.NuParcelaAdesao
			, pf.VrParcelaAdesao
			, pf.VrPagamentoAdesao
			, pf.VrContratoAdesao
			, pf.VrTaxaAdesao
			, pf.TxRateAdesao
			, d.cdfaturastring

			
			, coalesce(d.encargos_rotativo, 0)
				+ coalesce(d.juros_remuneratorios, 0)
				+ coalesce(e.mora, 0)
				+ coalesce(e.multa, 0)
				as encargos_M0
			, cred.VrCreditos
			, coalesce(e.mora, 0) + coalesce(e.multa, 0) as mora_multa_m0

			, coalesce(e.multa, 0) as multa_m0
		FROM faturas AS A
		left join fatura_anterior ant
			on a.cddimcontacartao = ant.cddimcontacartao
				and a.cdfaturatexto = ant.cdfaturaanttexto
		left join fatura_anterior ant2
			on a.cddimcontacartao = ant2.cddimcontacartao
				and a.cdfaturatexto = ant2.cdfaturatexto
		left join fatura_anterior prox
			on a.cddimcontacartao = prox.cddimcontacartao
				and a.cdfaturatexto = prox.cdfaturaanttexto
		left join tb_creditos as cred
			on prox.cdfaturatexto = cred.cdfaturastring
		LEFT Join Tb_EntradaParc2 AS C
			ON ant.CdDimContaCartao = c.CdDimContaCartao
				AND prox.cdfaturatexto = c.cdfaturastring
		left join tb_encargos as d
			on a.cddimcontacartao = d.cddimcontacartao
				and a.cdfaturatexto = d.cdfaturastring

				WHEN CdRubricaTransacaoCartao = '5578' THEN 'Parc Acordo'

				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) = a.VrSaldoFatura THEN 'PagtoTotal'

				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) between a.VrPagamentoMinimoFatura AND a.VrSaldoFatura  THEN 'Rotativo2'
				WHEN coalesce(-cred.vrCreditos, -a.VrPagamentoFatura) = 0 THEN 'SemPagto'

		left join fatura_anterior ant
			on a.cddimcontacartao = ant.cddimcontacartao
