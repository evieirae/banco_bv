PROC SQL;
CREATE TABLE SCORE_CONTRATOS_PISMO AS SELECT DISTINCT
INPUT(SUBSTR(cd_conta_cartao,3,8),11.) AS Contrato, 1 AS Score
FROM EstadoContaCartao_Pismo;
QUIT;

PROC SQL;
CREATE TABLE SCORE_CONTRATOS_PISMO_B AS SELECT
A.*,
B.nucontratocreditocartao,
INPUT(B.nucontratocreditocartao,9.) AS CONTRATO_CYBER
FROM
SCORE_CONTRATOS_PISMO A LEFT JOIN
(SELECT DISTINCT nucontratocreditocartao,nucontacartao FROM DEPARA_Contratos_PISMO) B
ON A.CONTRATO = B.nucontacartao;
QUIT;

DATA SCORE_CONTRATOS_PISMO_C;
SET SCORE_CONTRATOS_PISMO_B;

ISSUER = '0001';
BRANCH = '02';
PRODUCT = '05';
CONTRACT = PUT(CONTRATO_CYBER, Z9.);

FULL_CONTRACT = CATS (ISSUER,BRANCH,PRODUCT,CONTRACT);

WHERE CONTRATO_CYBER > 0 ;
