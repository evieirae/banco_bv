CREATE TABLE CP.RECEB_CP_&MES_REF. AS
SELECT
	NUCONTRATO,
	SQCONTRATOFINANCEIRO,
	DATEPART(DTRECEBIMENTO) AS DTRECEBIMENTO FORMAT DDMMYY10.,
	DATEPART(DTPROCESSAMENTO) AS DTPROCESSAMENTO FORMAT DDMMYY10.,
	VRRECEBIDO,
	VRPRINCIPALDEVIDO,
	VRPRINCIPALBAIXADO,
	TPACORDONEGOCIACAO
FROM CONNECTION TO DBCON
(SELECT
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTRECEBIMENTO,
	A.DTPROCESSAMENTO,
	B.TPACORDONEGOCIACAO,
	SUM(A.VRRECEBIDO) AS VRRECEBIDO,
	SUM(A.VRPRINCIPALDEVIDO) AS VRPRINCIPALDEVIDO,
	SUM(A.VRPRINCIPALBAIXADO) AS VRPRINCIPALBAIXADO
FROM
	DBA.VWTBPARCELARECEBIMENTO A LEFT JOIN DBCONTENCIOSO.TBNEGOCIACAOCONTRATO B
	ON
	A.NUCONTRATO = B.NUCONTRATOGESTAO
	AND
	A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
	AND
	A.DTRECEBIMENTO = B.DTPAGAMENTO
WHERE
	DTRECEBIMENTO BETWEEN &DT_INI. AND &DT_FIM.
	AND
	FLESTORNADO = 'N'
	AND
	BANCO NOT IN  ('DbGestaoCDCCG','DbGestaoLSG')
	AND
	CDAGENTERECEBEDOR IN (126,145,146,153,227,249)
GROUP BY
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTRECEBIMENTO,
	A.DTPROCESSAMENTO,
	B.TPACORDONEGOCIACAO);
QUIT;
RUN;

PROC SQL;
CREATE TABLE PERF_MASSIFICADOS_SCORE AS
SELECT DISTINCT
	A.DT_REFERENCIA,
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO,
	DATEPART(MAX(B.DTGRAVACAOSCORE)) AS DTGRAVACAOSCORE FORMAT DDMMYY10.
FROM
	PERF_MASSIFICADOS_&MES_REF._2 A LEFT JOIN BASE_SCORE B
	ON
	A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
	AND
	A.DT_ULTIMORECEBIMENTO >= DATEPART(B.DTGRAVACAOSCORE)
GROUP BY
	A.DT_REFERENCIA,
	A.NUCONTRATOGESTAO,
	A.SQCONTRATOFINANCEIRO;
