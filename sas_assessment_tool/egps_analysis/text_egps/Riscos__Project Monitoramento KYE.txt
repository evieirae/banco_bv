PROC SQL;
   CREATE TABLE BS_KYE_MONITORAMENTO AS
   SELECT DISTINCT
          t1.CPF,
		  (CASE WHEN T3.CARGO IS NOT MISSING THEN T3.CARGO ELSE CATX('','Terceiros','-','Empresa','-',t6.Empresa) end) FORMAT $2000. AS CARGO,
		  T3.'Admissão'n  AS DtAdmissao,
		  (CASE WHEN t1.'Status da consulta'n = 'APROVADO' THEN  'NAO' ELSE 'SIM' END) AS FlProcessos,
		  (CASE WHEN T2.DOC_NUM IS MISSING THEN 'NAO' ELSE 'SIM' end) AS FlMidia,
          (CASE WHEN T4.DOC_NUM IS MISSING THEN 'NAO' ELSE 'SIM' end) AS FlPep,
		  (CASE WHEN T5.DOC_NUM_QUADRO IS MISSING THEN 'NAO' ELSE 'SIM' end) AS FLQSA

      FROM BS_MONITORAMENTO_KYE t1
	  LEFT JOIN PLD.BS_MIDIA_AML t2 ON (T1.CPF = T2.DOC_NUM AND T2.DOC_NUM > 0)
	  LEFT JOIN BS_COLABORADORES_MONITORAMENTO t3 ON (T1.CPF =INPUT(compress(T3.CPF ,'.-'),BEST14.))
      LEFT JOIN PLD.BS_PEP_FULL t4 ON (T1.CPF = T4.DOC_NUM)
      LEFT JOIN BS_QUADROSOCIETARIO t5 ON (T1.CPF = T5.DOC_NUM_QUADRO)
      LEFT JOIN BASE_TERCEIROS t6 ON (T1.CPF = INPUT(T6.CPF,BEST14.) AND t6.'CATEGORIA TRATADA'n NOT = '0-NENHUM');

QUIT;
