PROC SQL;
   CREATE TABLE Mercado AS
   SELECT
      t1.NuAnoMes,
      t1.StVeiculo,
      t1.FxIdade,
      CASE
         WHEN t1.TpVeiculo = 'Leves' THEN 'EGV Leves'
         WHEN t1.TpVeiculo = 'Pesados' THEN 'EGV Pesados'
         ELSE ''
      END AS produto,
      SUM(t1.GravBV) AS GravBV,
      SUM(t1.GravTotal) AS GravTotal
   FROM BASES.TBRESULTADOMARKETSHARE t1
   WHERE t1.NuAnoMes BETWEEN 202506 AND 202506
     AND t1.TpArquivo = 'mf'
     AND t1.TpVeiculo IN ('Leves', 'Pesados')
     AND t1.flcvg = 'S'
   GROUP BY t1.NuAnoMes, t1.StVeiculo, t1.FxIdade, t1.TpVeiculo;

PROC SQL;
   CREATE TABLE TBPRODUCAO AS
   SELECT

          t1.nuanomes,
          case when t1.NmGrupoProdutoMeta = 'Leves-BVO' then 'Leves' else t1.NmGrupoProdutoMeta end as NmGrupoProdutoMeta,
          t1.DsTipoCanalProducao,
		  t1.NmSubEstruturaComercialC,
		  t1.NmCanalEstruturaComercial,
		  t1.NmAreaEstruturaComercialC,

          case when t1.fxidade = 'ZeroKM' then 'Novos' else 'Usados' end as Estado_bem,

		  case when t1.NmGrupoProdutoMeta in ('Leves', 'Motos', 'Pesados') then 'Rodas' when t1.NmGrupoProdutoMeta = 'Solar' then 'Solar' else 'Empréstimos' end as Grupo_Produto,

            (COUNT(t1.NuContrato)) AS QtdeContrato,
            (SUM(t1.VrProducao)) AS VrProducao,
            (SUM(t1.VrProducaoProjetado)) AS VrProducaoProjetado

      FROM bases.TBPRODUCAO t1

	  where t1.NuAnoMes BETWEEN 202501 AND 202504

	  and t1.NmGrupoProdutoMeta in ('Leves', 'Motos', 'Pesados', 'Leves-BVO', 'CVG Leves', 'CVG Pesados')
      GROUP BY
	  1,2,3,4,5,6,7,8;
