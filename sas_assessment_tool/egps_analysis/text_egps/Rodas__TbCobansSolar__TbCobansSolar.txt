%criavardb;
proc sql;
%CONECTADB (server=CTR);
CREATE TABLE TbCobansSolar_Aux AS
SELECT * FROM connection to dbcon
(
SELECT
         ParceiroComercial.dtcadastro                    AS DtCadastroCoban
        ,ParceiroComercial.NmFantasia                    AS RazaoSocialParceiro
		,Pessoa.NuCpfCnpj                                AS CNPJParceiro
        ,ParceiroComercial.CdParceiroComercial           AS Login
		,ParceiroComercial.FlAtivo                       AS StatusLogin
        ,TipoParceiroComercial.NmTipoParceiroComercial   AS TipoParceiro
		,AtividadeFinanciamento.DsAtividadeFinanciamento AS NmAtividade
        ,ParceiroSubProduto.CdSubProduto                 AS CdProduto
        ,SubProduto.DsSubProduto                         AS NmProduto
        ,FavorecidoAutorizado.dtcadastro                 AS DtCadastroFavorecido
        ,FavorecidoAutorizado.NmTitular                  AS RazaoSocialFavorecido
		,FavorecidoAutorizado.NuCpfCnpj                  AS CNPJFavorecido
        ,FavorecidoContaCorrente.NuBanco                 AS NroBancoFavorecido
		,FavorecidoContaCorrente.CdAgencia               AS NroAgenciaFavorecido
        ,FavorecidoContaCorrente.NuContaCorrente         AS NroContaFavorecido
		,FavorecidoAutorizado.FlAtivo                    AS StatusFavorecido
        ,FavorecidoContaCorrente.FlPrincipal             AS Principal
		,FavorecidoContaCorrente.FlContaServico          AS Servico
		,FavorecidoContaCorrente.FlComissao              AS Comissao
        ,FavorecidoContaCorrente.FlContrato              AS Contrato
		,FavorecidoContaCorrente.FlAtivoComissao         AS AtivoComissao
		,FavorecidoContaCorrente.FlAtivoContrato         AS AtivoContrato
        ,ColaboradorN1.NmColaborador                     AS GerenteComercial
        ,ColaboradorN2.NmColaborador                     AS GerenteRegional
        ,RepresentanteLegal.NmRepresentanteLegal         AS NmSocio
        ,RepresentanteLegal.NuCpf                        AS CPFSocio

FROM DBCOR..TbParceiroComercial ParceiroComercial

	LEFT  JOIN DBCOR..TbRegiao Regiao
    	ON Regiao.CdRegiao = ParceiroComercial.CdRegiao

	LEFT  JOIN DBCOR..TbFilial Filial
    	ON Regiao.CdFilial = Filial.CdFilial

	LEFT  JOIN DBCOR..TbColaborador ColaboradorN1
		ON ColaboradorN1.CdColaborador = Filial.CdGerenteComercial

    LEFT  JOIN DBCOR..TbColaborador ColaboradorN2
    	ON ColaboradorN2.CdColaborador = Filial.CdGerenteRegional

    LEFT  JOIN DBCOR..TbRepresentanteLegal RepresentanteLegal
		ON RepresentanteLegal.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

    LEFT  JOIN DBCOR..TbSocio Socio
    	ON Socio.CdParceiroComercial = ParceiroComercial.CdParceiroComercial
			AND Socio .SqRepresentanteLegal = RepresentanteLegal.SqRepresentanteLegal

    LEFT  JOIN DBCOR..TbPessoa Pessoa
    	ON ParceiroComercial.CdPessoa = Pessoa.CdPessoa

    LEFT  JOIN DBCOR..TbTipoParceiroComercial TipoParceiroComercial
    	ON TipoParceiroComercial.CdTipoParceiroComercial = ParceiroComercial.CdTipoParceiroComercial

    LEFT JOIN DBCOR..TbAtividFinancParcComercial AtividFinancParcComercial
    	ON AtividFinancParcComercial.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

    LEFT JOIN DBCOR..TbAtividadeFinanciamento AtividadeFinanciamento
    	ON AtividFinancParcComercial.CdAtividadeFinanciamento = AtividadeFinanciamento.CdAtividadeFinanciamento

    LEFT JOIN DBCOR..TbFavorecidoAutorizado FavorecidoAutorizado
    	ON FavorecidoAutorizado.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

    LEFT JOIN DBCOR..TbFavorecidoContaCorrente FavorecidoContaCorrente
    	ON FavorecidoContaCorrente.CdParceiroComercial = FavorecidoAutorizado.CdParceiroComercial

    LEFT JOIN DBCOR.dbo.TbParceiroSubProduto ParceiroSubProduto
    ON ParceiroSubProduto.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

    LEFT JOIN DBCOR.dbo.TbSubProduto SubProduto
    	ON SubProduto.CdSubProduto = ParceiroSubProduto.CdSubProduto

WHERE  ParceiroComercial.FlAtivo = 'S'

	AND ParceiroComercial.flconsignadoconvenio = 'N'
	AND ParceiroComercial.CdTipoParceiroComercial in (5)
	AND ParceiroComercial.FlParceiroVirtual = 'N'
	AND AtividadeFinanciamento.cdatividadefinanciamento in (7)
	AND FavorecidoContaCorrente.SqFavorecido = FavorecidoAutorizado.SqFavorecido
	AND ParceiroSubProduto.FlAtivo = 's'
	AND ParceiroSubProduto.CdSubProduto = 70

ORDER BY ParceiroComercial.dtcadastro
)

;run;

proc sql;
CREATE TABLE TbCobansSolar as
SELECT
		 DATEPART (DtCadastroCoban) as DtCadastroCoban format date9.
        ,RazaoSocialParceiro
		,CNPJParceiro
        ,Login
		,StatusLogin
        ,TipoParceiro
		,NmAtividade
        ,CdProduto
        ,NmProduto
        ,DATEPART (DtCadastroFavorecido) as DtCadastroFavorecido format date9.
        ,RazaoSocialFavorecido
		,CNPJFavorecido
        ,NroBancoFavorecido
		,NroAgenciaFavorecido
        ,NroContaFavorecido
		,StatusFavorecido
        ,Principal
		,Servico
		,Comissao
        ,Contrato
		,AtivoComissao
		,AtivoContrato
        ,GerenteComercial
        ,GerenteRegional
        ,NmSocio
        ,CPFSocio
FROM TbCobansSolar_Aux
;run;

proc export
data= TbCobansSolar
file='/riscosbvf/BA/processos/output/TbCobansSolar.csv' replace
dbms=dlm;
delimiter=';';
run;
