CREATE TABLE work.tblProcessoGarantiaSintProjuris  AS
SELECT

 CodCausa as CodCausa
,SerialImport as SerialImport
,PASTA as PASTA
,FlBaseOrigem as FlBaseOrigem
,Natureza as Natureza
,NaturezaUnidade as NaturezaUnidade
,StatusCausa as StatusCausa
,Carteira as Carteira
,StatusGarantia as StatusGarantia
,Garantias as Garantias
,ValorGarantia as ValorGarantia
,CodGarantia as CodGarantia
,QtGarantia as QtGarantia
,ValorAtualizado as ValorAtualizado
,DataInclusao as DataInclusao
,DataCadastro as DataCadastro
,DataLiberacao as DataLiberacao
,AdvogadoPrincipalDoCliente as AdvogadoPrincipalDoCliente
,TipoAdvogadoPrincipalDoCliente as TipoAdvogadoPrincipalDoCliente
,LiberadoTotal as LiberadoTotal
,LiberadoPerda as LiberadoPerda
,LiberadoRecuperacao as LiberadoRecuperacao
,LiberadoCustas as LiberadoCustas
,pasta as pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdCausaUnidade as CdCausaUnidade
,CdSistema as CdSistema
,CdUnidadeExyon as CdUnidadeExyon

FROM CONNECTION TO postgres(

 select
	case
 		when pr.serial_import like ('%CTO%') then (left(pr.SERIAL_IMPORT,10))
 		when pr.serial_import like ('%INQ%') then (left(pr.SERIAL_IMPORT,10))
    	when pr.serial_import like ('%BVT%') then (left(pr.SERIAL_IMPORT,10))
		when pr.serial_import like ('%TRI%') then (left(pr.SERIAL_IMPORT,10))
		when pr.serial_import like ('%CON%') then (left(pr.SERIAL_IMPORT,10))
		when pr.serial_import like ('%TRB%') then (left(pr.SERIAL_IMPORT,10))
		when pr.serial_import like ('%VAC%') then (left(pr.SERIAL_IMPORT,10))
			else pr.serial_import end       as CodCausa
	,pr.SERIAL_IMPORT                       as SerialImport
	,pr.PASTA
	,'Projuris'                             as FlBaseOrigem
	,na.Natureza
	,mu.nome                                as NaturezaUnidade
    ,case when pr.Data_Encerramento is null then 'EM CURSO' else 'ENCERRADO' end as StatusCausa
    ,cc.CARTEIRA_CUSTOM                     as Carteira
    ,case when gc.Status_Garantia IN ('N?O LIBERADA' ,'NÃO LIBERADA')  then 'NAO LIBERADA' else gc.Status_Garantia end as StatusGarantia
	,gc.GARANTIA                            as Garantias
 	,gc.VALOR_GARANTIA                      as ValorGarantia
    ,cast(gc.ID_PROCESSO_GARANTIA as varchar)                as CodGarantia
	,case when gc.VALOR_GARANTIA is not null then 1 else 0 end as QtGarantia
	,gc.VALOR_GARANTIA                      as ValorAtualizado
    ,cast(pg.data_inclusao as DATE)         as DataInclusao
    ,cast(pg.data_cadastro as DATE)         as DataCadastro
    ,cast(pl.data_liberacao as DATE)        as DataLiberacao
    ,ap.nome                                as AdvogadoPrincipalDoCliente
    ,case
      when (select ue.NOME from MM_PROCESSO_ADVOGADO ae inner join M_USUARIO ue on ue.ID_USUARIO = ae.ID_USUARIO and ue.LOGIN like 'escritorio.%'
  		where ae.ID_PROCESSO =pr.ID_PROCESSO and ae.ID_USUARIO = pr.ID_ADVOGADO_PRINCIPAL limit 1) = ap.nome then 'ASSESSORIA'
  	  when (select up.NOME from MM_PROCESSO_ADVOGADO ap inner join M_USUARIO up on up.ID_USUARIO = ap.ID_USUARIO and up.LOGIN like 'perito.%'
  		where ap.ID_PROCESSO = pr.ID_PROCESSO limit 1) = ap.nome then 'PERITO'
  	  when (selecT ui.NOME from MM_PROCESSO_ADVOGADO ai inner join M_USUARIO ui on UI.ID_USUARIO = ai.ID_USUARIO and ui.LOGIN NOT like 'escritorio.%' and ui.LOGIN not like 'perito.%'
  		where ai.ID_PROCESSO = pr.ID_PROCESSO limit 1) = ap.nome then 'INTERNO'
  			end as TipoAdvogadoPrincipalDoCliente

  	,pgl.LiberadoTotal
	,pgl.LiberadoPerda
	,pgl.LiberadoRecuperacao
	,pgl.LiberadoCustas
    ,pr.pasta
    ,pr.ID_NATUREZA 										as IdNatureza
    ,pr.NATUREZA_UNIDADE_CUSTOM 							as IdNaturezaUnidadeCustom
    ,null::numeric	 										as CdCausaUnidade
    ,'p' 													as CdSistema
    ,case
		when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 2             then 'TRI'
		when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 1             then 'TBR'
		when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 4             then 'CTO'
		when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 12            then 'CON'
		when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 13            then 'VAC'
		when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM in (1 ,7, 11)   then 'INQ'
		when pr.ID_NATUREZA = 27 and pr.NATUREZA_UNIDADE_CUSTOM = 5             then 'BVT'
		when pr.ID_NATUREZA = 32 and pr.NATUREZA_UNIDADE_CUSTOM in (7, 11 , 12) then 'INQ'
			else null end 									as CdUnidadeExyon

from public.M_PROCESSO pr
left join  public.VW_GARANTIAS_COMPLETA    gc on gc.ID_PROCESSO               = pr.ID_PROCESSO  and gc.REVERTIDA <> 'T'
inner join public.LT_NATUREZA              na on na.ID_NATUREZA               = pr.ID_NATUREZA
inner join public.M_UNIDADE_ORGANIZACIONAL mu on mu.ID_UNIDADE_ORGANIZACIONAL = pr.ID_UNIDADE_ORGANIZACIONAL
left join  public.LT_CARTEIRA_CUSTOM       cc on cc.ID_CARTEIRA_CUSTOM        = pr.ID_CARTEIRA_CUSTOM
left join  public.M_USUARIO			       ap on ap.ID_USUARIO                = pr.ID_ADVOGADO_PRINCIPAL
left join (select distinct pasta, status_processo from public.VW_OBJETOS_ATUALIZADOS) oa on oa.PASTA = pr.PASTA
left join (select distinct id_processo_garantia,data_inclusao,data_cadastro from public.D_PROCESSO_GARANTIA) pg on pg.id_processo_garantia = gc.id_processo_garantia
left join (select distinct id_processo_garantia,max(data_liberacao) as data_liberacao from public.D_PROCESSO_GARANTIA_LIBERACAO group by id_processo_garantia) pl on pl.id_processo_garantia = gc.id_processo_garantia
left join (

select
 IdProcessoGarantia
,sum(LiberadoTotal)       as LiberadoTotal
,sum(LiberadoPerda)       as LiberadoPerda
,sum(LiberadoRecuperacao) as LiberadoRecuperacao
,sum(LiberadoCustas)      as LiberadoCustas

from (
	select
	 	 pgl.ID_PROCESSO_GARANTIA                                                     as IdProcessoGarantia
		,pgl.valor_resgatado                                                          as LiberadoTotal
		,case when pgl.CLASSIFICACAO_CUSTOM = 'P' then pgl.VALOR_RESGATADO else 0 end as LiberadoPerda
		,case when pgl.CLASSIFICACAO_CUSTOM = 'R' then pgl.VALOR_RESGATADO else 0 end as LiberadoRecuperacao
		,case when pgl.CLASSIFICACAO_CUSTOM = 'C' then pgl.VALOR_RESGATADO else 0 end as LiberadoCustas
	from public.M_PROCESSO P
	inner join public.D_PROCESSO_GARANTIA PG on PG.ID_PROCESSO = P.ID_PROCESSO
	inner join public.D_PROCESSO_GARANTIA_LIBERACAO PGL on PGL.ID_PROCESSO_GARANTIA = PG.ID_PROCESSO_GARANTIA

	where PGL.REVERTIDA <> 'T'

	) as x group by IdProcessoGarantia
) pgl on pgl.IdProcessoGarantia = gc.ID_PROCESSO_GARANTIA

	where 1=1

	and (gc.GARANTIA like ('%Dep%') or gc.GARANTIA like ('%DEP%') or gc.GARANTIA like ('%Trans%') or gc.GARANTIA like ('%TRANS%'))
	

;

);DISCONNECT FROM postgres;
RUN;
