PROC SQL;
CREATE TABLE bases.TbInad AS
	SELECT Distinct
		PrimeiraParcela.NuProposta
		,PrimeiraParcela.DtEntradaProposta
		,PrimeiraParcela.NuContratoGestao AS NuContrato
		,PrimeiraParcela.DtCancelamento
		,PrimeiraParcela.DtVencimento format date9.
		,PrimeiraParcela.DtContrato
		,PrimeiraParcela.DtUltimoRecebimento AS DtPagamento

		,CASE WHEN PrimeiraParcela.StParcela = 'A' THEN 'Aberta'
			  WHEN PrimeiraParcela.StParcela = 'Q' THEN 'Quitada'
			  ELSE PrimeiraParcela.StParcela
		 END AS StParcela

		,Originacao.NmGrupoProdutoMeta
		,Originacao.FxIdade
		,Originacao.FxEntrada
		,Originacao.QtParcelas
		,Originacao.DsTipoCanalProducao
		,Originacao.CdColaborador
		,Originacao.NmColaborador
		,Originacao.NmGrupoCargoColaborador

		,Originacao.CdFilial
		,Originacao.NmSubEstruturaComercialA
		,Originacao.NmAreaEstruturaComercial
		,Originacao.NmCanalEstruturaComercial
		,Originacao.DsLoginN1
		,Originacao.DsLoginN2A
		,Originacao.DsLoginN3A
		,Originacao.LstSubstituicao
		,Originacao.NmParceiroComercial
		,Originacao.NmTipoParceiroComercial
		,Originacao.CdParceiroComercial
		,Originacao.NuCnpjLojaFavorecido
		,Originacao.NmParceiroComercialFavorecido

		,Originacao.NmPessoa
		,Originacao.TpPessoa
		,Originacao.NuCpfCnpj
		,Originacao.DsLoginOriginador
		,CanalOriginador.FiltroCVG as CanalOrigem
		,Telefone.NuTelefone1
		,Telefone.NuTelefone2
		,Case
			When Coalesce(Originacao.NuRating,0) < 5 Then 'RATING < 5'
			When Coalesce(Originacao.NuRating,0) < 6 Then 'RATING 5-6'
			When Coalesce(Originacao.NuRating,0) < 8 Then 'RATING 7-8'
			When Coalesce(Originacao.NuRating,0) < 9 Then 'RATING 8-9'
			When Coalesce(Originacao.NuRating,0) <= 10 Then 'RATING 9-10'
			Else 'Verificar'
		End as FxRating
		,Originacao.FxScoreFim
		,TbContrato.DsPlaca
		,PrimeiraParcela.Data_Base AS DtAtualizacao
		,Originacao.VrLiberado format 18.2
		,Originacao.VrFinanciamento format 18.2
		,PrimeiraParcela.VrParcela format 18.2
		,PrimeiraParcela.qtd_cto_elegiveis_inad7 AS FlElegivelInad7
		,PrimeiraParcela.Val_prt_7_1pmt AS VrParcelaElegivelInad7 format 18.2
		,PrimeiraParcela.Inad_7_1pmt AS VrParcelaInad7  format 18.2
		,PrimeiraParcela.qtd_cto_inad7  AS FlInad7
		,PrimeiraParcela.qtd_cto_elegiveis_inad15 AS FlElegivelInad15
		,PrimeiraParcela.Val_prt_15_1pmt AS VrParcelaElegivelInad15 format 18.2
		,PrimeiraParcela.Inad_15_1pmt AS VrParcelaInad15  format 18.2
		,PrimeiraParcela.qtd_cto_inad15  AS FlInad15
		,PrimeiraParcela.qtd_cto_elegiveis_inad30 AS FlElegivelInad30 format 18.2
		,PrimeiraParcela.Val_prt_30_1pmt AS VrParcelaElegivelInad30 format 18.2
		,PrimeiraParcela.Inad_30_1pmt AS VrParcelaInad30 format 18.2
		,PrimeiraParcela.qtd_cto_inad30 AS FlInad30
		,PrimeiraParcela.qtd_cto_elegiveis_inad60 as FlElegivelInad60
		,PrimeiraParcela.Val_prt_60_1pmt as VrParcelaElegivelInad60 format 18.2
		,PrimeiraParcela.Inad_60_1pmt as VrParcelaInad60 format 18.2
		,PrimeiraParcela.qtd_cto_inad60 as FlInad60
		,PrimeiraParcela.qtd_cto_elegiveis_inad90 as FlElegivelInad90
		,PrimeiraParcela.Val_prt_90_1pmt as VrParcelaElegivelInad90 format 18.2
		,PrimeiraParcela.Inad_90_1pmt as VrParcelaInad90 format 18.2
		,PrimeiraParcela.qtd_cto_inad90	as FlInad90

		,ever30.qtd_cto_elegiveis_inad30_3 as FlElegivelInad30_3
		,ever30.Val_prt_30_3pmt as VrParcelaElegivelInad30_3 format 18.2
		,ever30.Inad_30_3pmt as VrParcelaInad30_3 format 18.2

		,PrimeiraParcela.qtd_cto_elegiveis_inad7 as FlElegivelEver30_Mob3
		,Originacao.VrFinanciamento as VrParcelaElegivelEver30_Mob3 format 18.2
		,CASE WHEN ever30.fl_ever30=1 THEN Originacao.VrFinanciamento ELSE 0 END as VrParcelaEver30_Mob3 format 18.2
		,ever30.fl_ever30 as FlEver30_Mob3

	FROM TbInad PrimeiraParcela
		LEFT JOIN tmp.TbOriginacao Originacao
			ON PrimeiraParcela.NuProposta = Originacao.NuProposta

		LEFT JOIN TbTelefone Telefone
			ON PrimeiraParcela.NuCpfCnpj = Telefone.NuCpfCnpj

			LEft join TbInadP.TBINADTERCEIRAPARC ever30
				on PrimeiraParcela.NuProposta = TBINADTERCEIRAPARC.nuproposta

				Left Join BaseGRC.TbFiltroCVG as CanalOriginador
					on PrimeiraParcela.NuProposta = CanalOriginador.NuProposta

					Left Join Bases.TBEXTCONTRATO as TbContrato
						on PrimeiraParcela.NuProposta = TbContrato.NuProposta

	WHERE Originacao.NmGrupoProdutoMeta <> 'Privado'
	AND PrimeiraParcela.NuContratoGestao <> 12262000121859 /*Ajuste 7/ago/24, caso falecimento na 1pmt. Não é possível ajuste via Operações */

;RUN;

proc sql;
CREATE TABLE TbPreventivoInad AS
SELECT
	Inad.NuContrato format 18.
	,Inad.NmGrupoProdutoMeta
	,Inad.DtEntradaProposta
	,Inad.CdColaborador
	,Inad.NmColaborador
	,Inad.NmParceiroComercial
	,Inad.NmTipoParceiroComercial
	,Inad.NmParceiroComercialFavorecido
	,Inad.NuCpfCnpj
	,Inad.NmPessoa
	,Inad.TpPessoa
	,Inad.NuTelefone1
	,Inad.NuTelefone2
	,Inad.DtVencimento format ddmmyy10.
	,Inad.VrParcela  format 18.2
	,intck('day',Inad.DtVencimento,today()) as QtDiasAtras  format 10.
	,Inad.NmCanalEstruturaComercial
	,Inad.NmSubEstruturaComercialA
	,Inad.NmAreaEstruturaComercial
	,Inad.DsLoginN1
	,Inad.DsloginN2A
	,Inad.DsloginN3A
	,Inad.LstSubstituicao
	,ColaboradorFocal.LstColaboradorFocal
	,DsLoginOriginador
	,CanalOrigem
	,FxRating
	,FxScoreFim
	,DsPlaca
FROM bases.TbInad Inad
	LEFT JOIN basegrc.TbColaboradorFocal ColaboradorFocal
		ON  Inad.CdFilial = ColaboradorFocal.CdFilial
		AND Inad.NmGrupoProdutoMeta = ColaboradorFocal.NmGrupoProdutoMeta
WHERE today() > DtVencimento
	  AND intck('day',DtVencimento,today()) <= 30
	  AND DtPagamento IS NULL
	  AND DtCancelamento IS NULL
;run;

Proc Sql;
Create Table TbInadParcela (Where = (Filtro <=200)) as
Select DISTINCT
		TbParcela.NuContrato
		,TbProducao.NmGrupoProdutoMeta
		,TbProducao.DtContrato
		,TbProducao.DtPrimeiroVencimento
		,TbProducao.DtCancelamento
		,TbProducao.DtLiquidacaoEfetiva
		,Case
			When Coalesce(TbProducao.NuRating,0) < 5 Then 'RATING < 5'
			When Coalesce(TbProducao.NuRating,0) < 6 Then 'RATING 5-6'
			When Coalesce(TbProducao.NuRating,0) < 8 Then 'RATING 7-8'
			When Coalesce(TbProducao.NuRating,0) < 9 Then 'RATING 8-9'
			When Coalesce(TbProducao.NuRating,0) <= 10 Then 'RATING 9-10'
			Else 'Verificar'
		End as FxRating
		,TbProducao.FxScoreFim
		,TbContrato.DsPlaca
		,TbParcela.SqContratoFinanceiro
		,TbParcela.NuParcela
		,TbParcela.VrParcela
		,TbParcela.VrUltRecebimento
		,TbParcela.StParcela
		,TbParcela.DtVencimento AS DtVencimento
		,TbParcela.DtUltimoRecebimento AS DtUltimoRecebimento
		,TbParcela.DtVencimento - TbProducao.DtContrato as Carencia
		,intck('day',TbParcela.DtVencimento,today()) as QtDiasAtraso  format 10.
		,TbParcela.DtVencimento - TbProducao.DtPrimeiroVencimento as Filtro
From TbInadParcela2 as TbParcela
	 INNER JOIN bases.TbProducao TbProducao
			ON TbParcela.NuContrato = TbProducao.NuContrato
			Left join Bases.TbExtContrato TbContrato
				On TbParcela.NuContrato = TbContrato.NuContrato
Where TbParcela.StParcela = 'A' and TbProducao.DtContrato <> . and TbProducao.DtCancelamento = . and TbProducao.DtLiquidacaoEfetiva = .
	  AND today() > DtVencimento AND intck('day',DtVencimento,today()) <= 30
;Quit;
