PROC SQL;
   CREATE TABLE PAGAMENTOS_RF1  AS
   SELECT t1.cd_causa as Flag,
          sum(t1.VlLcto) as Valor_Devolucao

      FROM DEVOLUCAO_CHECK t1
      group by 1
;

QUIT;

PROC SQL;
   CREATE TABLE PAGAMENTOS_RF3  AS
   SELECT cd_causa,
		  SAFRA_ENCERRAMENTO,
		  TIPO_PGTO,
		  motivo_rro,
		  ASSESSORIA,
          Valor_Lancamento,
          count(cd_causa) as Qtdade,
		  Flag,
		  case when Valor_Devolucao is missing then 0 else Valor_Devolucao end as Valor_Devolucao

      FROM PAGAMENTOS_RF t1 left join PAGAMENTOS_RF1 t2 on cd_causa = flag
      group by 1,2,3,4,5,6
;

QUIT;

PROC SQL;
   CREATE TABLE PAGAMENTOS_RF4  AS
   SELECT cd_causa,
		  SAFRA_ENCERRAMENTO,
		  TIPO_PGTO,
		  motivo_rro,
		  ASSESSORIA,
		  case when Valor_Devolucao + Valor_Lancamento >= 0 and flag is not missing then 1 else 0 end as Flag2,
          Qtdade,
		  Valor_Devolucao + Valor_Lancamento as Valor_Lancamento

      FROM PAGAMENTOS_RF3 t1
;

QUIT;

PROC SQL;
   CREATE TABLE bases.PAGAMENTOS_RF  AS
   SELECT SAFRA_ENCERRAMENTO as AnoMes,
		  TIPO_PGTO,
			motivo_rro,
			ASSESSORIA,
          sum(Valor_Lancamento) as Valor_Lancamento,
          sum(Qtdade) as Qtdade_2,
		  count(t1.cd_causa) as Qtdade

      FROM PAGAMENTOS_RF4 t1
	  where AnoMes >= 201601 and Flag2 <> 1
      group by 1,2,3,4
;

QUIT;
