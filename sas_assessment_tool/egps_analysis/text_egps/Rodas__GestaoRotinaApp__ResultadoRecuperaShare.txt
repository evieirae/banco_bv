%include '/infogrc/utils/all-includes.sas';

%VerificaAtualizacao(/infogrc/convsimu/tbsimulacoes.sas7bdat,100,150);

/* Cria TbSimulacoes_V01 com a coluna Prioridade baseada em StatusSimulacao e SistemaInicial */
data TbSimulacoes_V01
     (keep=NuAnoMes StatusSimulacao Mensageria CdparceiroComercial );
  set convsimu.TbSimulacoes (where=(NuAnoMes >= &safraM0.
									AND Produto IN('Leves','CVG Leves')
									AND Canal IN ('MULTIMARCAS','CONCESSIONARIAS')
/*									and cdparceirocomercial =1018*/
									AND FlSimuValidaPDV=1));

run;

proc sql;
create table TbSimulacoes_V02 as
select
a.NuAnoMes
,a.StatusSimulacao
,a.CdparceiroComercial
,cats(Pc.CdParceiroComercial,'-',Pc.NmParceiroComercial) as NmParceiroComercial
,case when a.StatusSimulacao= ('Proposta APNP') then 1
      when a.StatusSimulacao='Card sem proposta originada' and a.Mensageria='0. Aprovado' then 2
      else 3
	  end as prioridade
,count(1) as Qtd
from TbSimulacoes_V01 as a
		left join tmp.TbParceiroComercial as Pc
		on a.CdParceiroComercial = Pc.CdParceiroComercial

where a.StatusSimulacao not in ('Proposta paga','Proposta negada')
group by 1,2,3,4,5
order by 5
;run;

/*Retira duplicidade considerando apenas um status por PDV*/
proc sort data=TbSimulacoes_V02 out=TbSimulacoes_V02 nodupkey;
  by NuAnoMes CdparceiroComercial ;
run;

proc sql ;
create table TbShareMissaoConfirmado as
select
	*
	from basegrc.TBSHAREMISSAO
	where 1=1
	and flCupom<>'GRUPO CONT'
	and nuanomes >= &safraM0.
/*	and NmParceiroComercial='1018-ASTRA VEICULOS'*/
;
quit;

PROC SQL;
CREATE TABLE TbShareMissaoPush AS
  SELECT distinct
	 sht.NuAnoMes
	,Sht.NuSemana
	,sht.NmParceiroComercial
/*	,sht.CdParceiroComercial	*/
	,sht.DsLoginN1C
	,sht.DsLoginN2C
	,sht.DsLoginN3C
/*	,simu.StatusSimulacao*/
	,CAse when simu.prioridade = 1  then 'Simulou, originou uma proposta aprovada'
	      when simu.prioridade = 2  then 'Simulou, e nao originou/aprovou'
		  when simu.prioridade = 3  then 'Nao simulou'
		  else 'Nao simulou'
	end as StatusSimulacao
	,CASE weekday(Today())   /*Monta o push pra cada dia da semana*/
        WHEN 4 THEN 'notificacao de quarta'
        WHEN 5 THEN 'notificacao de quinta'
        WHEN 6 THEN 'notificacao de sexta'
        ELSE 'Mensagem padrao'
    		END AS Push
	,sum(simu.Qtd) as Qtd
FROM   TbShareMissaoConfirmado Sht

  	left join TbSimulacoes_V02 as Simu
		on sht.cdparceiroComercial = Simu.cdparceiroComercial
		and sht.nuanomes = Simu.nuanomes

/* WHERE  Sht.NmParceiroComercial='1018-ASTRA VEICULOS'*/
group by 1,2,3,4,5,6,7,8
;run;

/*EXporta CSV */
proc export
data=TbShareMissaoPush
file='/riscosbvf/BA/processos/output/TbMissaoRecSharePush.csv' replace
dbms=dlm;
delimiter=';';
run;
