CREATE TABLE WORK.TB_BQ AS
SELECT *
FROM CONNECTION TO PERM (

   SELECT
        b.cd_contrato AS NUCONTRATO,
        b.sq_contrato_financeiro_ativo AS SQCONTRATOFINANCEIROATIVO,
        b.cd_cpf_cnpj AS NUCPFCNPJ,
        b.cd_pessoa AS CDPESSOA,
        b.dt_liquidacao_efetiva AS DTLIQUIDACAOEFETIVA,
        b.cd_situacao_contrato AS CDSITUACAOCONTRATO,
        b.cd_produto AS CDPRODUTO,
        b.cd_modalidade_produto AS CDMODALIDADEPRODUTO,
        d.cdsubproduto AS CDSUBPRODUTO,
        h.cdgruposubproduto AS CDGRUPOSUBPRODUTO
FROM bv-pdbd-prd.dvry_core.tbcontrato AS B
LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS D ON b.cd_contrato = d.nucontratogestao
AND d.stproposta='AP'
LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbmodalidadeproduto AS E ON CAST(b.cd_produto AS INT64) = e.cdproduto
AND CAST(b.cd_modalidade_produto AS INT64) = e.cdmodalidadeproduto
LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbsubproduto AS F ON CAST(b.cd_produto AS INT64) = f.cdproduto
AND CAST(b.cd_modalidade_produto AS INT64) = e.cdmodalidadeproduto
AND d.cdsubproduto = f.cdsubproduto
LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS H ON d.cdsubproduto = h.cdsubproduto
LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbgruposubproduto AS I ON h.cdgruposubproduto = i.cdgruposubproduto

);
DISCONNECT FROM PERM;
