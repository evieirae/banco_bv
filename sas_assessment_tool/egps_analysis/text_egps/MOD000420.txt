PROC SQL;
CONNECT TO sasiosiq as dbcon
(HOST= "IQPRDBI" server=IQPRDBI db=IQPRDBI user=&BI_USERNAME password=&BI_PASSWORD);
CREATE TABLE TbCpfsIIR AS SELECT * FROM connection to dbcon(

/*DEFINE DATA REFERÊNCIA*/
DECLARE @DtRef char(8) = &DATA_REFERENCIA.

/*CRIA TbTmp0, c/ apenas última versão da sequência do contrato
não filtra; Ou seja, não tem apenas LEVES*/
SELECT
	a.CdContrato,
	Max(a.CdSequenciaContratoFinanceiro) AS MaxSq
INTO #TbTmp0ContratoUltVersao
FROM DWBI_DTWH.TbContratoFinanceiro AS a
GROUP BY CdContrato;

/*CRIA TbTmp1 Usa a última versão da sequência do contrato
para extrair colunas TbContratoFinanceiro, filtrar duplicações*/
SELECT
	CF.CdContrato,
	CF.VrFinanciamento,
	CF.CdSequenciaContratoFinanceiro,
	/*s DtInicioVigencia; Qro dt q pessoa assumiu, ñ dt q seq muda; */
	CF.DtFimVigencia
INTO #TbTmp1ContratoUlt
FROM #TbTmp0ContratoUltVersao AS a
	INNER JOIN DWBI_DTWH.TbContratoFinanceiro AS CF
	/*INNER porque é TbContFin c/ ela mesma*/
	ON
		a.CdContrato=CF.CdContrato
		AND a.MaxSq=CF.CdSequenciaContratoFinanceiro
	LEFT JOIN DWBI_DTWH.TbContrato AS C ON CF.cdcontrato = C.cdcontrato
	/*COUNT(cdcont) on INNER = COUNT(cdcont) on LEFT)*/
WHERE
	C.cdprodutoorigem IN (12, 14);

/*Cria TbTmp1.1, filtra deixando última versão de cada contrato*/
SELECT
	ctpess.cdcontrato,
	max(ctpess.dtiniciorelacionamento) AS MaxDtInicio
INTO #TbTmpUltimaVersaoContrato
FROM
	DWBI_DTWH.TbContratoPessoa AS ctpess
WHERE
	cdcontrato IN (
					SELECT
						CDCONTRATO
					from
						#TbTmp1ContratoUlt
					)
	/*Esse where parece LEFTx'1''' JOIN, sem o JOIN; Usa contratos da "esquerda" para filtrar,
					e pegar última DtInicioRel*/
	AND ctpess.CdTipoPapelPessoa = 7
	AND ctpess.dtiniciorelacionamento < @DtRef
GROUP BY
	ctpess.cdcontrato;

/*Cria TbTmp1.2, com todas as colunas da TbContratoPessoa,
	apenas para última versão de cada contrato*/
SELECT
	b.CdContrato,
	b.CdPessoa,
	b.CdTipoPapelPessoa,
	A.MaxDtInicio
INTO #TbTmpContratoPessoa
FROM #TbTmpUltimaVersaoContrato AS A
INNER JOIN DWBI_DTWH.tbcontratopessoa AS B ON a.cdcontrato = b.cdcontrato AND A.MaxDtInicio = b.dtiniciorelacionamento
/*INNER porque é TbContPess c/ ela mesma*/
WHERE b.CdTipoPapelPessoa = 7;

/*--CRIA TbTmp2, adiciona CdPessoa da última versão do contrato,
para cada contrato na TbTmp1*/
SELECT
	a.*,
	CtPess.CdPessoa,
	CtPess.MaxDtInicio AS DtInicioContPessoa
INTO #TbTmp2UltLeftContPess
FROM
	#TbTmp1ContratoUlt 						AS a
	LEFT JOIN #TbTmpContratoPessoa 	AS CtPess ON a.CdContrato = CtPess.CdContrato
WHERE
	ctpess.CdTipoPapelPessoa = 7;

/*CRIA TbTmp3, adiciona NuCpfCnpj e DtNascimento,
	para cada CdPessoa na TbTmp2*/
SELECT
	a.*, /*9.922.356 CdPessoa distinct*/
	pess.nucpfcnpj, /*9.921.806 NuCpfCnpj distinct*/
	pessfis.dtnascimento /*9.747.484 PessFis, c DtNasc, distinct*/
INTO #TbTmp3ContPessLeftPessInnerPessFis
FROM
	#TbTmp2UltLeftContPess 						AS a
	LEFT JOIN DWBI_DTWH.tbpessoa 	AS pess ON a.cdpessoa = pess.cdpessoa
	INNER JOIN DWBI_DTWH.TbPessoaFisica AS pessfis ON pess.cdpessoa = pessfis.cdpessoa;
	/*INNER para filtrar apenas pessoas físicas; retira PJ*/

/*FILTRA, retirando pessoas vetadas (entraram para prejuízo, óbito, e algumas outras condições
	que impedem a pessoa de realizar financiamentos com o banco)*/
SELECT
	a.*,
	b.NuPonto
INTO #TbBaseContratos
FROM
	#TbTmp3ContPessLeftPessInnerPessFis AS a /*9.747.304 distinct cpf (c/ filtros)*/
	LEFT JOIN
	(
	SELECT
		CAST(nucpfcnpj AS NUMERIC(36,2)) AS nucpfcnpj,
		flregistroativo,
		nuponto
	FROM
		DWBI_DTWH.TbPessoaBehavior
	WHERE
		flregistroativo = 1
	) AS b
	ON a.NuCpfCnpj = b.NuCpfCnpj
WHERE b.nuponto <> 400;

/*Agrupamentos por CPF*/
SELECT
	a.NuCpfCnpj AS NuCPF,
	(DATEDIFF(hour,a.DtNascimento,@DtRef)/8766) AS NuIdade,
	count(DISTINCT a.CdContrato) AS QtContratos,
	min(a.DtInicioContPessoa) AS DtInicio1Contrato,
	max(a.DtInicioContPessoa) AS DtInicioUltimoContrato,
	max(a.DtFimVigencia) AS DtFimAtividade,
	avg(a.VrFinanciamento) AS MedTicketPorPessoa,
	sum(a.VrFinanciamento) AS SumTicketPorPessoa

INTO #TbCpfs
FROM #TbBaseContratos AS a

GROUP BY
	a.NuCpfCnpj,
	a.DtNascimento

ORDER BY a.NuCpfCnpj;

/*Segmentações por CPF*/
SELECT
	a.NuCPF,
	a.NuIdade,
	CASE
		WHEN a.NuIdade IS NULL THEN 'NULL'
		WHEN a.NuIdade < 18 THEN '<18' /*97 CPFs de menores*/
		WHEN a.NuIdade BETWEEN 18 AND 30 THEN '18-30'
		WHEN a.NuIdade BETWEEN 31 AND 40 THEN '31-40'
		WHEN a.NuIdade BETWEEN 41 AND 50 THEN '41-50'
		WHEN a.NuIdade BETWEEN 51 AND 60 THEN '51-60'
		WHEN a.NuIdade > 60 THEN '>60'
	END AS FxEtaria,
	a.QtContratos,
	CASE
		WHEN a.QtContratos = 1 THEN '1'
		WHEN a.QtContratos = 2 THEN '2'
		WHEN a.QtContratos = 3 THEN '3'
		WHEN a.QtContratos = 4 THEN '4'
		WHEN a.QtContratos BETWEEN 5 AND 10 THEN '5 - 10'
		WHEN a.QtContratos > 10 THEN '>10'
		WHEN a.QtContratos IS NULL THEN 'NULL'
		WHEN a.QtContratos = 0 THEN 'NULL'
	END AS FxContrato,
	CASE
		WHEN DATEDIFF(dd, a.DtInicio1Contrato, @DtRef) = 0 THEN a.QtContratos
		ELSE CAST(((a.QtContratos)*(365.25/(DATEDIFF(dd, a.DtInicio1Contrato, @DtRef)))) AS NUMERIC(36,5))
	END AS NuFreqContratoPorAno,
	a.DtInicio1Contrato,
	CASE
		WHEN a.DtInicio1Contrato IS NULL THEN 0000
		WHEN a.DtInicio1Contrato IS NOT NULL THEN YEAR(a.DtInicio1Contrato)
	END AS NuAno1Contrato,
	FLOOR((DATEDIFF(dd, a.DtInicio1Contrato, @DtRef))/365.25) AS NuIdadeAtividadeCliente,
	a.DtInicioUltimoContrato,
	CASE
		WHEN a.DtInicioUltimoContrato IS NULL THEN 0000
		WHEN a.DtInicioUltimoContrato IS NOT NULL THEN YEAR(a.DtInicioUltimoContrato)
	END AS NuAnoInicioUltimoContrato,
	FLOOR((DATEDIFF(dd, a.DtInicioUltimoContrato, @DtRef))/365.25) AS NuIdadeRecencia,
	a.DtFimAtividade,
	CASE
		WHEN a.DtFimAtividade >= @DtRef THEN 'Sim'
		WHEN a.DtFimAtividade IS NULL THEN 'NULL'
		WHEN a.DtFimAtividade < @DtRef THEN 'Nao'
	END AS FlAtivo,
	CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) AS MedTicketPorPessoa,
	CASE
		WHEN a.MedTicketPorPessoa IS NULL THEN 'NULL'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) < 40000 THEN '< R$40.000,00'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) BETWEEN 40000 AND 80000 THEN 'R$40.000,00 - R$80.000,00'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) BETWEEN 80000 AND 120000 THEN 'R$80.000,00 - R$120.000,00'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) BETWEEN 120000 AND 160000 THEN 'R$120.000,00 - R$160.000,00'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) BETWEEN 160000 AND 200000 THEN 'R$160.000,00 - R$200.000,00'
		WHEN CAST(round(a.MedTicketPorPessoa,2) AS NUMERIC(36,2)) > 200000 THEN '> R$200.000,00'
	END AS FxMedTicketPorPessoa,
	a.SumTicketPorPessoa
INTO #TbCpfsFaixas
FROM #TbCpfs AS a;

SELECT
	a.*,
	(
		SELECT
			avg(QtContratos)
		FROM #TbCpfsFaixas
	) AS MediaQtsContratos,
	(
		SELECT
			avg(MedTicketPorPessoa)
		FROM #TbCpfsFaixas
	) AS MediaTickets,
	(
		SELECT
			avg(NuIdadeRecencia)
		FROM #TbCpfsFaixas
	) AS MediaRecencias,
	(
		SELECT
			stddev(QtContratos)
		FROM #TbCpfsFaixas
	) AS StdDevQtsContratos,
	(
		SELECT
			stddev(MedTicketPorPessoa)
		FROM #TbCpfsFaixas
	) AS StdDevTickets,
	(
		SELECT
			stddev(NuIdadeRecencia)
		FROM #TbCpfsFaixas
	) AS StdDevRecencias,
	((QtContratos - MediaQtsContratos)/StdDevQtsContratos) AS PadronizacaoContratos,
	((MedTicketPorPessoa - MediaTickets)/StdDevTickets) AS PadronizacaoTicket,
	((MediaRecencias - NuIdadeRecencia)/StdDevRecencias) AS PadronizacaoRecencia,
	(PadronizacaoContratos + PadronizacaoTicket + PadronizacaoRecencia) AS iIR
FROM #TbCpfsFaixas AS a
);
RUN;
