CREATE TABLE RENEG_REALIZADA_&anomes. AS
SELECT DISTINCT
				A.NuContratoGestao
				,A.SqContratoFinanceiro
				,A.NuContratoGestaoNovo
				,A.sqcontratofinanceironovo
				,A.dt_baixa
				,A.dt_reneg
				,A.vr_presente
				,A.vr_financiado
				,A.qt_parcela
				,A.vr_prestacao
				,CASE
					WHEN B.Tx_Pactuada IS NOT NULL THEN B.Tx_Pactuada
					ELSE A.Tx_Pactuada
				END AS Tx_Pactuada
				,A.tx_contrato
				,B.vr_desconto_mora
				,B.vr_desconto_multa
				,B.vr_desconto_principal

					WHEN B.FlReestruturacao IS NOT NULL THEN B.FlReestruturacao
					ELSE A.FlRestruturacaoContrato
				END AS FlRestruturacaoContrato
				,B.TpEntrada

	FROM CONNECTION TO PERM(

		SELECT DISTINCT
			NuContratoGestao,
			MAX(SqContratoFinanceiro) 					AS SqContratoFinanceiro,
			NuContratoGestaoNovo,
			MAX(sqcontratofinanceironovo) 				AS sqcontratofinanceironovo,
			MAX(dt_baixa) 								AS dt_baixa,
			MAX(dt_reneg) 								AS dt_reneg,
			MAX(SAFE_CAST(vr_presente AS FLOAT64)) 		AS vr_presente,
			MAX(SAFE_CAST(vr_financiado AS FLOAT64)) 	AS vr_financiado,
			MAX(SAFE_CAST(Tx_Pactuada AS FLOAT64)) 		AS Tx_Pactuada,
			MAX(SAFE_CAST(tx_contrato AS FLOAT64)) 		AS tx_contrato,
			MAX(qt_parcela) 							AS qt_parcela,
			MAX(vr_prestacao) 							AS vr_prestacao,
			MAX(FlRestruturacaoContrato) 				AS FlRestruturacaoContrato

			FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador

			WHERE dt_baixa >= &dt_inicio. and dt_baixa < &dt_fim. and
				(dt_cancelamento is null or dt_reativacao is not null) and
				ST_RENEG = 3

			GROUP BY
				NuContratoGestao,
				NuContratoGestaoNovo
) AS A

			SELECT
				NUCONTRATO

				,MAX(VrDescontoMulta) 				AS vr_desconto_multa
				,MAX(VrDescontoPrincipal) 			AS vr_desconto_principal
				,MAX(FlReestruturacao) 				AS FlReestruturacao
				,MAX(TpEntrada)						AS TpEntrada

			FROM GOPRRenegociacaoSimulacao.TbRenegociacaoEscolhida
			WHERE ((DtAlteracao >= &dt_inicio. and DtAlteracao <= &dt_fim.)
			OR (DtFormalizacao >= &dt_inicio. and DtFormalizacao <= &dt_fim.))
			and CdStatusRenegociacao = 4

			GROUP BY NUCONTRATO

) AS B
ON INPUT(a.NuContratoGestao, 15.) = b.NUCONTRATO
AND a.SqContratoFinanceiro = b.SqContratoFinanceiro

;QUIT;

CREATE TABLE DBA AS
SELECT
	*
FROM CONNECTION TO PERM(
	WITH LISTA_CONTRATOS AS (

		SELECT DISTINCT
			NuContratoGestao
		FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
		WHERE
			dt_baixa >= &dt_inicio. AND dt_baixa < &dt_fim. AND
			(dt_cancelamento is null or dt_reativacao is not null) AND
			ST_RENEG = 3
)

	SELECT DISTINCT
		A.NuContratoGestao,
		B.sq_contrato_financeiro,
		B.vr_parcela,
		B.st_parcela,
		B.dt_ultimo_recebimento,
		B.dt_vencimento,
		C.tx_juros as Tx_Contrato_sq_1,
		CASE
			WHEN D.fl_restruturacao_contrato = FALSE THEN "N"
			WHEN D.fl_restruturacao_contrato = TRUE THEN "S"
			ELSE "N"
		END AS Fl_Reest_Anterior,

	FROM LISTA_CONTRATOS AS A

	LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
	ON A.NuContratoGestao = B.cd_contrato

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS C
	ON A.NuContratoGestao = C.cd_contrato AND C.sq_contrato_financeiro = 1

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS D
	ON B.cd_contrato = D.cd_contrato and B.sq_contrato_financeiro = D.sq_contrato_financeiro
)
;QUIT;

CREATE TABLE RENEG_SIMULADOR AS
SELECT DISTINCT
    A.NuContratoGestao AS NuContrato,
    A.SqContratoFinanceiro,
    A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo,
    A.dt_baixa,
    A.dt_reneg,
    A.vr_presente,
    A.vr_financiado,
    A.qt_parcela,
    A.vr_prestacao,
    A.vr_desconto_mora,
    A.vr_desconto_multa,
    A.vr_desconto_principal,
    A.Tx_Pactuada,
	A.Tx_Contrato,
    B.Tx_Contrato_sq_1,
    B.Fl_Reest_Anterior,
    A.FlRestruturacaoContrato,
	A.TpEntrada,

    MIN(B.Vr_Parcela) AS VrParcela_Anterior,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DtMin,

    COUNT(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_Abertas,

    COUNT(CASE
        WHEN (B.st_parcela <> 'Q' OR
              (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)) AND
             B.dt_vencimento < A.dt_reneg
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_em_Atraso,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DT_VENCIMENTO_NOVO

FROM RENEG_REALIZADA_&anomes. A

LEFT JOIN DBA B
  ON A.NuContratoGestao = B.NuContratoGestao
 AND A.SqContratoFinanceiro = B.sq_contrato_financeiro

GROUP BY
    A.NuContratoGestao, A.SqContratoFinanceiro, A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo, A.dt_baixa, A.dt_reneg, A.vr_presente,
    A.vr_financiado, A.qt_parcela, A.vr_prestacao, A.vr_desconto_mora,
    A.vr_desconto_multa, A.vr_desconto_principal, A.Tx_Pactuada, A.Tx_Contrato,
    A.FlRestruturacaoContrato, A.TpEntrada,
    B.Tx_Contrato_sq_1, B.Fl_Reest_Anterior
;
QUIT;

PROC SQL;

CONNECT TO BIGQUERY AS PERM2 (PROJECT='bv-safn-dtm' credfile='/projetos/acessos/bq/fn/safn-dtm.json'
     SCHEMA='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=25);

CREATE TABLE RENEG_SIMULADOR_D0 AS
SELECT DISTINCT
		A.*,

		CASE
			WHEN B.CdSItuacaoContabil IS NOT NULL THEN B.CdSItuacaoContabil
			ELSE C.CdSItuacaoContabil
		END AS CdSItuacaoContabil_0,

		B.VRTAXACONTRATO AS VRTAXACONTRATO_0,
		B.VRPERMANENCIAAPROPRIADO AS VRPERMANENCIAAPROPRIADO_0,
		B.VrTaxaContratoRestruturacao AS VrTaxaContratoRestruturacao_0,
		B.VrPDD AS VrPDD_0,

		CASE
			WHEN D.Carteira IS NOT NULL THEN D.CARTEIRA
			ELSE C.Carteira
		END AS Carteira,

		CASE
			WHEN D.PDD IS NOT NULL THEN D.PDD
			ELSE C.PDD
		END AS PDD,

		B.NuDiasAtraso AS NuDiasAtraso_0,
		INPUT(UF1.cd_cpf_cnpj, BEST32.) AS NuCpfCnpj,
		upcase(UF2.cduf) as UF

FROM	RENEG_SIMULADOR_C AS A

LEFT JOIN CONNECTION TO PERM	(

		SELECT DISTINCT
				NUCONTRATO,
				SqContratoFinanceiro,
				CdRatingAtual,
				CdSItuacaoContabil,
				SUM(SAFE_CAST(VRTAXACONTRATO AS FLOAT64)) 				AS VRTAXACONTRATO,
				SUM(SAFE_CAST(VRPERMANENCIAAPROPRIADO AS FLOAT64)) 		AS VRPERMANENCIAAPROPRIADO,
				SUM(SAFE_CAST(VrTaxaContratoRestruturacao AS FLOAT64)) 	AS VrTaxaContratoRestruturacao,
				SUM(SAFE_CAST(VrPDD AS FLOAT64)) 						AS VrPDD,
				MAX(NuDiasAtraso) 										AS NuDiasAtraso

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		WHERE DATE(DtProcessamento) = &dt_fech_m0.
			and DtProcessamentoLiquidacao is null
		GROUP BY NUCONTRATO, CdRatingAtual, CdSItuacaoContabil, SqContratoFinanceiro
) AS B
		ON	A.NuContrato = B.NuContrato and A.SqContratoFinanceiro = B.SqContratoFinanceiro

LEFT JOIN CONNECTION TO PERM	(

		SELECT DISTINCT
				NUCONTRATO,
				SqContratoFinanceiro,
				CdSItuacaoContabil,
				SUM(CAST(VrTaxaContrato AS FLOAT64) + CAST(VrPermanenciaApropriado AS FLOAT64)) AS Carteira,
				SUM(CAST(VrPDD AS FLOAT64)) AS PDD

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		WHERE DtProcessamento = (select MAX(DtProcessamento) from bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil )
		GROUP BY NUCONTRATO, CdSItuacaoContabil, SqContratoFinanceiro
) AS C
		ON	A.NuContrato = C.NuContrato

LEFT JOIN CONNECTION TO PERM2 (
	SELECT
			cd_contrato
			,SUM(vr_saldo_carteira) as CARTEIRA
			,SUM(vr_provisao_total) AS PDD

	FROM bv-pdbd-prd.dvry_corp_riscocredat.provisao_varejo_fato
	WHERE dt_referencia = &dt_fech_m0.
	GROUP BY cd_contrato
)as D
ON A.nucontrato = D.cd_contrato


LEFT JOIN CONNECTION TO PERM (
	SELECT
		cd_contrato,
		cd_cpf_cnpj
	FROM bv-pdbd-prd.dvry_core.tbcontrato
) AS UF1
ON A.NUCONTRATO = UF1.CD_CONTRATO


LEFT JOIN CONNECTION TO PERM (
	SELECT
		A.nucpfcnpj,
		A.cdpessoa,
		B.cduf,
		B.nmcidade
	FROM bv-pdbd-prd.trusted_dbcor.tbpessoa AS A
	LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS B
    ON A.cdpessoa = B.cdpessoa
) AS UF2
ON UF1.cd_cpf_cnpj = UF2.nucpfcnpj

;QUIT;

DATA RENEG_SIMULADOR_D;
SET RENEG_SIMULADOR_D0;

	IF CdSItuacaoContabil_0 = 1 THEN
		SALDO = Carteira;
	ELSE
		SALDO = VRTAXACONTRATO_0;

	IF CdSItuacaoContabil_0 = 1 THEN
		PDD_FINAL_0 = SALDO;
	ELSE
		PDD_FINAL_0 = 0;

	IF Qtd_Parcelas_Abertas < qt_parcela THEN ST_RENEG = '1. AUMENTO PRAZO'; ELSE
	IF Qtd_Parcelas_Abertas = qt_parcela THEN ST_RENEG = '2. MANUT. PRAZO '; ELSE
	IF Qtd_Parcelas_Abertas > qt_parcela THEN ST_RENEG = '3. REDUÇÃO PRAZO';

	Diff_Parcela = sum(qt_parcela,-Qtd_Parcelas_Abertas);

	if Diff_Parcela <   0  then fx_parcela = '0. Redução'; else
	if Diff_Parcela <=  1  then fx_parcela = '1. 00-01  '; else
	if Diff_Parcela <=  3  then fx_parcela = '2. 02-03  '; else
	if Diff_Parcela <=  6  then fx_parcela = '3. 04-06  '; else
	if Diff_Parcela <=  12 then fx_parcela = '4. 07-12  '; else
	if Diff_Parcela <=  24 then fx_parcela = '5. 13-24  '; else
	if Diff_Parcela <=  36 then fx_parcela = '6. 25-36  '; else
	if Diff_Parcela  >  36 then fx_parcela = '7. 36-99  ';

	IF tx_contrato = tx_pactuada THEN VAR_Taxa = "Manutencao"; ELSE
	IF tx_contrato > tx_pactuada THEN VAR_Taxa = "Reducao"; ELSE
	IF tx_pactuada < tx_contrato THEN VAR_Taxa = "Aumento";
	ELSE VAR_Taxa = "";

;RUN;

	DELTA_PMT = ROUND(vr_prestacao / VrParcela_Anterior, .1);

	IF DELTA_PMT <= 0.50 	THEN FX_NOVA_PMT = '2. 25-50% '; ELSE
	IF DELTA_PMT <= 0.75 	THEN FX_NOVA_PMT = '3. 50-75% '; ELSE
	IF DELTA_PMT <= 1.00 	THEN FX_NOVA_PMT = '4. 75-100%'; ELSE
							 FX_NOVA_PMT = '5. > 100% ';

	IF VrParcela_Anterior > vr_prestacao THEN VAR_PMT = "Reducao"; ELSE
	IF VrParcela_Anterior < vr_prestacao THEN VAR_PMT = "Aumento";
	ELSE VAR_PMT = "Manutencao";

	IF NuDiasAtraso_M3 > 30 THEN FL_OVER30_M3 = 1; ELSE FL_OVER30_M3 = 0;

RUN;

CREATE TABLE DBA AS
SELECT
	*
FROM CONNECTION TO PERM(
	WITH LISTA_CONTRATOS AS (

		SELECT DISTINCT
			NuContratoGestao
		FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
		WHERE
			dt_baixa >= &dt_inicio. AND dt_baixa < &dt_fim. AND
			(dt_cancelamento is null or dt_reativacao is not null) AND
			ST_RENEG = 3
)

	SELECT DISTINCT
		A.NuContratoGestao,
		B.sq_contrato_financeiro,
		B.vr_parcela,
		B.st_parcela,
		B.dt_ultimo_recebimento,
		B.dt_vencimento,
		C.tx_juros as Tx_Contrato_sq_1,
		CASE
			WHEN D.fl_restruturacao_contrato = FALSE THEN "N"
			WHEN D.fl_restruturacao_contrato = TRUE THEN "S"
			ELSE "N"
		END AS Fl_Reest_Anterior,

	FROM LISTA_CONTRATOS AS A

	LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
	ON A.NuContratoGestao = B.cd_contrato

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS C
	ON A.NuContratoGestao = C.cd_contrato AND C.sq_contrato_financeiro = 1

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS D
	ON B.cd_contrato = D.cd_contrato and B.sq_contrato_financeiro = D.sq_contrato_financeiro
)
;QUIT;

CREATE TABLE RENEG_SIMULADOR AS
SELECT DISTINCT
    A.NuContratoGestao AS NuContrato,
    A.SqContratoFinanceiro,
    A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo,
    A.dt_baixa,
    A.dt_reneg,
    A.vr_presente,
    A.vr_financiado,
    A.qt_parcela,
    A.vr_prestacao,
    A.vr_desconto_mora,
    A.vr_desconto_multa,
    A.vr_desconto_principal,
    A.Tx_Pactuada,
	A.Tx_Contrato,
    B.Tx_Contrato_sq_1,
    B.Fl_Reest_Anterior,
    A.FlRestruturacaoContrato,
	A.TpEntrada,

    MIN(B.Vr_Parcela) AS VrParcela_Anterior,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DtMin,

    COUNT(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_Abertas,

    COUNT(CASE
        WHEN (B.st_parcela <> 'Q' OR
              (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)) AND
             B.dt_vencimento < A.dt_reneg
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_em_Atraso,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DT_VENCIMENTO_NOVO

FROM RENEG_REALIZADA_&anomes. A

LEFT JOIN DBA B
  ON A.NuContratoGestao = B.NuContratoGestao
 AND A.SqContratoFinanceiro = B.sq_contrato_financeiro

GROUP BY
    A.NuContratoGestao, A.SqContratoFinanceiro, A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo, A.dt_baixa, A.dt_reneg, A.vr_presente,
    A.vr_financiado, A.qt_parcela, A.vr_prestacao, A.vr_desconto_mora,
    A.vr_desconto_multa, A.vr_desconto_principal, A.Tx_Pactuada, A.Tx_Contrato,
    A.FlRestruturacaoContrato, A.TpEntrada,
    B.Tx_Contrato_sq_1, B.Fl_Reest_Anterior
;
QUIT;

		SELECT DISTINCT
				NUCONTRATO,

				CdRatingAtual,

				SUM(SAFE_CAST(VRTAXACONTRATO AS FLOAT64)) 				AS VRTAXACONTRATO,

		ON	A.NuContrato = B.NuContrato and A.SqContratoFinanceiro = B.SqContratoFinanceiro

		SELECT DISTINCT
				NUCONTRATO,

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		WHERE DATE(DtProcessamento) = &dt_fim.
		GROUP BY NUCONTRATO, SqContratoFinanceiro

		ON	A.NuContrato = C.NuContrato

			NUCONTRATO,
			NmProdutoRisco AS Produto,

			MAX(PDD) AS PDD

ON A.nucontrato = D.nucontrato

ON A.nucontrato = E.nucontrato

PROC SQL;
CREATE TABLE RENEG_TIPO AS
SELECT *
	,CASE
		WHEN FL_REEST IS NOT NULL THEN FL_REEST
		ELSE FlRestruturacaoContrato
	END AS FlRestruturacao_Final

	,CASE
		WHEN calculated FlRestruturacao_Final = "S" 					   THEN "Reest"
		WHEN calculated FlRestruturacao_Final = "N" AND atraso <= 30 THEN "Reorg"
		ELSE "Reneg"
	END AS Tp_Repac

FROM RENEG_CORRECAO

PROC SQL;
CREATE TABLE RENEG_FINAL_&anomes. AS
SELECT
	A.*,
	MAX(CASE
		WHEN POL.Grupo_Final_Decisao IS NOT NULL THEN POL.Grupo_Final_Decisao
		WHEN POL2.Modelo_Final IS NOT NULL THEN POL2.Modelo_Final
		ELSE "N/A"
	END) AS POLITICA

FROM RENEG_TIPO AS A

LEFT JOIN WORK.empilha_reneg_final_&anomes. AS POL
ON INPUT(A.nucontrato, best32.) = POL.NUCONTRATO
AND A.SqContratoFinanceiro = POL.SQCONTRATOFINANCEIRO

LEFT JOIN WORK.base_eleg_cont_&anomes. AS POL2
ON INPUT(A.nucontrato, best32.)  = POL2.nucontrato
AND  A.sqcontratofinanceiro = POL2.sqcontratofinanceiro

GROUP BY A.NUCONTRATO
;QUIT;
