PROC SQL;
CREATE TABLE TbBvExper_Resumo_Pontos_REG AS
SELECT

'Regional' as TpNivel,
'Pontos' as TpBase,
A.NuAnoMes,
A.NmCanalEstruturaComercial,
'' as DsLoginN3A,
A.NmSubEstruturaComercialA,
'' as DsLoginN2A,
'' as NmAreaEstruturaComercial,
. as CdColaborador,
. as IdMatricula,
'' as DsLoginN1,
'' as NmColaborador,

A.NmGrupoCargo,
A.QtDiasAtividadeNormal,
A.QtDiasFerias,
A.QtDiasOutros,
A.NmElegAcelerador,

sum(A.Pontos_VrProducao) AS Producao FORMAT COMMAX20.6,
sum(A.Pontos_MktShareCart) AS MktShareCart FORMAT COMMAX20.6,
sum(A.Pontos_Spread) AS Spread FORMAT COMMAX20.6,
sum(A.Pontos_VrMC_Rodas) AS MC_Rodas FORMAT COMMAX20.6,
sum(A.Pontos_VrMC_EGV) AS MC_EGV FORMAT COMMAX20.6,
sum(A.Pontos_VrMCN) AS MCN FORMAT COMMAX20.6,
sum(A.Pontos_PenPDV) AS PenPDV FORMAT COMMAX20.6,

sum(A.Pontos_VrSPF) AS SPF FORMAT COMMAX20.6,
sum(A.Pontos_VrAP) AS AP FORMAT COMMAX20.6,
sum(A.Pontos_VrSegurosAuto) AS SegurosAuto FORMAT COMMAX20.6,
sum(A.Pontos_VrSPF_VrAP) AS SPF_AP FORMAT COMMAX20.6,
sum(A.Pontos_VrAP_VrPlacaSolar) as AP_PlacaSolar FORMAT COMMAX20.6,
sum(A.Pontos_VrMetaInad15) AS MetaInad15 FORMAT COMMAX20.6,
sum(A.Pontos_VrMetaInad30) AS MetaInad30 FORMAT COMMAX20.6,
sum(A.Pontos_SegurosCancelados) AS SegurosCancelados FORMAT COMMAX20.6,
sum(A.Pontos_Efetivacao) AS Efetivacao FORMAT COMMAX20.6,
sum(A.Pontos_ContaDigital) AS ContaDigital FORMAT COMMAX20.6,
sum(a.Pontos_ParceiroProduzindo) as ParceiroProduzindo FORMAT COMMAX20.6,

AVG(A.Pontos_Total) AS Pontos_Total FORMAT COMMAX20.6,
A.Faixa_Pontos AS Faixa_Pontos,
AVG(FlAting_PtsTurbo) AS FlAting_PtsTurbo FORMAT COMMAX20.6,
AVG(A.PONTOS_TURBO) AS PONTOS_TURBO FORMAT COMMAX20.6,
A.FAIXA_PONTOS_TURBO AS FAIXA_PONTOS_TURBO

FROM EXP_BU.TbBaseUnica_BvExper_Reg A

GROUP BY
Calculated TpNivel,
Calculated TpBase,
NuAnoMes,
NmCanalEstruturaComercial,
Calculated DsLoginN3A,
NmSubEstruturaComercialA,
Calculated DsLoginN2A,
Calculated NmAreaEstruturaComercial,
Calculated CdColaborador,
Calculated IdMatricula,
Calculated DsLoginN1,
Calculated NmColaborador,

A.NmGrupoCargo,
A.QtDiasAtividadeNormal,
A.QtDiasFerias,
A.QtDiasOutros,
A.NmElegAcelerador,
A.Faixa_Pontos,
A.FAIXA_PONTOS_TURBO

ORDER BY
A.NmCanalEstruturaComercial,
A.NmSubEstruturaComercialA,
A.NuAnoMes
/*A.DsLoginN3A*/
;RUN;
