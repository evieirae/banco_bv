options validvarname=any;
options nofmterr;

/* Marcar horário de início da atualização */
%LET W_ini = %SYSFUNC(datetime());

/* Enviar e-mail de início da atualização */
OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_INI EMAIL
	FROM=("inf-gerenciais@bv.com.br")
	TO=("inf-gerenciais@bv.com.br")
SUBJECT="Qualidade de Vendas Simu | Início da atualização";

DATA _NULL_;
	FILE MAIL_INI;
		PUT "Processamento das bases do QDV iniciado.";
		PUT "Este processo gerará as seguintes bases:";
		PUT " - TbQualidadeVendasIGESimu.csv";
RUN;

/* Execução dos processos */
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/00-parametros.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/01-simulacoes.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/02-base-esg.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/03-super-trunfo.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/04-mkt-share.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/05-originacao-e-producao.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/06-inad.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/07-propostas-e-contratos.sas';
%include '/infogrc/qualidadedevendas/codigos/qdv-simu/08-unificar-bases.sas';

/* Calcular tempo de processamento total */
%LET W_fim = %SYSFUNC(datetime());
%LET W_dur = %SYSFUNC(intck(SECOND, &W_ini, &W_fim), tod8.);

/* Enviar e-mail de finalização da atualização */
PROC SQL;
CREATE TABLE DATA_ATUALIZA AS
  SELECT
   TpDados,
   MAX(DtReferencia) FORMAT=ddmmyy10. AS DtAtualizacao
  FROM QDVCompleto
  GROUP BY 1;
RUN;

OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_FIM EMAIL
	FROM=("inf-gerenciais@bv.com.br")
	TO=("inf-gerenciais@bv.com.br")
    TYPE="text/html"
    SUBJECT="Qualidade de vendas Simu | Fim da atualização";

ODS html body=MAIL_FIM style=noline rs=none;

PROC REPORT DATA=DATA_ATUALIZA nowd HEADLINE HEADSKIP
style (report) = {background = none font_face = "Calibri" font_size = 9pt just=left OUTLINECOLOR = none}
style (column) = {background = none CELLHEIGHT = 2.5% font_face = "Calibri" font_size = 9pt just=center OUTLINECOLOR = none}
style (header) = {foreground = white font_face="Calibri" font_size = 9pt just=center background = #69c3cf OUTLINECOLOR = none} ;

COLUMNS TpDados DtAtualizacao;
RUN;

PROC ODSTEXT;
   p "Prezados,";
   p "";
   p "As bases do Qualidade de Vendas foram atualizadas.";
   p "A geração das bases demorou &W_dur. para ser concluída.";
   P "";
   p "Atenciosamente,";
   p "Informações Gerenciais Estratégicas";
RUN;

ods _all_ close;

DATA LOGEXE;
  IdProcesso = 'QDVSIMU';
  LENGTH Processo $40.;
  Processo = 'Qualidade de Vendas Simu';
  FORMAT Inicio Final DATETIME18.;
  Inicio = &W_ini.;
  Final = &W_fim.;
  Duracao = "&W_dur.";
RUN;

PROC APPEND DATA=LOGEXE BASE=IGELOG.LogExecucaoProcessos; RUN;
