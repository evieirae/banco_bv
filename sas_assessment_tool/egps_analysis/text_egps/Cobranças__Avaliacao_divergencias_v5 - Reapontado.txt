CREATE TABLE
	ANALITICO
AS

	CASE WHEN CALCULATED SEM_MARCACAO = 1 AND CALCULATED IMPEDITIVO = 1 THEN 0
		 WHEN CALCULATED SEM_MARCACAO = 1 AND CALCULATED IMPEDITIVO = 0 THEN 1
		 ELSE 0 END AS S_MARCA_CORRIGIDA,
	CASE
		WHEN CALCULATED IMPEDITIVO = 1 THEN '00. Impeditivo'
		WHEN CALCULATED S_MARCA_CORRIGIDA = 1 THEN '01. S/Marcacao'
		WHEN CALCULATED SALDO_REMANESC = 1 THEN '02. Saldo Remanescente'

	DIVERGENCIAS3;
