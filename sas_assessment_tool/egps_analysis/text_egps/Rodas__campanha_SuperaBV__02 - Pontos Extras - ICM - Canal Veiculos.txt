PROC SQL;
CREATE TABLE BASE_CALCULO_ICM_1 AS
SELECT

&Base_Mes. AS NuAnoMes,
c.DtAdmissaoColaborador,
c.NmEstadoColaborador,
A.DsLoginN1,
C.NmGrupoCargo AS NmGrupoCargoColaborador,
C.CdColaboradorAtivo AS CdColaborador,
'ICM' AS ICM,
A.VrICM*100 as VrICM format commax20.8,
B.media_ICM AS MEDIA_ORIG,/**/
CASE WHEN (B.media_ICM*100) IN (., 0) THEN 90 ELSE (B.media_ICM*100) END as media_ICM,

CASE WHEN CALCULATED media_ICM = 90 and (B.media_ICM*100) NOT IN (., 0) THEN 90 ELSE CALCULATED media_ICM + 10 END AS ICM_Meta

/*CASE WHEN VrICM >= CALCULATED ICM_Meta THEN 1 ELSE 0 END AS Fl_Ating_ICM,*/
/**/
/*CASE WHEN CALCULATED Fl_Ating_ICM = 1 THEN 20 ELSE 0 END AS VrPontosExtra*/

FROM WORK.BASE_ICM_CRESCIMENTO_2 A /*BASE ICM DO MÊS*/

LEFT JOIN CSBV.BASE_MEDIA_ICM_CV B ON B.DSLOGINN1 = A.DSLOGINN1 /*BASE FIXA COM O CALCULA DA MÉDIA DO TRIMESTRE*/
LEFT JOIN WORK.ESTRUTURA_COMERCIAL_01 C ON C.DsLoginN1 = A.DsLoginN1 AND C.NUANOMES = &Base_Mes. /*ESTRUTURA COMERCIAL PARA TRAZER O CARGO*/

;RUN;

PROC SQL;
CREATE TABLE BASE_CALCULO_ICM_2 AS
SELECT

NuAnoMes,
DtAdmissaoColaborador,
NmEstadoColaborador,
A.DsLoginN1,
a.NmGrupoCargoColaborador,
a.CdColaborador,
A.ICM,
ROUND(A.VrICM,0.01) AS VrICM,
ROUND(A.media_ICM, 0.01) AS media_ICM,

ROUND(A.ICM_Meta, 0.01) AS ICM_Meta,

CASE WHEN ROUND(VrICM, 0.01) >= ROUND(ICM_Meta, 0.01) THEN 1 ELSE 0 END AS Fl_Ating_ICM,

CASE WHEN CALCULATED Fl_Ating_ICM = 1 THEN 20 ELSE 0 END AS VrPontosExtra

FROM WORK.BASE_CALCULO_ICM_1 A

WHERE a.NmGrupoCargoColaborador in ('Gerente Relacionamento Veiculos')

;RUN;
