create table DESCONTOS_GRADATIVOS as
select distinct A.*
from connection to perm (
  with max_por_contrato as (
    select distinct
      NUCONTRATOORIGEM,
      max(date(DTINCLUSAO)) as max_d
    from `bv-pdbd-prd.trusted_dbsrcgmurc.tbdescontogradativo`
	where date(DTINCLUSAO) <= DATE(&DIA_REF.)
	AND SUBSTR(NUCONTRATOORIGEM, 1,2) IN ('12', '14')

    group by NUCONTRATOORIGEM
  )
  select
    F.NUCONTRATOORIGEM,
    F.QTDIAFIMFAIXAATRASO,
    F.QTDIAINICIOFAIXAATRASO,
    G.PCDESCONTOPARCELA,
	G.pcdescontomora
	,G.pcdescontomulta
	,G.pcdescontodespesa
	,G.pcdescontohonorario
    ,F.CDDESCONTOGRADATIVO,
    F.DTINCLUSAO,
    case when F.TPACORDO = '1' then 'Q'
         when F.TPACORDO = '2' then 'A'
         when F.TPACORDO = '3' then 'P' end as DSTPACORDO
  from `bv-pdbd-prd.trusted_dbsrcgmurc.tbdescontogradativo` as F
  join max_por_contrato M
    on F.NUCONTRATOORIGEM = M.NUCONTRATOORIGEM
   and date(F.DTINCLUSAO) = M.max_d
  left join `bv-pdbd-prd.trusted_dbsrcgmurc.tbdescontogradativofaixa` as G
    on F.CDDESCONTOGRADATIVO = G.CDDESCONTOGRADATIVO
   and G.CDFAIXADESCONTO = 3
) as A;

quit;
