/* ===========================================================================
 3. ATRASO MAX PAGO
==============================================================================*/

PROC SQL NOPRINT;
	connect to SYBASEIQ (&conex_syb.);

	CREATE TABLE TB_ATRASOMAX_&dtsafra. AS SELECT * FROM connection to SYBASEIQ
		(

		Declare @DtCalculo Varchar(8)
	Set @DtCalculo=&dtfimchar.

	SELECT
		Nucontrato,
		sqcontratofinanceiro,
		nuparcela,
		atraso = datediff(day,DtVencimento,DtUltimoRecebimento),
		convert(char(10),DtUltimoRecebimento,103)

	INTO #ER
		FROM cdccg.tbparcela
			WHERE
				stparcela ='Q' and
				DtVencimento < DtUltimoRecebimento and
				DtUltimoRecebimento < @DtCalculo

			insert into #ER
				SELECT
					Nucontrato,
					sqcontratofinanceiro,
					nuparcela,
					atraso = datediff(day,DtVencimento,DtUltimoRecebimento),
					convert(char(10),DtUltimoRecebimento,103)

				FROM lsg.tbparcela

				WHERE
					stparcela ='Q' AND
					DtVencimento < DtUltimoRecebimento AND
					DtUltimoRecebimento < @DtCalculo

				SELECT
					Nucontrato,
					sqcontratofinanceiro,
					safra = &dtsafra.,
					max(atraso) as AtrasoMaximoPago
				INTO #ER1
					FROM #ER
						GROUP BY Nucontrato,sqcontratofinanceiro

						SELECT
							a.*,
							b.DtUltimoRecebimento
						FROM #ER1 AS a LEFT JOIN #ER AS b ON a.Nucontrato=b.Nucontrato
							AND a.AtrasoMaximoPago=b.atraso
							AND a.sqcontratofinanceiro=b.sqcontratofinanceiro

		);
QUIT;

PROC SORT DATA=TB_ATRASOMAX_&dtsafra. OUT=TB_ATRASOMAX1_&dtsafra. (DROP=DtUltimoRecebimento safra) NODUPKEY;
	BY Nucontrato sqcontratofinanceiro;
RUN;

PROC DATASETS LIB= WORK;
	DELETE TB_ATRASOMAX_&dtsafra.;
RUN;

PROC SQL NOPRINT;
	connect to SYBASEIQ (&conex_syb.);

	CREATE TABLE TB_QTD_PARCELAS_&dtsafra. AS SELECT * FROM connection to SYBASEIQ
	(
	Declare @dtbase Varchar(8)
	Set @dtbase=&dtfimchar.

	SELECT
		a.Nucontrato,
		a.SqContratoFinanceiro,
		a.Nuparcela,
		atraso_15 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 1 and 15
					then 1 else 0 end,
		atraso_30 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 16 and 30
					then 1 else 0 end,
		atraso_45 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 31 and 45
					then 1 else 0 end,
		atraso_60 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 46 and 60
					then 1 else 0 end,
		atraso_61 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) >= 61
					then 1 else 0 end
	INTO #ER

	FROM cdccg.tbparcela as a left join cdccg.tbparcelarecebimento as b
		ON a.Nucontrato = b.Nucontrato and a.SqContratoFinanceiro=b.SqContratoFinanceiro AND a.Nuparcela=b.Nuparcela
	WHERE b.flestornado = 'N' AND b.cdagenterecebedor NOT IN (132,137,167)

	insert into #ER
	SELECT
		a.Nucontrato,
		a.SqContratoFinanceiro,
		a.Nuparcela,
		atraso_15 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 1 and 15
					then 1 else 0 end,
		atraso_30 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 16 and 30
					then 1 else 0 end,
		atraso_45 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 31 and 45
					then 1 else 0 end,
		atraso_60 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) between 46 and 60
					then 1 else 0 end,
		atraso_61 = case when a.StParcela = 'Q'  and a.DtUltimoRecebimento <=  @DtBase and datediff(day,a.DtVencimento,a.DtUltimoRecebimento) >= 61
					then 1 else 0 end

	FROM lsg.tbparcela as a left join lsg.tbparcelarecebimento as b
		ON a.Nucontrato = b.Nucontrato and a.SqContratoFinanceiro=b.SqContratoFinanceiro and a.Nuparcela=b.Nuparcela
	WHERE b.flestornado = 'N' AND b.cdagenterecebedor NOT IN (132,137,167)

	SELECT
		Nucontrato,
		SqContratoFinanceiro,
		sum(atraso_15) as qtd_parc_01_15,
		sum(atraso_30) as qtd_parc_16_30,
		sum(atraso_45) as qtd_parc_31_45,
		sum(atraso_60) as qtd_parc_46_60,
		sum(atraso_61) as qtd_parc_mais60
	FROM #ER
	GROUP BY Nucontrato,SqContratoFinanceiro
	);
QUIT;

			a.*,
			b.Score_Serasa as Prob_Serasa
		FROM tb_carteira_cons3_&dtsafra. AS a LEFT JOIN extrac.SERASA_COLLEC_BLENDING AS b
			ON a.nucpfcnpj = b.nucpfcnpj AND a.modelo=b.modelo And a.Produto=b.Produto
		WHERE a.Modelo = 1;

	CREATE TABLE BLENDING2_&dtsafra. AS

			a.*,
			b.Score_Serasa as Prob_Serasa
		FROM tb_carteira_cons3_&dtsafra. AS a LEFT JOIN extrac.SERASA_COLLEC_BLENDING AS b

	WHERE a.Modelo = 2;

	CREATE TABLE BLENDING3_&dtsafra. AS

			a.*,

		FROM tb_carteira_cons3_&dtsafra. AS a LEFT JOIN extrac.SERASA_COLLEC_BLENDING AS b
			ON a.nucpfcnpj = b.nucpfcnpj AND a.modelo=b.modelo And a.Produto=b.Produto
		WHERE a.Modelo = 3;

	SET  blending1_&dtsafra. blending2_&dtsafra. blending3_&dtsafra. blending4_&dtsafra.;

	IF modelo = 1 	   THEN	score_bld = - 3.3831 + 4.7966*prob + 1.3597*prob_serasa;
	ELSE IF modelo = 2 THEN score_bld = - 2.7962 + 4.1649*prob + 1.2576*prob_serasa;
	ELSE IF modelo = 3 THEN score_bld = - 2.829168 + 4.097172*prob + 2.053539*prob_serasa;

	IF modelo in (1,2,3) THEN
		prob_bld=1/(1+(exp(-score_bld)));
	ELSE prob_bld = prob;
