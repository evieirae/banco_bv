/*Somente GR Pesados - Inad 30 alterar a data para 3 meses antes da data de fechamento*/
/*Ex: Fechamento é de Abril então o NuAnoMes da entrada da proposta será de Janeiro, 3 meses antes*/

PROC SQL;
CREATE TABLE TbFechamentoInad30 AS
select
        month(Inad30.DtEntradaProposta) AS Mes_Base
        ,year(Inad30.DtEntradaProposta) AS Ano_Base
        ,Colaborador.IdMatricula AS ID_RH
        ,Inad30.NmColaborador
        ,11596 AS Produto
        ,sum(coalesce(Inad30.VrParcelaInad30,0)) AS VrParcelaInad30 format commax18.2
        ,sum(Inad30.VrParcelaElegivelInad30) AS  VrParcelaElegivelInad30 format commax18.2
        ,(sum(coalesce(Inad30.VrParcelaInad30,0))/sum(Inad30.VrParcelaElegivelInad30))*100 AS Valor format commax10.2

    from bases.TbInad Inad30
        inner join bases.TbColaborador Colaborador
            on Inad30.CdColaborador = Colaborador.CdColaborador
    where Inad30.FlElegivelInad30  = 1
        and year(Inad30.DtEntradaProposta)*100+month(Inad30.DtEntradaProposta) = &safraM4.
        and NmGrupoProdutoMeta = 'Pesados'
        and NmCanalEstruturaComercial = 'Canal Veiculos'
        and Colaborador.CdGrupoCargoColaborador in (20)
    group by 1,2,3,4,5

;run;

proc export
data=TbFechamentoInad30
file="/riscosbvf/BA/processos/tmp/Fechamento_Inad30_&safraM1..csv" replace
dbms=dlm;
delimiter=';';
run;
