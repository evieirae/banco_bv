			,
		case
			when A.PcEntradaSimu > 0.99
			THEN 1
		else round(A.PcEntradaSimu,0.01)
		END
	as  PcEntradaAjust
		,A.NmParceiroComercial
		,A.DsLoginN1 AS GerenteRegional
		,A.NmColaborador
		,A.DsLoginN2 AS GerenteFilial
		,A.FxPrazo
		,A.FlCard
		,A.Regional
		,A.Filial
		,A.Canal
		,A.Mensageria
		,A.TpPessoa
		,A.FlPorteA
		,A.FlZeroKm
		,A.FxIdade
		,A.IdadeBem
		,A.QtParcela
		,
	CASE
		WHEN A.ProbabilidadeAprovacao < 0
		THEN 0
	ELSE A.ProbabilidadeAprovacao
	END
AS ProbabilidadeAprovacao/*Esse é o score*/

	,A.DtSimulacaoInicial
	,A.DtInclusaoGravame
	,
	case
		when
		intck('day',A.DtSimulacaoInicial,A.DtInclusaoGravame) >= 0
			then intck('day',A.DtSimulacaoInicial,A.DtInclusaoGravame)
	end
	as MOBConc
		,
		case
			when intck('day',A.DtSimulacaoInicial,A.DtContrato) >= 0
				then intck('day',A.DtSimulacaoInicial,A.DtContrato)
		end
		as MOBConvContrato
			,
			case
				when intck('day',A.DtEntradaProposta,A.DtContrato) >= 0
					then intck('day',A.DtEntradaProposta,A.DtContrato)
			end
			as MOBEfet
				,A.NuCpfCnpj
				,A.FlSimulacao
				,A.FlSimuValida
				,A.FlSimuValidaBVO
				,A.FlSimuValidaEfet
				,A.FlSimuValidaDiaEfet
				,A.FlSimuValidaSemanaEfet
				,A.RatingCliente
				,A.RatingSimulacao
				,A.FlPropostaOriginada
				,A.FlPropostaAprovada
				,A.FlPropostaEfetivada
				,A.FlPropostaPaga
				,A.FlConversao
				,A.FlConcorrente
				,A.StPagoLoja
				,A.MotivoRecusa
				,A.DsSituacaoCredito
				,A.faixaScoreHSPN
				,A.FaixaMatriz
				,A.NotaGarantia
				,A.EntradaMinima
				,A.PcEntradaMinima
				,A.FlPap
				,A.Banco
				,A.VrVeiculoSimulacao
				,A.VrEntrada
				,A.VrProducao
				,A.VrProposta
				,
				case
					when A.FlSimulacao > 0
					and A.DtEntradaProposta IS NOT NULL
						THEN A.NuProposta
				END
				AS NuProposta
					,A.NuContrato
					From CONVSIMU.TbSimulacoes A
						Where 1=1
							and UPCASE(A.Produto) LIKE 'LEVES%'
							and A.NuAnoMes >= &MXDATASIMU.
/*											and A.NuAnomes >= 202501*/
	and A.DtSimulacaoInicial < TODAY()

	;
QUIT;

PROC SQL;
	CREATE TABLE  TbSimulacoes_00_leves_1 AS
		select
			TbSimu.*,
		CASE
			WHEN TbSimu.FxIdade = 'ZeroKM'
			then 'Novos'
		WHEN TbSimu.FxIdade = ''
		then 'nd'
	else 'Usados'
		end
	as Estado_Veiculo,
		CASE
			WHEN TbSimu.CdParceiroComercial = v7.Cod_Parceiro
			then 'PILOTO_V7'
		ELSE 'DEMAIS'
		end
	as PILOTO_V7,
		v7.Ondas as ONDA_V7,
		v7.V7 as V7,
	CASE
		WHEN TbSimu.CdParceiroComercial = Pdvcred.CdLoja and NuAnoMes <= 202402
		then Pdvcred.class_pdv_fev24
	WHEN TbSimu.CdParceiroComercial = Pdvcred.CdLoja and NuAnoMes >= 202403 then Pdvcred.class_pdv_mar24
	ELSE 'Sem Classificação'
	end
as classificacao_pdv,
	CASE
		WHEN TbSimu.CdParceiroComercial = PdvcredNew.CdLoja
		then PdvcredNew.class_pdv_nova
	ELSE 'Sem Classificação'
	end
as classificacao_pdv_nova,
	/*		TbSimu.PcEntradaAjust,*/
	V7EntMin.ent_min_1_ch,
	V7EntMin.ent_min_2_ch_v1 as ent_min_2_ch,
	PC.NmAreaEstruturaComercialC,
	MOTREC.ds_situacao_credito_pre

	FROM TbSimulacoes_00_leves TbSimu
		LEFT JOIN SOLITMP.ROLLOUT_V7 AS V7 ON TbSimu.CdParceiroComercial = V7.Cod_Parceiro
		left join solitmp.CLASS_PDVS_CREDITO_HIST AS Pdvcred ON TbSimu.CdParceiroComercial = Pdvcred.CdLoja
		left join solitmp.CLASS_PDVS_CREDITO_NOVA AS Pdvcrednew ON TbSimu.CdParceiroComercial = Pdvcrednew.CdLoja
		left join funil.v7_entr_min_pdec V7EntMin on TbSimu.CdSimulacaoInicial = V7EntMin.cd_simulacao_proposta
		left join BASES.TBPARCEIROCOMERCIAL AS PC ON PC.CDPARCEIROCOMERCIAL = TBSIMU.CDPARCEIROCOMERCIAL
		left join tb_motivo_recusa AS motrec ON INPUT(MOTREC.cd_simulacao_proposta,BEST.) = TbSimu.CdSimulacaoInicial

	;
QUIT;
