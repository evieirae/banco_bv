PROC SQL;
CREATE TABLE TbSLAGDoc as
SELECT
 RF.NuMonitoria
,RF.NuSolicitacao
,RF.NmAssessoria
,RF.NuContrato
,GDOC.DtSLA
,MAX(CASE WHEN GDOC.Fila = "CANCELADO"	THEN 1 ELSE 0 END ) 		AS FlCancelado

,MIN(CASE WHEN GDOC.Fila = "RESPONDIDA"	THEN GDOC.DtEntrada END )	AS MinDtEntrada	FORMAT  DATETIME22.

,DATEPART(CALCULATED MinDtEntrada )									AS DtResposta	FORMAT  DDMMYY8.
,CASE
  WHEN CALCULATED DtResposta  = .
   AND TODAY() <= GDOC.DtSLA				THEN 0

  AND RF.NmAssessoria  = GDOC.NmAssessoriaNovo

  AND RF.FlSLA = 0

GROUP BY  RF.NuMonitoria,RF.NuSolicitacao,RF.NmAssessoria,RF.NuContrato,GDOC.DtSLA
ORDER BY NuSolicitacao, NmAssessoria
;QUIT;RUN;
