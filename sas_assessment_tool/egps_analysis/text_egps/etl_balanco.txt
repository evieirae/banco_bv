PROC SQL NOPRINT;
	CONNECT TO SYBASE (&conex_moncred.);
	CREATE TABLE BASE_COMPLETA AS SELECT  * FROM CONNECTION TO SYBASE
	(
	          SELECT DISTINCT
	                t1.*,
	                t2.VrConta,
	                t2.VrContaPlanilhamento,
	                t2.CdContaContabilMonitor,
					t3.CdPessoaMonitoravel,
				    t3.FlControlador

	          FROM dbo.TbDemonstrativoContabilMonitor AS t1
	                INNER JOIN dbo.TbDemonstrativoContaMonitor AS t2
	                ON (t1.CdDemonstrativoContabilMonitor=t2.CdDemonstrativoContabilMonitor)
				LEFT JOIN dbo.TbComposicaoClienteMonitor AS t3
					ON (t1.CdDemonstrativoContabilMonitor=t3.CdDemonstrativoContabilMonitor)
			  WHERE DTCADASTRO <= &dt_fim.
	); disconnect from sybase;
QUIT;

PROC SQL NOPRINT;
	CONNECT TO SYBASE (&conex_moncred.);
	CREATE TABLE TbContaContabilMonitor AS SELECT  * FROM CONNECTION TO SYBASE
    (
			SELECT	NuConta,
					DsConta,
					CdContaContabilMonitor
			FROM dbo.TbContaContabilMonitor
	); disconnect from sybase;
QUIT;

PROC SQL NOPRINT;
	CONNECT TO SYBASE (&conex_moncred.);
	CREATE TABLE TbPessoaMonitoravel AS SELECT  * FROM CONNECTION TO SYBASE
    (
		    SELECT 	CdPessoa,
					NuCpfCnpjRaiz,
					NuCpfCnpjComplemento,
					NmPessoaMonitoravel,
					CdGrupoEconomico,
					CdPessoaMonitoravel
			FROM dbo.TbPessoaMonitoravel
	);disconnect from sybase;
QUIT;

PROC SQL;
CREATE TABLE BASE0  AS SELECT
t1.*,
(CASE WHEN CdAgrupamentoDemonstrativoMntr=1 THEN "NORM"
      WHEN CdAgrupamentoDemonstrativoMntr=2 THEN "CONS"
      WHEN CdAgrupamentoDemonstrativoMntr=3 THEN "COMB" END) AS AgrupamentoDemonstrativoMntr,

(CASE WHEN CdSegmentoMercado=2 THEN "MID"
      WHEN CdSegmentoMercado=3 OR CdSegmentoMercado=4 THEN "CORP" END) AS SegmentoMercado,

(CASE WHEN CdTipoDemonstrativo=1 THEN "Balanço"
      WHEN CdTipoDemonstrativo=2 THEN "Balancete" END) AS TipoDemonstrativo,

(CASE WHEN CdOrigemDemonstrativo=1 THEN "Serasa"
      WHEN CdOrigemDemonstrativo=2 THEN "Serasa Alterado"
      WHEN CdOrigemDemonstrativo=3 THEN "BV" END) AS OrigemDemonstrativo

FROM BASE_COMPLETA AS t1;

QUIT;

	PROC SQL NOPRINT;
		CREATE TABLE BASE1_&AGRUP AS
			SELECT		t1.CdPessoaMonitoravel,

	RUN;

	PROC TRANSPOSE DATA= BASE2_&AGRUP.
		OUT= BASE_TRANSP_&AGRUP. (DROP= _NAME_ _LABEL_);
		BY	CdPessoaMonitoravel
			DtDemonstrativo
			DtCadastro
			AgrupamentoDemonstrativoMntr
			CdContaContabilMonitor
			CdMoeda
			OrigemDemonstrativo
			SegmentoMercado
			TipoDemonstrativo
			FlFinalizado
			FlPlanilhamento
			TpMoeda;
		VAR VrConta;
		ID Conta;
	RUN;

		CREATE TABLE BASE_TRANSP1_1&AGRUP. AS

					t2.NuCpfCnpjRaiz,
					t2.NuCpfCnpjComplemento,
					t2.NmPessoaMonitoravel,
					t2.CdGrupoEconomico
			FROM BASE_TRANSP_&AGRUP. AS t1

				LEFT JOIN TbPessoaMonitoravel   AS t2
				ON (t1.CdPessoaMonitoravel=t2.CdPessoaMonitoravel);
	QUIT;

	   DATA  BASE_TRANSP1_1&AGRUP.;

			FORMAT CNPJ BEST8.;
			FORMAT DtDemonstrativo DATETIME16.;
			FORMAT DtCadastro DATETIME16.;
	RUN;
