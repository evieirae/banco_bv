/* ======================================================================================*/

%CRIAVARDB;
PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE extracao_pta_PROPOSTA AS
SELECT * FROM connection to dbcon
(

	SELECT
			 year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) NuAnoMes
			,Proposta.NuProposta
			,DATE(Proposta.DtEntradaProposta) DtEntradaProposta
			,Proposta.CdLoja CdParceiroComercial
			,SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0) CdGrupoSubProduto
			,Proposta.CdSubProduto
			,SubProduto.DsSubProduto
	FROM DBA.TbProposta Proposta

		LEFT JOIN DBA.TbSubProduto SubProduto
			ON Proposta.CdSubProduto = SubProduto.CdSubProduto

		LEFT JOIN DBCRED.TbPropostaFinanceiro PropostaFinanceiro
			on Proposta.NuProposta = PropostaFinanceiro.NuProposta

	WHERE 1=1
	and year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) >= 201801
    /*and SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0)
		in (12006010,12027010,12079010,12078010,12006001,12028001,12079001,12078001)*/
ORDER BY Proposta.NuProposta
)
;RUN;

PROC SQL;
CREATE TABLE FiltroCVG AS
SELECT  DISTINCT
		Proposta.nuproposta,
/*		CASE WHEN GrupoProdutoMeta.NmGrupoProdutoMeta LIKE 'CVG%' THEN */
				CASE    WHEN compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%GRCONSUL%' OR ParceiroComercial.NmParceiroComercial LIKE '%COCKPIT%'
					AND TipoCanalProducao.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS')
						THEN 'GR CONSULTOR'

				    WHEN compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%CPGARANTIA%' OR compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%BVPARTICULAR%'
					 OR compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%BVBALCAO%'
					AND TipoCanalProducao.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS')
						THEN 'BALCÃO'

				    WHEN Proposta.CdParceiroComercial in (select distinct INPUT(Codigos,BEST.) from AUXL.EGV_DEPARAPARCEIROSDIGITAIS)
						THEN 'PARCEIRO DIGITAL'

					WHEN (Proposta.CdParceiroComercial = 407516 AND Proposta.DtEntradaProposta  BETWEEN '01JUL2020'd AND '31AUG2020'd)
                         OR (Proposta.CdParceiroComercial = 416382 AND Proposta.DtEntradaProposta  >= '01AUG2020'd)
                         THEN 'JUST'

					WHEN compress(ParceiroComercial.NmParceiroComercial,' ') LIKE '%CPGARANTMLKT%'
						OR ParceiroComercial.CdParceiroComercial IN(466807,418607,468271) THEN 'TELEMARKETING'

					WHEN Proposta.CdParceiroComercial in(472272,474057,474058)
						THEN 'BANCO DIGITAL'

					WHEN Proposta.CdParceiroComercial = 414643
						THEN 'MINHA BV'

					WHEN Proposta.CdParceiroComercial in(424584,452470,465399)
						THEN 'CVG ÁREA ABERTA SITE'

					WHEN TipoCanalProducao.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS','PROMOTORA','BV FINANCEIRA - OPERACAO ESPECIAL')
						 AND (ParceiroComercial.NmParceiroComercial NOT LIKE '%FINANZERO%' OR compress(ParceiroComercial.NmParceiroComercial,' ') NOT LIKE '%QUEROFINANCIAR%')
                         AND (Proposta.CdParceiroComercial not in (407516) AND Proposta.DtEntradaProposta  BETWEEN '01JUL2020'd AND '31AUG2020'd)
                        THEN 'CANAL NEGOCIOS'

                    WHEN TipoCanalProducao.DsTipoCanalProducao IN ('MULTIMARCAS','CONCESSIONARIAS')
						THEN 'CANAL VEICULOS'

					WHEN TbFunilSimulacao_Solar.nuproposta is not null THEN UPPER(TbFunilSimulacao_Solar.Canal_Originação)   /*ORIGENS DO SOLAR*/

                    ELSE 'CANAL NEGOCIOS'
/*                    END ELSE 'OUTROS PRODUTOS'*/
		   END AS FiltroCVG,

		CASE
			WHEN Proposta.CdParceiroComercial in (select distinct INPUT(cdparceiro,BEST.) from AUXL.EGV_DEPARABALCAODIGITAL) THEN 'S'
		END AS EGV_FLBalcaoDigital,

		   	EGVBALCDIG.FILIAL AS EGV_FilialBalcaoDigital,
			EGVBALCDIG.NOME AS EGV_NomeBalcaoDIgital,

		CASE
			WHEN Proposta.CdParceiroComercial IN (418607,424584,471098,472245,414643,452470,475580,477503,471164,474057,461648,465399,477433,474058,467954,468906,461648,472272,472263,466807) THEN 'CX'
			 WHEN TipoCanalProducao.DsTipoCanalProducao IN ('MULTIMARCAS','CONCESSIONARIAS') THEN 'CANAL VEICULOS'
			 WHEN TbFunilSimulacao_Solar.nuproposta is not null THEN 'CANAL SOLAR'
			 ELSE 'CANAL NEGOCIOS'
		END AS CanalFinalizador

			,TODAY() format=date9. as DtAtualizacao
		    ,TIME() format= time5.  as HrAtualizacao

FROM extracao_pta_PROPOSTA Proposta

left join bases.TbTipoCanalProducaoProposta TipoCanalProducao
		on Proposta.NuProposta = TipoCanalProducao.NuProposta

left join tmp.TbParceiroComercial ParceiroComercial
		on  Proposta.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

left join basegrc.TbGrupoProdutoMeta GrupoProdutoMeta
		on  Proposta.CdGrupoSubProduto = GrupoProdutoMeta.CdGrupoSubProduto

left join TbFunilSimulacao_Solar as TbFunilSimulacao_Solar
		on TbFunilSimulacao_Solar.nuproposta = Proposta.nuproposta

LEFT JOIN AUXL.EGV_DEPARABALCAODIGITAL EGVBALCDIG
ON Proposta.CdParceiroComercial = INPUT(EGVBALCDIG.CDPARCEIRO,BEST.)

/*where GrupoProdutoMeta.NmGrupoProdutoMeta in ('CVG Leves','CVG Pesados')*/
/*and Proposta.NUproposta in (40260695, 40260728)*/
;run;
