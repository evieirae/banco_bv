%macro sazonalidade(base, vari, ano_fim);

	
	data &vari._des;
	set &base.;

		where anomes <= &ano_fim.12;

		keep data anomes &vari.;

	run;

	
	proc sql noprint;

		select min(&vari.) into :min_vari
		from &vari._des;

	quit;
	%put "&min_vari.";

	%let mode = ;
	data _null_;
	set &vari._des;

		if %sysevalf(&min_vari.) LE 0 then call symput('mode',"MODE %str(=) ADD");

	run;
	%put "&mode.";

	
	proc x12 data = &vari._des date = &tempo. ;

	   var &vari.;
	   x11 &mode.;
	   output out = out a1 d11;

	run;

	
	data &vari._des;
	merge &vari._des out;

		by &tempo.;
		drop &vari._a1;

	run;

	
	data &vari._des;
	set &vari._des;

		rename &vari._d11 = z_&vari. ;

	run;

%mend;










%macro amostra(difer, vari, n_cenarios);

	
	data boot_in;
	set boot_in;

		where anomes >= (&ano. - (&anos_amostra. -1) ) * 100;

		keep dif;

	run;

	
	data vme_amos_&vari.;
	set _NULL_;

		n_amostra = 0;
		delta_1a = 0;
		delta_2a = 0;
		delta_3a = 0;

	run;

	
	%do i = 1 %to &n_cenarios. %by 1;

		
		proc surveyselect data = boot_in

			method=urs sampsize=3 out=boot_out outhits noprint;

		run;

		
		data boot_out;
		set boot_out;

			retain delta;

			%if &difer. = S %then %do;

				delta = sum(delta, dif);

			%end;
			%else %do;

				delta = delta * dif;
				if _N_ = 1 then delta = dif;

			%end;

		run;

		
		
		data boot_out;
		set boot_out;

			if _N_ = 1 then call symput("d_1a", delta);
			if _N_ = 2 then call symput("d_2a", delta);
			if _N_ = 3 then call symput("d_3a", delta);

		run;

		
		proc sql;

			insert into vme_amos_&vari.
			set n_amostra = &i.,
				delta_1a = &d_1a.,
				delta_2a = &d_2a.,
				delta_3a = &d_3a.;

		quit;

		%symdel d_1a d_2a d_3a;

	%end;

proc sql;
create table
TE_AM.vme_amos_&vari. as
select *
from vme_amos_&vari.;

quit;

%mend;

%mend;












%macro gera_curvas(vari, nome, perc1, perc2, perc3, increm, adian);

	
	data _NULL_;
	set vme_hist_des;

		if anomes = (&ano. * 100 + 12) then call symput("pt_0", &vari.);

	run;
	%put pt0 = &pt_0.;

	
	data _NULL_;
	set vme_percentis;

		if vari = "&vari." and sum(&perc1., -0.000001) <= percentil <= sum(&perc1., 0.000001) then call symput("dt_1", delta_1a);
		if vari = "&vari." and sum(&perc2., -0.000001) <= percentil <= sum(&perc2., 0.000001) then call symput("dt_2", delta_2a);
		if vari = "&vari." and sum(&perc3., -0.000001) <= percentil <= sum(&perc3., 0.000001) then call symput("dt_3", delta_3a);

	run;
	%put deltas = &dt_1. &dt_2. &dt_3.;

	
	
	data curvas;
	set vme_curvas;

		
		%if &adian. = S %then %do;

			if anomes = ((&ano.) * 100 + 12) then delta = 0;

			else if anomes <= ((&ano. + 1) * 100 + 6) then

				%if &increm. = D %then delta = ((&dt_1. - 1) - &pt_0. ) / 6;
				%if &increm. = S %then delta = ((&pt_0. + &dt_1.) - &pt_0. ) / 6;
				%if &increm. = M %then delta = ((&pt_0. * &dt_1.) - &pt_0. ) / 6; ;

				%if &increm. = M %then delta = ((&pt_0. * &dt_3.) - (&pt_0. * &dt_2.)) / 24; ;

		%end;
		
		%else %do;

			if anomes = ((&ano.) * 100 + 12) then delta = 0;

			else if anomes <= ((&ano. + 1) * 100 + 12) then

				%if &increm. = D %then delta = ((&dt_1. - 1) - &pt_0. ) / 12;
				%if &increm. = S %then delta = ((&pt_0. + &dt_1.) - &pt_0. ) / 12;
				%if &increm. = M %then delta = ((&pt_0. * &dt_1.) - &pt_0. ) / 12; ;

			else if anomes <= ((&ano. + 2) * 100 + 12) then

				%if &increm. = D %then delta = ((&dt_2. / &dt_1. - 1) - (&dt_1. - 1)) / 12;
				%if &increm. = S %then delta = ((&pt_0. + &dt_2.) - (&pt_0. + &dt_1.)) / 12;
				%if &increm. = M %then delta = ((&pt_0. * &dt_2.) - (&pt_0. * &dt_1.)) / 12; ;

			else if anomes <= ((&ano. + 3) * 100 + 12) then

				%if &increm. = D %then delta = ((&dt_3. / &dt_2. - 1) - (&dt_2. / &dt_1. - 1)) / 12;

		%end;

	run;
