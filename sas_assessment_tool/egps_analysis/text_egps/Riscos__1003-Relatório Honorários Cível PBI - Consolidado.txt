proc sql;
    create table work.tblToday as
    select
        Today                                  as Today format=date.,
        intnx('month', Today, 0, 'B')          as DataInicio format=date.,
        intnx('month', Today, 0, 'E')          as DataFim format=date.
    from work.tblToday
    where 1=1;
run;

proc sql;
    create table work.tblToday as
    select
        a.Today,
        a.DataInicio                           as DtIniMesAtual,
        a.DataFim                              as DtFimMesAtual,
        year(max(a.DataInicio)) * 100 +
        month(max(a.DataInicio))              as DtBaseAnoMesProcessamentoMes,
        b.DtBaseAnoMesFechamento,
        'Mês Atual'                            as NmPeriodoMesAtual,
        b.DtRefIni                             as DtIniFechamento format=date.,
        b.DtRefFim                             as DtFimFechamento format=date.,
        case
            when b.FlFechamento = 'Sim' then 1
            else 0
        end                                    as FlFechamento,
        case
            when b.FlFechamento = 'Não' then ''
            else 'Fechamento'
        end                                    as NmPeriodoFechamento
    from work.tblToday as a
    left join work.tblParametroDtFechamento as b
        on a.Today = b.DtProcessamento
    where 1=1;

proc sql;
CREATE TABLE WORK.PAGAMENTOSHONORARIOS AS
SELECT

C.CdCausa
,A.DtLote
,A.NmRecurso
,A.TpLancamento
,A.VlPagamento
,A.CdPgtoHonorarioLoteHistGrupo
,B.NmPgtoHonorarioLoteHistGrupo
,A.NmPgtoHonorarioLoteHist
,A.SqPgtoHonorarioLoteHist
,intnx('day' , A.DtLote , 21916 ) as DtLote2C,case when A.CdPgtoHonorarioLoteHistGrupo = 1 then 1 else 0 end as FlEncerramento

,case when A.CdPgtoHonorarioLoteHistGrupo = 2 then 1 else 0 end as FlContestacao
,case when A.CdPgtoHonorarioLoteHistGrupo = 3 then 1 else 0 end as FlExito
,case when A.CdPgtoHonorarioLoteHistGrupo = 4 then 1 else 0 end as FlRessarcimentoErroAssessoria
,case when A.CdPgtoHonorarioLoteHistGrupo = 5 then 1 else 0 end as FlDescontoPgtoIndevido
,case when A.CdPgtoHonorarioLoteHistGrupo = 6 then 1 else 0 end as FlNaoDefinico
,case when A.CdPgtoHonorarioLoteHistGrupo = 7 then 1 else 0 end as FlTaxas

FROM WORK.CAUSAS AS C
LEFT JOIN YRIBEIRO.TBLPGTOHONORARIOLOTEHISTCAUSA AS A
ON C.CDCAUSA = A.CDCAUSA
LEFT JOIN work.tblPgtoHonLoteHistGrupo AS B
ON A.CdPgtoHonorarioLoteHistGrupo = B.CdPgtoHonorarioLoteHistGrupo

WHERE 1=1
;RUN;

PROC SQL;
CREATE TABLE WORK.FlResumo AS
SELECT
t1.CdCausa
, (MAX(t1.FlEncerramento)) 					AS FlEncerramento
, (MAX(t1.FlContestacao)) 					AS FlContestacao
, (MAX(t1.FlExito)) 						AS FlExito
, (MAX(t1.FlRessarcimentoErroAssessoria)) 	AS FlRessarcimentoErroAssessoria
, (MAX(t1.FlDescontoPgtoIndevido)) 			AS FlDescontoPgtoIndevido
, (MAX(t1.FlNaoDefinico)) 					AS FlNaoDefinico
, (MAX(t1.FlTaxas)) 						AS FlTaxas

FROM WORK.PAGAMENTOSHONORARIOS t1
GROUP BY t1.CdCausa
;RUN;
