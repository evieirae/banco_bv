%include '/infogrc/utils/all-includes.sas';
LIBNAME CMBV '/infogrc/Campanha_MissaoBV';
LIBNAME CMBV_BKP '/infogrc/Campanha_MissaoBV/Bases_BKP';

Data TbSimulacoes (Keep=NuAnoMes Produto Canal dsloginn1 SistemaInicial FlSimuValida FlSimuValidaEfet FlPAp Flconversao FlPropostaOriginada FlPropostaEfetivada Mensageria);
Set convsimu.TbSimulacoes (Where=(NuAnoMes between 202510 AND 202512))
;run;

Data TbSimulacoes_Final;
Set TbSimulacoes (Where=(Produto IN('Leves')
						 AND Canal IN ('MULTIMARCAS','CONCESSIONARIAS','LOJA CANAL NEGOCIOS')
						 AND Mensageria in ('0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%','7. Plano Aprovado',
							 				'0. Aprovado','1. Aprovado com rastreador','2. Reanálise psicométrico','0. Aprovado','1. Alta chance','2. Tem chance')
						 AND SistemaInicial = 'PPAR'
						 and FlSimuValidaEfet = 1
						 and FlPAp = 'Sim'
))
;run;

proc sql;
create table TbSimulacaoResumo as
select distinct
nuanomes,
dsloginn1,
Sum(Flconversao) as Convertidas,
Sum(FlPropostaEfetivada) as FlPropostaEfetivada
/*sum(FlPropostaOriginada) as FlPropostaOriginada*/
from TbSimulacoes_Final
where  1=1
group by 1,2
;run;

PROC SQL;
CREATE TABLE MAX_DATA AS
