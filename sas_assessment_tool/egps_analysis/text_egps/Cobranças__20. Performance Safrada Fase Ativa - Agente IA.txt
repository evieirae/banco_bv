%macro resumo(vars_num=, vars_nivel=, data=OUT_RANKED, target=performance, out=RESUMO_FINAL_&PRODUTO2.);
    %local i var rankvar n1 n2;

    
    %let n1 = %sysfunc(countw(&vars_num));
    %let n2 = %sysfunc(countw(&vars_nivel));
    %if &n1 ne &n2 %then %do;
        %put ERRO: As listas VARS_NUM e VARS_NIVEL devem ter o mesmo número de variáveis!;
        %return;
    %end;

    
    %if %sysfunc(exist(&out)) %then %do;
        proc datasets lib=work nolist; delete &out; quit;
    %end;

    %do i = 1 %to &n1;
        %let var     = %scan(&vars_num,&i);
        %let rankvar = %scan(&vars_nivel,&i);

        proc sql;
            create table resumo_&rankvar as
            select "&var" as Variavel length=32,
                   &rankvar as Nivel,
                   min(&var) as Min format=best12.,
                   max(&var) as Max format=best12.,
                   count(*) as Qtde_Contratos,
                   sum(&target) as Qtde_Pagos,
                   calculated Qtde_Pagos / calculated Qtde_Contratos as Performance format=percent8.2
            from &data
            group by &rankvar
            order by &rankvar;
        quit;

        %if &i = 1 %then %do;
            data &out; set resumo_&rankvar; run;
        %end;
        %else %do;
            proc append base=&out data=resumo_&rankvar force; run;
        %end;

        proc datasets lib=work nolist; delete resumo_&rankvar; quit;
    %end;
%mend;



%resumo(vars_num=&vars_num, vars_nivel=&vars_nivel, data=OUT_RANKED_&PRODUTO2., target=performance, out=RESUMO_FINAL_&PRODUTO2.);

PROC SQL;
CREATE TABLE DIAS_COMPARAVEIS AS
SELECT DISTINCT
DT_DISPONIBILIZACAO
,NR_DIAS_ATRASO
,SUM(QTDE) AS SUM_QTDE
,SUM(CASE WHEN GRANULAR_PUBLICO = "Agente IA" THEN QTDE ELSE 0 END) AS QTD_IA
,SUM(CASE WHEN GRANULAR_PUBLICO = "PA Fixa" THEN QTDE ELSE 0 END) AS QTD_PA_FIXA
,CASE WHEN CALCULATED QTD_IA > 10 AND CALCULATED QTD_PA_FIXA > 10 THEN 1 ELSE 0 END AS FL_DIAS_COMPARAVEIS
FROM CUBO_FIT
GROUP BY
DT_DISPONIBILIZACAO
,NR_DIAS_ATRASO
;QUIT;
