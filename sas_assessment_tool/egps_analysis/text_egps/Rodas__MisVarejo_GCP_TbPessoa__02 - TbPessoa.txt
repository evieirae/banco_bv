PROC SQL;
CONNECT TO bigquery AS perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'

      schema='bv-pdbd-prd' READ_MODE=storage);

   CREATE TABLE TbPessoa AS

   SELECT

	IdPessoa AS NuCpfCnpj
	,NmPessoa
	,catx('',NuDdd,NuTelefone) as NuTelefone1
	,catx('',NuDdd2,NuTelefone2) as NuTelefone2
	,catx('',NuDdd3,NuTelefone3) as NuTelefone3
	,DsEmail
	,DsEmail2
	,DtNascimento FORMAT=DATE9.

	FROM connection to perm

     	(
			SELECT
				  IdPessoa,
			      NmPessoa,
			      NuDdd,
			      NuTelefone,
			      NuDdd2,
			      NuTelefone2,
			      NuDdd3,
			      NuTelefone3,
			      DsEmail,
			      DsEmail2,
			      DATE(DtNascimento) AS DtNascimento

			FROM bv-pdbd-prd.dvry_clie_crm.tbpessoa
			WHERE NuTelefone <> '') ;

   DISCONNECT FROM perm
;QUIT;

data bases.TBPESSOA;

LENGTH	 DsEmail	$100.;

format	 DsEmail	$100.;

LENGTH	 DsEmail2	$100.;
informat DsEmail2	$100.;
format	 DsEmail2	$100.;

LENGTH	 NuCpfCnpj	$14.;
informat NuCpfCnpj	$14.;

format	 NuTelefone1	$14.;

format	 NuTelefone2	$14.;

LENGTH	 NuTelefone3	$14.;

format	 NuTelefone3	$14.;
