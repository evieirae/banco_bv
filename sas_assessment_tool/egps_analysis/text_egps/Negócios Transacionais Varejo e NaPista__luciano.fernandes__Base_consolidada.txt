proc sql;
create table basefase_2 as
select NuProposta
		,max(StProposta) as StProposta
		,max(DtPrimeiroVencimento) as DtPrimeiroVencimento /*format date9.*/
from basefase_1
where NuProposta>0
group by NuProposta

;
quit;

proc sql;
create table basefase_tratada3 as
select a.*
		,b.*
from basefase_3 as a
left join basefase_2 as b
on a.NuProposta=b.NuProposta
where StProposta = "Aprovada e Paga"
;
quit;
