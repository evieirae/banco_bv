PROC SQL;
CREATE TABLE MSG AS SELECT
A.CdSimulacaoInicial,
A.NuProposta,
A.DtSimulacaoInicial,
A.Mensageria AS Mensageria,
CASE WHEN A.ProbabilidadeAprovacao <= 0.5 THEN 'ZERO'
     WHEN A.ProbabilidadeAprovacao <= 6.0 THEN 'ATÉ 6,0'
	  WHEN A.ProbabilidadeAprovacao <= 7.3 THEN '6,25-7,25'
	  WHEN A.ProbabilidadeAprovacao <= 7.9 THEN '7,5-7,75'
	  WHEN A.ProbabilidadeAprovacao <= 8.5 THEN '8,25-8,5'
	  WHEN A.ProbabilidadeAprovacao <= 9.8 THEN '8,75-9,75'
	  WHEN A.ProbabilidadeAprovacao <= 10 THEN '10'
	  ELSE 'ZERO' END AS AGRUPAMENTO_RTG_SIMPLES,

CASE WHEN A.Mensageria in ('0. Aprovado','0. Aprovado 100%','1. Alta chance','1. Muita chance >85%','2. Grande chance >70%','2. Tem chance','3. Boa chance >50%','4. Tem chance >0%','7. Plano Aprovado')
and Mensageria2 not in ('3. Recusado', '4. Recusado Reversível') then 'Mensageria Positiva'
WHEN A.Mensageria  in ('3. Recusado','6. Recusado','6. Plano Recusado') or
A.Mensageria in ('0. Aprovado','0. Aprovado 100%','1. Alta chance','1. Muita chance >85%','2. Grande chance >70%','2. Tem chance','3. Boa chance >50%','4. Tem chance >0%','7. Plano Aprovado') and Mensageria2 in ('3. Recusado', '4. Recusado Reversível') then 'Recusado'
else 'Demais Msgs'
end as AGRUPAMENTO_MSG,

CASE WHEN A.EntradaMinima in ('00% a 4,99%','05% a 9,99%','00% a 9,99%')   THEN 'MÍNIMA ATÉ 10%'
     WHEN A.EntradaMinima in ('10% a 14,99%','15% a 19,99%','10% a 19,99%') THEN 'MÍNIMA ATÉ 20%'
	 WHEN A.EntradaMinima in ('20% a 24,99%','25% a 29,99%','20% a 29,99%') THEN 'MÍNIMA ATÉ 30%'
     WHEN A.EntradaMinima in ('30% a 34,99%','35% a 39,99%','30% a 39,99%') THEN 'MÍNIMA ATÉ 40%'
	 WHEN A.EntradaMinima in ('40% a 44,99%','45% a 49,99%','40% a 49,99%') THEN 'MÍNIMA ATÉ 50%'
	 WHEN A.EntradaMinima in ('50% a 54,99%','55% a 59,99%','60% a 64,99%','65% a 69,99%'
                              '70% a 74,99%','75% a 79,99%','80% a 84,99%'
                              '85% a 89,99%','90% a 94,99%','95% a 99,99%','50% a 100%') THEN 'MÍNIMA > 50%'
	 ELSE 'Desconhecido' END AS FAIXA_ENTRADA_MINIMA,
FlPap,
CASE WHEN A.CdParceiroComercial = D.CdParceiroComercial then 'AutoShopping' else 'Demais' end as FG_SHOPPING

from PRICING.Mensageria AS A
LEFT JOIN BASEGRC.TBPARCEIROAUTOSHOPPING AS D ON A.CdParceiroComercial=D.CdParceiroComercial
group by 1,2,3,4,5,6,7,8,9;
quit;

PROC SQL;
CREATE TABLE EFEITO_CANCELADAS AS SELECT
NuProposta,
SUM(CASE WHEN A.DtStatusProposta <= D0 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD0CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD0_AJUST,
SUM(CASE WHEN A.DtStatusProposta <= D3 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD3CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD3CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD0CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD0_AJUST

FROM QTPROPOSTAS AS A
LEFT JOIN TBD0 AS F ON A.DtEntradaProposta=F.DtTempo
LEFT JOIN TBD3 AS H ON A.DtEntradaProposta=H.DtTempo

GROUP BY 1;
QUIT;
