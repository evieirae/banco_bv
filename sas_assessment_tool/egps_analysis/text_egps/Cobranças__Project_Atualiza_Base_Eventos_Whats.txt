, A.DDD
, A.Telefone
, A.Telefone_Completo
, COUNT(DISTINCT A.Dt_Evento) AS Qtd_Dias_Erro
, Min(A.Dt_Evento) Format = yymmdd10. AS Min_Dt_Evento
, Max(A.Dt_Evento) Format = yymmdd10. AS Max_Dt_Evento
FROM WORK.INSUCESSO AS A
WHERE COMPRESS(A.NUCPFCNPJ) IS NOT MISSING
AND COMPRESS(A.NUCPFCNPJ) NOT IN ('00000000000','00000000000000','.','          .')
GROUP BY A.NUCPFCNPJ, A.DDD, A.Telefone, A.Telefone_Completo
;QUIT;




PROC SQL;
CREATE TABLE WORK.Insucesso_01 AS
SELECT DISTINCT
  A.NUCPFCNPJ
, A.DDD
, A.Telefone
, A.Telefone_Completo
, A.Qtd_Dias_Erro
, A.Min_Dt_Evento
, A.Max_Dt_Evento
, MAX(CASE
	WHEN A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo AND B.Dt_CPC > A.Max_Dt_Evento THEN 1
	WHEN A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo AND B.Dt_CPC BETWEEN A.Min_Dt_Evento AND A.Max_Dt_Evento THEN 2
	WHEN A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo AND A.Qtd_Dias_Erro <= 2 AND B.Dt_CPC < A.Min_Dt_Evento THEN 3
	WHEN A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo AND A.Qtd_Dias_Erro >= 3 AND B.Dt_CPC < A.Min_Dt_Evento THEN 4
	WHEN A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo THEN 5
	WHEN A.Qtd_Dias_Erro = 1 				THEN 94
	WHEN A.Qtd_Dias_Erro BETWEEN 2 AND 3 	THEN 95
	WHEN A.Qtd_Dias_Erro BETWEEN 4 AND 6 	THEN 96
	WHEN A.Qtd_Dias_Erro BETWEEN 7 AND 10 	THEN 97
	WHEN A.Qtd_Dias_Erro > 10 				THEN 98
ELSE 99 END) AS IdStatusTelWhats
FROM WORK.Insucesso_00 AS A
LEFT JOIN ( SELECT DISTINCT B.Dt_Evento As Dt_CPC, B.NUCPFCNPJ, B.NuContrato, B.Telefone_Completo
			FROM WORK.CPC AS B
			WHERE COMPRESS(B.NUCPFCNPJ) IS NOT MISSING
			AND COMPRESS(B.NUCPFCNPJ) NOT IN ('00000000000','00000000000000','.','          .')
			) AS B ON A.NuCPFCNPJ = B.NUCPFCNPJ AND A.Telefone_Completo = B.Telefone_Completo
Group By A.NUCPFCNPJ, A.DDD, A.Telefone, A.Telefone_Completo, A.Qtd_Dias_Erro, A.Min_Dt_Evento, A.Max_Dt_Evento
;QUIT;




PROC SQL;
CREATE TABLE WORK.Insucesso_02 AS
SELECT DISTINCT
  TODAY()  Format = yymmdd10. AS DT_REF
, DATETIME() Format = Datetime. AS Dt_Atualizacao
, A.*
, CASE
	WHEN A.IdStatusTelWhats =  1 THEN '01.CPC após última Falha'
	WHEN A.IdStatusTelWhats =  2 THEN '02.CPC durante período de Falha(s)'
	WHEN A.IdStatusTelWhats =  3 THEN '03.CPC anterior a até 2 falhas'
	WHEN A.IdStatusTelWhats =  4 THEN '04.CPC anterior a 3 ou mais falhas'
	WHEN A.IdStatusTelWhats =  5 THEN '05.CPC fora de parametros'
	WHEN A.IdStatusTelWhats = 99 THEN '99.Verificar Regra'
	WHEN A.IdStatusTelWhats = 94 THEN '94.Apenas 1 falha no período'
	WHEN A.IdStatusTelWhats = 95 THEN '95.De 2 a 3 falhas no período'
	WHEN A.IdStatusTelWhats = 96 THEN '96.De 4 a 6 falhas no período'
	WHEN A.IdStatusTelWhats = 97 THEN '97.De 7 a 10 falhas no período'
	WHEN A.IdStatusTelWhats = 98 THEN '98.Mais de 10 Falhas no período'
ELSE '99.Verificar Regra' END AS DescStatusTelWhats
, CASE WHEN B.Dias_Bloqueio > 0 THEN 1 ELSE 0 END AS Flag_Bloqueio
, CASE WHEN B.Dias_Bloqueio > 0 AND (A.Max_Dt_Evento + B.Dias_Bloqueio) >= TODAY() THEN 1 ELSE 0 END AS Flag_BloqueioAtivo
, B.Dias_Bloqueio
, CASE
	WHEN C.Dt_Expiracao_Bloqueio IS NOT MISSING AND C.Dt_Expiracao_Bloqueio > TODAY() THEN C.Dt_Expiracao_Bloqueio 
	WHEN B.Dias_Bloqueio > 0 THEN (A.Max_Dt_Evento + B.Dias_Bloqueio) ELSE .
	END Format = yymmdd10. AS Dt_Expiracao_Bloqueio
FROM WORK.Insucesso_01 AS A
LEFT JOIN TCONFIG.CONFIG_QUARENTENA AS B ON A.IdStatusTelWhats = B.ID_Bloqueio AND B.Canal = 'WhatsApp'
LEFT JOIN CONTATOS.AUX_Quarentena_Tel AS C ON A.NUCPFCNPJ = C.NUCPFCNPJ AND A.Telefone_Completo = C.Telefone_Completo
;QUIT;
