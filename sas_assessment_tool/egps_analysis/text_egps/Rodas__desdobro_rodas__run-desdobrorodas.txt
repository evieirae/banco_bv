/*
cria tabela com a identificação do(s) favorecido(s) do pagamento dos contratos
cada favorecido tem seu próprio registro, portanto, no caso de desdobro, a proposta é duplicada
*/

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE CONTRATO AS SELECT * FROM CONNECTION TO DBCON (
	SELECT
		  PROP.NuProposta
		, PROP.NuContratoGestao					AS NuContrato
		, CONVERT(DATE,PROP.DtEntradaProposta)	AS DtEntradaProposta
		, PESSOA.TpPessoa
		, PROP.CdLoja
		, ENDER.CdUf							AS UfProponente
		, CONVERT(DATE,PGTO.DtPagamento) 		AS DtPagamento
		, PGTO.VrLiberado 						AS VrLiberado
		, PESSOAPGTO.NuCpfCnpj
	FROM 		DBCRED.dbo.TbProposta PROP
	LEFT JOIN	DBCOR.dbo.TbSubProduto PROD
		ON PROD.CdSubProduto = PROP.CdSubProduto
	LEFT JOIN   DBCRED.dbo.TbPessoaSobAnalise PESSOA
		ON PROP.Nuproposta = PESSOA.Nuproposta
		AND PESSOA.CdPapel = 1
	LEFT JOIN	DBCRED.dbo.TbEnderecoTbPessoaSobAnalise ENDER
		ON PESSOA.CdPessoaSobAnalise = ENDER.CdPessoaSobAnalise
		AND ENDER.CdTipoEndereco = 1
	LEFT JOIN 	DBPAGAMENTO.dbo.TbPagamento PGTO
		ON PGTO.NuContratoGestao = PROP.NuContratoGestao
	LEFT JOIN 	DBPAGAMENTO.dbo.TbInformacaoBancariaPagamento INFOPGTO
		ON INFOPGTO.CdInformacaoBancariaPagto = PGTO.CdInforBancariaPagtoFavorecido
	LEFT JOIN	DBPAGAMENTO.dbo.TbPessoaPagamento PESSOAPGTO
		ON PESSOAPGTO.CdPessoaPagamento = INFOPGTO.CdPessoaPagamento
	WHERE	PROD.CdProduto = 12
	AND 	PGTO.DtPagamento >= '20210101'
	AND		PESSOAPGTO.NmNome <> ''
)
;QUIT;

PROC SORT DATA=CONTRATO; BY NuContrato; RUN;

DATA DESDOBRO_1; SET CONTRATO; BY NuContrato;
/*remove os contratos com favorecidos únicos da tabela*/
IF First.Nucontrato AND Last.NuContrato THEN Delete;

/*identifica o tipo de pessoa favorecida*/
FORMAT NuCpfCnpjFavorecido $14.;
NuCpfCnpjFavorecido  = NuCpfCnpj;
IF LENGTH(NuCpfCnpjFavorecido) > 11 THEN TpPessoaFavorecido = 'J'; ELSE TpPessoaFavorecido = 'F';
DROP NuCpfCnpj NuCpfCnpjFavorecido ;

/*Cria flags dos tipos de favorecidos*/
IF TpPessoaFavorecido = 'J' THEN FlPessJur = 1;
IF TpPessoaFavorecido = 'F' THEN FlPessFis = 1;
DROP TpPessoaFavorecido;

RUN;

/*
cria base final de desdobro;
em alguns casos cada favorecido é pago em uma data, neste caso está sendo considerada a maior data;
cria o parâmetro TpPessoaDesdobro indicando o tipo das pessoas que receberam pagamento
*/

PROC SQL;
CREATE TABLE DESDOBRO_2 AS
	SELECT
/*		  NuProposta*/
		  NuContrato
/*		, DtEntradaProposta*/
		, TpPessoa
		, CdLoja
/*		, UfProponente*/
		, CASE
			WHEN MAX(FlPessJur) NE . AND MAX(FlPessFis) NE . THEN 'PJ-PF'
			WHEN MAX(FlPessJur) NE . AND MAX(FlPessFis) EQ . THEN 'PJ-PJ'
			WHEN MAX(FlPessJur) EQ . AND MAX(FlPessFis) NE . THEN 'PF-PF'
															 ELSE ''
		  END 				AS TpPessoaDesdobro
		, MAX(DtPagamento)	AS DtPagamento FORMAT DATE9.
		, SUM(VrLiberado) 	AS VrLiberado
	FROM DESDOBRO_1
	GROUP BY 1,2,3
	ORDER BY 1
;QUIT;

DATA DESDOBRO.CONTRATO_DESDOBRO; SET DESDOBRO_2; RUN;
