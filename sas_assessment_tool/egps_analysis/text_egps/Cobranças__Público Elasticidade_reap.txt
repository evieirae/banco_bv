PROC SQL;

      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
CREATE TABLE info_parcela AS SELECT
    A.*,
	b.dtrecebimento,
	B.VrPrincipalBaixado,
	B.VrDevidoTotal,
	B.VrRecebido,
	B.VrPrincipalDevido,
	B.VrPrincipalDevido - B.VrPrincipalBaixado as desconto_prin,

	case when VrRecebido > 0 then 1 else 0 end as fl_recebimento

FROM (select * from base_inicial2) AS A LEFT JOIN
CONNECTION TO perm (SELECT
    z.cd_contrato AS NuContratoGestao,
    z.sq_contrato_financeiro as SqContratoFinanceiro,
    z.dt_recebimento as dtrecebimento,
    SUM(z.vr_principal_baixado) AS VrPrincipalBaixado,
    SUM(z.vr_devido_total) AS VrDevidoTotal,
    SUM(z.vr_recebido) AS VrRecebido,
    SUM(z.vr_principal_devido) AS VrPrincipalDevido
FROM
    bv-pdbd-prd.dvry_core.tbparcelarecebimento AS z
LEFT JOIN
    bv-pdbd-prd.trusted_scc_fin.saj_processo AS b
ON
    z.cd_contrato = b.nucontratogestao
    AND z.sq_contrato_financeiro = b.sqcontratofinanceiro
WHERE
    z.cd_agente_recebedor NOT IN ('104', '105', '125', '152', '167')
    AND NOT z.fl_estornado
    AND z.vr_recebido <> 0.01
    AND (DATE(b.dt_asses) <= &dtexecucao. AND (b.dt_ret_asses IS NULL OR DATE(b.dt_ret_asses) > &dtexecucao.))
    AND (z.dt_recebimento < CAST(b.dt_ret_asses AS DATE) OR b.dt_ret_asses IS NULL)
GROUP BY
    z.cd_contrato,
    z.sq_contrato_financeiro,
    z.dt_recebimento) AS B
ON	a.NuContrato = B.NuContratoGestao and a.SqContratoFinanceiro = B.SqContratoFinanceiro;
QUIT;



proc sql;
create table pagamento as
select NuContrato,
       SqContratoFinanceiro,
	   fl_recebimento,

	   	case
		when CdModalidadeProduto in (1,3,6,8,25,27,38,41,45,56,26,46) 		Then 'LEVES'

		when CdModalidadeProduto in (17) 									Then 'PRIVADO'
		when CdModalidadeProduto in (18,32,34,37,39,42,43) 				 	Then 'PUBLICO'
		when CdModalidadeProduto in (2,19,20,21,22,23,24,29,30,31,44,57) 	Then 'CP'
		when CdModalidadeProduto in (4,10,11,12,13,14,15,50,51,52,53,54,55) Then 'CDC'
        ELSE 'OUTROS' 	END AS PRODUTO,

       	SUM(VrPrincipalBaixado) AS VrPrincipalBaixado,
	    sum(VrDevidoTotal) as VrDevidoTotal,
     	SUM(VrRecebido) AS VrRecebido,
    	SUM(VrPrincipalDevido) AS VrPrincipalDevido,
	    max(datepart(DtRecebimento)) as DtRecebimento format date9.,
		max(desconto_prin/VrPrincipalDevido) as max_desc

from info_parcela
group by 1,2,3,4;
