PROC SQL;
CREATE TABLE BD_FLAG AS

SELECT *,

CASE
WHEN 'DATA DA SUBSTITUIÇÃO'N IS NULL AND 'SUB-MOTIVO'N IN ('COBRANÇA VEXATÓRIA', 'AUSÊNCIA/DEMORA NA BAIXA DO PROTESTO/PROTESTO INDEVIDO') THEN '1 - Não Distribuido Para Cob.'
WHEN 'Advogado Substituto'N IN ('Leticia Noguchi (ela/dela)' ) AND 'SUB-MOTIVO'N IN ('COBRANÇA VEXATÓRIA', 'AUSÊNCIA/DEMORA NA BAIXA DO PROTESTO/PROTESTO INDEVIDO') THEN '2 - Enviado Para Contratação'
WHEN 'Advogado Subtituído'N IN ('Leticia Noguchi (ela/dela)') AND 'SUB-MOTIVO'N IN ('COBRANÇA VEXATÓRIA', 'AUSÊNCIA/DEMORA NA BAIXA DO PROTESTO/PROTESTO INDEVIDO') THEN '3 - Escritório Contratado'
ELSE 'Fora do Escopo' END AS FLAG_SITUACAO

FROM WORK.BD_CONTROLE
WHERE 'SUB-MOTIVO'N IN ('COBRANÇA VEXATÓRIA', 'AUSÊNCIA/DEMORA NA BAIXA DO PROTESTO/PROTESTO INDEVIDO')
ORDER BY PASTA

;RUN;

PROC SQL;

CREATE TABLE VOL_JUR AS

SELECT * FROM WORK.BD_FLAG

WHERE FLAG_SITUACAO IN ('1 - Não Distribuido Para Cob.')

;RUN;

PROC SQL;

CREATE TABLE VOL_CONTRATACAO AS

SELECT * FROM WORK.BD_FLAG

WHERE FLAG_SITUACAO IN ('2 - Enviado Para Contratação');
RUN;

PROC SQL;

CREATE TABLE VOL_CONTRATADO AS

SELECT * FROM WORK.BD_FLAG

WHERE FLAG_SITUACAO IN ('3 - Escritório Contratado')

;RUN;

PROC SQL;

CREATE TABLE VOL_SINTETICO1 AS

SELECT DISTINCT A.PASTA,
B.'Data da substituição'N AS DT_ENVIO_COB,
C.'Data da substituição'N AS DT_CONTRATACAO,
A.'SUB-MOTIVO'N AS SUBMOTIVO,

CASE
WHEN B.'Data da substituição'N IS NOT NULL AND C.'Data da substituição'N IS NOT NULL THEN '3.ESCRITÓRIO CONTRATADO'
WHEN B.'Data da substituição'N IS NOT NULL AND C.'Data da substituição'N IS NULL THEN '2.AGUARDANDO CONTRATAÇÃO'
WHEN B.'Data da substituição'N IS NULL AND C.'Data da substituição'N IS NULL THEN '1.AGUARDANDO ENVIO DO JURÍDICO'
ELSE 'NULL' END AS FLAG_SITUACAO

FROM WORK.BD_FLAG AS A
LEFT JOIN WORK.VOL_CONTRATACAO AS B
ON A.PASTA = B.PASTA

LEFT JOIN WORK.VOL_CONTRATADO AS C
ON A.PASTA = C.PASTA

;RUN;

DATA WORK.VOL_SINTETICO;
SET VOL_SINTETICO1;

IF MONTH(DT_ENVIO_COB) < 10 THEN SAFRA_ENVIO = COMPRESS(YEAR(DT_ENVIO_COB) || '0' || MONTH(DT_ENVIO_COB));
ELSE SAFRA_ENVIO = COMPRESS(YEAR(DT_ENVIO_COB) || MONTH(DT_ENVIO_COB));

IF MONTH(DT_CONTRATACAO) < 10 THEN SAFRA_CONTRATACAO = COMPRESS(YEAR(DT_CONTRATACAO) || '0' || MONTH(DT_CONTRATACAO));
ELSE SAFRA_CONTRATACAO = COMPRESS(YEAR(DT_CONTRATACAO) || MONTH(DT_CONTRATACAO));

SLA_CONTRATACAO = DATDIF(DT_ENVIO_COB,DT_CONTRATACAO,'ACT/ACT');
SLA_A_CONTRATAR = DATDIF(DT_ENVIO_COB,TODAY(),'ACT/ACT');

;RUN;

PROC SQL;

CREATE TABLE VOL_FINAL AS

SELECT *,

CASE
WHEN DT_ENVIO_COB IS NOT NULL AND DT_CONTRATACAO IS NOT NULL AND SLA_CONTRATACAO BETWEEN 0 AND 3 THEN '4.CONTRATADO | NO PRAZO'
WHEN DT_ENVIO_COB IS NOT NULL AND DT_CONTRATACAO IS NOT NULL AND SLA_CONTRATACAO >= 4 THEN '5.CONTRATADO | FORA PRAZO'
WHEN DT_ENVIO_COB IS NOT NULL AND DT_CONTRATACAO IS NULL AND TODAY() - DT_ENVIO_COB < 4 THEN '2.A CONTRATAR | NO PRAZO'
WHEN DT_ENVIO_COB IS NOT NULL AND DT_CONTRATACAO IS NULL AND TODAY() - DT_ENVIO_COB >= 4 THEN '3.A CONTRATAR | FORA DO PRAZO'
WHEN DT_ENVIO_COB IS NULL AND DT_CONTRATACAO IS NULL THEN '1.AGUARDANDO JURÍDICO'
ELSE 'NULL' END AS FLAG_RESULTADO_SLA

FROM VOL_SINTETICO

;RUN;
