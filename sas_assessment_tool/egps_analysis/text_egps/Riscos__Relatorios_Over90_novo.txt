proc sql;
create table base1 as
select AnoMes_Referencia
		%if &anomes. >202402 %then %do;
		,case when tipo_carteira = 'SMES' or tipo_carteira_gerencial = 'SMES' then 'PM/UM'
			else 'CORE' end as Segmento
		,case when tipo_carteira = 'SMES' or tipo_carteira_gerencial = 'SMES' then 'PM/UM'
			  when tipo_carteira in ('GPR/AE') then tipo_carteira_gerencial
			  when tipo_carteira in ('PROJECT FINANCE') then tipo_carteira_financas
			  when tipo_carteira in ('CORRETORA') then 'CORPORATE'
			  else tipo_carteira end as SegmentoCliente
		%end;
		%else %do;
		,case when tipo_carteira = 'SMES' then 'PM/UM'
			else 'CORE' end as Segmento
		,case when tipo_carteira = 'SMES' then 'PM/UM'
			  when tipo_carteira in ('PROJECT FINANCE') then tipo_carteira_financas
			  when tipo_carteira in ('CORRETORA') then 'CORPORATE'
			  else tipo_carteira end as SegmentoCliente
		%end;		,DsGrupoEconomicoRisco
		,NuObjetoNegociacao
		,sum(VrSaldo) as Saldo
		,max(NumDiasVencidos) as Atraso
		,case when calculated Atraso > 90 then 'Over90'
			  when calculated Atraso > 60 then 'Over60'
			  when calculated Atraso > 30 then 'Over30'
			  when calculated Atraso > 15 then 'Over15'
			  else 'Em dia' end as Fx_Atraso

from base_unica_ajustada
%if &anomes. >202505 %then %do;
	where DsTipoCarteira in ('REGULAR', 'OFFSHORE') and tipo_carteira NE 'CORRETORA' and DsGrupoEconomicoRisco NE 'CREDZ' and
	      DsGrupoProduto not in ("Operações de Garantias" "Hedge" "Mercado Bancário" "TVM") and DsProduto ne ("Stand-by")
%end;
%else %do;
	where DsTipoCarteira in ('REGULAR', 'OFFSHORE') and DsGrupoEconomicoRisco NE 'CREDZ' and
	      DsGrupoProduto not in ("Operações de Garantias" "Hedge" "Mercado Bancário" "TVM") and DsProduto ne ("Stand-by")
%end;
group by 1,2,3,4,5;

create table OVER_&ANOMES. as
select
		AnoMes_Referencia
		,Segmento
		
		,sum(Saldo) as Saldo 									format commax26.2
		,sum(case when Atraso>15 then Saldo end) as Over15		format commax26.2
		,sum(case when Atraso>30 then Saldo end) as Over30		format commax26.2
		,sum(case when Atraso>60 then Saldo end) as Over60		format commax26.2
		,sum(case when Atraso>90 then Saldo end) as Over90 		format commax26.2
		,calculated over15 / calculated saldo as percent_over15 format percent10.2
		,calculated over30 / calculated saldo as percent_over30 format percent10.2
		,calculated over60 / calculated saldo as percent_over60 format percent10.2
		,calculated over90 / calculated saldo as percent_over90 format percent10.2
from base1
group by 1,2;

create table OVER_GRUPOS_&ANOMES. as
select
		AnoMes_Referencia
		,Segmento
		,DsGrupoEconomicoRisco
		,sum(case when Atraso>90 then Saldo end) as Over90 		format commax26.2
		,sum(case when Atraso>60 then Saldo end) as Over60		format commax26.2
		,sum(case when Atraso>30 then Saldo end) as Over30		format commax26.2
		,sum(case when Atraso>15 then Saldo end) as Over15		format commax26.2

from base1
group by 1,2,3;
quit;




proc sql;

    select

        count(*)/1000 as total_operacoes
    from
        base1

    id segmento;

    var Saldo_Over_15 Saldo_Over_30 Saldo_Over_60 Saldo_Over_90 Carteira;
