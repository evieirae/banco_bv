PROC SQL;
CREATE TABLE WORK.PARCEIROSATIVOS AS
SELECT

A.CdParceiroComercial,
A.NuCnpjLoja,
A.DsAtividadeFinanciamento,
A.NmFilialC,
A.NmTipoParceiroComercial,
A.NmAutoShopping,
A.NmGrupoConcessionaria,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN A.NmGrupoConcessionaria
	WHEN A.NmAutoShopping <> '' THEN A.NmAutoShopping
	WHEN A.NmTipoParceiroComercial <> '' THEN A.NmTipoParceiroComercial
	ELSE 'Outros'
END) AS NmClassificacaoGrupoParceiro,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN 'GRANDES GRUPOS'
	WHEN A.NmAutoShopping <> '' THEN 'AUTO SHOPPING'
	WHEN A.NmTipoParceiroComercial <> '' THEN UPCASE(A.NmTipoParceiroComercial)
	ELSE 'OUTROS'
END) AS NmClassificacaoTipoGrupoParceiro,

A.NmParceiroComercial,
A.CdUf,
A.CdColaboradorC,
A.NmColaboradorC,
A.DsLoginN1C,
B.NmGrupoCargoColaborador,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.DtCadastro,
A.FlAtivo
FROM BASES.TBPARCEIROCOMERCIAL A

LEFT JOIN BASES.TBCOLABORADOR B ON B.DSLOGIN = A.DsLoginN1C AND B.CDCOLABORADOR = A.CdColaboradorC

WHERE
A.FlAtivo = 'S'
AND A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.DsAtividadeFinanciamento = 'VEÍCULO'

ORDER BY 1

;RUN;

PROC SQL;
CREATE TABLE BASE_NAPISTA AS
SELECT

A.NuAnoMes,
A.FlAtivo,
A.NmGrupoProduto,
A.CdParceiroComercial,

B.NmTipoParceiroComercial,
B.NmAutoShopping,
B.NmGrupoConcessionaria,
B.NmClassificacaoGrupoParceiro,
B.NmClassificacaoTipoGrupoParceiro,

A.ChaveUFCidade,
A.NmCidade,
A.CdUf,
A.QtBancos,
A.NuCnpjLoja,
A.FlPorteA,
A.CdColaboradorC,
B.NmColaboradorC,
B.NmGrupoCargoColaborador,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.NmTipoParceiroComercial,
sum(A.qtdcontratos) as qtdcontratos,
sum(A.VrProducao) as VrProducao,
sum(A.VrMargemContribuicao) as VrMargemContribuicao,
sum(A.qtgravamesplayers) as qtgravamesplayers,

A.Regiao,
A.FaixaBancos

LEFT JOIN WORK.PARCEIROSATIVOS B ON B.CdParceiroComercial = A.CdParceiroComercial

where b.NmGrupoCargoColaborador = 'Gerente Relacionamento Veiculos'

group by
A.NuAnoMes,
A.FlAtivo,
A.NmGrupoProduto,
A.CdParceiroComercial,
B.NmTipoParceiroComercial,
B.NmAutoShopping,
B.NmGrupoConcessionaria,
B.NmClassificacaoGrupoParceiro,
B.NmClassificacaoTipoGrupoParceiro,
A.ChaveUFCidade,
A.NmCidade,
A.CdUf,
A.QtBancos,
A.NuCnpjLoja,
A.FlPorteA,
A.CdColaboradorC,
B.NmColaboradorC,
B.NmGrupoCargoColaborador,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.NmTipoParceiroComercial,
A.PorteCidade,
A.Regiao,
A.FaixaBancos

order by NuAnoMes

;RUN;
