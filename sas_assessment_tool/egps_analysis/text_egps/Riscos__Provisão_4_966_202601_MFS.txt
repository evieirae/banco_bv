class cd_contrato;

	   sum= soma_pe_contrato soma_pi_contrato;

data base_pct;
merge EXTRACAO.BASE_ROTEIRO_GCON_&DtBase.(in=a) somas(in=b);
by cd_contrato;
if a;

length perc_pe perc_pi 8;

tot_pe = abs(coalesce(soma_pe_contrato,0));
tot_pi = abs(coalesce(soma_pi_contrato,0));
pe_linha = abs(coalesce(vr_provisao_perda_esperada_Final,0));
pi_linha = abs(coalesce(vr_provisao_perda_incorrida,0));

IF tot_pe > 0 and tot_pi > 0 then do;
	denom = tot_pe + tot_pi;
	IF pe_linha > 0 then perc_pe = round(pe_linha/denom, 1e-6);
	ELSE perc_pe = .;
	IF pi_linha > 0 then perc_pi = round(pi_linha/denom, 1e-6);
	ELSE perc_pi = .;
END;

ELSE IF tot_pi > 0 then do;
	IF pi_linha >0 then perc_pi = round (pi_linha/tot_pi, 1e-6);
	else perc_pi = .;
	perc_pe = .;
END;

ELSE IF tot_pe > 0 then do;
	IF pe_linha > 0 then perc_pe = round (pe_linha/tot_pe, 1e-6);
	ELSE perc_pe = .;
	perc_pi = .;
END;

ELSE DO;
	perc_pe = .; perc_pi = .;
END;

FORMAT perc_pe perc_pi 12.6;
