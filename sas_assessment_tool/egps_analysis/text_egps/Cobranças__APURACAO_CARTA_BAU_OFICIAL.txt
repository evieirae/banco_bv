%MACRO CICLO_SALDO_GCON;


%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA., %str(*)));


%do i = 1 %to &NUM_ELEMENTOS.;

    
    %let ref_tab = %scan(&LISTA., &i., *);

    
    DATA _NULL_;
        MES = INPUT(scan("&LISTA.", &i., "*"), yymmdd10.);

        %CONNECT_PERM;
        CREATE TABLE GCON_&ref_tab. AS
        SELECT
            A.*,
            "&REF2." AS REF
        FROM CONNECTION TO PERM (
            WITH QUERY_1 AS (
                SELECT
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,

                WHERE DATE(a.dtprocessamento) = PARSE_DATE('%Y-%m-%d', "&REF.")
                  AND a.dtprocessamentoliquidacao IS NULL

                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
    A.*,
    D.SALDO AS SALDO_M0,
    B.SALDO AS SALDO_M1,
    C.SALDO AS SALDO_M2,

    
    CASE
        WHEN A.TIPO_PGTO = 'SEM PAGAMENTO' THEN 0
        ELSE 1
    END AS FL_PAGAMENTO,

    
    CASE
        WHEN A.DT_PRIMEIRO_PGTO IS MISSING THEN -100
        ELSE (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA)
    END AS QT_DIAS_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DT_PRIMEIRO_PGTO IS MISSING THEN -100 ELSE (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 10 THEN '01. Até 10 dias'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 20 THEN '02. 11-20 dias'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 60 THEN '04. 31-60 dias'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DT_PRIMEIRO_PGTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN A.ATRASO_ELEGIBILIDADE BETWEEN 1 AND 30 THEN '1-30'
        WHEN A.ATRASO_ELEGIBILIDADE BETWEEN 31 AND 60 THEN '31-60'
        WHEN A.ATRASO_ELEGIBILIDADE BETWEEN 61 AND 90 THEN '61-90'
        ELSE 'OUTRA'
    END AS FAIXA_ATRASO_MARC,

    
    CASE
        WHEN TODAY() - A.DATA_VIGENCIA >= 60 THEN 'Maturada'
        ELSE 'Nao maturada'
    END AS FL_MATURACAO,

    PUT(A.DATA_VIGENCIA, YYMMN6.) AS SAFRA

FROM BASE_UNICA_FIM2 A

LEFT JOIN SISTEMA_TOTAL D ON A.NUCONTRATO = INPUT(D.NUCONTRATO,14.) AND A.DATA_M0 = D.REF
LEFT JOIN SISTEMA_TOTAL B ON A.NUCONTRATO = INPUT(B.NUCONTRATO,14.) AND A.DATA_M1 = B.REF
LEFT JOIN SISTEMA_TOTAL C ON A.NUCONTRATO = INPUT(C.NUCONTRATO,14.) AND A.DATA_M2 = C.REF
WHERE A.ATRASO_ELEGIBILIDADE BETWEEN 1 AND 90

;QUIT;

PROC SQL;
%CONNECT_PERM;

CREATE TABLE BASE_PAGAMENTOS_RAW AS
SELECT *
FROM CONNECTION TO perm
(
  SELECT
    CAST(z.CD_CONTRATO AS INT64) AS NUCONTRATO,
    z.CD_CONTRATO,
    z.SQ_CONTRATO_FINANCEIRO     AS SQCONTRATOFINANCEIRO,
    z.DT_RECEBIMENTO            AS DtRecebimento,
    z.VR_PRINCIPAL_DEVIDO,
    z.VR_PRINCIPAL_BAIXADO,
    z.Nu_Parcela,

    EXTRACT(YEAR FROM z.DT_RECEBIMENTO) * 100
  + EXTRACT(MONTH FROM z.DT_RECEBIMENTO) AS SafraAjuste,

    b.NuDiasAtraso              AS NuDiasAtraso_raw,
    DATE(b.DTPROCESSAMENTO)     AS DtProcessamento,

    DATE_DIFF(
      DATE(b.DTPROCESSAMENTO),
      DATE(z.DT_RECEBIMENTO),
      DAY
    ) AS diff

  FROM `bv-pdbd-prd.dvry_core.tbparcelarecebimento` z

  LEFT JOIN `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` b
    ON z.CD_CONTRATO = b.NUCONTRATO
   AND z.SQ_CONTRATO_FINANCEIRO = b.SQCONTRATOFINANCEIRO
   AND DATE(b.DTPROCESSAMENTO)
       BETWEEN DATE(z.DT_RECEBIMENTO) - 5
           AND DATE(z.DT_RECEBIMENTO) + 5

  WHERE DATE(z.DT_RECEBIMENTO)
        BETWEEN DATE(&dt_inicio.) AND DATE(&dt_fim.)
    AND CAST(z.VR_RECEBIDO AS FLOAT64) NOT IN (0, 0.01)
    AND CAST(z.FL_ESTORNADO AS STRING) <> 'S'
);

QUIT;

PROC SORT DATA=BASE_PAGAMENTOS_RAW;
  BY
    NUCONTRATO
    SQCONTRATOFINANCEIRO
    NU_PARCELA
    DtRecebimento
    diff
    DtProcessamento;
RUN;

PROC SQL;
CREATE TABLE BASE_PAGAMENTOS_&anomes. AS

  NUCONTRATO,
  SQCONTRATOFINANCEIRO,

  SafraAjuste,

  MAX(CASE WHEN diff BETWEEN -5 AND -1
           THEN NuDiasAtraso_raw END) AS NuDiasAtraso_PRE_Acao,

  MAX(CASE WHEN diff = 0
           THEN NuDiasAtraso_raw END) AS NuDiasAtraso_Dia_Acao,

  MIN(CASE WHEN diff BETWEEN 1 AND 5
           THEN NuDiasAtraso_raw END) AS NuDiasAtraso_POS_Acao

FROM BASE_PAGAMENTOS_RAW
GROUP BY
  NUCONTRATO,
  SQCONTRATOFINANCEIRO,
  DtRecebimento,

PROC SQL;
CREATE TABLE PRIMEIRO_PGTO1 AS
SELECT
	PUT(A.NUCONTRATO,Z14.) AS NUCONTRATO,
    A.SQCONTRATOFINANCEIRO,
    A.DATA_VIGENCIA,
    MIN(
        CASE WHEN B.dt_ultimo_recebimento > A.DATA_VIGENCIA
            THEN B.dt_ultimo_recebimento
        END) AS dt_primeiro_pgto FORMAT=DATE9.

FROM CARTA_CARTORARIA_TOTAL A

LEFT JOIN BASE_ULTIMO_RECEB B
ON A.nucontrato = INPUT(B.cd_contrato,14.)
	AND A.SQCONTRATOFINANCEIRO = B.sq_contrato_financeiro

GROUP BY
    A.nucontrato,
    A.SQCONTRATOFINANCEIRO,
    A.DATA_VIGENCIA
;QUIT;


PROC SQL;
CREATE TABLE TIPO_PGTO AS
SELECT
A.nucontrato,
A.SQCONTRATOFINANCEIRO,
A.DATA_VIGENCIA,
A.dt_primeiro_pgto,
B.qtd_parcelas_total,

SUM(CASE WHEN B.dt_vencimento < A.DATA_VIGENCIA 
	AND (
		B.dt_ultimo_recebimento IS MISSING
		OR B.dt_ultimo_recebimento > A.DATA_VIGENCIA)
        THEN 1
        ELSE 0 end) AS qtd_vencidas_abertas, 

SUM(CASE WHEN
	NOT MISSING(A.dt_primeiro_pgto)
	AND NOT MISSING(B.dt_ultimo_recebimento)
	AND B.dt_ultimo_recebimento = A.dt_primeiro_pgto
	THEN 1
	ELSE 0 end) AS qtd_pagas, 

SUM(CASE WHEN
	B.dt_ultimo_recebimento IS MISSING
	THEN 1
	ELSE 0
	END) AS qtd_parcelas_abertas 

FROM PRIMEIRO_PGTO1 A

LEFT JOIN BASE_CRUZADA B
ON input(A.nucontrato, 14.) = B.nucontrato
	AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

GROUP BY
A.nucontrato,
A.SQCONTRATOFINANCEIRO,
A.DATA_VIGENCIA,
A.dt_primeiro_pgto,
B.qtd_parcelas_total
;QUIT;


PROC SQL;
CREATE TABLE TIPO_PAGAMENTO_FIM AS
SELECT *,
CASE
WHEN qtd_pagas = 0 THEN 'SEM PAGAMENTO'
WHEN dt_primeiro_pgto > DATA_VIGENCIA AND qtd_parcelas_abertas = 0 THEN 'QUITACAO'
WHEN qtd_pagas < qtd_vencidas_abertas THEN 'PARCIAL'
WHEN qtd_pagas >= qtd_vencidas_abertas THEN 'ATUALIZACAO'
ELSE 'OUTRO'
END AS TIPO_PGTO

FROM TIPO_PGTO
;RUN;
