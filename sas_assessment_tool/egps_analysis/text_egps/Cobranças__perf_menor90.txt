PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=15);
CREATE TABLE reneg_menor90 AS
SELECT
	A.*,
	B.atraso_m0,
	B.sqcontratofinanceiro_m0,
	C.atraso_m1,
	C.sqcontratofinanceiro_m1,
	C.DtProcessamentoLiquidacao,
	B.SALDO_M0

FROM BASE_ELEG AS A

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			A.Nucontrato,
			MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro_m0,
			MAX(A.nudiasatraso) as atraso_m0,
			CASE
				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO_M0

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE DATE(A.dtprocessamento) = &fim_mes_m0.
		GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato
) AS B

ON A.nucontrato = INPUT(B.NUCONTRATO,BEST32.)

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			Nucontrato,
			max(sqcontratofinanceiro) as sqcontratofinanceiro_m1,
			max(nudiasatraso) as atraso_m1,
			max(DtProcessamentoLiquidacao) as DtProcessamentoLiquidacao

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) = &fim_mes_m1.
		group by nucontrato
) AS C
ON A.nucontrato = INPUT(C.NUCONTRATO,BEST32.);

QUIT;


DATA reneg_menor90_&anomes.;
SET reneg_menor90;
FORMAT RESULTADO $15.;
IF atraso_m1 = 0 AND sqcontratofinanceiro_m1 > sqcontratofinanceiro_m0 		THEN RESULTADO = "Reneg"; ELSE
IF DtProcessamentoLiquidacao NE .		 									THEN RESULTADO = "Quitacao"; ELSE
IF nudiasatraso >= atraso_m1 												THEN RESULTADO = "Pagamento";
ELSE RESULTADO = "Sem acao";
RUN;

PROC SORT DATA= reneg_menor90_&anomes. NODUPKEY;
BY NUCONTRATO;
RUN;
