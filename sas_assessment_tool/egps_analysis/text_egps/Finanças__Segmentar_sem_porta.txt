Create Table WORK.Sum_Ajustados as
Select
Max(DtReferencia) as DtReferencia format DATE9.,
"Ajustado" as Origem,
cdGrupoConciliacao,
Case
			When CdGrupoConciliacao = 1 then "Box"
			When CdGrupoConciliacao = 2 then "Câmbio"
			When CdGrupoConciliacao = 3 then "Empréstimos"
			When CdGrupoConciliacao = 4 then "Fiança"
			When CdGrupoConciliacao = 15 then "Notas Estrangeiras"
			When CdGrupoConciliacao = 7 then "Off Shore"
			When CdGrupoConciliacao = 8 then "Renda Fixa"
			When CdGrupoConciliacao = 9 then "Renda Variável - Bov"
			When CdGrupoConciliacao = 10 then "Renda Variável - Fut"
			When CdGrupoConciliacao = 11 then "Repasse"
			When CdGrupoConciliacao = 12 then "Swaps e NDFs"
			When CdGrupoConciliacao = 16 then "Títulos Descontados"
			else ""
		end as GrupoConciliacao,
(Sum(coalesce(VrResultadoTesourariaNotDif,0)) + Sum(coalesce(VrResultadoComercial,0)) + Sum(coalesce(VrResultadoAcionista,0))) as Resultado format COMMAX17.2

Group by DtReferencia, Origem, cdGrupoConciliacao, GrupoConciliacao;
