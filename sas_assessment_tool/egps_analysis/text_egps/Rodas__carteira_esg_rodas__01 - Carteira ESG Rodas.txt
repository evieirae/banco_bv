/* Chamar macros principais */
%CRIAVARDB;
%SAFRAESG();

options compress=yes;

/* INFORMAR QTD MESES PARA O DASH */
%let qtMeses = 90;

DATA _NULL_;
  /* Define parâmetros com base no dia atual */
  hoje = TODAY();
  hojeXM = INTNX('MONTH', hoje, ((&qtMeses. - 1) * -1), 'BEGINNING');

  /* Calcula variáveis */
  safraXM = (YEAR(hojeXM) * 100) + MONTH(hojeXM);
  anoXM = YEAR(hojeXM);
  anoAtual = YEAR(hoje);
  dtIniPta = (safraXM * 100) + 1;

  /* Declarar macro variáveis */
  CALL SYMPUT('safraXM', safraXM);
  CALL SYMPUT('anoXM', PUT(anoXM,4.));
  CALL SYMPUT('anoAtual', PUT(anoAtual,4.));
  CALL SYMPUT('dtIniPta', "'"||PUT(dtIniPta,8.)||"'");
RUN;

PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE PropostasESG AS
  SELECT * FROM connection TO dbcon
  (
    SELECT
      PRP.NuProposta,
      PRP.DtEntradaProposta,
      PRP.StProposta,
      PRP.NuContrato,
	  PRP.NuContratoGestao,
      GRE.CdModelo,
      GRE.CdMolicar,
      GRE.CdMarca,
      GRE.AAModelo,
      GRE.AAFabricacao,
      GRE.DsCategoria,
      MLC.CdModelo AS CdModelo_Molicar,
      MLC.DsCombustivel
    FROM
      DBA.TbProposta PRP
    INNER JOIN
      DBA.TbPropGarantiaMaqEquip GRE
      ON PRP.NuProposta = GRE.NuProposta
    LEFT JOIN
      DBA.TbMolicar MLC
      ON GRE.CdMolicar = MLC.CdMolicar
    WHERE /*nucontratogestao = 12014000311726 and*/
      (
        MLC.DsCombustivel LIKE '%(EE)%' /*ELETRICO*/
        OR MLC.DsCombustivel LIKE '%(GE)%' /*HIBRIDO*/
        OR MLC.DsCombustivel LIKE '%(DE)%'/*HIBRIDO*/
		OR MLC.DsCombustivel LIKE '%Gas./Eta./E%'/*HIBRIDO*/
      )
      AND PRP.DtEntradaProposta >= &DtIniPta.
    ORDER BY PRP.DtEntradaProposta);
RUN;

%put &DtIniPta;

PROC SORT DATA=PropostasESG DUPOUT=DUPLI OUT=PROPOSTAS_ESG NODUPKEY;
BY NUPROPOSTA
   DTENTRADAPROPOSTA
;
RUN;

PROC SQL;
CREATE TABLE BaseESG AS
  SELECT DISTINCT
    UPCASE(CdMarca) AS CdMarca,
    UPCASE(CdModelo) AS CdModelo,
    UPCASE(combustivel) AS Combustivel
  FROM AUXL.BASE_MODELOS_ESG_&SafraESG.;
RUN;

PROC SQL;
CREATE TABLE TbModeloMolicar AS
  SELECT
    PRP.*,
    ESG.combustivel
  FROM
    work.PROPOSTAS_ESG PRP
  LEFT JOIN
    BaseESG ESG
    ON ESG.CdModelo = UPCASE(PRP.CdModelo_Molicar)
  WHERE
    PRP.StProposta NOT IN ('C','CE')
  ORDER BY 2,1;
RUN;

PROC SQL; CREATE INDEX NuProposta ON TbModeloMolicar; QUIT;

PROC SQL;
CREATE TABLE BASE_ESG_ANALITICA AS
SELECT
A.*,

 CASE
 WHEN A.DsCombustivel IN ('(EE)')                        THEN 'Elétrico'
 WHEN A.DsCombustivel IN ('(GE)','(DE)','Gas./Eta./E') THEN 'Híbrido'
 WHEN A.DsCombustivel IN ('(AA)')                        THEN 'Etanol'
 WHEN A.DsCombustivel IN ('(GG)')                        THEN 'Gasolina'
 WHEN A.DsCombustivel IN ('(DD)')                       THEN 'Diesel'
 WHEN A.DsCombustivel IN ('Bio/Diesel')                   THEN 'Biodiesel'
 WHEN A.DsCombustivel IN ('(GN)')                       THEN 'Gás Natural'
 WHEN A.DsCombustivel IN ('(AGX)','(GX)','(AG)')       THEN 'FLEX'
 ELSE 'N/D' END AS TpCombustivel,

B.NuAnoMes as NuAnoMes_Prod,
B.DtContrato,
B.NmGrupoProdutoMeta,
B.NUCONTRATO as NUCONTRATO_PROD,
B.VRPRODUCAO
FROM tmp.TBPRODUCAO B

LEFT JOIN work.TbModeloMolicar A ON B.NUPROPOSTA = A.NUPROPOSTA

WHERE B.NUANOMES between 202101 and 202512 AND B.NmGrupoProdutoMeta in ('Leves', 'Motos', 'Pesados') AND A.DsCombustivel <> ''

;RUN;

PROC SQL;
CREATE TABLE BASE_ESG_DRE AS
SELECT

A.NuAnoMes_Prod,
A.DtContrato,
A.NmGrupoProdutoMeta,
A.TpCombustivel,
A.DsCombustivel,
A.NUCONTRATO_PROD,
A.VRPRODUCAO

FROM BASE_ESG_ANALITICA A

order by
A.NuAnoMes_Prod,
A.DtContrato,
A.NmGrupoProdutoMeta,
a.DsCombustivel,
A.TpCombustivel

;RUN;

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
A.NUCONTRATO_PROD AS NuContrato format BEST20.,
A.DtContrato format date9.,
a.NuAnoMes_Prod AS NuAnoMes,
a.NmGrupoProdutoMeta,
A.TpCombustivel,
A.DsCombustivel
FROM WORK.BASE_ESG_DRE A

;RUN;

DATA AUTOMATI.CONTRATOS_ESG;
SET WORK.BASE_FINAL;
;RUN;

PROC SQL;
CREATE TABLE BASE_SUMARIZADA AS
SELECT
NuAnoMes,
NmGrupoProdutoMeta,
TpCombustivel,

COUNT(A.NuContrato) AS QTD FORMAT COMMAX20.
FROM AUTOMATI.CONTRATOS_ESG A

GROUP BY 1,2,3
ORDER BY 1,2,3
;RUN;
