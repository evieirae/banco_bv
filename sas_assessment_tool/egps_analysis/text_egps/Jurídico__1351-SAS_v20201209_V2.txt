create table work.AndamentoF2 as
select

 a.Pasta
,a.NmAndamento
,a.DtAndamento
,CASE 	WHEN b.NmAndamento = 'CASO EM ANÁLISE PARA AJUIZAMENTO DE BA' and a.NmAndamento = 'IMPOSSIBILIDADE PARA O AJUIZAMENTO DE BA'          THEN 1
		WHEN b.NmAndamento = 'CASO EM ANÁLISE PARA AJUIZAMENTO DE BA' and a.NmAndamento = 'REVISIONAIS - BUSCA E APREENSÃO - CASO MONITORADO' THEN 2

      	WHEN (b.NmAndamento = 'CASO EM ANÁLISE PARA AJUIZAMENTO DE BA')
       	and ((a.NmAndamento <> 'REVISIONAIS - BUSCA E APREENSÃO - CASO MONITORADO')
		or (a.NmAndamento <> 'IMPOSSIBILIDADE PARA O AJUIZAMENTO DE BA'))                                                             THEN 3

	  ELSE 4 END AS Peso

from dbep.TBLPROCESSOANDAMENTO a
left join work.AndamentoF1 b
on a.Pasta = b.Pasta

where 1=1

and a.CdSistema = 'p'
and a.NmAndamento in (
 	'CASO EM ANÁLISE PARA AJUIZAMENTO DE BA'
	,'IMPOSSIBILIDADE PARA O AJUIZAMENTO DE BA'
	,'REVISIONAIS - BUSCA E APREENSÃO - CASO MONITORADO'
	)

;run;

proc sort data = work.AndamentoF2; by Pasta peso ;run;
Data work.AndamentoF2; set work.AndamentoF2; by Pasta; if first.Pasta ;run;

create table work.TBLConsolidado as
select distinct

 a.Pasta
,case when b.LIMINAR_DEPOSITO_INTEGRAL = 1 then 1 else 0 end as FlLiminar_Deposito_Integral
,case when c.LIMINAR_DEPOSITO_PARCIAL = 1 then 1 else 0 end as FlLiminar_Deposito_Parcial
,case when d.LIMINAR_SEM_DEPOSITO = 1 then 1 else 0 end as FlLiminar_Sem_Deposito
,case when e.LIMINAR_CONCEDIDA = 1 then 1 else 0 end as FlLiminar_Concedida
,case when f.LIMINAR_DEFERIDA = 1 then 1 else 0 end as FlLiminar_Deferida
,case when g.LIMINAR_INDEFERIDA = 1 then 1 else 0 end as FlLiminar_Indeferida
,case when h.LIMINAR_REVOGADA = 1 then 1 else 0 end as FlLiminar_Revogada
,case when i.LIMINAR_DEF_SEM_SUSPENSAO_COB = 1 then 1 else 0 end as FlLiminar_Def_Sem_Suspensao_Cob

from dbep.TBLPROCESSOANDAMENTO a
left join work.T1 b on b.Pasta = a.Pasta
left join work.T2 c on c.Pasta = a.Pasta
left join work.T3 d on d.Pasta = a.Pasta
left join work.T4 e on e.Pasta = a.Pasta
left join work.T5 f on f.Pasta = a.Pasta
left join work.T6 g on g.Pasta = a.Pasta
left join work.T7 h on h.Pasta = a.Pasta
left join work.T8 i on i.Pasta = a.Pasta

where 1=1
and CdSistema = 'p'
;run;
