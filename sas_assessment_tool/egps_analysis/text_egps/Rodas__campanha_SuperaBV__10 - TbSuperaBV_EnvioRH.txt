/********************************************************/

PROC SQL;
CREATE TABLE TBSUPERABV_ENVIORH AS
SELECT

A.NUANOMES,
A.DsLoginN1,
B.CdColaboradorAtivo,
B.IdMatricula,
B.NmColaborador,
A.VrRecompensa

FROM CSBV.TBSUPERABV_RESUMO A

LEFT JOIN CSBV.ESTRUTURA_COMERCIAL_01 B ON B.NUANOMES = A.NUANOMES AND B.DSLOGINN1 = A.DSLOGINN1

WHERE A.NUANOMES = &Base_Mes

ORDER BY 1,2

;RUN;

/*****************************/
/*SALVA A ESTRUTURA COMERCIAL*/
/*****************************/

/*DATA CSBV.TBSUPERABV_ENVIORH;*/
/*SET WORK.TBSUPERABV_ENVIORH;*/
/*;RUN;*/

PROC SQL;
CREATE TABLE MAX_DATA AS
SELECT
MAX(NuAnoMes) AS Max_Estrutura_RH
FROM CSBV.TBSUPERABV_ENVIORH
WHERE NUANOMES = &Base_Mes.
;run;

DATA _NULL_;
SET WORK.MAX_DATA;
CALL SYMPUT ('Max_Estrutura_RH',Max_Estrutura_RH);
;run;
%PUT &Max_Estrutura_RH;

PROC SQL;
DELETE FROM CSBV.TBSUPERABV_ENVIORH
WHERE NUANOMES = &Max_Estrutura_RH.;
;RUN;
