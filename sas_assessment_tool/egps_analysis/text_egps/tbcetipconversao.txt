CREATE TABLE  TbCetipConversaoRetorno AS
SELECT * FROM connection to dbcon
(
    WITH DadosConversao AS (
      SELECT
        *,
        CASE WHEN NmArquivo LIKE 'R_901%' THEN CAST(substring(replace(nmarquivo,' ',''),7,8) AS DATE) END AS DtArquivo
      FROM DWBI_DTWH.TbConversao
    )
    SELECT DISTINCT
         CdTipoRegistro as tp_registro
        ,CdTipoDocumento as tp_documento
        ,NuCpfCnpj as cpf_cliente
        ,NuCnpjLoja as cnpj_loja
        ,VrAprovado as vl_aprov
        ,DtInclusaoGravame as dt_gravame
        ,DtContrato as dt_contrato
        ,DsMarcaModelo as marca_modelo
        ,DtAnoModelo as anomodelo
        ,DtAnoFabricacao as anofabricacao
        ,NuPrazoOperacao as qt_parcela
        ,CdFipe as cod_fipe
        ,VrFipe as vlfipe
        ,lpad(NufaixaVrAprovado,4,0) as fx_vl_aprovado
        ,NuCnpjAgenteFinanceiro as cnpj_agfin
        ,NmAgenteFinanceiro as nm_ag_fin
        ,StPagoLoja as pgto_loja
/*        ,CAST(trim(CdTipoRegistro)*/
/*       ||trim(str(CdTipoDocumento,1,0))*/
/*       ||trim(lpad(NuCpfCnpj,14,0))*/
/*       ||trim(lpad(NuCnpjLoja,14,0))*/
/*       ||trim(lpad(CAST(VrAprovado AS INTEGER),10,0)) AS VARCHAR(40)) AS chave*/
/*       ,NmArquivo*/
    FROM DadosConversao
    WHERE NmAgenteFinanceiro IS NOT NULL
      AND DtArquivo >= today()-30
/*  ORDER BY chave,cnpj_agfin*/
)
;RUN;

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
CREATE TABLE TbCetipConversaoEnvio AS
SELECT * FROM connection to dbcon
(
	WITH TbNuCpfCnpj AS
	(
		SELECT DISTINCT
			 PessoaSobAnalise.NuCpfCnpj
		FROM DBA.TbProposta Proposta

			INNER JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
				ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
				AND PessoaSobAnalise.CdPapel = 1

			INNER JOIN DBA.TbSubProduto SubProduto
				ON  Proposta.CdSubProduto = SubProduto.CdSubProduto

		WHERE SubProduto.CdProduto = 12 /* CDCCG */
		  AND Proposta.StProposta IN ('AP')
		  AND DtStatusProposta >= dateadd(day,-30,today())
		  AND DtStatusProposta  < today()
	)

	SELECT
		 Proposta.NuProposta
		,Proposta.CdTabelaFinanciamento
		,Proposta.PcEntrada
		,Proposta.QtParcelas
		,PessoaSobAnalise.NuCpfCnpj
		,coalesce(Loja.NuCpfCnpj,'00000000000000') AS NuCnpjLoja
		,Proposta.VrFinanciamento
		,Proposta.StProposta
		,date(Proposta.DtEntradaProposta) as DtEntradaProposta
		,today() as DtProcessamento
	FROM DBA.TbProposta Proposta

		INNER JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
			ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
			AND PessoaSobAnalise.CdPapel = 1

			LEFT JOIN TbNuCpfCnpj NuCpfCnpj
				ON PessoaSobAnalise.NuCpfCnpj = NuCpfCnpj.NuCpfCnpj

		INNER JOIN DBA.TbSubProduto SubProduto
			ON  Proposta.CdSubProduto = SubProduto.CdSubProduto

		LEFT JOIN DBA.TbLoja Loja
			ON Proposta.CdLoja = Loja.CdLoja

	WHERE SubProduto.CdProduto = 12 /* CDCCG */
	  AND PessoaSobAnalise.TpPessoa = 'F'
	  AND Proposta.StProposta IN ('CA','N','A','PA','C')
	  AND DtStatusProposta >= dateadd(day,-30,today())
	  AND NuCpfCnpj.NuCpfCnpj IS NULL

)
;RUN;

/* Separa apenas as propostas mais recentes */
proc sort data=TbCetipConversaoEnvio;by NuCpfCnpj descending DtEntradaProposta;run;
proc sort data=TbCetipConversaoEnvio nodupkey;by NuCpfCnpj;run;

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbArquivos as
SELECT * FROM connection to dbcon
(
    select distinct
        CASE WHEN NmArquivo LIKE 'R_901%' THEN CAST(substring(replace(nmarquivo,' ',''),7,8) AS DATE) END AS DtArquivo
        ,NmArquivo

    from DWBI_DTWH.TbConversao
    where nmarquivo like 'R_901%'
)
;RUN;

proc sql;
create table DtAtualizacaoCetip as
select
 	 max(DtArquivo) as DtAtualizacao format date9.
	,datetime() as TsProcessamento format datetime20.
from TbArquivos
;run;

%CRIAVARDB;
proc sql;
%CONECTADB(server = BI);
create table TbAgente as
select * from connection to dbcon
(
	SELECT
		*
	FROM DWBI_DTWH.TbAgenteMIS
	WHERE NuCnpjAgente <> 0
)
;run;

proc sql;
create table bases.TbCetipConversao as
select
	 CetipEnvio.NuProposta
	,CetipEnvio.NuCpfCnpj
	,max(case when Agente.NmSubgrupoAgenteMis = 'BANCO DO BRASIL' then 1 else 0 end) as FlBancoBrasil
	,max(case when Agente.NmSubgrupoAgenteMis = 'BRADESCO' then 1  else 0 end) as FlBradesco
	,max(case when Agente.NmSubgrupoAgenteMis = 'ITAU' then 1  else 0 end) as FlItau
	,max(case when Agente.NmSubgrupoAgenteMis = 'PANAMERICANO' then 1  else 0 end) as FlPan
	,max(case when Agente.NmSubgrupoAgenteMis = 'SAFRA' then 1  else 0 end) as FlSafra
	,max(case when Agente.NmSubgrupoAgenteMis = 'SANTANDER' then 1  else 0 end) as FlSantander
	,max(case when Agente.NmSubgrupoAgenteMis = 'C6' then 1  else 0 end) as FlC6
	,max(case when Agente.NmGrupoAgenteMIS    = 'MONTADORAS' then 1  else 0 end) as FlMontadoras
	,max(case when Agente.NmGrupoAgenteMIS   <> 'MONTADORAS' and Agente.NmSubGrupoAgenteMIS not in ('BV','C6','BANCO DO BRASIL','BRADESCO','ITAU','PANAMERICANO','SAFRA','SANTANDER') then 1  else 0 end) as FlOutros
	,max(1) as FlConcorrente
	,min(CetipRetorno.pgto_loja) as pgto_loja
	,DtAtualizacao.DtAtualizacao
	,DtAtualizacao.TsProcessamento

from  bases.TbCetipConversaoEnvio CetipEnvio

	left join DtAtualizacaoCetip DtAtualizacao
		on  1=1

	inner join bases.TbCetipConversaoRetorno CetipRetorno
		on  CetipEnvio.chave = CetipRetorno.chave

		inner join TbAgente Agente
			on  CetipRetorno.cnpj_agfin = Agente.NuCNPJAgente

where Agente.NmSubgrupoAgenteMis <> 'BV'
group by
	 CetipEnvio.NuProposta
	,DtAtualizacao.DtAtualizacao
	,DtAtualizacao.TsProcessamento
	,CetipEnvio.NuCpfCnpj
order by
	 CetipEnvio.NuProposta
;run;
