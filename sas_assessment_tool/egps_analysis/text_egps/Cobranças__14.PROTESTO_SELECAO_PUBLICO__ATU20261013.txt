create table BASE_ATRASO as
select * from connection to perm (

SELECT
  
  k.NUCONTRATO,
  k.SqContratoFinanceiro,
  t.cd_contrato,

  
  k.Dias_atraso,
  k.Cdmotivocontratofinanceiro,
  k.saldo,
  k.cdsituacaocontabil,

  
  c.cd_cpf_cnpj AS NuCpfCnpj,
  SAFE_CAST(c.Cd_Produto AS NUMERIC)            AS CdProduto,
  SAFE_CAST(c.Cd_Modalidade_Produto AS NUMERIC) AS CdModalidadeProduto,
  SAFE_CAST(c.Cd_Pessoa AS NUMERIC)             AS CdPessoa,
  a.CdSubProduto,
  h.CDGRUPOSUBPRODUTO,

  
  t.parcela,
  t.dt_venc_min,
  t.qtde_parcelas,
  t.qtde_parcelas_abertas,
  t.qtde_parcelas_atrasadas,
  t.risco_contrato_atrasado,
  t.risco_contrato_aberto,
  t.risco_contrato,
  t.vr_principal_aberto,
  t.vr_principal_atrasado,

  
  SUM(
    CASE
      WHEN s.cod_desp IN (14,80)
      THEN SAFE_CAST(s.valor_desp AS NUMERIC)
      ELSE 0
    END
  ) AS VRAJUIZ,

  
  CASE
    WHEN SAFE_CAST(b.NuContratoGestao AS NUMERIC) IS NOT NULL THEN 1
    ELSE 0
  END AS BNDU,

  
  MAX(SAFE_CAST(seg.VRPREMIOSEGURO AS NUMERIC))      AS VRPREMIOSEGURO,
  MAX(SAFE_CAST(seg.VRCOMISSAOCALCULADA AS NUMERIC)) AS VRCOMISSAOCALCULADA,
  MAX(CAST(seg2.FL_SEGURO_CONTRATADO AS INT64))      AS FLSEGUROCONTRATADO,

  CASE
    WHEN MAX(CAST(seg2.FL_SEGURO_CONTRATADO AS INT64)) = 1
       OR COUNT(seg.NUCONTRATO) > 0
    THEN 1 ELSE 0
  END AS FLAG_SEGURO

FROM
  
  (
    SELECT
      CAST(p.cd_contrato AS NUMERIC) AS NUCONTRATO,
      p.cd_contrato,
      p.sq_contrato_financeiro,

      MIN(
        CASE
          WHEN p.ST_PARCELA <> 'Q'
            OR (p.ST_PARCELA = 'Q'
                AND p.dt_processamento_ult_recebimento > DATE(&primeirodiames.))
          THEN CAST(p.nu_parcela AS NUMERIC)
        END
      ) AS parcela,

      MIN(
        CASE
          WHEN p.ST_PARCELA <> 'Q'
            OR (p.ST_PARCELA = 'Q'
                AND p.dt_processamento_ult_recebimento > DATE(&primeirodiames.))
          THEN p.dt_vencimento
        END
      ) AS dt_venc_min,
      COUNT(*) AS qtde_parcelas,
      COUNT(
        CASE
          WHEN p.ST_PARCELA <> 'Q'
            OR (p.ST_PARCELA = 'Q'
                AND p.dt_processamento_ult_recebimento > DATE(&primeirodiames.))
          THEN 1
        END
      ) AS qtde_parcelas_abertas,
      COUNT(
        CASE WHEN (p.ST_PARCELA <> 'Q'
                   OR (p.ST_PARCELA = 'Q' AND DATE(p.dt_processamento_ult_recebimento) > DATE(&primeirodiames.)))
                   AND DATE(p.Dt_Vencimento) < DATE(&primeirodiames.)
                   THEN p.Vr_Parcela ELSE NULL END) AS qtde_parcelas_atrasadas,
	  SUM(
        CASE WHEN (p.ST_PARCELA <> 'Q'
                   OR (p.ST_PARCELA = 'Q' AND DATE(p.dt_processamento_ult_recebimento) > DATE(&primeirodiames.)))
                   AND DATE(p.Dt_Vencimento) < DATE(&primeirodiames.)
                   THEN CAST(p.vr_parcela AS NUMERIC) ELSE NULL END) AS risco_contrato_atrasado,

      SUM(CASE WHEN p.ST_PARCELA <> 'Q' OR (p.ST_PARCELA = 'Q' AND p.dt_processamento_ult_recebimento > DATE(&primeirodiames.))
          THEN CAST(p.vr_parcela AS NUMERIC)
          ELSE 0 END) AS risco_contrato_aberto,

	  SUM(CASE WHEN (p.ST_PARCELA <> 'Q'
                   OR (p.ST_PARCELA = 'Q' AND DATE(p.dt_processamento_ult_recebimento) > DATE(&primeirodiames.)))
                   AND DATE(p.Dt_Vencimento) < DATE(&primeirodiames.)
                   THEN CAST(p.vr_principal AS NUMERIC) ELSE NULL END) AS vr_principal_atrasado,

	  SUM(CASE WHEN p.ST_PARCELA <> 'Q' OR (p.ST_PARCELA = 'Q' AND p.dt_processamento_ult_recebimento > DATE(&primeirodiames.))
          THEN CAST(p.vr_principal AS NUMERIC)
          ELSE 0 END) AS vr_principal_aberto,

	  sum(CAST(p.vr_parcela AS NUMERIC)) as risco_contrato

    FROM bv-pdbd-prd.dvry_core.tbparcela p
    GROUP BY
      CAST(p.cd_contrato AS NUMERIC),
      p.cd_contrato,
      p.sq_contrato_financeiro
  ) t


JOIN (
  SELECT
    CAST(Z.NUCONTRATO AS NUMERIC) AS NUCONTRATO,
    Z.SqContratoFinanceiro,
    MAX(Z.nudiasatraso) AS Dias_atraso,
    Z.Cdmotivocontratofinanceiro,
    CAST(Z.vrtaxacontrato AS NUMERIC)
      + CAST(Z.vrpermanenciaapropriado AS NUMERIC) AS saldo,
    Z.cdsituacaocontabil
  FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil Z
  WHERE Z.dtprocessamento = &primeirodiames.
    AND Z.nudiasatraso BETWEEN &atraso1. AND &atraso2.
  GROUP BY
    Z.NUCONTRATO,
    Z.SqContratoFinanceiro,
    Z.Cdmotivocontratofinanceiro,
    Z.vrtaxacontrato,
    Z.vrpermanenciaapropriado,
    Z.cdsituacaocontabil
) k
  ON t.NUCONTRATO = k.NUCONTRATO
 AND t.sq_contrato_financeiro = k.SqContratoFinanceiro


LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato c
  ON t.NUCONTRATO = SAFE_CAST(c.cd_contrato AS NUMERIC)

LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta a
  ON t.NUCONTRATO = SAFE_CAST(a.nucontrato AS NUMERIC)

LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_desp_contrato s
  ON t.NUCONTRATO = SAFE_CAST(s.NUCONTRATOGESTAO AS NUMERIC)
 AND t.sq_contrato_financeiro = s.SQCONTRATOFINANCEIRO

LEFT JOIN bv-pdbd-prd.trusted_dbpatio.tbentrada b
  ON t.NUCONTRATO = SAFE_CAST(b.NuContratoGestao AS NUMERIC)

LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo h
  ON a.CdSubProduto = h.CdSubProduto

LEFT JOIN bv-pdbd-prd.trusted_dbcontratoseguro.tbsegurocontratofinanceiro seg
  ON t.NUCONTRATO = SAFE_CAST(seg.NUCONTRATO AS NUMERIC)
 AND t.sq_contrato_financeiro = seg.SQCONTRATOFINANCEIRO
 AND seg.CDPRODUTOSEGURO IN (12,14)

LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro seg2
  ON t.NUCONTRATO = SAFE_CAST(seg2.cd_contrato AS NUMERIC)
 AND t.sq_contrato_financeiro = seg2.SQ_CONTRATO_FINANCEIRO
 AND seg2.FL_SEGURO_CONTRATADO = TRUE

WHERE SAFE_CAST(c.Cd_Produto AS NUMERIC) IN (12,14)

GROUP BY
  k.NUCONTRATO,
  k.SqContratoFinanceiro,
  t.cd_contrato,
  k.Dias_atraso,
  k.Cdmotivocontratofinanceiro,
  k.saldo,
  k.cdsituacaocontabil,
  c.cd_cpf_cnpj,
  c.Cd_Produto,
  c.Cd_Modalidade_Produto,
  c.Cd_Pessoa,
  a.CdSubProduto,
  h.CDGRUPOSUBPRODUTO,
  b.NuContratoGestao,
  t.parcela,
  t.dt_venc_min,
  t.qtde_parcelas,
  t.qtde_parcelas_abertas,
  t.qtde_parcelas_atrasadas,
  t.risco_contrato_atrasado,
  t.risco_contrato_aberto,
  t.risco_contrato,
  t.vr_principal_aberto,
  t.vr_principal_atrasado
);

disconnect from perm;
quit;

PROC SQL;
CREATE TABLE MELHOR_MODELO AS
SELECT
    NUCONTRATO,
    SQCONTRATOFINANCEIRO,
    COALESCE(GH_MODELO, GH_ANTERIOR) AS GH_MODELO,
    COALESCE(COLLECTION_MODELO, COLLECTION_ANTERIOR) AS COLLECTION_MODELO

    SELECT
        a.NUCONTRATO,
        a.SQCONTRATOFINANCEIRO,
        
        MAX(CASE
                WHEN a.GH_MODELO IS NULL THEN .
                ELSE a.GH_MODELO
            END) AS GH_MODELO,
        
        MAX(CASE
                WHEN a.COLLECTION_MODELO IN ('SEM MODELO', '') THEN ''
                ELSE a.COLLECTION_MODELO
            END) AS COLLECTION_MODELO,
        
        MAX(CASE
                WHEN b.GH_MODELO IS NULL THEN .
                ELSE b.GH_MODELO
            END) AS GH_ANTERIOR,
        
        MAX(CASE
                WHEN b.COLLECTION_MODELO IN ('SEM MODELO', '') THEN ''
                ELSE b.COLLECTION_MODELO
            END) AS COLLECTION_ANTERIOR
    FROM LTV.LTV_NOVO_202607 a
    LEFT JOIN LTV.LTV_NOVO_202606 b
        ON a.NUCONTRATO = b.NUCONTRATO
       AND a.SQCONTRATOFINANCEIRO = b.SQCONTRATOFINANCEIRO
    GROUP BY a.NUCONTRATO, a.SQCONTRATOFINANCEIRO
) AS c;

PROC SQL;
CREATE TABLE MELHOR_MODELO AS
SELECT
    NUCONTRATO,
    SQCONTRATOFINANCEIRO,
    COALESCE(GH_MODELO, GH_ANTERIOR) AS GH_MODELO,
    COALESCE(COLLECTION_MODELO, COLLECTION_ANTERIOR) AS COLLECTION_MODELO,
	COALESCE(CLUSTER_RISCO, CLUSTER_RISCO_ANTERIOR) AS CLUSTER_RISCO,
	COALESCE(PERC_PAGO,PERC_PAGO_ANTERIOR) AS PERC_PAGO
FROM (
    SELECT
        a.NUCONTRATO,
        a.SQCONTRATOFINANCEIRO,
        
        MAX(CASE
                WHEN a.GH_MODELO IS NULL THEN .
                ELSE a.GH_MODELO
            END) AS GH_MODELO,
        
        MAX(CASE
                WHEN a.COLLECTION_MODELO IN ('SEM MODELO', '') THEN ''
                ELSE a.COLLECTION_MODELO
            END) AS COLLECTION_MODELO,
		
		MAX(CASE
                WHEN a.CLUSTER_RISCO IN ('SEM MODELO', '') THEN ''
                ELSE a.CLUSTER_RISCO
            END) AS CLUSTER_RISCO,
			a.PERC_PAGO,
        
        MAX(CASE
                WHEN b.GH_MODELO IS NULL THEN .
                ELSE b.GH_MODELO
            END) AS GH_ANTERIOR,
        
        MAX(CASE
                WHEN b.COLLECTION_MODELO IN ('SEM MODELO', '') THEN ''
                ELSE b.COLLECTION_MODELO
            END) AS COLLECTION_ANTERIOR,
		
		MAX(CASE
                WHEN b.CLUSTER_RISCO IN ('SEM MODELO', '') THEN ''
                ELSE b.CLUSTER_RISCO
            END) AS CLUSTER_RISCO_ANTERIOR,
			b.PERC_PAGO as PERC_PAGO_ANTERIOR
    FROM LTV.LTV_NOVO_&anomesltv. a
    LEFT JOIN LTV.LTV_NOVO_&anomesltv_ant. b
        ON a.NUCONTRATO = b.NUCONTRATO
       AND a.SQCONTRATOFINANCEIRO = b.SQCONTRATOFINANCEIRO
    GROUP BY a.NUCONTRATO, a.SQCONTRATOFINANCEIRO
) AS c;

    FX_ATRASO,

CREATE TABLE MELHOR_MODELO AS

    COALESCE(GH_MODELO, GH_ANTERIOR) AS GH_MODELO,

        MAX(CASE
                WHEN a.GH_MODELO IS NULL THEN .
                ELSE a.GH_MODELO
            END) AS GH_MODELO,

                WHEN a.CLUSTER_RISCO IN ('SEM MODELO', '') THEN ''
                ELSE a.CLUSTER_RISCO

                ELSE b.GH_MODELO

                WHEN b.COLLECTION_MODELO IN ('SEM MODELO', '') THEN ''

                WHEN b.CLUSTER_RISCO IN ('SEM MODELO', '') THEN ''

WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("4. RESERVADO PARA AJUIZAMENTO - CONTROLE") THEN "Controle"
WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("6. IMPEDITIVOS AJUIZAMENTO","7. NAO ATINGIU SALDO MINIMO","9. BOM","8. REGULAR","10. RUIM") AND A.random >= 18 THEN "Teste"
WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("6. IMPEDITIVOS AJUIZAMENTO","7. NAO ATINGIU SALDO MINIMO","9. BOM","8. REGULAR","10. RUIM") THEN "Controle"

END AS Teste

, CASE WHEN CALCULATED Teste = "Teste" THEN 1 ELSE 0 END AS QTD_TESTE
, CASE WHEN CALCULATED Teste = "Controle" THEN 1 ELSE 0 END AS QTD_CONTROLE


, CASE
    WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("1. PUBLICO AJUIZAMENTO ESTRUTURANTE", "2. PROTESTADO ANTERIORMENTE") THEN  "A. PROTESTADO"
	WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("3. AJUIZADO") THEN  "B. AJUIZADO"
	WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("4. RESERVADO PARA AJUIZAMENRO - ANTECIPACAO","4. RESERVADO PARA AJUIZAMENRO - RENOTIFICAR") THEN  "C. A AJUIZAR"
	WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("5. SUSPENSO","6. SINISTRO SEGURO") THEN  "D. RESTRICOES"
	WHEN CALCULATED DS_ELEGIBILIDADE_2 IN ("4. RESERVADO PARA AJUIZAMENTO - CONTROLE") THEN  "E. CONTROLE AJUIZAMENTO"
	WHEN CALCULATED Teste IN ("Controle") THEN  "F. CONTROLE PROTESTO"
	WHEN CALCULATED Teste IN ("Teste") THEN  "G. A PROTESTAR"

    WHEN CALCULATED DS_ELEGIBILIDADE_2 = "DEMAIS COLLECTIONS"
         AND A.Dias_atraso > 90 THEN "DEMAIS COLLECTIONS"

    ELSE "NAO ELEGIVEL"
END AS DS_ELEGIBILIDADE_AGOSTO

FROM BASE_PROTESTO_TESTE_CONTROLE A

LEFT JOIN BASE_PROTESTO_CAROL B
    ON A.NUCONTRATO = B.NUCONTRATO
    AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
    AND A.PARCELA = B.PARCELA

LEFT JOIN CONTRATOS_BOLHA C
    ON A.NUCONTRATO = INPUT(C.NuContrato,32.)
    AND A.SqContratoFinanceiro = C.sqcontratofinanceiro

LEFT JOIN SINISTRO_ATUAL_FILTRADO D
    ON INPUT(A.NuCpfCnpj,32.) = D.CPF

LEFT JOIN AJUIZAMENTO_ESTOQUE E
    ON A.NUCONTRATO = INPUT(E.cd_contrato,32.)
    AND A.SQCONTRATOFINANCEIRO = E.sq_contrato_financeiro
    AND A.PARCELA = E.parcela

LEFT JOIN CONTROLE_AJUIZAMENTO F
    ON A.NUCONTRATO = INPUT(F.cd_contrato,32.)
    AND A.SQCONTRATOFINANCEIRO = F.sq_contrato_financeiro
    AND A.PARCELA = F.parcela

LEFT JOIN HIST_CONTROLE_CAROL G
    ON A.NUCONTRATO = INPUT(G.cd_contrato,32.)

LEFT JOIN IMPEDITIVOS_AJUIZAMENTO H
    ON A.NUCONTRATO = INPUT(H.CONTRATO,32.)

create table VALIDA_FX_ATRASO as
