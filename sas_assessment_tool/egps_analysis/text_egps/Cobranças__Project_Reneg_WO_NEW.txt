SELECT A.*

	 WHEN PUT(MesCompetencia,YYMMDDD10.) = '2025-10-01' THEN '202510'
	 ELSE 'VALIDAR'
     END AS SAFRA
	, CASE
		WHEN GpAgente = 'BELLINATI' THEN 'BELLINATI'
	 	WHEN GpAgente = 'CARVALHO' THEN 'CARVALHO'
	 	WHEN GpAgente = 'LM_ANGELIS' THEN 'LM_ANGELIS'
	 	WHEN GpAgente = 'MANDALITI' THEN 'MANDALITI'
	 	WHEN GpAgente = 'MENDES_E_CUNHA' THEN 'MENDES_E_CUNHA'
	 	WHEN GpAgente = 'PASQUALI' THEN 'PASQUALI'
		WHEN GpAgente = 'RN_ADV' THEN 'RN_ADV'
		WHEN GpAgente = 'SCHULZE' THEN 'SCHULZE'
	 	WHEN GpAgente = 'TOLEDO' THEN 'TOLEDO'
        ELSE 'OUT' END AS ASSESSORIA_2

		, CASE WHEN VrApreensao <> 0 AND VrApreensao <> . THEN 1 ELSE 0 END AS FLAG_APREENSAO
		, CASE WHEN VrEntrega <> 0 AND VrEntrega <> .  THEN 1 ELSE 0 END AS FLAG_ENTREGA
		, CASE WHEN VrReneg <> 0 AND VrReneg <> .  THEN 1 ELSE 0 END AS FLAG_RENEG
		, CASE WHEN VrRecebido <> 0 AND VrRecebido <> .  THEN 1 ELSE 0 END AS FLAG_CAIXA

		, CASE
		WHEN CALCULATED FLAG_APREENSAO = 1 THEN 'APREENSAO'
		WHEN CALCULATED FLAG_ENTREGA = 1 THEN 'ENTREGA'
		WHEN CALCULATED FLAG_RENEG = 1 THEN 'RENEG'
		WHEN CALCULATED FLAG_CAIXA = 1 THEN 'CAIXA'
		END AS TIPO_OFERTA
       , CASE
		WHEN AtrasoFinal <360 THEN 'A - ATÉ 360'
		WHEN AtrasoFinal BETWEEN 360 AND 540 THEN 'B - 360 a 540'
		WHEN AtrasoFinal BETWEEN 541 AND 720 THEN 'C - 541 a 720'
		WHEN AtrasoFinal BETWEEN 721 AND 1080 THEN 'D - 721 a 1080'
		WHEN AtrasoFinal BETWEEN 1081 AND 1800 THEN 'E - 1081 a 1800'

		, C.ELEG_RENEG
