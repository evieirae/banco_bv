%macro doHeader(nmBase,nmTabela,nmHeader, addDate);

	%if %lowcase("&addDate.") eq "linha1" %then %do;

			infile "&nmBase." dlm=',' dsd termstr=lf lrecl=32000 firstobs=1 missover obs=1;

					dtDate ddmmyy10.;

%macro doImport(nmLib,nmTabela,tpImport, addDate);
libname temp "&nmLib." compress=yes;

%if %lowcase("&addDate.") eq "coluna1" or %lowcase("&addDate.") eq "linha1" %then %do;
	%let nmTabela = &nmTabela._&hdDate;
%end;

%if %lowcase("&tpImport.") eq "append" and %checkTabela(tabela=temp.&nmTabela) %then %do;

	proc append base=&nmTabela data=temp.&nmTabela force;
	run;

	proc sql;
		create table temp.&nmTabela as
		select distinct * from &nmTabela;
	quit;

	%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);
%end;

/*and %checkTabela(tabela=temp.&nmTabela)*/
%else %if %lowcase("&tpImport.") eq "append2" %then %do;

	proc sql;
		create table &nmTabela as
		select * from &nmTabela
		union all
		select * from temp.&nmTabela;
	quit;

	proc sql;
		create table temp.&nmTabela as
		select distinct * from &nmTabela;
	quit;

	%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);
%end;

%else %if %lowcase("&tpImport.") eq "concat" and %checkTabela(tabela=temp.&nmTabela) %then %do;
	data temp.&nmTabela;
		set temp.&nmTabela &nmTabela;
	run;

	%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);
%end;
/*
Alterado em 24/08/2016 por Claudinei R. Pinto para a inclusão do tpImport="replace"
*/
%else %if %lowcase("&tpImport.") eq "replace" and %checkTabela(tabela=temp.&nmTabela) %then %do;
	data _null_;
		set temp.&nmTabela (obs=1);
		call symput('dtDate',dtReferencia);
	run;
	data temp.&nmTabela;
		set temp.&nmTabela(where=(dtReferencia ne &dtDate)) &nmTabela;
	run;

	%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);
%end;
%else %do;
	data temp.&nmTabela;
		set &nmTabela;
	run;

	%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);

	%chmod(dir=&nmLib,file=&nmTabela,permis=644,ext=sas7bdat);
%end;
%mend doImport;

%macro makeImportacoes(nmTabela, nmLib, nmBase, nmHeader, tpImport, addDate);
	data _null_;
		call symputx('_nmTabela',compress("&nmTabela."),'G');
		call symputx('_nmLib',compress("&nmLib."),'G');
		call symputx('_nmBase',compress("&nmBase."),'G');
		call symputx('_nmHeader',compress("&nmHeader."),'G');
		call symputx('_tpImport',compress("&tpImport."),'G');
		call symputx('_addDate',compress("&addDate."),'G');
	run;

	%if %sysfunc(fileexist(&nmBase)) %then %do;
		%if %lowcase("&tpImport.") eq "untar" %then %do;
			x "tar -xf &nmBase -C  &nmLib";
		%end;
		%else %do;
			%if %lowcase("&tpImport.") ne "move" %then %do;
				/*Cria a tabela temporária*/
				%doHeader(&nmBase,&nmTabela,&nmHeader, &addDate);

				/*Cria a tabela final com base no TpImport*/
				%doImport(&nmLib,&nmTabela,&tpImport, &addDate);
			%end;
			%else %do;
				%sysExec mv "&nmBase." "&nmLib.";
				%InsertOcorrencia(&sysnobs,&syserr,&syserrortext);
			%end;
		%end;

		%if %lowcase("&tpImport.") ne "move" %then %do;
			data _null_;
				call sleep(10000);
			run;

			%if &job_rc ne 0 %then
				%put job error=&job_rc.;
			%else
				x "mv &nmBase. &nmBase..old";

		%end;
	%end;
%mend makeImportacoes;
