CREATE TABLE GERAL_REVISIONAL AS
SELECT
	*
FROM CONNECTION TO DBCON

(
SELECT DISTINCT
	A.NUCONTRATO,
	MAX(B.SQCONTRATOFINANCEIRO) SQCONTRATOFINANCEIRO
INTO #BASE_SEQ
FROM #BASE A LEFT JOIN DBA.VWTBCONTRATOFINANCEIRO B
ON  A.NUCONTRATO = B.NUCONTRATO
WHERE DATE(DTPROCESSAMENTO) < &DT_INICIAL. 
GROUP BY
	A.NUCONTRATO



SELECT DISTINCT
	NUCONTRATOGESTAO,
	NUM_PROCESSO,
	DT_ASSES
INTO #01
FROM DBA.SAJ_PROCESSO
WHERE COD_ASSES IN (2038, 2036, 1663)		
AND (DATE(DT_ASSES) < &DT_INICIAL. AND (DT_RET_ASSES >= &DT_INICIAL. OR DT_RET_ASSES IS NULL))

SELECT DISTINCT
	NUCONTRATOGESTAO,
	MAX(DT_ASSES) AS DT_ASSES
INTO #02
FROM #01
GROUP BY
	NUCONTRATOGESTAO

SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.DT_ASSES,
	MAX(B.NUM_PROCESSO) AS NUM_PROCESSO
INTO #03
FROM #02 A LEFT JOIN #01 B
ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND A.DT_ASSES = B.DT_ASSES
GROUP BY
	A.NUCONTRATOGESTAO,
	A.DT_ASSES

SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.NUM_PROCESSO,
	B.COD_ASSES,
	A.DT_ASSES,
	B.DT_RET_ASSES,
	C.RAZAO_NOME AS ASSESSORIA_REVISIONAL
INTO #04
FROM #03 A LEFT JOIN DBA.SAJ_PROCESSO B
ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND A.NUM_PROCESSO = B.NUM_PROCESSO
AND A.DT_ASSES = B.DT_ASSES
LEFT JOIN SCC_FIN.SAJ_AGENTE C
ON B.COD_ASSES = C.COD_AGENTE



SELECT DISTINCT
	A.NUCONTRATO,
	B.SQCONTRATOFINANCEIRO,
	B.DTRECEBIMENTO,
	SUM(B.VRRECEBIDO) AS VRRECEBIDO
INTO #04_1
FROM #base A LEFT JOIN DBA.VWTBPARCELARECEBIMENTO B
ON A.NUCONTRATO = B.NUCONTRATO
WHERE DATE (B.DTRECEBIMENTO) >= &DT_INICIAL. AND DATE (B.DTRECEBIMENTO) <= &DT_FINAL.
AND B.FLESTORNADO = 'N' AND B.CDAGENTERECEBEDOR = 131
GROUP BY
	A.NUCONTRATO,
	B.SQCONTRATOFINANCEIRO,
	B.DTRECEBIMENTO

SELECT
	NUCONTRATO,
	SQCONTRATOFINANCEIRO,
	MIN(DTRECEBIMENTO) AS DTRECEBIMENTO,
	VRRECEBIDO
INTO #04_2
FROM #04_1
GROUP BY
	NUCONTRATO,
	SQCONTRATOFINANCEIRO,
	VRRECEBIDO

SELECT DISTINCT
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTRECEBIMENTO,
	A.VRRECEBIDO,
	MIN(B.DT_ASSES) AS DT_ASSES
INTO #04_3
FROM #04_2 A LEFT JOIN DBA.SAJ_PROCESSO B
ON A.NUCONTRATO = B.NUCONTRATOGESTAO
AND (A.DTRECEBIMENTO >= B.DT_ASSES AND A.DTRECEBIMENTO <= B.DT_RET_ASSES OR B.DT_RET_ASSES IS NULL)
WHERE ATIVA_PRINC = 'E'
GROUP BY
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTRECEBIMENTO,
	A.VRRECEBIDO

SELECT DISTINCT
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTRECEBIMENTO,
	A.VRRECEBIDO,
	A.DT_ASSES,
	B.DT_RET_ASSES,
	B.COD_ASSES,
	C.RAZAO_NOME
INTO #04_4
FROM #04_3 A LEFT JOIN DBA.SAJ_PROCESSO B
ON A.NUCONTRATO = B.NUCONTRATOGESTAO
AND A.DT_ASSES = B.DT_ASSES
LEFT JOIN SCC_FIN.SAJ_AGENTE C
ON B.COD_ASSES = C.COD_AGENTE



SELECT DISTINCT
	NUCONTRATOGESTAO,
	NUM_PROCESSO,
	DT_ASSES
INTO #05
FROM DBA.SAJ_PROCESSO
WHERE  ATIVA_PRINC = 'E' AND COD_ACAO IN ( 1,3,59,60) AND (DATE(DT_ASSES) < &DT_INICIAL.) 

SELECT DISTINCT
	NUCONTRATOGESTAO,
	MAX(DT_ASSES) AS DT_ASSES
INTO #05_1
FROM #05
GROUP BY
	NUCONTRATOGESTAO

SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.DT_ASSES,
	B.NUM_PROCESSO
INTO #06
FROM #05_1 A LEFT JOIN DBA.SAJ_PROCESSO B
ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND A.DT_ASSES = B.DT_ASSES

SELECT DISTINCT
	A.NUCONTRATOGESTAO,
	A.NUM_PROCESSO,
	B.COD_ASSES,
	A.DT_ASSES,
	B.DT_RET_ASSES,
	C.RAZAO_NOME AS ASSESSORIA_BA
INTO #07
FROM #06 A LEFT JOIN DBA.SAJ_PROCESSO B
ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND A.NUM_PROCESSO = B.NUM_PROCESSO

LEFT JOIN SCC_FIN.SAJ_AGENTE C
ON B.COD_ASSES = C.COD_AGENTE


SELECT DISTINCT
	A.*,
	D.ASSESSORIA_REVISIONAL,
	B.ASSESSORIA_BA AS ASSESSORIA_ATUAL_BA,
	C.RAZAO_NOME AS ASSESSORIA_PAGOU,
	C.DTRECEBIMENTO,
	C.VRRECEBIDO
INTO #08
FROM #BASE_SEQ A LEFT JOIN #07 B
ON A.NUCONTRATO = B.NUCONTRATOGESTAO
LEFT JOIN #04_4 C
ON A.NUCONTRATO = C.NUCONTRATO
LEFT JOIN #04 D
ON A.NUCONTRATO = D.NUCONTRATOGESTAO



SELECT DISTINCT
		A.*,
		B.SQCONTRATOFINANCEIROATIVO,
		CASE
			WHEN CDSITUACAOCONTRATO = 1 THEN 'ABERTO'
			ELSE 'FECHADO'
		END AS STATUS
INTO #10
FROM #08 A LEFT JOIN DBA.VWTBCONTRATO B
ON A.NUCONTRATO = B.NUCONTRATO



SELECT DISTINCT
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	SUM(B.VRPARCELA) AS VRPARCELA,
	MIN(B.NUPARCELA) AS NUPARCELA
INTO #11
FROM #10 A LEFT JOIN DBA.VWTBPARCELA B
ON A.NUCONTRATO = B.NUCONTRATO
AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
WHERE ( STPARCELA <> 'Q' OR STPARCELA = 'Q' AND  DATE(DTULTIMORECEBIMENTO) >= &DT_INICIAL.)
GROUP BY
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO

SELECT DISTINCT
	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.VRPARCELA AS RISCO,
	A.NUPARCELA,
	DATEDIFF(DAY,B.DTVENCIMENTO,&DT_INICIAL.) AS ATRASO
INTO #12
FROM #11 A LEFT JOIN DBA.VWTBPARCELA B
ON  A.NUCONTRATO           = B.NUCONTRATO
AND A.NUPARCELA            = B.NUPARCELA
AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

SELECT DISTINCT
	A.*,
	B.RISCO,
	B.ATRASO
INTO #13
FROM #10 A LEFT JOIN #12 B
ON A.NUCONTRATO = B.NUCONTRATO
AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO



SELECT
	NUCONTRATOGESTAO,
	MAX(DTINCLUSAO) AS DTENTRADA,
	MAX(CDBEM) AS CDBEM
INTO #VENDA
FROM DBA.TBVENDA
GROUP BY
	NUCONTRATOGESTAO

SELECT
	A.*,
	CASE
		WHEN B.NUCONTRATOGESTAO IS NOT NULL THEN 'S'
		ELSE 'N'
	END AS FLAG_VENDA
INTO #14
FROM #13 A LEFT JOIN #VENDA B
ON A.NUCONTRATO = B.NUCONTRATOGESTAO

CREATE TABLE GERAL_REVISIONAL AS
SELECT
	A.*,
	E.CDMODALIDADEPRODUTO,
	CASE
		WHEN E.CDMODALIDADEPRODUTO IN (1,2,3,6,8,27,38,41,45,56,26,46,79) 		    THEN 'LEVES'
		WHEN E.CDMODALIDADEPRODUTO IN (5,49) 								        THEN 'MOTOS'
		WHEN E.CDMODALIDADEPRODUTO IN (7,48)  							        	THEN 'VANS'
		WHEN E.CDMODALIDADEPRODUTO IN (9,28,47,78) 								    THEN 'PESADOS'
		WHEN E.CDMODALIDADEPRODUTO IN (16,33) 								    	THEN 'INSS'

		WHEN E.CDMODALIDADEPRODUTO IN (18,32,34,37,39,42,43) 				 	    THEN 'PUBLICO'
		WHEN E.CDMODALIDADEPRODUTO IN (2,19,20,21,22,23,24,29,30,31,44,57) 	    	THEN 'CP'
		WHEN E.CDMODALIDADEPRODUTO IN (4,10,11,12,13,14,15,50,51,52,53,54,55,25)  	THEN 'CDC'

		ELSE 'OUTROS'
	END AS PRODUTO
FROM GERAL_REVISIONAL A LEFT JOIN CONNECTION TO DBCON
			(SELECT
				B.NUCONTRATO,
				B.SQCONTRATOFINANCEIROATIVO,
			    B.NUCPFCNPJ,
			    B.CDPESSOA,
			    B.DTLIQUIDACAOEFETIVA,
				B.CDSITUACAOCONTRATO,
				B.CDMODALIDADEPRODUTO,
				C.CDMOTIVOCONTRATOFINANCEIRO,
			    C.CDTIPOCOBRANCA,
			    C.VRCONTRATO,
			    C.VRFINANCIAMENTO,
			    C.VRENTRADA,
				C.QTPARCELAS,
				C.VRPRESTACAO,
				D.CDUF,
				D.NMCIDADE
			FROM
				DBA.VWTBCONTRATO B LEFT JOIN DBA.VWTBCONTRATOFINANCEIRO C
				ON
				B.NUCONTRATO = C.NUCONTRATO
				AND
				B.SQCONTRATOFINANCEIROATIVO = C.SQCONTRATOFINANCEIRO
				LEFT JOIN DBA.TbPessoaEndereco D
				ON
				B.CDPESSOA = D.CDPESSOA
				AND
				D.SqEndereco = 1) E
ON
	A.NUCONTRATO = E.NUCONTRATO
	AND
	A.SQCONTRATOFINANCEIROATIVO = E.SQCONTRATOFINANCEIROATIVO;
DISCONNECT FROM DBCON;

PROC SQL;
   CREATE TABLE REMUNERACAO_REVISIONAL AS
   SELECT
	t1.SAFRA,
          t1.NUCONTRATO,
          t1.ASSESSORIA_REVISIONAL,
          t1.ASSESSORIA_ATUAL_BA,
          t1.ASSESSORIA_PAGOU,
          t1.DTRECEBIMENTO,
          t1.VRRECEBIDO,
          t1.SQCONTRATOFINANCEIROATIVO,
          t1.STATUS,
          t1.RISCO,
          t1.ATRASO,
          t1.FLAG_VENDA,
          t1.ASSESSORIA_BA,
          t1.ASSESSORIA_PGTO,
          t1.FAIXA,
		CASE
			WHEN t1.FAIXA = 'SEM ATRASO' 		THEN 0
			WHEN t1.FAIXA = '1 A 179' 			THEN 0
			WHEN t1.FAIXA = '180 A 360' 		THEN 0.0350
			WHEN t1.FAIXA = '361 A 450' 		THEN 0.0650
			WHEN t1.FAIXA = '451 A 540' 		THEN 0.1150
			WHEN t1.FAIXA = '541 A 1080' 		THEN 0.1350
			WHEN t1.FAIXA = 'ACIMA DE 1080'	THEN 0.1700
			ELSE 0 END AS PERCENT_REMUNERACAO,
		CASE
			WHEN t1.FAIXA = 'SEM ATRASO' 		THEN t1.VRRECEBIDO * 0
			WHEN t1.FAIXA = '1 A 179' 			THEN t1.VRRECEBIDO * 0
			WHEN t1.FAIXA = '180 A 360' 		THEN t1.VRRECEBIDO * 0.0350
			WHEN t1.FAIXA = '361 A 450' 		THEN t1.VRRECEBIDO * 0.0650
			WHEN t1.FAIXA = '451 A 540' 		THEN t1.VRRECEBIDO * 0.1150
			WHEN t1.FAIXA = '541 A 1080' 		THEN t1.VRRECEBIDO * 0.1350
			WHEN t1.FAIXA = 'ACIMA DE 1080'	THEN t1.VRRECEBIDO * 0.1700
			ELSE 0 END AS VALOR_REMUN,
          t1.ASSESSORIA_FINAL,
          t1.CDMODALIDADEPRODUTO,
          t1.PRODUTO,
          t1.Fl_Considera
      FROM REVI.GERAL_REVISIONAL t1
WHERE t1.SAFRA = &SAFRA.
AND t1.Fl_Considera = 0
AND T1.ASSESSORIA_final IN (&RANKING.)
and t1.FAIXA in ('180 A 360','361 A 450','451 A 540','541 A 1080','ACIMA DE 1080')
;
QUIT;

PROC SQL;
CREATE TABLE MARCACAO AS

	*,

		WHEN POLITICA_PADV = "W" AND MAIOR_DESCONTO = "W" THEN 'W'

	END AS MARCACAO_FINAL

FROM
	MARCACAO;

		WHEN PERCENT_COBERTURA >80 AND PERCENT_COBERTURA <=100 THEN '81_100%'

		END AS FX_COBERTURA

PROC SQL;
CREATE TABLE AVISO AS

	*,
	CASE
		WHEN (VLRDEPOSITO IS NULL OR VLRDEPOSITO <100) THEN 0
		WHEN PERCENT_COBERTURA >=100 AND (VLR_QUITACAO * 1.1)> VLRDEPOSITO THEN (VLRDEPOSITO-VLR_QUITACAO)
		WHEN PERCENT_COBERTURA >=100 AND (VLR_QUITACAO * 1.1)< VLRDEPOSITO THEN (VLR_QUITACAO * 0.1)
		ELSE (VLR_QUITACAO * 0.1)
	END AS REMUNERACAO
FROM
	AVISO;
QUIT;
RUN;

PROC SQL;

        WHEN  FX_COBERTURA = 'MAIOR 100%'  THEN (VLRDEPOSITO-REMUNERACAO)
        ELSE (VLR_QUITACAO)
	END AS REPASSEBV,
	CASE
        WHEN VLRDEPOSITO IS MISSING OR VLRDEPOSITO <=100 THEN 0
        WHEN FX_COBERTURA = 'MAIOR 100%'  THEN 0

FROM
	AVISO;

		WHEN C.ATRASO <=70 							THEN '00_70'
		WHEN C.ATRASO >=71 		AND C.ATRASO <=90 	THEN '71_90'
		WHEN C.ATRASO >=91 		AND C.ATRASO <=120 	THEN '91_120'
		WHEN C.ATRASO >=121 	AND C.ATRASO <=150 	THEN '121_150'
		WHEN C.ATRASO >=151 	AND C.ATRASO <=180 	THEN '151_180'
		WHEN C.ATRASO >=181 	AND C.ATRASO <=210 	THEN '181_210'
		WHEN C.ATRASO >=211 	AND C.ATRASO <=240 	THEN '211_240'
		WHEN C.ATRASO >=241 	AND C.ATRASO <=270 	THEN '241_270'
		WHEN C.ATRASO >=271 	AND C.ATRASO <=300 	THEN '271_300'
		WHEN C.ATRASO >=301 	AND C.ATRASO <=330 	THEN '301_330'
		WHEN C.ATRASO >=331 	AND C.ATRASO <=360 	THEN '331_360'
		WHEN C.ATRASO >=361 	AND C.ATRASO <=390 	THEN '361_390'
		WHEN C.ATRASO >=391 	AND C.ATRASO <=450 	THEN '391_450'
		WHEN C.ATRASO >=451 	AND C.ATRASO <=540 	THEN '451_540'
		WHEN C.ATRASO >=540 	AND C.ATRASO <=700 	THEN '540_700'
		WHEN C.ATRASO >700 	THEN 'MAIOR_700'
		ELSE 'OUTRO'
	END AS FAIXA_ATRASO,
	C.MAX_DTVENC,
	C.MAX_PARCATRASO,
	C.PARC_ABERTO,
	C.PARC_ABERTO *  A.VRPRESTACAO AS VLR_RISCO
FROM
	ESTOQUE_&DATABASE. A LEFT JOIN CONNECTION TO DBCON
			(SELECT
				NUCONTRATO,
				SQCONTRATOFINANCEIRO,
				MAX(DATEDIFF(DAY,DTVENCIMENTO,GETDATE())) AS ATRASO,
				MIN(DTVENCIMENTO) AS MAX_DTVENC,
				MIN(NUPARCELA) AS MAX_PARCATRASO,
				COUNT(NUPARCELA) AS PARC_ABERTO
			FROM
				DBA.VWTBPARCELA
			WHERE
				STPARCELA <> 'Q'
			GROUP BY
				NUCONTRATO,
				SQCONTRATOFINANCEIRO) C
ON
	A.NUM_CONTRATO = C.NUCONTRATO
AND
	A.SQCONTRATOFINANCEIROATIVO = C.SQCONTRATOFINANCEIRO;
DISCONNECT FROM DBCON;
QUIT;
