select * from Planctrl.BASE_SAIDA_SIC&Base.

select * from work.Cotas_SICSaida

update Work.Zerar_DRE
set template_margem = 'NAO EXPLICADO DRE';

create table Work.Calculo_Cotas as
select * from work.Cotas_SICSaida
where template_margem not in ('NAO EXPLICADO DRE', 'NAO EXPLICADO DRE TESOURARIA MANUAL', 'NAO EXPLICADO DRE CONCILIACAO');

update Work.Zerar_DRE
set res_dia_gerencial_sas_brl = res_dia_gerencial_sas_brl*-1,
    resultado_tesouraria_brl = resultado_tesouraria_brl*-1,
    resultado_comercial_brl = resultado_comercial_brl*-1,
    resultado_acionista_brl = resultado_acionista_brl*-1,
	vrmesacaixa = vrmesacaixa*-1;

create table work.segmentado as
select distinct cdobjetonegociacao, CdModalidade_Produto_SAS, CdSegmento_Mercado_SAS, Tipo_Contabil, anomes,  Chave_Unica,
CdCliente, CdOfficer_SAS, CdDivisao_Comercial_SAS, CdGrupo_Comercial_SAS, CdGrupo_Economico_SAS,
CdVeiculo, CdBook, cdGrupoConciliacao
from Work.Calculo_Cotas
where chave_unica not in ('')
and origem = 'SIC'
order by chave_unica;

create table work.duplicados as
select chave_unica, count(*) as qtd
from work.segmentado
group by chave_unica
having count(*) > 1
order by chave_unica;

delete from work.segmentado
where chave_unica in (select chave_unica from work.duplicados)
and cdbook = 169;

create table work.segmentado_margem as
select distinct cdobjetonegociacao, CdModalidade_Produto_SAS, CdSegmento_Mercado_SAS, Tipo_Contabil, anomes,  Chave_Unica,
CdCliente, CdOfficer_SAS, CdDivisao_Comercial_SAS, CdGrupo_Comercial_SAS, CdGrupo_Economico_SAS,
CdVeiculo, CdBook, cdGrupoConciliacao
from Work.Calculo_Cotas
where chave_unica not in ('')
and origem = 'Margem'
and chave_unica not in (select chave_unica from segmentado);

insert into work.segmentado
select * from segmentado_margem;

create table work.duplicados as
select chave_unica, count(*) as qtd
from work.segmentado
group by chave_unica
having count(*) > 1
order by chave_unica;

delete from work.segmentado
where chave_unica in (select chave_unica from work.duplicados)
and cdsegmento_mercado_sas = 8;

create table work.duplicados as
select chave_unica, count(*) as qtd
from work.segmentado
group by chave_unica
having count(*) > 1
order by chave_unica;

create table Calculo_segm as
select * 
from quebra_cotas as q
inner join work.segmentado as s
on s.cdobjetonegociacao = q.cota;

resultado_comercial_brl, resultado_acionista_brl, vrmesacaixa, CdModalidade_Produto_SAS, CdSegmento_Mercado_SAS, Tipo_Contabil,anomes, Chave_Unica,
CdCliente, CdOfficer_SAS, CdDivisao_Comercial_SAS, CdGrupo_Comercial_SAS, CdGrupo_Economico_SAS,
CdVeiculo, CdBook, cdGrupoConciliacao )
select cota, contabil, tesouraria, comercial, acionista, mesacaixa, CdModalidade_Produto_SAS, CdSegmento_Mercado_SAS, Tipo_Contabil,anomes, Chave_Unica,
CdCliente, CdOfficer_SAS, CdDivisao_Comercial_SAS, CdGrupo_Comercial_SAS, CdGrupo_Economico_SAS,
CdVeiculo, CdBook, cdGrupoConciliacao
from work.Calculo_segm;

quit;
