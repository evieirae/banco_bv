CREATE TABLE PROVISAO_VAREJO AS
   SELECT
        *,
	CASE
		WHEN cd_produto_risco = '1' THEN 'LEVES'
		WHEN cd_produto_risco = '2' THEN 'CARTOES'
		WHEN cd_produto_risco = '3' THEN 'CONSIGNADO PUBLICO'
		WHEN cd_produto_risco = '4' THEN 'CONSIGNADO PRIVADO'
		WHEN cd_produto_risco = '5' THEN 'CONSIGNADO INSS'
		WHEN cd_produto_risco = '6' THEN 'CP'
		WHEN cd_produto_risco = '7' THEN 'EGV'
		WHEN cd_produto_risco = '8' THEN 'EGV'
		WHEN cd_produto_risco = '9' THEN 'MOTOS'
		WHEN cd_produto_risco = '10' THEN 'PESADOS'
		WHEN cd_produto_risco = '11' THEN 'TURISMO'
		WHEN cd_produto_risco = '13' THEN 'SAUDE'
		WHEN cd_produto_risco = '14' THEN 'SOLAR'
		WHEN cd_produto_risco = '19' THEN 'FGTS'
		WHEN cd_produto_risco = '20' THEN 'CARTOES BANKLY'
		WHEN cd_produto_risco = '21' THEN 'DESENROLA BRASIL'
		WHEN cd_produto_risco = '22' THEN 'CARTOES DARWIN'
 	ELSE 'OUTROS'
	END AS PRODUTO
   FROM CONNECTION TO PERM
     (SELECT
        *
      FROM
        bv-pdbd-prd.dvry_corp_riscocredat.provisao_varejo_fato
   WHERE
        DT_REFERENCIA >= '2023-01-01');
   DISCONNECT FROM PERM;
QUIT;

CREATE TABLE PROVISAO_VAREJO_V2 AS
SELECT A.*,
	   CASE WHEN B.safra_Concessao >= 202501 THEN SUM(vr_saldo_carteira, vr_custo)
	   		ELSE A.vr_saldo_carteira END AS vr_saldo_carteira_ajustado

FROM PROVISAO_VAREJO AS A
LEFT JOIN BU AS B
	ON A.CD_CONTRATO = PUT(B.NUCONTRATO, 14.)

;RUN;
