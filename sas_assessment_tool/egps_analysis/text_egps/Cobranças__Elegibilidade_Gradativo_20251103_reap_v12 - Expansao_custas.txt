PROC SQL;
CREATE TABLE
	DESPESAS AS
SELECT DISTINCT
	A.NUCONTRATO
	,A.num_processo
	,A.FX_ATRASO
	,A.STATUS
	,SUM(CASE WHEN A.cod_oper = 'D' THEN A.VALOR_DESP ELSE 0 END) AS Desp_Totais
	,CASE WHEN
		CALCULATED Desp_Totais > 100 AND A.STATUS = 'ATIVA' THEN 'ELEGIVEL' ELSE 'INELEGIVEL' END AS ELEGIBILIDADE_CUSTAS
	,CASE WHEN
		CALCULATED ELEGIBILIDADE_CUSTAS = 'ELEGIVEL' THEN
			CASE
				WHEN
					MOD(input(hashing('sha1',PUT(A.NUCONTRATO+&SEED_CUSTAS.,z14.)),hex8.),100) < 50 THEN 'TESTE'
				ELSE 'CONTROLE' END
		ELSE
			'INELEGIVEL'
		END AS TESTE_CONTROLE_CUSTAS
	,Z4.TESTE_CONTROLE_CUSTAS AS TESTE_CONTROLE_CUSTAS_ANT
	,Z4.DATA_ELEGIBILIDADE


	,CASE WHEN
		CALCULATED ELEGIBILIDADE_CUSTAS = 'ELEGIVEL' AND Z4.NUCONTRATO IS NULL THEN 1 ELSE 0 END AS ENTRANTE_CUSTAS

FROM
	GRADATIVO_1_4_UF_VF_1_2 A


LEFT JOIN ALVESDS.BASE_TESTE_CUSTAS Z4
ON A.NUCONTRATO = Z4.NUCONTRATO

GROUP BY
	A.NUCONTRATO
	,A.num_processo
	,A.FX_ATRASO
	,A.STATUS
HAVING
	A.Dt_DESP = MAX(A.Dt_DESP)
;

PROC SQL;
CREATE TABLE
	VALIDA_ENTRANTES
AS
SELECT DISTINCT
	A.*,
	CASE WHEN
		C.NUCONTRATO IS NOT MISSING
		OR A.ENTRANTE_CUSTAS = 1
	
	
	
	
	

	THEN 1 ELSE 0 END AS FLAG_ENTRANTE


	,CASE WHEN
			A.ENTRANTE_CUSTAS = 1 THEN 'ENTRANTE - TESTE CUSTAS'
		WHEN
			B.NUCONTRATO IS MISSING THEN 'NOVO CONTRATO - ENTRANTE 90D'
		WHEN
			A.BASE_CONTRA_MANUAL = 1 THEN 'CONTRAPROPOSTA MANUAL - AJUSTE DESCONTO'
		WHEN
			C.NUCONTRATO IS NOT MISSING AND B.NUCONTRATO IS NOT MISSING THEN 'MOTIVO BAU'
		WHEN
			A.FLAG_AJUSTA_CUSTAS = 1 THEN 'AJUSTE DE CUSTAS'
		WHEN
			A.DESCONTO_CONTRA IS NOT NULL THEN 'CONTRAOFERTA - NEUROTECH'
	ELSE ''
	END AS MOTIVO_ENTRANTE
	,C.FLAG_MOTIVO
FROM
	BASE_CONFERENCIA A
LEFT JOIN
	LDUARTE.BASE_TESTE_GRADATIVO B
ON
	A.NUCONTRATO = B.NUCONTRATO

LEFT JOIN
	BASE_ENVIO_A C
ON
	A.NUCONTRATO = C.NUCONTRATO
;QUIT;

PROC SQL;
CREATE TABLE ENTRANTES_ENRIQ AS
SELECT DISTINCT
A.*
,H.DESC_FAIXA
,C2.QTDIAINICIOFAIXAATRASO AS FAIXA_INICIO
,CASE WHEN C2.QTDIAINICIOFAIXAATRASO > 120 THEN 1 ELSE 0 END AS FX_INICIO_120

,C2.LIMITADORPRINCIPAL3 AS DESCONTO_MIX_CENARIOS
,FLOOR((H.DESC_FAIXA*C2.VARIACAOPRINCIPAL3)/100)  AS VARICAO_DESC

,CASE WHEN H.DESC_FAIXA  > 50 THEN 'ACIMA 50' ELSE 'ABAIXO 50' END AS FX_DESC_50

,I.RISCO_ABERTO


,&DIA_REF3. AS DATA_MARCACAO

AND A.ATRASO >= C2.QTDIAINICIOFAIXAATRASO
AND A.ATRASO <= C2.QTDIAFIMFAIXAATRASO

AND H.TPACORDO2 = 1

LEFT JOIN RISCO_ABERTO I ON

PROC SQL;
CREATE TABLE
	VALIDA_ENTRANTES
AS
SELECT DISTINCT
	A.*,
	CASE WHEN
		C.NUCONTRATO IS NOT MISSING
		OR A.ENTRANTE_CUSTAS = 1
	
	
	
	
	

	THEN 1 ELSE 0 END AS FLAG_ENTRANTE


	,CASE WHEN
			A.ENTRANTE_CUSTAS = 1 THEN 'ENTRANTE - TESTE CUSTAS'

			A.FLAG_AJUSTA_CUSTAS = 1 THEN 'AJUSTE DE CUSTAS'
		WHEN
			A.DESCONTO_CONTRA IS NOT NULL THEN 'CONTRAOFERTA - NEUROTECH'
	ELSE ''
	END AS MOTIVO_ENTRANTE
	,C.FLAG_MOTIVO

	BASE_CONFERENCIA A

	LDUARTE.BASE_TESTE_GRADATIVO B

	A.NUCONTRATO = B.NUCONTRATO

	A.NUCONTRATO = C.NUCONTRATO
