PROC SQL;
CREATE TABLE WORK.BoletimCivelDetalhe AS

SELECT

  A.CdCausa
, case when A.DtCadastramento is null then '' else
	cats(
		day(A.DtCadastramento) , '/' ,
		month(A.DtCadastramento) , '/' ,
		year(A.DtCadastramento)
	) end as DtCadastramento
, case when A.DtEncerramento is null then '' else
	cats(
		day(A.DtEncerramento) , '/' ,
		month(A.DtEncerramento) , '/' ,
		year(A.DtEncerramento)
	) end as DtEncerramento
, upper(A.NmMunicipio) as NmMunicipio
, upper(A.CdUF) as CdUF
, A.NmMotivoRRO
, A.NmAssessoria
, A.NmProdutoGrupo
, A.NmSubMotivo
, A.FlAcaoAtiva
, year(A.DtCadastramento) as AnoCadastramento
, year(A.DtEncerramento) as  AnoEncerramento
, 1 as Contador

FROM  DBEP.VWTBLCAUSACONS AS A

WHERE 1=1
AND A.CDSISTEMA = 'p'
AND A.CDUNIDADEEXYON = 'VAC'

;RUN;
