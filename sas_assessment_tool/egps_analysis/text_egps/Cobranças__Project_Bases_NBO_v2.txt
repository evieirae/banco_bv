PROC SQL ;
connect to bigquery as perm
	(
	 project				= 'bv-sadi-dtm'
	 credfile				= '/projetos/acessos/bq/di/sadi-dtm.json'
	 schema					= 'bv_pdbd_prd'
	 READ_MODE				= storage
	 ALLOW_LARGE_RESULTS	= must
	 MAX_CHAR_LEN			= 100
	 SCANSTRINGCOLUMNS		= YES
	 );

	 CREATE TABLE WORK.Hist_Internalizacao_&DT_ARQ. AS

		SELECT DISTINCT
			*
		FROM CONNECTION TO PERM	(SELECT DISTINCT
									  dt_acionamento AS Dt
									, cd_cpf_cnpj as IdPessoa
									, cd_contrato as NuContrato
									, SUM(CASE WHEN tp_resultado_acionamento in ('1','2','3','4','5','6') 	THEN 1 ELSE 0 END) AS Flag_Acio
									, SUM(CASE WHEN tp_resultado_acionamento in ('2','3','4','5','6') 		THEN 1 ELSE 0 END) AS Flag_Alo
									, SUM(CASE 	WHEN tp_resultado_acionamento = '4' THEN 1
												WHEN tp_resultado_acionamento = '5' THEN 1
												WHEN tp_resultado_acionamento = '6' THEN 1
											ELSE 0 END) AS Flag_CPC
									, SUM(CASE WHEN tp_resultado_acionamento = '5' THEN 1 WHEN tp_resultado_acionamento = '6' THEN 1 ELSE 0 END) AS Flag_CPCA
									, SUM(CASE WHEN tp_resultado_acionamento = '6' THEN 1 ELSE 0 END) AS Flag_Promessa
								FROM `bv-pdbd-prd.dvry_corp_cobranca.assessoria_acionamentos_cobranca_fato`
								WHERE 1=1
								AND FORMAT_DATE('%Y%m', dt_acionamento) in (&REF_LIGACOES.)
								AND dt_acionamento < CURRENT_DATE()
								AND tp_resultado_acionamento in ('1','2','3','4','5','6')
								AND cd_cpf_cnpj IS NOT NULL
								AND cd_contrato IS NOT NULL
								GROUP BY dt_acionamento, cd_cpf_cnpj, cd_contrato
								ORDER BY dt_acionamento DESC

	)
	;DISCONNECT FROM PERM

;QUIT;
