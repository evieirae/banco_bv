PROC SQL;
CREATE TABLE TbPropostas_Executivo AS
SELECT
	Originacao.NuAnoMes
	,Originacao.DtEntradaProposta format ddmmyy10.
	,Originacao.NuProposta
	,Originacao.StProposta
	,Producao.NuContrato
	,Producao.DtContrato
	,Originacao.NmTipoCobranca
	,Originacao.CdParceiroComercial
	,Originacao.NuCnpjLoja
	,case
		when Originacao.NmParceiroComercial in ("CP GARAN FINANZERO","FINANZERO BRASIL")
			then "FINANZERO"
		else Originacao.NmParceiroComercial
	end
	as NmParceiroComercial
	,Originacao.NmTipoParceiroComercial
	,Originacao.NmPessoa
	,Originacao.TpPessoa
	,Originacao.CdUfPessoa
	,Originacao.NmCidadePessoa
    ,Originacao.NuCpfCnpj
	,Originacao.DsTipoCanalProducao
	,Originacao.NmCanalEstruturaComercial
	,Originacao.NmAreaEstruturaComercial
	,Originacao.NmSubEstruturaComercialA
	,Originacao.NmGrupoProdutoMeta as NmGrupoSubProduto
	,Originacao.QtAnosIdade
	,Originacao.FxIdade
	,Originacao.QtParcelas
	,Originacao.CdMarca
	,Originacao.CdModelo
	,Originacao.AaModelo
	,Originacao.AaFabricacao
	,Originacao.FlPropostaOriginada
    ,Originacao.FlPropostaNegada
    ,Originacao.FlPropostaAprovada
    ,Originacao.FlPropostaAPNP
    ,Originacao.FlPropostaBipartida
    ,Originacao.FlPropostaUtilizacaoFgts
	,Originacao.FlPropostaEfetivada
	,case
		when Producao.VrCliente < Producao.VrProducao
			then 1
				else 0
	END
	AS FlContratoDesdobro
	,CASE WHEN PARCDIG.CODIGOS IS NOT NULL THEN 'S' ELSE 'N' END AS FlParceiroDigital
	,CASE WHEN BALCAODIG.CDPARCEIRO IS NOT NULL THEN 'S' ELSE 'N' END AS FlBalcaoDigital
	,BALCAODIG.FILIAL AS FilialBalcaoDigital
	,BALCAODIG.NOME AS NomeBalcaoDIgital
	,Originacao.VrPropostaOriginada format commax18.2
	,Originacao.VrPropostaAprovada format commax18.2
	,Originacao.VrLiberado format commax18.2
	,Originacao.VrEntrada format commax18.2
	,Originacao.VrFinanciamento format commax18.2
	,Originacao.VrPrestacao format commax18.2
	,Producao.VrProducao format commax18.2
	,Producao.VrCliente format commax18.2
	,Producao.VrComissaoTotal format commax18.2
	,Originacao.QtCorridosDiaUtilMes
	,Originacao.DtAtualizacao
	,Originacao.TsProcessamento
	,originacao.cdColaborador
	,originacao.nmColaborador
	,MRT.ds_motivo_recusa
	,MRT.Motivo_Selecao
	,case when Originacao.StProposta = "AP" AND MRT.Motivo_Selecao = 1 then 1 else 0 end as FlConvMotRecusa
	,PAP.FLPAP

FROM MISV_TMP.TbOriginacao Originacao
	left JOIN MISV_TMP.TBEXTPROPOSTA PRO
	ON ORIGINACAO.NUPROPOSTA = PRO.NUPROPOSTA

	LEFT JOIN
		(select DISTINCT
			nuproposta,
			flpap
			from TRIBO.EGV_PAP
			where nuproposta is not null
	) PAP
	ON Originacao.NUPROPOSTA = PAP.NUPROPOSTA

	LEFT JOIN MISV_TMP.TbProducao Producao
		ON Originacao.NuProposta = Producao.NuProposta

		LEFT JOIN TRIBO.base_MotivoRecusaTargetEgv MRT
		ON Originacao.NuProposta = MRT.Nu_Proposta

			LEFT JOIN AUXL.EGV_DEPARAPARCEIROSDIGITAIS PARCDIG
/*			on CASE WHEN Originacao.NmGrupoProdutoMeta CONTAINS 'CVG' THEN 'EGV' ELSE Originacao.NmGrupoProdutoMeta END = */
/*			PARCDIG.PRODUTO AND Originacao.CdParceiroComercial = input(PARCDIG.CODIGOS,best.)*/
			on Originacao.CdParceiroComercial = input(PARCDIG.CODIGOS,best.)
			LEFT JOIN AUXL.EGV_DEPARABALCAODIGITAL BALCAODIG
/*			on CASE WHEN Originacao.NmGrupoProdutoMeta CONTAINS 'CVG' THEN 'EGV' ELSE Originacao.NmGrupoProdutoMeta END = */
/*			PARCDIG.PRODUTO AND Originacao.CdParceiroComercial = input(PARCDIG.CODIGOS,best.)*/
			on Originacao.CdParceiroComercial = input(BALCAODIG.CDPARCEIRO,best.)

WHERE Originacao.DtEntradaProposta >= "&a1dt."d
/*and Originacao.NuCnpjLoja in ('14439116000136','23722194000134','34697707000110','31392500000140','34656383000172','28812324000143','41149615000161','18249116000124','44391578000191')*/
and Originacao.NmGrupoProdutoMeta
	and Originacao.NmGrupoProdutoMeta
	in (
	'CP',
	'CP DESENROLA',
	'CVG Leves',
	'CVG Pesados',
	'Leves',
	'Leves-BVO',
	'Leves-Multibanco',
	'Motos',
	'Pesados'
/*	'FGTS',*/
/*	'Privado',*/
/*	'Saude'*/
	)
and Originacao.StProposta in ('A','AP','CA','PA','N')
;QUIT;

Proc sql;

/*lidi*/
Create table EGV_RecusaTargetEnvioEmail_Tlmkt as
Select *
From TbPropostas_Executivo
Where 1=1
AND NmGrupoSubProduto in ('CVG Leves','CVG Pesados')
AND FlParceiroDigital = 'S'
AND Motivo_selecao = 1
AND NmCanalEstruturaComercial IS NULL
AND YEAR(DtEntradaProposta) = YEAR(TODAY())
;quit;

PROC EXPORT
	DATA= EGV_RecusaTargetEnvioEmail_Tlmkt
	FILE="&dir_csv./EGV_RecusaTargetEnvioEmail_Tlmkt.csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;

PROC SQL;

CREATE TABLE EGV_dParceiroComercial as
select Distinct
compress(case when NmGrupoSubProduto contains 'CVG' THEN 'EGV' ELSE UPPER(NmGrupoSubProduto) END||"-"||put(CdParceiroComercial,best.)) AS KeyProduto
,NmGrupoSubProduto
,CdParceiroComercial
,NmParceiroComercial
,FlParceiroDigital
,FlBalcaoDigital
from TbPropostas_Executivo;quit;

PROC EXPORT
	DATA= EGV_dParceiroComercial
	FILE="&dir_csv./EGV_dParceiroComercial.csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;

PROC SQL;
   CREATE TABLE PARCEIROS_DIGITAIS_SUM AS
   SELECT
			NuAnoMes,
			DtEntradaProposta,
			DtContrato,
			NmGrupoSubProduto,
			CdParceiroComercial,
			NmParceiroComercial,
			DsTipoCanalProducao,
			Ds_motivo_recusa,
			NmCanalEstruturaComercial,
			NmAreaEstruturaComercial,
			FlParceiroDigital,
			FlBalcaoDigital,
			FilialBalcaoDigital,
			NomeBalcaoDIgital,
			FLPAP,
            (SUM(FlPropostaOriginada)) AS FlPropostaOriginada,
            (SUM(FlPropostaNegada)) AS FlPropostaNegada,
            (SUM(FlPropostaAprovada)) AS FlPropostaAprovada,
            (SUM(FlPropostaAPNP)) AS FlPropostaAPNP,
            (SUM(FlPropostaEfetivada)) AS FlPropostaEfetivada,
			(SUM(Motivo_Selecao)) AS FlPropostaMotRecusa,
			(SUM(FlConvMotRecusa)) AS FlConvMotRecusa,
            (SUM(FlPropostaBipartida)) FORMAT=6. AS FlPropostaBipartida,
            (SUM(FlPropostaUtilizacaoFgts)) FORMAT=6. AS FlPropostaUtilizacaoFgts,
            (SUM(FlContratoDesdobro)) AS FlContratoDesdobro,
            (SUM(VrPropostaOriginada)) FORMAT=COMMAX18.2 AS VrPropostaOriginada,
            (SUM(VrPropostaAprovada)) FORMAT=COMMAX18.2 AS VrPropostaAprovada,
            (SUM(VrProducao)) FORMAT=COMMAX18.2 AS VrProducao
      FROM WORK.TbPropostas_Executivo
		GROUP BY
			NuAnoMes,
			DtEntradaProposta,
			DtContrato,
			NmGrupoSubProduto,
			CdParceiroComercial,
			NmParceiroComercial,
			DsTipoCanalProducao,
			Ds_motivo_recusa,
			NmCanalEstruturaComercial,
			NmAreaEstruturaComercial,
			FlParceiroDigital,
			FlBalcaoDigital,
			FilialBalcaoDigital,
			NomeBalcaoDIgital,
			FLPAP
;
QUIT;

PROC EXPORT
	DATA= PARCEIROS_DIGITAIS_SUM
	FILE="&dir_csv./EGV_ONEPAGE_PARCEIROS_DIGITAIS_SUM.csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;
