PROC SQL;
CREATE TABLE CART_VCTOS_2 AS
	SELECT
		B.TIPO_CARTEIRA,

		A.DsGrupoEconomicoRisco,
		A.DsClienteRisco,
		A.DsSetorClienteRisco_novo,
		A.DsRatingCliente,
		A.DsStatus_Monitoramento,
		SUM(A.VrSaldo) FORMAT=COMMAX22.2 AS SUM_of_VrSaldo,
		SUM(A.VrProvisao) FORMAT=COMMAX22.2 AS SUM_of_VrProvisao,
		SUM(A.VrSaldoContabil) FORMAT=COMMAX22.2 AS SUM_of_VrSaldoContabil,

	LEFT JOIN LISTA_SEGMENTO_ANT AS B
	ON A.DsGrupoEconomicoRisco = B.DsGrupoEconomicoRisco

			B.TIPO_CARTEIRA,

			A.DsGrupoEconomicoRisco,
			A.DsClienteRisco,
			A.DsSetorClienteRisco_novo,
			A.DsRatingCliente,

b1.max_VrLimiteGrupoAprovado/&unidade. as Limite_Aprovado_BU,

d.sum_total as Saldo_Total,
	(case when c.status_limite eq 'VIGENTE' then
		(case when abs(b1.max_VrLimiteGrupoAprovado/&unidade.-(coalesce(b1.max_VrLimiteGrupoDisponivel/&unidade.,0))-d.sum_total)<0.01 and (d.sum_total>(1/&unidade.)) then 1 else
			(case when abs(c.Limite/&unidade.-(coalesce(c.Disponivel/&unidade.,0))-d.sum_total)<0.01 then 2 else 0 end) end) else -1 end)  as Validar
from populacao_limiteout as a

on a.Grupo=b1.DsGrupoEconomicoRisco

on a.Grupo=c.GrupoEconomico
left join tab as d
on a.Grupo=d.Grupo
left join verificar as e
on a.Grupo=e.Grupo;

(case when Validar=0 then . else
	(case when Validar=-1 then Saldo_Total else
		(case when Validar=1 then Limite_Aprovado_BU else
			(case when Validar=2 then Limite_Aprovado_OUT else
				(case when min(Limite_Aprovado_BU,Limite_Aprovado_OUT)>Saldo_Total then min(Limite_Aprovado_BU,Limite_Aprovado_OUT) else Saldo_Total end) end) end) end) end) as Limite_Aprovado_ORIG

(case when Validar=0 then . else
	(case when Validar=-1 then 0 else
		(case when Validar=1 then Limite_Disponivel_BU else
			(case when Validar=2 then Limite_Disponivel_OUT else
				(case when Limite_Aprovado_ORIG>Saldo_total then Limite_Aprovado_ORIG-Saldo_total else 0 end) end) end) end) end) as Limite_Disponivel_ORIG
from limites_out2;

proc sql;
create table BASE_DERIV as

		substr(a.Status_Monitoramento, length(a.Status_Monitoramento), 1) as nivel_monit
from CART_VENCIMENTOS_FINAL_DERIV as a

proc sql;
create table tab_deriv as
select grupo, 
max(nivel_monit) as max_nivel_monit,
max(nivel_rating) as max_nivel_rating,
sum(total) as sum_total,
sum(tvm) as sum_tvm,
sum(derivativos) as sum_derivativos,
sum(fianca_passivel) as sum_fianca_passivel,
sum(emprestimos) as sum_emprestimos,
sum(pdd) as sum_pdd,
sum(ajuste_mtm) as sum_ajuste_mtm
from BASE_DERIV
group by 1 ;
quit;

proc sql;
create table tab_deriv as
select a.*,
(-1)*a.sum_pdd/(coalesce(a.sum_fianca_passivel,0)+coalesce(a.sum_emprestimos,0)) as perct_pdd,
b.rating as pior_rating,
c.Status_Monitoramento as pior_monit,
a.sum_ajuste_mtm/(coalesce(a.sum_tvm,0)+coalesce(a.sum_derivativos,0)) as perct_ajuste_mtm
from tab_deriv as a
left join mapeamento_ratings as b
on a.max_nivel_rating=b.ID
left join CART_VENCIMENTOS_FINAL_DERIV as c
on a.Grupo = c.Grupo and a.max_nivel_monit = substr(c.Status_Monitoramento, length(c.Status_Monitoramento), 1)
order by Grupo;
quit;






proc sql;
create table classificacao_deriv as
select grupo,
sum(total) as sum_total,
sum(tvm) as sum_tvm,
sum(derivativos) as sum_derivativos,
sum(fianca_passivel) as sum_fianca_passivel,
sum(emprestimos) as sum_emprestimos
from BASE_DERIV
group by 1
order by calculated sum_total desc;
quit;
