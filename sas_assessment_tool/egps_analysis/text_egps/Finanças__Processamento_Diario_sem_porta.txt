			TODAY() as HOJE format ddmmyy10.,

- coalesce((ABS(ac.ResultadoTesouraria)+ABS(ac.ResultadoComercial)+ABS(ac.ResultadoAcionista)),0)) > 100

from Work.Agrup_analitico as ag
Left join Work.Agrup_consolidado ac

;

(Where = (Calcular = "Calcula" and Coalesce(Spread,0) NE 0 and Segmento_Mercado_SAS in ("CORPORATE", "CORRETORA", "MIDDLE", "PRIVATE", "INSTITUICAO FINANCEIRA", "LARGE", "ATIVOS ESPECIAIS"))) as

	When a.CdObjetoNegociacao NE "" Then Cats (a.cdVeiculoLegal, '||', a.cdModalidadeProduto, '||', a.CdObjetoNegociacao)
	else "Verificar"
End as ChaveUnica,

Case
	When a.CdSituacaoPosicaoContrato = 60 Then "Não Calcula"
	When b.Excecao = "Não Calcula" Then "Não Calcula"

	When Calculated Calcular = "Calcula" Then Sum (coalesce(a.AccrualTxOpeDm1,0)*(c.PctRiskSharingResultado/100) - coalesce(a.AccrualTxTirDm1,0)*(c.PctRiskSharingResultado/100)) * &CDI_DIA.

Create Table Work.Agrup_analitico as
select max(DataRef) as DataRef format ddmmyy10., GrupoConciliacao, sum(ResultadoDia) as ResultadoDia, sum(ResultadoTesouraria) as ResultadoTesouraria,

from BASESIC.TBREPOSITORIOANALITICO
where GrupoConciliacao not in ('Financeira', 'Swap Interareas')

Create Table Work.Agrup_consolidado as
select DataRef, GrupoConciliacao, sum(ResultadoDia) as ResultadoDia, sum(ResultadoTesouraria) as ResultadoTesouraria,

where GrupoConciliacao not in ('Financeira', 'Swap Interareas')
group by DataRef, GrupoConciliacao
order by GrupoConciliacao;

Create table Work.Validacao_Base as
select  ag.DataRef, ag.DataInicio,  ag.GrupoConciliacao,
Case When ABS((ABS(ag.ResultadoTesouraria)+ABS(ag.ResultadoComercial)+ABS(ag.ResultadoAcionista))
- coalesce((ABS(ac.ResultadoTesouraria)+ABS(ac.ResultadoComercial)+ABS(ac.ResultadoAcionista)),0)) > 100
then 'Diferença na base' else 'OK' end as Validacao
from Work.Agrup_analitico as ag
Left join Work.Agrup_consolidado ac
on ac.DataRef = ag.DataRef
and ac.GrupoConciliacao = ag.GrupoConciliacao
;
