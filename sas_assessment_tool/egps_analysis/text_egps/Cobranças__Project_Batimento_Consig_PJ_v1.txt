PROC SQL;
CREATE TABLE SELECAO_ATUAL AS SELECT

	CODWEB,
	EMPRESA,
	FLAG_ACORDO_ATIVO,
	FLAG_CONTENCIOSO,

	MIN(DT_VCT_CONVENIO) AS DT_VCT_CONVENIO_ATU FORMAT DDMMYY10.,
	MAX(ATRASO) AS MAIOR_ATRASO_ATU,

	SUM(VALOR_ATRASO_REPASSE_ATU) AS RISCO_ATRASO_ATU

FROM PRIVADO.CARGA_PJ_PRIVADO_&DATA_ARQ_ATU.
WHERE
	STATUS_COB IN ('ESTOQUE','ATUALIZACAO','NOVO','RETIRAR - PGTO')
AND
	FLAG_EMP_GRUPO='NAO'
AND
	FLAG_PROIBIDO='NAO'
AND
	FLAG_CONTENCIOSO='NAO'

GROUP BY
	CODWEB,
	EMPRESA,
	FLAG_ACORDO_ATIVO,
	FLAG_CONTENCIOSO;
QUIT;



PROC SQL;
CREATE TABLE SELECAO_ATUAL_CONTENCIOSO AS SELECT

	CODWEB,
	EMPRESA,
	FLAG_ACORDO_ATIVO,
	FLAG_CONTENCIOSO,

	MIN(DT_VCT_CONVENIO) AS DT_VCT_CONVENIO_ATU FORMAT DDMMYY10.,
	MAX(ATRASO) AS MAIOR_ATRASO_ATU,

	SUM(VALOR_ATRASO_REPASSE_ATU) AS RISCO_ATRASO_ATU

FROM PRIVADO.CART_CONTENCIOSA_&DATA_ARQ_ATU.
WHERE
	STATUS_COB IN ('ESTOQUE','ATUALIZACAO','NOVO')
AND
	FLAG_EMP_GRUPO='NAO'
AND
	FLAG_PROIBIDO='NAO'
AND
	FLAG_CONTENCIOSO='SIM'

GROUP BY
	CODWEB,
	EMPRESA,
	FLAG_ACORDO_ATIVO,
	FLAG_CONTENCIOSO;
QUIT;



DATA FOTO_CARTEIRA_PJ;
SET SELECAO_ATUAL
	SELECAO_ATUAL_CONTENCIOSO;
RUN;



PROC SQL;
CREATE TABLE FOTO_CARTEIRA_PJ AS SELECT distinct
	A.*,
	B.CNPJ,
	put(MAIOR_ATRASO_ATU,FX_ATRASO.) as FXATRASO

FROM FOTO_CARTEIRA_PJ A left JOIN PRIVADO.BASE_CADASTRO_EMPRESAS B
ON A.CODWEB=B.CODWEB;
QUIT;



DATA FOTO_CARTEIRA_PJ;
SET FOTO_CARTEIRA_PJ;

	IF  MAIOR_ATRASO_ATU >= 60 AND MAIOR_ATRASO_ATU =< 1790 AND FLAG_ACORDO_ATIVO = 'SEM ACORDO'
	AND RISCO_ATRASO_ATU > 5000 THEN FLAG_ELEG_BCO=1 ; ELSE FLAG_ELEG_BCO=0;

	FORMAT CNPJ_CONVENIO BEST15.;
	CNPJ_CONVENIO = INPUT(CNPJ,BEST15.);

RUN;



PROC SQL;
CREATE TABLE SELECAO_NEGATIVACAO AS SELECT DISTINCT

	A.*,

	CASE WHEN (B.CNPJ_CONVENIO IS NOT NULL) THEN 1 ELSE 0 END AS FLAG_NEGATIVADO_ORG

FROM  FOTO_CARTEIRA_PJ A LEFT JOIN ARQUIVO_ORG_PJ B
ON A.CNPJ_CONVENIO=B.CNPJ_CONVENIO;
QUIT;

PROC FREQ DATA=SELECAO_NEGATIVACAO;
TABLE FLAG_ELEG_BCO*FLAG_NEGATIVADO_ORG / LIST MISSING;
RUN;

DATA BA_RESULTADO_BATIMENTO;
SET SELECAO_NEGATIVACAO;

FORMAT DTREF DATA_NEGATIVACAO ;
DTREF=&DT_EXEC.;

FORMAT DECISAO $30.;

	IF (FLAG_ELEG_BCO=1 AND FLAG_NEGATIVADO_ORG=1) or (FLAG_ELEG_BCO=0 AND FLAG_NEGATIVADO_ORG=0)  THEN DO;

	DECISAO='OK';
END;

ELSE ;

	IF FLAG_ELEG_BCO=0 AND FLAG_NEGATIVADO_ORG=1 THEN DO;

		DECISAO='REABILITAR';
END;

ELSE ;

	IF FLAG_ELEG_BCO=1 AND FLAG_NEGATIVADO_ORG=0 THEN DO;

	DECISAO='NEGATIVAR';
END;

RUN;

PROC FREQ DATA=BA_RESULTADO_BATIMENTO;
TABLE DECISAO DECISAO*fxatraso / LIST MISSING;

RUN;

data CONFERENCIA_PF;
set BA_RESULTADO_BATIMENTO;
where DECISAO in ('NEGATIVAR','REABILITAR');
