  a.CT006_CD_OPESIS,
  b.CT005_DT_CADSIS

  cont.CT006 as a
  INNER JOIN cont.CT005 as b ON
      b.CT005_NR_INST  = a.CT006_NR_INST
  AND b.CT005_NR_AGEN  = a.CT006_NR_AGEN
  AND b.CT005_DT_MOVTO = a.CT006_DT_MOVTO
  AND b.CT005_NR_LOTE  = a.CT006_NR_LOTE

      c.CT001_SQ_CONTA = b.CT005_SQ_CONTA

    a.CT006_TP_LOTE = 'M'
AND b.CT005_DT_MOVTO BETWEEN 20250701 AND 20250731
AND b.CT005_NR_INST IN (51,6556,175,213,183) 
AND a.CT006_ST_LOTE = 9

  b.CT005_NR_INST,
  b.CT005_DT_MOVTO,

  b.CT005_VL_LANC;
