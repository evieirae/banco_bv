%include '/infogrc/utils/all-includes.sas';

OPTIONS COMPRESS=BINARY;
%LET TIMESTAMP = %SYSFUNC(DATE(),ddmmyy10.);
%LET TIMESTAMPDATA = %SYSFUNC(date(),YYMMDDN8.);
%LET TIMESTAMPHORA = %sysfunc(TIME(),hour4.);

/* ======================================================================================*/
/* =====================Gera tabela de leads MFS =========================*/
/* ======================================================================================*/

PROC SQL;
%CONECTADB(server=CTR);

  CREATE TABLE TbMFS AS
	SELECT *
	FROM CONNECTION TO DBCON
				(
SELECT
			pc.dtcadastro AS 'Data cadastro COBAN',
			pc.NmFantasia AS 'Razão Social do Parceiro',
			p.NuCpfCnpj AS 'CNPJ do Parceiro',
			pc.CdParceiroComercial AS 'Login',
			pc.FlAtivo AS 'Status Login',
			tpc.NmTipoParceiroComercial AS 'Tipo Parceiro',
			taf.DsAtividadeFinanciamento AS 'Nome da Atividade' ,
			psc.CdSubProduto AS 'Cod.Produto',
			sc.DsSubProduto AS 'Nome Produto',
			fa.dtcadastro AS 'Data Cadastro Favorecido',
			fa.NmTitular AS 'Razão Social Favorecido',
			fa.NuCpfCnpj AS 'CNPJ Favorecido',
			fcc.NuBanco AS 'Nro Banco Favorecido',
			fcc.CdAgencia AS 'Nro Agência Favorecido',
			fcc.NuContaCorrente AS 'Nro Conta Favorecido',
			fa.FlAtivo AS 'Status Favorecido',
			fcc.FlPrincipal AS 'Principal',
			fcc.FlContaServico AS 'Serviço',
			fcc.FlComissao AS 'Comissão',
			fcc.FlContrato AS 'Contrato',
			fcc.FlAtivoComissao AS 'Ativo Comissão',
			fcc.FlAtivoContrato 'Ativo Contrato',
			cgc.NmColaborador AS 'Gerente Comercial',
			cgr.NmColaborador AS 'Gerente Regional',
			rl.NmRepresentanteLegal AS 'Nome Sócio',
			rl.NuCpf AS 'CPF Sócio'

			/* table parceiro comercial*/
		FROM dbcor..TbParceiroComercial pc

			/*table regiao*/
		LEFT  JOIN DBCOR..TbRegiao r
			ON r.CdRegiao = pc.CdRegiao

			/*table filial*/
		LEFT  JOIN DBCOR..TbFilial f
			ON r.CdFilial = f.CdFilial

			/* Gerente Comercial*/
		LEFT  JOIN DBCOR..TbColaborador cgc
			ON cgc.CdColaborador = f.CdGerenteComercial

			/* Gerente Regional*/
		LEFT  JOIN DBCOR..TbColaborador cgr
			ON cgr.CdColaborador = f.CdGerenteRegional

			/* Representante legal*/
		LEFT  JOIN DBCOR..TbRepresentanteLegal rl
			ON rl.CdParceiroComercial = pc.CdParceiroComercial

			/* Sócios*/
		LEFT  JOIN DBCOR..TbSocio s
			ON s.CdParceiroComercial = pc.CdParceiroComercial
			AND s.SqRepresentanteLegal = rl.SqRepresentanteLegal

			/*table pessoa*/
		LEFT  JOIN DBCOR..TbPessoa p
			ON pc.CdPessoa = p.CdPessoa

			/*table tipo parceiro comercial */
		LEFT  JOIN DBCOR..TbTipoParceiroComercial tpc
			ON tpc.CdTipoParceiroComercial = pc.CdTipoParceiroComercial

			/*table atividade financeira parc comercial*/
		LEFT JOIN DBCOR..TbAtividFinancParcComercial af
			ON af.CdParceiroComercial = pc.CdParceiroComercial

			/*table atividade financiamento*/
		LEFT JOIN DBCOR..TbAtividadeFinanciamento taf
			ON af.CdAtividadeFinanciamento = taf.CdAtividadeFinanciamento

			/*table favorecido */
		LEFT JOIN DBCOR..TbFavorecidoAutorizado fa
			ON fa.CdParceiroComercial = pc.CdParceiroComercial

			/*table favorecido conta corrente */
		LEFT JOIN DBCOR..TbFavorecidoContaCorrente fcc
			ON fcc.CdParceiroComercial = fa.CdParceiroComercial

			/*table Produtos*/
		LEFT JOIN DBCOR.dbo.TbParceiroSubProduto psc
			ON psc.CdParceiroComercial = pc.CdParceiroComercial

			/*table De/Para Produtos*/
		LEFT JOIN DBCOR.dbo.TbSubProduto sc
			ON sc.CdSubProduto = psc.CdSubProduto

		WHERE (1=1)
			AND pc.FlAtivo = 'S' /*flag parceiro ativo/inativo*/
			AND pc.flconsignadoconvenio = 'N' /*flag exclusao consignado*/
			AND pc.CdTipoParceiroComercial in (5) /*tipo de parceiro*/
			AND pc.FlParceiroVirtual = 'N' /*flag exclusão parceiro virtual*/
			AND taf.cdatividadefinanciamento in (7) /*tipo atividade financiamento*/
			AND fcc.SqFavorecido = fa.SqFavorecido
			AND psc.FlAtivo = 's'
			AND pc.NmFantasia = 'MEU FINANCIAMENTO SOLAR LTDA'

		ORDER BY pc.dtcadastro);
RUN;

%let dir_csv = /riscosbvf/BA/processos/output;

proc export

  OUTFILE= "&dir_csv./Base_MFS_&TIMESTAMPDATA..csv"
  replace;
