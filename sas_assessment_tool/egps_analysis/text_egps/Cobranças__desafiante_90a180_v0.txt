		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas)  > 0.9 		THEN "91,1-100%"
		ELSE "ERRO"
	END AS PERCENT_PAGO

FROM BASE_ELEG AS A

LEFT JOIN CONNECTION TO PERM(
	SELECT DISTINCT
		A.NuContrato,
		MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro,
		Date(A.DtProcessamento)		AS Data_Execucao,
		A.CdMotivoContratoFinanceiro,
		MAX(A.nudiasatraso) 		AS NuDiasAtraso,
		A.CdSituacaoContabil,
		CASE
			WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
			ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
		END AS SALDO_M0

	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

	ON A.NUCONTRATO = J.NUCONTRATO
	AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

	WHERE DATE(A.DtProcessamento) = &fim_mes_m0.
	AND A.CdProduto IN(12, 14)
	AND (A.NuDiasAtraso > 90 AND A.NuDiasAtraso <= 360)


	GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato, A.CdMotivoContratoFinanceiro, Data_Execucao

ON A.NUCONTRATO = INPUT(B.NuContrato, BEST32.)

				SUM(CASE WHEN st_parcela='Q' AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END)        AS Parcelas_Pagas,
                SUM(CASE WHEN st_parcela='A' OR (st_parcela='Q' AND dt_ultimo_recebimento > &fim_mes_m0.) THEN 1 ELSE 0 END)        AS Parcelas_Abertas,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) < 30 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_30d,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) BETWEEN 1 AND 90 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_90d,

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G

ON A.NuContrato = INPUT(G.cd_contrato,BEST32.) AND A.SqContratoFinanceiro = G.sq_contrato_financeiro

PROC SQL;

      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=20);
CREATE TABLE BASE_ELEG_SEGMENTACAO AS
SELECT
	A.*,
	CASE
		WHEN MOD90.IDChaveModelo IS NOT NULL 		THEN INPUT(MOD90.GH, BEST32.)
		WHEN MOD120.CdContratoOrigem_2 IS NOT NULL	THEN INPUT(MOD120.GH, BEST32.)
	END AS GH_COLLECTION,

	CASE
		WHEN SAFRADO60.IDChaveModelo IS NOT NULL 			THEN SAFRADO60.GH
		WHEN CANIB_120.CdContratoOrigem_2 IS NOT NULL		THEN CANIB_120.GH
	END AS GH_DESAF,

	CASE
		WHEN CANIB_90B.CdContratoOrigem_2 IS NOT NULL 		THEN INPUT(CANIB_90B.GH, BEST32.)
		WHEN CANIB_120B.CdContratoOrigem_2 IS NOT NULL		THEN INPUT(CANIB_120B.GH, BEST32.)
	END AS GH_CAMP,

	B.SALDO_M0,

	G.Parcelas_pagas_ate_30d,
	CASE
		WHEN G.Parcelas_pagas_ate_30d <= 2  THEN '1. 0 a 2'
		WHEN G.Parcelas_pagas_ate_30d <= 4  THEN '2. 3 a 4'
		WHEN G.Parcelas_pagas_ate_30d <= 7  THEN '3. 5 a 7'
		WHEN G.Parcelas_pagas_ate_30d <= 14 THEN '4. 8 a 14'
		WHEN G.Parcelas_pagas_ate_30d >  14 THEN '5. 15+'
	END AS GROUP_Parc_Pag_30d,

	CASE
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.05 		THEN "00,0-05,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.1 		THEN "05,1-10,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.2 		THEN "10,1-20,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.3 		THEN "20,1-30,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.4 		THEN "30,1-40,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.5 		THEN "40,1-50,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.6 		THEN "50,1-60,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.7 		THEN "60,1-70,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.8 		THEN "70,1-80,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.9 		THEN "80,1-90,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas)  > 0.9 		THEN "91,1-100%"
		ELSE "ERRO"
	END AS PERCENT_PAGO,

	G.Parcelas_pagas_ate_90d/(G.Parcelas_Abertas + G.Parcelas_Pagas) as PCT_PG_ATE_90D,

	CASE
		WHEN B.SALDO_M0 <= 10000 	THEN "01. ATÉ 10K"
		WHEN B.SALDO_M0 <= 20000 	THEN "02. ATÉ 20K"
		WHEN B.SALDO_M0 <= 30000 	THEN "03. ATÉ 30K"
		WHEN B.SALDO_M0 <= 45000 	THEN "04. ATÉ 45K"
		WHEN B.SALDO_M0  > 45000 	THEN "05. ACIMA DE 45K"
	END AS FX_SALDO

FROM BASE_ELEG AS A

LEFT JOIN CONNECTION TO PERM(
	SELECT DISTINCT
		A.NuContrato,
		MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro,
		Date(A.DtProcessamento)		AS Data_Execucao,
		A.CdMotivoContratoFinanceiro,
		MAX(A.nudiasatraso) 		AS NuDiasAtraso,
		A.CdSituacaoContabil,
		CASE
			WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
			ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
		END AS SALDO_M0

	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

	LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
	ON A.NUCONTRATO = J.NUCONTRATO
	AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

	WHERE DATE(A.DtProcessamento) = &fim_mes_m0.
	AND A.CdProduto IN(12, 14)
	AND (A.NuDiasAtraso > 90 AND A.NuDiasAtraso <= 360)


	GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato, A.CdMotivoContratoFinanceiro, Data_Execucao

) AS B
ON A.NUCONTRATO = B.NuContrato

LEFT JOIN CONNECTION TO PERM(
SELECT DISTINCT cd_contrato,
             	sq_contrato_financeiro,
				SUM(CASE WHEN st_parcela='Q' AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END)        AS Parcelas_Pagas,
                SUM(CASE WHEN st_parcela='A' OR (st_parcela='Q' AND dt_ultimo_recebimento > &fim_mes_m0.) THEN 1 ELSE 0 END)        AS Parcelas_Abertas,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) < 30 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_30d,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) BETWEEN 1 AND 90 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_90d,

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G
ON A.NuContrato = G.cd_contrato AND A.SqContratoFinanceiro = G.sq_contrato_financeiro



LEFT JOIN collection_90_120 MOD90
ON INPUT(A.NuContrato, BEST32.) = MOD90.IDChaveModelo


LEFT JOIN collection_120_360 MOD120
ON INPUT(A.NuContrato, BEST32.) = MOD120.CdContratoOrigem_2


LEFT JOIN SAFRADO_60 SAFRADO60
ON INPUT(A.NuContrato, BEST32.) = SAFRADO60.nucontrato_2


LEFT JOIN MODELO_CANIB_120A180 CANIB_120
ON INPUT(A.NuContrato, BEST32.) = CANIB_120.CdContratoOrigem_2


LEFT JOIN MODELO_CANIB_90A120_BUREAU CANIB_90B
ON INPUT(A.NuContrato, BEST32.) = CANIB_90B.CdContratoOrigem_2


LEFT JOIN MODELO_CANIB_120A180_BUREAU CANIB_120B
ON INPUT(A.NuContrato, BEST32.) = CANIB_120B.CdContratoOrigem_2

;QUIT;

CREATE TABLE apuracao_maior90 AS
SELECT
	A.*,
	B.atraso_m0,
	B.sqcontratofinanceiro_m0,
	C.atraso_m1,
	C.sqcontratofinanceiro_m1,
	C.DtProcessamentoLiquidacao

FROM BASE_ELEG_SEGMENTACAO AS A

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			A.Nucontrato,
			MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro_m0,
			MAX(A.nudiasatraso) as atraso_m0,
			CASE
				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO_M0

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE DATE(A.dtprocessamento) = &fim_mes_m0.
		GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato
) AS B
ON A.nucontrato = B.NUCONTRATO

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			Nucontrato,
			max(sqcontratofinanceiro) as sqcontratofinanceiro_m1,
			max(nudiasatraso) as atraso_m1,
			max(DtProcessamentoLiquidacao) as DtProcessamentoLiquidacao
		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) = &fim_mes_m1.
		group by nucontrato
) AS C
ON A.nucontrato = C.NUCONTRATO;
QUIT;


DATA apuracao_maior90_&anomes.;
SET apuracao_maior90;
FORMAT RESULTADO $15.;
IF atraso_m1 = 0 AND sqcontratofinanceiro_m1 > sqcontratofinanceiro_m0 		THEN RESULTADO = "Reneg"; ELSE
IF DtProcessamentoLiquidacao NE .		 									THEN RESULTADO = "Quitacao"; ELSE
IF nudiasatraso > atraso_m1 												THEN RESULTADO = "Pagamento";
ELSE RESULTADO = "Sem acao";
RUN;

PROC SORT DATA= apuracao_maior90_&anomes. NODUPKEY;
BY NUCONTRATO;
RUN;

CREATE TABLE WORK.RENEG_SIMULADOR_&ANOMES. AS
SELECT *

FROM CONNECTION TO PERM(

WITH QUERY_1 AS (


SELECT DISTINCT
	nucontratogestao,
	sqcontratofinanceiro,
	nucontratogestaonovo,
	CASE
		WHEN nucontratogestao = nucontratogestaonovo THEN sqcontratofinanceiro + 1
		ELSE 1
	END AS SQCONTRATONOVO,
	dt_baixa,
	dt_reneg,
	CAST(vr_presente AS NUMERIC) AS vr_presente,
	CAST(vr_financiado AS NUMERIC) AS vr_financiado,
	qt_parcela,
	vr_prestacao,
	vr_desconto_mora,
	vr_desconto_multa,
	vr_desconto_principal,
	tx_pactuada
FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
WHERE dt_baixa >= &dt_inicio. AND dt_baixa < &dt_fim.
	AND (dt_cancelamento IS NULL OR dt_reativacao IS NOT NULL)
	AND st_reneg = 3

),
QUERY_2 AS (


SELECT
    A.NuContratoGestao AS NuContrato,
    A.SqContratoFinanceiro,
    A.NuContratoGestaoNovo,
    A.SQCONTRATONOVO,
    A.dt_baixa,
    A.dt_reneg,
    A.vr_presente,
    A.vr_financiado,
    A.qt_parcela,
    A.vr_prestacao,
    A.vr_desconto_mora,
    A.vr_desconto_multa,
    A.vr_desconto_principal,
    A.Tx_Pactuada,
    MIN(B.vr_parcela) AS VrParcela_Anterior,
    DATE_DIFF(CAST(A.dt_reneg AS DATE), MIN(CASE
        WHEN B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND CAST(B.dt_ultimo_recebimento AS DATE) >= CAST(A.dt_reneg AS DATE)) THEN B.dt_vencimento
        ELSE NULL
    END), DAY) AS Atraso,
    SUM(CASE
        WHEN B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND CAST(B.dt_ultimo_recebimento AS DATE) >= CAST(A.dt_reneg AS DATE)) THEN B.vr_parcela
        ELSE 0
    END) AS Risco_Aberto,
    COUNT(CASE
        WHEN B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND CAST(B.dt_ultimo_recebimento AS DATE) >= CAST(A.dt_reneg AS DATE)) THEN B.vr_parcela
        ELSE NULL
    END) AS Qtd_Parcelas_Abertas,
    SUM(CASE
        WHEN (B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND (CAST(B.dt_ultimo_recebimento AS DATE)) >= CAST(A.dt_reneg AS DATE)))
        AND (B.dt_vencimento) < CAST(A.dt_baixa AS DATE) THEN B.vr_parcela
        ELSE 0
    END) AS Risco_Aberto_em_Atraso,
    COUNT(CASE
        WHEN (B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND (CAST(B.dt_ultimo_recebimento AS DATE)) >= CAST(A.dt_reneg AS DATE)))
        AND (B.dt_vencimento) < CAST(A.dt_reneg AS DATE) THEN B.vr_parcela
        ELSE NULL
    END) AS Qtd_Parcelas_em_Atraso,
    MIN(CASE
        WHEN B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND CAST(B.dt_ultimo_recebimento AS DATE) >= CAST(A.dt_reneg AS DATE)) THEN B.dt_vencimento
        ELSE NULL
    END) AS DT_VENCIMENTO_NOVO,
    MIN(CASE
        WHEN B.st_parcela <> 'Q'
        OR (B.st_parcela = 'Q'
        AND CAST(B.dt_ultimo_recebimento AS DATE) >= CAST(A.dt_reneg AS DATE)) THEN B.nu_parcela
        ELSE NULL
    END) AS PARCELA,
FROM
    QUERY_1 AS A

    bv-pdbd-prd.dvry_core.tbparcela AS B

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=25);
CREATE TABLE BASE_SALDO AS
SELECT
	A.*

FROM CONNECTION TO PERM(
		SELECT
			A.NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			CASE
				WHEN B.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO_M0

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS B
		ON
		A.NUCONTRATO = B.NUCONTRATO
		AND A.SqContratoFinanceiro = B.SqContratoFinanceiro

		WHERE

			DATE(A.DTPROCESSAMENTO) = &fim_mes_M0.

		GROUP BY
			A.NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			B.FlRestruturacaoContrato
		) AS A
;QUIT;

proc sort data=BASE_SALDO out=BASE_SALDO NODUPKEY;
   by NUCONTRATO;
run;
