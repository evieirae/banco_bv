%include '/projetos/MISV/sas-misv-base-lib/scripts/ConectaDB.sas';

Libname intgr '/riscosbvf/mis_varejo/integracao';
libname Inpo '/infogrc/batidao_leves/Import/';

PROC SQL;
CREATE TABLE TbColaboradorGr AS
SELECT
	NuAnoMes
	,DsLoginN1	AS DsLogin
	,NmGrupoCargo
FROM  bases.TbEstruturaComercial
WHERE NuAnoMes >= &safraM7.
AND DsLoginN1 <> ''
AND NmNivel = 'N1'
;run;

PROC SQL;
CREATE TABLE TbColaboradorGf AS
SELECT
	NuAnoMes
	,DsLoginN2A	AS DsLogin
	,NmGrupoCargo
FROM  bases.TbEstruturaComercial
WHERE NuAnoMes >= &safraM7.
AND DsLoginN2A  <> ''
AND NmNivel = 'N2'
AND NmFilial NOT like 'CP %'
;run;

data TbColaborador;
set
	TbColaboradorGr
	TbColaboradorGf
;run;

proc sort data=TbColaborador;by descending NuAnoMes  ;run;

proc sort data=TbColaborador  nodupkey;by DsLogin;run;

proc export
data=TbColaborador
file='/riscosbvf/BA/processos/tmp/TbColaborador.csv' replace
dbms=dlm;
delimiter=';';
run;

X "mv  /riscosbvf/BA/processos/tmp/TbColaborador.csv  /riscosbvf/BA/processos/output/ ";

PROC SQL;
CREATE TABLE Arq_Insights AS
	SELECT

		TbColaborador.DsLogin
		,TbColaborador.NmGrupoCargo
		,CASE WHEN TbMensagem.Data = . THEN TODAY() ELSE TbMensagem.Data END AS Data FORMAT DDMMYY10.
		,CASE WHEN TbMensagem.Analise = '' THEN 'Vamos fortalecer a participação no jogo e conversão, aumentando nossa Originação!' ELSE TbMensagem.Analise END AS Analise
		,CASE WHEN TbMensagem.Ação = '' THEN 'O foco tático está na prospecção. Vamos aumentar nossa base de parceiros!' ELSE TbMensagem.Ação END AS Acao

	FROM TbColaborador TbColaborador
	LEFT JOIN Inpo.TbMensagem TbMensagem
	ON TbColaborador.DsLogin = TbMensagem.Login
;Run;

PROC SQL;
CREATE TABLE Arq_Insights_Matriz AS
	SELECT DISTINCT

		 '' AS DsLogin
		,'Matriz' AS NmGrupoCargo
		,TODAY() AS Data FORMAT DDMMYY10.
		,'A geração de análises e ações específicas para PDVs que necessitam de atenção já está disponível para o GR' AS Analise
		,'Fornecer recomendações baseadas em análises para otimizar a operação nos PDVs.' AS Acao

	FROM TbColaborador TbColaborador
;Run;

data Arq_Insights;
set
	Arq_Insights
	Arq_Insights_Matriz
;run;

proc export
data=Arq_Insights
file='/riscosbvf/BA/processos/tmp/Arq_Insights.csv' replace
dbms=dlm;
delimiter=';';
run;

X "mv  /riscosbvf/BA/processos/tmp/Arq_Insights.csv  /riscosbvf/BA/processos/output/ ";
