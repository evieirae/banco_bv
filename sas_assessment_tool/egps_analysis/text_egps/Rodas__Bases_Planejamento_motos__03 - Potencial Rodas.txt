%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table BASE_DTREF_M2 as
select * from connection to dbcon
(

WITH

	 TbDimTempoData AS
	(
		SELECT
			 DtTempo
			,NuAnoMes
			,NuAnoMesDia
			,CASE WHEN TempoData.DtTempo = '20201231' THEN 'N'
			      WHEN TempoData.DtTempo IN ('20211224','20211231','20241231') THEN 'N'
			      ELSE TempoData.FlDiaUtil END AS FlDiaUtil
			,CASE WHEN TempoData.NuAnoMes IN (202412) THEN 20 ELSE TempoData.QtTotalDiaUtilMes END AS QtTotalDiaUtilMes
			,NmDiaSemana
			,SgDiaSemana
			,CASE WHEN FlDiaUtil = 'S' THEN 1
                  WHEN NuDiaSemana = 7 AND FlFeriadoNacional = 'N' THEN 0.4
                  ELSE 0 END AS FlDiaUtilSab
		FROM DWBI_DIME.TbDimTempoData TempoData

	)

	,TbTotalDUMesSab AS
    (
        SELECT
             NuAnoMes,
             SUM(FlDiaUtilSab) AS QtTotalDiaUtilMes
        FROM TbDimTempoData
        GROUP BY NuAnoMes
    )

	,TbStData AS
	(
		SELECT
			 1 StData
			,max(CASE WHEN TempoData.FlDiaUtil = 'S' THEN TempoData.DtTempo END) AS DtTempo

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20240101'
		  AND TempoData.DtTempo <= &DtAtualizacao.
		GROUP BY
			1
	)
	,TbStMes AS
	(
		SELECT
			 NuAnoMes
			,1 StMes
			,max(CASE WHEN TempoData.FlDiaUtil = 'S' THEN TempoData.DtTempo END) AS DtTempo

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20240101'
		  AND TempoData.DtTempo <= &DtAtualizacao.
		GROUP BY
			1,
			NuAnoMes
	)
	,TbDiaUteisMaturacao AS
	(
		SELECT
			 DtTempo
			,sum(CASE WHEN TempoData.FlDiaUtil = 'S' THEN 1 END) over (ORDER BY CAST(DtTempo AS  INTEGER) desc) AS QtDiasUteisMaturacao

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20240101'
		  AND TempoData.DtTempo <= &DtAtualizacao.
	)

	,TbNuAnoMes AS

	(

		SELECT

			 TempoData.NuAnoMes
			,max(TempoData.QtTotalDiaUtilMes) AS QtTotalDiaUtilMes
			,sum(CASE WHEN TempoData.FlDiaUtil = 'S' THEN 1 ELSE 0 END) AS QtCorridosDiaUtilMes
		    ,sum(FlDiaUtilSab) AS QtCorridosDiaUtilMesSab
		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20240101'
		  AND TempoData.DtTempo <= &DtAtualizacao.
		GROUP BY
			 TempoData.NuAnoMes

	),TbCorridosDiaUtilDia AS
	(
		SELECT
			 DtTempo
			,sum(CASE WHEN FlDiaUtil = 'S' THEN 1 ELSE 0 END) over (partition BY NuAnoMes ORDER BY cast(DtTempo as integer)) AS QtCorridosDiaUtilDia
		FROM TbDimTempoData
		WHERE 1 = 1
		AND DtTempo >= '20240101'
		AND DtTempo <= &DtAtualizacao.
	)

		SELECT DISTINCT

			 TempoData.NuAnoMes
			,NuAnoMes.QtCorridosDiaUtilMes
			,NuANoMes.QtTotalDiaUtilMes
			,(NuAnoMes.QtCorridosDiaUtilMes/NuANoMes.QtTotalDiaUtilMes) as QtRacionalDuTotal
			,NuAnoMesSab.QtTotalDiaUtilMes AS QtTotalDiaUtilMesSab
			,NuAnoMes.QtCorridosDiaUtilMesSab
			,(NuAnoMes.QtCorridosDiaUtilMesSab / NuAnoMesSab.QtTotalDiaUtilMes) as QtRacionalDuSabTotal

		FROM TbDimTempoData TempoData

			LEFT JOIN TbStData StData
				ON  TempoData.DtTempo = StData.DtTempo

			LEFT JOIN TbStMes StMes
				ON  TempoData.DtTempo = StMes.DtTempo

			LEFT JOIN TbNuAnoMes NuAnoMes
				ON  TempoData.NuAnoMes = NuANoMes.NuAnoMes

			LEFT JOIN TbDiaUteisMaturacao DiaUteisMaturacao
				ON  TempoData.DtTempo = DiaUteisMaturacao.DtTempo

			LEFT JOIN TbCorridosDiaUtilDia CorridosDiaUtilDia
				ON  TempoData.DtTempo = CorridosDiaUtilDia.DtTempo

			LEFT JOIN TbTotalDUMesSab NuAnoMesSab
			    ON  TempoData.NuAnoMes = NuAnoMesSab.NuAnoMes

		WHERE 1 = 1
		  AND TempoData.DtTempo >='20240101'
		  AND TempoData.DtTempo <= &DtAtualizacao.

)
;run;

CREATE TABLE MKT_SHARE_IDEAL_M AS
SELECT

Chave1,
/*ChaveCdParceiro_FxIdade,*/
SUM(A.QtdContratos_Proj) as ctt,
SUM(A.qtgravamesplayers_Proj) as grav,

SUM(A.QtdContratos_Proj) as ctt_Sab,

SUM(A.QtdContratos_Proj)/SUM(A.qtgravamesplayers_Proj) AS Mkt_Share_Proj_1 format commax20.6,
SUM(A.QtdContratos_ProjSab)/SUM(A.qtgravamesplayers_ProjSab) AS Mkt_Share_ProjSab_1 format commax20.6,

	WHEN CALCULATED Mkt_Share_Proj_1 > 0.5 THEN 0.5

CASE
	WHEN CALCULATED Mkt_Share_ProjSab_1 > 0.5 THEN 0.5
	ELSE CALCULATED Mkt_Share_ProjSab_1
END AS Mkt_Share_ProjSab format commax20.6
