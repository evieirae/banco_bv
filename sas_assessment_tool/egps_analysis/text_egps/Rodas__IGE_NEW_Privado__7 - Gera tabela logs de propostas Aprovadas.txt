PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.Propostas_Aprovadas_0 AS
	SELECT *
	FROM CONNECTION TO DBCON
				(

SELECT
cast(a.dtStatusProposta as date) AS dtaprovacao,
month(a.dtStatusProposta)||year(a.dtStatusProposta) AS mesano,
b.cdloja,
a.nuproposta,
b.VrProposta
FROM DBA.TbLogStatusProposta a
INNER JOIN DBA.TbProposta b
ON a.nuproposta = b.nuproposta
WHERE b.CdSubProduto = 12
AND a.stproposta = 'A'
and year(a.dtStatusProposta) >= year(getdate())-1
/*AND b.cdloja in (&cobans.)*/
/*GROUP BY cast(a.dtStatusProposta as date), month(a.dtStatusProposta)||year(a.dtStatusProposta),b.cdloja*/
ORDER BY cast(a.dtStatusProposta as date) DESC);RUN;

proc sql;

create table tempw.Propostas_Aprovadas_1 as

select dtaprovacao,
mesano,
cdloja,
nuproposta,
VrProposta format commax12.2  as VrProposta
from tempw.Propostas_Aprovadas_0;RUN;

proc sql;

create table tempw.Propostas_Aprovadas as

select dtaprovacao,
mesano,
cdloja,
count(nuproposta) as ttl_pp_aprovadas,
sum(VrProposta) format commax12.2  as vlr_pp_aprovadas

from tempw.Propostas_Aprovadas_1
GROUP BY dtaprovacao, mesano, cdloja;RUN;

%let dir_csv = /riscosbvf/BA/processos/output;

proc export
	data=tempw.Propostas_Aprovadas
	dbms=csv
	OUTFILE= "&dir_csv./Privado_Propostas_Aprovadas.csv"
	replace;
run;
