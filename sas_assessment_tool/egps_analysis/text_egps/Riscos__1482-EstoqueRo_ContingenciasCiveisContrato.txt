					 when t1.MotivoPadrao = 'NOVA CET' 				then 'Nova CET'

				 t1.Produto2,
				 t1.Flag_Dep,

				case when t1.UF = '' then 'ND' else t1.UF end as UF,
				case when t1.Municipio = ' ' then 'ND' else t1.Municipio end as Municipio,
			   -t1.VlPrincipalINPC	format=commax22.2 	as Perda,
			   -t1.VlPrincipal	format=commax22.2 	as Perda_sem_INPC,
				t1.PROCURADOR,
				t1.Subsidios

						t2.DenVara NOT CONTAINS 'CEJUSC'
		
		
		group by Database, t1.NaturezaAcao, t1.OrgLegalProv, CALCULATED MotivoPadrao, t1.Motivo,

PROC SQL;
   CREATE TABLE WORK.SUMARIO (COMPRESS=YES) AS
   SELECT t1.SAFRA_CTO,
		  t1.ANO_CTO,
		  t1.PRODUTO,
		  T1.UF,
          t1.MotivoPadrao,
          t1.Motivo,
          t1.Submotivo,
		  T1.SAFRA_CADASTRO,
          t1.IntMesEnt,
          
            (COUNT(t1.NuContrato)) AS QT_CONTRATOS,
          
            (COUNT(t1.Cd_Causa)) AS QT_CAUSAS
      FROM work.SUMARIO_2010 t1
      GROUP BY t1.SAFRA_CTO,
			   t1.ANO_CTO,
			   t1.PRODUTO,
               t1.MotivoPadrao,
               t1.Motivo,
               t1.Submotivo,
			   T1.SAFRA_CADASTRO,
		       T1.UF,
               t1.IntMesEnt;
QUIT;

PROC SQL;
create table work.total_safra as
select ANO_CTO,
		SUM(QT_CONTRATOS) AS PRODUCAO,
   		CASE WHEN ANO_CTO=2010 THEN INTCK("MONTH",'1JAN11'D,TODAY())
		WHEN ANO_CTO=2011 THEN INTCK("MONTH",'1JAN12'D,TODAY())
		WHEN ANO_CTO=2012 THEN INTCK("MONTH",'1JAN13'D,TODAY())
		WHEN ANO_CTO=2013 THEN INTCK("MONTH",'1JAN14'D,TODAY())
		WHEN ANO_CTO=2014 THEN INTCK("MONTH",'1JAN15'D,TODAY())
		WHEN ANO_CTO=2015 THEN INTCK("MONTH",'1JAN16'D,TODAY())
		WHEN ANO_CTO=2016 THEN INTCK("MONTH",'1JAN17'D,TODAY())
		WHEN ANO_CTO=2017 THEN INTCK("MONTH",'1JAN18'D,TODAY())
		WHEN ANO_CTO=2018 THEN INTCK("MONTH",'1JAN19'D,TODAY())
		WHEN ANO_CTO=2019 THEN INTCK("MONTH",'1JAN20'D,TODAY())
   		WHEN ANO_CTO=2020 THEN INTCK("MONTH",'1JAN21'D,TODAY())
		WHEN ANO_CTO=2021 THEN INTCK("MONTH",'1JAN22'D,TODAY())
		WHEN ANO_CTO=2022 THEN INTCK("MONTH",'1JAN23'D,TODAY())
		WHEN ANO_CTO=2023 THEN INTCK("MONTH",'1JAN24'D,TODAY())
		WHEN ANO_CTO=2024 THEN INTCK("MONTH",'1JAN25'D,TODAY())
		WHEN ANO_CTO=2025 THEN INTCK("MONTH",'1JAN26'D,TODAY())
   		WHEN ANO_CTO=2026 THEN -1
		ELSE . END AS Max_IntMesEntrada_safra
FROM WORK.SUMARIO
GROUP BY ANO_CTO;
QUIT;

PROC SQL;
create table work.BD_ACOES_CIVEIS_DE_CONTRATOS AS
SELECT T1.SAFRA_CTO,
		T1.ANO_CTO,
		T1.PRODUTO,
		T1.UF,
		T1.MotivoPadrao,
		T1.Motivo,
		T1.Submotivo,
		T1.SAFRA_CADASTRO,
		T1.IntMesEnt,
		T1.QT_CONTRATOS,
		T1.QT_CAUSAS,
		T2.PRODUCAO AS Produção,
		case
			when T1.IntMesEnt=. then 0
			when T2.Max_IntMesEntrada_safra < T1.IntMesEnt then 0 else 1 end as Max_IntMesEntrada_safra
		from WORK.SUMARIO T1,
			work.total_safra T2
		where T1.ANO_CTO = T2.ANO_CTO;
QUIT;
