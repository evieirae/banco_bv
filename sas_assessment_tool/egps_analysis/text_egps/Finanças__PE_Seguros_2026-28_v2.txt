%let AnoMes_i2 = '01Jan2025'd;


proc sql;
	create table historico_PremioLA as
		select
			Dtcontrato-day(Dtcontrato)+1 as data format date9.,
			sum(VrBVMais) as Premio_Bruto
		from mis2.tbproducao
		where Dtcontrato >= '01jan2019'd and Dtcontrato < &AnoMes_i2.
		group by 1
;quit;

proc sql;
	create table premio_LA as
	select
		data,
		sum(Premio_bruto) as Premio_Bruto
	from work.base
	where  seguro ='SPF'  and Data >= &AnoMes_i2.
 	and Produto_Financiado in ('CVG Leves', 'Leves', 'Motos')
	group by 1
;quit;


data LA;

	ano=year(data);
run;


proc sql;
	create table LA_mob as
		select
			a.*,
			b.mob,
			intnx('month',a.data,b.mob,'s') as dataref format date9.,
			b.Curva,
			a.Premio_bruto*-b.Curva*0.944212 as LA_BRUTA
		from work.la as a
		left join Work.CurvaLA as b on (a.ano=b.versao)
		order by 1,4;
	quit;

proc sql;
create table base_spf as select
*,
Premio_Bruto/sum(Premio_Bruto) as Driver_LA,
Case When Produto_Financiado in ('Leves', 'Motos', 'Pesados') AND Negocio <> 'Banco Digital' then Premio_Bruto / SUM(case when Produto_Financiado in ('Leves', 'Motos', 'Pesados') AND Negocio <> 'Banco Digital' then Premio_Bruto else 0 end) end as Driver_Cancelamento
from work.base_spf
group by data
;
quit;

proc sql;
	create table base_spf_la as
		select
			a.*,
			a.Driver_La*coalesce(b.LA_Total,0) as LA_total,
			(a.Driver_Cancelamento*coalesce(b.Canc_novo_prod,0)) as Canc_novo,
			coalesce(CANCELAMENTO,0) as CANCELAMENTOasis
		from work.base_spf a
		left join work.LA_resumo2 as b on (a.data=b.data)
;quit;

proc sql;
	Create table BaseCancelado01 as
		select
			a.negocio,
			a.Produto,
			a.Seguro,
			a.Data,
			a.Produto_seguro,
			a.Produto_Financiado,
			a.CANAL,
			Canc_novo,
			CANCELAMENTOasis as Canc,
			coalesce(Canc_novo,0) + coalesce(CANCELAMENTOasis,0) as  Cancelamento
		From base_spf_la a

		Union all
		select
			a.negocio,
			a.Produto,
			a.Seguro,
			a.Data,
			a.Produto_seguro,
			a.Produto_Financiado,
			a.CANAL,
			a.Canc_novo_prod as Canc_novo,
			a.Cancelamento as Canc,
			coalesce(a.Canc_novo_prod,0) + coalesce(a.Cancelamento,0) as  Cancelamento
		From base_Outros a

		Union all
		select
			a.negocio,
			a.Produto,
			a.Seguro,
			a.Data,
			a.Produto_seguro,
			a.Produto_Financiado,
			a.CANAL,
			a.Canc_novo_prod as Canc_novo,
			a.Cancelamento as Canc,
			coalesce(a.Canc_novo_prod,0) + coalesce(a.Cancelamento,0) as  Cancelamento
		From base_cartoes a
;quit;

proc sql;
	create table kpis2 as select

		produto,

		sum('Qtd Contratos'n) as 'Qtd Contratos'n

proc sql;
	create table premio as select
		data,
		negocio,
		case when produto = 'Rodas' then Produto_Financiado else
		produto end as Produto,
		seguro,
		sum(Premio_Bruto) as Premio_Bruto,
		sum(qtd_contratos) as qtd_contratos
	from work.base_final group by 1,2,3,4;
quit;

proc sql;
	create table Kpis_1 as select
		a.*,
		b.'Produção'n,
		b.'Qtd Contratos'n as Qtd_Contrato_Prod,
		a.Qtd_Contratos as Qtd_Contrato_Seg,
		a.Premio_Bruto/b.'Produção'n as Pen_Financeira format percent8.4
	from work.premio as a left join work.kpis2 as b on (a.Data=b.Data and a.produto=b.produto and a.negocio=b.negocio)
		where a.Produto in ('Leves','Motos','Pesados','CVG','Privado','Solar','CP','FGTS', 'Crédito do Trabalhador', 'Crédito Trabalhador')
			order by 1,2,3;

	Where  Data >=&dt_i. and Data <=&dt_f.
quit;

data Kpis_2;
set kpis_1;
format Pen_Fisica percent8.4;
Pen_Fisica = Qtd_Contrato_Seg/Qtd_Contrato_Prod;
drop 'Produção'n;
drop Qtd_Contrato_Prod;
drop qtd_contratos;
run;

proc sql;
	create table Template_VCS as select
		negocio as 'Negócio'n,
		Seguro,
		Produto,
		Data,
		Produto_seguro as 'Produto Seguro'n,
		Produto_Financiado as 'Produto Financiado'n,
		Canal,
		Qtd_Contratos as Contratos,
		coalesce(Premio_Bruto,0) as 'Premio Bruto'n,
		coalesce(Premio_Liquido,0) as 'Premio Liquido'n,
		(coalesce(Receita_Bruta,0) + coalesce(Ajust_cx_comp,0) + coalesce(CANCELAMENTO,0) + coalesce(LA_total,0))  as 'Receita Líquida'n,
		coalesce(Repase_cartoes,0) as 'Repase cartoes'n,

		coalesce(Desp_adm,0) as 'Desp Adm'n,
		coalesce(Cont_civis,0) as 'Cont Cíveis'n,
		coalesce(Cont_Trabalhistas,0) as 'Cont Trab'n,
		coalesce(Cont_Fiscais,0) as 'Cont Fiscais'n,
		(coalesce(Outras_Desp,0) - coalesce(Cont_Fiscais,0)) as 'Outras Desp'n,
		coalesce(RV,0) as 'PLR'n,

		coalesce(CANCELAMENTO,0) + coalesce(Ajust_cx_comp,0))*-0.34 as 'IR/CS'n,

		coalesce(LA_total,0) as 'LA'n,
		coalesce(CANCELAMENTO,0) AS 'Cancelamento'n,
		coalesce(Receita_Bruta,0) + coalesce(Ajust_cx_comp,0) as 'Receita Bruta de LA/Canc'n,
		coalesce(Ajust_cx_comp,0) as 'Ajuste Caixa/Competencia'n,

		&Versao. as 'Versão'n

	Where  Data >=&dt_i. and Data <=&dt_f.
