%include '/infogrc/utils/all-includes.sas';
%let dir_csv = /riscosbvf/BA/processos/output;
LIBNAME TRIBO '/riscosbvf/coaching/tribo_veiculos/SAS';

/* ======================================================================================*/
/* =====================Gera dColaborador =========================*/
/* ======================================================================================*/

PROC SQL;
   CREATE TABLE d_colaborador AS
   SELECT
		DISTINCT
/*			t1.nuanomes,*/
			t1.DsLoginN1,
			t1.DsLoginN2A,
			t1.DsLoginN3A,
			t1.LstSubstituicao,
			t1.LstColaboradorFocal,
			t1.NmAreaEstruturaComercial as Filial,
			t1.NmSubEstruturaComercialA as Regional,
			t1.NmCanalEstruturaComercial as Canal,
			t1.NmColaborador as Nm_Colaborador,
			CATS(t1.NmCanalEstruturaComercial,"-",t1.NmSubEstruturaComercialA,"-",t1.NmAreaEstruturaComercial,"-",t1.NmColaborador) as Key_dimensoes,
			CATS(t1.NmSubEstruturaComercialA,"-",t1.NmAreaEstruturaComercial,"-",t1.NmColaborador) as Key_Rito

      FROM BASES.TBMETAPRODUCAOFINAL_DASH t1;

QUIT;

%let dir_csv = /riscosbvf/BA/processos/output;

PROC EXPORT
		DATA= d_colaborador
		outfile="&dir_csv./d_colaborador_RITO_RC_TATICO.csv" REPLACE
		DBMS=DLM;
	DELIMITER=';';
RUN;
