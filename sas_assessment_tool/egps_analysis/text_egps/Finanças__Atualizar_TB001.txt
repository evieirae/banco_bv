Create Table Work.Segmento as
Select distinct
	Case
		When cdSegmento = 1 Then 11
		else cdSegmento
	End as cdSegmento_CIB FORMAT = BEST11. INFORMAT = BEST11.,
	Case
		 When Segmento = "INSTITUICAO FINANCEIRA" Then "FI"
		 When Segmento = "MIDDLE" Then "ATIVOS ESPECIAIS"
		 When Segmento = "COMERCIAL" Then "LARGE"
		 When Segmento = "PRIVATE" Then "CORRETORA"
		 else Segmento
	End as Segmento_CIB FORMAT = $50. INFORMAT = $50.
From TOSTI.BASE_CLIENTES_CIB;

Create Table Work.Banker as
Select distinct
cdPessoa as cdBanker_CIB,
nmPessoa as Banker_CIB
From DBGLOBAL.VwOfficer;

Create Table Work.Divisao_Comercial as
Select distinct
	Case
		When Divisao_Comercial like "PLATAFORMA%" then 83
		When Divisao_Comercial like "LARGE CORPORATE%" then 304
		When Divisao_Comercial = "CIB." then 74
		When Divisao_Comercial = "SE/NE" then 356 
		When Divisao_Comercial like "RJ, MG E ES%" then 306
		else cdDivisao_Comercial
	End as cdDivisao_Comercial_CIB,

	Case
		When Divisao_Comercial like "PLATAFORMA%" then "GPR"
		When Divisao_Comercial = "MIDDLE" then "GPR"
		When Divisao_Comercial like "LARGE CORPORATE%" then "LARGE CORPORATE"
		When Divisao_Comercial = "CIB." then "CIB"
		When Divisao_Comercial = "SE/NE" then "CIB - HIBRIDO"
		When Divisao_Comercial like "RJ, MG E ES%" then "RJ, MG E ES & CO E NE"
		else tranwrd (Divisao_Comercial, ".", "")
	End as Divisao_Comercial_CIB
From TOSTI.BASE_CLIENTES_CIB;

		a.'NOVO BANKER 2018'n as Officer,

		Case
			When b.Segmento_CIB = "FI" Then "INSTITUICAO FINANCEIRA"
			else b.Segmento_CIB
		End as Segmento_Ajustado,

		a.cdTrader as cdOfficer_Passivo,
		a.Trader as Officer_Passivo,

	Case
		When a.'NOVO TRADER CAPTAÇÃO'n = "CAMILA RAMOS DE OLIVEIRA" Then 769069
		When a.'NOVO TRADER CAPTAÇÃO'n = "ANA PAULA KOWALSKY TABORDA" Then 43523

Create table WORK.Verificar_duplicados as

select cdmodalidade, count(*) as qtde
from SHIS.AREA_PRODUTO
group by cdmodalidade
having count(*) > 1;

Create table WORK.Verificar_duplicados_comp as
select cdcomponente, count(*) as qtde
from SHIS.AREA_PRODUTO_comp
group by cdcomponente
having count(*) > 1;
