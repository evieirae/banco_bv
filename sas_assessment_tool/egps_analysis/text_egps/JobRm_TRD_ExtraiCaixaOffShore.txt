CREATE TABLE TMPCAIXAOFFSHORETRD AS

(

	  FROM Tb_Lancamento_Caixa_Cash_Flow      lccf (NOLOCK)
	       INNER JOIN Tb_Conta_Liquidacao     ctlq (NOLOCK) ON ctlq.CD_CONTA_LIQUIDACAO = lccf.CD_CONTA_LIQUIDACAO
	       INNER JOIN Tb_Conta_Caixa          ctcx (NOLOCK) ON ctcx.CD_CONTA            = lccf.CD_CONTA
	       INNER JOIN TB_PARITY par (NOLOCK) ON par.PARITY_CD = lccf.CD_MOEDA
	        LEFT JOIN Tb_Operacao             oper (NOLOCK) ON oper.CD_OPERACAO         = lccf.CD_OPERACAO
	        LEFT JOIN Tb_Produto              prod (NOLOCK) ON prod.CD_PRODUTO          = oper.CD_PRODUTO
	        LEFT JOIN Tb_Instituicao          inst (NOLOCK) ON inst.CD_INSTITUICAO      = oper.CD_INSTITUICAO_CONTRA_PARTE
	        LEFT JOIN Tb_Papel                pape (NOLOCK) ON pape.CD_PAPEL            = oper.CD_PAPEL
	        LEFT JOIN Tb_Operacao_Complemento opcp (NOLOCK) ON opcp.CD_OPERACAO         = oper.CD_OPERACAO
	        LEFT JOIN Tb_Contrato_futuro      ctft (NOLOCK) ON ctft.CD_DE_PARA_SERIE_FUTURO = opcp.CD_DE_PARA_SERIE_FUTURO
	WHERE lccf.DT_LANCAMENTO = &PDTREFERENCIA.
	  AND lccf.CD_INSTITUICAO = 611
	  AND lccf.ID_REFERENCIA = 'V'
	  AND (((ctlq.TP_calculo IN ('FUN','REP','LOA') AND PATINDEX('%' + oper.ST_OPERACAO  + '%' ,'S') > 0) AND (lccf.NM_ORIGEM NOT IN ('MGN', 'ABE'))) OR
	       (ctlq.TP_calculo IN ('SWP')))
	UNION
	SELECT lccf.DT_LANCAMENTO      'DataLancamento'
	      ,lccf.CD_OPERACAO        'Operacao'
	      ,CAST(COALESCE(prod.NM_PRODUTO, pro2.NM_PRODUTO, '') AS VARCHAR) 'Produto'
	      ,inst.NM_INSTITUICAO     'ContraParte'
	      ,SUM(lccf.VR_LANCAMENTO) 'ValorLancamento'
	      ,lccf.CD_MOEDA           'CodMoeda'
	      ,CAST(COALESCE(pape.NM_PAPEL, ctft.CD_TICKER, '') AS VARCHAR(40)) 'Papel'
	FROM Tb_Lancamento_Caixa_Cash_Flow   lccf (nolock)
	      INNER JOIN Tb_Conta_Liquidacao ctlq (nolock) ON ctlq.CD_CONTA_LIQUIDACAO     = lccf.CD_CONTA_LIQUIDACAO
	       LEFT JOIN Tb_Ajuste_caixa     ajcx (nolock) ON ajcx.CD_AJUSTE               = lccf.CD_OPERACAO
	       LEFT JOIN Tb_Produto          prod (nolock) ON prod.CD_PRODUTO              = ajcx.CD_PRODUTO
	       LEFT JOIN Tb_Instituicao      inst (nolock) ON inst.CD_INSTITUICAO          = ajcx.CD_INSTITUICAO_CONTRAPARTE

	       LEFT JOIN Tb_Contrato_futuro  ctft (nolock) ON ctft.CD_DE_PARA_SERIE_FUTURO = lccf.CD_PAPEL
	       LEFT JOIN Tb_Produto          pro2 (nolock) ON pro2.CD_PRODUTO              = ctft.CD_PRODUTO
	       LEFT JOIN Tb_Operacao         oper (nolock) ON oper.CD_OPERACAO             = lccf.CD_OPERACAO
	  WHERE lccf.DT_LANCAMENTO   = &PDTREFERENCIA.
	    AND ctlq.TP_CALCULO = 'CDV'
	    AND lccf.NM_ORIGEM       = 'ADJ'
	    AND lccf.CD_OPERACAO     = 0
	  GROUP BY lccf.DT_LANCAMENTO
	          ,lccf.CD_OPERACAO
	          ,prod.NM_PRODUTO
	          ,pro2.NM_PRODUTO
	          ,inst.NM_INSTITUICAO
	          ,lccf.CD_MOEDA
	          ,pape.NM_PAPEL
	          ,ctft.CD_TICKER;
