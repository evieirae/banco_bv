PROC SQL; CREATE TABLE TbCib AS
SELECT
 A.NuCpfCnpjRaiz
,COALESCE(B.DsClassificacao,'ND') AS CMRCIB
,MAX(B.FlAtivoCIB)				  AS FlAtivoCIB
,MAX(B.FlGPAtivoCIB)			  AS FlGPAtivoCIB
,MAX(CALCULATED FlAtivoCIB
    ,CALCULATED FlGPAtivoCIB)	  AS FlCIB

FROM TbBrutaPME AS A
LEFT JOIN &LIBIT..TbBruta AS B ON A.NuCpfCnpjRaiz = B.NuCpfCnpjRaiz AND B.TpLancamento = 'Pessoa'

GROUP BY A.NuCpfCnpjRaiz,B.DsClassificacao
;quit;run;

PROC SQL; CREATE TABLE TbRFPME AS
SELECT
 A.DtProcessamento						AS DtReferencia
,A.NuCpfCnpj							AS NuCnpj
,A.NuCpfCnpjRaiz 						AS NuCnpjRaiz
,INPUT(SUBSTR(A.NuCpfCnpj,9,4),BEST.)	AS NuFilial
,A.NmPessoa						  		AS NmPessoa
,A.CdGpEconomico
,A.NmGpEconomico
,A.FlAtivoP						  		AS FlAtivo
,A.CdPessoa
,A.DsClassificacaoSME			  		AS DsClassificacao
,A.FlSuspensoP					  		AS FlSuspenso
,A.VrTotalAprovadoCliente		  		AS VrLimite
,A.VrTotalRiscoCliente			  		AS VrRisco
,A.FlNMGPSME
,A.FlNMGPPEAC
,B.CMRCIB
,B.FlCIB

FROM TbBrutaPME AS A
LEFT JOIN TbCib AS B ON A.NuCpfCnpjRaiz = B.NuCpfCnpjRaiz

;QUIT;RUN;

 A.NuCnpjRaiz
,B.DtInclusao

,B.NuCMRNv		AS NuCMRAT

ORDER BY A.NuCnpjRaiz , B.DtInclusao DESC

PROC SQL; CREATE TABLE TbCManual AS
SELECT
'Data Inclusão'n					AS DtInclusao		FORMAT DDMMYY8.
,INPUT('23:59:59',TIME8.)			as HrInclusao		FORMAT TIME8.
,PUT('CNPJ Raiz'n,z8.)  			AS NuCnpjRaiz

,UPCASE(SUBSTR(Indeterminado,1,1))	AS FlIndeterminado

	WHEN VrPercentualILM <= 0.80 AND VrPercentualILM > 0.70 THEN 3

FROM TbILM AS A

PROC SQL; CREATE TABLE TbVisaoGarantias	AS
SELECT
 "Garantias"							AS TpVisao 		LENGTH = 50
,DATA_ARQUIVO							AS DtConsulta
,TIPO_GARANTIA							AS TpRegistro	LENGTH = 100
,CD_PESSOA 								AS CdPessoa
,GRUPO_ECONOMICO                        AS GpEconomico
, ''                    				AS Sigla1       LENGTH=100
,GARANTIA_DETALHE_OPERACIONAL 			AS DsGrupo1		LENGTH = 100
,MAX(INPUT(DIAS_PENDENTES, 8.))			AS QdDiasAtraso

AND GARANTIA_DETALHE_OPERACIONAL NOT IN ('Contrato de Repasse BNDES', 'Garantia Com Consituição Futura - Dentro do Prazo', 'Cesta de Garantias', 'Contrato em Ativos Especiais')
AND DATA_ARQUIVO = (SELECT MAX(DATA_ARQUIVO) FROM INSIGHT.TB_GARANTIAS_ANALITICO)
AND CD_PESSOA IS NOT NULL
GROUP BY 1,2,3,4,5,6
;QUIT;RUN;

	A.*
	,CASE
		WHEN TpRegistro IN ('Duplicatas (custódia própria)', 'Duplicatas (custódia terceiros)') AND A.VrPcInsuficiente >= 0.3 THEN 4
		WHEN TpRegistro IN ('Duplicatas (custódia própria)', 'Duplicatas (custódia terceiros)') AND A.VrPcInsuficiente >= 0.1 AND A.VrPcInsuficiente < 0.3 THEN 3

		WHEN TpRegistro NOT IN ('Duplicatas (custódia própria)', 'Duplicatas (custódia terceiros') AND A.VrPcInsuficiente >= 0.1 AND A.VrPcInsuficiente < 0.3 THEN 2
		ELSE 0 END AS NuPontoGARANTIAS

	,C.NuCpfCnpjRaiz AS NuCnpjRaiz

FROM TbVisaoGarantias AS A
INNER JOIN TbPMEEmprestimos AS C ON A.CdPessoa = C.CdPessoa
;RUN;

PROC SQL; CREATE TABLE TbPME AS
SELECT
	A.*
	,CASE
		WHEN B1.FlAtivoPME NE . THEN 'M'
		WHEN B2.FlAtivoPME NE . THEN 'J'
		ELSE 'S' END AS TIPO_ESTEIRA_PME

	,C.VrPercentualILM
	,CASE WHEN C.NuPontoILM = . THEN 0
		ELSE C.NuPontoILM END AS NuPontoILM

	,D.VrPcInsuficiente


	,CASE WHEN D.NuPontoDUPLICATAS = . THEN 0
		ELSE D.NuPontoDUPLICATAS END AS NuPontoDUPLICATAS
	,CASE WHEN D.NuPontoGARANTIAS = . THEN 0
		ELSE D.NuPontoGARANTIAS END AS NuPontoGARANTIAS

	,CASE WHEN E.NuPontoBalanco = . THEN 0
		ELSE E.NuPontoBalanco END AS NuPontoBalanco

	,CASE WHEN F.ATRASO_SACADO = . THEN 0
		ELSE F.ATRASO_SACADO END AS DiasAtrasoSacado

	,CASE WHEN G.FLAG_SOCIOPF = . THEN 0
		ELSE G.FLAG_SOCIOPF END AS FlagSocioPF

	,CASE WHEN H.FlagLiminar = . THEN 0
		ELSE H.FlagLiminar END AS FlagLiminar

	,CASE WHEN I.FLGARANTIASGERAIS = . THEN 0
		ELSE i.FLGARANTIASGERAIS END AS FlagGarantiasGerais

	,CASE WHEN J.CNPJRAIZ = "" THEN .
		ELSE J.PercCobertura END AS PercCobertura

	,CASE WHEN K.FlagBloqueio = . THEN 0
		ELSE K.FlagBloqueio END AS FlagBloqueio

FROM TbPME_temp AS A

LEFT JOIN TbPMEMassificada 		AS B1 ON A.NuCnpjRaiz = B1.NuCpfCnpjRaiz
LEFT JOIN TbPMEJulgamental 		AS B2 ON A.NuCnpjRaiz = B2.NuCpfCnpjRaiz

LEFT JOIN TbVisaoILM 			AS C ON A.NuCnpjRaiz = C.NuCnpjRaiz
LEFT JOIN TbVisaoGarantias2		AS D ON A.NuCnpjRaiz = D.NuCnpjRaiz
LEFT JOIN TbVisaoBalanco2 		AS E ON A.NuCnpjRaiz = E.NuCnpjRaiz
LEFT JOIN TbVisaoAtrasoSacado 	AS F ON A.NuCnpjRaiz = F.CNPJ_RAIZ_SACADO

LEFT JOIN TbVisaoSocioPF 		AS G ON A.NuCnpjRaiz = G.NuCnpjRaiz

LEFT JOIN BASE_NADA_CONSTA 		AS H ON A.NuCnpjRaiz = H.NuCnpjRaiz
LEFT JOIN RELATORIOPOSICAO_2	AS I ON A.NuCnpjRaiz = I.CNPJRAIZ
LEFT JOIN RELATORIODIRCRED_2	AS j ON A.NuCnpjRaiz = j.CNPJRAIZ
LEFT JOIN BASE_BLOQUEIO_4		AS K ON A.NuCnpjRaiz = K.CNPJRaizSacado

HAVING CMRAT NE 'PCMR4'
;RUN;
