PROC SQL;

CREATE TABLE TMPPRD AS
SELECT
DTENTRADAPROPOSTA,
YEAR(DTENTRADAPROPOSTA)*100+MONTH(DTENTRADAPROPOSTA) AS MESANOPROPOSTA,
DTCONTRATO,
YEAR(DTCONTRATO)*100+MONTH(DTCONTRATO) AS MESANOCONTRATO,
NUPROPOSTA,
NUCONTRATO,
CDPARCEIROCOMERCIAL,
NMGRUPOPRODUTOMETA
from BASES.TBPRODUCAO
WHERE 1=1
AND YEAR(DTCONTRATO)*100+MONTH(DTCONTRATO) >= 202501
AND NmGrupoProdutoMeta in
(
'CVG Leves',
'CVG Pesados',
'Leves',
'Leves-BVO',
'Motos',
'Pesados'
)
;QUIT;
