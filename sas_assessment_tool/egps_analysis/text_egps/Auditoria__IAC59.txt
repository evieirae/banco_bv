proc sql;
CREATE TABLE PAGAMENTOS_&nome_mes_anterior (COMPRESS=YES) AS
    SELECT
        E2.*,
        A2.A2_XPARC,
        E5.E5_FILIAL, E5.E5_DATA, E5.E5_VALOR, E5.E5_BANCO, E5.E5_AGENCIA, E5.E5_CONTA,
        E5.E5_BENEF, E5.E5_VLMOED2, E5.E5_FORNECE, E5.E5_CCUSTO, E5.E5_DOCUMEN, E5.E5_HISTOR, E5.E5_TIPODOC,
        C7.C7_NUM, C7.C7_DESCRI,
        CR.CR_NUM, CR.CR_USERLIB, CR.CR_TIPO, CR.CR_USER, CR.CR_APROV, CR.CR_NIVEL, CR.CR_STATUS, CR.CR_DATALIB, CR.CR_TOTAL
    FROM PRT.SE2700 E2
    INNER JOIN PRT.SE5700 E5
        ON E2.E2_FILIAL  = E5.E5_FILIAL
        AND E5.E5_PREFIXO= E2.E2_PREFIXO
        AND E5.E5_NUMERO = E2.E2_NUM
        AND E5.E5_PARCELA= E2_PARCELA
        AND E5.E5_TIPO   = E2.E2_TIPO
        AND E5.E5_CLIFOR = E2.E2_FORNECE
        AND E5.E5_LOJA  = E2.E2_LOJA
        AND E5.D_E_L_E_T_= ''
    LEFT JOIN PRT.SD1700 D1
        ON SUBSTR(D1.D1_FILIAL,1,4) = E2.E2_FILIAL
        AND D1.D1_DOC   = E2.E2_NUM
        AND D1.D1_SERIE = E2.E2_PREFIXO
        AND D1.D1_FORNECE= E2.E2_FORNECE
        AND D1.D1_LOJA  = E2.E2_LOJA
        AND D1.D_E_L_E_T_= ''
    LEFT JOIN PRT.SC7700 C7
        ON C7.C7_FILIAL = E2.E2_FILIAL
        AND C7.C7_NUM   = D1_PEDIDO
        AND C7.D_E_L_E_T_= ''
    LEFT JOIN PRT.SCR700 CR
        ON CR.CR_FILIAL = C7.C7_FILIAL
        AND CR.CR_NUM   = C7.C7_NUM
        AND CR.D_E_L_E_T_= ''
    LEFT JOIN PRT.SA2700 A2
        ON A2.A2_COD = E2.E2_FORNECE
        AND A2.D_E_L_E_T_ <> '*'
    WHERE E2.E2_VENCREA BETWEEN "&inicio_mes_anterior" AND "&fim_mes_anterior"
        AND E2.E2_PREFIXO NOT IN ('RE ','COM')
        AND E2.E2_NATUREZ NOT IN ('0000000999','0000000998')
        AND E2.D_E_L_E_T_= '';
QUIT;
