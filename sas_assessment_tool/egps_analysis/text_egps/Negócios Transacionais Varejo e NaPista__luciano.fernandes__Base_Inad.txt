proc sql;
create table base_inad as
select *
from Base_BVO_Atraso
where nuproposta not in (select distinct NuProposta from Base where StProposta="Aprovada e Paga" )
;
quit;

proc sql;
create table Base_BVO_Atraso_out as
select a.*
,b.NuCpfCnpj

from Base_BVO_Atraso as a
left join Base as b
on a.nuproposta=b.NuProposta
where NuCpfCnpj>0

;
quit;

proc sort data=Base_BVO_Atraso_out nodupkey; by _all_; run;
