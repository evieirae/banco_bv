PASSWORD 	= &pw_PTASYBCTR.   );
CREATE TABLE WORK.BASE_ACORDO_00 AS
SELECT A.*
FROM
			CONNECTION TO DBCON(
								SELECT DISTINCT
					              CON.DTGERACAO
					            , CON.NUCONTRATOGESTAO
					            , CON.DTVENCIMENTO
					            , CON.DTPAGAMENTO
					            , CON.STNEGOCIACAO
					            , CON.TPENTRADA
					            , CASE
					                WHEN CON.STNEGOCIACAO = 'AN' THEN 'A NEGOCIAR'
					                WHEN CON.STNEGOCIACAO = 'N'  THEN 'NEGOCIADO'
					                WHEN CON.STNEGOCIACAO = 'PA' THEN 'PENDENTE DE APROVACAO'
					                WHEN CON.STNEGOCIACAO = 'A'  THEN 'APROVADO PELA BV'
					                WHEN CON.STNEGOCIACAO = 'R'  THEN 'RECUSADO PELA BV'
					                WHEN CON.STNEGOCIACAO = 'NA' THEN 'NEGOCIACAO EM ANDAMENTO'
					                WHEN CON.STNEGOCIACAO = 'C'  THEN 'NEGOCIACAO CANCELADA'
					                WHEN CON.STNEGOCIACAO = 'LP' THEN 'LIBERADO PAGAMENTO'
					                WHEN CON.STNEGOCIACAO = 'PG' THEN 'PAGO'
					                WHEN CON.STNEGOCIACAO = 'PL' THEN 'PAGAMENTO REALIZADO'
					                WHEN CON.STNEGOCIACAO = 'DA' THEN 'DEVOLVIDO ASSESSORIA'
					                WHEN CON.STNEGOCIACAO = 'CD' THEN 'CHEQUE DEVOLVIDO'
					                WHEN CON.STNEGOCIACAO = 'AC' THEN 'CHEQUE AGUARDANDO COMPENSACAO'
					             ELSE '99.VERIFICAR' END STATUS_BOLETO
						FROM DBCONTENCIOSO..TBNEGOCIACAOCONTRATO CON
						WHERE CON.DTCANCELAMENTO IS NULL
						AND CON.DTPAGAMENTO IS NULL
						AND CON.DTVENCIMENTO >= GETDATE()
						AND CON.DTVENCIMENTO <= DATEADD(DD, 15, GETDATE())
						ORDER BY CON.DTVENCIMENTO DESC) A
;QUIT;

PROC SQL;
	CREATE TABLE WORK.BASE_ACORDO AS

		SELECT DISTINCT
			  A.*
			, intck('DAY', DATEPART(A.DtGeracao), DATEPART(A.DtVencimento), 'C')  AS Intervalo_GerVenc_Bol
			, DATEPART(A.DtGeracao) Format yymmdd10. AS Dt_Geracao_Boleto
			, DATEPART(A.DtVencimento) Format yymmdd10. AS Dt_Vencimento_Boleto
			, HOUR(A.DtGeracao) AS Hr_Geracao
			, Case
				WHEN B.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) <=  7 THEN 1
				WHEN B.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) >= 20 THEN 1
				ELSE 0 END AS Flag_ForaExped
			, B.Canal As Canal_Boleto
			, B.Canal_Macro
		FROM WORK.BASE_ACORDO_00 AS A
		LEFT JOIN TCONFIG.CONFIG_TPENTRADA AS B ON A.TpEntrada = B.TpEntrada
		HAVING Flag_ForaExped = 0
;QUIT;

proc sql;
create table work.aux_base_cyber as

select distinct
case
	when a.cod_assessoria = 2074 then 'google'
	when a.cod_assessoria = 2075 then 'jabuti'
	when a.cod_assessoria = 2076 then 'domu'

	else 'outros'
end as nm_escob
,a.*
,case
	when a.nucontratogestao = b.nucontrato  then 'S'
	else 'N'
 end as Flag_Filtro_List_Rest_Int
,case
	when a.nucontratogestao = c.nucontrato then 'S'
	else 'N'
end as Flag_Mand_Exp
,case
	when a.nucontratogestao = d.nucontratogestao then 'S'
	else 'N'
 end as Flag_Acordo_New
from contatos.aux_base_cyber as a
left join (	select distinct a.nucontrato
			from zbloq.bloqueio_reguas_intersec as a
			where 1=1
			and A.DT_REF = Today()
			and A.Flag_Bloqueio = 1
			and Motivo in ('Bloq_Cessao_Rodas','Bloq_Cessao_Cartao','Regra_Base_D0')) as b on a.nucontratogestao = b.nucontrato
left join work.base_mandado_exp as c on a.nucontratogestao = c.nucontrato
left join work.base_acordo as d on a.nucontratogestao = d.nucontratogestao
where 1=1
and a.cod_assessoria in (2074,2075,2076)


order by a.cod_assessoria asc

;quit;
