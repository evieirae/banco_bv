PROC SQL;
CREATE TABLE BASE_CL AS
SELECT
	A.*,
	CASE
		WHEN A.RECEBIDO_PARCELADO > 0 AND A.RECEBIDO_QUITADO > 0 THEN (A.RECEBIDO_PARCELADO + A.RECEBIDO_QUITADO)
		WHEN A.RECEBIDO_PARCELADO =. AND A.RECEBIDO_QUITADO > 0 THEN A.RECEBIDO_QUITADO
		WHEN A.RECEBIDO_PARCELADO >0 AND A.RECEBIDO_QUITADO =. THEN A.RECEBIDO_PARCELADO
		END AS RECEBIMENTO_TOTAL
FROM PERF_CL.CL_TABLEAU AS A
JOIN (SELECT DISTINCT
				MAX(DT_REF) AS DT_REF FORMAT DDMMYY10.
				FROM PERF_CL.CL_TABLEAU
				WHERE DATA_BASE = &SAFRA.) AS B
ON A.DT_REF = B.DT_REF
WHERE A.ASSESSORIA <> "BV"
AND A.TIPO <> "COLCHAO";
RUN;

PROC SQL;
CREATE TABLE BASE_CL_1 AS
SELECT
	ASSESSORIA,
	FX_ATRASO,
	(SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO)) AS PERFORMANCE
FROM BASE_CL
GROUP BY
	ASSESSORIA,
	FX_ATRASO
;QUIT;

PROC SQL;
CREATE TABLE BASE_CL_1 AS
SELECT
	ASSESSORIA,
	CL_STATUS,
	FX_ATRASO,
	(SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO)) AS PERFORMANCE,
	CASE
		WHEN FX_ATRASO = "75 A 120" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.50)
		WHEN FX_ATRASO = "121 A 180" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.20)
		WHEN FX_ATRASO = "181 A 360" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.30)
		WHEN FX_ATRASO = "361 A 720" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.40)
		WHEN FX_ATRASO = "ACIMA DE 720" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.60)
	END AS PERFORMANCE_PESO
FROM BASE_CL
WHERE ASSESSORIA <> "BV"
GROUP BY
	ASSESSORIA,
	CL_STATUS,
	FX_ATRASO
;QUIT;

PROC SQL;
CREATE TABLE BASE_CL_2 AS
SELECT
	ASSESSORIA,
	CL_STATUS,
	CASE
		WHEN ASSESSORIA IN ('VILELA IBANEZ','SERVICES') THEN 0
		WHEN CL_STATUS = 1 THEN round(SUM(PERFORMANCE_PESO),0.0000001)
		WHEN CL_STATUS = 2 THEN round(SUM(PERFORMANCE_PESO),0.0000001)
	END AS PERF_FINAL
FROM BASE_CL_1
GROUP BY
	ASSESSORIA,
	CL_STATUS
;QUIT;

PROC SQL;

	CL_STATUS,
	ASSESSORIA,

		WHEN CL_STATUS = 2 THEN &METACL2.
	END AS META,
	PERF_FINAL,
	CASE
		WHEN ASSESSORIA IN ('VILELA IBANEZ','SERVICES') THEN 0
		WHEN CL_STATUS = 1 THEN ROUND(PERF_FINAL / &METACL1.,0.0001)
		WHEN CL_STATUS = 2 THEN ROUND(PERF_FINAL / &METACL2.,0.0001)
	END AS ATINGIMENTO_META
FROM  BASE_CL_2
