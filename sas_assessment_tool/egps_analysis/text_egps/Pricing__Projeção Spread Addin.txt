CREATE TABLE pricing. PROJECAO_SPREAD AS
SELECT

A.NuAnoMes,
A.DtContrato as Dia,

SUM(A.VpTxGerencial)/SUM(A.VrProducao) as TaxaNet,
SUM(A.VpTirContabil)/SUM(A.VrProducao) as TirC,
SUM(A.VpSpreadContabil)/SUM(A.VrProducao) as SpreadNet,
SUM(A.FlTxAlterada)/COUNT(A.NuContrato) as PA,
SUM(A.VrProducao) as VrProducao,
SUM(A.VrMargemContribuicao)/SUM(A.VrProducao) as MC,
SUM(A.VpTirDivulgada)/SUM(A.VrProducao) as TirAgrv,
SUM(A.VpSpread)/SUM(A.VrProducao) as Spread_Agrv,
(SUM(A.VpSpreadContabil)-SUM(A.VpSpread))/SUM(A.VrProducao) as Agrv,
SUM(A.VrPerdaEstimada)/SUM(C.VrFinanciamento) as Perda,
SUM(A.VpScore)/SUM(A.VrProducao) as Score_Medio,
COUNT(A.NuContrato) as QtdContratos,
SUM(A.VpPrazo)/SUM(A.VrProducao) as Prazo_Medio,
SUM(A.VpEntrada)/SUM(A.VrProducao) as Entrada_Media,
SUM(A.VpIdade)/SUM(A.VrProducao) as Idade_Media
FROM TMP.TBPRODUCAO AS A
LEFT JOIN PROJETOS.TBTEMPODATA AS B ON A.DtContrato=B.DtTempo
LEFT JOIN PPR.PPR_ANALITICO AS C ON A.NuContrato=C.NuContrato


WHERE A.NmGrupoProdutoMeta = 'Motos' and A.NuAnoMes >= 202501
group by 1,2;
quit;
