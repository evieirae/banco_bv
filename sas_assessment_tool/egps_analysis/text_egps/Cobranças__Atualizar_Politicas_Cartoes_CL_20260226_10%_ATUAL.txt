PROC SQL;
CREATE TABLE CARTEIRA_V20 AS
SELECT DISTINCT
A.*


,CASE WHEN A.MOVTO_QUITACAO = '02 - Manutenção' AND A.MOVTO_PARCELA = '01 - Redução' 	THEN '01 - MANTÉM Q E REDUZ P'
	  WHEN A.MOVTO_QUITACAO = '02 - Manutenção' AND A.MOVTO_PARCELA = '02 - Manutenção' THEN '02 - MANTÉM Q E MANTÉM P'
	  WHEN A.MOVTO_QUITACAO = '02 - Manutenção' AND A.MOVTO_PARCELA = '03 - Aumento' 	THEN '03 - MANTÉM Q E AUMENTA P'

	  WHEN A.MOVTO_QUITACAO = '03 - Aumento' AND A.MOVTO_PARCELA = '01 - Redução' 		THEN '04 - AUMENTA Q E REDUZ P'
	  WHEN A.MOVTO_QUITACAO = '03 - Aumento' AND A.MOVTO_PARCELA = '02 - Manutenção' 	THEN '05 - AUMENTA Q E MANTÉM P'
	  WHEN A.MOVTO_QUITACAO = '03 - Aumento' AND A.MOVTO_PARCELA = '03 - Aumento' 		THEN '06 - AUMENTA Q E AUMENTA P'

	  WHEN A.MOVTO_QUITACAO = '01 - Redução' AND A.MOVTO_PARCELA = '01 - Redução' 		THEN '07 - REDUZ Q E REDUZ P'
	  WHEN A.MOVTO_QUITACAO = '01 - Redução' AND A.MOVTO_PARCELA = '02 - Manutenção' 	THEN '08 - REDUZ Q E MANTÉM P'
	  WHEN A.MOVTO_QUITACAO = '01 - Redução' AND A.MOVTO_PARCELA = '03 - Aumento' 		THEN '09 - REDUZ Q E AUMENTA P'

	  ELSE '10 - CORREÇÃO' END AS MOVIMENTACAO FORMAT $30.


,B.COD_SCORE AS COD_CS_CL
,C.SCORE_VU  AS COD_CS_VU

,CASE WHEN DMQUE='CLVU' AND B.COD_SCORE = 8032 THEN 8032
	  WHEN DMQUE='CLVU' AND B.COD_SCORE NE 8032 AND C.SCORE_VU NE . THEN C.SCORE_VU ELSE B.COD_SCORE END AS COD_CS


,CASE WHEN A.ATRASO_ATUALIZADO >= 0   AND A.ATRASO_ATUALIZADO <= 120 THEN '0075 A 0120'
	  WHEN A.ATRASO_ATUALIZADO >= 121 AND A.ATRASO_ATUALIZADO <= 180 THEN '0121 A 0180'
	  WHEN A.ATRASO_ATUALIZADO >= 181 AND A.ATRASO_ATUALIZADO <= 360 THEN '0181 A 0360'
	  WHEN A.ATRASO_ATUALIZADO >= 361 AND A.ATRASO_ATUALIZADO <= 540 THEN '0361 A 0540'
	  WHEN A.ATRASO_ATUALIZADO >= 541 AND A.ATRASO_ATUALIZADO <= 900 THEN '0541 A 0900' ELSE '0901 A 9999' END AS FX_ATRASO_A

FROM CARTEIRA_V19 AS A

LEFT JOIN (SELECT DISTINCT SCORE_ESPELHO, SCORE_VU FROM ALCADAS_CARTAO_CL_VU) 	AS C ON A.SCORE_FINAL = C.SCORE_ESPELHO
;QUIT;

	  WHEN A.MOVTO_QUITACAO = '02 - Manutenção' AND A.MOVTO_PARCELA = '02 - Manutenção' THEN '02 - MANTÉM Q E MANTÉM P'
	  WHEN A.MOVTO_QUITACAO = '02 - Manutenção' AND A.MOVTO_PARCELA = '03 - Aumento' 	THEN '03 - MANTÉM Q E AUMENTA P'

	  WHEN A.MOVTO_QUITACAO = '03 - Aumento' AND A.MOVTO_PARCELA = '03 - Aumento' 		THEN '06 - AUMENTA Q E AUMENTA P'

	  WHEN A.MOVTO_QUITACAO = '01 - Redução' AND A.MOVTO_PARCELA = '03 - Aumento' 		THEN '09 - REDUZ Q E AUMENTA P'
