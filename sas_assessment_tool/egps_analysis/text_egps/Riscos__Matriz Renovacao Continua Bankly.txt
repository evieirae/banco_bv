PROC SQL;
   CREATE TABLE BS_QSA_FINAL_F AS
   SELECT DISTINCT
          t1.TP_QUADRO FORMAT $100. AS TP_QUADRO,
          t1.NOME_QUADRO,
          t1.NOME,
          t1.NuConsultado,
          t1.docConsultado,
          t1.DOC_NUM_QUADRO,
          t1.FL_QSA FORMAT $100. AS FL_QSA

      FROM PLD.BS_QSA_FIM_TESTE t1
	  LEFT JOIN WORK.BS_Fornecedores_Bankly T5 ON (t1.docConsultado = INPUT(compress(t5.DOCUMENTO, , 'kn'),BEST14.))
      WHERE T1.NOME_QUADRO NOT IN ('NÃO SE APLICA','ACOES EM TESOURARIA','FASE PRÉ-OPERACIONAL - 05/04/2024',
            'PREOPERACIONAL') AND INPUT(compress(t5.DOCUMENTO, , 'kn'),BEST14.) IS MISSING

UNION ALL

   SELECT DISTINCT
          (CASE WHEN T1.TIPO = 'FORNECEDOR' THEN 'PRINCIPAL'
                WHEN LENGTH(compress(t1.'CNPJ/CPF'n, , 'kn')) = 11 THEN 'SOCIO PF'
                ELSE 'SOCIO PJ' END) FORMAT $100. AS TP_QUADRO,
		  t1.'RAZÃO SOCIAL/NOME'n AS NOME_QUADRO,
          t1.NOME,
          compress(t1.DOCUMENTO, , 'kn') AS NuConsultado,
          INPUT(compress(t1.DOCUMENTO, , 'kn'),BEST14.) FORMAT BEST14. AS docConsultado,
          (CASE WHEN t1.'CNPJ/CPF'n = 'FORNECEDOR ESTRANGEIRO' THEN 0
                ELSE INPUT(compress(t1.'CNPJ/CPF'n, , 'kn'),BEST14.) END) FORMAT BEST14. AS DOC_NUM_QUADRO,
          (CASE WHEN t1.'CNPJ/CPF'n = 'FORNECEDOR ESTRANGEIRO' THEN 'ESTRANGEIRO'
		        ELSE 'ABERTO' END) FORMAT $100. AS FL_QSA

      FROM WORK.BS_Fornecedores_Bankly t1
	  WHERE (CASE WHEN T1.TIPO = 'FORNECEDOR' THEN 'PRINCIPAL'
                WHEN LENGTH(compress(t1.'CNPJ/CPF'n, , 'kn')) = 11 THEN 'SOCIO PF'
                ELSE 'SOCIO PJ' END) NOT = 'PRINCIPAL'

;QUIT;

PROC SORT DATA=BS_QSA_FINAL_F NODUPKEY; BY _ALL_;RUN;



 PROC SQL;
   CREATE TABLE BS_QSA_FINAL_F AS
   SELECT DISTINCT
          t1.*,
		 (CASE WHEN T2.NUCONSULTADO IS NOT MISSING AND T1.TP_QUADRO IN ('RESPONSAVEL','SOCIO PF') THEN 'SOCIO PF'
		       WHEN T3.NUCONSULTADO IS NOT MISSING AND T1.TP_QUADRO IN ('ESTRANGEIRO PF','SOCIO PF') THEN 'ESTRANGEIRO PF'
		       WHEN T4.NUCONSULTADO IS NOT MISSING AND T1.TP_QUADRO IN ('PRINCIPAL QSA','SOCIO PJ') THEN 'SOCIO PJ'
			   WHEN T5.NUCONSULTADO IS NOT MISSING AND T1.TP_QUADRO IN ('ESTRANGEIRO PJ','SOCIO PJ','PRINCIPAL QSA') THEN 'ESTRANGEIRO PJ'
               ELSE T1.TP_QUADRO END) FORMAT $150. AS TP_QUADRO1

      FROM BS_QSA_FINAL_F t1
	  LEFT JOIN (SELECT DISTINCT t1.NUCONSULTADO,t1.DOC_NUM_QUADRO
                 FROM BS_QSA_FINAL_F t1 WHERE t1.TP_QUADRO = 'RESPONSAVEL' )T2
				ON (T1.NUCONSULTADO = T2.NUCONSULTADO AND T1.DOC_NUM_QUADRO = T2.DOC_NUM_QUADRO)
	  LEFT JOIN (SELECT DISTINCT t1.NUCONSULTADO,t1.DOC_NUM_QUADRO
                 FROM BS_QSA_FINAL_F t1 WHERE t1.TP_QUADRO = 'ESTRANGEIRO PF' )T3
				ON (T1.NUCONSULTADO = T3.NUCONSULTADO AND T1.DOC_NUM_QUADRO = T3.DOC_NUM_QUADRO)
	 LEFT JOIN (SELECT DISTINCT t1.NUCONSULTADO,t1.DOC_NUM_QUADRO
                 FROM BS_QSA_FINAL_F t1 WHERE t1.TP_QUADRO = 'PRINCIPAL QSA' )T4
				ON (T1.NUCONSULTADO = T4.NUCONSULTADO AND T1.DOC_NUM_QUADRO = T4.DOC_NUM_QUADRO)
	 LEFT JOIN (SELECT DISTINCT t1.NUCONSULTADO,t1.DOC_NUM_QUADRO
                 FROM BS_QSA_FINAL_F t1 WHERE t1.TP_QUADRO = 'ESTRANGEIRO PJ' )T5
				ON (T1.NUCONSULTADO = T5.NUCONSULTADO AND T1.DOC_NUM_QUADRO = T5.DOC_NUM_QUADRO)

;QUIT;

DATA BS_QSA_FINAL_F;
SET BS_QSA_FINAL_F;
DROP TP_QUADRO;
RENAME TP_QUADRO1 = TP_QUADRO;
RUN;

PROC SORT DATA=BS_QSA_FINAL_F NODUPKEY; BY NuConsultado docConsultado docConsultado DOC_NUM_QUADRO FL_QSA TP_QUADRO
;RUN;

CREATE TABLE BS_BANKLY_COAF AS
SELECT DISTINCT
			A.*,
			(CASE WHEN B.QTDE IS MISSING THEN 0 ELSE B.QTDE END) AS QTDE_COAF,

	ON (A.DOC_NUM = B.DOC_NUM);

PROC SQL;
CREATE TABLE BS_PEP_SCORE AS
	SELECT DISTINCT
			A.*,
			(CASE WHEN A.NmCargo = 'S/INFO' THEN 27
			      WHEN A.NmCargo NOT= 'S/INFO' AND A.NmOrgao IS MISSING THEN 27
                  WHEN B.Risco = 'Altissimo' THEN 27
			      WHEN B.Risco = 'Alto' THEN 9
                  WHEN B.Risco in ( 'Médio','médio') THEN 3
				  WHEN B.Risco in ('Baixo','baixo') THEN 1
				  WHEN B.Risco in ('Sem definição com o governo') THEN 0
				  WHEN B.Risco IN ('todos','Todos')
                  THEN (CASE WHEN C.RISCO_1 = 'Altíssimo' THEN 27
						 	 WHEN C.RISCO_1 = 'ALTO' THEN 9
						 	 WHEN C.RISCO_1 in ( 'MÉDIO') THEN 3
						 	 WHEN C.RISCO_1 in ('BAIXO') THEN 1 ELSE 100 END) ELSE 100 END) AS SCORE_PEP_INI

	FROM BS_PEP_FULL_MUNI A
		LEFT JOIN De_Para_PEP B
 			ON (A.NmCargo = B.NmCargo) AND (A.NmOrgao = B.NmOrgao)
		LEFT JOIN BS_MUNICIPIOS C
 			ON (A.MUNICIPIO = C.MUNICIPIO) AND (A.UF = C.UF);
QUIT;

PROC SQL;
CREATE TABLE BS_PEP_SCORE AS
	SELECT DISTINCT
			A.*,
			(CASE WHEN A.GrauRelacionamento CONTAINS 'TITULAR' THEN A.SCORE_PEP_INI
				  ELSE (CASE WHEN SCORE_PEP_INI = 27 THEN 9
				  		     WHEN SCORE_PEP_INI = 9 THEN 3
							 WHEN SCORE_PEP_INI = 3 THEN 1
							 WHEN SCORE_PEP_INI = 0 THEN 0
							 WHEN SCORE_PEP_INI = 100 THEN 100
                             WHEN SCORE_PEP_INI = 1 THEN 1 ELSE 0 END)END) AS SCORE_PEP_FIM

	FROM BS_PEP_SCORE A
		;
QUIT;

PROC FREQ DATA=BS_PEP_SCORE ; TABLES SCORE_PEP_INI*SCORE_PEP_FIM/nopercent nocum; RUN ;

PROC SQL;
   CREATE TABLE BS_BANKLY_PEP AS
   SELECT DISTINCT
          T1.*,
          (CASE WHEN T2.DOC_NUM IS MISSING THEN 'NÃO' ELSE 'SIM' END) AS PEP,
          COALESCE(T2.SCORE_PEP,0) AS SCORE_PEP,
		  t2.NmCargo AS CARGO_PEP,
		  t2.NmOrgao AS ORGAO_PEP,
		  t2.SCORE_PEP,
		  t2.UF AS UF_PEP,
		  t2.MUNICIPIO AS MUNICIPIO_PEP,
		  t2.GRAURELACIONAMENTO AS RELACAO_PEP

      FROM BS_BANKLY_LISTAS T1
           LEFT JOIN BS_PEP_SCORE t2 ON (t1.DOC_NUM = t2.DOC_NUM);
QUIT;

PROC SQL;
   CREATE TABLE BS_BANKLY_PEP AS
   		SELECT DISTINCT
		  	t1.*,
		  (CASE WHEN T3.DOC_NUM IS NOT MISSING THEN 'SIM'
                WHEN T1.DOC_NUM = 0 AND T4.NOME IS NOT MISSING THEN 'SIM' ELSE T1.PEP END) AS PEP1,
		  (CASE WHEN T3.DOC_NUM IS NOT MISSING THEN 9
                WHEN T1.DOC_NUM = 0 AND T4.NOME IS NOT MISSING THEN 9 ELSE 0 END) AS SCORE_PEPINTERNACIONAL,
		  (CASE WHEN T3.DOC_NUM IS NOT MISSING AND t1.CARGO_PEP IS NOT MISSING THEN
                CATX(' ',t1.CARGO_PEP,' e Match em PEP Internacional')
                WHEN T3.DOC_NUM IS NOT MISSING THEn 'Match em PEP Internacional'
				WHEN T4.NOME IS NOT MISSING AND t1.CARGO_PEP IS NOT MISSING THEN
                CATX(' ',t1.CARGO_PEP,' e Match em PEP Internacional')
                WHEN T4.NOME IS NOT MISSING THEn 'Match em PEP Internacional'
                ELSE t1.CARGO_PEP END) format $200. as CARGO_PEP1

   FROM BS_BANKLY_PEP t1
   LEFT JOIN WORK.'_Clone___Renovação_Bankly_PF__Co'n T3 ON (T1.DOC_NUM = T3.DOC_NUM)
   LEFT JOIN WORK.'_Clone___Renovação_Bankly_PF__Co'n T4 ON (T1.NOME = T4.NOME)
    ;

QUIT;

DATA BS_BANKLY_PEP;
SET BS_BANKLY_PEP;
DROP CARGO_PEP PEP;
RENAME CARGO_PEP1 = CARGO_PEP;
RENAME PEP1 = PEP;
RUN;

PROC SQL;
CREATE TABLE BS_BANKLY_ESPECIE AS
SELECT DISTINCT
			A.*,
			COALESCE(B.TOTAL,0) AS Vr_Especie,
			(CASE WHEN B.TOTAL IS MISSING THEN  'Sem Espécie'
                  WHEN B.TOTAL <10000 THEN  'Até 10 mil'
  				  WHEN B.TOTAL >=10000 AND B.TOTAL <20000 THEN 'de 10 a 20 mil'
  	    		  WHEN B.TOTAL >=20000 AND B.TOTAL <=50000 THEN 'de 20 a 50 mil'
		          	ELSE 'Acima de 50 mil' END) AS FAIXA_ESPECIE,
			(CASE WHEN B.TOTAL > 50000 THEN 27
			      WHEN B.TOTAL >=20000 AND B.TOTAL <=50000 THEN 9
                  WHEN B.TOTAL >=10000 AND B.TOTAL < 20000 THEN 3
				  	ELSE 0 END) AS SCORE_ESPECIE

FROM BS_BANKLY_PEP A LEFT JOIN bs_saque B
	ON (A.DOC_NUM = B.DOC_NUM);
QUIT;

PROC FREQ DATA=BS_BANKLY_ESPECIE ; TABLES SCORE_ESPECIE; RUN ;

PROC SQL;
	CREATE TABLE BS_FILTRO_2_PF AS
		SELECT DISTINCT
                DOC_NUM,
				SUM(CASE WHEN SCORE_LISTA > 0 THEN 1 ELSE 0 END)  AS SCORE_LISTA,
                SUM(CASE WHEN SCORE_COAF > 0 THEN 1 ELSE 0 END) AS SCORE_COAF,
                SUM(CASE WHEN SCORE_ALERTA6M > 0 THEN 1 ELSE 0 END) AS SCORE_ALERTA6M,
                SUM(CASE WHEN SCORE_ESPECIE > 0 THEN 1 ELSE 0 END) AS SCORE_ESPECIE,

	GROUP BY DOC_NUM

;
QUIT;

PROC SQL;
   CREATE TABLE WORK.BS_FILTRO_2_FIM AS
   SELECT DISTINCT
          t1.DOC_NUM,
          (CASE WHEN (CASE WHEN t1.SCORE_COAF > 0 THEN 1 ELSE 0 END) <= (CASE WHEN t2.SCORE_COAF > 0 THEN 1 ELSE 0 END) THEN 0 ELSE 1 END) AS SCORE_COAF,
          (CASE WHEN (CASE WHEN t1.SCORE_ALERTA6M > 0 THEN 1 ELSE 0 END) <= (CASE WHEN t2.SCORE_ALERTA6M > 0 THEN 1 ELSE 0 END) THEN 0 ELSE 1 END) AS SCORE_ALERTA6M,
          (CASE WHEN (CASE WHEN t1.SCORE_LISTA > 0 THEN 1 ELSE 0 END) <= (CASE WHEN t2.SCORE_LISTA > 0 THEN 1 ELSE 0 END) THEN 0 ELSE 1 END) AS SCORE_LISTA,
          (CASE WHEN (CASE WHEN t1.SCORE_PEP > 0 THEN 1 ELSE 0 END) <= (CASE WHEN t2.SCORE_PEP > 0 THEN 1 ELSE 0 END) THEN 0 ELSE 1 END) AS SCORE_PEP,
          (CASE WHEN (CASE WHEN t1.SCORE_ESPECIE > 0 THEN 1 ELSE 0 END) <= (CASE WHEN t2.SCORE_ESPECIE > 0 THEN 1 ELSE 0 END) THEN 0 ELSE 1 END) AS SCORE_ESPECIE
		  

      FROM BS_PF_BANKLY_SCORE t1
      LEFT JOIN BS_FILTRO_2_PF t2 ON (t1.DOC_NUM = t2.DOC_NUM)
	  WHERE T1.riscoMatriz = 3;
QUIT;

FLAG = (SCORE_LISTA + SCORE_ALERTA6M + SCORE_COAF + SCORE_PEP + SCORE_ESPECIE);
RUN;

	CREATE TABLE BS_RENOVACAO_PF_OTIMIZADA AS

          t1.SCORE_ESPECIE,

                     t1.LISTA_CEPIM) > 0 THEN 1 ELSE 0 END) AS ListaExterna,

                THEN 1 ELSE 0 END) AS ListaInterna,

 LEFT JOIN BS_UNICA_PF T2 ON (T1.NuCpfCnpj = T2.NuCpfCnpj)

PROC SQL;
   CREATE TABLE BS_PF_OTIMIZADA_FRAUDES AS
   SELECT DISTINCT
          t1.*,
		  (CASE WHEN t2.DOC_NUM IS NOT MISSING THEN 'Análise Otimizada - Nao Renovar - Lista de Fraudes'
                ELSE 'Análise Otimizada - Renovar - Retirado da Lista de Fraudes'  END) FORMAT $200. AS FLAG_OTIMIZADA

      FROM BS_RENOVACAO_PF_OTIMIZADA t1
	  LEFT JOIN (SELECT T1.DOC_NUM FROM PLD.LISTA_BLOQUEADOS t1
                  WHERE t1.LISTA = 'FRAUDES') T2 ON (T2.DOC_NUM = t1.DOC_NUM)

PROC SQL;
   CREATE TABLE BS_PF_OTIMIZADA_IMPEDITIVA AS
   SELECT DISTINCT
          t1.*,
		  'Análise Otimizada - Nao Renovar - Lista Impeditiva' FORMAT $200. AS FLAG_OTIMIZADA

     FROM BS_RENOVACAO_PF_OTIMIZADA t1
     INNER JOIN PLD.LISTA_BLOQUEADOS T3 ON (T1.DOC_NUM = T3.DOC_NUM)
	 LEFT JOIN BS_PF_OTIMIZADA_FRAUDES T2 ON (T1.DOC_NUM = T2.DOC_NUM)
	 WHERE t3.TPLISTA = 'IMPEDITIVO' AND T2.DOC_NUM IS MISSING AND T1.SCORE_LISTA = 27
           and t1.DescAlertas NOT CONTAINS 'Fraudes' AND  t1.DescAlertas NOT CONTAINS  'Recusa'
		   and t1.DescAlertas NOT CONTAINS 'PEPInternacional'
		;

SET BS_PF_OTIMIZADA_IMPEDITIVA BS_PF_OTIMIZADA_FRAUDES;
RUN;



PROC SQL;
   CREATE TABLE BS_PF_OTIMIZADA_RESTRITIVA AS
   SELECT DISTINCT
          t1.*,
		  (CASE WHEN (T1.ListaExterna = 1 or T1.ListaInterna = 1) AND T3.DOC_NUM IS MISSING
                THEN 'Análise Otimizada - Renovar - Saiu da Lista Restritiva'
		        WHEN T1.ListaExterna = 1 AND T3.DOC_NUM IS NOT MISSING THEN 'Analisar - Lista Externa'
		        WHEN T1.ListaInterna = 1 AND T3.DOC_NUM IS NOT MISSING THEN
				     (CASE WHEN t2.'Status da consulta'n = 'APROVADO' THEN 'Análise Otimizada - Renovar - Aprovado Dimensa'
					       WHEN t2.'Status da consulta'n = 'REPROVADO' THEN 'Análise Otimizada - Nao Renovar - Reprovado Dimensa'
                           ELSE 'Rodar Dimensa' END) ELSE 'Nada' END) FORMAT $200. AS FLAG_OTIMIZADA

     FROM BS_RENOVACAO_PF_OTIMIZADA t1
     LEFT JOIN PLD.LISTA_BLOQUEADOS T3 ON (T1.DOC_NUM = T3.DOC_NUM)
	 LEFT JOIN BS_PROCESSOS_MATRIZ_ND T2 ON (T1.DOC_NUM = T2.DOCUMENTO)
	 LEFT JOIN BS_PF_OTIMIZADA_FIM T4 ON (T1.DOC_NUM = T4.DOC_NUM)
	 WHERE T4.DOC_NUM IS MISSING  AND t1.DescAlertas not CONTAINS 'PEPInternacional' AND
     (CASE WHEN (T1.ListaExterna = 1 or T1.ListaInterna = 1) AND T3.DOC_NUM IS MISSING
                THEN 'Análise Otimizada - Renovar - Saiu da Lista Restritiva'
		        WHEN T1.ListaExterna = 1 AND T3.DOC_NUM IS NOT MISSING THEN 'Analisar - Lista Externa'
		        WHEN T1.ListaInterna = 1 AND T3.DOC_NUM IS NOT MISSING THEN
				     (CASE WHEN t2.'Status da consulta'n = 'APROVADO' THEN 'Análise Otimizada - Renovar - Aprovado Dimensa'
					       WHEN t2.'Status da consulta'n = 'REPROVADO' THEN 'Análise Otimizada - Nao Renovar - Reprovado Dimensa'
                           ELSE 'Rodar Dimensa' END) ELSE 'Nada' END) not = 'Nada'
       and t1.DescAlertas NOT CONTAINS 'Fraudes' AND T1.SCORE_FINAL = 27
		;

QUIT;

DATA BS_PF_OTIMIZADA_FIM;
FORMAT FLAG_OTIMIZADA $char300.;
SET BS_PF_OTIMIZADA_FIM BS_PF_OTIMIZADA_RESTRITIVA;
RUN;
