			SELECT @mesant=convert(CHAR(10),dateadd(dd,-1,dateadd(mm,-1,@today)),101)

     		SELECT DISTINCT
			co.NuContrato as NuContrato_con,
            co.DtContrato,
            prp.NuProposta as NuProposta_con,
            co.NuCpfCnpj as NuCpfCnpj_con,

				CREATE TABLE DATASEG.MktPlace_Contratos_Final AS
				SELECT DISTINCT

				(VrSeguro/prp.VrMercado) as Pct_VrSeguro,
				CASE
					WHEN prp.VrMercado <= 10000 THEN 'Até 10K'
					WHEN prp.VrMercado > 10000 AND prp.VrMercado <= 20000 THEN '10K - 20K'
					WHEN prp.VrMercado > 20000 AND prp.VrMercado <= 30000 THEN '20K - 30K'
					WHEN prp.VrMercado > 30000 AND prp.VrMercado <= 40000 THEN '30K - 40K'
					WHEN prp.VrMercado > 40000 AND prp.VrMercado <= 50000 THEN '40K - 50K'
					WHEN prp.VrMercado > 50000 AND prp.VrMercado <= 60000 THEN '50K - 60K'
					WHEN prp.VrMercado > 60000 AND prp.VrMercado <= 70000 THEN '60K - 70K'
					WHEN prp.VrMercado > 70000 AND prp.VrMercado <= 80000 THEN '70K - 80K'
					WHEN prp.VrMercado > 80000 AND prp.VrMercado <= 90000 THEN '80K - 90K'
					WHEN prp.VrMercado > 90000 AND prp.VrMercado <= 100000 THEN '90K - 100K'
					WHEN prp.VrMercado > 100000 AND prp.VrMercado <= 110000 THEN '100K - 110K'
					WHEN prp.VrMercado > 110000 AND prp.VrMercado <= 120000 THEN '110K - 120K'
					WHEN prp.VrMercado > 120000 AND prp.VrMercado <= 130000 THEN '120K - 130K'
					WHEN prp.VrMercado > 130000 AND prp.VrMercado <= 140000 THEN '130K - 140K'
					WHEN prp.VrMercado > 140000 AND prp.VrMercado <= 150000 THEN '140K - 150K'
					WHEN prp.VrMercado > 150000 AND prp.VrMercado <= 160000 THEN '150K - 160K'
					WHEN prp.VrMercado > 160000 AND prp.VrMercado <= 170000 THEN '160K - 170K'
					WHEN prp.VrMercado > 170000 AND prp.VrMercado <= 180000 THEN '170K - 180K'
					WHEN prp.VrMercado > 180000 AND prp.VrMercado <= 190000 THEN '180K - 190K'
					WHEN prp.VrMercado > 190000 AND prp.VrMercado <= 200000 THEN '190K - 200K'
					WHEN prp.VrMercado > 200000 THEN 'Acima de 200K'
					ELSE 'Outros'
				END as Faixa_VrMercado,
				Case when substr(NuChassis,1,2) in ("93","94","95","96","97","98","99","9B","9D","9b") then "Nacional"
				When NuChassis = "" then ""

				AND prp.AAModelo = moli.AaModelo
