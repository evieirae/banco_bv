PROC SQL;
CREATE TABLE SIMULACAO_TRATADA AS
SELECT
A.*,D0,D3,D10,D15,D20,D30,
CASE
	WHEN Mensageria in ('0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%',
						'0. Aprovado','1. Aprovado com rastreador','2. Reanálise psicométrico')
	AND Mensageria2 NOT IN ('3. Recusado', '4. Recusado Reversível','6. Recusado','6. Plano Recusado')
	THEN 'Mensageria Positiva'
	WHEN A.Mensageria IN ('6. Recusado','3. Recusado','6. Plano Recusado')
	OR A.Mensageria in ('0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%',
						'0. Aprovado','1. Aprovado com rastreador','2. Reanálise psicométrico')
	AND Mensageria2 in ('3. Recusado', '4. Recusado Reversível','6. Recusado','6. Plano Recusado')
	THEN 'Recusado'
	ELSE 'Demais Msgs' END AS AgrupMensageria,
CASE
	WHEN A.PcEntradaMinima BETWEEN 0 AND 10 THEN 'Ate 10%'
	WHEN A.PcEntradaMinima BETWEEN 10 AND 20 THEN 'MINIMA ATE 20%'
	WHEN A.PcEntradaMinima BETWEEN 20 AND 30 THEN 'MINIMA ATE 30%'
	WHEN A.PcEntradaMinima BETWEEN 30 AND 40 THEN 'MINIMA ATE 40%'
	WHEN A.PcEntradaMinima BETWEEN 40 AND 50 THEN 'MINIMA ATE 50%'
	WHEN A.PcEntradaMinima BETWEEN 50 AND 60 THEN 'MINIMA ATE 60%'
	WHEN A.PcEntradaMinima BETWEEN 60 AND 70 THEN 'MINIMA ATE 70%'
	WHEN A.PcEntradaMinima BETWEEN 70 AND 80 THEN 'MINIMA ATE 80%'
	WHEN A.PcEntradaMinima BETWEEN 80 AND 90 THEN 'MINIMA ATE 90%'
	WHEN A.PcEntradaMinima BETWEEN 90 AND 100 THEN 'MINIMA ATE 100%'
	ELSE 'Desconhecido' END AS FaixaEntradaMinima,
B.VrPropostaOriginada,
B.VpScore,
CASE WHEN B.VrTotalSeguro IS NOT NULL THEN 1 ELSE 0 END AS FlagSeguro,
CASE
	WHEN B.NuScoreFim < 5 THEN 'Ate 5'
	WHEN B.NuScoreFim >= 5 AND B.NuScoreFim < 6 THEN '5 - 6'
	WHEN B.NuScoreFim >= 6 AND B.NuScoreFim < 7 THEN '6 - 7'
	WHEN B.NuScoreFim >= 7 AND B.NuScoreFim < 8 THEN '7 - 8'
	WHEN B.NuScoreFim >= 8 AND B.NuScoreFim < 9 THEN '8 - 9'
	WHEN B.NuScoreFim >= 9 AND B.NuScoreFim < 10 THEN '9 - 10'
	ELSE 'Desconhecido' END AS FxScore,
CASE
	WHEN AH.NmSubEstruturaComercialC = "Canal Pesados" THEN "Canal Exclusivo"
	ELSE "Canal Hibrido" END AS TipoCanal,
CASE
	WHEN A.Mensageria IN( '0. Aprovado 100%','0. Aprovado') THEN 'PAP'
	WHEN A.DtSimulacaoInicial >= '09Apr2024'd AND SUBSTR(A.NuCPFCNPJ,6,2) BETWEEN '00' AND '04' THEN 'Politica Desafiante'
	ELSE 'Politica Atual' END AS TpPoliticaCredito,
	AH.NmTipoParceiroComercial AS TipoParceiro
FROM SIMULACAO AS A
LEFT JOIN PRODUCAO AS B ON A.NuProposta = B.NuProposta
LEFT JOIN TBD0 AS C ON A.DtSimulacaoInicial=C.DtTempo
LEFT JOIN TBD3 AS F ON A.DtSimulacaoInicial=F.DtTempo
LEFT JOIN TBD10 AS M ON A.DtSimulacaoInicial=M.DtTempo
LEFT JOIN TBD15 AS R ON A.DtSimulacaoInicial=R.DtTempo
LEFT JOIN TBD20 AS W ON A.DtSimulacaoInicial=W.DtTempo
LEFT JOIN TBD30 AS AG ON A.DtSimulacaoInicial=AG.DtTempo
LEFT JOIN bases.TBPARCEIROCOMERCIAL AS AH ON A.CdParceiroComercial = AH.CdParceiroComercial
;RUN;
