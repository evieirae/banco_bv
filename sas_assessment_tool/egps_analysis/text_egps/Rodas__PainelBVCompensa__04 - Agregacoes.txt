/* Criação da tabela agregada */
PROC SQL;
   CREATE TABLE WORK.TbProducaoPricing_Aggregated AS
   SELECT AnoMes,
          CdParceiroComercial,
          SUM(VrProducaoBVCompensa) AS VrProducaoLiquidaTotal
   FROM TbProducao
   GROUP BY AnoMes, CdParceiroComercial;
QUIT;

DATA WORK.TbProducaoPricing_Aggregated;
   SET WORK.TbProducaoPricing_Aggregated;
   AnoMesNum = INPUT(AnoMes, 6.);
   AnoMesPrev = AnoMesNum - 1;
   IF MOD(AnoMesPrev, 100) = 0 THEN AnoMesPrev = AnoMesPrev - 88; /* Ajuste para transição de ano */
   FORMAT AnoMesPrev 6.;
   AnoMesPrevChar = PUT(AnoMesPrev, 6.);
RUN;

/* Junção para obter os valores do mês atual e anterior */
PROC SQL;
   CREATE TABLE WORK.TbProducaoPricing_WithPrev AS
   SELECT a.AnoMes,
          a.CdParceiroComercial,
          a.VrProducaoLiquidaTotal,
          b.VrProducaoLiquidaTotal AS VrProducaoLiquidaPrev
   FROM WORK.TbProducaoPricing_Aggregated a
   LEFT JOIN WORK.TbProducaoPricing_Aggregated b
   ON a.CdParceiroComercial = b.CdParceiroComercial AND a.AnoMesPrevChar = b.AnoMes;
QUIT;

/* Cálculo da média dos últimos três meses para cada AnoMes e CdParceiroComercial */
PROC SQL;
   CREATE TABLE WORK.TbProducaoPricing_WithTriAvg AS
   SELECT a.AnoMes,
          a.CdParceiroComercial,
          a.VrProducaoLiquidaTotal,
          (SELECT MEAN(b.VrProducaoLiquidaTotal)
           FROM WORK.TbProducaoPricing_Aggregated b
           WHERE b.CdParceiroComercial = a.CdParceiroComercial
             AND INTCK('month', INPUT(b.AnoMes, yymmn6.), INPUT(a.AnoMes, yymmn6.)) BETWEEN -3 AND -1) AS VrProducaoLiquidaTriAvg
   FROM WORK.TbProducaoPricing_Aggregated a;
QUIT;
