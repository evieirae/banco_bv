	CREATE TABLE CONSULTA_AJUIZAMENTO AS SELECT
	* from
	CONNECTION TO perm	(
			SELECT	A.NuContratoGestao,
					A.SqContratoFinanceiro,
					A.VRAJUIZ,
					DATE(A.Dt_DESP) AS Dt_DESP,
					DATE_DIFF(Dt_DESP, MIN(CASE WHEN	B.St_Parcela <> 'Q' OR
												(B.St_Parcela = 'Q' AND
												B.Dt_Ultimo_Recebimento >= Dt_DESP)  THEN Dt_Vencimento ELSE NULL END), DAY) AS Atraso,
					SUM(CASE WHEN	B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									B.Dt_Ultimo_Recebimento >= Dt_DESP)  THEN Vr_Parcela ELSE 0 END) AS Risco_Aberto,
					COUNT(CASE WHEN	B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									B.Dt_Ultimo_Recebimento >= Dt_DESP)  THEN Vr_Parcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
					SUM(CASE WHEN	(B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									DATE(B.Dt_Ultimo_Recebimento) >= Dt_DESP)) AND
									DATE(B.Dt_Vencimento) < A.Dt_DESP THEN Vr_Parcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
					COUNT(CASE WHEN	(B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									DATE(B.Dt_Ultimo_Recebimento) >= Dt_DESP)) AND
									DATE(B.Dt_Vencimento) < A.Dt_DESP THEN Vr_Parcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,

												B.Dt_Ultimo_Recebimento >= Dt_DESP)  THEN Dt_Vencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,

												(B.St_Parcela = 'Q' AND
												B.Dt_Ultimo_Recebimento >= Dt_DESP)  THEN Nu_parcela ELSE NULL END) as PARCELA
			FROM	(

SELECT NuContratoGestao,
					SqContratoFinanceiro,
					DATE(Dt_DESP) AS Dt_DESP,
					sum(SAFE_CAST(valor_desp AS FLOAT64)) AS VRAJUIZ

			WHERE	dt_desp > '2021-01-01 00:00:00' and cod_desp in (14,80)
			GROUP BY NuContratoGestao, SqContratoFinanceiro, Dt_DESP

			) AS A
					LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B ON	A.NuContratoGestao = B.cd_contrato AND
														A.SqContratoFinanceiro = B.Sq_Contrato_Financeiro
			GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP, A.VRAJUIZ
	)
