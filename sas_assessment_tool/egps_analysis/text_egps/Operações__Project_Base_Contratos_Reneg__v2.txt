CREATE TABLE WORK.BASE_RENEG_GCON_GESTAO AS

SELECT
	e.nucontrato,
	e.NuContratoLegado,
	e.dtcontratofinanceiro,
	e.vrfinanciamento,
	e.VrContrato,
	e.QtParcelas,
	e.VrPrestacao,
	e.total_pmt_abertas,
	e.FLUXO_ATUAL,
	e.VRBNDU,
	e.BNDU,
	e.VrAviso,
	e.Aviso_Cliente,
	CASE WHEN a.Nucontrato = . THEN 0 ELSE a.Nucontrato END as Nucontrato_Reneg_Gcon,
	CASE WHEN b.Nucontrato = . THEN 0 ELSE b.Nucontrato END as Nucontrato_Atraso_Gcon,
	CASE WHEN c.Nucontrato = . THEN 0 ELSE e.Nucontrato END as Nucontrato_EmDia_Gcon,

	CASE WHEN a.VrTaxaContrato IS NULL THEN 0 ELSE a.VrTaxaContrato END AS VrTaxaContrato_Reneg,
	CASE WHEN b.VrTaxaContrato_Atraso IS NULL THEN 0 ELSE b.VrTaxaContrato_Atraso END AS VrTaxaContrato_Atraso,
	CASE WHEN c.VrTaxaContrato_EmDia IS NULL THEN 0 ELSE c.VrTaxaContrato_EmDia END AS VrTaxaContrato_EmDia,

	CASE WHEN a.VrFuturo IS NULL THEN 0 ELSE a.VrFuturo END AS VrFuturo_Reneg,
	CASE WHEN b.VrFuturo IS NULL THEN 0 ELSE b.VrFuturo END  AS VrFutro_Atraso,
	CASE WHEN c.VrFuturo IS NULL THEN 0 ELSE c.VrFuturo END AS VrFutro_EmDia,

	CASE
	WHEN b.Nucontrato = . AND c.Nucontrato = . THEN 0
	WHEN b.Nucontrato = . AND c.Nucontrato IS NOT NULL THEN (0+c.VrFuturo)
	WHEN c.Nucontrato = . AND B.Nucontrato THEN (0+b.VrFuturo)
	ELSE (b.VrFuturo+c.VrFuturo) END AS Total_Valor_Futuro,

	CASE
	WHEN b.Nucontrato = . AND c.Nucontrato = . THEN 0
	WHEN b.Nucontrato = . AND C.Nucontrato IS NOT NULL THEN ROUND((0+c.VrFuturo)-(0+c.VrTaxaContrato_EmDia),2.2)
	WHEN c.Nucontrato = . AND B.Nucontrato IS NOT NULL THEN ROUND((b.VrFuturo+0)-(b.VrTaxaContrato_Atraso+0),2.2)
	ELSE ROUND((b.VrFuturo+c.VrFuturo)-(b.VrTaxaContrato_Atraso+c.VrTaxaContrato_EmDia),2.2) END AS saldo_Ganho,

	CASE WHEN A.VrFuturo IS NULL THEN 0 ELSE A.VrFuturo END AS VrFuturo_Reneg,
	CASE WHEN a.Valida_Ganho_Auferido IS NULL THEN 0 ELSE a.Valida_Ganho_Auferido END AS Valida_Ganho_Auferido,
	e.QtParcelas AS QtParcelas_Reneg,
	
	CASE
	WHEN c.Nucontrato = . AND B.Nucontrato = . THEN 0
	WHEN c.Nucontrato IS NOT NULL THEN ROUND((c.VrTaxaContrato_EmDia/e.QtParcelas),2.2)
	WHEN c.Nucontrato = . AND b.VrTaxaContrato_Atraso IS NOT NULL THEN ROUND((b.VrTaxaContrato_Atraso/e.QtParcelas),2.2)
	ELSE ROUND((a.VrTaxaContrato/e.QtParcelas),2.2) END AS Estorno
FROM WORK.BASE_RENEG_COMPLEMENTO AS E
LEFT JOIN WORK.BASE_RENEG_CONTABIL AS A ON E.Nucontrato = A.Nucontrato
LEFT JOIN WORK.BASE_RENEG_PMT_ATRASO b ON E.nucontrato = b.nucontrato
LEFT JOIN WORK.BASE_RENEG_PMT_EM_DIA  c ON E.nucontrato = c.nucontrato

;
QUIT;

CREATE TABLE WORK.BASE_RENEG_GCON_CALCULO AS

SELECT
	A.nucontrato,
	A.NuContratoLegado,
	A.vrfinanciamento,
	A.VrContrato,
	A.VrPrestacao,
	datepart(A.dtcontratofinanceiro) AS Data_Renegociado FORMAT DDMMYY10.,
CASE
	WHEN A.Nucontrato_Reneg_Gcon = 0 THEN 'NÃO' ELSE 'SIM' END AS RENEG_NA_BASE_GCON,
CASE
	WHEN A.Nucontrato_EmDia_Gcon = 0 AND A.Nucontrato_Atraso_Gcon = 0 THEN 'NÃO' ELSE 'SIM' END AS INFORMACAO_FLUXO_INICIAL_GCON,
	A.VrTaxaContrato_Reneg,
	A.VrTaxaContrato_Atraso,
	A.VrTaxaContrato_EmDia,
	A.VrFuturo_Reneg,
	A.VrFutro_Atraso,
	A.VrFutro_EmDia,
	A.Total_Valor_Futuro,
	A.saldo_Ganho,
	A.Valida_Ganho_Auferido,
	CASE WHEN a.total_pmt_abertas IS NULL THEN 'FECHADO' ELSE 'ABERTO' END AS STATUS,
	A.QtParcelas_Reneg,
	a.total_pmt_abertas,
	a.FLUXO_ATUAL,
	a.VRBNDU,
	a.BNDU,
	a.VrAviso,
	a.Aviso_Cliente,
	a.Estorno,
	
	ROUND(CASE WHEN A.Nucontrato_Reneg_Gcon = 0 THEN 0 ELSE ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg) END,0.01) AS Calc_Sistema_Reneg,
	ROUND(CASE WHEN A.Nucontrato_Reneg_Gcon = 0 THEN 0 ELSE (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg) END,0.01) AS Calc_Sistema_Ganho_Auferido,
	ROUND(CASE WHEN A.Nucontrato_Reneg_Gcon = 0 THEN 0 ELSE ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.Valida_Ganho_Auferido/a.QtParcelas_Reneg) END,0.01) AS Calc_Sistema_Total_Pmt,


	ROUND(CASE
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.Nucontrato_Atraso_Gcon = 0 THEN 0
	WHEN a.Nucontrato_EmDia_Gcon > 0 THEN ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_EmDia/a.QtParcelas_Reneg)
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.Nucontrato_Atraso_Gcon > 0 THEN ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_Atraso/a.QtParcelas_Reneg)
	ELSE ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_Reneg/a.QtParcelas_Reneg) END,0.01) AS Calc_Select_Reneg,

	ROUND(CASE
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.Nucontrato_Atraso_Gcon = 0 THEN 0
	WHEN a.Nucontrato_EmDia_Gcon > 0 THEN (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_EmDia/a.QtParcelas_Reneg)
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.Nucontrato_Atraso_Gcon > 0 THEN (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_Atraso/a.QtParcelas_Reneg)
	ELSE (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_Reneg/a.QtParcelas_Reneg) END,0.01) AS Calc_Select_Ganho_Auferido,

	ROUND(CASE
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.Nucontrato_Atraso_Gcon = 0 THEN 0
	WHEN a.Nucontrato_EmDia_Gcon > 0 THEN ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_EmDia/a.QtParcelas_Reneg) + (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_EmDia/a.QtParcelas_Reneg)
	WHEN a.Nucontrato_EmDia_Gcon = 0 AND a.VrTaxaContrato_Atraso > 0 THEN ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_Atraso/a.QtParcelas_Reneg) + (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_Atraso/a.QtParcelas_Reneg)
	ELSE ((A.VrFuturo_Reneg-A.Valida_Ganho_Auferido)/a.QtParcelas_Reneg)+(a.VrTaxaContrato_Reneg/a.QtParcelas_Reneg) + (a.Valida_Ganho_Auferido/a.QtParcelas_Reneg)-(a.VrTaxaContrato_Reneg/a.QtParcelas_Reneg)END,0.01) AS Calc_Select_Total_Pmt,



	ROUND(CASE 	WHEN a.VrFuturo_Reneg = 0 THEN 0 ELSE (a.VrFuturo_Reneg/a.QtParcelas_Reneg)-(a.saldo_Ganho/a.QtParcelas_Reneg) END,0.01) AS Calc_Novo_Reneg,
	ROUND(CASE WHEN a.saldo_Ganho = 0 THEN 0 ELSE (a.saldo_Ganho/a.QtParcelas_Reneg) END,0.01) AS Calc_Novo_Ganho_Auferido,
	ROUND(CASE WHEN a.VrFuturo_Reneg = 0 THEN 0 ELSE (a.VrFuturo_Reneg/a.QtParcelas_Reneg)-(a.saldo_Ganho/a.QtParcelas_Reneg)+(a.saldo_Ganho/a.QtParcelas_Reneg) END,0.01) AS  Calc_Novo_Total_Pmt

FROM WORK.BASE_RENEG_GCON_GESTAO AS A
;
QUIT;
