  %let path      = %sysfunc(pathname(&libname.));
  %let work      = %sysfunc(pathname(work));
  %let libname_l = %sysfunc(lowcase(&libname.));
  %let base_l    = %sysfunc(lowcase(&base.));

  %put NOTE: path=&path.;
  %put NOTE: work=&work.;
  %put NOTE: libname=&libname_l.;
  %put NOTE: base=&base_l.;

  
  x "cp &path./&base_l..sas7* &work.";

  
  x "gunzip -f &work./&base_l..sas7*";
%mend;

proc sql;
  create table work.TABELA_FINAL as
  select
      a.*,
      b.SAFRA as SAFRA_C,
      b.Fase_BB,
      b.FxRtgCPF_BV,
      b.FxRtgCPF_Backtest,
      z.NmFxRecebimento,
      z.Fx_Recebimento_NEW,
      case
        when z.NuContrato_key is not null then 1
        else 0
      end as ENTRADA_COB
  from acum_key as a
  left join bvo_key as b
    on a.NuContrato_key = b.NuContrato_key
  left join zero_key as z
    on a.NuContrato_key = z.NuContrato_key
   and a.SAFRA = z.SAFRA_M1
  ;
quit;

data work.base_final;
  set work.tabela_final;

  length
    Fase_BB_norm         $30
    NmProdutoRisco_norm  $60
    FxRtgCPF_BV_norm     $40
    FxRec_norm           $80
    FX_ATRASO_norm       $80
  ;

  
  if missing(strip(Fase_BB)) then Fase_BB_norm = 'OUTROS';
  else do;
    Fase_BB_norm = upcase(strip(Fase_BB));
    if Fase_BB_norm in (
      '-', '.', 'NA', 'N/A', 'N.D.', 'ND', 'N I', 'NI', 'NULL',
      'SEM INFORMACAO', 'SEM INFORMAÇÃO'
    ) then Fase_BB_norm = 'OUTROS';
  end;

  
  NmProdutoRisco_norm = upcase(strip(NmProdutoRisco));
  FxRtgCPF_BV_norm    = upcase(strip(FxRtgCPF_BV));

  
  if vtype(Carteira)='N' then Carteira_num = Carteira;
  else Carteira_num = input(strip(Carteira), commax17.2);

  
  fl_entrada_cob = (ENTRADA_COB = 1);

  
  FxRec_norm = upcase(strip(Fx_Recebimento_NEW));

  
  if missing(Fx_Recebimento_NEW) then fl_performou = 0;
  else if index(FxRec_norm,'SEM RECEBIMENTO') > 0 then fl_performou = 0;
  else fl_performou = 1;

  
  FX_ATRASO_norm = upcase(strip(FX_ATRASO));

  
  fl_em_dia = (index(FX_ATRASO_norm,'EM DIA') > 0);

  
  FX_ATRASO_NUM = .;

  
  if prxmatch('/^\d{1,4}/', FX_ATRASO_norm) then do;
    length _lead $4;
    _lead = prxchange('s/^(\d{1,4}).*$/$1/', 1, FX_ATRASO_norm);
    FX_ATRASO_NUM = input(_lead, best.);
  end;
  
  else if prxmatch('/\d{4}/', FX_ATRASO_norm) then do;
    length _m $4;
    _m = prxchange('s/.*?(\d{4}).*$/$1/', 1, FX_ATRASO_norm);
    FX_ATRASO_NUM = input(_m, best.);
  end;

  fl_acima_07 = (index(FX_ATRASO_norm,'ACIMA 07') > 0);

  
  fl_over90  = (fl_acima_07 or (not missing(FX_ATRASO_NUM) and FX_ATRASO_NUM >= 91));

  
  fl_over180 = (fl_acima_07 or (not missing(FX_ATRASO_NUM) and FX_ATRASO_NUM >= 151));

  
  Carteira_EmDia          = Carteira_num * fl_em_dia;
  Carteira_Entrada_EmDia  = Carteira_num * (fl_em_dia and fl_entrada_cob);

  Carteira_Perf           = Carteira_num * fl_performou;
  Carteira_PerfEnt        = Carteira_num * (fl_entrada_cob and fl_performou);

  Carteira_Over90         = Carteira_num * fl_over90;
  Carteira_Over180        = Carteira_num * fl_over180;

  format Carteira_num commax17.2;
run;

%macro cnt_distinct(in=, out=, byvars=, keepvars=);
  proc sort data=&in(keep=&keepvars NuContrato)
            out=&out nodupkey;
    by &byvars NuContrato;
  run;

  proc summary data=&out nway missing;
    class &byvars;
    output out=&out._cnt(drop=_type_);
  run;

  data &out;
    set &out._cnt;
    Qtd_Contratos = _freq_;
    drop _freq_;
  run;

  proc datasets lib=work nolist;
    delete &out._cnt;
  quit;
%mend;
