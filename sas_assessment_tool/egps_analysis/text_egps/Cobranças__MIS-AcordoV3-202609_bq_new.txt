CREATE TABLE BA_ACORDO AS
SELECT *
FROM CONNECTION TO PERM (WITH QUERY_1 AS (
SELECT
    ac.ahcollid AS CdAgente,
    CAST(ac.ahacct AS STRING) AS Chave,
    ac.ahdscqb AS DsMtvoQuebra,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(pb.pbdtciclo)) AS DtCicloPgto,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtClicloCriacao)) AS DtClicloCriacao,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtCriacao)) AS DtCriacao,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtFim)) AS DtFim,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtInicio)) AS DtInicio,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(pb.pbdttransac)) AS DtPgto,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(u3.u3dtpriatr)) AS DtPrimeiroAtraso,
    (
        SELECT MIN(FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(a.ahdt))) AS MIN_AHDT
        FROM bv-pdbd-prd.trusted_rcvry.agrhdr AS A
        WHERE ac.ahacct = a.ahacct
        AND a.ahdt >= P2.DtQuebra
    ) AS DtPromessa,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtQuebra)) AS DtQuebra,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtQuebraAcordo)) AS DtQuebraAcordo,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtQuitacao)) AS DtQuitacao,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(pc.apduedt)) AS DtVencimento,
    FORMAT_DATETIME('%d/%m/%Y %H:%M:%S',DATETIME(P2.DtVencQuebra)) AS DtVencQuebra,
    ag.agfirm AS NmAgente,
    ac.ahssnum AS NuCpfCnpj,
    CASE
        WHEN DATE(P2.DtClicloCriacao) >= P2.DtInicio THEN 'FLUXO'
        WHEN P2.DtClicloCriacao IS NULL AND DATE(P2.DtCriacao) >= P2.DtInicio THEN 'FLUXO1'
        ELSE 'COLCHAO'
    END AS Perfil,
    pc.apflag AS StParcela,
    P2.Tipo AS Tipo,
    P2.TpEntrada AS TpEntrada,
    CASE
        WHEN UPPER(ac.ahdscqb) LIKE '%WEBSERVICE%' THEN 1
        ELSE 0
    END AS FlQuebraManual,
    P2.NuAcordo AS NuAcordo,
    CAST(pc.apdetid AS INT64) + 1 AS NuParcela,
    P2.Plano AS Plano,
    P2.PMTQuebra + 1 AS PMTQuebra,
    ac.ahtotpmt AS VrAcordo,
    pc.apamt AS VrParcela
FROM
(
    SELECT
        P1.DtInicio AS DtInicio,
        P1.DtFim AS DtFim,
        P1.Tipo AS Tipo,
        P1.NuAcordo AS NuAcordo,
        P1.DtCriacao AS DtCriacao,
        P1.Plano AS Plano,
        P1.DtQuebraAcordo AS DtQuebraAcordo,
        P1.QdPMTPgto AS QdPMTPgto,
        P1.MAXVencPgto AS MAXVencPgto,
        P1.DtCriacaoCliclo AS DtCriacaoCliclo,
        P1.MAXDtCicloPgto AS MAXDtCicloPgto,
        P1.DtVencQuebra AS DtVencQuebra,
        P1.PMTQuebra AS PMTQuebra,
        COALESCE(P1.DtCriacaoCliclo, P1.DtCriacao) AS DtClicloCriacao,
        COALESCE(P1.DtQuebraAcordo, P1.DtVencQuebra) AS DtQuebra,
        CASE
            WHEN P1.Plano = P1.QdPMTPgto THEN COALESCE(P1.MAXDtCicloPgto, P1.MAXVencPgto)
        END AS DtQuitacao,
        CASE
            WHEN P1.Tipo <> 'MIGRA' THEN 'CYBER'
            ELSE P1.Tipo
        END AS TpEntrada
    FROM
    (
        SELECT
            DATE('2026-09-01') AS DtInicio,
            DATE_SUB(DATE_ADD(DATE('2026-09-01'), INTERVAL 1 MONTH), INTERVAL 1 DAY) AS DtFim,
            a.ahtype AS Tipo,
            a.ahid AS NuAcordo,
            a.ahdt AS DtCriacao,
            CAST(a.ahprd AS INT64) + 1 AS Plano,
            a.ahdtqb AS DtQuebraAcordo,
            SUM(CASE WHEN b.apflag = 'C' THEN 1 ELSE 0 END) AS QdPMTPgto,
            MAX(CASE WHEN b.apflag = 'C' THEN b.apduedt END) AS MAXVencPgto,
            MIN(CASE WHEN b.apflag = 'C' AND CAST(b.apdetid AS INT64) = 0 THEN c.pbdtciclo END) AS DtCriacaoCliclo,
            MAX(CASE WHEN b.apflag = 'C' THEN c.pbdtciclo END) AS MAXDtCicloPgto,
            MIN(CASE WHEN b.apflag = 'R' THEN b.apduedt END) AS DtVencQuebra,
            MIN(CASE WHEN b.apflag = 'R' THEN CAST(b.apdetid AS INT64) END) AS PMTQuebra
        FROM
            bv-pdbd-prd.trusted_rcvry.agrhdr AS A
        INNER JOIN
            bv-pdbd-prd.trusted_rcvry.agpmtdet AS B ON a.ahid = b.apahid
        LEFT JOIN
            bv-pdbd-prd.trusted_rcvry.tb_pagto_boleto AS C
              ON CAST(b.apahid AS INT64) = CAST(c.pbnumacordo AS INT64)
             AND CAST(b.apdetid AS INT64) = CAST(c.pbnumparc AS INT64)
        WHERE
            CAST(a.ahprd AS INT64) > 0
            AND DATE(a.ahdt) <= LAST_DAY(DATE('2026-09-01'))
        GROUP BY
            a.ahid,
            a.ahdt,
            a.ahprd,
            a.ahtype,
            a.ahdtqb
    ) AS P1
) AS P2
JOIN
    bv-pdbd-prd.trusted_rcvry.agrhdr AS AC ON ac.ahid = P2.NuAcordo
JOIN
    bv-pdbd-prd.trusted_rcvry.agency AS AG ON ac.ahcollid = ag.agagency
LEFT JOIN
    bv-pdbd-prd.trusted_rcvry.agpmtdet AS PC
      ON pc.apahid = P2.NuAcordo
     AND DATE(pc.apduedt) BETWEEN P2.DtInicio AND P2.DtFim
LEFT JOIN
    bv-pdbd-prd.trusted_rcvry.uda3 AS U3
      ON ac.ahacct = u3.u3acct
LEFT JOIN
    bv-pdbd-prd.trusted_rcvry.tb_pagto_boleto AS PB
      ON CAST(pc.apahid AS INT64) = CAST(pb.pbnumacordo AS INT64)
     AND CAST(pc.apdetid AS INT64) = CAST(pb.pbnumparc AS INT64)
WHERE
    P2.QdPMTPgto > 0
    AND DATE(P2.DtClicloCriacao) <= P2.DtFim
    AND (DATE(P2.DtQuebra) >= DATE('2026-09-01') OR P2.DtQuebra IS NULL)
    AND (DATE(P2.DtQuitacao) >= DATE('2026-09-01') OR P2.DtQuitacao IS NULL)
ORDER BY
    NuAcordo,
    CdAgente,
    NuParcela
)

SELECT * FROM QUERY_1
) ;

DISCONNECT FROM PERM;

PROC SQL ;
CREATE TABLE TbMISAcordo AS
SELECT
 *
,CASE
 WHEN STPARCELA = 'C'
  AND DTPGTO < DTINICIO AND DTCICLOPGTO < DTINICIO	THEN 0 
 WHEN NUPARCELA = 1									THEN 0 
 WHEN DTVENCIMENTO = .	 							THEN 0 
 ELSE 1	END 					AS R_CxMes

FROM TbMISAcordo;

;QUIT;


PROC SQL ;
CREATE TABLE TbMISAcordo AS
SELECT
 *
,CASE
 WHEN PERFIL = 'COLCHAO'					THEN 1 ELSE 0 END 	AS FlEInicio

,CASE
 WHEN R_CxMes = 1
  AND DTCICLOPGTO IS NOT NULL
  AND DTCICLOPGTO <= DtLimite				THEN 1 ELSE 0 END 	AS FlPago

,CASE
 WHEN CALCULATED FlPago = 1 AND PLANO = NUPARCELA	THEN 1
 WHEN DTQUITACAO BETWEEN DTINICIO AND DTFIM			THEN 1
 ELSE 0 END 													AS FlQuitacao

,CASE
 WHEN DTQUEBRA BETWEEN DTINICIO AND DTFIM	THEN 1 ELSE 0 END 	AS FlQuebra

,CASE
 WHEN CALCULATED FlQuebra = 1
  AND DTPROMESSA IS NOT NULL
  AND DTPROMESSA  = DTQUEBRA 				THEN 1 ELSE 0 END 	AS FlPromessa

FROM TbMISAcordo

ORDER BY DTCRIACAO , NUACORDO, NUPARCELA
;QUIT;RUN;


PROC SQL;

,CDAGENTE

PROC SQL ;
CREATE TABLE TbMISAcordo AS
SELECT
 A.*
,B.QdPMTMes
,CASE
 WHEN A.DTVENCIMENTO = B.DTVENCIMENTO 	THEN 1 ELSE 0 END 	AS QdAcordo
,CALCULATED QdAcordo * A.FlEInicio							AS QdEstoque
,CASE
 WHEN A.FlEInicio = 0 THEN CALCULATED QdAcordo ELSE 0 END	AS QdFluxo
,CALCULATED QdAcordo * B.QdQuitacao							AS QdQuitacao
,CALCULATED QdAcordo * B.QdQuebra							AS QdQuebra
,CALCULATED QdAcordo * A.VrAcordo							AS VrCarteira
,A.R_CxMes * A.VRPARCELA 									AS VrCaixa
,CASE
 WHEN A.R_CxMes = 1 THEN DAY(A.DTVENCIMENTO) END 			AS DiaEntradaCx

FROM TbMISAcordo   	AS A
LEFT JOIN TbResumo	AS B
  ON A.DtInicio 	= B.DtInicio
 AND A.NUACORDO		= B.NUACORDO
 AND A.CDAGENTE		= B.CDAGENTE
;QUIT;RUN;

PROC SQL;
CREATE TABLE TbData AS
SELECT
 DU.Safra
,DU.Data
,DU.Du
,MAX(COALESCE(A.DTCLICLOCRIACAO,DU1.Data))	AS DataDU			FORMAT DDMMYY10.

FROM B_GERAL.DIAS_UTEIS 		AS DU

  on DU.Du    = A.Du

  ON DU.Du    = DU1.Du
 AND DU.Safra = DU1.Safra

where DU.Safra = &AnoMesCarga.
GROUP BY DU.Safra,DU.Data ,DU.Du

having calculated DataDU <= &DtMxProces.
;QUIT;RUN;
