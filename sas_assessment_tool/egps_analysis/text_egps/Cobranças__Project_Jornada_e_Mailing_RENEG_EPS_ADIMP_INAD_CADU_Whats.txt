PROC SQL;
CREATE TABLE WORK.BASE_GERAL_RENEGINDAD_ AS
SELECT
    A.*
, CASE WHEN A.NUCONTRATO = SMS.NUCONTRATO 		THEN 1 ELSE 0 END AS FLAG_SMS
, CASE WHEN A.NUCONTRATO = FLASH.NUCONTRATO		THEN 1 ELSE 0 END AS FLAG_FLASH
, CASE WHEN A.NUCONTRATO = EMAIL.NUCONTRATO 	THEN 1 ELSE 0 END AS FLAG_EMAIL

, CASE
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.mil.br%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.jus.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.gov.%")   	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.sj.%")    	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.tj.br%")  	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%.procon.%") 	THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') AND A.EMAIL LIKE ("%;%") 		THEN 0
	WHEN A.EMAIL LIKE ('%@%') AND A.EMAIL LIKE ('%.%') 									THEN 1
	ELSE 0
	END AS FLAG_EMAIL_VALIDO
, CASE WHEN A.NUCONTRATO = WHATS.NUCONTRATO 	THEN 1 ELSE 0 END AS FLAG_WHATS
, 0 AS FLAG_PUSH
FROM WORK.BASE_FINAL					 AS A
LEFT JOIN WORK.TEMP_SMS_RENEGINDAD_02    AS SMS   ON A.NUCONTRATO = SMS.NUCONTRATO
LEFT JOIN WORK.TEMP_FLASH_RENEGINDAD_02  AS FLASH ON A.NUCONTRATO = FLASH.NUCONTRATO
LEFT JOIN WORK.TEMP_EMAIL_RENEGINDAD_02  AS EMAIL ON A.NUCONTRATO = EMAIL.NUCONTRATO

LEFT JOIN WORK.TEMP_WHATS_RENEGINDAD_02  AS WHATS ON A.NUCONTRATO = WHATS.NUCONTRATO
;QUIT;

  DBMS=csv
  REPLACE;

 DELIMITER=';';
RUN;
