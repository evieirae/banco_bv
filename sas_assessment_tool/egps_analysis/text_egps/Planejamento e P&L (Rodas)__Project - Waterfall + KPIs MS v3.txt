PROC SQL;
   CREATE TABLE Gravames_Ano_ProdutoEGV_Estado AS
   SELECT
          FLOOR(t1.NuAnoMes/100) AS Ano,

          
          CASE
            WHEN UPCASE(STRIP(t1.FlCVG)) = 'S' AND t1.TpVeiculo IN ('Leves','Motos')   THEN 'EGV Leves'
            WHEN UPCASE(STRIP(t1.FlCVG)) = 'S' AND t1.TpVeiculo =  'Pesados'           THEN 'EGV Pesados'
            ELSE t1.TpVeiculo
          END AS Produto_Reclass,

          
          CASE WHEN UPCASE(STRIP(t1.StVeiculo)) IN ('NOVO','NOVOS') THEN 'Novos'
               ELSE 'Usados'
          END AS Estado_Bem,

          SUM(t1.GravTotal) AS GravTotal,
          SUM(t1.GravBV)    AS GravBV
     FROM BASES.TBRESULTADOMARKETSHARE t1
    WHERE t1.NuAnoMes BETWEEN 202001 AND 202612
      AND t1.TpArquivo = 'mf'
      
      AND t1.TpVeiculo IN ('Leves','Motos','Pesados')
    GROUP BY 1,2,3
    ORDER BY 1,2,3;

PROC SQL;
   CREATE TABLE TBPRODUCAO AS
   SELECT

          t1.nuanomes,
          case when t1.NmGrupoProdutoMeta = 'Leves-BVO' then 'Leves' else t1.NmGrupoProdutoMeta end as Produto,
          t1.DsTipoCanalProducao,
		  t1.NmSubEstruturaComercialC,
		  t1.NmCanalEstruturaComercial,
		  t1.NmAreaEstruturaComercialC,

          case when t1.fxidade = 'ZeroKM' then 'Novos' else 'Usados' end as Estado_bem,

		  case when t1.NmGrupoProdutoMeta in ('Leves', 'Leves-BVO',  'Motos', 'Pesados') then 'Rodas' when t1.NmGrupoProdutoMeta = 'Solar' then 'Solar' else 'Empréstimos' end as Grupo_Produto,

            (COUNT(t1.NuContrato)) AS QtdeContrato,
            (SUM(t1.VrProducao)) AS VrProducao,
            (SUM(t1.VrProducaoProjetado)) AS VrProducaoProjetado

      FROM bases.TBPRODUCAO t1

	  where t1.NuAnoMes BETWEEN 202601 AND 202602

	  and t1.NmGrupoProdutoMeta in ('Leves', 'Leves-BVO', 'Motos', 'Pesados', 'CVG Leves', 'CVG Pesados')
      GROUP BY
	  1,2,3,4,5,6,7,8;

QUIT;
