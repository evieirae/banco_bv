%Macro BVO(dt_AtuMin,dt_AtuMax,base);

	proc sql ;
	connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
	      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
	   Create table base_bvo_&base. as
	   select * from connection to perm
	     (
	 WITH
	  tb_simulacao_proposta AS (
	 SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_proposta`
	  WHERE TIMESTAMP_TRUNC(DT_SIMULACAO_PROPOSTA,MONTH) between &dt_AtuMin. and &dt_AtuMax. ),
	  tb_simulacao_proposta_garantia AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_proposta_garantia`
	  WHERE TIMESTAMP_TRUNC(DT_SIMULACAO_GARANTIA,MONTH) between &dt_AtuMin. and &dt_AtuMax. ),
	  tb_simulacao_dado_fnnco AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_dado_fnnco`
	  WHERE TIMESTAMP_TRUNC(dt_simulacao_financiamento,MONTH) between &dt_AtuMin. and &dt_AtuMax. ),
	  tb_simulacao_prazo_fnnco AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_prazo_fnnco`
	  WHERE TIMESTAMP_TRUNC(dt_simulacao_prazo_fnnco,MONTH) between &dt_AtuMin. and &dt_AtuMax. ),
	  tb_simulacao_przo_fnnco_prpsa AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_przo_fnnco_prpsa`
	  WHERE DATE_TRUNC(dt_inclusao,MONTH) between DATE(&dt_AtuMin.) and DATE(&dt_AtuMax.) ),
	  tb_simulacao_prazo_faixa_aprvo AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_simulacao_prazo_faixa_aprvo`
	  WHERE TIMESTAMP_TRUNC(dt_inclusao,MONTH) between &dt_AtuMin. and &dt_AtuMax. ),
	  tb_proposta_parceria  AS (
	  SELECT * FROM `bv-pdbd-prd.trusted_apro_simu.tb_proposta_parceria`
	  WHERE (dt_inclusao) between &dt_AtuMin. and &dt_AtuMax. ),

	  tb_apro_simu AS (
	  SELECT distinct
	    cast(a.cd_simulacao_proposta as int64) as cd_simulacao_proposta
	    ,date(a.dt_simulacao_proposta) as dt_simulacao_proposta
	    ,cast(e.nu_proposta as int64) as nu_proposta
	    ,cast(g.cd_parceiro_externo as int64) as cd_parceiro_externo
		,f.CD_SIMULACAO_PRAZO_FNNCO                     AS Simu_Parc
		,c.cd_simulacao_dado_fnnco                      AS cd_simu_dado
		,f.DS_MOTIVO_DECISAO_PDEC                       AS motivo_decisao
		,a.st_pre_analise_bvo                        	AS Decisao_preAnalise_Bvo
		,f.PC_PROBABILIDADE_APROVACAO                   AS ProbabilidadeAprovacao
		,g.cd_parceiro_externo 							AS cd_parceiro_bb
		,e.NU_PROPOSTA 									as nuProposta

	  FROM tb_simulacao_proposta a
	LEFT JOIN tb_simulacao_proposta_garantia b
	 ON b.cd_simulacao_proposta = a.cd_simulacao_proposta
	LEFT JOIN tb_simulacao_dado_fnnco c
	ON c.cd_simulacao_dado_fnnco = b.cd_simulacao_dado_fnnco
	LEFT JOIN tb_simulacao_prazo_fnnco d
	ON d.cd_simulacao_dado_fnnco = c.cd_simulacao_dado_fnnco
	LEFT JOIN tb_simulacao_przo_fnnco_prpsa e
	ON e.cd_simulacao_prazo_fnnco = d.cd_simulacao_prazo_fnnco
	LEFT JOIN tb_simulacao_prazo_faixa_aprvo f
	ON d.cd_simulacao_prazo_fnnco = f.cd_simulacao_prazo_fnnco
	LEFT JOIN tb_proposta_parceria g
	ON g.cd_simulacao_proposta = a.cd_simulacao_proposta
	 WHERE e.nu_proposta is not null)

	SELECT * FROM tb_apro_simu

	);
	   disconnect from perm;
	quit;

%Mend BVO;

/* Mantendo apenas as propostas com cd_parceiro_externo = 41 e suas respectivas simula  es */
proc sql;
	create table aprosimu_bvo_bb_1 as
	select t1.*, t2.cd_parceiro_externo as cd_parceiro_externo_rm
	from base_bvo_full_nd  t1 inner join bvo_prop_41_nd t2
						  on t1.nu_proposta = t2.nu_proposta;

PROC SORT DATA=aprosimu_bvo_bb_1
	OUT=aprosimu_bvo_bb_1_1;
	BY DESCENDING cd_simulacao_proposta DESCENDING cd_simu_dado Simu_Parc;
RUN;

data work.aprosimu_bvo_bb_1_2;
	set work.aprosimu_bvo_bb_1_1;
	BY DESCENDING cd_simulacao_proposta DESCENDING cd_simu_dado Simu_Parc;
	if first.cd_simulacao_proposta then
		n=1;
	else n+1;
run;

proc sql;
create table work.aprosimu_bvo_bb_1_2_1 as
	select
		cd_simulacao_proposta,
		avg(input(ProbabilidadeAprovacao,3.)) format commax12.2 as PcProbAproAjustada
	from aprosimu_bvo_bb_1_2
group by cd_simulacao_proposta;quit;

proc sql;
create table aprosimu_bvo_bb_1_3 as
Select *
from aprosimu_bvo_bb_1_2
where n = 1;quit;

proc sql;

create table aprosimu_bvo_bb_2 as

	Select
		a.cd_simulacao_proposta
		,a.Simu_Parc
		,a.cd_simu_dado
		,a.Decisao_preAnalise_Bvo
		,a.motivo_decisao
		,a.ProbabilidadeAprovacao
		,a.nu_proposta
		,a.cd_parceiro_externo_rm as cd_parceiro_bb
		,a.nuProposta
		,a.n
		,b.PcProbAproAjustada
		,case when b.PcProbAproAjustada > 0 and a.motivo_decisao = 'OK' then 1 else 0 end as AprovacaoAjustada
	from aprosimu_bvo_bb_1_3 a
		left join work.aprosimu_bvo_bb_1_2_1 b
			on a.cd_simulacao_proposta = b.cd_simulacao_proposta

;quit;
