proc sql;
    create table base_solar as
    select distinct
        a.*,
        b.TxRebate,
        case
            when b.TABELA is not null then b.TxRebate * a.VrProducao
            else a.VpTxGerencialRebate
        end as VpTxNet
    from HOML.tbproducao as a
    left join tuyra.solar_rebate_&Data_Rebate as b
        on (a.CdTabelaFinanciamento = b.TABELA and a.QtParcelas = b.PRAZO)
    left join HOML.tboriginacao as c
        on (a.NuProposta = c.NuProposta)
    left join AUXL.tbinfocupom as d
        on (d.NuProposta = a.NuProposta)
    where a.nmgrupoprodutometa in ('Solar')
      and a.NuAnoMes >= &Data_Ref
    order by a.NuAnoMes;
quit;
