CREATE TABLE EFEITO_CANCELADAS AS
SELECT
NuProposta,
SUM(CASE WHEN A.DtStatusProposta <= D0 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD0CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD0_AJUST,
SUM(CASE WHEN A.DtStatusProposta <= D1 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD1CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD1_AJUST,
SUM(CASE WHEN A.DtStatusProposta <= D2 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD2CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD2_AJUST,
SUM(CASE WHEN A.DtStatusProposta <= D3 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD3CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD3CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD2CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD2_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD1CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD1_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD0CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD0_AJUST

FROM QTPROPOSTAS AS A
LEFT JOIN TBD0 AS F ON A.DtEntradaProposta=F.DtTempo
LEFT JOIN TBD1 AS G ON A.DtEntradaProposta=G.DtTempo
LEFT JOIN TBD2 AS J ON A.DtEntradaProposta=J.DtTempo
LEFT JOIN TBD3 AS H ON A.DtEntradaProposta=H.DtTempo

GROUP BY 1;
QUIT;
