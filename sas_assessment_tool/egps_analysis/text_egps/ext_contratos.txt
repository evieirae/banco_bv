/* ================== Gera base de rating usando a tbsimulacoes ========== */
/* ======================================================================= */
PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TBSIMURATING AS SELECT * FROM connection to dbcon (
  SELECT Distinct
    b.nuproposta,
    CAST((a.PcProbabilidadeAprovacao*10) AS FLOAT) as PcProbabilidadeAprovacao
   FROM STGORA_APRO_SIMU.TbSimulacaoPrazoFnnco a
  LEFT JOIN STGORA_APRO_SIMU.TbSimulacaoPrzoFnncoPrpsa b
  ON a.CdSimulacaoPrazoFnnco = b.CdSimulacaoPrazoFnnco
  WHERE 1=1
  AND a.DtSimulacaoPrazoFnnco >= &iqA0.
  AND b.nuproposta IS NOT NULL
);
RUN;

DATA TBSIMURATING_1;
	SET TBSIMURATING;
	FORMAT PcProbabilidadeAprovacao COMMAX12.2;RUN;
