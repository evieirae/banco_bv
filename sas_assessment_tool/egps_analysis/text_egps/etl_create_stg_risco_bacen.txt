/*---- Define load data macro  ----*/

/* --------------------------------------------------------------
   Load Technique Selection: Replace - EntireTable
   Constraint and index action selections: 'ASIS','ASIS','ASIS','ASIS'
   Additional options selections...

   -------------------------------------------------------------- */
%macro etls_loader;

   %let etls_tableOptions = ;

   /*---- Map the columns  ----*/
   proc datasets lib = work nolist nowarn memtype = (data view);
      delete WC50PDPN;
   quit;

   data _null_;
      put "NOTE: The following column(s) do not have a column mapping, so the"
           " value(s) will be set to missing: NuControleProtocolo";
   run;

   %put %str(NOTE: Mapping columns ...);
   proc sql;
      create view work.WC50PDPN as
         select
            DtBaseDocumento,
            StControleProtocolo,
            DtEnvioArquivo,
            DtRecebimentoRespostaBacen,
            DtGeracaoArquivo,
            QtClienteValido,
            QtClienteLote,
            TpArquivo,
            PcDocumentoProcessado,
            PcVolumeProcessado,
            CdControleProtocolo,
            QtOperacaoCliente,
            QtInstuicaoFinanceiraCliente,
            QtOperacaoSubJudice,
            VrReponsabilidadeTotalSubJudce,

            DtConsultaDocumento,
            CdRaizCpfCnpCliente,
            CdInformacaoConsolidadaCliente,
            FlMoedaEstrangeira,
            CdControleOperacaoCliente,
            CdSubProdutoBacen,
            CdProdutoBacen,
            VlVencimentoOperacaocliente,
            CdVencimentoCredito,
            NmVencimentoCredito,
            NmSubProdutoBacen,
            "" as NuControleProtocolo length = 15
               format = $15.
               informat = $15.
      from &etls_lastTable
      ;
   quit;

   %let SYSLAST = work.WC50PDPN;

   %let etls_lastTable = &SYSLAST;
   %let etls_tableOptions = ;

   /* Determine if the target table exists  */
   %let etls_tableExist = %eval(%sysfunc(exist(PEPRSTG.STG_BCO_RISCO_BACEN, DATA)) or
         %sysfunc(exist(PEPRSTG.STG_BCO_RISCO_BACEN, VIEW)));

   %if &etls_tableExist %then
   %do;/* table exists  */
      %let etls_hasPreExistingConstraint=0;

      proc datasets lib = work nolist nowarn memtype = (data view);
         delete etls_commands etls_commands_F;
      quit;

      %let etls_otherTablesReferToThisTable=0;

      %macro etls_CIContents(table=,workTableOut=,inDSOptions=);
         %put NOTE: Building table listing Constraints and Indexes for: &table;
         proc datasets lib=work nolist; delete &workTableOut; quit;
         proc contents data=&table&inDSOptions out2=&workTableOut noprint; run;

         data &workTableOut;
            length name $60 type $20 icown idxUnique idxNoMiss $3 recreate $600;
            set &workTableOut;
            type=upcase(type);
            if type eq 'REFERENTIAL' then
            do;
               put "WARNING%QUOTE(:) Target table is referenced by constraints in"
                    " another table: " ref;
               call symput('etls_otherTablesReferToThisTable','1');
               delete;
            end;
            if type='INDEX' and ICOwn eq 'YES' then delete;
         run;
         %rcSet(&syserr);

      %mend etls_CIContents;

      %etls_CIContents(table=PEPRSTG.STG_BCO_RISCO_BACEN, workTableOut=etls_commands, inDSOptions=);

      %if &etls_otherTablesReferToThisTable %then
         %put WARNING%QUOTE(:) Replacing entire table will fail. Consider an alternate load technique or revising constraints.;
      %else
      %do; /* okay - remove foreign keys  */

         data etls_commands_F;
            set etls_commands;
            if upcase(type)="FOREIGN KEY" then
            do;
               command='ic delete '||trim(name)||';';
               output etls_commands_F;
            end;
         run;

         %put %str(NOTE: Removing foreign keys before dropping table...);
         data _null_;
            set etls_commands_F;
            %rcSet(&syserr);

      %end; /* okay - remove foreign keys  */

      proc datasets lib = work nolist nowarn memtype = (data view);
         delete etls_commands etls_commands_F;
      quit;

      /*---- Drop a table  ----*/
      %put %str(NOTE: Dropping table ...);
      proc datasets lib = PEPRSTG nolist nowarn memtype = (data view);
         delete STG_BCO_RISCO_BACEN;
      quit;

      %rcSet(&syserr);

      %let etls_tableExist = 0;

   %end; /* table exists  */

   /*---- Create a new table  ----*/
   %if (&etls_tableExist eq 0) %then
   %do;  /* if table does not exist  */

      %put %str(NOTE: Creating table ...);

      data PEPRSTG.STG_BCO_RISCO_BACEN
              (label = 'Informações do Bacen');
         attrib DtBaseDocumento length = 8
            format = DATETIME22.3
            informat = DATETIME22.3
            label = 'DtBaseDocumento';
         attrib StControleProtocolo length = $1
            format = $1.
            informat = $1.
            label = 'StControleProtocolo';
         attrib DtEnvioArquivo length = 8
            format = DATETIME22.3
            informat = DATETIME22.3
            label = 'DtEnvioArquivo';
         attrib DtRecebimentoRespostaBacen length = 8
            format = DATETIME22.3
            informat = DATETIME22.3
            label = 'DtRecebimentoRespostaBacen';
         attrib DtGeracaoArquivo length = 8
            format = DATETIME22.3
            informat = DATETIME22.3
            label = 'DtGeracaoArquivo';
         attrib QtClienteValido length = 8
            format = 11.
            informat = 11.
            label = 'QtClienteValido';
         attrib QtClienteLote length = 8
            format = 11.
            informat = 11.
            label = 'QtClienteLote';
         attrib TpArquivo length = $1
            format = $1.
            informat = $1.
            label = 'TpArquivo';
         attrib PcDocumentoProcessado length = 8
            format = 7.2
            informat = 7.2
            label = 'PcDocumentoProcessado';
         attrib PcVolumeProcessado length = 8
            format = 7.2
            informat = 7.2
            label = 'PcVolumeProcessado';
         attrib CdControleProtocolo length = 8
            format = 11.
            informat = 11.
            label = 'CdControleProtocolo';
         attrib QtOperacaoCliente length = 8
            format = 11.
            informat = 11.
            label = 'QtOperacaoCliente';
         attrib QtInstuicaoFinanceiraCliente length = 8
            format = 11.
            informat = 11.
            label = 'QtInstuicaoFinanceiraCliente';
         attrib QtOperacaoSubJudice length = 8
            format = 11.
            informat = 11.
            label = 'QtOperacaoSubJudice';
         attrib VrReponsabilidadeTotalSubJudce length = 8
            format = 21.2
            informat = 21.2
            label = 'VrReponsabilidadeTotalSubJudce';
         attrib TpConsultaDocumento length = $1
            format = $1.
            informat = $1.
            label = 'TpConsultaDocumento';
         attrib DtConsultaDocumento length = 8
            format = DATETIME22.3
            informat = DATETIME22.3
            label = 'DtConsultaDocumento';
         attrib CdRaizCpfCnpCliente length = $11
            format = $11.
            informat = $11.
            label = 'CdRaizCpfCnpCliente';
         attrib CdInformacaoConsolidadaCliente length = 8
            format = 11.
            informat = 11.
            label = 'CdInformacaoConsolidadaCliente';
         attrib FlMoedaEstrangeira length = $1
            format = $1.
            informat = $1.
            label = 'FlMoedaEstrangeira';
         attrib CdControleOperacaoCliente length = 8
            format = 11.
            informat = 11.
            label = 'CdControleOperacaoCliente';
         attrib CdSubProdutoBacen length = $2
            format = $2.
            informat = $2.
            label = 'CdSubProdutoBacen';
         attrib CdProdutoBacen length = $2
            format = $2.
            informat = $2.
            label = 'CdProdutoBacen';
         attrib VlVencimentoOperacaocliente length = 8
            format = 21.2
            informat = 21.2
            label = 'VlVencimentoOperacaocliente';
         attrib CdVencimentoCredito length = 8
            format = 11.
            informat = 11.
            label = 'CdVencimentoCredito';
         attrib NmVencimentoCredito length = $70
            format = $70.
            informat = $70.
            label = 'NmVencimentoCredito';
         attrib NmSubProdutoBacen length = $50
            format = $50.
            informat = $50.
            label = 'NmSubProdutoBacen';
         attrib NuControleProtocolo length = $15
            format = $15.
            informat = $15.
            label = 'NuControleProtocolo';
         call missing(of _all_);
         stop;
      run;

      %rcSet(&syserr);

   %end;  /* if table does not exist  */

   /*---- Append  ----*/
   %put %str(NOTE: Appending data ...);

   proc append base = PEPRSTG.STG_BCO_RISCO_BACEN
      data = &etls_lastTable (&etls_tableOptions)  force ;
    run;

   %rcSet(&syserr);

   proc datasets lib = work nolist nowarn memtype = (data view);
      delete WC50PDPN;
   quit;

%mend etls_loader;
%etls_loader;

%let etls_recCheckExist = 0;
%let etls_recnt = 0;
%macro etls_recordCheck;
   %let etls_recCheckExist = %eval(%sysfunc(exist(PEPRSTG.STG_BCO_RISCO_BACEN, DATA)) or
         %sysfunc(exist(PEPRSTG.STG_BCO_RISCO_BACEN, VIEW)));

   %if (&etls_recCheckExist) %then
   %do;
      %local etls_syntaxcheck;
      %let etls_syntaxcheck = %sysfunc(getoption(syntaxcheck));
      /* Turn off syntaxcheck option to perform following steps  */
      options nosyntaxcheck;

      proc contents data = PEPRSTG.STG_BCO_RISCO_BACEN out = work.etls_contents(keep = nobs) noprint;
      run;

      data _null_;
         set work.etls_contents (obs = 1);
         call symput("etls_recnt", left(put(nobs,32.)));
      run;

      proc datasets lib = work nolist nowarn memtype = (data view);
         delete etls_contents;
      quit;

      /* Reset syntaxcheck option to previous setting  */
      options &etls_syntaxcheck;
   %end;
%mend etls_recordCheck;
%etls_recordCheck;
