create table  Work.Compro_soma as
select sum(resultado_comercial_brl)/2 as soma_resultado_comercial_brl, 'DEBENTURES' as PRODUTO
from  Work.Debentures_Compromissadas;


Create table Work.lastros_debenture as
select PRODUTO, SUBSTR(PAPEL,1,6) as PAPEL, POSICAO, VL_ATUAL_FECH
from work.LASTROS_CRI_CRA_DEBENTURES
where PRODUTO in ("DEBENTURES")
and POSICAO in ("PRÓPRIA FINANCIADA");

Create table Work.lastros_debenture_estoque as
select d.*, l.papel, l.produto
from Work.Debentures_Estoque d
inner join Work.lastros_debenture as l
on l.papel = d.cdobjetonegociacao;


drop table Work.Debentures_Estoque;

Create table Work.deb_estoque_agru as
select Chave_Unica, cdCliente, sum(carteira_contabil_brl) as soma_carteira_contabil_brl
from Work.lastros_debenture_estoque;

Create table Work.deb_estoque_somado as
select ld.*, soma_carteira_contabil_brl, 0.00 as percentual
from Work.lastros_debenture_estoque as ld
inner join deb_estoque_agru as ea
on ea.Chave_Unica = ld.Chave_Unica
and ea.cdCliente = ld.cdCliente;

update deb_estoque_somado
set percentual = carteira_contabil_brl / soma_carteira_contabil_brl;

create table  Work.deb_soma_total as
select d.*, cos.soma_resultado_comercial_brl, 0.00 as novo_resultado_comercial
from deb_estoque_somado as d
inner join Work.Compro_soma as cos
on cos.produto = d.produto;

update Work.deb_soma_total
set novo_resultado_comercial = soma_resultado_comercial_brl * percentual;
