proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table ParametroGrupo as
   select * from connection to perm
		(
	SELECT

	b.NmGrupoComissao,
	a.CdParceiroComercial,

	case when c.CdParametroGrupoCategoriaVclo = 1 then 'Leves'
	when c.CdParametroGrupoCategoriaVclo = 2 then 'Motos'
	when c.CdParametroGrupoCategoriaVclo = 3 then 'Pesados'
	else 'Erro' end as Produto

FROM bv-pdbd-prd.trusted_precportalprecificacao.tbvinculogrupocomissaoparceiro a

LEFT JOIN bv-pdbd-prd.trusted_precportalprecificacao.tbgrupocomissaoprecificacao  b  on b.CdGrupoComissaoPrecificacao = a.CdGrupoComissaoPrecificacao

LEFT JOIN bv-pdbd-prd.trusted_precportalprecificacao.tbgrupocomissaoproduto c on b.CdGrupoComissaoPrecificacao  = c.CdGrupoComissaoPrecificacao

WHERE a.FlAtivo = 'S'
);
disconnect from perm;
QUIT;

data SOLITMP.ParametroGrupo;
   set ParametroGrupo;
run;
