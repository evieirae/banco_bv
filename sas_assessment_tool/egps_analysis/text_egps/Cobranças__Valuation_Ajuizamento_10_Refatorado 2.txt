	AS SELECT DISTINCT
	A.MesCompetencia,

	min(E.Dt_DESP) AS DtDESP_REMUNERACAO format DATE9.,

	min(G.Dt_DESP) AS DtDESP_OUTRAS format DATE9.

	FROM SIMULACAO_AJUIZAMENTO2 AS A
	LEFT JOIN DESPESAS_AJUIZAMENTO2 AS B
		ON A.NUCONTRATO = B.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = B.SqContratoFinanceiro
		AND "AJUIZAMENTO" = B.DESC_DESPESA
		AND B.MesCompetencia = a.MesCompetencia
		and B.PARCELA = a.Nuparcela

	LEFT JOIN DESPESAS_BNDU2 AS C
		ON A.NUCONTRATO = C.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = C.SqContratoFinanceiro
		AND "BNDU" = C.DESC_DESPESA
		AND C.MesCompetencia = a.MesCompetencia
		and C.PARCELA = a.Nuparcela
		

	LEFT JOIN DESPESAS_HONORARIO2 AS D
		ON A.NUCONTRATO = D.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = D.SqContratoFinanceiro
		AND "HONORARIO" = D.DESC_DESPESA
		AND D.MesCompetencia = a.MesCompetencia
		and D.PARCELA = a.Nuparcela

		ON A.NUCONTRATO = E.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = E.SqContratoFinanceiro
		AND "REMUNERACAO" = E.DESC_DESPESA
		AND E.MesCompetencia = a.MesCompetencia
		and e.PARCELA = a.Nuparcela
		

	LEFT JOIN DESPESAS_PROTESTO2 AS F
		ON A.NUCONTRATO = F.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = F.SqContratoFinanceiro
		AND "PROTESTO" = F.DESC_DESPESA
		AND F.MesCompetencia = a.MesCompetencia
		and F.PARCELA = a.Nuparcela

	LEFT JOIN DESPESAS_OUTRAS2 AS G
		ON A.NUCONTRATO = G.NUCONTRATOGESTAO
		AND A.SqContratoFinanceiro = G.SqContratoFinanceiro
		AND "OUTRAS" = G.DESC_DESPESA
		AND G.MesCompetencia = a.MesCompetencia
		and g.PARCELA = a.Nuparcela

	A.MesCompetencia,
	A.NUCONTRATO,
	A.SqContratoFinanceiro,
	A.Nuparcela
