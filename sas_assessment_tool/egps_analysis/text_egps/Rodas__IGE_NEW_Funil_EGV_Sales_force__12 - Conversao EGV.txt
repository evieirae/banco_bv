/*CONVERSÃO PROPOSTA*/
proc sql;
	create table CONV_DGT_1 as
		select
			COALESCE(leads.data,DGTProd.dt_proposta) FORMAT DDMMYY10. AS DATA,
			leads.NUCPF,
			leads.TEL,
			leads.NMGR,
			UPPER(scan(leads.NMGR, 1, ' ')) as priNMGR,
			leads.ORIGEM,
			leads.DETALHAMENTO,
			COALESCE(leads.CTA,'SEM ENTRADA NO CTA') AS CTA,

			leads.OportunitStage,

			DGTProd.CdSubProduto,
			DGTProd.DsSubProduto,
			DGTProd.CdParceiroComercial,
			DGTProd.Convenio,
			DGTProd.cpf_prod,
			DGTProd.nmoperador,
			DGTProd.prinmoperador,
			DGTProd.dt_proposta,
			DGTProd.dt_status_proposta,
			DGTProd.nuproposta,
			DGTProd.nucontratogestao as nucontrato,
			DGTProd.StProposta,
/*			DGTProd.FLAG_SEGUROS,*/
/*			DGTProd.QTDE_SEGUROS,*/
/*			DGTProd.VRSEGURO,*/
			DGTProd.SomaVrProposta,
			INTCK('day',leads.data,DGTProd.dt_proposta) as difdata,
			case
				when UPPER(scan(leads.NMGR, 1, ' ')) = DGTProd.prinmoperador
					then 'S'
					else 'N'
				end
			as Mesmo_Operador,
		case
			when INTCK('day',leads.data,DGTProd.dt_proposta) between 0 and 30 then 1
			else 0
		end
	as proposta_valida

			from PRIVADO.DGT_CTA_LEADS_STAGE leads
			right join (
				select
					A.CdParceiroComercial,
					A.NmParceiroComercial as Convenio,
					A.nucpfcnpj as cpf_prod,
					(A.DtEntradaProposta)  as dt_proposta,
					(A.DtStatusProposta)  as dt_status_proposta,
					'' as CdSubProduto,
					A.NmGrupoProdutoMeta as DsSubProduto,
					A.nuproposta,
					A.nucontrato as nucontratogestao,
					A.stproposta,
					A.VrProposta as SomaVrProposta,
					UPPER(A.NmColaboradorC) AS nmoperador,
					UPPER(scan(A.NmColaboradorC, 1, ' ')) as prinmoperador
/*					A.FLAG_SEGUROS,*/
/*					A.QTDE_SEGUROS,*/
/*					A.VRSEGURO */
					from TMP_DGT_PROPOSTAS A) DGTProd

				on  PUT(NUCPF,Z11.) = DGTProd.cpf_prod
				and	 leads.data <= DGTProd.dt_proposta
				and	month(leads.data) = month(DGTProd.dt_proposta) and
				year(leads.data) = year(DGTProd.dt_proposta)
/*WHERE INTCK('day',leads.data,DGTProd.dt_proposta) between 0 and 30*/
;
QUIT;

PROC SORT DATA=CONV_DGT_1
