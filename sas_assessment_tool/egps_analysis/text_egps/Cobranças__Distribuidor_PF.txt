PROC SQL;

CREATE TABLE BA_PRIVADO AS SELECT

	A.*,
	CASE WHEN C.CDLOJA IS NOT NULL THEN C.ATRASO ELSE 0 END AS ATRASOREPASSE_PJ,
	CASE WHEN MOTIVOS IS NOT NULL THEN 1 ELSE 0 END AS FLAG_FILTRO

FROM BA_PRIVADO A LEFT JOIN CARGAPF.FILTRO_PRIVADO_INSS B
ON UPCASE(A.DS_MOTIVO)=UPCASE(B.MOTIVOS)
LEFT JOIN ATRASO_REPASSE_PRIV C
ON A.CD_LOJA_WEB=C.CDLOJA;

QUIT;

PROC SQL;

CREATE TABLE BA_PUBLICO AS SELECT

	A.*,
	CASE WHEN C.CDLOJA IS NOT NULL THEN C.ATRASO ELSE 0 END AS ATRASOREPASSE_PJ,
	CASE WHEN MOTIVOS IS NOT NULL THEN 1 ELSE 0 END AS FLAG_FILTRO

FROM BA_PUBLICO A LEFT JOIN CARGAPF.FILTRO_PB B
ON UPCASE(A.DS_MOTIVO)=UPCASE(B.MOTIVOS)
LEFT JOIN ATRASO_REPASSE_PUB C
ON A.CD_LOJA_WEB=C.CDLOJA;
QUIT;

PROC SQL;

CREATE TABLE BA_INSS AS SELECT

	A.*,
	CASE WHEN MOTIVOS IS NOT NULL THEN 1 ELSE 0 END AS FLAG_FILTRO

FROM BA_INSS A LEFT JOIN CARGAPF.FILTRO_PRIVADO_INSS B
ON UPCASE(A.DS_MOTIVO)=UPCASE(B.MOTIVOS);
QUIT;

PROC SQL;
CONNECT TO sybaseiq AS dbcon
(host='pta-iqprd01' server=ptaiqprd db=PTAIQCONT user=&ID password=&senha);
CREATE TABLE BA_ATRASO  AS
SELECT DISTINCT
	A.*,
	C.QTDEPARCATRASO

from PF_PUB_PRIV a LEFT JOIN CONNECTION TO dbcon

(SELECT DISTINCT
	B.NUCONTRATO,
	MAX(B.SQCONTRATOFINANCEIRO) AS SQCONTRATOFINANCEIRO,
	SUM(CASE WHEN  B.STPARCELA IN ('A','E') AND B.DTVENCIMENTO < &PRI_DIA.
	AND DATEDIFF(DAY,B.DTVENCIMENTO,&PRI_DIA.) > 29  THEN 1 ELSE 0 END) AS QTDEPARCATRASO

FROM  CPC.TBPARCELA B
WHERE B.SQCONTRATOFINANCEIRO=(SELECT MAX(SQCONTRATOFINANCEIRO) FROM  CPC.TBPARCELA P
                              WHERE B.NUCONTRATO=P.NUCONTRATO AND P.STPARCELA IN ('A','E'))
GROUP BY B.NUCONTRATO) C

ON A.CONTRATO=C.NUCONTRATO;
QUIT;

DATA BA_ATRASO2;
SET BA_ATRASO;

	IF DESCR_PRODUTO='CONSIGNADO PRIVADO' THEN PRODUTO='PRIVADO'; ELSE  PRODUTO='PUBLICO';

	ATRASO_PF=(QTDEPARCATRASO-ATRASOREPASSE_PJ);
