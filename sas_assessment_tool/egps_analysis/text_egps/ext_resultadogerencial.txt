PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE ext_pta_ResultadoGerencial AS
SELECT * FROM connection to dbcon
(
	SELECT
		     ResultadoGerencial.NuContrato
			,ResultadoGerencial.SqContratoFinanceiro
			,ResultadoGerencial.VrFinanciado VrFinanciamento
			,ResultadoGerencial.VrAcordoComercial
			,ResultadoGerencial.VrLiberado
			,ResultadoGerencial.VrCustoVariavel
			,ResultadoGerencial.VrPerdaEstimada
			,ResultadoGerencial.TxTir
			,ResultadoGerencial.VrDurationVP
			,ResultadoGerencial.VrMargemBruta
			,ResultadoGerencial.VrMargemLiquida
			,ResultadoGerencial.VrMargemContribuicaoFinal
			,ResultadoGerencial.VrMargemContribuicao
			,((ResultadoGerencial.VrFinanciado * ResultadoGerencial.PcROE) * ResultadoGerencial.VrDurationVP) as VpRoe
			,ResultadoGerencial.PcROE
			,ResultadoGerencial.TsReplicacao
	FROM DBGERENCIAL.TbResultadoGerencial ResultadoGerencial

	WHERE ResultadoGerencial.TsReplicacao >= today()-45
)
;RUN;

PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE ext_pta_ResultadoGerencialImblro AS
SELECT * FROM connection to dbcon
(
	SELECT
			Contrato.NuContrato
			,Contrato.SqContratoFinanceiro
			,ResultadoGerencial.VrLiberado
			,ResultadoGerencial.VrFinanciado VrFinanciamento
			,ResultadoGerencial.VrCusto as VrCustoVariavel
			,ResultadoGerencial.VrPdd as VrPerdaEstimada
			,ResultadoGerencial.TxTIR
			,ResultadoGerencial.VrMargemBruta
			,ResultadoGerencial.VrMargemLiquida
			,((Proposta.PcMargemLiquida*Proposta.VrProposta)/100) AS VrMargemContribuicaoFinal
			,((Proposta.PcMargemLiquida*Proposta.VrProposta)/100) AS VrMargemContribuicao
	FROM DBGERENCIAL.TbContratoRefinanciamentImblro ResultadoGerencial

		LEFT JOIN EGI.TbContratoFinanceiro Contrato
			ON  ResultadoGerencial.NuContrato = Contrato.NuContrato

		LEFT JOIN DbImobiliario.TbProposta Proposta
			ON  ResultadoGerencial.NuProposta = Proposta.NuProposta

	WHERE Contrato.NuContrato > 0
		AND Contrato.DtContratoFinanceiro >= today()-180
		AND ResultadoGerencial.VrLiberado is not null
)
;RUN;

data ext_pta;
set
	ext_pta_ResultadoGerencial
	ext_pta_ResultadoGerencialImblro
;run;

proc sort data=ext_pta; by NuContrato SqContratoFinanceiro;run;

/*incluindo a data do contrato na base e gerando outra base apenas com contratos que tiveram ajustes antes do envio da comissão ao RH*/
Proc Sql;
Create table ext_pta_datas (Where = (Filtro = 'Ajustar')) as
	Select
			a.NuContrato
			,a.SqContratoFinanceiro
			,a.VrFinanciamento
			,a.VrAcordoComercial
			,a.VrLiberado
			,a.VrCustoVariavel
			,a.VrPerdaEstimada
			,a.TxTir
			,a.VrDurationVP
			,a.VrMargemBruta
			,a.VrMargemLiquida
			,a.VrMargemContribuicaoFinal
			,a.VrMargemContribuicao
			,a.VpRoe
			,a.PcROE
			,DatePart(a.TsReplicacao) as TsReplicacao Format ddmmyy10.
			,prod.DtContrato
			,Case
				When day(prod.DtContrato) < 16  Then INTNX('month', prod.DtContrato+31, 0, 'BEGINNING')
				When day(prod.DtContrato) >= 16 Then INTNX('month', prod.DtContrato+16, 0, 'BEGINNING')
			End as MesRef format ddmmyy10.
			,mdy(MONTH(Calculated MesRef),Day('17Jun2024'd), YEAR(Calculated MesRef)) as DtLimAjuste format ddmmyy10.
			,Case When DatePart(a.TsReplicacao) <= Calculated DtLimAjuste Then "Ajustar" Else "Não Ajustar" End as Filtro
From ext_pta as a
	Left join tmp.tbproducao as prod On A.NuContrato = Prod.NuContrato;
Quit;

proc sort data=ext_pta_datas;
	by NuContrato SqContratoFinanceiro;
run;
