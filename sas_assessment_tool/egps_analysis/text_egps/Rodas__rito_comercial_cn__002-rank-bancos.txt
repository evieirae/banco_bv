/*Busca bancos de mensageria positiva Visao PDV*/

proc sql;
create table BancosRankinPDV as
Select
 NuAnoMes
,CdParceiroComercial
,FxEntradaMinima
,FxIdade
,Banco
,Sum(Qtd_APNP_Conco_MesmaLoja) + Sum(Qtd_Sem_Prop_MesmaLoja)  as TotalConcorrencia
from ResumoSimulacoesRito as cb
where 1=1
and banco not in ('N/A','BV')
/*and CdParceiroComercial = 9695*/
/*and safra = 202407*/
group by 1,2,3,4,5
ORDER by 1,2,3,4 , 6 asc,5 desc
;quit;

data BancosRankin_TopPDV;
    set BancosRankinPDV (Where=(TotalConcorrencia > 0));
    by  NuAnoMes CdParceiroComercial FxEntradaMinima FxIdade;
    retain Top 0;
    if first.FxEntradaMinima then Top = 0;
    Top + 1;
	BancoQtd = strip(Banco)||'('||strip(TotalConcorrencia)||')'; /* Nova coluna concatenada */
	TopBanco = 'Banco'||strip(Top);
run;

/*Busca bancos de mensageria recusada Visao PDV*/
proc sql;
create table BancosRankinPDVRec as
Select
 NuAnoMes
,CdParceiroComercial
/*,FxEntradaMinima	*/
,FxIdade
,Banco
,Sum(Qtd_Sem_Prop_MesmaLojaRec)  as TotalConcorrencia
from ResumoSimulacoesRito as cb
where 1=1
and banco not in ('N/A','BV')
/*and CdParceiroComercial = 9695*/
/*and safra = 202407*/
group by 1,2,3,4
ORDER by 1,2,3 , 5 asc
;quit;

data BancosRankin_TopPDVRec;
    set BancosRankinPDVRec (Where=(TotalConcorrencia > 0));
    by  NuAnoMes CdParceiroComercial FxIdade ;
    retain Top 0;
    if first.FxIdade then Top = 0;
    Top + 1;
	BancoQtd = strip(Banco)||'('||strip(TotalConcorrencia)||')'; /* Nova coluna concatenada */
	TopBanco = 'Banco'||strip(Top);
run;

/*base simu com ranking de bancos*/
proc sql;
create table TbRitoPDV_A001 as
select
	 B.NuAnoMes
	,B.CdParceiroComercial
/*	,B.NmParceiroComercial*/
	,B.FxIdade
	,B.FxEntradaMinima
	,C.Banco  as Banco
	,D.Banco  as BancoRec

	,coalesce(sum(B.Qtd_APNP_Conco_MesmaLoja),0)  	as Qtd_APNP_Conco_MesmaLj
	,coalesce(sum(B.Qtd_Sem_Prop_MesmaLoja),0)  	as Qtd_Sem_Prop_MesmaLj
	,coalesce(sum(B.Qtd_Efetivadas),0)  			as Qtd_Efetivadas
/*	,coalesce(sum(B.Qtd_APNP),0)  					as Qtd_APNP		*/
/*	,coalesce(sum(B.Qtd_APNP_Conco),0)  			as Qtd_APNP_Conco	*/
/*	,coalesce(sum(B.Qtd_Aprovadas),0)  				as Qtd_Aprovadas	*/
/*	,coalesce(sum(B.Qtd_Originadas),0)  			as Qtd_Originadas	*/
/*	,coalesce(sum(B.Qtd_Sem_Prop),0)  				as Qtd_Sem_Prop*/
/*	,coalesce(sum(B.Qtd_Simulacoes),0)  			as Qtd_Simulacoes	*/
/*	,coalesce(sum(B.VrPropostaEfetivada),0)  		as VrPropostaEfetivada*/

	,coalesce(sum(B.Qtd_Sem_Prop_MesmaLojaREC),0)  	as Qtd_Sem_Prop_MesmaLjRec
	,coalesce(SUM(B.QtdSimuRG),0) AS QtdSimuRG

from  ResumoSimulacoesRito B
	LEFT JOIN BancosRankin_TopPDV C
		on B.CdParceiroComercial = C.CdParceiroComercial
		and B.NuAnoMes = C.NuAnoMes
		and B.FxEntradaMinima = c.FxEntradaMinima
		and B.FxIdade = c.FxIdade
		and B.Banco = c.Banco

	LEFT JOIN BancosRankin_TopPDVRec D
		on B.CdParceiroComercial = D.CdParceiroComercial
		and B.NuAnoMes = D.NuAnoMes
		and B.FxIdade = D.FxIdade
		and B.Banco = D.Banco
/*where b.cdparceirocomercial in (9695,365991,394451)*/
GROUP BY 1,2,3,4,5,6
;run;
