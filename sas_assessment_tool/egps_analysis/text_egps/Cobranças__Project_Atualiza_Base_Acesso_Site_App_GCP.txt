CREATE TABLE WORK.AUX_ACESSOS AS
SELECT DISTINCT A.*
FROM CONNECTION TO PERM (
SELECT DISTINCT
  a.cd_bv
, b.idpessoa
, b.idbv
, a.tp_plataforma
, Min(a.dt_evento) as Min_acesso
, MAX(a.dt_evento) as Max_acesso
, Count(distinct a.dt_evento) AS Qtd_Acesso
from `bv-pdbd-prd.dvry_clie_expdigital.navegacao_appweb_fato` as a
Left join `bv-pdbd-prd.dvry_clie_crm.tbpessoacadastro`   as b on a.cd_bv = b.idbv
where a.dt_particao >= CAST(CURRENT_DATE()-30 AS DATE)
AND a.cd_bv is not null
AND a.cd_bv not in ('','%%BVID%%','/','/?utm_source=salesforce')
AND A.tp_plataforma = 'APP'
group by a.cd_bv, a.tp_plataforma, b.idpessoa, b.idbv
) AS A
;DISCONNECT FROM PERM
;QUIT;

PROC SQL;
CREATE TABLE WORK.AUX_HIST_ACESSO_APP AS
SELECT DISTINCT
  'APP' AS CANAL
, Today() Format = yymmdd10. Informat = yymmdd10. AS Dt_Ref
, Datetime() Format = e8601dt19. Informat = e8601dt19. AS Dt_Extracao
, (Year(A.MAX_ACESSO)*100+Month(A.MAX_ACESSO)) Format = best6. Informat = best6. AS SAFRA
, A.MIN_ACESSO Format = yymmdd10. Informat = yymmdd10. As MIN_ACESSO
, A.MAX_ACESSO Format = yymmdd10. Informat = yymmdd10. As MAX_ACESSO
, A.QTD_ACESSO Format = best2. Informat = best2. AS Qtd_Acesso
, CASE WHEN A.IdPessoa IS MISSING THEN B.IDPESSOA ELSE A.IDPESSOA END AS IdPessoa
, Compress(A.cd_bv) Format = $36. Informat = $36. AS HashBV
FROM WORK.AUX_ACESSOS AS A
LEFT JOIN CONTATOS.AUX_BASE_HASH AS B ON COMPRESS(UPCASE(A.cd_bv)) = COMPRESS(UPCASE(B.HashBV))
WHERE A.cd_bv IS NOT MISSING
AND A.tp_plataforma = 'APP'
Having IdPessoa IS NOT MISSING
;QUIT;




PROC SQL;
CREATE INDEX IdPessoa ON WORK.AUX_HIST_ACESSO_APP(IdPessoa);
