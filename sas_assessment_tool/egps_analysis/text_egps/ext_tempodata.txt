%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table bases.TbTempoData as
select * from connection to dbcon
(

WITH

	 TbDimTempoData AS
	(
		SELECT
			 DtTempo
			,NuAnoMes
			,NuAnoMesDia
			,CASE WHEN TempoData.DtTempo = '20201231' THEN 'N'
			      WHEN TempoData.DtTempo IN ('20211224','20211231','20241231') THEN 'N'
				  WHEN TempoData.DtTempo IN ('20251120') THEN 'N' /*FERIADO NACIONAL - CONSCIENCIA NEGRA - GFM - 05/11/2025*/
			      ELSE TempoData.FlDiaUtil END AS FlDiaUtil
			,CASE
				  WHEN TempoData.NuAnoMes IN (202012,202112) THEN 21
				  WHEN TempoData.NuAnoMes IN (202412) THEN 20
				  WHEN TempoData.NuAnoMes IN (202511) THEN 19 /*FERIADO NACIONAL - CONSCIENCIA NEGRA - GFM - 05/11/2025*/
				  ELSE TempoData.QtTotalDiaUtilMes END AS QtTotalDiaUtilMes
			,NmDiaSemana
			,SgDiaSemana
			,CASE WHEN FlDiaUtil = 'S' THEN 1
                  WHEN NuDiaSemana = 7 AND FlFeriadoNacional = 'N' THEN 0.4
                  ELSE 0 END AS FlDiaUtilSab
		FROM DWBI_DIME.TbDimTempoData TempoData

	)

	,TbTotalDUMesSab AS
    (
        SELECT
             NuAnoMes,
             SUM(FlDiaUtilSab) AS QtTotalDiaUtilMes
        FROM TbDimTempoData
        GROUP BY NuAnoMes
    )

	,TbStData AS
	(
		SELECT
			 1 StData
			,max(CASE WHEN TempoData.FlDiaUtil = 'S' THEN TempoData.DtTempo END) AS DtTempo

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20150101'
		  AND TempoData.DtTempo < today()
		GROUP BY
			1
	)
	,TbStMes AS
	(
		SELECT
			 NuAnoMes
			,1 StMes
			,max(CASE WHEN TempoData.FlDiaUtil = 'S' THEN TempoData.DtTempo END) AS DtTempo

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20150101'
		  AND TempoData.DtTempo < today()
		GROUP BY
			1,
			NuAnoMes
	)
	,TbDiaUteisMaturacao AS
	(
		SELECT
			 DtTempo
			,sum(CASE WHEN TempoData.FlDiaUtil = 'S' THEN 1 END) over (ORDER BY CAST(DtTempo AS  INTEGER) desc) AS QtDiasUteisMaturacao

		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20150101'
		  AND TempoData.DtTempo < today()
	)

	,TbNuAnoMes AS

	(

		SELECT

			 TempoData.NuAnoMes
			,max(TempoData.QtTotalDiaUtilMes) AS QtTotalDiaUtilMes
			,sum(CASE WHEN TempoData.FlDiaUtil = 'S' THEN 1 ELSE 0 END) AS QtCorridosDiaUtilMes
		    ,sum(FlDiaUtilSab) AS QtCorridosDiaUtilMesSab
		FROM TbDimTempoData TempoData
		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20150101'
		  AND TempoData.DtTempo < today()
		GROUP BY
			 TempoData.NuAnoMes

	),TbCorridosDiaUtilDia AS
	(
		SELECT
			 DtTempo
			,sum(CASE WHEN FlDiaUtil = 'S' THEN 1 ELSE 0 END) over (partition BY NuAnoMes ORDER BY cast(DtTempo as integer)) AS QtCorridosDiaUtilDia
		FROM TbDimTempoData
		WHERE 1 = 1
		AND DtTempo >= '20150101'
		AND DtTempo < today()
	)

		SELECT

			 TempoData.NuAnoMes
			,TempoData.NuAnoMesDia
			,TempoData.DtTempo
			,TempoData.FlDiaUtil
			,TempoData.NmDiaSemana
			,TempoData.SgDiaSemana
			,coalesce(StData.StData,0) AS StData
			,coalesce(StMes.StMes,0) AS StMes

			,CASE WHEN coalesce(DiaUteisMaturacao.QtDiasUteisMaturacao,0) >= 8 THEN 1 ELSE 0 END AS FlDiasUteisMaturacao
			,DiaUteisMaturacao.QtDiasUteisMaturacao
			,NuANoMes.QtTotalDiaUtilMes
			,NuAnoMesSab.QtTotalDiaUtilMes AS QtTotalDiaUtilMesSab
			,DAY(TempoData.DtTempo) AS QtCorridosDiaMes
			,NuAnoMes.QtCorridosDiaUtilMes
			,NuAnoMes.QtCorridosDiaUtilMesSab
			,CorridosDiaUtilDia.QtCorridosDiaUtilDia
			,NuANoMes.QtTotalDiaUtilMes-CorridosDiaUtilDia.QtCorridosDiaUtilDia QtDiasUtilFinalMes
			,NuAnoMesSab.QtTotalDiaUtilMes - NuAnoMes.QtCorridosDiaUtilMesSab as QtDiasUtilFinalMesSab
		FROM TbDimTempoData TempoData

			LEFT JOIN TbStData StData
				ON  TempoData.DtTempo = StData.DtTempo

			LEFT JOIN TbStMes StMes
				ON  TempoData.DtTempo = StMes.DtTempo

			LEFT JOIN TbNuAnoMes NuAnoMes
				ON  TempoData.NuAnoMes = NuANoMes.NuAnoMes

			LEFT JOIN TbDiaUteisMaturacao DiaUteisMaturacao
				ON  TempoData.DtTempo = DiaUteisMaturacao.DtTempo

			LEFT JOIN TbCorridosDiaUtilDia CorridosDiaUtilDia
				ON  TempoData.DtTempo = CorridosDiaUtilDia.DtTempo

			LEFT JOIN TbTotalDUMesSab NuAnoMesSab
			    ON  TempoData.NuAnoMes = NuAnoMesSab.NuAnoMes

		WHERE 1 = 1
		  AND TempoData.DtTempo >= '20150101'
		  AND TempoData.DtTempo < today()

)
;run;

proc sql;create index DtTempo on bases.TbTempoData;run;
