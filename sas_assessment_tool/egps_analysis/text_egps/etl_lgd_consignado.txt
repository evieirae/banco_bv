PROC SQL NOPRINT;
	connect to SYBASEIQ (&conex_syb.);
    CREATE TABLE TbCadastral_&dtsafra. AS SELECT  * FROM CONNECTION TO SYBASEIQ
(

declare @DtCalculo date
SELECT @DtCalculo = &dtfim.

SELECT
	a.NuProposta,
	a.nucontratoGestao AS NuContrato,
	c.CdEstadoCivil,
	c.VrTotalReceitas,
	c.CdPessoaSobAnalise,
	c.DtNascimento,
	c.CdSexo
FROM dba.tbproposta AS a LEFT JOIN dba.tbpessoasobanalise AS b
								ON a.nuproposta = b.nuproposta
						 INNER JOIN dba.tbpessoafisicasobanalise AS c
								ON b.cdpessoasobanalise=c.cdpessoasobanalise
						 LEFT JOIN dba.TbSubProduto AS d
								ON a.CdSubProduto=d.CdSubProduto

WHERE a.StProposta in ('A', 'AP') AND
	  d.CdProduto IN (10,11) AND
	  a.DtEntradaProposta <=@DtCalculo

);
	QUIT;

PROC SORT DATA= TbCadastral_&dtsafra. NODUPKEY;
BY NuContrato;
Where NuContrato ne .;
RUN;

PROC SQL NOPRINT ;
	connect to SYBASEIQ (&conex_syb.);
CREATE TABLE tbAtrasoMaxPacVenc_PG_&dtsafra. AS SELECT * FROM connection to SYBASEIQ
(
DECLARE @DtCalculo date
SELECT @DtCalculo = &dtfim.

SELECT
	NuContrato,
	max(SqContratoFinanceiro) SqContratoFinanceiro,
	max(DtProcessamento ) AS DataProcessamento
INTO #x
FROM CPC.tbcontratoFinanceiro
WHERE DtProcessamento < @DtCalculo
GROUP BY NuContrato

SELECT
	a.NuContrato,
	AtrasoMaxParce_IQ = max(datediff(day,b.DtVencimento,b.DtUltimoRecebimento))
FROM #x AS a LEFT JOIN CPC.TbParcela AS b
				ON a.NuContrato = b.NuContrato AND a.SqContratoFinanceiro=b.SqContratoFinanceiro
WHERE b.DtVencimento <= @DtCalculo
AND b.DtUltimoRecebimento <= @DtCalculo
GROUP BY a.NuContrato
);
	QUIT;

/*Retira a Duplicidade*/
PROC SORT DATA=tbAtrasoMaxPacVenc_PG_&dtsafra. NODUPKEY;
BY NuContrato;
RUN;

PROC SQL NOPRINT;
	connect to SYBASEIQ (&conex_syb.);

CREATE TABLE tbQtdeAtrasoMaior30d12M_&dtsafra. AS SELECT * FROM CONNECTION TO SYBASEIQ
(

DECLARE @DtFim date
DECLARE @DtFim12 date
SELECT @DtFim = &dtfim.
SELECT @DtFim12 = &dtfim12.

SELECT
	a.Nucontrato,
	a.SqContratoFinanceiro,
	Atraso12 = case when (a.StParcela = 'Q'  AND
	a.dtUltimoRecebimento BETWEEN  convert(DATETIME,@DtFim12)  and convert(DATETIME,@DtFim) AND
	a.DtVencimento	BETWEEN convert(DATETIME,@DtFim12) AND convert(DATETIME,@DtFim)
	     		AND datediff(day,a.DtVencimento,a.DtUltimoRecebimento) >= 30)
				THEN 1 ELSE 0 END,
	Produto = Case when b.CdModalidadeProduto in (17)          		  THEN 'Privado'
				   when b.CdModalidadeProduto in (18,32,34,37,39)     THEN 'Publico'
			       when b.CdModalidadeProduto in (16,33,42,43)        THEN 'INSS'
	ELSE 'Sem Informacao'
		END
	INTO #ER
FROM cpc.tbparcela AS a INNER JOIN cpc.tbcontrato AS b ON a.nucontrato = b.nucontrato AND
		a.sqcontratoFinanceiro = b.sqcontratoFinanceiroAtivo

SELECT
	a.NuCpfCnpj,
	b.Produto,
	qtdeatrasomaior30d12m = sum(b.atraso12)
FROM cpc.TbContrato AS a INNER JOIN #ER AS b ON a.NuContrato = b.NuContrato AND
				a.SqContratoFinanceiroAtivo = b.SqContratoFinanceiro

GROUP BY a.NuCpfCnpj, b.Produto
);
	QUIT;
