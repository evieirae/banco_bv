PROC SQL;
  CREATE TABLE PRODUCAO_SEGUROS_01 AS
    SELECT
      DtAtualizacao AS DtBaseProducao FORMAT=mmddyy10.,
	  QtTotalDiaUtilMesSab,
	  QtCorridosDiaUtilMesSab,
      SUBSTR(PUT(NuAnoMes,6.),5,2)||"/01/"||SUBSTR(PUT(NuAnoMes,6.),1,4) AS Safra,
      NmGrupoProdutoMeta AS Produto,
      CASE WHEN NuAnoMes >= 202201 THEN
        CASE WHEN NmGrupoProdutoMeta IN ('Leves','Motos','Pesados') THEN CASE WHEN FxIdade = 'ZeroKM' THEN 'Novos' ELSE 'Usados' END END
      END AS StVeiculo,
	  CASE WHEN NuAnoMes >= 202201 THEN
	    CASE WHEN NmGrupoProdutoMeta = 'Leves' THEN FxIdade END
      END AS FxIdade,
      NmSubEstruturaComercialA AS Regional,
      CASE WHEN DsTipoCanalProducao = 'NEGOCIOS CENTRALIZADOS' AND NuAnoMes < 202201 THEN 'MULTIMARCAS' ELSE DsTipoCanalProducao END AS Canal,
      SUM(VrBVMais) AS VrSPFPrd FORMAT=num18.2,
      SUM(VrCap) AS VrCapPrd FORMAT=num18.2,
      SUM(VrGap + VrRCF + VrGarantiaMecanica + VrCasco) AS VrDemaisPrd FORMAT=num18.2,
	  SUM(VrCompleto) AS VrCompletoPrd FORMAT=num18.2,
	  SUM(VrPlacaSolar) AS VrPlacaSolarPrd FORMAT=num18.2,
      SUM(VrBVMaisProjetado) AS VrSPFPrdProjetado FORMAT=num18.2,
      SUM(VrCapProjetado) AS VrCapPrdProjetado FORMAT=num18.2,
      SUM(VrCompletoProjetado) AS VrCompletoPrdProjetado FORMAT=num18.2,
	  SUM(VrPlacaSolarProjetado) AS VrPlacaSolarPrdProjetado FORMAT=num18.2,
      SUM(CASE WHEN StData = 1 THEN VrBVMais ELSE 0 END) AS VrSPFPrdDia FORMAT=num18.2,
      SUM(CASE WHEN StData = 1 THEN VrCap ELSE 0 END) AS VrCapPrdDia FORMAT=num18.2,
      SUM(CASE WHEN StData = 1 THEN VrDemais ELSE 0 END) AS VrDemaisPrdDia FORMAT=num18.2,
	  SUM(CASE WHEN StData = 1 THEN VrDemaisSemCompletoMax ELSE 0 END) AS VrCompletoPrdDia FORMAT=num18.2,
	  SUM(CASE WHEN StData = 1 THEN VrPlacaSolar ELSE 0 END) AS VrPlacaSolarPrdDia FORMAT=num18.2,
      SUM(FlBVMais) AS QtSPFPrd,
      SUM(FlCap) AS QtCapPrd,
      SUM(FlGap + FlRCF + FlGarantiaMecanica + FlCasco) AS QtDemaisPrd,
      SUM(FlCompleto) AS QtCompletoPrd,
	  SUM(FlPlacaSolar) AS QtPlacaSolarPrd,
      SUM(CASE WHEN FlBVMais = 1
            OR FlCap = 1
            OR FlDemaisSemCompleto = 1
            OR FlCompleto = 1
			OR FlPlacaSolar = 1
          THEN 1 ELSE 0 END) AS QtContratosSeguros,
      SUM(FlBVMaisEleg) AS QtSPFPrdEleg,
      SUM(FlCapEleg) AS QtCapPrdEleg,
      SUM(FlDemaisEleg) AS QtDemaisPrdEleg,
      SUM(FlDemaisEleg) AS QtCompletoPrdEleg,
	  SUM(FlPlacaSolarEleg) AS QtPlacaSolarPrdEleg
    FROM TMP.TBPRODUCAO
    WHERE NuAnoMes >= &safra13M.
      AND NmGrupoProdutoMeta IN ('Leves','Motos','Pesados','CVG Leves','CVG Pesados','Solar','Leves-BVO')
/*	  AND NmGrupoProdutoMeta IN ('FGTS','Leves','Motos','Pesados','CVG Leves', 'CVG Pesados','CP','Privado','Solar','CIG','Leves-BVO')*/

	  /*AND DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS','MULTIMARCAS','CONCESSIONARIAS','NEGOCIOS CENTRALIZADOS')*/
    GROUP BY 1,2,3,4,5,6,7,8,9
    ORDER BY 1,2,3,4,5,6,7,8,9;
RUN;

PROC SQL;
  CREATE TABLE PRODUCAO_SEGUROS AS
    SELECT
      DtBaseProducao,
      Safra,
      Produto,
	  StVeiculo,
	  FxIdade,
      Regional,
      Canal,
      VrSPFPrd,
      VrCapPrd,
      VrDemaisPrd,
	  VrCompletoPrd,
	  VrPlacaSolarPrd,
      VrSPFPrdProjetado,
      VrCapPrdProjetado,
      (VrDemaisPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS VrDemaisPrdProjetado FORMAT=num18.2,
      VrCompletoPrdProjetado,
	  VrPlacaSolarPrdProjetado,
      VrSPFPrdDia,
      VrCapPrdDia,
      VrDemaisPrdDia,
	  VrCompletoPrdDia,
	  VrPlacaSolarPrdDia,
      QtSPFPrd,
      QtCapPrd,
      QtDemaisPrd,
      QtCompletoPrd,
	  QtPlacaSolarPrd,
	  (QtContratosSeguros / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtContratosSegurosProjetado FORMAT=num18.2,
      (QtSPFPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtSPFPrdProjetado FORMAT=num18.2,
      (QtCapPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtCapPrdProjetado FORMAT=num18.2,
      (QtDemaisPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtDemaisPrdProjetado FORMAT=num18.2,
      (QtCompletoPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtCompletoPrdProjetado FORMAT=num18.2,
	  (QtPlacaSolarPrd / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtPlacaSolarPrdProjetado FORMAT=num18.2,
      QtContratosSeguros,
      QtSPFPrdEleg,
      QtCapPrdEleg,
      QtDemaisPrdEleg,
      QtCompletoPrdEleg,
	  QtPlacaSolarPrdEleg,
	  (QtSPFPrdEleg / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtSPFPrdElegProjetado FORMAT=num18.2,
	  (QtCapPrdEleg / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtCapPrdElegProjetado FORMAT=num18.2,
	  (QtDemaisPrdEleg / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtDemaisPrdElegProjetado FORMAT=num18.2,
	  (QtCompletoPrdEleg / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtCompletoPrdElegProjetado FORMAT=num18.2,
	  (QtPlacaSolarPrdEleg / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS QtPlacaSolarPrdElegProjetado FORMAT=num18.2
    FROM PRODUCAO_SEGUROS_01;
RUN;

PROC SQL;
  CREATE TABLE SEGUROS_CANCELADOS AS
	SELECT
      SUBSTR(PUT(NuAnoMes,6.),5,2)||"/01/"||SUBSTR(PUT(NuAnoMes,6.),1,4) AS Safra,
      NmGrupoProdutoMeta AS Produto,
      CASE WHEN NuAnoMes >= 202201 THEN
        CASE WHEN NmGrupoProdutoMeta IN ('Leves','Motos','Pesados') THEN CASE WHEN FxIdade = 'ZeroKM' THEN 'Novos' ELSE 'Usados' END END
      END AS StVeiculo,
	  CASE WHEN NuAnoMes >= 202201 THEN
	    CASE WHEN NmGrupoProdutoMeta = 'Leves' THEN FxIdade END
      END AS FxIdade,
      NmSubEstruturaComercialA AS Regional,
      CASE WHEN DsTipoCanalProducao = 'NEGOCIOS CENTRALIZADOS' AND NuAnoMes < 202201 THEN 'MULTIMARCAS' ELSE DsTipoCanalProducao END AS Canal,
      SUM(VrBVMais) as VrSPFCanc,
	  SUM(VrCap) as VrCapCanc,
	  SUM(VrDemais) as VrDemaisCanc,
	  SUM(VrPlacaSolar) as VrPlacaSolarCanc,
	  SUM(FlBVMais) as QtSPFCanc,
	  SUM(FlCap) as QtCapCanc,
	  SUM(FlDemais) as QtDemaisCanc,
	  SUM(FlPlacaSolar) as QtPlacaSolarCanc
    FROM BASES.TBSEGUROSPRODUCAOCANCELADOS
    GROUP BY 1,2,3,4,5,6;
RUN;

PROC SQL;
  CREATE TABLE SEGUROS AS
    SELECT
	  A.*,
	  B.VrSPFCanc,
	  B.VrCapCanc,
	  B.VrDemaisCanc,
	  B.VrPlacaSolarCanc,
	  B.QtSPFCanc,
	  B.QtCapCanc,
	  B.QtDemaisCanc,
	  B.QtPlacaSolarCanc
	FROM PRODUCAO_SEGUROS A
	LEFT JOIN SEGUROS_CANCELADOS B
	  ON A.Safra = B.Safra
	 AND A.Produto = B.Produto
	 AND A.StVeiculo = B.StVeiculo
	 AND A.FxIdade = B.FxIdade
	 AND A.Regional = B.Regional
	 AND A.Canal = B.Canal;
RUN;

PROC SQL;
  CREATE TABLE PRODUCAO_SEGUROS3 AS
    SELECT
	  DtBaseProducao,
	  Safra,
	  Produto,
	  StVeiculo,
	  FxIdade,
	  Regional,
	  CASE WHEN Canal NOT IN ('CONCESSIONARIAS','LOJA CANAL NEGOCIOS','MULTIMARCAS', 'NEGOCIOS CENTRALIZADOS') THEN 'OUTROS' ELSE Canal END AS Canal,
	  CASE WHEN UPPER(Variavel) LIKE '%SPF%' THEN 'SPF'
	       WHEN UPPER(Variavel) LIKE '%CAP%' THEN 'CAP/Ap'
		   WHEN UPPER(Variavel) LIKE '%DEMAIS%' THEN 'Auto'
		   WHEN UPPER(Variavel) LIKE '%COMPLETO%' THEN 'Completo'
		   WHEN UPPER(Variavel) LIKE '%PLACASOLAR%' THEN 'PlacaSolar'
		   ELSE 'Não selecionado'
	  END AS TpSeguro,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%qt%' AND LOWER(Variavel) LIKE '%eleg%' AND LOWER(Variavel) LIKE '%prd%' AND LOWER(Variavel) NOT LIKE '%projetado%' THEN Valor1 END),0) FORMAT=num12. AS QtdeElegivel,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%qt%' AND LOWER(Variavel) LIKE '%eleg%' AND LOWER(Variavel) LIKE '%prd%' AND LOWER(Variavel) LIKE '%projetado%' THEN Valor1 END),0) FORMAT=num12. AS QtdeElegivelProjetado,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%qt%' AND LOWER(Variavel) NOT LIKE '%eleg%' AND LOWER(Variavel) NOT LIKE '%projetado%' AND LOWER(Variavel) LIKE '%prd%' THEN Valor1 END),0) FORMAT=num12. AS QtdeContratos,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%qt%' AND LOWER(Variavel) NOT LIKE '%eleg%' AND LOWER(Variavel) LIKE '%projetado%' AND LOWER(Variavel) LIKE '%prd%' THEN Valor1 END),0) FORMAT=num12. AS QtdeContratosProjetado,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%qt%' AND LOWER(Variavel) LIKE '%canc%' THEN Valor1 END),0) FORMAT=num12. AS QtSegurosCancelados,
      COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%vr%' AND LOWER(Variavel) NOT LIKE '%dia%' AND LOWER(Variavel) LIKE '%prd%' AND LOWER(Variavel) NOT LIKE '%projetado%' THEN Valor1 END),0) FORMAT=num12.2 AS VrPremio,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%vr%' AND LOWER(Variavel) LIKE '%projetado%' AND LOWER(Variavel) LIKE '%prd%' THEN Valor1 END),0) FORMAT=num12.2 AS VrPremioProjetado,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%vr%' AND LOWER(Variavel) LIKE '%canc%' THEN Valor1 END),0) FORMAT=num12.2 AS VrPremioCancelados,
	  COALESCE(SUM(CASE WHEN LOWER(Variavel) LIKE '%vr%' AND LOWER(Variavel) LIKE '%dia%' AND LOWER(Variavel) LIKE '%prd%' THEN Valor1 END),0) FORMAT=num12.2 AS VrPremioDia
	FROM PRODUCAO_SEGUROS2
	WHERE calculated TpSeguro <> 'Não selecionado'
	GROUP BY 1,2,3,4,5,6,7,8;
RUN;

PROC SQL;
CREATE TABLE SegurosPainelExecutivo001 AS
  SELECT
    INPUT(SUBSTR(Safra,7,4) || SUBSTR(Safra,1,2),6.) AS NuAnoMes,
    CASE WHEN Produto IN ('Leves','Motos','Pesados') THEN 'Rodas'
		 WHEN Produto IN ('Leves-BVO') THEN 'Leves-BVO'
         WHEN Produto IN ('CVG Pesados','CVG Leves') THEN 'EVG'
         WHEN Produto IN ('CP') THEN 'CP'
		 WHEN Produto IN ('Privado') THEN 'Privado'
         WHEN Produto = 'Solar' THEN 'Solar'
		 WHEN Produto = 'FGTS' THEN 'FGTS'
    END AS Produto,
    CASE WHEN TpSeguro = 'CAP/Ap' THEN 'AP'
         WHEN TpSeguro IN ('Completo','Auto') THEN 'Auto'
         WHEN TpSeguro = 'SPF' THEN 'SPF'
		 WHEN TpSeguro = 'PlacaSolar' THEN 'PlacaSolar'
    END AS TpSeguro,
    SUM(VrPremio) AS VrPremio,
	SUM(VrPremioProjetado) AS VrPremioProjetado
  FROM
    PRODUCAO_SEGUROS_FINAL
  WHERE
    (calculated Produto = 'Rodas' AND calculated TpSeguro IN ('AP','Auto','SPF'))
	OR (calculated Produto = 'Leves-BVO' AND calculated TpSeguro IN ('AP','Auto','SPF'))
	OR (calculated Produto = 'EVG' AND calculated TpSeguro IN ('AP','Auto','SPF'))
	OR (calculated Produto = 'Privado' AND calculated TpSeguro IN ('AP','SPF'))
	OR (calculated Produto = 'CP' AND calculated TpSeguro IN ('AP','SPF'))
	OR (calculated Produto = 'Solar' AND calculated TpSeguro IN ('AP','SPF','PlacaSolar'))
	OR (calculated Produto = 'FGTS' AND calculated TpSeguro IN ('AP','SPF'))
  GROUP BY
    1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbProducao001 AS
  SELECT
    NuAnoMes,
    CASE WHEN NmGrupoProdutoMeta IN ('Leves','Motos','Pesados') THEN 'Rodas'
		 WHEN NmGrupoProdutoMeta = 'Leves-BVO' THEN 'Leves-BVO'
         WHEN NmGrupoProdutoMeta IN ('CVG Pesados','CVG Leves') THEN 'EVG'
         WHEN NmGrupoProdutoMeta IN ('Privado') THEN 'Privado'
		 WHEN NmGrupoProdutoMeta IN ('CP') THEN 'CP'
         WHEN NmGrupoProdutoMeta = 'Solar' THEN 'Solar'
		 WHEN NmGrupoProdutoMeta = 'FGTS' THEN 'FGTS'
    END AS Produto,
    SUM(VrProducao) AS VrProducao
  FROM
    TMP.TbProducao
  WHERE
    NuAnoMes >= &safra13M.
	AND NmGrupoProdutoMeta IN ('Leves','Motos','Pesados','CVG Pesados','CVG Leves','Solar','Leves-BVO')
	/*	AND NmGrupoProdutoMeta IN ('FGTS','Leves','Motos','Pesados','CVG Pesados','CVG Leves','CP','Privado','Solar','Leves-BVO')*/
  GROUP BY
    1,2;
RUN;

	A.TpSeguro,
