PROC SQL;

    CREATE TABLE BASE_MONITORAMENTO_1 AS SELECT

        MAX(CdPessoa) AS COD_CLIENTE,
        SUBSTR(NuCpfCnpj,1,8) AS RAIZ_CNPJ,
        NuCpfCnpj AS CNPJ,
        MAX(NmPessoa) AS NOME_CLIENTE,
        MAX(CdGpEconomico) AS COD_GP_ECONOMICO,
        MAX(NmGpEconomico) AS NOME_GP_ECONOMICO,
        MAX(FlAtivoP) AS FLAG_ATIVO_CLIENTE,
        MAX(FlAtivoGP) AS FLAG_ATIVO_GRUPO,
		MAX(FlAtivoCIB) AS FLAG_ATIVO_CIB,
        MAX(FlSuspensoP) AS FLAG_SUSPENSO_CLIENTE,
        MAX(FlSuspensoGrupo) AS FLAG_SUSPENSO_GRUPO,
        MAX(DtVencimentoRiscoCreditoGLGrupo) AS DAT_VENC_GRUPO FORMAT=DDMMYY10.,
        MAX(DtVencLimiteCliente) AS DAT_VENC_CLIENTE FORMAT=DDMMYY10.,
        SUM(VrTotalAprovadoGLGrupo) AS VLR_LIMITE_GRUPO,
        SUM(VrTotalRiscoGLGrupo) AS VLR_RISCO_GRUPO,
		CASE WHEN MAX(DtVencimentoRiscoCreditoGLGrupo) > today() THEN
		SUM(VrTotalAprovadoGLGrupo) - SUM(VrTotalRiscoGLGrupo) ELSE 0 END AS VLR_DISPONIVEL_GRUPO,
        SUM(VrTotalAprovadoGLCliente) AS VLR_LIMITE_CLIENTE,
        SUM(VrTotalRiscoGLCliente) AS VLR_RISCO_CLIENTE,
        MAX(NmGerenteExecutivo) AS GEREX_CREDITO,
        MAX(NmAnalistaCredito) AS ANALISTA_CREDITO,
        MAX(DtClassificacao) AS DAT_CLASS_CIB FORMAT=DDMMYY10.,
        MAX(DsClassificacao) AS DES_CLASS_CIB,
        MAX(DtClassificacaoSME) AS DAT_CLASS_SME FORMAT=DDMMYY10.,
        MAX(DsClassificacaoSME) AS DES_CLASS_SME,
        MAX(VrFaturamentoGP) AS VLR_FATUR_GRUPO,
        MAX(FlLRGrupo) AS FLAG_LR_GRUPO,
        MAX(FlLRCliente) AS FLAG_LR_CLIENTE,
        CASE
            WHEN UPCASE(MAX(NmGpEconomico)) LIKE 'SACADO SME%' THEN 3
			WHEN UPCASE(MAX(NmGpEconomico)) LIKE 'PME%' THEN 4
            ELSE 2
        END AS PRIORIDADE,
		CASE WHEN CALCULATED PRIORIDADE = 2 AND MAX(FLATIVOGP) = 1 THEN 1 ELSE CALCULATED PRIORIDADE END AS ORDEM_PRIORIDADE

    FROM INSIGHT.TBBRUTA
    WHERE TpPessoa = 'J' AND TpLancamento = 'Pessoa'
    GROUP BY 3
    ORDER BY ORDEM_PRIORIDADE ASC, FLATIVOCIB DESC;
QUIT;

PROC SQL;

	CREATE TABLE BASE_ENTRADA_CALCRAT AS SELECT

	 CNPJ
	,DATA_BALANCO
	,VALOR_FATUR
	,VALOR_PL
	,(VALOR_EBITDA / MONTH(DATA_BALANCO))* 12 AS VALOR_EBITDA
	,VALOR_RESULT
    ,VALOR_PASS_CIRC
    ,VALOR_SAL
    ,VALOR_ADIANT
	,(VALOR_PASS_CIRC-VALOR_SAL-VALOR_ADIANT-VALOR_FORNE) AS VALOR_PASS_FIN
	,VALOR_DISP
	,VALOR_ATV_CIRC
	,VALOR_EMP_FIN
	,VALOR_REAL_LP
	,VALOR_ATV_PERM
	,CASE WHEN VALOR_APLIC_FINAN >= 0 THEN VALOR_APLIC_FINAN ELSE 0 END AS VALOR_APLIC_FINAN
	,CASE WHEN VALOR_TOT_INV >= 0 THEN VALOR_TOT_INV ELSE 0 END AS VALOR_TOT_INV
	,SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM) AS VALOR_ATV
	,(SUM(CALCULATED VALOR_PASS_FIN,VALOR_EMP_FIN) - MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN)) AS VALOR_DIVLIQ
	,(CASE WHEN VALOR_PL<=0 THEN 1 ELSE 0 END) AS FLAG_PL_NEG
	,(CASE WHEN CALCULATED VALOR_EBITDA<=0 THEN 1 ELSE 0 END) AS FLAG_EBTDA_NEG
	,(CASE WHEN VALOR_RESULT<=0 THEN 1 ELSE 0 END) AS PREJUIZO_EXERC
	,(CASE WHEN CALCULATED VALOR_PASS_FIN=0 THEN 10 ELSE VALOR_DISP/CALCULATED VALOR_PASS_FIN END) AS DISPONIVEL_ENDIVIDAMENTO
	,(CASE WHEN ((VALOR_ATV_CIRC - VALOR_PASS_CIRC)/SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM))=0 THEN 0.0001 ELSE ((VALOR_ATV_CIRC - VALOR_PASS_CIRC)/SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM)) END) AS CAPITAL_DE_GIRO_LIQUIDO
	,MIN(15,(CASE WHEN CALCULATED VALOR_EBITDA < 0 THEN
	                (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN) - CALCULATED VALOR_EBITDA
	                ELSE CALCULATED VALOR_DIVLIQ - CALCULATED VALOR_EBITDA END)
	          ELSE (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN (CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/CALCULATED VALOR_EBITDA
	                ELSE CALCULATED VALOR_DIVLIQ / CALCULATED VALOR_EBITDA END)
	          END)) AS DIVIDA_LIQUIDA_EBTDA
	,MIN(8,(CASE WHEN VALOR_PL < 0 THEN
	                (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN) - VALOR_PL
	                ELSE CALCULATED VALOR_DIVLIQ - VALOR_PL END)
	          ELSE (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN (CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/VALOR_PL
	                ELSE CALCULATED VALOR_DIVLIQ / VALOR_PL END)
	          END)) AS DIVIDA_LIQUIDA_PL
	,(CASE WHEN VALOR_PASS_CIRC=0 THEN 28 ELSE (MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/VALOR_PASS_CIRC END) AS LIQUIDEZ_IMEDIATA

	FROM INFO_BALANCO_SUM3;

%macro carregar_base_bhv;

    
    %let hoje = %sysfunc(today());
    %let safra_atual = %sysfunc(putn(&hoje, yymmn6.));
    %let safra_anterior = %sysfunc(putn(%sysfunc(intnx(month, &hoje, -1)), yymmn6.));

    %put >>> Tentando carregar safra: &safra_atual;

    
    %if %sysfunc(exist(SME.BASE_BHV_SF_&safra_atual)) %then %do;
        %let safrabhv = &safra_atual;
        %put >>> Safra encontrada: &safrabhv;
    %end;
    %else %do;
        %put >>> Safra atual não encontrada. Tentando safra anterior: &safra_anterior;
        %if %sysfunc(exist(SME.BASE_BHV_SF_202507)) %then %do;
            %let safrabhv = &safra_anterior;
            %put >>> Safra anterior encontrada: &safrabhv;
        %end;
        %else %do;
            %put >>> Nenhuma base encontrada para safra atual ou anterior. Macro encerrada.;
            %return;
        %end;
    %end;

    
    DATA BASE_BHV_SF;
        SET SME.BASE_BHV_SF_202507;

        WHERE FLG_TOT_DIA_ATR_M00 NE .;

        IF FLG_TOT_FIN_DES_12M = 0      THEN CLUSTER_BHV = 'Sem histórico    '; ELSE
        IF FLG_TOT_FIN_DES_12M = 1 AND QTD_TOT_DIA_ATR_M00 > 15 THEN CLUSTER_BHV = 'Em atraso        '; ELSE
        IF FLG_TOT_FIN_DES_12M = 1 AND QTD_TOT_DIA_ATR_M00 <= 15 AND QTD_MAX_DIA_ATR_12M = 0 AND QTD_CNT_QUI_PAG_12M = 0 THEN CLUSTER_BHV = 'Em construção    '; ELSE
        IF FLG_TOT_FIN_DES_12M = 1 AND QTD_TOT_DIA_ATR_M00 <= 15 AND QTD_MAX_DIA_ATR_12M = 0 AND QTD_CNT_QUI_PAG_12M > 0 THEN CLUSTER_BHV = 'Bom pagador      '; ELSE
        IF FLG_TOT_FIN_DES_12M = 1 AND QTD_TOT_DIA_ATR_M00 <= 15 AND QTD_MAX_DIA_ATR_12M > 0 AND QTD_TOT_MES_ATR_12M < 2 THEN CLUSTER_BHV = 'Atraso pontual   '; ELSE
        IF FLG_TOT_FIN_DES_12M = 1 AND QTD_TOT_DIA_ATR_M00 <= 15 AND QTD_MAX_DIA_ATR_12M > 0 AND QTD_TOT_MES_ATR_12M > 1 THEN CLUSTER_BHV = 'Atraso recorrente';
    RUN;

%mend carregar_base_bhv;

				ELSE

				   'SITUACAO DO CNPJ: NULO',

             WHEN DESCRICAO_HBV8 =
           'EMPRESA COM MENOS DE 1 ANO DE FUNDACAO' THEN 'Recusado'
	         WHEN FLAG_BLOQUEIO = 1 OR
                  FLAG_BLOQUEIO_CREDITO = 1 OR
		          FLAG_BLOQUEIO_GRUPO = 1           THEN 'Recusado'
		 	 WHEN FLAG_SUSPENSO_CLIENTE = 1 OR FLAG_SUSPENSO_GRUPO = 1 THEN 'Recusado'

			      PRINAD_FINAL >= 5                       THEN 'Recusado'

		     WHEN FLAG_SOCIO_SCORE = 1              THEN 'Recusado'

				WHEN LIMITE_GRUPO_CIB >= 4000000 AND LIMITE_TOTAL = 0 AND UPPER(TIPO) = 'INDÚSTRIA' THEN 'Recusado - Restrições de Crédito         '

	CREATE TABLE BASE_LIMITES_FINAL_1 AS SELECT

   		 *
		,CASE WHEN FLAG_REVENDAS = 1 THEN MIN(3000000,0.6*VLR_FATUR_FINAL)
			  WHEN UPPER(TIPO) = 'INDÚSTRIA' THEN MIN(4000000,0.3*VLR_FATUR_FINAL)
			  WHEN UPPER(TIPO) = 'AGRO'      THEN MIN(6000000,0.6*VLR_FATUR_FINAL)
		 ELSE 0 END AS LIMITE_GLOBAL

		,CASE

			  WHEN VISAO_SETOR_MATR_RESI = 'Fortemente Negativo' THEN 0.30
		      WHEN VISAO_SETOR_MATR_RESI = 'Levemente Negativo'  THEN 0.60
			  WHEN VISAO_SETOR_MATR_RESI = 'Neutro'              THEN 0.80
			  WHEN VISAO_SETOR_MATR_RESI = 'Levemente Positivo'  THEN 1.00
			  WHEN VISAO_SETOR_MATR_RESI = 'Fortemente Positivo' THEN 1.20
		                                                         ELSE 0.30 END AS FATOR_SETORIAL

		 ,CASE WHEN PRINAD_FINAL > 0  AND PRINAD_FINAL <= 1  THEN 1.00
			   WHEN PRINAD_FINAL > 1  AND PRINAD_FINAL <= 3  THEN 0.90
 			   WHEN PRINAD_FINAL > 3  AND PRINAD_FINAL <= 5  THEN 0.80
			   WHEN PRINAD_FINAL > 5  AND PRINAD_FINAL <= 8  THEN 0.60
			   WHEN PRINAD_FINAL > 8  AND PRINAD_FINAL <= 10 THEN 0.40
			   WHEN PRINAD_FINAL > 10 AND PRINAD_FINAL <= 12 THEN 0.20
			   WHEN PRINAD_FINAL > 12 AND PRINAD_FINAL <= 15 THEN 0.10
			                                                 ELSE 0.05 END AS FATOR_PRINAD

		,CASE WHEN ILM_MEDIO = . THEN 1.00
			  WHEN ILM_MEDIO > 0.8 THEN 1.00
			  ELSE 0.8 END AS FATOR_ILM

		,CASE WHEN FLAG_REVENDAS = 1 THEN 0.6 ELSE 1.00 END AS FATOR_REVENDAS

		 ,CASE WHEN UPPER(TIPO) = 'INDÚSTRIA'
               THEN MIN(4000000, CALCULATED LIMITE_GLOBAL * CALCULATED FATOR_SETORIAL * CALCULATED FATOR_PRINAD * CALCULATED FATOR_ILM * CALCULATED FATOR_REVENDAS)
		       WHEN UPPER(TIPO) = 'AGRO'
               THEN MIN(6000000, CALCULATED LIMITE_GLOBAL * CALCULATED FATOR_SETORIAL * CALCULATED FATOR_PRINAD * CALCULATED FATOR_ILM * CALCULATED FATOR_REVENDAS)
		  END AS LIMITE_GLOBAL_FATORES

		 ,CALCULATED LIMITE_GLOBAL_FATORES - LIMITE_TOTAL + LIMITE_PORTFOLIO AS LIMITE_DISPONIVEL_AUX

		 , CASE WHEN UPPER(TIPO) = 'INDÚSTRIA' AND FLAG_CADASTRO_CIB = 1 THEN 4000000 - LIMITE_GRUPO_CIB
		   WHEN UPPER(TIPO) = 'AGRO' AND FLAG_CADASTRO_CIB = 1 THEN 6000000 - LIMITE_GRUPO_CIB END AS LIMITE_DISP_GRUPOCIB

		 , CASE WHEN UPPER(TIPO) = 'INDÚSTRIA' AND FLAG_CADASTRO_CIB = 0 THEN 4000000 - LIMITE_GRUPOECON
		   WHEN UPPER(TIPO) = 'AGRO' AND FLAG_CADASTRO_CIB = 0 THEN 6000000 - LIMITE_GRUPOECON END AS LIMITE_DISP_GRUPOECON

		 , CASE WHEN LIMITE_GRUPO_CIB = 0 AND LIMITE_GRUPOECON = 0 THEN CALCULATED LIMITE_DISPONIVEL_AUX
		   WHEN FLAG_CADASTRO_CIB = 1 AND LIMITE_GRUPO_CIB > 0 THEN MIN(CALCULATED LIMITE_DISPONIVEL_AUX, CALCULATED LIMITE_DISP_GRUPOCIB)
		   WHEN FLAG_CADASTRO_CIB = 0 AND LIMITE_GRUPOECON > 0 THEN MIN(CALCULATED LIMITE_DISPONIVEL_AUX, CALCULATED LIMITE_DISP_GRUPOECON)
		   ELSE CALCULATED LIMITE_DISPONIVEL_AUX END AS LIMITE_DISPONIVEL

		 ,CASE WHEN FLAG_CADASTRO_CIB = 1 THEN MIN(CALCULATED LIMITE_DISP_GRUPOCIB, VLR_DISPONIVEL_GRUPO * 0.05)
			   WHEN LIMITE_SACADO > 0 THEN MIN(CALCULATED LIMITE_DISPONIVEL,LIMITE_SACADO)
		       WHEN (LIMITE_SACADO = 0 OR LIMITE_SACADO = .) AND FATUR_SACADO > 0 AND UPPER(TIPO) = 'AGRO'      THEN MIN(CALCULATED LIMITE_DISPONIVEL,0.6*FATUR_SACADO)
			   WHEN (LIMITE_SACADO = 0 OR LIMITE_SACADO = .) AND FATUR_SACADO > 0 AND UPPER(TIPO) = 'INDÚSTRIA' THEN MIN(CALCULATED LIMITE_DISPONIVEL,0.3*FATUR_SACADO)
			   ELSE MIN(2000000,CALCULATED LIMITE_DISPONIVEL,0.2*VLR_FATUR_FINAL) END AS LIMITE_POTENCIAL

		 ,CASE WHEN DECISAO = 'Elegível' THEN 100*INT(CALCULATED LIMITE_POTENCIAL/100) ELSE 0 END AS LIMITE_FINAL

	          WHEN PRINAD_FINAL >  0.90  AND PRINAD_FINAL <= 1.00   THEN 'BB'
	          WHEN PRINAD_FINAL >  1.00  AND PRINAD_FINAL <= 1.35   THEN 'BB-'

	          WHEN PRINAD_FINAL > 70.00  AND PRINAD_FINAL <= 100.00 THEN 'DF'
		      ELSE "N/A"
