proc sql;

CONNECT TO BIGQUERY AS PERM(project='bv-sadi-dtm' 
credfile='/projetos/acessos/bq/di/sadi-dtm.json' 
schema='bv-pdbd-prd' READ_MODE=storage);

CREATE TABLE CONTADOR AS SELECT *
FROM CONNECTION TO PERM (WITH HIST_GOE as (

         select distinct
            nm_assessoria,
            format_timestamp('%Y-%m-%d', A.dt_acionamento) as Dt,
            format_timestamp('%Y-%m-%d %H:%M:%S',
               timestamp(concat(cast(A.dt_acionamento as string), ' ', cast(A.hr_inicio_acionamento as string)))) as Dth,
            A.tp_resultado_acionamento as tp_resultado_acionamento,
            A.cd_cpf_cnpj,
            A.cd_contrato,
            A.ds_ocorrencia_chamada,
            substr(cast(A.cd_telefone_contato as string), 1, 2) as DDD,
            substr(cast(A.cd_telefone_contato as string), 3, 10) as Numero,
            case
               when substr(cast(A.cd_telefone_contato as string), 3, 1) = '9' then 'CELULAR'
               else 'FIXO'
            end as Tipo_Numero
         from `bv-pdbd-prd.dvry_corp_cobranca.assessoria_acionamentos_cobranca_fato` as A
         where cast(A.dt_acionamento as date) = '2025-01-02'
           AND A.tp_resultado_acionamento IS NOT NULL
           and cast(A.cd_assessoria as int64) = 220
           and A.cd_contrato is not null
      )

      select *,
         row_number() over (
         partition by cd_contrato, Dt, DDD, Numero order by Dth) as SPIN,
         case when tp_resultado_acionamento > '0' then 1 else 0 end as Flag_Registro,
         case when tp_resultado_acionamento >= '2' then 1 else 0 end as Flag_HitRate,
         case when tp_resultado_acionamento >= '3' then 1 else 0 end as Flag_Alo,
         case when tp_resultado_acionamento >= '4' then 1 else 0 end as Flag_CPC,
         case when tp_resultado_acionamento = '6' then 1 else 0 end as Flag_Promessa
      from HIST_GOE
      order by Dth asc);quit;
