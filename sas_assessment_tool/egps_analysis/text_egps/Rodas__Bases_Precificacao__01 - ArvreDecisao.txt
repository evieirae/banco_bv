proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table ArvreDecisao as
   select * from connection to perm
		(
		SELECT
		arv.nmarvoredecisao,
		grupo.nmgrupoparceiroprecificacao,
		arvparc.pctaxaajuste,
		arvparc.flativo,
		PGrupo.cdparceirocomercial,

		case when arvproduto.cdparametrogrupocategoriavclo = 1 then 'Leves'
		when arvproduto.cdparametrogrupocategoriavclo = 2 then 'Motos'
		when arvproduto.cdparametrogrupocategoriavclo = 3 then 'Pesados'
		else 'Erro' end as Produto

		from `bv-pdbd-prd.trusted_precportalprecificacao.tbgrupoparceiroarvoredecisao` arvparc

		left join `bv-pdbd-prd.trusted_precportalprecificacao.tbarvoredecisao` arv on arvparc.cdarvoredecisao = arv.cdarvoredecisao
		left join `bv-pdbd-prd.trusted_precportalprecificacao.tbgrupoparceiroprecificacao` grupo on arvparc.cdgrupoparceiroprecificacao=grupo.cdgrupoparceiroprecificacao
		left join `bv-pdbd-prd.trusted_precportalprecificacao.tbvinculoparceirogrupo` PGrupo on grupo.cdgrupoparceiroprecificacao = PGrupo.cdgrupoparceiroprecificacao
		left join `bv-pdbd-prd.trusted_precportalprecificacao.tbarvoredecisaoproduto` arvproduto on arvparc.cdarvoredecisao = arvproduto.cdarvoredecisao
		where arvparc.flativo = 'S'
		and arv.flativo  = 'S'
		AND grupo.flativo = 'S'
		and PGrupo.flativo = 'S'
);
disconnect from perm;
QUIT;

data SOLITMP.ArvreDecisao;
   set ArvreDecisao;
run;
