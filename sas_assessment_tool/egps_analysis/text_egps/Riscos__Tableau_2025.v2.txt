PROC SQL;
CREATE TABLE Garantias_GL_01 AS
	SELECT
		A.*,
		B.TIPO_GAR,
		B.PODER_CLASSIFICACAO AS PODER_MITIGACAO
	FROM MISCRED.TB_GL_OPERACOES_&ANOMES. AS A
	LEFT JOIN MISCRED.GARANTIAS_GL_VF AS B
		ON A.NmTipoGarantia = B.NmTipoGarantia
			AND A.NmNaturezaJuridica = B.NmNaturezaJuridica;
















			CREATE TABLE Garantias_GL_02 AS
	SELECT
		A.CdPessoaClienteRisco,
		A.DsClienteRisco,
		A.NuContratoOperacao,
		A.DsProduto,
		A.TIPO_GAR,
		A.PODER_MITIGACAO,
		MAX(A.PcExigibilidade) AS PcExigibilidade
	FROM Garantias_GL_01 AS A
	INNER JOIN (SELECT DISTINCT NuObjetoNegociacao FROM WORK.PRODUCAO) AS B
		ON A.NuContratoOperacao  = B.NuObjetoNegociacao
	WHERE Desc_Status IN ('EFETIVADA','PRE-ALOCADA')
		AND TIPO_GAR NE 'CLEAN'
	GROUP BY 1,2,3,4,5,6;















CREATE TABLE Garantias_GL_03 (DROP=PcExigibilidade) AS
	SELECT
		A.*,
		(A.PcExigibilidade * B.SALDO) / 100 AS VrExigida,
		C.QTDE
	FROM Garantias_GL_02 AS A
	LEFT JOIN (
		SELECT
			NuObjetoNegociacao,
			SUM(Vol_Producao) AS SALDO
		FROM WORK.PRODUCAO
		GROUP BY 1
	) AS B
		ON A.NuContratoOperacao = B.NuObjetoNegociacao
	LEFT JOIN (
		SELECT
			NuContratoOperacao,
			COUNT(NuContratoOperacao) AS QTDE
		FROM Garantias_GL_02
		GROUP BY 1
	) AS C
		ON A.NuContratoOperacao = C.NuContratoOperacao;

%MACRO LOG_CHANGES(DATABASE_WORK, DATABASE_FINAL, VALUE_FIELD=NAO_POSSUI);

	%IF %SYSFUNC(EXIST(&DATABASE_FINAL.)) %THEN %DO;
		%put >>>>> EXISTE;

		
		%IF &VALUE_FIELD NE NAO_POSSUI %THEN %DO;
			PROC SQL;
				SELECT SUM(&VALUE_FIELD.) INTO :WORK_VAL
				FROM &DATABASE_WORK.;

				SELECT SUM(&VALUE_FIELD.) INTO :FINAL_VAL
				FROM &DATABASE_FINAL.;
			QUIT;

			%IF %SYSFUNC(ABS(%SYSEVALF(&WORK_VAL. - &FINAL_VAL.))) > 1 %THEN %DO;
				%put >>>>> EXISTE E TEM DIFERENÇA DE SALDOS;
				PROC SQL;
					INSERT INTO TABLEAU.CHANGE_DATABASES_LOG
						SET USER_LOGIN="&_CLIENTUSERID.",
							MACHINE_ID="&_CLIENTMACHINE.",
							DATABASE="&DATABASE_FINAL.",
							DATA=TODAY(),
							CREATED_OR_UPDATE="UPDATE",
							CHANGED_VALUES="SIM",
							VALUE_FIELD="&VALUE_FIELD.",
							VALUE_DIFF=%SYSEVALF(&FINAL_VAL. - &WORK_VAL.);
				QUIT;
			%END;
			%ELSE %DO;
				%put >>>>> EXISTE, MAS NÃO TEM DIFERENÇAS DE SALDOS;
				PROC SQL;
					INSERT INTO TABLEAU.CHANGE_DATABASES_LOG
						SET USER_LOGIN="&_CLIENTUSERID.",
							MACHINE_ID="&_CLIENTMACHINE.",
							DATABASE="&DATABASE_FINAL.",
							DATA=TODAY(),
							CREATED_OR_UPDATE="UPDATE",
							CHANGED_VALUES="NÃO",
							VALUE_FIELD="&VALUE_FIELD.",
							VALUE_DIFF=0;
				QUIT;

			%END;
		%END;

		%ELSE %DO;
			PROC SQL;
				INSERT INTO TABLEAU.CHANGE_DATABASES_LOG
					SET USER_LOGIN="&_CLIENTUSERID.",
						MACHINE_ID="&_CLIENTMACHINE.",
						DATABASE="&DATABASE_FINAL.",
						DATA=TODAY(),
						CREATED_OR_UPDATE="UPDATE",
						CHANGED_VALUES="NÃO",
						VALUE_FIELD="&VALUE_FIELD.",
						VALUE_DIFF=0;
			QUIT;

		%END;

	%END;
	%ELSE %DO;
		PROC SQL;
			INSERT INTO TABLEAU.CHANGE_DATABASES_LOG
				SET USER_LOGIN="&_CLIENTUSERID.",
					MACHINE_ID="&_CLIENTMACHINE.",
					DATABASE="&DATABASE_FINAL.",
					DATA=TODAY(),
					CREATED_OR_UPDATE="CREATE",
					CHANGED_VALUES="NÃO",
					VALUE_FIELD="&VALUE_FIELD.",
					VALUE_DIFF=0;
		QUIT;
	%END;
%MEND;

TITLE 'COMPARAÇÃO DE SALDO E PDD DA BASE DE CARTEIRA';
PROC SQL;
	SELECT
		'BASE_UNICA' AS FONTE,

		SUM(VrProvisao) FORMAT=COMMAX25.2 as PROVISAO
	FROM MISCRED.BASE_UNICA_ATACADO_&ANOMES.
	WHERE DsTipoCarteira = 'REGULAR'
	UNION ALL (
		SELECT
			'TABLEAU' AS FONTE,
			SUM(Carteira_Gerencial_SAS_BR) FORMAT=COMMAX25.2 AS SALDO,
			SUM('Saldo PDD'n) FORMAT=COMMAX25.2 AS PROVISAO
		FROM WORK.BASE_CARTEIRA

TITLE 'COMPARAÇÃO DE SALDO DOS GRUPOS ECONOMICOS DIFERENTES DA BU';
PROC SQL;
	SELECT DISTINCT
		A.DsSetorClienteRisco_novo as SETOR,
		SUM(VrSaldo) FORMAT=COMMAX25.2 as SALDO,
		SUM(VrProvisao) FORMAT=COMMAX25.2 as PROVISAO,
		B.SALDO_TABLEAU,
		b.PROVISAO_TABLEAU,
		SUM(CALCULATED SALDO, -B.SALDO_TABLEAU) AS DIFF_SALDO FORMAT=COMMAX25.2,
		SUM(CALCULATED PROVISAO, -B.PROVISAO_TABLEAU) AS DIFF_PROVISAO FORMAT=COMMAX25.2
	FROM (
		SELECT *
		FROM MISCRED.BASE_UNICA_ATACADO_&ANOMES.
		WHERE DsTipoCarteira = 'REGULAR'
	) AS A
		LEFT JOIN (
			SELECT
				DsSetorClienteRisco_novo,
				SUM(Carteira_Gerencial_SAS_BR) FORMAT=COMMAX25.2 AS SALDO_TABLEAU,
				SUM('Saldo PDD'n) FORMAT=COMMAX25.2 AS PROVISAO_TABLEAU
			FROM WORK.BASE_CARTEIRA
			GROUP BY 1
		) AS B ON A.DsSetorClienteRisco_novo = B.DsSetorClienteRisco_novo
	GROUP BY 1
	HAVING ABS(DIFF_SALDO) > 1 OR ABS(DIFF_PROVISAO) > 1;
QUIT;

%macro loop_atualiza_historico(inicio=202109, fim=202506);

    %local data_ref ano mes;

    
    %let ano = %substr(&inicio, 1, 4);
    %let mes = %substr(&inicio, 5, 2);

    %do %while (%eval(&ano * 100 + &mes) <= &fim);

        %let data_ref = %sysfunc(putn(%eval(&ano * 100 + &mes), z6.));
        %put Gerando para &data_ref;

        proc sql;
            create table PRODUÇÃO_FRN_&data_ref. as
            select
                'Ref Ano'n AS Anomes,
				CdClienteRisco,
				Grupo_Final AS GrupoEconomico,
				Nome_Cliente,
                Modalidade_Produto_SAS,
				Data_Inicio_Oper,
                sum(Vol_Producao) as Producao_FRN
            from TABLEAU.BASE_TABLEAU_&data_ref.
            where fonte = 'Producao'
                  and Modalidade_Produto_SAS contains 'FRN'
            group by 1,2,3,4,5,6;
        quit;

        
        %let mes = %eval(&mes + 1);
        %if &mes > 12 %then %do;
            %let mes = 1;
            %let ano = %eval(&ano + 1);
        %end;

    %end;

%mend;

%loop_atualiza_historico(inicio=202109, fim=202506);

set PRODUÇÃO_FRN_:;
run;
