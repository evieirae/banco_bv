and t1.NmGrupoProdutoMeta = 'Leves'

PROC SQL;

CASE WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV I - Regional I','Canal Rodas e EGV - Regional I')  THEN 'C1R1'
     WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV I - Regional II','Canal Rodas e EGV - Regional II')  THEN 'C1R2'
     WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV I - Regional III','Canal Rodas e EGV II - Regional II','Canal Rodas e EGV - Regional III')  THEN 'C1R3'
     WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV II - Regional IV','Canal Rodas e EGV II - Regional I','Canal Rodas e EGV - Regional IV' ) THEN 'C2R4'
     WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV II - Regional V','Canal Rodas e EGV III - Regional I','Canal Rodas e EGV - Regional V' )  THEN 'C2R5'
     WHEN A.NmSubEstruturaComercialC  in ('Canal Rodas e EGV II - Regional VI','Canal Rodas e EGV III - Regional II','Canal Rodas e EGV - Regional VI') THEN 'C2R6'
 ELSE 'DEMAIS' END AS Regional,
      A.NmAreaEstruturaComercialC,
      A.NmParceiroComercial
   FROM PROJETOS.TBPARCEIROCOMERCIAL AS A
   WHERE A.FlAtivo = 'S'
     AND A.NmCanalEstruturaComercialC in  ('Canal Veiculos','Negocios Centralizados')
     AND A.NmTipoParceiroComercial = 'Loja';
