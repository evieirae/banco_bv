PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE LogStatusProposta_01 AS SELECT * FROM connection to dbcon (
  SELECT
    NuProposta,
    cast (DtStatusProposta as date) as DtStatusProposta,
    CASE WHEN StProposta IN ('A','CA','PA') THEN 'A' ELSE StProposta END AS StProposta,
    1 AS Prioridade

  FROM DBA.TbLogStatusProposta

  WHERE DtStatusProposta >= &iqM12.
    AND StProposta IN ('A','AP','C','CA','CE','PA','N')

);
RUN;

PROC SQL;
CREATE TABLE LogStatusProposta_01_Aux AS
  SELECT
    NuProposta,
    StProposta,
    MIN(DtStatusProposta) AS UltDtStatus FORMAT=DATE9.
  FROM LogStatusProposta_01

  GROUP BY 1,2;
RUN;

PROC SQL;
CREATE TABLE PropostasNegadasAprovadas AS
  SELECT
    NuProposta,
    MAX(CASE WHEN StProposta = 'A' THEN 1 ELSE 0 END) AS FlA,
    MAX(CASE WHEN StProposta = 'N' THEN 1 ELSE 0 END) AS FlN

  FROM LogStatusProposta_01

  GROUP BY 1
  HAVING calculated FlA + calculated FlN = 2;

RUN;

PROC SQL;
CREATE TABLE StatusAtual AS
  SELECT NuProposta, StProposta FROM BASE_LEVES
  WHERE NuProposta IN (SELECT DISTINCT NuProposta FROM PropostasNegadasAprovadas);
RUN;

PROC SQL;
CREATE TABLE LogStatusProposta_02 AS
  SELECT
    A.NuProposta,
    A.DtStatusProposta,
    A.StProposta,
    A.Prioridade

  FROM LogStatusProposta_01 A
  INNER JOIN LogStatusProposta_01_Aux B ON A.NuProposta = B.NuProposta AND A.StProposta = B.StProposta
  AND A.DtStatusProposta = B.UltDtStatus
  LEFT JOIN PropostasNegadasAprovadas C ON A.NuProposta = C.NuProposta
  LEFT JOIN StatusAtual D ON A.NuProposta = D.NuProposta
  WHERE  CASE WHEN A.StProposta <> 'N' THEN 1
              WHEN A.StProposta = 'N' AND C.NuProposta <> . AND D.StProposta = 'N' THEN 1
              ELSE 0 END = 1;
RUN;

PROC SQL;

    DtEntradaProposta AS DtStatusProposta,
    'O' AS StProposta,
    0 AS Prioridade

  FROM BASE_LEVES WHERE Safra >= &safraM12.;
RUN;

DATA LogStatusProposta;

RUN;

RUN;

PROC SORT DATA=LogStatusProposta NODUPKEY;

PROC SQL;
CREATE TABLE Propostas AS
  SELECT
    Safra AS NuAnoMes,
    NuProposta,
    Produto,
    DsTipoCanalProducao AS Canal,
	DtEntradaProposta,
    CASE WHEN Produto LIKE 'CVG%'
           THEN CASE WHEN NmParceiroComercial LIKE '%FINANZERO%' OR NmParceiroComercial LIKE '%QUERO FINANCIAR%'
                          AND DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS','PROMOTORA','BV FINANCEIRA - OPERACAO ESPECIAL')
                          AND CdConvenio not in (407516, 449109 ,450421, 452671,455884)
                          AND DtEntradaProposta BETWEEN '01JUL2020'd AND '31AUG2020'd
                          THEN 'CANAL NEGOCIOS'
                     WHEN NmParceiroComercial LIKE '%FINANZERO%' OR NmParceiroComercial LIKE '%QUERO FINANCIAR%' OR CdConvenio in (449109,450421,452671,455884,460298,448756,433956,454428,451257,462320,460903,462486) THEN 'PARCEIRO DIGITAL'
                     WHEN CdConvenio in(424584 , 452470) THEN 'CVG Área Aberta Site'
                     WHEN DsTipoCanalProducao IN ('MULTIMARCAS','CONCESSIONARIAS') THEN 'CANAL VEICULOS'
                     WHEN (CdConvenio = 407516 AND DtEntradaProposta BETWEEN '01JUL2020'd AND '31AUG2020'd)
                          OR (CdConvenio = 416382 AND DtEntradaProposta >= '01AUG2020'd)
                          THEN 'JUST'
                     WHEN NmParceiroComercial LIKE '%CP GARAN TMLKT%' THEN 'TELEMARKETING'
                     WHEN CdConvenio = 414643 THEN 'MINHA BV'
                     ELSE 'CANAL NEGOCIOS'
                     END
           ELSE 'OUTROS PRODUTOS'
         END AS FiltroCVG,
    FxIdade,
    FaixaEntrada as FxEntrada,
    PrazoFinanciamento as FxParcela

  FROM BASE_LEVES AS A
  LEFT JOIN PROJETOS.TBPARCEIROCOMERCIAL AS B ON A.CdConvenio = B.CdParceiroComercial
  WHERE Safra >= &safraM12.;
RUN;

PROC SQL;
CREATE TABLE QtPropostas AS
  SELECT
    A.NuProposta,
    A.NuAnoMes,
    INPUT(STRIP(PUT(A.NuAnoMes,6.) || '01'),yymmdd8.) AS DtMin FORMAT=DATE9.,
    MIN(INTNX('DAY', calculated DtMin, 90, 'END'),TODAY()-1) AS DtMax FORMAT=DATE9.,
    a.DtEntradaProposta,
	b.DtStatusProposta,
    A.Produto,
    SUM(CASE WHEN B.StProposta = 'O' THEN 1 ELSE 0 END) AS QtOriginadas,
    SUM(CASE WHEN B.StProposta = 'A' THEN 1 ELSE 0 END) AS QtAprovadas,
    SUM(CASE WHEN B.StProposta = 'AP' THEN 1 ELSE 0 END) AS QtEfetivadas,
    SUM(CASE WHEN B.StProposta = 'C' THEN 1 ELSE 0 END) AS QtCanceladas,
    SUM(CASE WHEN B.StProposta = 'CE' THEN 1 ELSE 0 END) AS QtCanceladasAE,
    SUM(CASE WHEN B.StProposta = 'N' THEN 1 ELSE 0 END) AS QtNegadas
  FROM Propostas A
  LEFT JOIN LogStatusProposta B ON A.NuProposta = B.NuProposta
  where A.Produto IN ('Leves','Motos')
  GROUP BY
    1,2,3,4,5,6,7;

RUN;

PROC SQL;
CREATE TABLE EFEITO_CANCELADAS AS SELECT
NuProposta,
SUM(CASE WHEN A.DtStatusProposta <= D0 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD0CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD0_AJUST,
SUM(CASE WHEN A.DtStatusProposta <= D3 and A.DtStatusProposta >= A.DtEntradaProposta and A.DtEntradaProposta <= &LstD3CSm. then QtEfetivadas ELSE 0 END) AS EFETIVADASD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD3CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD3_AJUST,
SUM(CASE WHEN A.DtEntradaProposta <= &LstD0CSm. then QtAprovadas else 0 end) as FlPropostaAprovadaD0_AJUST
