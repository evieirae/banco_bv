/* =====================Atualiza Processo=============================================*/

data BASE_EXPER_1;
    set EXP_BU.BvExper_BaseUnica_Resumo_Total;
    /* Ajuste conforme sua variável de data */
   	data = input(put((nuAnoMes*100)+1, 8.), yymmdd8.);
    format data date9.;
    ano = year(data);
    trimestre = qtr(data);
    mes = month(data);
run;

proc sql print;
    select distinct cats(ano, trimestre)
    into :trimestres SEPARATED BY ' '
    from BASE_EXPER_1
    order by ano desc, trimestre desc;
quit;

%let trim1 = %scan(&trimestres, 1);
%let trim2 = %scan(&trimestres, 2);
%let trim3 = %scan(&trimestres, 3);
%let trim4= %scan(&trimestres, 4);
%let trim_atual = %scan(&trimestres, 1);
%let trim_anterior = %scan(&trimestres, 2);

proc sql;
    create table resumo_trimestre_EXPER as
    select
		"BASE_TRI" AS TP_base_Exper,
        ano,
		. as NuAnoMes,
		ano*100+trimestre as NuAnoTri,
/*		put(ano*100+trimestre,best.)||"t" as NuAnoMes_t,*/
        trimestre,
          NmCanalEstruturaComercial,
          NmSubEstruturaComercialA,
          NmAreaEstruturaComercial,
          NmColaborador,
          FAIXA_PONTOS_TURBO,
          SUM(PONTOS_TURBO) AS PONTOS_TURBO FORMAT COMMAX12.2
      FROM BASE_EXPER_1
	   where cats(ano, trimestre) in ("&trim1", "&trim2","&trim3")
GROUP BY
        ano,
		ano*100+trimestre,
/*		put(ano*100+trimestre,best.)||"t" as NuAnoMes_t,*/
        trimestre,
          NmCanalEstruturaComercial,
          NmSubEstruturaComercialA,
          NmAreaEstruturaComercial,
          NmColaborador,
          FAIXA_PONTOS_TURBO;
quit;

proc sql;
    create table resumo_mes_EXPER as
select "BASE_MES" AS TP_base_Exper,
         ano,
		NuAnomes,
		ano*100+trimestre as NuAnoTri,
        trimestre,
          NmCanalEstruturaComercial,
          NmSubEstruturaComercialA,
          NmAreaEstruturaComercial,
          NmColaborador,
          FAIXA_PONTOS_TURBO,
          SUM(PONTOS_TURBO) AS PONTOS_TURBO FORMAT COMMAX12.2
      FROM BASE_EXPER_1
	  where cats(ano, trimestre) in ("&trim1", "&trim2","&trim3")
GROUP BY
        ano,
		NuAnomes,
		ano*100+trimestre,
        trimestre,
          NmCanalEstruturaComercial,
          NmSubEstruturaComercialA,
          NmAreaEstruturaComercial,
          NmColaborador,
          FAIXA_PONTOS_TURBO;
	quit;

data base_final_RC_Tatico_BVEXPER;
    set resumo_trimestre_EXPER resumo_mes_EXPER;
run;

	%let dir_csv = /riscosbvf/BA/processos/output;

		OUTFILE="&dir_csv./BASE_FINAL_RC_TATICO_BVEXPER.csv" REPLACE
		DBMS=DLM;
	DELIMITER=';';
RUN;
