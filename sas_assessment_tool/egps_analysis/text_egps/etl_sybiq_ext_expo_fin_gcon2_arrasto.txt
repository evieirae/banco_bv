%macro etls_recordCheck;
   %let etls_recCheckExist = %eval(%sysfunc(exist(PEPREXT.EXT_EXPO_FIN_GCON2_ARRASTO, DATA)) or
         %sysfunc(exist(PEPREXT.EXT_EXPO_FIN_GCON2_ARRASTO, VIEW)));

   %if (&etls_recCheckExist) %then
   %do;
      %local etls_syntaxcheck;
      %let etls_syntaxcheck = %sysfunc(getoption(syntaxcheck));
      /* Turn off syntaxcheck option to perform following steps  */
      options nosyntaxcheck;

      proc contents data = PEPREXT.EXT_EXPO_FIN_GCON2_ARRASTO out = work.etls_contents(keep = nobs) noprint;

         delete etls_contents;
      quit;

      /* Reset syntaxcheck option to previous setting  */
      options &etls_syntaxcheck;
   %end;
%mend etls_recordCheck;
