Create table WORK.final as select * from connection to perm
( WITH base_aju AS (
  SELECT
    CONCAT(
      CAST(EXTRACT(YEAR FROM dt_inicio_ajuizamento) AS STRING),
      LPAD(CAST(EXTRACT(MONTH FROM dt_inicio_ajuizamento) AS STRING), 2, '0')
    ) AS SafraAjuizamento,
    CAST(cd_contrato AS STRING) AS cd_contrato,
    CAST(sq_contrato_financeiro AS STRING) AS sq_contrato_financeiro_base,
    cd_status,
    cd_pasta,
    fl_ultimo_ajuizamento,
    dt_inicio_ajuizamento,
    dt_parcela_ajuizada,
    CASE WHEN dt_inicio_ajuizamento IS NOT NULL THEN 1 ELSE 0 END AS fl_ajuizamento,
    CASE
      WHEN DATE_DIFF(dt_inicio_ajuizamento, dt_parcela_ajuizada, DAY) < 15  THEN '00 - MENOR 15 DIAS'
      WHEN DATE_DIFF(dt_inicio_ajuizamento, dt_parcela_ajuizada, DAY) < 71  THEN '01 - ENTRE 15 E 70'
      WHEN DATE_DIFF(dt_inicio_ajuizamento, dt_parcela_ajuizada, DAY) < 141 THEN '02 - ENTRE 71 E 140'
      WHEN DATE_DIFF(dt_inicio_ajuizamento, dt_parcela_ajuizada, DAY) < 181 THEN '03 - ENTRE 141 E 180'
      WHEN DATE_DIFF(dt_inicio_ajuizamento, dt_parcela_ajuizada, DAY) < 361 THEN '04 - ENTRE 181 E 360'
      ELSE '05 - MAIOR 360'
    END AS FAIXA_ATRASO,
    COALESCE(fl_revisional_ajuizamento, FALSE) AS fl_revisional_ajuizamento,
    CASE
      WHEN nm_assessoria LIKE '%Bellinati%'      THEN 'BELLINATI'
      WHEN nm_assessoria LIKE '%BELLINATI%'      THEN 'BELLINATI'
      WHEN nm_assessoria LIKE '%Toledo%'         THEN 'TOLEDO'
      WHEN nm_assessoria LIKE '%Schulze%'        THEN 'SCHULZE'
      WHEN nm_assessoria LIKE '%LM Angelis%'     THEN 'LM ANGELIS'
      WHEN nm_assessoria LIKE '%Mendes e Cunha%' THEN 'MENDES E CUNHA'
      WHEN nm_assessoria LIKE '%Paschoalotto%'   THEN 'PASCHOALOTTO'
      WHEN nm_assessoria LIKE '%Mandaliti%'      THEN 'MANDALITI'
      ELSE nm_assessoria
    END AS ASSESSORIA,
    cd_uf AS UF,
    dt_liminar,
    dt_mandado,
    dt_retomado
  FROM `bv-pdbd-prd.dvry_corp_cobranca.ajuizamento_contratos_fato`
  WHERE DATE(dt_inicio_ajuizamento) >= DATE_SUB(CURRENT_DATE(), INTERVAL 26 MONTH)
),

maior_seq AS (
  SELECT
    CAST(cd_contrato AS STRING) AS cd_contrato,
    CAST(MAX(SAFE_CAST(sq_contrato_financeiro AS INT64)) AS STRING) AS sq_contrato_financeiro_max
  FROM `bv-pdbd-prd.dvry_core.tbparcela`
  WHERE dt_vencimento >= DATE_SUB(CURRENT_DATE(), INTERVAL 72 MONTH)
  GROUP BY 1
),

tbparcela_ms AS (
  SELECT p.cd_contrato,
         p.sq_contrato_financeiro,
         p.dt_vencimento,
         p.dt_ultimo_recebimento,
         p.st_parcela
  FROM `bv-pdbd-prd.dvry_core.tbparcela` p
  JOIN maior_seq m
    ON CAST(p.cd_contrato AS STRING) = m.cd_contrato
   AND SAFE_CAST(p.sq_contrato_financeiro AS INT64) = SAFE_CAST(m.sq_contrato_financeiro_max AS INT64)
  WHERE p.dt_vencimento >= DATE_SUB(CURRENT_DATE(), INTERVAL 72 MONTH)
),

tbparcela_exist AS (
  SELECT
    CAST(p.cd_contrato AS STRING) AS cd_contrato,
    CAST(p.sq_contrato_financeiro AS STRING) AS sq_contrato_financeiro,
    COUNT(*) AS qtd_parcelas_seq
  FROM tbparcela_ms p
  GROUP BY 1,2
),

base_aju_seq_pre AS (
  SELECT
    b.SafraAjuizamento,
    b.cd_contrato,
    COALESCE(b.sq_contrato_financeiro_base, m.sq_contrato_financeiro_max) AS sq_contrato_financeiro,
    b.cd_status,
    b.cd_pasta,
    b.fl_ultimo_ajuizamento,
    b.dt_inicio_ajuizamento,
    b.dt_parcela_ajuizada,
    b.fl_ajuizamento,
    b.FAIXA_ATRASO,
    b.fl_revisional_ajuizamento,
    b.ASSESSORIA,
    b.UF,
    b.dt_liminar,
    b.dt_mandado,
    b.dt_retomado
  FROM base_aju b
  LEFT JOIN maior_seq m
    ON m.cd_contrato = b.cd_contrato
),

base_aju_seq AS (
  SELECT * EXCEPT(rn) FROM (
    SELECT
      bas.*,
      ROW_NUMBER() OVER (
        PARTITION BY bas.cd_contrato
        ORDER BY SAFE_CAST(bas.sq_contrato_financeiro AS INT64) DESC,
                 bas.fl_ultimo_ajuizamento DESC,
                 bas.dt_inicio_ajuizamento DESC,
                 bas.dt_parcela_ajuizada DESC
      ) AS rn
    FROM base_aju_seq_pre bas
  )
  WHERE rn = 1
),

emenda_pos_aju AS (
  SELECT
    b.cd_pasta,
    MIN(e.dt_evento) AS dt_emenda
  FROM base_aju_seq b
  JOIN `bv-pdbd-prd.dvry_corp_cobranca.ajuizamento_eventos_fato` e
    ON e.cd_pasta = b.cd_pasta
  WHERE e.ds_evento LIKE 'ADITAMENTO DA INICIAL'
    AND DATE(e.dt_evento) >= DATE(b.dt_inicio_ajuizamento)
    AND DATE(e.dt_evento) >= DATE_SUB(CURRENT_DATE(), INTERVAL 28 MONTH)
  GROUP BY b.cd_pasta
),

base_aju_seq_emenda AS (
  SELECT
    b.*,
    em.dt_emenda
  FROM base_aju_seq b
  LEFT JOIN emenda_pos_aju em
    USING (cd_pasta)
),

mobs AS (
  SELECT
    b.*,
    m.mob,
    m.off AS mob_offset,
    DATE_TRUNC(DATE_ADD(b.dt_inicio_ajuizamento, INTERVAL m.off MONTH), MONTH) AS ref_month,
    LAST_DAY(DATE_ADD(b.dt_inicio_ajuizamento, INTERVAL m.off MONTH)) AS fechamento,
    LAST_DAY(DATE_SUB(DATE_ADD(b.dt_inicio_ajuizamento, INTERVAL m.off MONTH), INTERVAL 1 MONTH)) AS fechamento_mes_anterior
  FROM base_aju_seq_emenda b
  CROSS JOIN UNNEST([
    STRUCT('M0' AS mob, 0 AS off),
    STRUCT('M1' AS mob, 1 AS off),
    STRUCT('M2' AS mob, 2 AS off),
    STRUCT('M3' AS mob, 3 AS off),
    STRUCT('M4' AS mob, 4 AS off),
    STRUCT('M5' AS mob, 5 AS off),
    STRUCT('M6' AS mob, 6 AS off),
    STRUCT('M7' AS mob, 7 AS off),
    STRUCT('M8' AS mob, 8 AS off),
    STRUCT('M9' AS mob, 9 AS off),
    STRUCT('M10' AS mob, 10 AS off)
  ]) AS m
),


contratos_base AS (
  SELECT
    CAST(NUCONTRATO AS STRING)            AS cd_contrato,
    CAST(SQCONTRATOFINANCEIRO AS STRING)  AS sq_contrato_financeiro,
    DATE(DTPROCESSAMENTO)                 AS dtprocessamento,
    DATE(dtprocessamentoliquidacao)       AS dt_quitacao,
    DATE(dtcessao)                        AS dt_cessao,
    CDPRODUTO,
    CDMODALIDADEPRODUTO,
    CDSITUACAOCONTABIL,
    ROUND(SUM(CAST(VRPERMANENCIAAPROPRIADO AS FLOAT64) + CAST(VRTAXACONTRATO AS FLOAT64)), 2) AS SALDO_BRUTO,
    ROUND(SUM(CAST(VRPDD AS FLOAT64)), 2) AS PDD_BRUTO
  FROM `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil`
  WHERE DATE(DTPROCESSAMENTO) >= DATE_SUB(CURRENT_DATE(), INTERVAL 80 MONTH)
  AND NUCONTRATO IS NOT NULL
  GROUP BY cd_contrato, sq_contrato_financeiro, dtprocessamento, dt_quitacao, dt_cessao,
           CDPRODUTO, CDMODALIDADEPRODUTO, CDSITUACAOCONTABIL
),


eventos_finalizadores_base AS (
  SELECT cd_contrato, sq_contrato_financeiro, dt_quitacao AS evento_data, 'Quitacao' AS evento_tipo
  FROM contratos_base
  WHERE dt_quitacao IS NOT NULL
    AND dt_quitacao >= DATE_SUB(CURRENT_DATE(), INTERVAL 26 MONTH)

  UNION ALL
  SELECT cd_contrato, sq_contrato_financeiro, dt_cessao AS evento_data, 'Cessao | FIDC' AS evento_tipo
  FROM contratos_base
  WHERE dt_cessao IS NOT NULL
    AND dt_cessao >= DATE_SUB(CURRENT_DATE(), INTERVAL 26 MONTH)

  UNION ALL
  SELECT
    CAST(NUCONTRATO AS STRING) AS cd_contrato,
    NULL AS sq_contrato_financeiro,
    DATE(DTINCLUSAO) AS evento_data,
    CASE
      WHEN TPAPREENSAO IN (1, 2) THEN 'BUSCA'
      WHEN TPAPREENSAO = 3       THEN 'EA_C_QUITACAO'
      WHEN TPAPREENSAO = 4       THEN 'EA_S_QUITACAO'
      WHEN TPAPREENSAO = 5       THEN 'EA_CONVER'
    END AS evento_tipo
  FROM `bv-pdbd-prd.trusted_dbcontencioso.tbprocessojuridico`
  WHERE DTINCLUSAO IS NOT NULL AND TPAPREENSAO IN (1,2,3,4,5)
  AND DATE(DTINCLUSAO) >= DATE_SUB(CURRENT_DATE(), INTERVAL 80 MONTH)
),

evento_finalizador_flag AS (
  SELECT
    m.SafraAjuizamento,
    m.cd_contrato,
    m.sq_contrato_financeiro,
    m.mob,
    CASE WHEN COUNTIF(
      ev.evento_tipo IN ('Quitacao','Cessao | FIDC','EA_C_QUITACAO','EA_S_QUITACAO','EA_CONVER','BUSCA')
      AND ev.evento_data <= m.fechamento
    ) > 0 THEN 1 ELSE 0 END AS fl_evento_finalizador_ate_fechamento
  FROM mobs m
  LEFT JOIN eventos_finalizadores_base ev
    ON ev.cd_contrato = m.cd_contrato
   AND (ev.sq_contrato_financeiro IS NULL
        OR SAFE_CAST(ev.sq_contrato_financeiro AS INT64) = SAFE_CAST(m.sq_contrato_financeiro AS INT64))
  GROUP BY 1,2,3,4
),


atraso_por_mob AS (
  SELECT
    m.SafraAjuizamento,
    m.cd_contrato,
    m.sq_contrato_financeiro,
    m.mob,
    m.mob_offset,
    m.ref_month,
    m.fechamento,
    m.fechamento_mes_anterior,

    MAX(
      IF(
        p.dt_vencimento <= m.fechamento
        AND NOT ( (p.dt_ultimo_recebimento IS NOT NULL AND DATE(p.dt_ultimo_recebimento) <= m.fechamento) OR p.st_parcela = 'Q' ),
        p.dt_vencimento, NULL
      )
    ) AS dt_ult_venc_aberto,

    COALESCE(SUM(
      IF(
        p.dt_vencimento <= m.fechamento
        AND NOT ( (p.dt_ultimo_recebimento IS NOT NULL AND DATE(p.dt_ultimo_recebimento) <= m.fechamento) OR p.st_parcela = 'Q' ),
        1, 0
      )
    ), 0) AS qtd_parc_abertas,

    COALESCE(SUM(
      IF(
        p.dt_vencimento <= m.fechamento_mes_anterior
        AND NOT ( (p.dt_ultimo_recebimento IS NOT NULL AND DATE(p.dt_ultimo_recebimento) <= m.fechamento_mes_anterior) OR p.st_parcela = 'Q' ),
        1, 0
      )
    ), 0) AS qtd_parc_abertas_mes_anterior,

    COALESCE(SUM(
      IF(
        p.dt_vencimento <= m.fechamento_mes_anterior
        AND NOT ( (p.dt_ultimo_recebimento IS NOT NULL AND DATE(p.dt_ultimo_recebimento) <= m.fechamento) OR p.st_parcela = 'Q' ),
        1, 0
      )
    ), 0) AS qtd_prev_abertas_restantes,

    
    COALESCE(MAX( IF(p.dt_vencimento > m.fechamento, 1, 0) ), 0) AS fl_tem_parcelas_futuras_agenda,

    
    COALESCE(MAX( IF(p.dt_vencimento > m.fechamento AND p.st_parcela != 'Q', 1, 0) ), 0) AS fl_tem_parcelas_futuras,

    COALESCE(SUM(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY) AND m.fechamento,
        1, 0
      )
    ), 0) AS qtd_baixas_no_mes,

    MAX(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY) AND m.fechamento,
        DATE(p.dt_ultimo_recebimento), NULL
      )
    ) AS dt_ult_baixa_mes,

    MIN(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY) AND m.fechamento,
        DATE(p.dt_ultimo_recebimento), NULL
      )
    ) AS dt_prim_baixa_mes,

    
    COALESCE(SUM(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN
              GREATEST(DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY), DATE(m.dt_inicio_ajuizamento))
          AND m.fechamento,
        1, 0
      )
    ), 0) AS qtd_baixas_no_mes_pos_aju,

    MIN(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN
              GREATEST(DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY), DATE(m.dt_inicio_ajuizamento))
          AND m.fechamento,
        DATE(p.dt_ultimo_recebimento), NULL
      )
    ) AS dt_prim_baixa_mes_pos_aju,

    
    COALESCE(SUM(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN
              GREATEST(DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY), DATE(m.dt_inicio_ajuizamento))
          AND m.fechamento
        AND DATE(p.dt_vencimento) < DATE(p.dt_ultimo_recebimento),
        1, 0
      )
    ), 0) AS qtd_baixas_no_mes_pos_aju_em_atraso,

    MIN(
      IF(
        p.dt_ultimo_recebimento IS NOT NULL
        AND DATE(p.dt_ultimo_recebimento) BETWEEN
              GREATEST(DATE_ADD(m.fechamento_mes_anterior, INTERVAL 1 DAY), DATE(m.dt_inicio_ajuizamento))
          AND m.fechamento
        AND DATE(p.dt_vencimento) < DATE(p.dt_ultimo_recebimento),
        DATE(p.dt_ultimo_recebimento), NULL
      )
    ) AS dt_prim_baixa_mes_pos_aju_em_atraso,

    ef.fl_evento_finalizador_ate_fechamento

  FROM mobs m
  LEFT JOIN tbparcela_ms p
    ON CAST(p.cd_contrato AS STRING) = m.cd_contrato
   AND SAFE_CAST(p.sq_contrato_financeiro AS INT64) = SAFE_CAST(m.sq_contrato_financeiro AS INT64)
  LEFT JOIN evento_finalizador_flag ef
    ON ef.SafraAjuizamento = m.SafraAjuizamento
   AND ef.cd_contrato      = m.cd_contrato
   AND ef.sq_contrato_financeiro = m.sq_contrato_financeiro
   AND ef.mob              = m.mob
  GROUP BY
    m.SafraAjuizamento, m.cd_contrato, m.sq_contrato_financeiro,
    m.mob, m.mob_offset, m.ref_month, m.fechamento, m.fechamento_mes_anterior,
    ef.fl_evento_finalizador_ate_fechamento
),


abertas_ate_dt_ult_baixa_calc AS (
  SELECT
    a.SafraAjuizamento, a.cd_contrato, a.sq_contrato_financeiro, a.mob,
    IFNULL(SUM(
      IF(
        a.dt_ult_baixa_mes IS NOT NULL
        AND p2.dt_vencimento <= a.dt_ult_baixa_mes
        AND NOT ( (p2.dt_ultimo_recebimento IS NOT NULL AND DATE(p2.dt_ultimo_recebimento) <= a.dt_ult_baixa_mes) OR p2.st_parcela = 'Q' ),
        1, 0)
    ), 0) AS qtd_abertas_ate_dt_ult_baixa
  FROM atraso_por_mob a
  LEFT JOIN tbparcela_ms p2
    ON CAST(p2.cd_contrato AS STRING) = a.cd_contrato
   AND SAFE_CAST(p2.sq_contrato_financeiro AS INT64) = SAFE_CAST(a.sq_contrato_financeiro AS INT64)
  GROUP BY 1,2,3,4
),

atraso_por_mob_2 AS (
  SELECT
    a.*,
    h.qtd_abertas_ate_dt_ult_baixa,
    CASE
      WHEN a.dt_ult_baixa_mes IS NOT NULL AND IFNULL(h.qtd_abertas_ate_dt_ult_baixa,0) = 0 THEN 1 ELSE 0
    END AS fl_zerou_ate_data_pag
  FROM atraso_por_mob a
  LEFT JOIN abertas_ate_dt_ult_baixa_calc h
    ON h.SafraAjuizamento = a.SafraAjuizamento
   AND h.cd_contrato      = a.cd_contrato

classificado_pre AS (
  SELECT
    d.*,
    CASE
      WHEN COALESCE(te.qtd_parcelas_seq, 0) = 0 THEN 'sem acao'
      WHEN d.qtd_baixas_no_mes_pos_aju_em_atraso > 0 THEN
        CASE
          WHEN d.qtd_parc_abertas = 0
           AND (d.fl_tem_parcelas_futuras_agenda = 0 OR d.fl_evento_finalizador_ate_fechamento = 1) THEN 'Quitacao'
          WHEN d.qtd_parc_abertas = 0
           AND d.fl_tem_parcelas_futuras_agenda = 1 THEN 'Atualizacao'
          ELSE 'pagamento parcial'
        END
      ELSE 'sem acao'
    END AS tipo_pag_v2_raw
  FROM atraso_dias d
  LEFT JOIN tbparcela_exist te
    ON te.cd_contrato = d.cd_contrato
   AND te.sq_contrato_financeiro = d.sq_contrato_financeiro

classificado AS (
  SELECT
    cp.* EXCEPT(tipo_pag_v2_raw),
    CASE
      WHEN cp.tipo_pag_v2_raw IN ('Atualizacao','pagamento parcial') AND cp.dt_prim_baixa_mes IS NULL THEN 'sem acao'
      WHEN cp.mob = 'M0'
       AND cp.dt_prim_baixa_mes_pos_aju IS NULL
       AND cp.dt_prim_baixa_mes IS NOT NULL
       AND DATE(cp.dt_prim_baixa_mes) < DATE(cp.dt_inicio_ajuizamento) THEN 'sem acao'
      ELSE cp.tipo_pag_v2_raw
    END AS tipo_pag_v2
  FROM (
    SELECT d.*, bas.dt_inicio_ajuizamento
    FROM classificado_pre d
    LEFT JOIN base_aju_seq_emenda bas
      ON bas.SafraAjuizamento = d.SafraAjuizamento
     AND bas.cd_contrato      = d.cd_contrato
     AND bas.sq_contrato_financeiro = d.sq_contrato_financeiro
  ) cp

eventos_internos_mob AS (
  SELECT
    d.SafraAjuizamento,
    d.cd_contrato,
    d.sq_contrato_financeiro,
    d.mob,
    d.dt_prim_baixa_mes_pos_aju_em_atraso AS evento_data,
    CASE
      WHEN d.qtd_parc_abertas = 0
       AND (d.fl_tem_parcelas_futuras_agenda = 0 OR d.fl_evento_finalizador_ate_fechamento = 1) THEN 'Quitacao'
      WHEN d.qtd_parc_abertas = 0 AND d.fl_tem_parcelas_futuras_agenda = 1 THEN 'Atualizacao'

  FROM atraso_dias d
  WHERE d.dt_prim_baixa_mes_pos_aju_em_atraso IS NOT NULL

evento_pagamento_por_mob AS (
  SELECT
    c.SafraAjuizamento, c.cd_contrato, c.sq_contrato_financeiro, c.mob, c.mob_offset, c.ref_month, c.fechamento,
    e.evento_data, e.evento_tipo
  FROM classificado c
  JOIN eventos_externos e
    ON e.cd_contrato = c.cd_contrato
   AND SAFE_CAST(e.sq_contrato_financeiro AS INT64) = SAFE_CAST(c.sq_contrato_financeiro AS INT64)
   AND e.evento_data <= c.fechamento
   AND (e.evento_tipo != 'Cessao | FIDC'
        OR e.evento_data >= DATE_TRUNC(DATE_ADD(c.fechamento, INTERVAL -c.mob_offset MONTH), MONTH))
   AND (e.evento_tipo != 'Reneg' OR e.dt_cancelamento IS NULL OR e.dt_cancelamento > c.fechamento)

  UNION ALL
  SELECT
    c.SafraAjuizamento, c.cd_contrato, c.sq_contrato_financeiro, c.mob, c.mob_offset, c.ref_month, c.fechamento,
    a.evento_data, a.evento_tipo
  FROM classificado c
  JOIN apreensao_valida a
    ON a.cd_contrato = c.cd_contrato
   AND a.evento_data <= c.fechamento

  UNION ALL
  SELECT
    i.SafraAjuizamento, i.cd_contrato, i.sq_contrato_financeiro, i.mob,
    m.mob_offset, m.ref_month, m.fechamento,
    i.evento_data, i.evento_tipo
  FROM eventos_internos_mob i
  JOIN mobs m
    ON m.SafraAjuizamento = i.SafraAjuizamento
   AND m.cd_contrato = i.cd_contrato
   AND m.sq_contrato_financeiro = i.sq_contrato_financeiro
   AND m.mob = i.mob
),

evento_pagamento_por_mob_norm AS (
  SELECT
    e.*,
    CASE
      WHEN e.evento_tipo IN ('EA_C_QUITACAO','EA_S_QUITACAO','EA_CONVER','BUSCA') THEN 'BNDU (EA e BAP)'
      ELSE e.evento_tipo
    END AS evento_tipo_exibicao,
    CASE
      WHEN e.evento_tipo IN ('Atualizacao','pagamento parcial') THEN d.dt_prim_baixa_mes_pos_aju_em_atraso
      ELSE e.evento_data
    END AS evento_data_pag
  FROM evento_pagamento_por_mob e
  LEFT JOIN atraso_dias d
    ON d.SafraAjuizamento       = e.SafraAjuizamento
   AND d.cd_contrato            = e.cd_contrato
   AND d.sq_contrato_financeiro = e.sq_contrato_financeiro
   AND d.mob                    = e.mob
),

primeiro_evento_pagamento_mob AS (
  SELECT
    SafraAjuizamento, cd_contrato, sq_contrato_financeiro, mob,
    MIN(evento_data_pag) AS dt_primeiro_evento_pagamento_mob
  FROM evento_pagamento_por_mob_norm
  WHERE evento_tipo_exibicao IN ('Quitacao','Reneg','Atualizacao','pagamento parcial','BNDU (EA e BAP)')
    AND evento_data_pag IS NOT NULL
  GROUP BY 1,2,3,4
),

primeiro_evento_pagamento_pos_aju AS (
  SELECT
    b.cd_contrato, b.sq_contrato_financeiro,
    MIN(n.evento_data_pag) AS dt_primeiro_evento_pagamento_pos_aju
  FROM base_aju_seq_emenda b
  JOIN evento_pagamento_por_mob_norm n
    ON n.cd_contrato = b.cd_contrato
   AND n.sq_contrato_financeiro = b.sq_contrato_financeiro
  WHERE n.evento_tipo_exibicao IN ('Quitacao','Reneg','Atualizacao','pagamento parcial','BNDU (EA e BAP)')
    AND n.evento_data_pag IS NOT NULL
    AND n.evento_data_pag >= DATE(b.dt_inicio_ajuizamento)
  GROUP BY 1,2
),

primeiro_evento_global AS (
  SELECT
    n.cd_contrato,
    n.sq_contrato_financeiro,
    MIN(n.evento_data_pag) AS evento_data_primeiro,
    (ARRAY_AGG(n.evento_tipo
      ORDER BY n.evento_data_pag,
        CASE evento_tipo
          WHEN 'Quitacao' THEN 1
          WHEN 'EA_C_QUITACAO' THEN 2
          WHEN 'EA_S_QUITACAO' THEN 3
          WHEN 'EA_CONVER' THEN 4
          WHEN 'BUSCA' THEN 5
          WHEN 'Reneg' THEN 6
          WHEN 'Atualizacao' THEN 7
          WHEN 'pagamento parcial' THEN 8
          ELSE 99 END
      LIMIT 1))[OFFSET(0)] AS evento_tipo_primeiro_raw
  FROM evento_pagamento_por_mob_norm n
  JOIN base_aju_seq_emenda b
    ON b.cd_contrato = n.cd_contrato
   AND b.sq_contrato_financeiro = n.sq_contrato_financeiro
  WHERE n.evento_tipo IN ('Quitacao','EA_C_QUITACAO','EA_S_QUITACAO','EA_CONVER','BUSCA','Reneg','Atualizacao','pagamento parcial')
    AND n.evento_data_pag IS NOT NULL
    AND n.evento_data_pag >= DATE(b.dt_inicio_ajuizamento)
  GROUP BY 1,2
),


saida_base AS (
  SELECT
    c.SafraAjuizamento,
    c.cd_contrato,
    c.sq_contrato_financeiro,
    c.mob,
    c.ref_month,
    c.fechamento,
    c.fechamento_mes_anterior,

    c.qtd_parc_abertas,
    c.qtd_parc_abertas_mes_anterior,
    c.qtd_prev_abertas_restantes,
    c.qtd_baixas_no_mes,
    c.dt_prim_baixa_mes,
    c.dt_ult_baixa_mes,
    c.qtd_abertas_ate_dt_ult_baixa,
    c.fl_zerou_ate_data_pag,
    c.fl_tem_parcelas_futuras,

    c.QtDiaAtraso_fechamento_legacy,

    bas.ASSESSORIA,
    bas.UF,
    bas.dt_liminar,
    bas.dt_mandado,
    bas.dt_retomado,
    bas.dt_emenda,

    DATE(bas.dt_parcela_ajuizada)   AS dt_base_atraso_aju,
    DATE(bas.dt_inicio_ajuizamento) AS dt_inicio_ajuizamento,

    GREATEST(0, DATE_DIFF(DATE(bas.dt_inicio_ajuizamento), DATE(bas.dt_parcela_ajuizada), DAY)) AS DiasAtrasoNoAjuizamento,

    pep.dt_primeiro_evento_pagamento_mob,

    peaj.dt_primeiro_evento_pagamento_pos_aju,

    GREATEST(
      0,
      DATE_DIFF(
        COALESCE(
          CASE WHEN pep.dt_primeiro_evento_pagamento_mob IS NOT NULL
                 AND pep.dt_primeiro_evento_pagamento_mob BETWEEN DATE_TRUNC(c.ref_month, MONTH) AND c.fechamento
               THEN pep.dt_primeiro_evento_pagamento_mob
               ELSE NULL END,
          c.fechamento
        ),
        DATE(bas.dt_parcela_ajuizada),
        DAY
      )
    ) AS QtDiaAtraso_fechamento,

    GREATEST(
      0,
      CASE
        WHEN peaj.dt_primeiro_evento_pagamento_pos_aju IS NOT NULL
         AND c.ref_month >= DATE_TRUNC(peaj.dt_primeiro_evento_pagamento_pos_aju, MONTH)
        THEN DATE_DIFF(peaj.dt_primeiro_evento_pagamento_pos_aju, DATE(bas.dt_parcela_ajuizada), DAY)
        ELSE DATE_DIFF(c.fechamento, DATE(bas.dt_parcela_ajuizada), DAY)
      END
    ) AS QtDiaAtraso_base_propag,

    c.tipo_pag_v2,

    pg.evento_tipo_primeiro_raw AS evento_tipo_primeiro,
    pg.evento_data_primeiro     AS evento_data_primeiro,

    CASE
      WHEN c.mob = 'M0'
       AND c.dt_prim_baixa_mes_pos_aju IS NULL
       AND c.dt_prim_baixa_mes IS NOT NULL
       AND DATE(c.dt_prim_baixa_mes) < DATE(bas.dt_inicio_ajuizamento) THEN 'sem acao'
      WHEN peaj.dt_primeiro_evento_pagamento_pos_aju IS NULL
        THEN c.tipo_pag_v2
      WHEN c.ref_month < DATE_TRUNC(peaj.dt_primeiro_evento_pagamento_pos_aju, MONTH)
        THEN 'sem acao'
      ELSE pg.evento_tipo_primeiro_raw
    END AS tipo_pag_final,

    CASE
      WHEN (
        CASE
          WHEN c.mob = 'M0'
           AND c.dt_prim_baixa_mes_pos_aju IS NULL
           AND c.dt_prim_baixa_mes IS NOT NULL
           AND DATE(c.dt_prim_baixa_mes) < DATE(bas.dt_inicio_ajuizamento) THEN 'sem acao'
          WHEN peaj.dt_primeiro_evento_pagamento_pos_aju IS NULL THEN c.tipo_pag_v2
          WHEN c.ref_month < DATE_TRUNC(peaj.dt_primeiro_evento_pagamento_pos_aju, MONTH) THEN 'sem acao'
          ELSE pg.evento_tipo_primeiro_raw
        END
      ) IN ('EA_C_QUITACAO','EA_S_QUITACAO','EA_CONVER','BUSCA') THEN 'BNDU (EA e BAP)'
      ELSE (
        CASE
          WHEN c.mob = 'M0'
           AND c.dt_prim_baixa_mes_pos_aju IS NULL
           AND c.dt_prim_baixa_mes IS NOT NULL
           AND DATE(c.dt_prim_baixa_mes) < DATE(bas.dt_inicio_ajuizamento) THEN 'sem acao'
          WHEN peaj.dt_primeiro_evento_pagamento_pos_aju IS NULL THEN c.tipo_pag_v2
          WHEN c.ref_month < DATE_TRUNC(peaj.dt_primeiro_evento_pagamento_pos_aju, MONTH) THEN 'sem acao'
          ELSE pg.evento_tipo_primeiro_raw
        END
      )
    END AS tipo_pag_final_exibicao
  FROM classificado c
  LEFT JOIN primeiro_evento_pagamento_mob pep
    ON pep.SafraAjuizamento       = c.SafraAjuizamento
   AND pep.cd_contrato            = c.cd_contrato
   AND pep.sq_contrato_financeiro = c.sq_contrato_financeiro
   AND pep.mob                    = c.mob
  LEFT JOIN primeiro_evento_global pg
    ON pg.cd_contrato            = c.cd_contrato
   AND pg.sq_contrato_financeiro = c.sq_contrato_financeiro
  LEFT JOIN base_aju_seq_emenda bas
    ON bas.SafraAjuizamento      = c.SafraAjuizamento
   AND bas.cd_contrato           = c.cd_contrato
   AND bas.sq_contrato_financeiro= c.sq_contrato_financeiro
  LEFT JOIN primeiro_evento_pagamento_pos_aju peaj
    ON peaj.cd_contrato          = c.cd_contrato
   AND peaj.sq_contrato_financeiro = c.sq_contrato_financeiro
  WHERE c.fechamento <= LAST_DAY(CURRENT_DATE())
),

      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 5 THEN 'MOTOS'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 7  AND tbproposta.CDSUBPRODUTO IS NULL THEN 'LEVES'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 7  AND tbproposta.CDSUBPRODUTO = 4  THEN 'LEVES'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 7  AND tbproposta.CDSUBPRODUTO = 3  THEN 'PESADOS'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 8  AND tbproposta.CDSUBPRODUTO IS NULL THEN 'LEVES'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 8  AND tbproposta.CDSUBPRODUTO IN (1,58) THEN 'LEVES'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 9  AND tbproposta.CDSUBPRODUTO IS NULL THEN 'PESADOS'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 9  AND tbproposta.CDSUBPRODUTO IN (2,3,61) THEN 'PESADOS'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 38 AND tbproposta.CDSUBPRODUTO = 29 THEN 'LEVES'
      WHEN cb.CDPRODUTO = 12 AND cb.CDMODALIDADEPRODUTO = 38 AND tbproposta.CDSUBPRODUTO = 31 THEN 'PESADOS'
      WHEN cb.CDPRODUTO = 16 AND cb.CDMODALIDADEPRODUTO = 60 THEN 'COBAN'
    END AS PRODUTO
  FROM (
    SELECT DISTINCT cd_contrato, CDPRODUTO, CDMODALIDADEPRODUTO
    FROM contratos_base
  ) cb
  LEFT JOIN (
    SELECT NUCONTRATOGESTAO, STPROPOSTA, CDSUBPRODUTO, dtentradaproposta
    FROM `bv-pdbd-prd.trusted_dbcred.tbproposta`
    WHERE DATE(dtentradaproposta) >= DATE_SUB(CURRENT_DATE(), INTERVAL 80 MONTH)
      AND STPROPOSTA = 'AP'
  ) tbproposta
    ON cb.cd_contrato = tbproposta.NUCONTRATOGESTAO
  LEFT JOIN (
    SELECT CDSUBPRODUTO, CDGRUPOSUBPRODUTO
    FROM `bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo`
  ) tbsubprodutogrupo
    ON tbproposta.CDSUBPRODUTO = tbsubprodutogrupo.CDSUBPRODUTO
  LEFT JOIN (
    SELECT CDGRUPOSUBPRODUTO
    FROM `bv-pdbd-prd.trusted_dbcred.tbgruposubproduto`
  ) tbgruposubproduto
    ON tbsubprodutogrupo.CDGRUPOSUBPRODUTO = tbgruposubproduto.CDGRUPOSUBPRODUTO
),

provisao_produto AS (
  SELECT
    nucontrato,
    (ARRAY_AGG(DISTINCT PRODUTO ORDER BY PRODUTO LIMIT 1))[OFFSET(0)] AS PRODUTO
  FROM provisao_produto_raw
  GROUP BY nucontrato
),


final AS (

    sb.evento_tipo_primeiro,
    sb.evento_data_primeiro,

    sb.tipo_pag_final,
    sb.tipo_pag_final_exibicao,

    CASE
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju IS NULL THEN NULL
      WHEN sb.ref_month < DATE_TRUNC(sb.dt_primeiro_evento_pagamento_pos_aju, MONTH) THEN NULL
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju > sb.fechamento THEN NULL
      ELSE sb.dt_primeiro_evento_pagamento_pos_aju
    END AS dt_primeiro_evento_pagamento_pos_aju,

    CASE
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju IS NULL THEN NULL
      WHEN sb.ref_month < DATE_TRUNC(sb.dt_primeiro_evento_pagamento_pos_aju, MONTH) THEN NULL
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju > sb.fechamento THEN NULL
      ELSE GREATEST(0, DATE_DIFF(sb.dt_primeiro_evento_pagamento_pos_aju, sb.dt_inicio_ajuizamento, DAY))
    END AS DiasResolucaoAjuizamento,

    
    CASE
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju IS NOT NULL
       AND sb.ref_month >= DATE_TRUNC(sb.dt_primeiro_evento_pagamento_pos_aju, MONTH)
      THEN sb.QtDiaAtraso_base_propag
      WHEN sb.ref_month = DATE_TRUNC(CURRENT_DATE(), MONTH)
       AND sb.tipo_pag_final_exibicao = 'sem acao'
      THEN GREATEST(0, DATE_DIFF(CURRENT_DATE(), sb.dt_base_atraso_aju, DAY))
      ELSE sb.QtDiaAtraso_fechamento
    END AS QtDiaAtraso,

    CASE
      WHEN sb.dt_primeiro_evento_pagamento_pos_aju IS NOT NULL
       AND sb.ref_month >= DATE_TRUNC(sb.dt_primeiro_evento_pagamento_pos_aju, MONTH)
      THEN COALESCE(sa.SALDO_ANCHOR, sc.SALDO)
      ELSE sc.SALDO
    END AS SALDO,

    pp.PRODUTO

  FROM saida_base sb
  LEFT JOIN saldos_contabil sc
    ON sc.cd_contrato            = sb.cd_contrato
   AND sc.sq_contrato_financeiro = sb.sq_contrato_financeiro
   AND sc.dtprocessamento        = sb.fechamento_mes_anterior
  LEFT JOIN saldo_anchor sa
    ON sa.cd_contrato            = sb.cd_contrato
   AND sa.sq_contrato_financeiro = sb.sq_contrato_financeiro
  LEFT JOIN provisao_produto pp
    ON pp.nucontrato             = sb.cd_contrato
)
