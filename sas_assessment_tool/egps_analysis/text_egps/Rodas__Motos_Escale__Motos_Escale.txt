proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table negc_rodas_pdec_motos as
   select * from connection to perm
     (
		SELECT DISTINCT
			   dt_simulacao_proposta,
			   cd_simulacao_proposta	as _cd_simulacao_proposta,
			   ds_situacao_credito_pre  as  _ds_situacao_credito_pre
		FROM bv-saem-dtm.mis_varejo_rodas_emprestimos.pdec_motos	a
		WHERE a.dt_simulacao_proposta >= &dt_corte_gcp.

		);
   disconnect from perm;
quit;

data motos_parcial;
	retain dt_simulacao_proposta cd_simulacao_proposta;
	set negc_rodas_pdec_motos (where=(_cd_simulacao_proposta ne ""));

	format ds_situacao_credito_pre $20.;

	cd_simulacao_proposta = input(_cd_simulacao_proposta,20.);
	ds_situacao_credito_pre = _ds_situacao_credito_pre;

	drop _cd_simulacao_proposta _ds_situacao_credito_pre;

	anomes = compress(put(year(dt_simulacao_proposta),4.)||put(month(dt_simulacao_proposta),z2.));

run;

proc sort data=motos_parcial out=teste dupout=dup nodupkey;
	by cd_simulacao_proposta;
run;

proc append data=motos_parcial
			base=solitmp.motos_sit_credito force;

PROC SQL;

	CREATE TABLE escale_parcial AS
	SELECT
		A.CdSimulacao,
		A.NuAnoMes_Simu,
/*		a.DsProduto,*/
/*	 	B.CdSimulacaoInicial,*/
		C.NuProposta,
		C.NuContrato,
		A.DsMensageria,
		A.FlContratoC2c,
		COALESCE(C.Mensageria2,B.Mensageria2) AS Mensageria2,
		COALESCE(C.EntradaMinima,B.EntradaMinima) as EntradaMinima,
		COALESCE(C.NuAnoMes,B.NuAnoMes) AS NuAnoMes,
		COALESCE(C.Regional,B.Regional) AS Regional,
		COALESCE(C.Filial,B.Filial) AS Filial,
		COALESCE(C.FxIdade,B.FxIdade) as FxIdade,
		COALESCE(C.ProbabilidadeAprovacao,B.ProbabilidadeAprovacao ) as ProbabilidadeAprovacao,
		COALESCE(C.StatusSimulacao,B.StatusSimulacao) as StatusSimulacao,
		COALESCE(C.DtInclusaoGravame,B.DtInclusaoGravame) as DtInclusaoGravame format date9.,
		COALESCE(C.DtEntradaProposta,B.DtEntradaProposta) as DtEntradaProposta format date9.,
		COALESCE(C.FlConcorrente,B.FlConcorrente) as FlConcorrente,
		COALESCE(C.FlPagoLoja,B.FlPagoLoja) as FlPagoLoja,
		COALESCE(C.nucpfcnpj,B.nucpfcnpj) as nucpfcnpj,
		B.flcard ,
		COALESCE(C.DtSimulacaoInicial,B.DtSimulacaoInicial) AS DtSimulacaoInicial format date9.,
		COALESCE(C.FlPropostaOriginada,B.FlPropostaOriginada) AS FlPropostaOriginada,
		COALESCE(C.VrProposta,B.VrProposta) AS VrProposta,
		COALESCE(C.TxInicioFinanciamento,B.TxInicioFinanciamento) AS TxInicioFinanciamento,
		COALESCE(C.TxTirContabil,B.TxTirContabil) AS TxTirContabil,
		COALESCE(C.FlPropostaAprovada,B.FlPropostaAprovada) AS FlPropostaAprovada,
		COALESCE(C.FlPropostaEfetivada,B.FlPropostaEfetivada) AS FlPropostaEfetivada,
		COALESCE(C.DtContrato,B.DtContrato) AS DtContrato format date9.,
		COALESCE(C.Banco,B.Banco) AS Banco,
		COALESCE(C.Produto,B.Produto) AS Produto,
		COALESCE(C.TpPessoa,B.TpPessoa) AS TpPessoa,
		COALESCE(C.SistemaInicial,B.SistemaInicial) AS SistemaInicial,
		COALESCE(C.Canal,B.Canal) AS Canal ,
		COALESCE(C.FlSimuValidaDiaEfet,B.FlSimuValidaDiaEfet) AS FlSimuValidaDiaEfet,
		COALESCE(C.CdParceiroComercial,B.CdParceiroComercial) AS CdParceiroComercial,
		COALESCE(C.PcEntradaMinima,B.PcEntradaMinima) AS PcEntradaMinima,
		COALESCE(C.FlPap,B.FlPap) AS FlPap,
		COALESCE(C.CdTabelaFinanciamento,B.CdTabelaFinanciamento) AS CdTabelaFinanciamento,
		COALESCE(C.FxPrazo,B.FxPrazo) AS FxPrazo,
		COALESCE(C.PcVariacaoTabelaFinanciamento,B.PcVariacaoTabelaFinanciamento) AS PcVariacaoTabelaFinanciamento

	FROM escale AS A LEFT JOIN SIMU_ESCALE AS B
			  		 ON A.CdSimulacao=B.CdSimulacaoInicial
					 LEFT JOIN PROP_ESCALE AS C
					 ON A.NuProposta=C.NuProposta
	WHERE COALESCE(C.DtSimulacaoInicial,B.DtSimulacaoInicial) <= "&dt_atu."d;
quit;
