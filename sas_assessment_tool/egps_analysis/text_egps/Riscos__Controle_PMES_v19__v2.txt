	create table inad_&dt_ref. as
	select  input(substr(put(&dt_ref.,8.), 1,6),8.) as Ref
			,(YEAR(a.dt_ven)*100+MONTH(a.dt_ven)) as Safra_Venc
			,a.Tp_Portfolio
			,e.Região as Região_Final
			,a.Classificação
			,sum(a.vr_tit) as Saldo format commax22.
			,sum(case when b.atraso gt 15 then a.vr_tit else 0 end) as Inad15 format commax22.

			,sum(case when b.atraso gt 30 then a.vr_tit else 0 end) as Inad30 format commax22.

			%if &dt_ref60. ne . %then %do;
			,sum(case when c.atraso gt 60 then a.vr_tit else 0 end) as Inad60 format commax22.
			%end;

			%if &dt_ref90. ne . %then %do;
			,sum(case when d.atraso gt 90 then a.vr_tit else 0 end) as Inad90 format commax22.
			%end;

left join MONITRCA.BASE_PME_&dt_ref30. b on a.nr_nos_nr = b.nr_nos_nr

%if &dt_ref60. ne . %then %do;
left join MONITRCA.BASE_PME_&dt_ref60. c on a.nr_nos_nr = c.nr_nos_nr
%end;

%if &dt_ref90. ne . %then %do;
left join MONITRCA.BASE_PME_&dt_ref90. d on a.nr_nos_nr = d.nr_nos_nr
%end;

left join REGIAO_CNPJ 				   e on a.cnpj_cliente = e.cnpj_cliente

where a.Classificação not in ('aux_resumo') and a.Classificação not contains ('PEAC') and
		(YEAR(a.dt_ven)*100+MONTH(a.dt_ven)) in (&anomes.)
group by 1,2,3,4,5;

		,sum(Inad15)/sum(Saldo) as Percent_Inad15 format percent10.2

		,sum(Inad30)/sum(Saldo) as Percent_Inad30 format percent10.2

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60)/sum(Saldo) as Percent_Inad60 format percent10.2
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90)/sum(Saldo) as Percent_Inad90 format percent10.2
		%end;

from inad_&dt_ref.
group by 1,2,3,4,5
union all
select Ref
		,Safra_Venc
		,'' as Tp_Portfolio
		,'' as Região_Final
		,'Total' as Classificação
		,sum(saldo) as Saldo format commax22.
		,sum(Inad15) as Inad15 format commax22.
		,sum(Inad30) as Inad30 format commax22.

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60) as Inad60 format commax22.
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90) as Inad90 format commax22.
		%end;

		,sum(Inad15)/sum(saldo) as Percent_Inad15 format percent10.2

		,sum(Inad30)/sum(saldo) as Percent_Inad30 format percent10.2

		%if &dt_ref60. ne . %then %do;
		,sum(Inad60)/sum(saldo) as Percent_Inad60 format percent10.2
		%end;

		%if &dt_ref90. ne . %then %do;
		,sum(Inad90)/sum(saldo) as Percent_Inad90 format percent10.2
		%end;

from inad_&dt_ref.
group by 1,2,3,4,5;
QUIT;

proc sql;
create table calcula_atraso as
select *,
(case when missing(dt_lqd_aux) then vr_tit else 0 end) as carteira,
(case when missing(dt_lqd_aux) then max(intck('DAY', dt_ven,&dataref.),0) else 0 end) as dias_atraso
from filtra_escopo;
quit;

proc sql;
create table agrupa_contraparte as
select Classificacao,
ds_portfolio,
Região,

nr_ctr,
sum(vr_tit) as vr_tit,
sum(carteira) as carteira,
max(dias_atraso) as dias_atraso
from calcula_atraso
group by 1,2,3,4;

create table agrupa_contraparte as
select *,
(case when dias_atraso>15 then carteira else 0 end) as over15,
(case when dias_atraso>30 then carteira else 0 end) as over30,
(case when dias_atraso>60 then carteira else 0 end) as over60,
(case when dias_atraso>90 then carteira else 0 end) as over90
from agrupa_contraparte;
quit;
