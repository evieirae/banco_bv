PROC SQL;
CREATE TABLE RESULTADO_FINALCARTAO_PROJ AS
SELECT
	B.DTPROCESSAMENTO AS DATA_BASE,
'ATIVA' AS STATUS,
'DXC' AS BASE,
'CARTAO' AS PRODUTO ,
FlPagamento,
CASE
	WHEN B.ATRASO + 1 <= 0    THEN '01 - EM DIA'
	WHEN B.ATRASO + 1 <= 30   THEN '03 - 0001 A 0030'
	WHEN B.ATRASO + 1 <= 60   THEN '04 - 0031 A 0060'
	WHEN B.ATRASO + 1 <= 90   THEN '05 - 0061 A 0090'
	WHEN B.ATRASO + 1 <= 120  THEN '06 - 0091 A 0120'
	WHEN B.ATRASO + 1 <= 150  THEN '07 - 0121 A 0150'
	WHEN B.ATRASO + 1 <= 180  THEN '08 - 0151 A 0180'
	WHEN B.ATRASO + 1 <= 210  THEN '09 - 0181 A 0210'
	WHEN B.ATRASO + 1 <= 240  THEN '10 - 0211 A 0240'
	WHEN B.ATRASO + 1 <= 270  THEN '11 - 0241 A 0270'
	WHEN B.ATRASO + 1 <= 300  THEN '12 - 0271 A 0300'
	WHEN B.ATRASO + 1 <= 330  THEN '13 - 0301 A 0330'
	WHEN B.ATRASO + 1 <= 360  THEN '14 - 0331 A 0360'
	WHEN B.ATRASO + 1 > 360   THEN '15 - 0361 A 0540'
	ELSE 'EM DIA' END AS FX_ATRASO_k,
	'NAO' AS TIPO_ACORDO,
	COUNT(B.NUCPFCNPJ) AS QTD,
	SUM(B.SALDO) AS SALDO_CONTABIL
FROM TBORIGEM_HIST_2706_PAG B
GROUP BY
	DATA_BASE,
	STATUS,
	BASE,
	PRODUTO,
	FX_ATRASO_k,
	TIPO_ACORDO,
	FlPagamento;
QUIT;
RUN;
