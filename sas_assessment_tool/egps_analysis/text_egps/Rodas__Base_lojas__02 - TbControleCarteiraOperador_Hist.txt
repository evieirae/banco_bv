%include '/infogrc/utils/all-includes.sas';
LIBNAME BL '/infogrc/Base_Lojas';

%CRIAVARDB;
proc sql;
%CONECTADB(server=CTR);
create table TbControleCarteiraOperador_Hist as
select * from connection to dbcon
(

SELECT

a.CdColaborador AS CdColaboradorGF,
D.CdOperador AS CdOperadorGF,
D.CdRh AS CdRhGF,
D.NmColaborador AS NmColaboradorGF,

a.CdOperador AS CdColaboradorGR,
b.CdOperador AS CdOperadorGR,
B.CdRh AS CdRhGR,
B.NmColaborador AS NmColaboradorGR,

A.CdParceiroComercial,
C.NMFANTASIA AS NMPARCEIROCOMERCIAL,

a.DtInicioVigencia,
a.DtFimVigenciaEfetiva,
a.DtInclusao,
a.DtFimVigenciaPrevisao,
a.DtAlteracao,
a.DsControleCarteiraOperador

FROM DBCOR..TbControleCarteiraOperador A

LEFT JOIN DBCOR..TBCOLABORADOR B ON B.CdColaborador = A.CdOperador
LEFT JOIN DBCOR..TBPARCEIROCOMERCIAL C ON C.CdParceiroComercial = A.CdParceiroComercial
LEFT JOIN DBCOR..TBCOLABORADOR D ON D.CdColaborador = A.CdColaborador

ORDER BY a.CdParceiroComercial, A.DtInicioVigencia

)
;run;

/***************************************************************/
/*BASE HISTORICA COM TODAS AS ALTERAÇÕES NO ATENDIMENTO DA LOJA*/
/***************************************************************/

DATA BL.TbControleCarteiraOperador_Hist;
SET WORK.TbControleCarteiraOperador_Hist;
;RUN;

/*********************************/
/*BASE COM A CARTEIRA ATIVA ATUAL*/
/*********************************/

/*COMO TEMOS ALGUMAS LOJAS APARECENDO 2 VEZES OU + NA BASE POIS NÃO TEM DTFIMVIGENCIA*/
/*TEMOS QUE PEGAR O MAIOR REGISTRO NO CONTADOR DE LINHAS*/

DATA BASE_001;
SET BL.TbControleCarteiraOperador_Hist;
RETAIN QTD_PC;
IF FIRST.CdParceiroComercial THEN QTD_PC = 1;
ELSE QTD_PC = QTD_PC + 1;
BY CdParceiroComercial;
RUN;

PROC SQL;
CREATE TABLE MAX_REG_PC AS
SELECT
CdParceiroComercial,
MAX(QTD_PC) AS MAX_REG_PC

FROM WORK.BASE_001 A
GROUP BY 1
;RUN;

PROC SQL;
CREATE TABLE TbControleCarteiraOperador_Ativa AS
SELECT

A.*

FROM WORK.BASE_001 A

INNER JOIN WORK.MAX_REG_PC B ON
B.CDPARCEIROCOMERCIAL = A.CDPARCEIROCOMERCIAL
AND B.MAX_REG_PC = A.QTD_PC

;RUN;

DATA BL.TbControleCarteiraOperador_Ativa;
SET WORK.TbControleCarteiraOperador_Ativa;
WHERE DtFimVigenciaEfetiva IS null;
;RUN;

/*NÃO PODEMOS TEM QTD >= 2*/
PROC SQL;
CREATE TABLE VALIDACAO_LOJAS_ATIVAS AS
SELECT
CDPARCEIROCOMERCIAL,
COUNT(CDPARCEIROCOMERCIAL) AS QTD
FROM BL.TbControleCarteiraOperador_Ativa
GROUP BY 1
;RUN;
