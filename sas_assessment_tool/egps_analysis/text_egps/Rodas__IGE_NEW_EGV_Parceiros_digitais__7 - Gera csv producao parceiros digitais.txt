DATA _null_;
	SET SDD31.TBPARAMS;
	IF PARAM = 'MAILINGX' THEN CALL SYMPUT('MAILINGX',compress(VALOR));
	IF PARAM = 'MAILINGF' THEN CALL SYMPUT('MAILINGF',compress(VALOR));

	IF PARAM = 'DESENV1' THEN CALL SYMPUT('DESENV',"'"||compress(VALOR)||"'");
	IF PARAM = 'UC4_1' THEN CALL SYMPUT('UC4',"'"||compress(VALOR)||"'");
	IF PARAM = 'MAILING1' THEN CALL SYMPUT('MAILING1',compress(VALOR));
	IF PARAM = 'MAILING2' THEN CALL SYMPUT('MAILING2',compress(VALOR));
	IF COMPRESS(PUT(DATE(),DOWNAME.)) = 'Monday' THEN CALL SYMPUT('BVNET',"'S'");
;RUN;

OPTIONS COMPRESS=BINARY;

%LET TIMESTAMP = %SYSFUNC(DATE(),YYMMDDN8.);

/*Define os 3 últimos meses a partir da data de ontem*/
proc sql noprint;
SELECT intnx('month',DtTempo,-3) into: DtBase
FROM BASES.TbTempoData
WHERE StData = 1
;quit;

PROC SQL;
CREATE TABLE TbPropostas AS
SELECT
	Originacao.NuAnoMes
	,Originacao.DtEntradaProposta format ddmmyy10.
	,Originacao.NuProposta
	,Originacao.StProposta
	,Producao.NuContrato
	,Producao.DtContrato
	,Originacao.NmTipoCobranca
	,Originacao.CdParceiroComercial
	,Originacao.NuCnpjLoja
	,Originacao.NmParceiroComercial
	,Originacao.NmTipoParceiroComercial
	,case when Producao.VrCliente < Producao.VrProducao then "S" else "N" END AS Desdobro
    ,Originacao.NmPessoa
	,Originacao.TpPessoa
	,Originacao.CdUfPessoa
	,Originacao.NmCidadePessoa
    ,Originacao.NuCpfCnpj
	,Originacao.DsTipoCanalProducao
	,Originacao.NmCanalEstruturaComercial
	,Originacao.NmAreaEstruturaComercial
	,Originacao.NmSubEstruturaComercialA
	,GrupoProdutoMeta.NmGrupoSubProduto
	,Originacao.QtAnosIdade
	,Originacao.FxIdade
	,Originacao.QtParcelas
	,Originacao.CdMarca
	,Originacao.CdModelo
	,Originacao.AaModelo
	,Originacao.AaFabricacao
	,Originacao.VrPropostaOriginada format commax18.2
	,Originacao.VrPropostaAprovada format commax18.2
	,Originacao.VrLiberado format commax18.2
	,Originacao.VrEntrada format commax18.2
	,Originacao.VrFinanciamento format commax18.2
	,Originacao.VrPrestacao format commax18.2
	,Producao.VrProducao format commax18.2
	,Producao.VrCliente format commax18.2
	,Producao.VrComissaoTotal format commax18.2
	,Originacao.QtCorridosDiaUtilMes
	,Originacao.DtAtualizacao
	,Originacao.TsProcessamento
	,originacao.cdColaborador
	,originacao.nmColaborador

FROM BASES.TbOriginacao Originacao
	left JOIN BASES.TBEXTPROPOSTA PRO
	ON ORIGINACAO.NUPROPOSTA = PRO.NUPROPOSTA

	LEFT JOIN BASES.TbProducao Producao
		ON Originacao.NuProposta = Producao.NuProposta

	LEFT JOIN SDD31.TbSubProduto SubProduto
		ON SubProduto.CdSubProduto = PRO.CdSubProduto

	left join sdd31.TbGrupoProdutoMeta GrupoProdutoMeta
		on  GrupoProdutoMeta.CdGrupoSubProduto = (SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0))

/*INCLUIMOS O CNPJ DA TATOS, FRANC E SERASA*/
WHERE Originacao.DtEntradaProposta >= &DtBase.
and Originacao.NuCnpjLoja in ('21738217000173','62173620010495','32709829000154','14439116000136','23722194000134','34697707000110','31392500000140','34656383000172','28812324000143','41149615000161','18249116000124','44391578000191','32778350000170','14110585000107')
and GrupoProdutoMeta.NmGrupoSubProduto in ('CVG LEVES','CVG PESADOS','LEVES','MOTOS','PESADOS')
and Originacao.StProposta in ('A','AP','CA','PA','N')
;QUIT;

/* FINANZERO */

DATA TbFinanzero;
	set TbPropostas (where=(NuCnpjLoja = '23722194000134'));
run;

PROC EXPORT
	DATA= TbFinanzero
	FILE="&dir_csv./EGV_DLV_Finanzero_Propostas_&timestamp..csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;
