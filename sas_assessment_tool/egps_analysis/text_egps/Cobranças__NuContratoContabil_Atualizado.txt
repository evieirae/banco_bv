create table PARCELAMENTO_FATURA as
    select distinct
	input(cdcontacartao,best22.) as cdcontacartao
	,dtadesao
	,input(cdtipoparcelado,best22.) as cdtipoparcelado

    from connection to perm (

	select distinct
	cdcontacartao
	,dtadesao
	,cdtipoparcelado

	from bv-pdbd-prd.trusted_dwbi_dtwh.tbcontratoparcelamentofatura
	where dtadesao >= '2018-12-01'
	);

disconnect from perm;
quit;

DATA PARCELAMENTO_FATURA;

ELSE IF cdtipoparcelado = 7 THEN TP = "E";

DATA_EFET = dtadesao;

RUN;

PROC SQL;
CREATE TABLE BASE_PARCELADOS_B AS SELECT
A.*, B.CPF FROM
BASE_PARCELADOS A
LEFT JOIN
CARNEIRO.INFO_CONTA B
ON
1E9*A.PRODUTO+A.NUCONTA = B.CDDIMCONTACARTAO;
QUIT;

PROC SQL;
CREATE TABLE BASE_2_PARC_B AS SELECT
*
FROM
(SELECT
A.*,
B.NUCONTA AS NUCONTA_PARC,
B.DATA_EFET,
B.TP

FROM BASE_2_PARC A
LEFT JOIN
BASE_PARCELADOS_B B

ON A.CPF = B.CPF AND A.PRODUTO = B.PRODUTO
AND
SUBSTR(A.NUCONTRATO,4,1) = B.TP)
WHERE DATA_EFET > 0
;
QUIT;

PROC SQL;
CREATE TABLE BASE_2_PARC_C AS SELECT
A.*, B.contador
FROM
(SELECT DISTINCT
nucontrato,	cpf,	NUCONTA_PARC AS nuconta,	produto, MIN(min_ref) AS min_ref format date10., MAX(max_ref) AS max_ref format date10.
FROM
BASE_2_PARC_B
GROUP BY nucontrato,cpf,nuconta_parc,produto) A
LEFT JOIN
(SELECT nucontrato, COUNT(*) AS contador from
(SELECT DISTINCT
nucontrato,	cpf,	NUCONTA_PARC AS nuconta,	produto, MIN(min_ref) AS min_ref, MAX(max_ref) AS max_ref
FROM
BASE_2_PARC_B
GROUP BY nucontrato,cpf,nuconta_parc,produto)
GROUP BY NUCONTRATO) B
ON A.nucontrato = b.nucontrato
;
QUIT;



DATA BASE_2_PARC_UNICO;
SET BASE_2_PARC_C;
WHERE CONTADOR = 1;
RUN;
