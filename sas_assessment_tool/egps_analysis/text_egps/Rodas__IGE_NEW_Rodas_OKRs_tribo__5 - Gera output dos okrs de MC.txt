%let nuanomes = (YEAR(TODAY())-1)*100+1;

/* ======================================================================================*/
/* =====================Gera output dos okrs de MC=========================*/
/* ======================================================================================*/

PROC SQL;
   CREATE TABLE MC_EXECUTIVA AS
   SELECT
   		DtAtualizacaoMCExecutiva,
   		Anomes as Anomes1,
		Year(Anomes)*100+day(Anomes) as Anomes,
          'Producao_Leves'n,
          'Producao_Motos'n,
          'Producao_Pesados'n,
          'Producao_EGV'n,
          Margem_Leves,
          Margem_Motos,
          Margem_Pesados,
          Margem_EGV
      FROM WORK.CONSOLIDADO_MARGEM_EXECUTIVA;
QUIT;

%let dir_csv = /riscosbvf/BA/processos/output;

PROC EXPORT
	DATA= MC_EXECUTIVA
	FILE="&dir_csv./Rodas_BaseOKR_MC_EXECUTIVA.csv" REPLACE
	DBMS=DLM;
	DELIMITER=';';
RUN;
