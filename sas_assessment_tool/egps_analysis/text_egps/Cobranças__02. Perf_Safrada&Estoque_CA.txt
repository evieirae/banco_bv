CREATE TABLE ATRASOS_CA AS
SELECT DISTINCT
A.*

,CASE WHEN E.DT_ENQUAD NE . THEN 1 ELSE 0 END AS FL_ENQUADRADO
,E.DT_ENQUAD AS DT_ENQUAD_CL

FROM SATIE.CARTEIRA_CA_PISMO AS A
LEFT JOIN SATIE.ENQUADRADOS_PISMO AS E ON A.NUCONTA = E.NUCONTA AND A.PRODUTO = E.PRODUTO AND A.DT_INICIO_ATRASO = E.DT_INICIO_ATRASO AND A.DT_RETIRADA = E.DT_ENVIO
;QUIT;

DATA ATRASOS_CA_A (DROP=DT_ENQUAD);
SET ATRASOS_CA;
FORMAT DT_REG DATE9.;

WHERE DT_INICIO_ATRASO >= MDY(10,01,2024);

IF DT_RETIRADA - DT_INICIO_ATRASO >= 6 OR (DT_RETIRADA = . AND &ONTEM. - DT_INICIO_ATRASO >=5) THEN FLAG_5 = 1; ELSE FLAG_5 = 0; 

IF DT_ENQUAD_CL = . OR DT_ENQUAD_CL > DT_RETIRADA + 5 THEN DT_REG = DT_RETIRADA;
ELSE IF DT_ENQUAD_CL >= DT_RETIRADA - 5 AND DT_ENQUAD_CL <= DT_RETIRADA + 5 THEN DT_REG = .;

IF DT_ENQUAD_CL = . OR DT_ENQUAD_CL > DT_RETIRADA + 5 THEN DO;
	DIAS_REG = DT_RETIRADA - DT_INICIO_ATRASO;
	SAFRA_RETIRADA = 100*YEAR(DT_RETIRADA)+MONTH(DT_RETIRADA); END;
ELSE DO; DIAS_REG = .;
		 SAFRA_RETIRADA = .;END; 

VCTO = DAY(DT_INICIO_ATRASO);
SAFRA_ATR = 100*YEAR(DT_INICIO_ATRASO)+MONTH(DT_INICIO_ATRASO);
SAFRA_ENVIO = 100*YEAR(DT_ENVIO)+MONTH(DT_ENVIO);
ATRASO_ENVIO = DT_ENVIO - DT_INICIO_ATRASO;
IF DT_RETIRADA = . THEN ATRASO_ATUAL = &ONTEM. - DT_INICIO_ATRASO; ELSE ATRASO_ATUAL = .;
