PROC SQL;
CREATE TABLE TbTempoData_DU AS
SELECT
NUANOMES,

max(DtTempo) as DtTempo format ddmmyy10.,
MAX(QtCorridosDiaUtilMes) AS Max_QtCorridosDiaUtilMes,
MAX(QtTotalDiaUtilMes) AS Max_QtTotalDiaUtilMes,

MAX(QtCorridosDiaUtilMesSab) AS Max_QtCorridosDiaUtilMesSab format commax20.1,
MAX(QtTotalDiaUtilMesSab) AS Max_QtTotalDiaUtilMesSab format commax20.1

FROM BASES.TBTEMPODATA
/*WHERE StData = 1*/
WHERE NUANOMES BETWEEN 202509 AND 202512
group by 1
;RUN;

DATA CMBV.TbTempoData_DU;
SET WORK.TbTempoData_DU;
;RUN;
