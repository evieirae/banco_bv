PROC SQL;
CREATE TABLE BASE_PDEC AS
SELECT
	A.*
	,FX_ATRASO_M0
	,B.PILOTO
	,B.aleat
	,B.SALDO
	,A.TxJuros_Contrato as TxContrato
	,TxJuros_Contrato*0.7 as Tx_Juros_Min
	,TxJuros_Contrato - 0.2 as TxJuros_Acao

	,CASE WHEN CALCULATED TxJuros_Acao < CALCULATED Tx_Juros_Min THEN CALCULATED Tx_Juros_Min
	ELSE CALCULATED TxJuros_Acao
	END AS TxJuros_Acao_Final

	,Filtro_P0 as Filtro_Acao

	,CASE WHEN CALCULATED TxJuros_Acao_Final <> CALCULATED TxJuros_Acao 			THEN "Filtro"
	WHEN TpPessoa = "J" 															THEN "Filtro"
	WHEN Piloto = '1. Padrão' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat = 0 				THEN "Controle"
	WHEN Piloto = '1. Padrão' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat > 0 				THEN "Teste"




	ELSE "Filtro"
	END AS Teste_Controle_Acao,

	CASE WHEN PILOTO = '2. Flex_2_meses' THEN 1
	ELSE 0
	END AS FL_FLEX_2_MESES,

	CASE WHEN PILOTO = '3. Flex_Pag_Parcial' THEN 1
	ELSE 0
	END AS FL_FLEX_PAG_PARCIAL

FROM &Base_PDEC. A

LEFT JOIN BASE_RECORRENTE B
ON INPUT(A.NuContrato,best32.) = B.NuContrato_Num

WHERE PILOTO in ('1. Padrão','2. Flex_2_meses', '3. Flex_Pag_Parcial')

CREATE TABLE BASE_PDEC AS
SELECT
	A.*
	,B.PILOTO, B.aleat
	,A.TxJuros_Contrato as TxContrato
	,TxJuros_Contrato*0.7 as Tx_Juros_Min
	,TxJuros_Contrato - 0.2 as TxJuros_Acao

	,CASE WHEN CALCULATED TxJuros_Acao < CALCULATED Tx_Juros_Min THEN CALCULATED Tx_Juros_Min
	ELSE CALCULATED TxJuros_Acao
	END AS TxJuros_Acao_Final

	,Filtro_P0 as Filtro_Acao

	,CASE WHEN CALCULATED TxJuros_Acao_Final <> CALCULATED TxJuros_Acao 			THEN "Filtro"
	WHEN TpPessoa = "J" 															THEN "Filtro"
	WHEN Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat = 0 	THEN "Controle"
	WHEN Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat > 0 	THEN "Teste"

PROC SQL;
CREATE TABLE BASE_PDEC AS
SELECT
	A.*
	,FX_ATRASO_M0
	,B.PILOTO
	,B.aleat
	,B.SALDO
	,A.TxJuros_Contrato as TxContrato
	,TxJuros_Contrato*0.7 as Tx_Juros_Min
	,TxJuros_Contrato - 0.2 as TxJuros_Acao

	,CASE WHEN CALCULATED TxJuros_Acao < CALCULATED Tx_Juros_Min THEN CALCULATED Tx_Juros_Min
	ELSE CALCULATED TxJuros_Acao
	END AS TxJuros_Acao_Final

	,Filtro_P0 as Filtro_Acao

	,CASE WHEN CALCULATED TxJuros_Acao_Final <> CALCULATED TxJuros_Acao 			THEN "Filtro"
	WHEN TpPessoa = "J" 															THEN "Filtro"
	WHEN Piloto = '1. Padrão' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat = 0 	THEN "Controle"
	WHEN Piloto = '1. Padrão' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat > 0 	THEN "Teste"
	WHEN Piloto = '2. Flex' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat <= 3 	THEN "Controle"
	WHEN Piloto = '2. Flex' AND Filtro_P0 IN ("99. Sem Filtro", "9. Fora do corte pol") AND B.aleat >  3 	THEN "Teste"

	ELSE "Filtro"
	END AS Teste_Controle_Acao,

	CASE WHEN PILOTO = '2. Flex' THEN 1
	ELSE 0
	END AS FL_FLEXIBILIZACAO_&SAFRA.

FROM &Base_PDEC. A

LEFT JOIN BASE_RECORRENTE B
ON INPUT(A.NuContrato,best32.) = B.NuContrato_Num

WHERE PILOTO in ('1. Padrão','2. Flex')

;QUIT;

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=15);
CREATE TABLE GCON_M0 AS SELECT
	M0.*,
	M3.atraso AS QtDiaAtraso_m3,
	M2.atraso AS QtDiaAtraso_m2,
	M1.atraso AS QtDiaAtraso_m1,
	M0.atraso AS QtDiaAtraso_m0,
	CASE
		WHEN M0.atraso <= 0 THEN '0000-0000'
		WHEN M0.atraso <= 30 THEN '0001-0030'
		WHEN M0.atraso <= 60 THEN '0031-0060'
		WHEN M0.atraso <= 90 THEN '0061-0090'
		WHEN M0.atraso  > 90 THEN '0091-0720'
	END AS FX_ATRASO_M0

FROM CONNECTION TO PERM (

		SELECT
			A.DTPROCESSAMENTO AS DATA_BASE,
			A.NUCONTRATO,
			C.cd_cpf_cnpj,
			C.dt_contrato,
			A.SQCONTRATOFINANCEIRO,

			CASE
				WHEN A.CdSItuacaoContabil = '1' THEN 'ATIVA'
				WHEN A.CdSItuacaoContabil = '2' THEN 'WO'
				ELSE ""
			END AS STATUS,

			'GCON' AS BASE,
			C.cd_produto,
			D.cdsubproduto,
			C.cd_modalidade_produto,
			I.nmmodalidadeproduto,
			H.NMGRUPOSUBPRODUTO,
			J.FlRestruturacaoContrato,

		CASE
			WHEN C.cd_produto = '14' 									AND H.CDGRUPOSUBPRODUTO IN (4, 7)                    THEN 'LEVES'
			WHEN C.cd_produto = '14' 									AND H.CDGRUPOSUBPRODUTO IS NULL                      THEN 'LEVES'
			WHEN C.cd_produto = '14' 									AND H.CDGRUPOSUBPRODUTO = 6                          THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto = 8                               THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto = 13                              THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '27' AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '27' AND D.cdsubproduto IN (19,60)                        THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '28' AND D.cdsubproduto IN (20,63)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '78' AND D.cdsubproduto IN (13,20)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '79' AND D.cdsubproduto IN (8,19,60)                      THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '79' AND D.cdsubproduto IN (13,20)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '2'  AND D.cdsubproduto = 9                               THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto = 16                              THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto = 30                              THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '5'                                                       THEN 'MOTOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto IS NULL                           THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto = 4                               THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto = 3                               THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '8'  AND D.cdsubproduto IS NULL                           THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '8'  AND D.cdsubproduto IN (1,58)                         THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '9' 	AND D.cdsubproduto IS NULL                           THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '9' 	AND D.cdsubproduto IN ( 2,3,61)                      THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '38' AND D.cdsubproduto = 29                              THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '38' AND D.cdsubproduto = 31                              THEN 'PESADOS'
		END AS PRODUTO,
			MAX(A.NuDiasAtraso) AS ATRASO,

		CASE
			WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
			ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
		END AS SALDO

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato C
		ON A.NUCONTRATO = C.cd_contrato

		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS D
		ON A.NUCONTRATO = D.NUCONTRATOGESTAO AND D.STPROPOSTA='AP'

		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS G
		ON D.CDSUBPRODUTO = G.CDSUBPRODUTO

		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbgruposubproduto AS H
		ON G.CDGRUPOSUBPRODUTO =H.CDGRUPOSUBPRODUTO

		LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbmodalidadeproduto AS I
		ON A.cdmodalidadeproduto=I.CDMODALIDADEPRODUTO

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE
		DATE(A.DTPROCESSAMENTO) = &fim_mes_m0.
		AND A.CDCARTEIRA IN (1,3,4,5,12)
		AND A.CdSItuacaoContabil = "1"
		AND A.NUDIASATRASO BETWEEN 31 AND 90

		GROUP BY
		DATA_BASE,
		A.NUCONTRATO,
		C.cd_cpf_cnpj,
		C.dt_contrato,
		A.SQCONTRATOFINANCEIRO,
		STATUS,
		BASE,
		C.cd_produto,
		D.cdsubproduto,
		C.cd_modalidade_produto,
		I.nmmodalidadeproduto,
		H.nmgruposubproduto,
		PRODUTO,
		J.FlRestruturacaoContrato

) AS M0

LEFT JOIN CONNECTION TO PERM (
	SELECT
		NUCONTRATO,
		SQCONTRATOFINANCEIRO,

	M0.*,

	M2.atraso AS QtDiaAtraso_m2,
	M1.atraso AS QtDiaAtraso_m1,

		WHEN M0.atraso <= 30 THEN '0001-0030'
		WHEN M0.atraso <= 60 THEN '0031-0060'
		WHEN M0.atraso <= 90 THEN '0061-0090'
		WHEN M0.atraso  > 90 THEN '0091-0720'
	END AS FX_ATRASO_M0

FROM CONNECTION TO PERM (

			A.DTPROCESSAMENTO AS DATA_BASE,

			C.cd_cpf_cnpj,

			WHEN C.cd_produto = '14' 									AND H.CDGRUPOSUBPRODUTO = 6                          THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto = 8                               THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '6'  AND D.cdsubproduto = 13                              THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '27' AND D.cdsubproduto IS NULL                           THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '27' AND D.cdsubproduto IN (19,60)                        THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '28' AND D.cdsubproduto IN (20,63)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '78' AND D.cdsubproduto IN (13,20)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '79' AND D.cdsubproduto IN (8,19,60)                      THEN 'CVG_LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '79' AND D.cdsubproduto IN (13,20)                        THEN 'CVG_PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '2'  AND D.cdsubproduto = 9                               THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto = 16                              THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '3'  AND D.cdsubproduto = 30                              THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '5'                                                       THEN 'MOTOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto IS NULL                           THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto = 4                               THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '7'  AND D.cdsubproduto = 3                               THEN 'PESADOS'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '8'  AND D.cdsubproduto IS NULL                           THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '8'  AND D.cdsubproduto IN (1,58)                         THEN 'LEVES'
			WHEN C.cd_produto = '12' AND C.cd_modalidade_produto = '9' 	AND D.cdsubproduto IS NULL                           THEN 'PESADOS'

		LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbmodalidadeproduto AS I
		ON A.cdmodalidadeproduto=I.CDMODALIDADEPRODUTO

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE
		DATE(A.DTPROCESSAMENTO) = &fim_mes_m0.
		AND A.CDCARTEIRA IN (1,3,4,5,12)
		AND A.CdSItuacaoContabil = "1"
		AND A.NUDIASATRASO BETWEEN 31 AND 90

		GROUP BY
		DATA_BASE,
		A.NUCONTRATO,
		C.cd_cpf_cnpj,
		C.dt_contrato,
		A.SQCONTRATOFINANCEIRO,
		STATUS,
		BASE,
		C.cd_produto,
		D.cdsubproduto,
		C.cd_modalidade_produto,
		I.nmmodalidadeproduto,
		H.nmgruposubproduto,

PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage MAX_CHAR_LEN=15);
CREATE TABLE GCON_V2 AS SELECT
	a.*,
	B.atraso_atual,
	C.dt_ultimo_recebimento,
	C.fl_pagou,
	CASE
		WHEN G.Parcelas_pagas_ate_30d <= 2  THEN '1. 0 a 2'
		WHEN G.Parcelas_pagas_ate_30d <= 4  THEN '2. 3 a 4'
		WHEN G.Parcelas_pagas_ate_30d <= 7  THEN '3. 5 a 7'
		WHEN G.Parcelas_pagas_ate_30d <= 14 THEN '4. 8 a 14'
		WHEN G.Parcelas_pagas_ate_30d >  14 THEN '5. 15+'
	END AS GROUP_Parc_Pag_30d

FROM GCON_V1 as a

LEFT JOIN CONNECTION TO PERM (
	SELECT
		NUCONTRATO,
		MAX(NUDIASATRASO) as atraso_atual
	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
	WHERE DTPROCESSAMENTO = &ontem.
	GROUP BY NUCONTRATO
) AS B
ON a.nucontrato = b.nucontrato

LEFT JOIN CONNECTION TO PERM (
	SELECT
		cd_Contrato,
		st_parcela,
		MAX(sq_contrato_financeiro) AS SQCONTRATOFINANCEIRO,
		MAX(DATE(dt_ultimo_recebimento)) AS dt_ultimo_recebimento,
		CASE
			WHEN MAX(DATE(dt_ultimo_recebimento)) >= &inic_mes. and MAX(DATE(dt_ultimo_recebimento)) < &fim_mes. THEN 1
			ELSE 0
		END AS fl_pagou
	FROM bv-pdbd-prd.dvry_core.tbparcela
	WHERE st_parcela = 'Q'
	GROUP BY 1,2
) as C
ON A.nucontrato = C.cd_Contrato

LEFT JOIN CONNECTION TO PERM(
SELECT DISTINCT cd_contrato,
             	sq_contrato_financeiro,
				SUM(CASE WHEN st_parcela='Q' AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END)        AS Parcelas_Pagas,
                SUM(CASE WHEN st_parcela='A' OR (st_parcela='Q' AND dt_ultimo_recebimento > &fim_mes_m0.) THEN 1 ELSE 0 END)        AS Parcelas_Abertas,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) < 30 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_30d,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) BETWEEN 1 AND 90 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_90d,

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G
ON A.NuContrato = G.cd_contrato AND A.SqContratoFinanceiro = G.sq_contrato_financeiro

;QUIT;





DATA GCON_V3;
SET GCON_V2;
FORMAT PILOTO $20.;
IF atraso_atual >= DAY(TODAY()) and (RECORRENTE_M2 IN (1) OR RECORRENTE_M1 IN (1)) AND fl_pagou = 0 THEN PILOTO = '4. Flex_s_pagamento_atual';
