   on isnull(PO.CdCnsldrEstruturaProduto,-1) = pr.CdCnsldrEstruturaProduto
   left outer join POSICAOCONSOLIDADA.TbEstruturaGerencial as ger
   on po.CdCnsldrEstruturaGerencial = ger.CdCnsldrEstruturaGerencial
   left outer join PosicaoConsolidadaVarejo.TbPosicaoContabil as po1
   on po.NuObjetoNegociacao = po1.NuObjetoNegociacao
   and po.CdSeqFinanceira = po1.CdSeqFinanceira
   and po.TpContabil = po1.TpContabil
   and po.CdCarteira = po1.CdCarteira
   and po1.DtReferencia = @DtFechamentoAnt 
   where po.DtReferencia = @DtReferencia 
   and po.cdcarteira not in (13)
   
   group by po.DtReferencia,
   po.TpContabil,
   po.cdcarteira,
   ca.NmCarteira,
   po.CdVeiculoLegal,
   CdModalidadeCatalogoProduto,
   ProdutoTerceiraGeracao,
   ger.NmMesaOperacao,
   ger.NmCarteiraGerencial,
   ger.cdbook,
   ger.NmBook,
   po.CdSituacaoPosicao,
   ISNULL(po.FlHedgeAccount,C.FlHedgeAccount)
   order BY po.CdVeiculoLegal, ProdutoTerceiraGeracao, Book
	);

Quit;

   VrObjetoSpread=SUM(isnull(po.VrObjetoSpread,0)),
   VrObjetoAccrualSpread=SUM(isnull(po.VrObjetoAccrualSpread,0)),

   VrObjetoResultadoAjustado=SUM(isnull(po.VrObjetoResultadoAjustado,0)),
   VrAjusteMtmPdd=SUM(isnull(po.VrAjusteMtmPdd,0)),

   VrTIRAjustadoPDD = SUM(isnull(po.VrTIRAjustadoPDD,0)),
   PcRatingAtual  = ROUND(CASE WHEN SUM(isnull(po.VrResgateTIRComercialInicial,0)) = 0 OR SUM(isnull(po.VrTIRAjustadoPDD,0)) = 0  THEN 0 ELSE 1.0 - (SUM(isnull(po.VrTIRAjustadoPDD,0)) / SUM(isnull(po.VrResgateTIRComercialInicial,0))) END,4),
   PcRatingAnterior = ROUND(CASE WHEN SUM(isnull(po.VrResgateTIRComercialInicial,0)) = 0 OR SUM(isnull(po.VrTIRAjustadoPDDM1,0)) = 0    THEN 0 ELSE 1.0 - (SUM(isnull(po.VrTIRAjustadoPDDM1,0)) / SUM(isnull(po.VrResgateTIRComercialInicial,0))) END,4),
   PrazoMedio = ROUND(CASE WHEN SUM(isnull(po.VrPosAccrualTxContratoFluxo,0)) = 0 OR po.cdcarteira = 5 THEN 0 ELSE SUM(isnull(po.VrPosAccrualTxContratoFluxo,0) * DATEDIFF(dd, ISNULL(c.DtInicioContrato,po.DtReferencia) , po.DtReferencia )) / SUM(isnull(po.VrPosAccrualTxContratoFluxo,0)) END, 2),

   VrMtmAditamento = SUM(isnull(po.VrMtmAditamento,0)),

   VrResLiqAntTIRRenegAprop = SUM(isnull(po.VrRestTIRLqudoAntcpaRngcoAprpr,0)),
   VrResLiqAntTIRRenegMes = SUM(isnull(po.VrResTIRLiquidacoAntcpdaRngMes,0)),
   VrResLiqAnCCXReneg = SUM(isnull(po.VrRsltdoCCXLqudcoAntcpdaRngcco,0)),
   VrResLiqAnCCXRenegAprop = SUM(isnull(po.VrRsltoCCXLqudoAntcaRngcoAprpr,0)),
   VrResLiqAnCCXRenegMes = SUM(isnull(po.VrRsltdoCCXLqudcoAntcpdaRngMes,0)),
   VRAcionistaMTM =SUM(isnull(po.VrAjusteMTMGer,0)) * @CDI_DIA
   From PosicaoConsolidadaVarejo.TbPosicaoContabil as po
   Left outer join PosicaoConsolidadaVarejo.TbContratoResumido as c
   on po.NuObjetoNegociacao = c.NuObjetoNegociacao
   and po.CdSeqFinanceira = c.CdSeqFinanceira
   and po.TpContabil = c.TpContabil
   left outer join gestaocontabil.TbCarteira as ca
   on po.cdcarteira = ca.cdcarteira
   left outer join posicaoconsolidada.TbEstruturaProduto as pr
   on isnull(PO.CdCnsldrEstruturaProduto,-1) = pr.CdCnsldrEstruturaProduto
   left outer join POSICAOCONSOLIDADA.TbEstruturaGerencial as ger
   on po.CdCnsldrEstruturaGerencial = ger.CdCnsldrEstruturaGerencial
   left outer join PosicaoConsolidadaVarejo.TbPosicaoContabil as po1
   on po.NuObjetoNegociacao = po1.NuObjetoNegociacao
   and po.CdSeqFinanceira = po1.CdSeqFinanceira
   and po.TpContabil = po1.TpContabil
   and po.CdCarteira = po1.CdCarteira
   and po1.DtReferencia = @DtFechamentoAnt 
   where po.DtReferencia = @DtReferencia 
   and po.cdcarteira not in (13)
   
   group by po.DtReferencia,
   po.TpContabil,
   po.cdcarteira,
   ca.NmCarteira,
   po.CdVeiculoLegal,

   ProdutoTerceiraGeracao,
   ger.NmMesaOperacao,
   ger.NmCarteiraGerencial,
   ger.cdbook,
   ger.NmBook,
   po.CdSituacaoPosicao,
   ISNULL(po.FlHedgeAccount,C.FlHedgeAccount)
   order BY po.CdVeiculoLegal, ProdutoTerceiraGeracao, Book
	);

PROC SQL;
   CREATE TABLE WORK.VAREJO_BPOS as
   select v1.*,
   coalesce(v2.VrResLiqAntTIRRenegAprop,0)- v1.VrResLiqAntTIRRenegAprop as DifResLiqAntTIRRenegAprop,
   v1.VrResLiqAntTIRRenegMes - coalesce(v2.VrResLiqAntTIRRenegMes,0) as DifResLiqAntTIRRenegMes,
   v1.VrResLiqAnCCXReneg - coalesce(v2.VrResLiqAnCCXReneg,0) as DifResLiqAnCCXReneg,
   v1.VrResLiqAnCCXRenegAprop - coalesce(v2.VrResLiqAnCCXRenegAprop,0) as DifResLiqAnCCXRenegAprop,
   v1.VrResLiqAnCCXRenegMes - coalesce(v2.VrResLiqAnCCXRenegMes,0) as DifResLiqAnCCXRenegMes,
   v1.VrMtmAditamentoAprop - coalesce(v2.VrMtmAditamentoAprop,0) as DifMtmAditamentoApropriar
   FROM WORK.VAREJO_BPOS as v1
   LEFT JOIN PLANCTRL.VAREJO_BPOS as v2
   ON v2.TpContabil= v1.TpContabil
   and v2.cdcarteira = v1.cdcarteira
   AND v2.CarteiraContabil= v1.CarteiraContabil
   AND v2.CodVeiculoLegal = v1.CodVeiculoLegal
   AND v2.CdModalidadeCatalogoProduto = v1.CdModalidadeCatalogoProduto
   AND v2.Mesa = v1.Mesa
   AND v2.Carteira = v1.Carteira
   and v2.cdbook = v1.cdbook
   AND v2.SituacaoPosicao = v1.SituacaoPosicao
   AND v2.Marcado = v1.Marcado
   AND v2.DtReferencia = select dtreferencia from DtAnterior
   WHERE v1.DtReferencia = select dtreferencia from DtAtual
    ;

   UPDATE WORK.VAREJO_BPOS
   SET DifResLiqAntTIRRenegMes = VrResLiqAntTIRRenegMes,
       DifResLiqAnCCXRenegMes = VrResLiqAnCCXRenegMes,
	   DifMtmAditamentoApropriar = 0
   WHERE DtReferencia = &PrimeiroDiaUtilMes.;

Quit;
