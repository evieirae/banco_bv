proc sql;create table producao as
select
	nuanomes,
	nuproposta,
	dtentradaproposta,
	nmgrupoprodutometa,
	nmareaestruturacomercial,
	nmcanalestruturacomercial,
	nmsubestruturacomercialA,
	nmcolaborador,
	catx(" - ",cdparceirocomercial,nmparceirocomercial) as nmparceirocomercial,
	dsloginn1c,
	dsloginn2c,
	dsloginn3c,
	nucontrato,
	dtcontrato,
	vrproducao
from bases.tbproducao
where dtcontrato >= "&dtmerge."d
	and nmgrupoprodutometa in ("Leves", "Leves-BVO", "Motos", "Pesados", "CVG Leves", "CVG Pesados")
;run;

proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage);
create table tbtipomodelo_proposta as
select
	*
	,case when cdtipomodelo = 1 then 'Web'
		  	when cdtipomodelo = 10 then 'App'
		  		when cdtipomodelo in (14,15)then 'Física'
		  			else 'nd' end as tpassinatura
from connection to perm(
	select
		 cast(NUPROPOSTA as numeric) as nuproposta
		,cast(cdtipomodeloproposta as numeric) as cdtipomodeloproposta
		,cast(DTINCLUSAO as datetime) as dtinclusao
		,cast(dataescolhaassinatura as datetime) as dataescolhaassinatura
		,cast(datafinalizacaoassinatura as datetime) as datafinalizacaoassinatura
		,cast(CDMOTIVOCANCELAMENTOASSINATURA as string) as cdmotivocancelamentoassintura
		,cast(CDTIPOMODELO as numeric) as cdtipomodelo
		,cast(SGCANALOPCAOASSINATURA as string) AS dscanalassinatura
	from bv-pdbd-prd.trusted_dbcntf.tbtipomodeloproposta
	where cast(DTINCLUSAO as date) >= &dtprop_gcp
);
disconnect from perm;
quit;

proc sql;
create table tpassinatura_novo as
select
	nuproposta,
	tpassinaturalog,
	fl_gapregistro
from(
	select
		a.nuproposta,
	/*	b.dtinclusao,*/
	/*	c.dataescolhaassinatura as dataescolhaassinatura,*/
		c.tpassinatura as tpassinaturalog,
		case when d.datafinalizacaoassinatura > c.dataescolhaassinatura then 1 else . end as fl_assinou,
		case when d.datafinalizacaoassinatura is not null and
				  d.datafinalizacaoassinatura > c.dataescolhaassinatura  and
				  d.datafinalizacaoassinatura <> c.datafinalizacaoassinatura then 1 else . end as fl_gapregistro
	from(
		select
			nuproposta,
			min(cdtipomodeloproposta) as cdtipomodeloproposta_min,
			max(cdtipomodeloproposta) as cdtipomodeloproposta_max
		from tbtipomodelo_proposta
		group by 1
		) a
	left join tbtipomodelo_proposta b on a.cdtipomodeloproposta_min = b.cdtipomodeloproposta
	left join tbtipomodelo_proposta c on a.cdtipomodeloproposta_max = c.cdtipomodeloproposta
	left join (
		select
			nuproposta,
			max(datafinalizacaoassinatura) as datafinalizacaoassinatura
		from tbtipomodelo_proposta
		where datafinalizacaoassinatura is not null
		and cdmotivocancelamentoassintura is null
		group by 1
		) d on a.nuproposta = d.nuproposta
)where fl_assinou = 1 or tpassinaturalog = "Física"
;run;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE proposta_documentacao AS SELECT * FROM CONNECTION TO DBCON (
SELECT
t1.NuProposta,
/*t1.DtEntradaProposta,*/
/*t1.StProposta,*/
/*t2.DsSubProduto, */
/*t5.NmGrupoSubProduto,*/
CASE WHEN t3.CdDocumento in (161)	 THEN 'Eletrônica'
	 WHEN t3.CdDocumento in (42,147) THEN 'Física'
	 								 ELSE 'ND' END AS TpAssinaturadoc
FROM DBCRED..TbProposta t1
INNER JOIN DBCOR..TbSubProduto t2 ON t1.CdSubProduto = t2.CdSubProduto and t2.CdProduto = 12
LEFT JOIN DBPENDENCIA..TbDocumentoContratoPendencia t3 ON t1.NuContratoGestao = t3.NuContrato AND t3.CdDocumento  IN (161,42,147)
LEFT JOIN DBCRED..TbSubProdutoGrupo t4 ON t1.cdSubProduto = t4.cdsubproduto
LEFT JOIN DBCRED..TbGrupoSubProduto t5 ON t4.CdGrupoSubProduto = t5.CdGrupoSubProduto
WHERE  t1.StProposta IN ('AP')
AND t1.DtEntradaProposta >= &dtprop_gcp
)
;QUIT;

proc sql; create table proposta_assinatura as
select
	a.nuanomes,
	a.dtentradaproposta,
	a.nuproposta,
	a.nmgrupoprodutometa,
	a.nmareaestruturacomercial,
	a.nmcanalestruturacomercial,
	a.nmsubestruturacomercialA,
	a.nmcolaborador,
	a.nmparceirocomercial,
	a.dsloginn1c,
	a.dsloginn2c,
	a.dsloginn3c,
	a.nucontrato,
	a.dtcontrato,
	a.vrproducao,
	COALESCE(b.tpassinaturadoc, "ND") as tpassinaturadoc,
	COALESCE(c.tpassinaturalog, "ND") as tpassinaturalog,
	c.fl_gapregistro
from producao a
left join proposta_documentacao b on a.nuproposta = b.nuproposta
left join tpassinatura_novo c on a.nuproposta = c.nuproposta
;run;
