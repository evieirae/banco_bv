PROC SQL;
    CREATE TABLE PESSOA_RESPONSAVEL AS
    SELECT
        A.CdPropostaLimiteCredito,
        A.DtConclusaoAtividade,
        A.DsLogin AS LOGIN_RESPONSAVEL,
        A.NmPessoa AS NOME_RESPONSAVEL
    FROM RESPONSAVEL A
    INNER JOIN (
        SELECT
            CdPropostaLimiteCredito,
            MAX(DtConclusaoAtividade) AS DtConclusaoAtividade
        FROM RESPONSAVEL
        GROUP BY CdPropostaLimiteCredito
    ) B
    ON A.CdPropostaLimiteCredito = B.CdPropostaLimiteCredito
    AND A.DtConclusaoAtividade = B.DtConclusaoAtividade
    ORDER BY A.DtConclusaoAtividade DESC;
QUIT;

PROC SQL;
	CREATE TABLE ANALISTA_GERENTE AS SELECT DISTINCT
		A.*,
		C.CDPESSOA_GERENTE,
		CASE
			 WHEN UPPER(A.DsLogin) EQ B.NmGerenteCredito THEN 'GERENTE'
			 WHEN A.CdPessoaFuncSuperior EQ C.CDPESSOA_GERENTE THEN 'ANALISTA'
			 WHEN A.CdPessoaFuncSuperior in ( 10068330, 506158163) THEN 'ANALISTA' ELSE '' END AS GESTAO

PROC SQL;
	CREATE TABLE CERCADINHO_UPPERMIDDLE AS SELECT DISTINCT
		A.*,
		CASE WHEN A.DsSetorMatriz EQ 'Fortemente Negativo' THEN 0
			 WHEN A.DsSetorMatriz EQ 'Levemente Negativo' THEN 0.4
			 WHEN A.DsSetorMatriz IN ('Neutro' 'NA' 'Sem informação') THEN 0.8
			 WHEN A.DsSetorMatriz EQ 'Levemente Positivo' THEN 1.1
			 WHEN A.DsSetorMatriz EQ 'Fortemente Positivo' THEN 1.5 ELSE 0.8 END AS PESO_SETORIAL,

		CASE WHEN VlVencPrejuSCRM00 > 100000 THEN 0
			 WHEN A.VlRespTotalSCRM00/A.Faturamento > 3.5 THEN 0
			 WHEN A.VlRespTotalSCRM00/A.Faturamento BETWEEN 2.51 AND 3.5 THEN 0.8
			 WHEN A.VlRespTotalSCRM00/A.Faturamento < 2.50 THEN 1.5 ELSE 0.8 END AS PESO_ENDIVIDAMENTOSCR,

		CASE WHEN A.C351PLTotalAtivos < 0.2 THEN 0
			 WHEN A.C351PLTotalAtivos BETWEEN 0.2 AND 0.3 THEN 0.8
			 WHEN A.C351PLTotalAtivos > 0.3 THEN 1.5 ELSE 0.8 END AS PESO_SOLVENCIA,

		CASE WHEN A.C354DividaLiquidaEBITDA > 3.5 THEN 0
			 WHEN A.C354DividaLiquidaEBITDA BETWEEN 2.6 AND 3.5 THEN 1
			 WHEN A.C354DividaLiquidaEBITDA < 2.6 THEN 2 ELSE 1 END AS PESO_ALAVANCAGEM,

		CASE WHEN A.CBEBITDADespesaFinanceira < 1.5 THEN 0
			 WHEN A.CBEBITDADespesaFinanceira BETWEEN 1.5 AND 2 THEN 1
			 WHEN A.CBEBITDADespesaFinanceira > 2 THEN 2 ELSE 1 END AS PESO_CAPPAGJUROS,

		CASE WHEN A.Faturamento > 0.1 THEN
				CASE WHEN A.VlRestritivos/(A.Faturamento/12) > 0.05 THEN 0
					 WHEN A.VlRestritivos/(A.Faturamento/12) BETWEEN 0.015 AND 0.05 THEN 0.8
					 WHEN A.VlRestritivos/(A.Faturamento/12) < 0.015 THEN 1.5 END ELSE 0.8 END AS PESO_RESTRITIVOS,

		CALCULATED PESO_SETORIAL +
		CALCULATED PESO_ENDIVIDAMENTOSCR +
		CALCULATED PESO_SOLVENCIA +
		CALCULATED PESO_ALAVANCAGEM +
		CALCULATED PESO_CAPPAGJUROS +
		CALCULATED PESO_RESTRITIVOS AS SCORE_CERCADINHO,

		CASE WHEN CALCULATED SCORE_CERCADINHO <= 3.5 THEN 'VERMELHO'
			 WHEN CALCULATED SCORE_CERCADINHO BETWEEN 3.6 AND 4 THEN 'LARANJA'
			 WHEN CALCULATED SCORE_CERCADINHO BETWEEN 4.1 AND 6.4 THEN 'AMARELO'
			 WHEN CALCULATED SCORE_CERCADINHO >= 6.5 THEN 'VERDE' END AS FAROL_CERCADINHO

	FROM PROPOSTAS11 AS A
	WHERE NmModeloProposta EQ 'PME - Prospecção Massificada'
