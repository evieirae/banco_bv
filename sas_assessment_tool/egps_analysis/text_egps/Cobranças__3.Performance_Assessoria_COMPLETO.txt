CREATE TABLE PERF_AUX_1 AS SELECT
A.*, B.ContadorFluxo, C.ContadorReg,
	CASE
		WHEN A.DT_ARQUIVO = C.ArquivoMax AND A.DTENVIO = C.EnvioMax THEN A.DT_REGULARIZADO
		ELSE .
	END AS DT_REGULARIZACAO FORMAT DDMMYY10.,
	CASE
		WHEN A.SAFRA - A.DT_ATRASO <= 0 THEN "FLUXO"
		WHEN A.SAFRA - A.DT_ATRASO > 0 AND A.SAFRA - A.DT_ATRASO <= 16 THEN "0001 A 0015"

		WHEN A.SAFRA - A.DT_ATRASO > 46 AND A.SAFRA - A.DT_ATRASO <= 61 THEN "0046 A 0060"
		WHEN A.SAFRA - A.DT_ATRASO > 60 AND A.SAFRA - A.DT_ATRASO <= 76 THEN "0061 A 0075"
	END AS FX_ATRASO_REAL
FROM PERF_AMIGAVEL_B A LEFT JOIN
(SELECT CDDIMCONTACARTAO, DT_ATRASO, COUNT(*) AS ContadorFluxo

schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS = must MAX_CHAR_LEN = 30);

create table PARCELAMENTO_FATURA as
    select distinct
	input(cdcontacartao,best22.) as cdcontacartao
	,dtadesao
	,input(NuParcelaAdesao,best22.) as NuParcelaAdesao
	,input(VrContratoAdesao,best22.) as VrContratoAdesao
	,input(cdtipoparcelado,best22.) as cdtipoparcelado

    from connection to perm (

	select distinct
	cdcontacartao
	,dtadesao
	,NuParcelaAdesao
	,VrContratoAdesao
	,cdtipoparcelado

	from bv-pdbd-prd.trusted_dwbi_dtwh.tbcontratoparcelamentofatura
	where dtadesao >= '2023-12-01'
	);

disconnect from perm;
quit;

DATA PARCELAMENTO_FATURA;
SET PARCELAMENTO_FATURA;
SAFRA = 100*YEAR(DtAdesao)+MONTH(DtAdesao);
RUN;

PROC FREQ DATA=PARCELAMENTO_FATURA;
TABLE SAFRA;
RUN;

LIBNAME CARNEIRO	BASE '/riscosbv/cobranca/jcarneiro/';




PROC SQL;
CREATE TABLE CARTEIRA_AMIGAVEL_E AS SELECT
A.*,
	CASE
		WHEN B.DtAdesao > 0 THEN 1
		ELSE 0
	END AS FLAG_PARCELAMENTO,
	CASE
		WHEN B.CdTipoParcelado = 2 THEN 'Convencional'
		WHEN B.CdTipoParcelado = 3 THEN 'Compulsorio'
		WHEN B.CdTipoParcelado = 4 THEN 'Saque Parcelado'
		WHEN B.CdTipoParcelado = 5 THEN 'Credito Pessoal'
		WHEN B.CdTipoParcelado = 6 THEN 'Crediario'
		WHEN B.CdTipoParcelado = 7 THEN 'Pagto Contas'
	END AS TipoParcelamento,
B.DtAdesao AS DtParcelamento, B.NuParcelaAdesao AS QtdeParcelasParcelamento, B.VrContratoAdesao AS VlrParcelamento
FROM CARTEIRA_AMIGAVEL_D A LEFT JOIN PARCELAMENTO_FATURA B ON
A.CDDIMCONTACARTAO = B.CdContaCartao AND
A.DT_REGULARIZACAO + 3 >= B.DtAdesao AND A.DT_REGULARIZACAO - 3 <= B.DtAdesao
AND B.CdTipoParcelado NOT IN (4,5,6,7);
QUIT;

PROC FREQ DATA=CARTEIRA_AMIGAVEL_E;
TABLE TipoParcelamento*SAFRA;
RUN;





PROC SQL;
CREATE TABLE CONFERE_CARTEIRA AS SELECT
A.*, B.CONTADOR FROM
CARTEIRA_AMIGAVEL_E A LEFT JOIN
(SELECT CDDIMCONTACARTAO,DT_REGULARIZACAO,COUNT(*) AS CONTADOR
FROM CARTEIRA_AMIGAVEL_E GROUP BY CDDIMCONTACARTAO,DT_REGULARIZACAO) B
ON A.CDDIMCONTACARTAO = B.CDDIMCONTACARTAO AND A.DT_REGULARIZACAO = B.DT_REGULARIZACAO;
RUN;

DATA CARTEIRA_AMIGAVEL_F;
SET CARTEIRA_AMIGAVEL_E;
IF DT_REGULARIZACAO > 0 AND FLAG_ACORDO = 0 AND FLAG_PARCELAMENTO = 1 THEN TIPO_PAGTO = 'PARCELAMENTO';
ELSE IF DT_REGULARIZACAO > 0 AND FLAG_ACORDO = 1 AND FLAG_PARCELAMENTO = 0 THEN TIPO_PAGTO = 'ACORDO';
ELSE IF DT_REGULARIZACAO > 0 AND FLAG_ACORDO = 1 AND FLAG_PARCELAMENTO = 1 THEN TIPO_PAGTO = 'ACORDO';
ELSE IF DT_REGULARIZACAO > 0 AND FLAG_ACORDO = 0 AND FLAG_PARCELAMENTO = 0 THEN TIPO_PAGTO = 'OUTROS';

IF VLR_COBRANCA <= 1000 THEN FX_TCKT = '00000 A 01000';
ELSE IF VLR_COBRANCA > 1000 AND VLR_COBRANCA <= 2000 THEN FX_TCKT = '01000 A 02000';
ELSE IF VLR_COBRANCA > 2000 AND VLR_COBRANCA <= 5000 THEN FX_TCKT = '02000 A 05000';
ELSE IF VLR_COBRANCA > 5000 AND VLR_COBRANCA <= 10000 THEN FX_TCKT = '05000 A 10000';
ELSE IF VLR_COBRANCA > 10000 AND VLR_COBRANCA <= 15000 THEN FX_TCKT = '10000 A 01000';
ELSE IF VLR_COBRANCA > 15000 THEN FX_TCKT = '15000 A 99999';

IF VLR_COB_INI <= 2000 THEN FAIXA_SALDO = 'BS';
ELSE IF VLR_COB_INI > 2000 AND VLR_COB_INI <= 4000 THEN FAIXA_SALDO = 'MS';
ELSE IF VLR_COB_INI > 2000 THEN FAIXA_SALDO = 'AS';

IF TPCLASSE = 'E' THEN DU = DU_REGULARIZADO;
ELSE IF TPCLASSE = 'F' THEN DU = DU_ENVIO;

RUN;

PROC SQL;
CREATE TABLE CUBO_COMPLETO AS SELECT
SAFRA, TPCLASSE, ASSESSORIA, FX_ATRASO_NEW,

TIPO_PAGTO,

CENARIO_Novo, FaixaTicket, FaixaSaldo,



COUNT(*) AS QUANTIDADE,
SUM(VLR_COBRANCA) AS VlrCobranca,
SUM(VLR_REGULARIZADO) AS VlrRegularizado,
SUM(VLR_A) AS VlrAcordo,
SUM(ENTRADA_A) AS VlrEntradaAcordo,
SUM(VlrParcelamento) AS VlrParcelamento,
DU_REGULARIZADO,DU_ENVIO
FROM CARTEIRA_AMIGAVEL_F
GROUP BY
SAFRA, TPCLASSE, ASSESSORIA, FX_ATRASO_NEW,

TIPO_PAGTO,

CENARIO_Novo, FaixaTicket, FaixaSaldo,
DU_REGULARIZADO,DU_ENVIO



;QUIT;

PROC SQL;

COUNT(*) AS QUANTIDADE, SUM(VLR_COBRANCA) AS VLR_COBRANCA, SUM(VLR_REGULARIZADO) AS VLR_REGULARIZADO ,
SUM(VLR_A) AS VALOR_ACORDO, SUM(ENTRADA_A) AS ENTRADA_ACORDO
FROM CARTEIRA_AMIGAVEL_E
GROUP BY SAFRA,	FX_ATRASO_2,	ASSESSORIA,

FX_ATRASO_NEW,	FX_TCKT,	TIPO_PAGTO;
