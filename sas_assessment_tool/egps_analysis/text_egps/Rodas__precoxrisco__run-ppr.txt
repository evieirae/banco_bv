/* Enviar e-mail de início da atualização */

FILENAME MAIL_INI EMAIL
	FROM=("inf-gerenciais@bv.com.br")
	TO=("inf-gerenciais@bv.com.br")
        BCC=("fernanda.varani@bv.com.br" "andressa.souza@bv.com.br" "hellen.dantas@bv.com.br" "felipe.ikezaki@bv.com.br" "rafaeli.trigone@bv.com.br")
SUBJECT="Preço por risco | Início da atualização";

DATA _NULL_;
	FILE MAIL_INI;
		PUT "Processamento da base de Preço por Risco iniciado.";
RUN;

/* Executar código de geração da base */
%include '/infogrc/precoxrisco/codigos/00.parametros-e-variaveis.sas';
%include '/infogrc/precoxrisco/codigos/01.input-bases-iqprdbi.sas';

%include '/infogrc/precoxrisco/codigos/06.tabela-taxas-propostas.sas';

/* Calcular tempo de processamento total */
%LET W_fimp = %SYSFUNC(datetime());
%LET W_durp = %SYSFUNC(intck(SECOND, &W_inip, &W_fimp), tod8.);

/* Enviar e-mail de finalização da atualização */
PROC SQL;
CREATE TABLE STATUSPPR AS
SELECT
MAX(DtContrato) format=ddmmyy10. AS MaxDtContrato,
MAX(DtEntradaProposta) format=ddmmyy10. AS MaxDtProposta
FROM INFGER.PPR_ANALITICO;
RUN;

OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_FIM EMAIL
	FROM=("inf-gerenciais@bv.com.br")
	TO=("inf-gerenciais@bv.com.br")
	BCC=("mariana.cardoso@bv.com.br" "larissa.borghi@bv.com.br" "fernanda.varani@bv.com.br" "andressa.souza@bv.com.br" "hellen.dantas@bv.com.br" "felipe.ikezaki@bv.com.br" "rafaeli.trigone@bv.com.br")
    TYPE="text/html"
    SUBJECT="Preço por risco | Fim da atualização";

ODS html body=MAIL_FIM style=noline rs=none;

PROC REPORT DATA=STATUSPPR nowd HEADLINE HEADSKIP
style (report) = {background = none font_face = "Calibri" font_size = 9pt just=left OUTLINECOLOR = none}
style (column) = {background = none CELLHEIGHT = 2.5% font_face = "Calibri" font_size = 9pt just=center OUTLINECOLOR = none}
style (header) = {foreground = white font_face="Calibri" font_size = 9pt just=center background = #69c3cf OUTLINECOLOR = none} ;

COLUMNS MaxDtContrato MaxDtProposta;
RUN;

PROC ODSTEXT;
   p "Prezados,";
   p "";
   p "A base de preço por risco foi atualizada.";
   p "A geração da base demorou &W_durp. para ser concluída.";
   P "";
   p "Atenciosamente,";
   p "Informações Gerenciais Estratégicas";
RUN;

ods _all_ close;

DATA LOGEXE;
  IdProcesso = 'BPRCRSCO';
  LENGTH Processo $40.;
  Processo = 'Preço por risco';
  FORMAT Inicio Final DATETIME18.;
  Inicio = &W_inip.;
  Final = &W_fimp.;
  Duracao = "&W_durp.";
RUN;

PROC APPEND DATA=LOGEXE BASE=IGELOG.LogExecucaoProcessos; RUN;
