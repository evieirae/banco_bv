PROC SQL;
%CONNECT_GCP;
CREATE TABLE CARTEIRA_GCON AS SELECT * FROM CONNECTION TO PERM (

	
	WITH QUERY1 AS (
	SELECT DISTINCT
		A.CD_CONTRATO AS NUCONTRATO,
		A.SQ_CONTRATO_FINANCEIRO AS SQCONTRATOFINANCEIRO,
		MIN(A.NU_PARCELA) AS NUPARCELA,
		SUM(A.VR_PARCELA) AS VLR_RISCO

	FROM bv-pdbd-prd.dvry_core.tbparcela A
	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato B ON
	A.CD_CONTRATO = B.cd_contrato
	AND A.SQ_CONTRATO_FINANCEIRO = B.sq_contrato_financeiro_ativo
	WHERE A.st_parcela <> 'Q'
	AND b.cd_produto in ('10','11','13')
	GROUP BY
		A.CD_CONTRATO,
		A.SQ_CONTRATO_FINANCEIRO),

	QUERY1_AUX as ( SELECT DISTINCT
		A.NUCONTRATO,
		A.SQCONTRATOFINANCEIRO,
		B.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO AS DTINI,
		CASE
			WHEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO IS NOT NULL AND B.fl_contrato_financeiro_ativo = false THEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO
		END AS DTFIM

	FROM QUERY1 A

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro B ON
	A.NUCONTRATO = B.cd_contrato
	AND A.SQCONTRATOFINANCEIRO = B.sq_contrato_financeiro

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro C ON
	A.NUCONTRATO = C.cd_contrato
	AND (A.SQCONTRATOFINANCEIRO + 1) = C.sq_contrato_financeiro

	WHERE B.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO < &FIM. AND ((CASE
			WHEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO IS NOT NULL AND B.fl_contrato_financeiro_ativo = false THEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO END) IS NULL OR (CASE
			WHEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO IS NOT NULL AND B.fl_contrato_financeiro_ativo = false THEN C.DT_INICIO_VIGENCIA_CONTRATO_FINANCEIRO END) >= &INICIO.)),

	QUERY2 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			A.DTINI,
			A.DTFIM,
			B.NUPARCELA,
			B.VLR_RISCO

	FROM QUERY1_AUX A LEFT JOIN QUERY1 B
	ON A.NUCONTRATO = B.NUCONTRATO
	AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO ),

	QUERY3 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			A.DTINI,
			A.DTFIM,
			A.NUPARCELA,
			A.VLR_RISCO,
			B.DT_LIQUIDACAO_EFETIVA as DTLIQUIDACAOEFETIVA,
			B.DT_CANCELAMENTO AS DTCANCELAMENTO,
			B.CD_SITUACAO_CONTRATO AS CDSITUACAOCONTRATO,
			CASE
				WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
				WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 	ELSE 0
		 	END AS FLAG_LIQ

		FROM QUERY2 A
    	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato B ON
    	A.NUCONTRATO = B.cd_contrato
		AND A.SQCONTRATOFINANCEIRO = B.sq_contrato_financeiro_ativo
		WHERE (CASE	WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
					WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 			ELSE 0
		 			END) = 0),

	QUERY4 AS (SELECT DISTINCT
			CAST(A.NUCONTRATO AS INT64) AS NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			A.DTINI,
			A.DTFIM,
			A.NUPARCELA,
			B.VR_PARCELA AS VRPARCELA,
			A.VLR_RISCO,
			A.DTLIQUIDACAOEFETIVA,
			A.DTCANCELAMENTO,
			CAST(A.CDSITUACAOCONTRATO AS INT64) AS CDSITUACAOCONTRATO,
			A.FLAG_LIQ,
			B.DT_VENCIMENTO AS DTVENCIMENTO,
			DATE_DIFF(&FIMM1., B.DT_VENCIMENTO, DAY) AS ATRASO_INICIAL,
			DATE_DIFF(&REF11., B.DT_VENCIMENTO, DAY) AS ATRASO_ATUAL,
			CASE
				WHEN (DATE_DIFF(B.DT_VENCIMENTO, &FIMM1., DAY)) >= 5 THEN 'E'
				ELSE 'F'
			END AS TP_CART,
			CASE
				WHEN K.CD_MOTIVO_CONTRATO_FINANCEIRO = '4' THEN 'RENEG'
				WHEN K.CD_MOTIVO_CONTRATO_FINANCEIRO IS NULL THEN 'OUTROS'
				ELSE 'OUTROS'
			END AS FL_RENEG,
			CASE
				WHEN CAST(C.CD_PRODUTO AS INT64) = 11 AND K.CD_TIPO_COBRANCA = '1' THEN 'RENEG CONSIG'
				ELSE 'OUTROS'
			END AS FL_RENEG_CONSIG,
			CAST(C.CD_PRODUTO AS INT64) AS CDPRODUTO,
			CAST(C.CD_MODALIDADE_PRODUTO AS INT64) AS CDMODALIDADEPRODUTO,
			E.NMMODALIDADEPRODUTO,
			H.CDGRUPOSUBPRODUTO,
			I.NMGRUPOSUBPRODUTO,
			D.CDSUBPRODUTO,
			F.DSSUBPRODUTO,
			G.TPPESSOA,
			CASE
				WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO IS NULL     THEN 'CP_BV'
				WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) IN ( 21, 22, 29, 44 )               THEN 'CP_BV'
				WHEN CAST(C.CD_PRODUTO AS INT64) = 10 AND CAST(C.CD_MODALIDADE_PRODUTO AS INT64) = 19 AND D.CDSUBPRODUTO = 18        THEN 'CP_BV_MAR_ABERTO'

%CONNECT_GCP;
CREATE TABLE CARTEIRA_FLEX AS SELECT * FROM CONNECTION TO PERM (

	
	WITH QUERY1 AS (
	SELECT DISTINCT
		A.CD_CONTRATO AS NUCONTRATO,
		MIN(A.NU_PARCELA) AS NUPARCELA,
		SUM(A.VR_PARCELA) AS VLR_RISCO

	FROM bv-pdbd-prd.dvry_core.tbparcela A
	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato B ON
	A.CD_CONTRATO = B.cd_contrato
	WHERE A.st_parcela <> 'Q'
	AND b.cd_produto in ('10','11','13')
	GROUP BY
		A.CD_CONTRATO),

	QUERY1_AUX as ( SELECT DISTINCT
		A.NUCONTRATO,
		B.DTPROCESSAMENTO AS DTINI,
		B.DTFIMVIGENCIA AS DTFIM

	FROM QUERY1 A

	LEFT JOIN bv-pdbd-prd.trusted_stgtxt_flex.tbcontratofinanceiro B ON
	A.NUCONTRATO = B.NUCONTRATO),

	QUERY2 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.DTINI,
			A.DTFIM,
			B.NUPARCELA,
			B.VLR_RISCO

	FROM QUERY1_AUX A LEFT JOIN QUERY1 B
	ON A.NUCONTRATO = B.NUCONTRATO),

	QUERY3 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.DTINI,
			A.DTFIM,
			A.NUPARCELA,
			A.VLR_RISCO,
			B.DT_LIQUIDACAO_EFETIVA as DTLIQUIDACAOEFETIVA,
			B.DT_CANCELAMENTO AS DTCANCELAMENTO,
			B.CD_SITUACAO_CONTRATO AS CDSITUACAOCONTRATO,
			CASE
				WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
				WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 	ELSE 0
		 	END AS FLAG_LIQ

		FROM QUERY2 A
    	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato B ON
    	A.NUCONTRATO = B.cd_contrato
		WHERE (CASE	WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
					WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 			ELSE 0
		 			END) = 0),

	QUERY4 AS (SELECT DISTINCT
			CAST(A.NUCONTRATO AS INT64) AS NUCONTRATO,
			A.DTINI,
			A.DTFIM,
			A.NUPARCELA,
			B.VR_PARCELA AS VRPARCELA,
			A.VLR_RISCO,
			A.DTLIQUIDACAOEFETIVA,
			A.DTCANCELAMENTO,
			CAST(A.CDSITUACAOCONTRATO AS INT64) AS CDSITUACAOCONTRATO,
			A.FLAG_LIQ,
			B.DT_VENCIMENTO AS DTVENCIMENTO,
			DATE_DIFF(&FIMM1., B.DT_VENCIMENTO, DAY) AS ATRASO_INICIAL,
			DATE_DIFF(&REF11., B.DT_VENCIMENTO, DAY) AS ATRASO_ATUAL,
			CASE
				WHEN (DATE_DIFF(B.DT_VENCIMENTO, &FIMM1., DAY)) >= 5 THEN 'E'
				ELSE 'F'
			END AS TP_CART,
			'OUTROS' AS FL_RENEG,
			'OUTROS' AS FL_RENEG_CONSIG,
			CAST(C.CD_PRODUTO AS INT64) AS CDPRODUTO,
			'CDC_SOLAR_FLEX' AS PRODUTO_NOVO,
			G.TPPESSOA,
			C.CD_CPF_CNPJ as NUCPFCNPJ

		FROM QUERY3 A
		LEFT JOIN  bv-pdbd-prd.dvry_core.tbparcela B
			ON A.NUCONTRATO = B.CD_CONTRATO
			AND A.NUPARCELA = B.NU_PARCELA

		LEFT JOIN bv-pdbd-prd.trusted_stgtxt_flex.tbpessoa G

PROC SQL;
	%CONNECT_GCP;
	CREATE TABLE TBPESSOA_FIM AS SELECT * FROM CONNECTION TO PERM (
	SELECT A.CDPESSOA,
	A.NUCPFCNPJ,
	A.NMPESSOA,
	B.DSLOGRADOURO,
	B.NULOGRADOURO,
	B.DSCOMPLEMENTOLOGRADOURO,
	B.CDCEP,
	B.DSBAIRRO,
	B.NMCIDADE,
	B.CDUF,
	MAX(B.SQENDERECO) AS SQENDERECO
	FROM bv-pdbd-prd.trusted_dbcor.tbpessoa AS A
	LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS B ON A.CDPESSOA = B.CDPESSOA
	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato C ON A.CDPESSOA = c.CD_PESSOA
	WHERE c.cd_produto in ('10','11','13')
	GROUP BY A.CDPESSOA,
	A.NUCPFCNPJ,
	A.NMPESSOA,
	B.DSLOGRADOURO,
	B.NULOGRADOURO,
	B.DSCOMPLEMENTOLOGRADOURO,
	B.CDCEP,
	B.DSBAIRRO,
	B.NMCIDADE,
	B.CDUF
	ORDER BY A.CDPESSOA
	);
	DISCONNECT FROM PERM;
	QUIT;
	RUN;

	A.CD_CONTRATO = B.cd_contrato

		A.CD_CONTRATO),

		A.NUCONTRATO,
		B.DTPROCESSAMENTO AS DTINI,
		B.DTFIMVIGENCIA AS DTFIM

	FROM QUERY1 A

	LEFT JOIN bv-pdbd-prd.trusted_stgtxt_flex.tbcontratofinanceiro B ON
	A.NUCONTRATO = B.NUCONTRATO),

	QUERY2 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.DTINI,
			A.DTFIM,
			B.NUPARCELA,
			B.VLR_RISCO

	FROM QUERY1_AUX A LEFT JOIN QUERY1 B
	ON A.NUCONTRATO = B.NUCONTRATO),

	QUERY3 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.DTINI,

			A.NUPARCELA,

				WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
				WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 	ELSE 0
		 	END AS FLAG_LIQ

		FROM QUERY2 A
    	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato B ON
    	A.NUCONTRATO = B.cd_contrato
		WHERE (CASE	WHEN B.DT_LIQUIDACAO_EFETIVA < &INICIO. THEN 1
					WHEN B.DT_CANCELAMENTO      < &INICIO. THEN 1
		 			ELSE 0
		 			END) = 0),

	QUERY4 AS (SELECT DISTINCT
			A.NUCONTRATO,
			A.DTINI,
			A.DTFIM,
			A.NUPARCELA,
