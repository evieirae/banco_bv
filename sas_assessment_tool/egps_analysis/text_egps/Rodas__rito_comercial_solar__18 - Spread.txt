/* Verificar se a base rebate.solar_rebate_1123 é a mais atual */
proc sql;
Create table Base_Spread as
Select distinct
'Spread' as Base
,Prod.NuAnoMes
,Prod.NmAreaEstruturaComercialC
,Prod.NmColaboradorC
,Prod.DsLoginN1C
,Prod.DsLoginN2C
,Prod.DsLoginN3C
,Case
	When Prod.CdParceiroComercial = 418254 and Parc.CdParceiroComercial NE . Then Parc.CdParceiroComercial
	Else Prod.CdParceiroComercial
End as CdParceiroComercial
,Case
	When Prod.CdParceiroComercial = 418254 and Parc.CdParceiroComercial NE . Then Parc.NmParceiroComercial
	Else Prod.NmParceiroComercial
End as NmParceiroComercial
,'Mês Fechado' as Semana
,Porte.Porte
,Porte.TipoParceiro
,Max(DatePart(Parc.DtCadastro)) as DtCadastro Format ddmmyy10.

,Prod.QtCorridosDiaUtilMes
,Case
	when Spread.TABELA is not null then Spread.TxRebate*Prod.VrProducao
	else Prod.VpTxGerencialRebate
end as VpTxNet
,Prod.VrProducao
,Prod.VpTirContabil
,Max(&DtAtualizacao.) as DataBase Format DDMMYY10.

From Bases.TbProducao as Prod
Left Join Base_Cobans_Solar_Saida as Parc On Prod.NuCnpjLojaFavorecido = Parc.NuCnpjLoja
left join Rebate_Solar as Spread on (Prod.CdTabelaFinanciamento = Spread.TABELA and Prod.QtParcelas = Spread.PRAZO)
Left Join Base_Porte_Cobans_CNPJ as Porte On Prod.NuCnpjLojaFavorecido = Porte.NuCnpjLojaFavorecido and Prod.NuAnomes = Porte.NuAnomes

Where Prod.NuANomes >= &Base_m5. and Prod.NmGrupoProdutoMeta = 'Solar' and Prod.NmCanalEstruturaComercial = 'Canal Solar' and
		Prod.NmAreaEstruturaComercial in
		(
			'Canal Solar CO e NO',
			'Canal Solar MG, BA, AL e SE',
			'Canal Solar NE',
			'Canal Solar SP Interior',
			'Canal Solar SP, RJ e ES',
			'Canal Solar Sul'
		)
;quit;

Proc Sql;
Create Table Base_Spread_Canal_Total as
Select
Base
,NuAnoMes
,'Canal' as Visao
,"" as DsLoginN1C
,"" as DsLoginN2C
,"" as DsLoginN3C
,'Canal Solar' as Dimensao1
,'Canal Solar' as Dimensao2
,'Canal Solar' as Dimensao3
,Semana
,Porte
,TipoParceiro
,QtCorridosDiaUtilMes
,Sum(VpTxNet) as VpTxNet
,Sum(VrProducao) as VrProducao
,Sum(VpTirContabil) as VpTirContabil

,Max(DataBase) as DataBase Format DDMMYY10.
