 Cd_Causa as Cd_Causa

,Pasta as Pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdCausaUnidade as CdCausaUnidade
,CdSistema as CdSistema
,CdUnidadeExyon as CdUnidadeExyon
,NmRequisitante as NmRequisitante
,NmExecutor as NmExecutor
,NmUsuarioResponsavel as NmUsuarioResponsavel
,TipoRoteiroAprovacao as TipoRoteiroAprovacao
,NmStatusRequisicao as NmStatusRequisicao
,NuRequisicao as NuRequisicao
,NmTituloRequisicao as NmTituloRequisicao
,TipoRequisicao as TipoRequisicao
,DtRequisicao as DtRequisicao
,DtUltimaAprovacao as DtUltimaAprovacao
,DtVencimentoAprovacao as DtVencimentoAprovacao
,DtUltimaModificacao as DtUltimaModificacao
,DtVencimentoExecucao as DtVencimentoExecucao
,DtUltimaExecucao as DtUltimaExecucao
,DtFinalizacao as DtFinalizacao

FROM CONNECTION TO postgres(



select
case
 	when pr.serial_import like ('%INQ%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%BVT%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%CON%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%VAC%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%CTO%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRI%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRB%') then (left(pr.SERIAL_IMPORT,10))
		else pr.serial_import end                           as Cd_Causa
 ,pr.SERIAL_IMPORT 											as SERIAL_IMPORT
,pr.pasta                               					as Pasta
,pr.ID_NATUREZA												as IdNatureza
,pr.NATUREZA_UNIDADE_CUSTOM									as IdNaturezaUnidadeCustom
,null::numeric												as CdCausaUnidade
,'p' 														as CdSistema
,case
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 2             then 'TRI'
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 1             then 'TBR'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 4             then 'CTO'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 12            then 'CON'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 13            then 'VAC'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM in (1 ,7, 11)   then 'INQ'
	when pr.ID_NATUREZA = 27 and pr.NATUREZA_UNIDADE_CUSTOM = 5             then 'BVT'
	when pr.ID_NATUREZA = 32 and pr.NATUREZA_UNIDADE_CUSTOM in (7, 11 , 12) then 'INQ'
	else null end 										     as CdUnidadeExyon

,us.nome                                      				 as NmRequisitante
,ex.nome                                      				 as NmExecutor
,ur.nome                                      				 as NmUsuarioResponsavel
,ra.nome                                      				 as TipoRoteiroAprovacao

,case
	when re.fase = 'E' then 'AGUARDANDO EXECUÇÃO'
	when re.fase = 'A' then 'AGUARDANDO AVALIAÇÃO'
	when re.fase = 'X' then 'EXECUTADA'
	when re.fase = 'Q' then 'EM REQUISIÇÃO'
	when re.fase = 'R' then 'REPROVADA'
	when re.fase = 'Z' then 'FINALIZADA'
	when re.fase = 'C' then 'CANCELADA'
	when re.fase = 'I' then 'AGUARDANDO INFORMAÇÕES'
 end                                          				 as NmStatusRequisicao

,re.NUMERO_REQUISICAO                         				 as NuRequisicao
,' '                                     				 as NmTituloRequisicao
,tr.TITULO                                    				 as TipoRequisicao
,cast(re.data_cadastro as date)               				 as DtRequisicao
,cast(re.ultima_modificacao as date)          				 as DtUltimaAprovacao
,cast(re.vencimento_aprovacao as date)        				 as DtVencimentoAprovacao
,cast(re.ultima_modificacao as date)          				 as DtUltimaModificacao
,cast(re.vencimento_execucao as date)         				 as DtVencimentoExecucao
,cast(re.ultima_execucao as date)             				 as DtUltimaExecucao
,cast(re.data_finalizacao as date)            				 as DtFinalizacao

from public.M_PROCESSO pr
	left join public.M_REQUISICAO         re on re.ID_PROCESSO_REQUISICAO = pr.ID_PROCESSO
	left join public.LT_TIPO_REQUISICAO   tr on tr.ID_TIPO_REQUISICAO     = re.ID_TIPO_REQUISICAO
	left join public.M_USUARIO            us on us.ID_USUARIO             = re.ID_REQUISITANTE
	left join public.M_USUARIO            ex on ex.ID_USUARIO             = re.ID_EXECUTOR
	left join public.M_USUARIO            ur on ur.ID_USUARIO             = re.ID_USUARIO_RESPONSAVEL
	left join public.LT_ROTEIRO_APROVACAO ra on ra.ID_ROTEIRO_APROVACAO   = re.ID_ROTEIRO_APROVACAO
where re.NUMERO_REQUISICAO is not null

;

);DISCONNECT FROM postgres;

RUN;
