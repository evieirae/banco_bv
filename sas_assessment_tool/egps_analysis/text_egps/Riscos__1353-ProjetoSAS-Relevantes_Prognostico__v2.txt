PROC SQL;
CREATE TABLE WORK.PAGMES AS
select
YEAR(intnx('day',DtBase, -21916))* 100 + MONTH(intnx('day',DtBase, -21916)) AS AnoMes
,CdCausa
,Pasta
,VlProvisaoAtualizado as VlPagamentos

FROM WORK.TBL1353_PROV_ATUAL_ACUM
WHERE NmBase = 'PAGAMENTO'
order by 2 , 1
;RUN;

proc sql;
create table work.Tbl1353MES_ANT AS
SELECT
A.CdCausa
,A.Pasta
,A.Prognostico
,A.VlProvisaoAtualizado
,A.FlFechamento
,A.FlUltCargaDiaria
,A.NmCarteira
,A.AnoMes

FROM WORK.TBL1353_PROV_ATUAL_ACUM AS A

WHERE 1=1
AND a.FlFechamento in (1 , 2)
AND a.Prognostico NOT IN  ('PEDIDO','PAGAMENTO')
AND a.FlUltCargaDiaria = 1
;RUN;

PROC SQL;
CREATE TABLE WORK.DtRef AS

select
AnoMes
,year(DtRefAnt) * 100 + month(DtRefAnt) as AnoMesAnt
,DtRef
,intnx('month' ,DtRef  ,0,'END'  ) format= ddmmyy10. 	AS DtRefFim
,DtRefAnt
,intnx('month' ,DtRefAnt  ,0,'END'  ) format= ddmmyy10. 	AS DtRefAntFim
from (

SELECT DISTINCT
a.AnoMes
,mdy(input(substr(cats(a.AnoMes),5,2) , 5.),1,input(substr(cats(a.AnoMes),1,4) , 5.) ) as DtRef	format= ddmmyy10.
,intnx('month' , mdy(input(substr(cats(a.AnoMes),5,2) , 5.),1,input(substr(cats(a.AnoMes),1,4) , 5.) ) , -1 ) 	AS DtRefAnt format= ddmmyy10.
FROM WORK.TBL1353_PROV_ATUAL_ACUM as a
WHERE a.AnoMes IS NOT NULL
)
;run;

PROC SQL;
CREATE TABLE WORK.TBL1353_PROV_ATUAL_ACUM AS
SELECT
A.*
,B.AnoMesAnt
,CASE WHEN a.Prognostico IN  ('PEDIDO','PAGAMENTO')
		THEN .
		ELSE C.VlPagamentos END AS VlPagamentos

FROM WORK.TBL1353_PROV_ATUAL_ACUM AS A
LEFT JOIN WORK.DtRef AS B
ON A.AnoMes = B.AnoMes

LEFT JOIN WORK.PAGMES AS C
ON A.AnoMes = C.AnoMes
AND A.CdCausa = C.CdCausa
AND A.Pasta = C.Pasta

;run;

PROC SQL;
CREATE TABLE WORK.TBL1353_PROV_ATUAL_ACUM AS
SELECT
A.*
,B.VlProvisaoAtualizado AS VlProvisaoMesAnt
,(A.VlProvisaoAtualizado - B.VlProvisaoAtualizado) AS VlDifProvAtualxAnt
,((A.VlProvisaoAtualizado - B.VlProvisaoAtualizado) / A.VlProvisaoAtualizado)  AS PercentDifProvAtualxAnt

FROM WORK.TBL1353_PROV_ATUAL_ACUM AS A

LEFT JOIN WORK.Tbl1353MES_ANT AS B
ON A.AnoMesAnt = B.AnoMes
AND A.CdCausa = B.CdCausa
AND A.Pasta = B.Pasta
and A.Prognostico = B.Prognostico

;run;
