%include '/infogrc/utils/all-includes.sas';

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbAgenteShare AS SELECT * FROM connection to dbcon (
  SELECT DISTINCT
    CAST(Gravame.NuCNPJAgenteFinanceiro AS NUMERIC(14)) as NuCnpjAgente,
    Gravame.NmAgenteFinanceiro as NmAgente
  FROM
    DWBI_DTWH.TbGravamePanorama Gravame
  WHERE 1=1
			AND Gravame.CdStatusGravame <> 'B'
            AND Gravame.TpVeiculo IN ('AUTOMOVEL','CAMIONETA','CAMINHONETE','UTILITARIO','CICLOMOTOR','MOTONETA','MOTOCICLETA','TRICICLO','QUADRICICLO','MICROONIBUS','ONIBUS','REBOQUE','SEMI-REBOQUE','CAMINHAO TRATOR','CHASSI/PLATAFORMA','TRATOR DE RODAS','CAMINHAO')

);
RUN;

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbAgente AS SELECT * FROM connection to dbcon (
  SELECT * FROM DWBI_DTWH.TbAgenteMIS
);
RUN;

PROC SQL;
CREATE TABLE TbDisparoEmail AS
  SELECT
    MKT.NuCnpjAgente,
    MKT.NmAgente
  FROM
    TbAgenteShare MKT
  LEFT JOIN
    TbAgente TBI
    ON MKT.NuCnpjAgente = TBI.NuCnpjAgente
  WHERE
    TBI.NuCnpjAgente IS NULL;
RUN;

PROC SORT DATA=TbDisparoEmail NODUPKEY;
BY NuCnpjAgente;
RUN;

PROC SQL;
SELECT COUNT(NuCnpjAgente) INTO :QtdNovo FROM TbDisparoEmail;
RUN;

PROC SQL;
SELECT MAX(DtAtualizacao) FORMAT=DATE9. INTO :DtCetip FROM RELEXC.tbmarketsharediariaaberta;
RUN;
