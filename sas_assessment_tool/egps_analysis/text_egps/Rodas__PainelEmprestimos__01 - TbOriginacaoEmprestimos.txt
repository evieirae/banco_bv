PROC SQL;
   CREATE TABLE TbOriginacaoEmprestimos AS
   SELECT t1.DtEntradaProposta FORMAT=DDMMYYS10. AS DtEntradaProposta,
          t1.DtStatusProposta FORMAT=DDMMYYS10. AS DtStatusProposta,
          t1.DtAtualizacao FORMAT=DDMMYYS10. AS DtAtualizacao,
          t1.NmGrupoProdutoMeta,
          t1.NuProposta,
          t1.DsTipoSubCanalProducao,
          t1.CdParceiroComercial,
          t1.NmParceiroComercial,
          t1.NmTipoParceiroComercial,
          t1.TpPessoa,
          t1.CdUfPessoa,
          t1.NmCidadePessoa,
          t2.cdsexo,
          t2.DsEstadoCivil,
          t2.DsTipoProfissional,
          t2.DsProfissao,
          t2.dstiporesidencia,
          t1.FlPropostaOriginada AS QtPropostaOriginada,
          t1.FlPropostaAprovada AS QtPropostaAprovada,
          t1.FlPropostaEfetivada AS QtPropostaEfetivada,
          t1.FlPropostaAPNP AS QtPropostaAPNP,
          t1.VrPropostaOriginada FORMAT=COMMAX18.2 AS VrPropostaOriginada,
          t1.VrPropostaAprovada FORMAT=COMMAX18.2 AS VrPropostaAprovada,
          t1.VrPropostaEfetivada FORMAT=COMMAX18.2 AS VrPropostaEfetivada,
          t1.VrPropostaAPNP FORMAT=COMMAX18.2 AS VrPropostaAPNP,
          t2.Renda FORMAT=COMMAX18.2 AS Renda
      FROM BASES.TBORIGINACAO t1
           LEFT JOIN MESAS.BU_NN t2 ON (t1.NuProposta = t2.NuProposta)
      WHERE t1.DtEntradaProposta >= '1Jan2024'd AND t1.NmGrupoProdutoMeta IN
           (
          	'Solar',
		'Privado',
		'CP',
		'CP DESENROLA'
/*           	'CVG Leves',*/
/*           	'CVG Pesados',*/
/*           	'Motos',*/
/*           	'Leves',*/
/*           	'Pesados',*/
/*           	'Saude',*/
           );
QUIT;
