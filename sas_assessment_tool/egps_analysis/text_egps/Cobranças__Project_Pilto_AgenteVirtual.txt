PROC SQL;
CREATE TABLE BASE_PILOTO_ATIVA_AV AS
SELECT NUCONTRATO
    , PUT(NR_CPFCNPJ,Z14.) AS CPF
	, NR_CPFCNPJ AS CPF_2
	, RISCO_ENVIO
	, CD_ASSESSORIA
	, ASSESSORIA
	, DSGRUPO_2
	, CASE
          WHEN RISCO_ENVIO <=5000 THEN 'A.ATE 5K'
		  WHEN RISCO_ENVIO <=10000 THEN 'B.ATE 10K'
		  WHEN RISCO_ENVIO <=15000 THEN 'C.ATE 15K'
		  WHEN RISCO_ENVIO <=20000 THEN 'D.ATE 20K'
		  WHEN RISCO_ENVIO <=25000 THEN 'E.ATE 25K'
		  WHEN RISCO_ENVIO <=30000 THEN 'F.ATE 30K'
		  WHEN RISCO_ENVIO <=35000 THEN 'G.ATE 35K'
		  ELSE 'H.MAIOR35K'
		  END AS FX_RISCO
FROM ATIVA.BASE_ATIVA_202510
WHERE CD_ASSESSORIA = 2065
AND DSGRUPO_2 = 'PREDITIVO'
;QUIT;

        AND DT_CRIACAO_BASE IN ( DATE '2025-10-01')
		AND CD_ASSESSORIA = '2065'
    );

quit;
