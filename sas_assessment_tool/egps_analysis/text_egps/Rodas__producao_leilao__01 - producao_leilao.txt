proc sql;
create table PropostasAP as
select
epro.Nuproposta,
epro.Nuchassis,
Ext.DsPlaca,
epro.PcEntradaMolicar,
Prod.DtContrato as DtContrato,
Prod.NmAreaEstruturaComercial,
Prod.NmSubEstruturaComercialA,
coalesce(Prod.DsLoginN1,Prod.DsLoginN1C) 	as GR,
coalesce(coalesce(Prod.DsLoginN2A,Prod.DsLoginN2C),Prod.DsLoginN2H) as GF,
coalesce(coalesce(Prod.DsLoginN3A,Prod.DsLoginN3C),,Prod.DsLoginN3H)	as GER
from tmp.TBEXTPROPOSTA  as epro
	inner join bases.TbExtContrato as Ext
			on ext.Nuproposta = epro.Nuproposta

	left join tmp.TBPRODUCAO as Prod
			on Prod.Nuproposta = epro.Nuproposta

WHere 1=1
and prod.DtContrato between Today()-&Dia. and Today()
/*and epro.Nuanomes = 202303*/
and epro.stproposta = "AP"
and epro.PcEntradaMolicar < 30
and epro.PcEntradaMolicar <> .
and prod.NmGrupoProdutoMeta in ('Leves','Motos','Pesados')
/*and Prod.NmAreaEstruturaComercial in (	'Canal ABC','Canal Bauru','Canal Cascavel','Canal Curitiba I',*/
/*										'Canal Curitiba II','Canal Goiânia','Canal Londrina',*/
/*										'Canal Rio de Janeiro II','Canal Salvador',	*/
/*										'Canal S J Campos','Canal Tatuapé','Canal Varginha')*/

/*and epro.nuproposta = 271654897*/
;run;

/* Monta base de Leilão */
/*********************************************/

%CRIAVARDB;
PROC SQL;
%CONECTADB(server=CRE);
CREATE TABLE TbBaseLeilao AS
SELECT * FROM connection to dbcon
(
SELECT distinct
				  TbRetornoBaseLeilao.DsChassi
				 ,TbRetornoBaseLeilao.DtRetorno as DtRetorno
				 ,TbRetornoBaseLeilao.DtLeilao as DtLeilao
				 ,TbRetornoBaseLeilao.TpSucateamento as TpSucateamento
			FROM DbConsultaInformacaoBem..TbRetornoBaseLeilao TbRetornoBaseLeilao
/*			where DsChassi = '9BWAA05ZXA4120896'*/
						order by TbRetornoBaseLeilao.DtLeilao desc , TbRetornoBaseLeilao.TpSucateamento

);quit;

	PROC SORT DATA=TbBaseLeilao NODUPKEY;
BY DsChassi;
RUN;

data basegrc.TbBaseLeilao ;
set TbBaseLeilao
;run;

/*********************************************/
/* Monta base final */
/*********************************************/

proc sql;
create table ProducaoLeilao as
select  distinct
	 Pr.nuproposta
	,Bl.TpSucateamento
	,Bl.DtRetorno as DTConsulta
	,Bl.DtLeilao
	,Pr.DtContrato
	,Pr.NmAreaEstruturaComercial
	,Pr.NmSubEstruturaComercialA
	,Pr.DsPlaca
	,Pr.NuChassis
	,catt(put(Pr.PcEntradaMolicar,2.),'%') as PcEntradaMolicar
	,GR
	,GF
	,GER
from PropostasAP as Pr
	inner join TbBaseLeilao as BL
		on	Pr.NuChassis = bl.DsChassi
where 1=1
and	Bl.TpSucateamento in (4,5)
/*and	Bl.TpSucateamento in (9)*/
order by GER, GF, GR
/*and month (Bl.DtRetorno) = 10 and year (Bl.DtRetorno) = 2022*/
;run;
