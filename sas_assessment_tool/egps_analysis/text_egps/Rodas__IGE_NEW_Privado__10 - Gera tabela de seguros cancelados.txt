/* ======================================================================================*/
/* =====================Gera tabela de seguros cancelados=========================*/
/* ======================================================================================*/

PROC SQL;
   CREATE TABLE tempw.Tb_Seguros_canc AS
   SELECT (t1.NuAnoMes*100)+1 as nuanomes,
          t1.NmColaborador,
          t1.TpValor,
          t1.QtSegurosCancelados,
          t1.VrSegurosCancelados FORMAT COMMAX10.2
      FROM MISV.TBMETASEGUROSCANCELADOS t1

      WHERE t1.TpValor IN ('VrSPF','VrAP')
AND t1.NmColaborador IN (SELECT * FROM tempw.Depara_colaborador);
RUN;

%let dir_csv = /riscosbvf/BA/processos/output;

proc export
	data=tempw.Tb_Seguros_canc
	dbms=csv
	OUTFILE= "&dir_csv./Privado_Tb_Seguros_canc.csv"
	replace;
run;
