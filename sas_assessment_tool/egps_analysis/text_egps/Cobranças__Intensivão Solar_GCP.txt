CREATE TABLE BASE_ORACLE AS
SELECT CONTRATO,
CPFCNPJ,
ATRASO,
COD_ASSESSORIA,
SEGMENTO,
LETRA,
RISCO,
PARCELA,
VALOR_PARC,
LOJA,
PLANO,
QTD_PARC_PG,
QTD_PARC_NPG,
QTD_PA_VINC,
PARC_N_PG,
VLR_TOT_N_PG,
FLG_ORIGEM,
ULT_PARC,
VL_TOT_PARC,
TKT_MEDIO_PARCELA,
NOME_CLIENTE,
CPF_CLIENTE
FROM CONNECTION TO perm (SELECT
        DMACCT AS CONTRATO,
        DMSSNUM AS CPFCNPJ,
        DMDAYS AS ATRASO,
        DMAGENCY AS COD_ASSESSORIA,
        DMSPDESC AS SEGMENTO,
        DMQUE3 AS LETRA,
        DMCURBAL AS RISCO,
        U2NRPAANAT AS PARCELA,
        U2VLPARCREN AS VALOR_PARC,
        U2CDPACOM AS LOJA,
        U2PLANO AS PLANO,
        U2QTDPAPG AS QTD_PARC_PG,
        U2QTDPANPG AS QTD_PARC_NPG,
        U2QTDPAVINC AS QTD_PA_VINC,
        U2NRPAANAT AS PARC_N_PG,
        U2VLTOTPARC AS VLR_TOT_N_PG,
        CASE
            WHEN U2CODORISIST = '3' AND U2DESORISIST = 'FLEX' THEN 'FLEX'
            ELSE 'GCON'
        END AS FLG_ORIGEM,
        MAX(PANRPARC) AS ULT_PARC,
        SUM(CAST(PAVLPARC AS FLOAT64)) AS VL_TOT_PARC,
        ROUND(SUM(CAST(PAVLPARC AS FLOAT64)) / MAX(CAST(PANRPARC AS INT64)), 2)
 AS TKT_MEDIO_PARCELA,
        DMNAME AS NOME_CLIENTE,
        DMSSNUM AS CPF_CLIENTE
    FROM
        `bv-pdbd-prd.trusted_rcvry.delqmst` AS DELQMST,
        `bv-pdbd-prd.trusted_rcvry.uda2` AS UDA2,
        `bv-pdbd-prd.trusted_rcvry.tb_parcela` AS TB_PARCELA
    WHERE
        
DMSPDESC = 'EQUIPAMENTOS' AND
DMACCT = U2ACCT
        AND DMACCT = PAACCT
        AND DMSTATUS IS NULL
        AND CAST(DMDAYS AS INT64) BETWEEN 0 AND 540
    GROUP BY
        DMACCT, DMSSNUM, DMDAYS, DMAGENCY, DMSPDESC,
        DMCURBAL, U2NRPAANAT, U2VLPARCREN, U2CDPACOM, DMNAME, U2PLANO,
        U2QTDPAPG, U2QTDPANPG, U2QTDPAVINC, U2VLTOTPARC, DMQUE3,
        CASE WHEN U2CODORISIST = '3' AND U2DESORISIST = 'FLEX' THEN 'FLEX' ELSE 'GCON' END
);
QUIT;
