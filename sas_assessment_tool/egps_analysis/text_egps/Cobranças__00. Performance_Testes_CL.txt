CREATE TABLE PAGAMENTOS AS
SELECT DISTINCT

,CASE WHEN C.CDACORDO NE . THEN C.RECEBIDO_BV 				ELSE B.RECEBIDO_BV 				END AS RECEBIDO_BV

,CASE WHEN C.CDACORDO NE . THEN C.VRIOF 					ELSE B.VRIOF 					END AS VRIOF

,CASE WHEN C.CDACORDO NE . THEN C.VRTOTALACORDO 			ELSE B.VRTOTALACORDO 			END AS VRTOTALACORDO
,CASE WHEN C.CDACORDO NE . THEN C.SALDO_NEGOCIACAO 			ELSE B.SALDO_NEGOCIACAO 		END AS SALDO_NEGOCIACAO
,CASE WHEN C.CDACORDO NE . THEN C.VRPRINCIPALNEGOCIACAO 	ELSE B.VRPRINCIPALNEGOCIACAO 	END AS VRPRINCIPALNEGOCIACAO
,CASE WHEN C.CDACORDO NE . THEN C.DESCONTO 					ELSE B.DESCONTO 				END AS DESCONTO
,CASE WHEN C.CDACORDO NE . THEN C.DESCONTO_DIFERIDO 		ELSE B.DESCONTO_DIFERIDO 		END AS DESCONTO_DIFERIDO
,CASE WHEN C.CDACORDO NE . THEN C.CDACORDO 					ELSE B.CDACORDO 				END AS CDACORDO

LEFT JOIN PAGOS_CYBER_A AS B ON A.NUCONTA=B.NUCONTA AND A.PRODUTO=B.PRODUTO AND B.DATA_PAGTO2 BETWEEN A.DT_INICIO AND A.DT_FIM
LEFT JOIN PAGOS_CYBER_A AS C ON A.CD_ACORDO=C.CDACORDO
;QUIT;

DESCONTO_DIFERIDO = (DESCONTO/VRPRINCIPALNEGOCIACAO)*VRPRINCIPALPARCELAACORDO;

PROC SQL;
CREATE TABLE MARCACAO_CLUSTER_COMPLETA_B AS
SELECT DISTINCT
A.*
,B.N_PARCELAS
,B.QTDE_PARCELAS_ACORDO
,B.DT_PROCESSAMENTO_PAGO
,B.DATA_PAGTO
,B.CD_ASSESSORIA
,B.ASSESSORIA AS ASSESSORIA2
,B.DTACORDO

,B.RECEBIDO_BV
,B.VL_PAGO
,B.VRPARCELAACORDO
,B.VRPRINCIPALPARCELAACORDO
,B.VRIOF

,B.VRTOTALACORDO
,B.SALDO_NEGOCIACAO
,B.VRPRINCIPALNEGOCIACAO
,B.DESCONTO
,B.DESCONTO_DIFERIDO

FROM MARCACAO_CLUSTER_COMPLETA_A AS A
LEFT JOIN PAGOS_CYBER AS B ON A.NUCONTA = B.NUCONTA AND A.PRODUTO = B.PRODUTO AND A.DT_INICIO < DATEPART(B.DATA_PAGTO) AND A.DT_FIM >= DATEPART(B.DATA_PAGTO)

;QUIT;

PROC SQL;
CREATE TABLE DUP AS
SELECT DISTINCT NUCONTA, PRODUTO, DT_PROCESSAMENTO, Count(*) FROM MARCACAO_CLUSTER_COMPLETA_B
GROUP BY NUCONTA, PRODUTO, DT_PROCESSAMENTO
HAVING Count(*) > 1;
QUIT;

PROC SQL;
CREATE TABLE MARCACAO_CLUSTER_COMPLETA_C AS
SELECT DISTINCT
DT_PROCESSAMENTO
,dmque
,COD_ASSESS
,ASSESSORIA
,PERFIL_PROP_2026
,CLUSTER
,FX_ATRASO_A
,STOPERACAO
,PUBLICO
,CASE WHEN DATA_PAGTO > 0 THEN 1 ELSE 0 END AS FLAG_PAGO
,CASE WHEN QTDE_PARCELAS_ACORDO > 1 THEN 'PARCELADO'
	  WHEN QTDE_PARCELAS_ACORDO = 1 THEN 'QUITADO' ELSE '' END AS TP_PAGTO

,COUNT(CONTRATO)				AS QTDE_CONTRATO
,MIN(DATEPART(DATA_PAGTO)) 		AS DATA_PAGTO FORMAT DDMMYY10.
,SUM(SALDO_ATUAL) 				AS SALDO_ATUAL
,SUM(VL_PAGO) 					AS VLR_PAGO
,SUM(VRPRINCIPALPARCELAACORDO) 	AS VLR_PARCELA
,SUM(VRIOF) 					AS VLRIOF
,SUM(SALDO_NEGOCIACAO) 			AS SALDO_DEVEDOR
,SUM(VRPRINCIPALNEGOCIACAO) 	AS SALDO_PRINCIPAL
,SUM(DESCONTO) 					AS DESCONTO
,SUM(DESCONTO_DIFERIDO) 		AS DESCONTO_DIFERIDO
,SUM(N_PARCELAS) 				AS N_PARCELAS

FROM MARCACAO_CLUSTER_COMPLETA_B
GROUP BY
DT_PROCESSAMENTO
,dmque
,COD_ASSESS
,ASSESSORIA
,PERFIL_PROP_2026
,CLUSTER
,FX_ATRASO_A
,STOPERACAO
,PUBLICO
,FLAG_PAGO
,TP_PAGTO

;QUIT;
