/* Gerar tabela de log de status */

PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE LogStatusProposta_01 AS SELECT * FROM connection to dbcon (
  SELECT
    NuProposta,
    DATE(DtStatusProposta) AS DtStatusProposta,
    CASE WHEN StProposta IN ('A','CA','PA') THEN 'A' ELSE StProposta END AS StProposta,
    1 AS Prioridade

  FROM DBA.TbLogStatusProposta

  WHERE DtStatusProposta >= &iqM6.
    AND StProposta IN ('A','AP','C','CA','CE','PA','N')

);
RUN;

PROC SQL;
CREATE TABLE LogStatusProposta_01_Aux AS
  SELECT
    NuProposta,
    StProposta,
    MIN(DtStatusProposta) AS UltDtStatus FORMAT=DATE9.
  FROM LogStatusProposta_01

  GROUP BY 1,2;
RUN;

PROC SQL;
CREATE TABLE PropostasNegadasAprovadas AS
  SELECT
    NuProposta,
    MAX(CASE WHEN StProposta = 'A' THEN 1 ELSE 0 END) AS FlA,
    MAX(CASE WHEN StProposta = 'N' THEN 1 ELSE 0 END) AS FlN

  FROM LogStatusProposta_01

  GROUP BY 1
  HAVING calculated FlA + calculated FlN = 2;

RUN;

PROC SQL;
CREATE TABLE StatusAtual AS
  SELECT NuProposta, StProposta FROM projetos.TbOriginacao
  WHERE NuProposta IN (SELECT DISTINCT NuProposta FROM PropostasNegadasAprovadas);
RUN;

PROC SQL;
CREATE TABLE LogStatusProposta_02 AS
  SELECT
    A.NuProposta,
    A.DtStatusProposta,
    A.StProposta,
    A.Prioridade

  FROM LogStatusProposta_01 A
  INNER JOIN LogStatusProposta_01_Aux B ON A.NuProposta = B.NuProposta AND A.StProposta = B.StProposta
  AND A.DtStatusProposta = B.UltDtStatus
  LEFT JOIN PropostasNegadasAprovadas C ON A.NuProposta = C.NuProposta
  LEFT JOIN StatusAtual D ON A.NuProposta = D.NuProposta
  WHERE  CASE WHEN A.StProposta <> 'N' THEN 1
              WHEN A.StProposta = 'N' AND C.NuProposta <> . AND D.StProposta = 'N' THEN 1
              ELSE 0 END = 1;
RUN;

PROC SQL;
CREATE TABLE PropostasOriginadas AS
  SELECT
    NuProposta,
    DtEntradaProposta AS DtStatusProposta,
    'O' AS StProposta,
    0 AS Prioridade

  FROM projetos.TbOriginacao WHERE NuAnoMes >= &safraM6.;
RUN;

DATA LogStatusProposta;
SET LogStatusProposta_02 PropostasOriginadas;
RUN;

PROC SORT DATA=LogStatusProposta;
BY NuProposta DtStatusProposta Prioridade;
RUN;

PROC SORT DATA=LogStatusProposta NODUPKEY;
BY NuProposta DtStatusProposta Prioridade StProposta;
RUN;

PROC SQL;
PROC SQL;
CREATE TABLE Propostas AS
  SELECT
    ORI.NuAnoMes,
    ORI.NuProposta,
    ORI.NmGrupoProdutoMeta AS Produto,
    ORI.DsTipoCanalProducao AS Canal,
	ORI.DtEntradaProposta,
    COALESCE(CVG.FiltroCVG,'OUTROS PRODUTOS') as FiltroCVG,
    ORI.FxIdade,
    ORI.FxEntrada,
    ORI.FxParcela

  FROM projetos.TbOriginacao as ORI
    LEFT JOIN BASEGRC.TbFiltroCVG CVG
	ON CVG.nuproposta = ORI.nuproposta
  WHERE NuAnoMes >= &safraM6.;
RUN;

PROC SQL;
CREATE TABLE QtPropostas AS
  SELECT
    A.NuProposta,
    A.NuAnoMes,
    INPUT(STRIP(PUT(A.NuAnoMes,6.) || '01'),yymmdd8.) AS DtMin FORMAT=DATE9.,
    MIN(INTNX('DAY', calculated DtMin, 90, 'END'),TODAY()-1) AS DtMax FORMAT=DATE9.,
    a.DtEntradaProposta,
	b.DtStatusProposta,
    A.Produto,
    SUM(CASE WHEN B.StProposta = 'O' THEN 1 ELSE 0 END) AS QtOriginadas,
    SUM(CASE WHEN B.StProposta = 'A' THEN 1 ELSE 0 END) AS QtAprovadas,
    SUM(CASE WHEN B.StProposta = 'AP' THEN 1 ELSE 0 END) AS QtEfetivadas,
    SUM(CASE WHEN B.StProposta = 'C' THEN 1 ELSE 0 END) AS QtCanceladas,
    SUM(CASE WHEN B.StProposta = 'CE' THEN 1 ELSE 0 END) AS QtCanceladasAE,
    SUM(CASE WHEN B.StProposta = 'N' THEN 1 ELSE 0 END) AS QtNegadas
  FROM Propostas A
  LEFT JOIN LogStatusProposta B ON A.NuProposta = B.NuProposta
  where A.Produto IN ('Motos')
  GROUP BY
    1,2,3,4,5,6,7;

RUN;
