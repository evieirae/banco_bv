PROC SQL;
CREATE TABLE BASE_VENDAS_1 AS
SELECT
	A.SAFRA_VENDA,
	A.DT_VENDA,

	A.SEQ_CONTRATO_FINANCEIRO,

;RUN;
PROC SQL;
CREATE TABLE ALC_MIN AS
SELECT
	SAFRA_VENDA,
	LEILOEIRO_CAMPANHA,
	REGIAO_CAMPANHA,
	(SUM(
			CASE WHEN STATUS_FINAL = 'VENDIDO' THEN LANCE_2 END)
	/
	SUM(CASE WHEN STATUS_FINAL = 'VENDIDO' THEN VALOR_DESEJADO END)) AS ALCANCE_MINIMO,
	(CALCULATED ALCANCE_MINIMO) * 0.60 AS ALCANCE_MINIMO_60P,
	(COUNT(
			CASE WHEN STATUS_FINAL = 'VENDIDO' AND LEILOEIRO <> "NAPISTA" THEN CONTRATO_GESTAO END)
	/
	COUNT(CASE WHEN LEILOEIRO <> "NAPISTA" THEN CONTRATO_GESTAO END)) AS EXITO_VENDAS,
	(CALCULATED EXITO_VENDAS) * 0.40 AS EXITO_VENDAS_40p,
	(CALCULATED ALCANCE_MINIMO_60P) + (CALCULATED EXITO_VENDAS_40p) AS PONTUACAO
FROM BASE_VENDAS_1
WHERE SITUACAO_LEILOEIRO_CAMPANHA IS NULL AND LEILOEIRO_CAMPANHA IS NOT NULL
GROUP BY
	SAFRA_VENDA,
	LEILOEIRO_CAMPANHA,
	REGIAO_CAMPANHA
;RUN;
PROC RANK
DATA = ALC_MIN
OUT = ALC_MIN2
DESCENDING;
BY SAFRA_VENDA;
