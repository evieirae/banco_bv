    function DataPascoa(dt);
        
        array TabelaGauss [5,2] (   24  4
                                    24  5
                                    24  5
                                    24  6
                                    25  7
                                );

ano = year(dt);
        iGauss = int(ano/100)-17;
        a = mod(ano, 19);
        b = mod(ano, 4);
        c = mod(ano, 7);
        d = mod(19 * a + TabelaGauss[iGauss,1],30);
        e = mod(2 * b + 4 * c + 6 * d + TabelaGauss[iGauss,2],7);
        if (d + e) gt 9 then do;
            dia = (d + e - 9);
            if dia eq 26 then dia +(-7);
            else if dia eq 25 and d eq 28 and a gt 10 then dia +(-7);
            Pascoa = mdy(4, dia, ano);
        end;
        else do;
            Pascoa = mdy(3,(d + e + 22), ano);
        end;
        return (Pascoa);

endsub;
    function DiaUtil(dt);
        dt_ant = dt - 1;
        return(dt eq intnx('DiaUtil', dt_ant, 1, 's'));
    endsub;
quit;
options cmplib=work.funcao;

    end;

		        WHEN T6.NuCpfCnpj IS NOT MISSING THEN 'OMIE'
		        WHEN T3.DOCUMENTO IS NOT MISSING THEN 'OMIE'
				WHEN T4.DOCUMENTO IS NOT MISSING THEN 'OMIE'
                WHEN T1.FLAG_FIM IN ('BLOQUEADOS_PLD','BLOQUEAR') AND UPCASE(t7.STATUS) = 'ATIVO' AND
                     t8.NuCpfCnpj IS NOT MISSING THEN 'RETIRADO'
				WHEN T1.FLAG_FIM IN ('BLOQUEADOS_PLD','BLOQUEAR') AND UPCASE(t7.STATUS) = 'ATIVO' THEN 'RETIRAR'
				WHEN T1.FLAG_FIM IN ('BLOQUEAR') AND T9.DOC_NUM IS NOT MISSING THEN 'RETIRAR'
				WHEN T1.FLAG_FIM = 'BLOQUEAR' AND T5.PAPEL_GLOBAL IS NOT MISSING THEN T5.PAPEL_GLOBAL
		        WHEN T1.FLAG_FIM = '' THEN 'BLOQUEAR'
		        ELSE T1.FLAG_FIM END) FORMAT $50. AS FLAG_FIM,
		t1.DtNova

      FROM PLD.BASE_BLOQUEIO_KYC_TOTAL t1
      LEFT JOIN BS_CLIENTES_BLOCK_PLD t5 ON (t1.NuCpfCnpj = t5.NuCpfCnpj)
	  LEFT JOIN BASE_RENOVACAO_FIM t9 ON (INPUT(t1.NuCpfCnpj,BEST14.) = t9.DOC_NUM)
	  LEFT JOIN SACADO_OMIE T2 ON (INPUT(t1.NuCpfCnpj,BEST14.) = T2.CNPJ)
	  LEFT JOIN BS_OMIE T6 ON (T1.NuCpfCnpj = T6.NuCpfCnpj)
	  LEFT JOIN PLD.BS_OMIE_FIM T3 ON (INPUT(T1.NuCpfCnpj,BEST14.) = T3.DOCUMENTO AND T3.TP_QUADRO = 'PRINCIPAL')
	  LEFT JOIN PLD.BS_CEDENTES_SCORE T4 ON (INPUT(T1.NuCpfCnpj,BEST14.) = T4.DOCUMENTO AND T4.TP_QUADRO = 'PRINCIPAL')
	  LEFT JOIN BS_SALES_FORCE T7 ON (INPUT(T1.NuCpfCnpj,BEST14.) = T7.DOC_NUM)
      LEFT JOIN BS_CLIENTES_ATIVOS T8 ON (t1.NuCpfCnpj = t8.NuCpfCnpj)
			;
QUIT;

PROC FREQ DATA=PLD.BASE_BLOQUEIO_KYC_TOTAL; TABLES FLAG*FLAG_FIM; RUN ;
