PROC SQL;
	CREATE TABLE WORK.DARK_POST2 AS
		SELECT DISTINCT
		A.*
		,CASE
			WHEN A.Flag_SaidaRisco = 'N'								   THEN 'y.nao_elegivel'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN -999 AND -31 THEN	'a.999_31aVencer'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN -30  AND -16 THEN	'b.30_16aVencer'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN -15  AND -1  THEN	'c.15_1aVencer'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO 				  =  0 THEN	'd.VctoHoje'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN 1    AND   4 THEN	'e.1_4EmAtraso'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN 5    AND  15 THEN	'f.5_15EmAtraso'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO BETWEEN 16   AND  30 THEN	'g.16_30EmAtraso'
			WHEN A.Flag_SaidaRisco = 'S' AND A.ATRASO 				 >  30 THEN	'h.+30EmAtraso'
			ELSE 'z.Verificar'
		END AS FX_SAIDA_RISCO
		FROM CONTATOS.AUX_BASE_CYBER AS A
		WHERE 1=1
		
		AND A.SISTEMA NOT IN ('FLEX')

PROC SQL;
CREATE TABLE WORK.BASE_01 AS
SELECT DISTINCT
	  TODAY() FORMAT yymmdd10. as DT_REF_PROC
		, A.*
		, SUBSTR(NOME_COMPLETO,LENGTH(COMPRESS(A.PRI_NOME))+2,(LENGTH(A.NOME_COMPLETO))) AS SOBRENOME
		, CASE
			WHEN MONTH(TODAY()) >= INPUT(SUBSTR(A.DT_NASCIMENTO,4,2),BEST2.) THEN (YEAR(TODAY())-INPUT(SUBSTR(A.DT_NASCIMENTO,7,4),BEST4.))
			WHEN MONTH(TODAY()) < INPUT(SUBSTR(A.DT_NASCIMENTO,4,2),BEST2.) THEN (YEAR(TODAY())-INPUT(SUBSTR(A.DT_NASCIMENTO,7,4),BEST4.))-1
			ELSE 9999
		END AS IDADE
		, CASE
			
			WHEN COMPRESS(A.GRUPO) NOT IN ('CARTAO','RODAS','CP') THEN 'OUTROS_PRODUTOS'
			WHEN COMPRESS(A.GRUPO) = 'CARTAO' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 90 AND 99 THEN 'DESCONSIDERAR_CONTROLE_JORNADAS'
			WHEN COMPRESS(A.GRUPO) = 'RODAS' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 95 AND 99 THEN 'DESCONSIDERAR_CONTROLE_JORNADAS'
			WHEN COMPRESS(A.GRUPO) = 'CP' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 90 AND 99 THEN 'DESCONSIDERAR_CONTROLE_JORNADAS'

			
			WHEN COMPRESS(A.GRUPO) = 'CARTAO' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 0 AND 80 THEN 'CTX_CARTOES'
			WHEN COMPRESS(A.GRUPO) = 'CARTAO' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 81 AND 89 THEN 'CONTROLE_CARTOES'

			
			WHEN COMPRESS(A.GRUPO) = 'RODAS' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 0 AND 84 THEN 'CTX_RODAS'
			WHEN COMPRESS(A.GRUPO) = 'RODAS' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 85 AND 94 THEN 'CONTROLE_RODAS'

			
			WHEN COMPRESS(A.CARTEIRA) = 'SOLAR' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 0 AND 80 THEN 'CLICOU_NEGOCIOU_SOLAR'
			WHEN COMPRESS(A.CARTEIRA) = 'SOLAR' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 81 AND 89 THEN 'CONTROLE_SOLAR'

			
			WHEN COMPRESS(A.GRUPO) = 'CP' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 0 AND 80 THEN 'CTX_CP'
			WHEN COMPRESS(A.GRUPO) = 'CP' AND INPUT(SUBSTR(A.NUCPFCNPJ,7,2),Z2.) BETWEEN 81 AND 89 THEN 'CONTROLE_CP'

			ELSE '99.VERIFICAR'
		  END AS PUBLICO_DARKPOST
		FROM WORK.DARK_POST AS A
		WHERE A.COD_ASSESSORIA NOT IN (996,809,627,718,738,757,787,300,3000,3001,1020,3002,1033,1019,1029,1030,1032,1427,1284,1240,1279,3031,5000,5002,3032,3033,1021,1238,703,1412,1513,1458,1596,1505,4050,1035,1018,1022,1447,9999)

PROC SQL;
	CREATE TABLE WORK.BASE_DARKPOST_1 AS

			  A.*
			, EMAIL1.EMAIL AS EMAIL1

		LEFT JOIN WORK.BASE_SYNAPSE_EMAIL	AS EMAIL2		ON A.NUCPFCNPJ = EMAIL2.IDPESSOA AND EMAIL2.Ranking = 2
		LEFT JOIN WORK.BASE_SYNAPSE_EMAIL	AS EMAIL3		ON A.NUCPFCNPJ = EMAIL3.IDPESSOA AND EMAIL1.Ranking = 3

		LEFT JOIN WORK.SYNAPSE_TEL			AS TELEFONE2	ON A.NUCPFCNPJ = TELEFONE2.IDPESSOA AND TELEFONE2.Ranking = 2
