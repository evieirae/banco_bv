PROC SQL;
%CONNECT_GCP;
CREATE TABLE CINCO_DIAS_ANALITICO_FLEX_ AS
SELECT * FROM CONNECTION TO PERM(

WITH QUERY_1 AS (

SELECT DISTINCT
    CAST(&SAFRAINICIO. AS STRING) AS SafraInicio,
    &QdMeses. AS QdMeses,
    &RefAtraso. AS RefAtraso,
    PARSE_DATE('%Y%m%d', CONCAT(SUBSTR(CAST(&SAFRAINICIO. AS STRING), 3, 4), SUBSTR(CAST(&SAFRAINICIO. AS STRING), 1, 2), '01')) AS DtInicioPesquisa,
    DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(SUBSTR(CAST(&SAFRAINICIO. AS STRING), 3, 4), SUBSTR(CAST(&SAFRAINICIO. AS STRING), 1, 2), '01')), INTERVAL CAST(&QdMeses. AS INT64) MONTH), INTERVAL 1 DAY) AS DtFimPesquisa,
    PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')) AS MesCompetencia,
    DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY) AS DtMesFim,
    CURRENT_DATE() AS DtAtrasoSafra,
    DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) AS AtrasoSafra,
    CAST(SUBSTR(
        CASE
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 360 THEN '11-Até 360 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 180 THEN '10-Até 180 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 90 THEN '09-Até 90 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 60 THEN '08-Até 60 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 45 THEN '07-Até 45 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 30 THEN '06-Até 30 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 25 THEN '05-Até 25 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 20 THEN '04-Até 20 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 15 THEN '03-Até 15 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 10 THEN '02-Até 10 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 5 THEN '01-Até 05 dias'
        END, 1, 2) AS INT) AS CdFxAtraso,
    SUBSTR(
        CASE
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 360 THEN '11-Até 360 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 180 THEN '10-Até 180 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 90 THEN '09-Até 90 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 60 THEN '08-Até 60 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 45 THEN '07-Até 45 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 30 THEN '06-Até 30 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 25 THEN '05-Até 25 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 20 THEN '04-Até 20 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 15 THEN '03-Até 15 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 10 THEN '02-Até 10 dias'
            WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 5 THEN '01-Até 05 dias'
        END, 4) AS FxAtraso,
    CASE
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 360 THEN '11-Até 360 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 180 THEN '10-Até 180 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 90 THEN '09-Até 90 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 60 THEN '08-Até 60 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 45 THEN '07-Até 45 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 30 THEN '06-Até 30 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 25 THEN '05-Até 25 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 20 THEN '04-Até 20 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 15 THEN '03-Até 15 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 10 THEN '02-Até 10 dias'
        WHEN DATE_DIFF(CURRENT_DATE(), DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(FORMAT_DATE('%Y%m', t1.dt_contrato), '01')), INTERVAL 1 MONTH), INTERVAL 1 DAY), DAY) >= 5 THEN '01-Até 05 dias'
    END AS NmFxAtraso
FROM
    bv-pdbd-prd.dvry_core.tbcontrato AS t1
FULL JOIN
    (SELECT DISTINCT
        t3.dt_contrato
    FROM
        bv-pdbd-prd.dvry_core.tbcontrato AS t3) AS t2
ON
    t1.dt_contrato = t1.dt_contrato

WHERE
    t1.dt_contrato BETWEEN PARSE_DATE('%Y%m%d', CONCAT(SUBSTR(CAST(&SAFRAINICIO. AS STRING), 3, 4), SUBSTR(CAST(&SAFRAINICIO. AS STRING), 1, 2), '01')) AND DATE_SUB(DATE_ADD(PARSE_DATE('%Y%m%d', CONCAT(SUBSTR(CAST(&SAFRAINICIO. AS STRING), 3, 4), SUBSTR(CAST(&SAFRAINICIO. AS STRING), 1, 2), '01')), INTERVAL CAST(&QdMeses. AS INT64) MONTH), INTERVAL 1 DAY)
ORDER BY
    MesCompetencia

),
QUERY_2 AS (

SELECT
    P.MesCompetencia,
    P.CdFxAtraso,
    P.FxAtraso,
    P.RefAtraso,
    PC.cd_contrato,
    PC.st_parcela,
    PC.dt_vencimento,
    DATE_ADD(PC.dt_vencimento, INTERVAL P.RefAtraso DAY) AS DtRefAtraso,
    RANK() OVER (PARTITION BY P.MesCompetencia, PC.cd_contrato ORDER BY PC.nu_parcela ASC) AS SqParcela,
    PC.nu_parcela,
    PC.vr_parcela,
    PC.dt_ultimo_recebimento,
    CASE
        WHEN PC.dt_ultimo_recebimento <= P.DtMesFim THEN PC.dt_ultimo_recebimento
        ELSE P.DtMesFim
    END AS DtCalculoAtrasoFim,
    P.DtMesFim,
    DATE_DIFF(P.MesCompetencia, PC.dt_vencimento, DAY) AS A_InicioMes,
    DATE_DIFF(
        CASE
            WHEN PC.dt_ultimo_recebimento <= P.DtMesFim THEN PC.dt_ultimo_recebimento
            ELSE P.DtMesFim
        END,
        PC.dt_vencimento,
        DAY
    ) AS A_FimMes,
    CASE
        WHEN P.RefAtraso BETWEEN DATE_DIFF(P.MesCompetencia, PC.dt_vencimento, DAY) AND DATE_DIFF(
            CASE
                WHEN PC.dt_ultimo_recebimento <= P.DtMesFim THEN PC.dt_ultimo_recebimento
                ELSE P.DtMesFim
            END,
            PC.dt_vencimento,
            DAY
        ) THEN 1
        ELSE 0
    END AS R_Atraso,
    CASE
        WHEN (
            CASE
                WHEN P.RefAtraso BETWEEN DATE_DIFF(P.MesCompetencia, PC.dt_vencimento, DAY) AND DATE_DIFF(
                    CASE
                        WHEN PC.dt_ultimo_recebimento <= P.DtMesFim THEN PC.dt_ultimo_recebimento
                        ELSE P.DtMesFim
                    END,
                    PC.dt_vencimento,
                    DAY
                ) THEN 1
                ELSE 0
            END
        ) = 1
        AND (RANK() OVER (PARTITION BY P.MesCompetencia, PC.cd_contrato ORDER BY PC.nu_parcela ASC)) = 1 THEN 1
        ELSE 0
    END AS FlCarteira,
    CASE
        WHEN (
            CASE
                WHEN P.RefAtraso BETWEEN DATE_DIFF(P.MesCompetencia, PC.dt_vencimento, DAY) AND DATE_DIFF(
                    CASE
                        WHEN PC.dt_ultimo_recebimento <= P.DtMesFim THEN PC.dt_ultimo_recebimento
                        ELSE P.DtMesFim
                    END,
                    PC.dt_vencimento,
                    DAY
                ) THEN 1
                ELSE 0
            END
        ) = 1
        AND (RANK() OVER (PARTITION BY P.MesCompetencia, PC.cd_contrato ORDER BY PC.nu_parcela ASC)) = 1 THEN 1
        ELSE 0
    END AS RegraCarteira
FROM
    QUERY_1 AS P
JOIN
    bv-pdbd-prd.dvry_core.tbparcela AS PC
    ON PC.dt_vencimento <= P.DtMesFim
JOIN
    bv-pdbd-prd.trusted_stgtxt_flex.tbcontratofinanceiro AS CF
    ON CF.nucontrato = PC.cd_contrato
LEFT JOIN (SELECT DISTINCT NUCONTRATO FROM bv-pdbd-prd.trusted_stgtxt_flex.tbcontratocontabil WHERE CDPRODUTO = 13) CCF
	 ON PC.cd_contrato = CCF.NUCONTRATO
WHERE
    (PC.st_parcela <> 'Q'
    OR PC.dt_ultimo_recebimento >= P.MesCompetencia)
    AND DATE(CF.dtprocessamento) <= P.MesCompetencia
	AND CCF.NUCONTRATO IS NOT NULL
GROUP BY
    P.MesCompetencia,
    P.CdFxAtraso,
    P.FxAtraso,
    P.RefAtraso,
    P.DtMesFim,
    PC.cd_contrato,
    PC.st_parcela,
    PC.dt_vencimento,
    PC.nu_parcela,
    PC.vr_parcela,
    PC.dt_ultimo_recebimento
ORDER BY
    P.MesCompetencia,
    PC.cd_contrato,
    PC.nu_parcela

),
QUERY_3 AS (

SELECT
    t.MesCompetencia,
    t.RefAtraso,
    t.cd_contrato AS NuContrato,
    t.nu_parcela AS RefNuParcela,
    t.dt_vencimento AS RefDtVencimento,
    MAX(t.FlCarteira) AS FlNaoCarteira
FROM
    QUERY_2 AS t
WHERE
    t.R_Atraso = 1
GROUP BY
    t.MesCompetencia,
    t.RefAtraso,
    t.cd_contrato,
    t.nu_parcela,
    t.dt_vencimento
HAVING
    FlNaoCarteira = 0

),
QUERY_4 AS (

SELECT
    A.MesCompetencia,
    A.RefAtraso,
    A.NuContrato,
    A.RefNuParcela,
    A.RefDtVencimento,
    B.nu_parcela,
    B.dt_vencimento,
    B.dt_ultimo_recebimento,
    CASE
        WHEN B.dt_ultimo_recebimento IS NULL THEN B.dt_vencimento
        ELSE A.RefDtVencimento
    END AS DtInicio,
    COALESCE(B.dt_ultimo_recebimento, B.DtMesFim) AS DtFim,
    DATE_DIFF(COALESCE(B.dt_ultimo_recebimento, B.DtMesFim), CASE WHEN B.dt_ultimo_recebimento IS NULL THEN B.dt_vencimento ELSE A.RefDtVencimento END, DAY) AS AtrasoNoPgto
FROM
    QUERY_3 AS A
INNER JOIN
    QUERY_2 AS B
ON
    B.MesCompetencia = A.MesCompetencia
    AND B.cd_contrato = A.NuContrato
    AND B.nu_parcela < A.RefNuParcela

),
QUERY_5 AS (

SELECT
    t.MesCompetencia,
    t.NuContrato,
    t.RefNuParcela,
    t.RefAtraso,
    MAX(t.AtrasoNoPgto) AS MaxAtraso,
    CASE
        WHEN MAX(t.AtrasoNoPgto) <= t.RefAtraso THEN 1
        ELSE 0
    END AS FlCarteira,
    CASE
        WHEN (CASE WHEN MAX(t.AtrasoNoPgto) <= t.RefAtraso THEN 1 ELSE 0 END) = 1 THEN 2
        ELSE 0
    END AS RegraCarteira
FROM
    QUERY_4 AS t
GROUP BY
    t.MesCompetencia,
    t.NuContrato,
    t.RefNuParcela,
    t.RefAtraso

),
QUERY_6 AS (

SELECT
    RF.MesCompetencia,
    RF.CdFxAtraso,
    RF.FxAtraso,
    RF.RefAtraso,
    RF.DtMesFim,
    RF.cd_contrato AS NuContrato,
    RF.nu_parcela AS NuParcela,
    RF.dt_vencimento AS DtVencimento,
    RF.A_InicioMes,
    RF.A_FimMes,
    CASE
        WHEN A.FlCarteira = 1 THEN A.FlCarteira
        ELSE RF.FlCarteira
    END AS FlCarteira,
    CASE
        WHEN A.FlCarteira = 1 THEN A.RegraCarteira
        ELSE RF.RegraCarteira
    END AS RegraCarteira
FROM
    QUERY_2 AS RF
LEFT JOIN
    QUERY_5 AS A
ON
    A.MesCompetencia = RF.MesCompetencia
    AND A.NuContrato = RF.cd_contrato
    AND A.RefNuParcela = RF.nu_parcela

),
QUERY_7 AS (

SELECT DISTINCT
    RF.MesCompetencia,
    RF.CdFxAtraso,
    RF.FxAtraso,
    RF.RefAtraso,
    RF.DtMesFim,
    RF.NuContrato,
    'BV' AS NmBanco,
    C.cd_modalidade_produto,
    70 AS CdSubProduto,
    MP.nmmodalidadeproduto,
    'CDC_SOLAR_FLEX' AS Produto,
    CASE
        WHEN CAST(C.cd_produto AS INT64) IN (10, 13) THEN 'EMPRESTIMO'
        WHEN CAST(C.cd_produto AS INT64) = 11 AND D.cdmotivocontratofinanceiro = 4 AND D.cdtipocobranca = 1 THEN 'EMPRESTIMO'
        WHEN CAST(C.cd_produto AS INT64) IN (12, 14) THEN 'RODAS'
        WHEN CAST(C.cd_produto AS INT64) IN (11) THEN 'CONSIGNADO'
        WHEN CAST(C.cd_produto AS INT64) IN (15) THEN 'CARTAO_CREDITO'
        WHEN CAST(C.cd_produto AS INT64) IN (16) THEN 'RESSARCIMENTO'
    END AS Familia,
    RF.NuParcela,
    RF.DtVencimento,
    RF.A_InicioMes,
    RF.A_FimMes,
    RF.RegraCarteira,
    DATE_ADD(RF.DtVencimento, INTERVAL RF.RefAtraso DAY) AS DtRefRisco,
    CAST(NULL AS FLOAT64) AS VrRisco,
    CAST(NULL AS DATE) AS DtRecebimento,
    CAST(NULL AS INT64) AS DiasRecebimento,
    CAST(NULL AS INT64) AS CdFxRecebimento,
    CAST(NULL AS STRING) AS NmFxRecebimento,
    CAST(NULL AS STRING) AS FxRecebimento
FROM
    QUERY_6 AS RF
JOIN
    bv-pdbd-prd.dvry_core.tbcontrato AS C ON RF.NuContrato = C.cd_contrato
JOIN
    bv-pdbd-prd.trusted_stgtxt_flex.tbcontratofinanceiro AS D ON RF.NuContrato = D.nucontrato
JOIN
    bv-pdbd-prd.trusted_dbcor.tbmodalidadeproduto AS MP ON CAST(C.cd_modalidade_produto AS INT64) = CAST(MP.cdmodalidadeproduto AS INT64)
WHERE
    RF.FlCarteira = 1
    AND CAST(C.cd_produto AS INT64) = 13
    AND CAST(C.cd_modalidade_produto AS INT64) = 55

),
QUERY_8 AS (

SELECT
    A.MesCompetencia,
    A.NuContrato,
    A.DtVencimento,
    A.DtRefRisco,
    A.DtVencimento AS DataRef,
    SUM(PC.vr_parcela) AS VrRisco,
    MIN(CASE
        WHEN PC.st_parcela = 'Q' AND PC.nu_parcela = A.NUPARCELA THEN PC.dt_ultimo_recebimento
    END) AS DtRecebimento,
    DATE_DIFF(MIN(CASE
        WHEN PC.st_parcela = 'Q' AND PC.nu_parcela = A.NUPARCELA THEN PC.dt_ultimo_recebimento
    END), A.DtVencimento, DAY) AS DiasRecebimento
FROM
    QUERY_7 AS A
JOIN
    bv-pdbd-prd.dvry_core.tbparcela AS PC ON A.NuContrato = PC.cd_contrato
WHERE
    PC.st_parcela <> 'Q'
    OR PC.dt_ultimo_recebimento >= A.DtRefRisco
GROUP BY
    A.MesCompetencia,
    A.NuContrato,
    A.DtRefRisco,
    A.DtVencimento
ORDER BY
    A.MesCompetencia,
    A.NuContrato,
    A.DtRefRisco

),
QUERY_9 AS (

SELECT
    RF.MesCompetencia,
	RF.CdFxAtraso,
	RF.FxAtraso,
	RF.RefAtraso,
	RF.DtMesFim,
	RF.NuContrato,
	RF.NmBanco,
	RF.cd_modalidade_produto,
	RF.CdSubProduto,
	RF.nmmodalidadeproduto,
	RF.Produto,
	RF.Familia,
	RF.NuParcela,
	RF.DtVencimento,
	RF.A_InicioMes,
	RF.A_FimMes,
	RF.RegraCarteira,
	RF.DtRefRisco,
    A.VrRisco,
    A.DtRecebimento,
    A.DiasRecebimento,
    CASE
        WHEN A.DiasRecebimento <= 5 THEN '01-Até 05 dias'
        WHEN A.DiasRecebimento <= 10 THEN '02-Até 10 dias'
        WHEN A.DiasRecebimento <= 15 THEN '03-Até 15 dias'
        WHEN A.DiasRecebimento <= 20 THEN '04-Até 20 dias'
        WHEN A.DiasRecebimento <= 25 THEN '05-Até 25 dias'
        WHEN A.DiasRecebimento <= 30 THEN '06-Até 30 dias'
        WHEN A.DiasRecebimento <= 45 THEN '07-Até 45 dias'
        WHEN A.DiasRecebimento <= 60 THEN '08-Até 60 dias'
        WHEN A.DiasRecebimento <= 90 THEN '09-Até 90 dias'
        WHEN A.DiasRecebimento <= 180 THEN '10-Até 180 dias'
        WHEN A.DiasRecebimento <= 360 THEN '11-Até 360 dias'
    END AS NmFxRecebimento,
	CAST(LEFT(CASE
        WHEN A.DiasRecebimento <= 5 THEN '01-Até 05 dias'
        WHEN A.DiasRecebimento <= 10 THEN '02-Até 10 dias'
        WHEN A.DiasRecebimento <= 15 THEN '03-Até 15 dias'
        WHEN A.DiasRecebimento <= 20 THEN '04-Até 20 dias'
        WHEN A.DiasRecebimento <= 25 THEN '05-Até 25 dias'
        WHEN A.DiasRecebimento <= 30 THEN '06-Até 30 dias'
        WHEN A.DiasRecebimento <= 45 THEN '07-Até 45 dias'
        WHEN A.DiasRecebimento <= 60 THEN '08-Até 60 dias'
        WHEN A.DiasRecebimento <= 90 THEN '09-Até 90 dias'
        WHEN A.DiasRecebimento <= 180 THEN '10-Até 180 dias'
        WHEN A.DiasRecebimento <= 360 THEN '11-Até 360 dias'
    END, 2) AS INT) AS CdFxRecebimento,
