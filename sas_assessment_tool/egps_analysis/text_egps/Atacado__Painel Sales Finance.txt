%macro tratamento(anomes,dataref);

data tbconsolidada_safra&anomes.;
	set tbconsolidada;

	format dataref date9.;
	dataref = &dataref.;
	safraref = &anomes.;

	if dtdesembolso >  &dataref. then nmoperacao   = 'Não existe'; else
                                      nmoperacao   = 'Existe    ';

	if dtvencimento >= &dataref. then nmvencimento = 'A vencer'; else
                                      nmvencimento = 'Vencida ';

	if dtmovimento = .           then nmpagamento  = 'Não paga'; else
	if dtmovimento <= &dataref.  then nmpagamento  = 'Paga    '; else
                                      nmpagamento  = 'Não paga';

	if nmoperacao = 'Não existe'                              then nmstatusfinal = 'Operação futura'; else
	if nmvencimento = 'A vencer' and nmpagamento = 'Não paga' then nmstatusfinal = 'A vencer       '; else
	if nmvencimento = 'A vencer' and nmpagamento = 'Paga    ' then nmstatusfinal = 'Liquidada      '; else
	if nmvencimento = 'Vencida ' and nmpagamento = 'Não paga' and flprejuizo = 1 then nmstatusfinal = 'Prejuízo      '; else
	if nmvencimento = 'Vencida ' and nmpagamento = 'Não paga' then nmstatusfinal = 'Em atraso      '; else
	if nmvencimento = 'Vencida ' and nmpagamento = 'Paga    ' then nmstatusfinal = 'Liquidada      ';

	if nmstatusfinal = 'Em atraso      ' then qtdiasatraso =  &dataref. - dtvencimento; else qtdiasatraso = 0;

	if qtdiasatraso > 5  and flprejuizo ne 1 then flover05 = 1; else flover05 = 0;
	if qtdiasatraso > 15 and flprejuizo ne 1 then flover15 = 1; else flover15 = 0;
	if qtdiasatraso > 30 and flprejuizo ne 1 then flover30 = 1; else flover30 = 0;
	if qtdiasatraso > 60 and flprejuizo ne 1 then flover60 = 1; else flover60 = 0;
	if qtdiasatraso > 90 and flprejuizo ne 1 then flover90 = 1; else flover90 = 0;







	if dtdesembolso = . then safradesembolso = .; else safradesembolso = year(dtdesembolso)*100+month(dtdesembolso);
	if dtvencimento = . then safravencimento = .; else safravencimento = year(dtvencimento)*100+month(dtvencimento);
	if dtmovimento = . then saframovimento = .; else saframovimento = year(dtmovimento)*100+month(dtmovimento);
	if dtprejuizo = . then safraprejuizo = .; else safraprejuizo = year(dtprejuizo)*100+month(dtprejuizo);

run;
%mend;

create table tbindicadores_empilhada as select
 a.safra
,a.cdprograma
,case when b01.vrdesembolso = . then 0 else b01.vrdesembolso end as vrdesembolso
,case when b02.vrvencimento = . then 0 else b02.vrvencimento end as vrvencimento
,case when b03.vrmovimento  = . then 0 else b03.vrmovimento  end as vrmovimento
,case when b04.vrcarteira   = . then 0 else b04.vrcarteira   end as vrcarteira
,case when b04.qtoperacoes  = . then 0 else b04.qtoperacoes  end as qtoperacoes
,case when b04.vrprazomedio = . then 0 else b04.vrprazomedio end as vrprazomedio
,case when b05.vrematraso   = . then 0 else b05.vrematraso   end as vrematraso
,case when b06.vrover05     = . then 0 else b06.vrover05     end as vrover05
,case when b07.vrover15     = . then 0 else b07.vrover15     end as vrover15
,case when b08.vrover30     = . then 0 else b08.vrover30     end as vrover30
,case when b09.vrover60     = . then 0 else b09.vrover60     end as vrover60
,case when b10.vrover90     = . then 0 else b10.vrover90     end as vrover90
from tbindicadores_inicial as a

%let safraliq = %sysfunc(today());

proc sql;
    create table INDICE_LIQUIDEZ as
    select
        cdprograma,
        intnx('month', dtvencimento, 0, 'b') as mes_vencimento format=date9.,
        put(year(dtvencimento)*100 + month(dtvencimento), 6.) as safra,
        sum(case when dtvencimento <= today() - 2 then vroperacao else 0
            end) as valor_total,
        sum(case when dtvencimento <= today() - 2 and dtmovimento is not null and dtmovimento - dtvencimento <= 5 then vroperacao else 0
            end) as valor_liquidado,
        calculated valor_liquidado / calculated valor_total as perc_liquidez format=percent8.2
    from
        TBOPERACOESHISTORICAS
    where
        intnx('month', dtvencimento, 0, 'b') <= intnx('month', &safraliq., 0, 'b')
        and (year(dtvencimento)*100 + month(dtvencimento)) >= &a24.
    group by
        cdprograma, calculated mes_vencimento, calculated safra
    order by
        cdprograma, calculated mes_vencimento;
quit;

proc sql;
    create table INDICE_LIQUIDEZ_SETOR as
    select
	    substr(nudocumentosacado,1,8) as cnpjraiz,
        cdprograma,
        sum(case when dtvencimento <= today() - 2 then vroperacao else 0
            end) as valor_total,
        sum(case when dtvencimento <= today() - 2 and dtmovimento is not null and dtmovimento - dtvencimento <= 5 then vroperacao else 0
            end) as valor_liquidado,
        calculated valor_liquidado / calculated valor_total as perc_liquidez format=percent8.2
    from
        TBOPERACOESHISTORICAS
    where
        intnx('month', dtvencimento, 0, 'b') <= intnx('month', &safraliq., 0, 'b')
        and (year(dtvencimento)*100 + month(dtvencimento)) >= &a24.
    group by
        calculated cnpjraiz, cdprograma
    order by
        calculated cnpjraiz, cdprograma;
quit;

	CREATE TABLE BASE_ENTRADA_CALCRAT AS SELECT

	 CNPJ
	,DATA_BALANCO
	,VALOR_FATUR
	,VALOR_PL
	,(VALOR_EBITDA / MONTH(DATA_BALANCO))* 12 AS VALOR_EBITDA
	,VALOR_RESULT
    ,VALOR_PASS_CIRC
    ,VALOR_SAL
    ,VALOR_ADIANT
	,(VALOR_PASS_CIRC-VALOR_SAL-VALOR_ADIANT-VALOR_FORNE) AS VALOR_PASS_FIN
	,VALOR_DISP
	,VALOR_ATV_CIRC
	,VALOR_EMP_FIN
	,VALOR_REAL_LP
	,VALOR_ATV_PERM
	,CASE WHEN VALOR_APLIC_FINAN >= 0 THEN VALOR_APLIC_FINAN ELSE 0 END AS VALOR_APLIC_FINAN
	,CASE WHEN VALOR_TOT_INV >= 0 THEN VALOR_TOT_INV ELSE 0 END AS VALOR_TOT_INV
	,SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM) AS VALOR_ATV
	,(SUM(CALCULATED VALOR_PASS_FIN,VALOR_EMP_FIN) - MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN)) AS VALOR_DIVLIQ
	,(CASE WHEN VALOR_PL<=0 THEN 1 ELSE 0 END) AS FLAG_PL_NEG
	,(CASE WHEN CALCULATED VALOR_EBITDA<=0 THEN 1 ELSE 0 END) AS FLAG_EBTDA_NEG
	,(CASE WHEN VALOR_RESULT<=0 THEN 1 ELSE 0 END) AS PREJUIZO_EXERC
	,(CASE WHEN CALCULATED VALOR_PASS_FIN=0 THEN 10 ELSE VALOR_DISP/CALCULATED VALOR_PASS_FIN END) AS DISPONIVEL_ENDIVIDAMENTO
	,(CASE WHEN ((VALOR_ATV_CIRC - VALOR_PASS_CIRC)/SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM))=0 THEN 0.0001 ELSE ((VALOR_ATV_CIRC - VALOR_PASS_CIRC)/SUM(VALOR_ATV_CIRC,VALOR_REAL_LP,VALOR_ATV_PERM)) END) AS CAPITAL_DE_GIRO_LIQUIDO
	,MIN(15,(CASE WHEN CALCULATED VALOR_EBITDA < 0 THEN
	                (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN) - CALCULATED VALOR_EBITDA
	                ELSE CALCULATED VALOR_DIVLIQ - CALCULATED VALOR_EBITDA END)
	          ELSE (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN (CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/CALCULATED VALOR_EBITDA
	                ELSE CALCULATED VALOR_DIVLIQ / CALCULATED VALOR_EBITDA END)
	          END)) AS DIVIDA_LIQUIDA_EBTDA
	,MIN(8,(CASE WHEN VALOR_PL < 0 THEN
	                (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN) - VALOR_PL
	                ELSE CALCULATED VALOR_DIVLIQ - VALOR_PL END)
	          ELSE (CASE WHEN CALCULATED VALOR_DIVLIQ < 0 THEN (CALCULATED VALOR_DIVLIQ + MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/VALOR_PL
	                ELSE CALCULATED VALOR_DIVLIQ / VALOR_PL END)
	          END)) AS DIVIDA_LIQUIDA_PL
	,(CASE WHEN VALOR_PASS_CIRC=0 THEN 28 ELSE (MAX(VALOR_DISP,CALCULATED VALOR_APLIC_FINAN))/VALOR_PASS_CIRC END) AS LIQUIDEZ_IMEDIATA

	FROM INFO_BALANCO_SUM3;

  CREATE TABLE PRODUCAO AS
  SELECT
      Produto,
      
      INTNX('MONTH', DtInicioOperacao, 0, 'B') AS Data_Mes FORMAT=ddmmyy10.,
      SUM(VrPrincipalOperacao) AS vrproducaoproduto
  FROM OPERACAO_CREDITO
  WHERE NOT MISSING(DtInicioOperacao)
    AND NOT MISSING(Produto)
  GROUP BY
      Produto,
      CALCULATED Data_Mes
  ORDER BY
      Produto,
      Data_Mes;
QUIT;

      CALCULATED Data_Mes
  ORDER BY

      Data_Mes;
