create table Tbproducao as
select
*
    ,case when AssCont.cdmodeloassinatura_ajustado = 10 then 1 else 0 end as QtAssAppBV
    ,case when AssCont.flelegivel_Assinapp = 1 then 1 else 0 end as QtAssElegiveis

proc sql;
create table TbSegurosCancelados as
select
	 NuAnoMes
	,CdColaborador
	,NmColaborador
	,NmGrupoCargoColaborador
	,DsLoginN1
	,NmGrupoProdutoMeta
	,DsTipoCanalProducao
	,CdFilial
	,DtAtualizacao
	,sum(FlBVMais) as QtBVMais
	,sum(FlCap) as QtCap
	,sum(FlPlacaSolar) as QtPlacaSolar

	,sum(case when DsTipoCanalProducao not in ('LOJA CANAL NEGOCIOS','PROMOTORA','CANAL PRIVADO') and NuAnoMes >= 202101 then QtDemaisSemCompleto else QtDemais end) as QtDemais
	,sum(case when DsTipoCanalProducao not in ('LOJA CANAL NEGOCIOS','PROMOTORA','CANAL PRIVADO') and NuAnoMes >= 202101 then FlCompleto else 0 end) as QtAutoTotal

	,sum(VrBVMais) as VrBVMais
	,sum(VrCap) as VrCap
	,sum(VrPlacaSolar) as VrPlacaSolar

	,sum(case when DsTipoCanalProducao not in ('LOJA CANAL NEGOCIOS','PROMOTORA','CANAL PRIVADO') and NuAnoMes >= 202101 then VrDemaisSemCompleto else VrDemais end) as VrDemais
	,sum(case when DsTipoCanalProducao not in ('LOJA CANAL NEGOCIOS','PROMOTORA','CANAL PRIVADO') and NuAnoMes >= 202101 then VrCompleto else 0 end) as VrAutoTotal

	,sum(case when DsTipoCanalProducao not in ('NEGOCIOS CENTRALIZADOS','PROMOTORA','CANAL PRIVADO') and NuAnoMes >= 202201 then VrDemaisComCompletoMax end) as VrSegurosAuto

from bases.TbSegurosProducaoCancelados
where NuAnoMes <= year(today()-1)*100+month(today()-1)
group by 1,2,3,4,5,6,7,8,9
order by 1,2,3,4,5,6,7,8,9
;run;

by
	NuAnoMes

	NmGrupoCargoColaborador
	DsLoginN1

	CdFilial
