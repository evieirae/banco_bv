proc sql;
  create table gc_tb_cliente_scr_bacen as
  select
    DataRef as anomes_ref,
    NuCpfCnpj,
    "GLOBAL" as ds_tipo,

    
    sum(case when Mod in ("1304", "0204", "0210", "0406", "0303")
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_cartao,

    sum(case when Mod in ("1304", "0204", "0210", "0406", "0303", "1904")
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_cartao,

    
    max(case when Mod = '0202'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_consignado,

    max(case when Mod = '0202'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_consignado,

    
    max(case when Mod = '0203'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_cp,

    max(case when Mod = '0203'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_cp,

    
    max(case when Mod = '0211'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_home_equity,

    max(case when Mod = '0211'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_home_equity,

    max(case when Mod in ('0901', '0902', '0903')
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_imovel,

    max(case when Mod in ('0901', '0902', '0903')
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_imovel,

    max(case when Mod = '0401'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_veiculos,

    max(case when Mod = '0401'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_veiculos

  from scr_&v_DT_AAAAMM_M0.

  group by 1,2,3
  order by 1 desc;
quit;

      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 0.2 and  vl_saldo_total_bv / vl_saldo_total_global_c_imovel <= 0.3 then "d. 20-30%"
      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 0.3 and  vl_saldo_total_bv / vl_saldo_total_global_c_imovel <= 0.4 then "e. 30-40%"
      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 0.4 and  vl_saldo_total_bv / vl_saldo_total_global_c_imovel <= 0.5 then "f. 40-50%"
      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 0.5 and vl_saldo_total_bv / vl_saldo_total_global_c_imovel <= 0.6 then "g. 50-60%"

      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 0.8 and  vl_saldo_total_bv / vl_saldo_total_global_c_imovel <= 0.9 then "j. 80-90%"

      when vl_saldo_total_bv / vl_saldo_total_global_c_imovel > 1 then "l. Acima de 100%"
      else "l. S/ Credito"
  end as fx_sow_c_imovel,
  case

      when vl_saldo_total_bv / vl_saldo_total_global_s_imovel > 0.5 and vl_saldo_total_bv / vl_saldo_total_global_s_imovel <= 0.6 then "g. 50-60%"

	create table gc_tb_anali_sow_prin_cont_corr_1 as
SELECT
    *,
    nucpfcnpj as cd_cpf_cnpj,
    CASE
      WHEN Classif_Credito = 'a. 0 a 50%' AND Classif_Conta = 'a. 0 a 15%' THEN 'Zona D'
      WHEN Classif_Credito = 'a. 0 a 50%' AND Classif_Conta IN ('b. 15 a 20%', 'c. 20 a 50%', 'd. 50%+') THEN 'Zona C'
      WHEN Classif_Credito = 'b. 50%+' AND Classif_Conta = 'a. 0 a 15%' THEN 'Zona B'

    *,
    CASE
      WHEN fx_sow_s_imovel IN ('a. 0', 'b. 0-10%', 'c. 10-20%', 'd. 20-30%', 'e. 30-40%', 'f. 40-50%') THEN 'a. 0 a 50%'

      WHEN fx_sow_s_imovel = 'l. S/ Credito' THEN 'a. 0 a 50%'
    END AS Classif_Credito,
    CASE
      WHEN pts_principalidade_conta IN ('a. 0', 'b. 0-10%', 'c. 10-15%') THEN 'a. 0 a 15%'
      WHEN pts_principalidade_conta = 'd. 15-20%' THEN 'b. 15 a 20%'

      WHEN pts_principalidade_conta = 'n. S/ Conta' THEN 'a. 0 a 15%'
    END AS Classif_Conta
  FROM gc_tb_anali_sow_prin_cont_corr
  WHERE nucpfcnpj  not in (select cliente from dre_geral_&v_DT_AAAAMM_M2._new_sem_cedidos)
  )

%macro PROC_SAFRA(DATA_REF,SAFRA);

PROC SQL;
CREATE TABLE UNIAO_GERAL_&SAFRA.  AS
	SELECT A.*

		,1 AS QTDE

		,CASE WHEN (COALESCE(RESULTADO_CARTEIRA,0)+COALESCE(RESULTADO_PRODUCAO,0)) <> 0
					THEN '1' ELSE '0' END AS FLAG_RECEITA

		,CASE WHEN (COALESCE(RESULTADO_CARTEIRA,0)+COALESCE(RESULTADO_PRODUCAO,0)) > 0 THEN '1' ELSE '0' END AS FLAG_RECEITA_POS

		,CASE WHEN NEGOCIO = 'RODAS' THEN B.P_RODAS ELSE 0 END AS P_RODAS

		,CASE WHEN NEGOCIO = 'EMPRESTIMOS' THEN B.P_EMPRESTIMOS ELSE 0 END AS P_EMPRESTIMOS

		,CASE WHEN NEGOCIO = 'CARTAO'  THEN B.P_CARTAO ELSE 0 END AS P_CARTAO

		,CASE WHEN NEGOCIO = 'CONTA'   THEN B.P_CONTA_DIGITAL ELSE 0 END AS P_CONTA_DIGITAL

		,CASE WHEN CALCULATED P_RODAS <> 0 THEN 1 ELSE 0 END AS P_RODAS_2
		,CASE WHEN CALCULATED P_EMPRESTIMOS <> 0 THEN 1 ELSE 0 END AS P_EMPRESTIMOS_2
		,CASE WHEN CALCULATED P_CARTAO <> 0 THEN 1 ELSE 0 END AS P_CARTAO_2
		,CASE WHEN CALCULATED P_CONTA_DIGITAL <> 0 THEN 1 ELSE 0 END AS P_CONTA_DIGITAL_2

		,CALCULATED P_RODAS + CALCULATED P_EMPRESTIMOS + CALCULATED P_CARTAO + CALCULATED P_CONTA_DIGITAL AS P_GERAL

		,CALCULATED P_RODAS_2 + CALCULATED P_EMPRESTIMOS_2 + CALCULATED P_CARTAO_2 + CALCULATED P_CONTA_DIGITAL_2 AS P_GERAL_2

		,PUT((CALCULATED P_GERAL / CALCULATED P_GERAL_2),Z2.2) AS P_MEDIO_GERAL
		,case when DATA_ENTRADA_CICLO = . then PROPCASE(CATT(SUBSTR(STRIP(UPCASE(PUT(&DATA_REF.,PTGDFMN.))),1,3),'/', YEAR(&DATA_REF.)))
			  when DATA_ENTRADA_CICLO < '01APR2021'D then 'Jan/1960'
			  else PROPCASE(CATT(SUBSTR(STRIP(UPCASE(PUT(a.DATA_ENTRADA_CICLO,PTGDFMN.))),1,3),'/', YEAR(a.DATA_ENTRADA_CICLO)))
		 end as ANOMES_SAFRA_ENT_CICLO

		,case when INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF) < 0 then 0 else INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF) end AS MOB

		,case when month(a.DATA_REF) in (1, 2, 3) 		then CATX('/',YEAR(a.DATA_REF),'T1')
			  when month(a.DATA_REF) in (4, 5, 6) 		then CATX('/',YEAR(a.DATA_REF),'T2')
			  when month(a.DATA_REF) in (7, 8, 9) 		then CATX('/',YEAR(a.DATA_REF),'T3')
			  when month(a.DATA_REF) in (10, 11, 12) 	then CATX('/',YEAR(a.DATA_REF),'T4')
		 end as Anotri_dt_ref

		,case when DATA_ENTRADA_CICLO = . and month(&DATA_REF.) in (1, 2, 3) then CATX('/',YEAR(&DATA_REF.),'T1')
			  when DATA_ENTRADA_CICLO = . and month(&DATA_REF.) in (4, 5, 6) then CATX('/',YEAR(&DATA_REF.),'T2')
			  when DATA_ENTRADA_CICLO = . and month(&DATA_REF.) in (7, 8, 9) then CATX('/',YEAR(&DATA_REF.),'T3')
			  when DATA_ENTRADA_CICLO = . and month(&DATA_REF.) in (10, 11, 12) then CATX('/',YEAR(&DATA_REF.),'T2')
			  when DATA_ENTRADA_CICLO < '01APR2021'D 				then 'Outras Safras' 
			  when month(a.DATA_ENTRADA_CICLO) in (1, 2, 3) 		then CATX('/',YEAR(a.DATA_ENTRADA_CICLO),'T1')
			  when month(a.DATA_ENTRADA_CICLO) in (4, 5, 6) 		then CATX('/',YEAR(a.DATA_ENTRADA_CICLO),'T2')
			  when month(a.DATA_ENTRADA_CICLO) in (7, 8, 9) 		then CATX('/',YEAR(a.DATA_ENTRADA_CICLO),'T3')
			  when month(a.DATA_ENTRADA_CICLO) in (10, 11, 12) 	    then CATX('/',YEAR(a.DATA_ENTRADA_CICLO),'T4')
		 end as Anotri_ent_ciclo
		,case when DATA_ENTRADA_CICLO = . then 0
			  when DATA_ENTRADA_CICLO < '01APR2021'D 				then -1 
			  when month(a.DATA_ENTRADA_CICLO) in (3) 		then INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF)
			  when month(a.DATA_ENTRADA_CICLO) in (6) 		then INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF)
			  when month(a.DATA_ENTRADA_CICLO) in (9) 		then INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF)
			  when month(a.DATA_ENTRADA_CICLO) in (12) 	    then INTCK('MONTH',a.DATA_ENTRADA_CICLO , a.DATA_REF)
		 end as mob_max_safra
		,g.para AS PRODUTO_RAC_PARA
		,f.para as PRODUTO_ENTRADA_CICLO_PARA
		,case when compress(h.class_contrato) = '' then 'Contrato Normal'
												   else h.class_contrato
		 end as class_contrato
		,case when input(i.nucpfcnpj,best32.) ne . then 1 else 0 end as Flag_Oferta_integrada

		'N/A' as cesta_ideal



		,sow.Classif_Credito
		,sow.Classif_Conta
		,sow.Zona

	FROM clientes.dre_geral_&safra._new   AS A
		LEFT JOIN clientes.PARAMETROS_PRAZO AS B
	ON 1 = 1 AND B.DATA_REF = &DATA_REF.
		left join depara_prod_entrada_ciclo as f
	on a.produto_entrada_ciclo = f.de
		left join depara_prod_rac as g
	on a.produto_rac = g.de
		left join depara_contratos as h
	on a.NUCONTRATOGESTAO = h.numcontrato
		left join clientes.tb_cpf_oferta_integrada as i
	on a.cliente = input(i.nucpfcnpj,best32.)


		left join tb_&SAFRA. as sow
	on a.cliente = sow.nucpfcnpj

	ORDER BY CLIENTE
;QUIT;

proc sql;
  create table gc_tb_cliente_scr_bacen as
  select
    DataRef as anomes_ref,
    NuCpfCnpj,
    "GLOBAL" as ds_tipo,

    
    sum(case when Mod in ("1304", "0204", "0210", "0406", "0303")
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_cartao,

    sum(case when Mod in ("1304", "0204", "0210", "0406", "0303", "1904")
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_cartao,

    
    max(case when Mod = '0202'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_consignado,

    max(case when Mod = '0202'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_consignado,

    
    max(case when Mod = '0203'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_cp,

    max(case when Mod = '0203'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_cp,

    
    max(case when Mod = '0211'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_home_equity,

    max(case when Mod = '0211'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_home_equity,

    max(case when Mod in ('0901', '0902', '0903')
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_imovel,

    max(case when Mod in ('0901', '0902', '0903')
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_imovel,

    max(case when Mod = '0401'
             then Em_Dia + Em_Atraso + Em_Prejuizo else 0 end) as vl_saldo_veiculos,

    max(case when Mod = '0401'
             then Em_Dia + Em_Atraso + Em_Prejuizo + Limite else 0 end) as vl_limite_veiculos

  from scr_&v_DT_AAAAMM_M0.

  group by 1,2,3
  order by 1 desc;

proc sql;
CREATE TABLE work.gc_princip_exceto_mesma_titu AS
   SELECT
	    A.cd_cpf_cnpj,
	    A.anomes_ref,
	    b.vr_renda_presumida as vl_renda_bv,
	    coalesce(vl_saque,0) + coalesce(vl_debito,0) + coalesce(vl_dacc,0) + coalesce(vl_boleto,0) + coalesce(vl_cdb,0) + coalesce(vl_ted_out,0) + coalesce(vl_pix_out_s_msm_titularidade,0) as vl_movimentacao,
	    Case when coalesce(vl_saque,0) + coalesce(vl_debito,0) + coalesce(vl_dacc,0) + coalesce(vl_boleto,0) + coalesce(vl_cdb,0) + coalesce(vl_ted_out,0) + coalesce(vl_pix_out_s_msm_titularidade,0) > (0.5* vl_renda_bv) then ("Banco Principal") Else ("Banco não Principal") end as Princ_vlr_pix_out_sem_titu,
	    date() as dt_atualizacao format=date9.

   FROM work.gc_principalidade_bd_ie as A
   	LEFT JOIN tb_renda_bv as B
   on input(A.cd_cpf_cnpj,best32.) = input(B.cd_cpf_cnpj,best32.)
   order by A.cd_cpf_cnpj, A.anomes_ref
;quit;

  set work.gc_princip_exceto_mesma_titu;
  by cd_cpf_cnpj anomes_ref;
  if first.cd_cpf_cnpj then rn = 1;
  else rn + 1;
run;

CREATE TABLE work.gc_princip_exceto_mesma_titu AS
   SELECT
	    A.cd_cpf_cnpj,

   FROM work.gc_principalidade_bd_ie as A
   	LEFT JOIN tb_renda_bv as B
   on input(A.cd_cpf_cnpj,best32.) = input(B.cd_cpf_cnpj,best32.)
   order by A.cd_cpf_cnpj, A.anomes_ref

  case
    when vl_movimentacao / vl_renda_bv = 0                                then "a. 0"
    when vl_movimentacao / vl_renda_bv > 0   and vl_movimentacao / vl_renda_bv <= 0.1  then "b. 0-10%"
    when vl_movimentacao / vl_renda_bv > 0.1 and vl_movimentacao / vl_renda_bv <= 0.15  then "c. 10-15%"
    when vl_movimentacao / vl_renda_bv > 0.15 and vl_movimentacao / vl_renda_bv <= 0.2  then "d. 15-20%"
    when vl_movimentacao / vl_renda_bv > 0.2 and vl_movimentacao / vl_renda_bv <= 0.3  then "e. 20-30%"
    when vl_movimentacao / vl_renda_bv > 0.3 and vl_movimentacao / vl_renda_bv <= 0.4  then "f. 30-40%"
    when vl_movimentacao / vl_renda_bv > 0.4 and vl_movimentacao / vl_renda_bv <= 0.5  then "g. 40-50%"
    when vl_movimentacao / vl_renda_bv > 0.5 and vl_movimentacao / vl_renda_bv <= 0.6  then "h. 50-60%"
    when vl_movimentacao / vl_renda_bv > 0.6 and vl_movimentacao / vl_renda_bv <= 0.7  then "i. 60-70%"
    when vl_movimentacao / vl_renda_bv > 0.7 and vl_movimentacao / vl_renda_bv <= 0.8  then "j. 70-80%"
    when vl_movimentacao / vl_renda_bv > 0.8 and vl_movimentacao / vl_renda_bv <= 0.9  then "k. 80-90%"
    when vl_movimentacao / vl_renda_bv > 0.9 and vl_movimentacao / vl_renda_bv <= 1.0  then "l. 90-100%"
    when vl_movimentacao / vl_renda_bv > 1  then "m. + de 100%"
    else  "n. S/ Conta"
  end as pts_principalidade_conta
from (
	  select
	    a.anomes as anomes_ref,
	    a.cliente as nucpfcnpj,

	    max(coalesce(b.vl_saldo_total_bv, 0)) as vl_saldo_total_bv,
	    max(coalesce(b.vl_saldo_total_global_c_imovel, 0)) as vl_saldo_total_global_c_imovel,
	    max(coalesce(b.vl_saldo_total_global_s_imovel, 0)) as vl_saldo_total_global_s_imovel,
	    max(coalesce(i.vl_movimentacao, 0)) as vl_movimentacao,
	    max(coalesce(j.vl_renda_bv * 1.0, 0)) as vl_renda_bv,
	    max(coalesce(a.fl_relacionado, 0)) as fl_cliente_relacionado,

	    
	    max(case when a.fl_relacionado = 1 then "Gera REC" else "Nao gera REC" end) as fl_gerou_receita,

	    max(coalesce(k.flag_cessao_fgts, 0)) as fl_cessao_fgts

	  from relacionados a

	  left join gc_tb_cli_cons_bv_bacen_dados b
	    on a.cliente = b.cd_cpf_cnpj


	  left join movimentacao i
	    on a.cliente = input(i.cd_cpf_cnpj, best32.)


	  left join renda j
	    on a.cliente = j.cd_cpf_cnpj


	  left join dre_geral_&v_DT_AAAAMM_M2._new_sem_cedidos k
	  	on a.cliente = k.cliente
	
	
	
	
	  group by 1,2
	)
;quit;

proc sql;
	create table gc_tb_anali_sow_prin_cont_corr_1 as
SELECT
    *,
    nucpfcnpj as cd_cpf_cnpj,
    CASE
      WHEN Classif_Credito = 'a. 0 a 50%' AND Classif_Conta = 'a. 0 a 15%' THEN 'Zona D'
      WHEN Classif_Credito = 'a. 0 a 50%' AND Classif_Conta IN ('b. 15 a 20%', 'c. 20 a 50%', 'd. 50%+') THEN 'Zona C'
      WHEN Classif_Credito = 'b. 50%+' AND Classif_Conta = 'a. 0 a 15%' THEN 'Zona B'
      WHEN Classif_Credito = 'b. 50%+' AND Classif_Conta IN ('b. 15 a 20%', 'c. 20 a 50%', 'd. 50%+') THEN 'Zona A'
    END AS Zona_cliente
  FROM (
  SELECT
    *,
    CASE
      WHEN fx_sow_s_imovel IN ('a. 0', 'b. 0-10%', 'c. 10-20%', 'd. 20-30%', 'e. 30-40%', 'f. 40-50%') THEN 'a. 0 a 50%'
      WHEN fx_sow_s_imovel IN ('g. 50-60%', 'h. 60-70%', 'i. 70-80%', 'j. 80-90%', 'k. 90-100%', 'l. Acima de 100%') THEN 'b. 50%+'
      WHEN fx_sow_s_imovel = 'l. S/ Credito' THEN 'a. 0 a 50%'
    END AS Classif_Credito,
    CASE
      WHEN pts_principalidade_conta IN ('a. 0', 'b. 0-10%', 'c. 10-15%') THEN 'a. 0 a 15%'
      WHEN pts_principalidade_conta = 'd. 15-20%' THEN 'b. 15 a 20%'
      WHEN pts_principalidade_conta IN ('e. 20-30%', 'f. 30-40%', 'g. 40-50%') THEN 'c. 20 a 50%'
      WHEN pts_principalidade_conta IN ('h. 50-60%', 'i. 60-70%', 'j. 70-80%', 'k. 80-90%', 'l. 90-100%', 'm. + de 100%', 'm. + de 100') THEN 'd. 50%+'
      WHEN pts_principalidade_conta = 'n. S/ Conta' THEN 'a. 0 a 15%'
    END AS Classif_Conta
  FROM gc_tb_anali_sow_prin_cont_corr
  WHERE nucpfcnpj  not in (select cliente from dre_geral_&v_DT_AAAAMM_M2._new_sem_cedidos)
  )
;quit;
