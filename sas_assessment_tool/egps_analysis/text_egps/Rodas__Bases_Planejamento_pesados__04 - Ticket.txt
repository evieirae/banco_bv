PROC SQL;
CREATE TABLE BASE_TICKET_00_P AS
SELECT distinct

NmAreaEstruturaComercialC,
FxIdade,
strip(NmAreaEstruturaComercialC)||FxIdade as Chave,

SUM(QtdContratos_ProjSab) AS QtdContratos_ProjSab FORMAT COMMAX20.6,
SUM(VrProducao_ProjSab) AS VrProducao_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED VrProducao_ProjSab / CALCULATED QtdContratos_ProjSab,0) AS Ticket_BV_ProjSab FORMAT COMMAX20.2,

SUM(QtGravamesMercadoAjustado) AS qtgravamesplayers_ProjSab FORMAT COMMAX20.6,
SUM(VrGravamesMercadoAjustado) AS Vrgravamesplayers_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED Vrgravamesplayers_ProjSab / CALCULATED qtgravamesplayers_ProjSab,0) AS Ticket_Mercado_ProjSab FORMAT COMMAX20.2,

case when CALCULATED Ticket_BV_ProjSab >0 then CALCULATED Ticket_BV_ProjSab else CALCULATED Ticket_Mercado_ProjSab end as TicketCluster FORMAT COMMAX20.2

FROM WORK.POTENCIAL_RODAS_04_P A
/*where strip(NmAreaEstruturaComercialC)||FxIdade = 'Canal Pesados0 a 2 anos' */
GROUP BY 1,2,3
;RUN;

PROC SQL;
CREATE TABLE BASE_TICKET_01_P AS
SELECT distinct

a.ChaveCdParceiro_FxIdade,
a.CdParceiroComercial,
a.FxIdade,
a.NmAreaEstruturaComercialC,

SUM(a.QtdContratos_ProjSab) AS QtdContratos_ProjSab FORMAT COMMAX20.6,
SUM(a.VrProducao_ProjSab) AS VrProducao_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED VrProducao_ProjSab / CALCULATED QtdContratos_ProjSab,0) AS Ticket_BV_ProjSab FORMAT COMMAX20.2,

SUM(a.QtGravamesMercadoAjustado) AS qtgravamesplayers_ProjSab FORMAT COMMAX20.6,
SUM(a.VrGravamesMercadoAjustado) AS Vrgravamesplayers_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED Vrgravamesplayers_ProjSab / CALCULATED qtgravamesplayers_ProjSab,0) AS Ticket_Mercado_ProjSab FORMAT COMMAX20.2,

case when CALCULATED Ticket_BV_ProjSab >0 then CALCULATED Ticket_BV_ProjSab else CALCULATED Ticket_Mercado_ProjSab end as TicketPDV FORMAT COMMAX20.2,
Sum(b.TicketCluster) as TicketCluster FORMAT COMMAX20.2,
case when CALCULATED TicketPDV >0 then CALCULATED TicketPDV else Sum(b.TicketCluster) end as TicketFinal FORMAT COMMAX20.2,
CALCULATED TicketFinal / Sum(b.TicketCluster)  as Multiplo,

case when CALCULATED Multiplo >3 then Sum(b.TicketCluster) * 3 else CALCULATED TicketFinal end as TicletFinalCap FORMAT COMMAX20.2

FROM WORK.BASE_TICKET_PoT_P A
	left join BASE_TICKET_00_P as B
	on strip(a.NmAreaEstruturaComercialC)||a.FxIdade = b.Chave
/*where strip(a.NmAreaEstruturaComercialC)||a.FxIdade= 'Canal Pesados0 a 2 anos' and CdParceiroComercial = 7554*/
GROUP BY 1,2,3,4
;RUN;
