PROC SQL;
CREATE TABLE ZENVIA_VISAO_UNICA_3 AS
SELECT
	A.*,
	CASE
		WHEN NOTA IN (0,1,2,3,4,5,6) AND Resposta_Qualitativa = 1  	THEN 'Abordagem do Atendente'
		WHEN NOTA IN (0,1,2,3,4,5,6) AND Resposta_Qualitativa = 2  	THEN 'As ofertas de negociação'
		WHEN NOTA IN (0,1,2,3,4,5,6) AND Resposta_Qualitativa = 3  	THEN 'Esclarecimento de suas dúvidas'
		WHEN NOTA IN (0,1,2,3,4,5,6) AND Resposta_Qualitativa = 4  	THEN 'Cobrança após o pagamento'
		WHEN NOTA IN (0,1,2,3,4,5,6) AND Resposta_Qualitativa = 5  	THEN 'Acessos aos canais digitais'
		WHEN NOTA IN (0,1,2,3,4,5,6) 								THEN 'Sem resposta'

		WHEN NOTA IN(7,8) AND Sugestao = 1   					THEN 'Abordagem do Atendente'
		WHEN NOTA IN(7,8) AND Sugestao = 2  					THEN 'As ofertas de negociação'
		WHEN NOTA IN(7,8) AND Sugestao = 3  					THEN 'Esclarecimento de suas dúvidas'
		WHEN NOTA IN(7,8) AND Sugestao = 4  					THEN 'Cobrança após o pagamento'
		WHEN NOTA IN(7,8) AND Sugestao = 5  					THEN 'Acessos aos canais digitais'
		WHEN NOTA IN(7,8) 										THEN 'Sem resposta'

		WHEN NOTA IN(9,10) AND MotivoNota = 1  		   	THEN 'Abordagem do Atendente'
		WHEN NOTA IN(9,10) AND MotivoNota = 2  			THEN 'As ofertas de negociação'
		WHEN NOTA IN(9,10) AND MotivoNota = 3  			THEN 'Esclarecimento de suas dúvidas'
		WHEN NOTA IN(9,10) AND MotivoNota = 4  			THEN 'Facilidade para usar os canais digitais'
		WHEN NOTA IN(9,10) AND MotivoNota = 5  			THEN 'BV como um todo'
		WHEN NOTA IN(9,10) 								THEN 'Sem resposta'

	END AS RESPOSTA_QUALITATIVA_CLI,
	CASE
		WHEN NOTA IN (0,1,2,3,4,5,6) 	THEN 'DETRATOR'
		WHEN NOTA IN (7,8)				THEN 'NEUTRO'
		WHEN NOTA IN (9,10)				THEN 'PROMOTOR'
	END AS CATEGORIA_NPS
FROM ZENVIA_VISAO_UNICA_2 AS A
;RUN;

PROC SQL;
CREATE TABLE NPS.HIST_ZENVIA_VISAO_UNICA AS

	DT_ENVIO,
	CHAVE_CONTRATO,
	CHAVE_CPF,
	NOTA,
	RESPOSTA_QUALITATIVA_CLI,
	CATEGORIA_NPS

;RUN;


PROC SQL;
CREATE TABLE ULT_RESPOSTA_VU AS
SELECT
	A.*,
	(SELECT
		MAX(B.DT_ENVIO)
	FROM NPS.HIST_ZENVIA_VISAO_UNICA  AS B
	WHERE A.CHAVE_CONTRATO = B.CHAVE_CONTRATO
	AND A.CHAVE_CPF = B.CHAVE_CPF
	AND B.DT_ENVIO < A.DT_ENVIO
	AND NOTA >= 0 AND NOTA <=10) AS MAX_DT_ENVIO FORMAT DDMMYY10.
FROM ZENVIA_VISAO_UNICA_3 AS A
WHERE A.NOTA >= 0 AND A.NOTA <=10
;RUN;


PROC SQL;
CREATE TABLE ZENVIA_VISAO_UNICA_3 AS
SELECT DISTINCT
	A.*,
	CASE
		WHEN C.CHAVE_CONTRATO IS NOT NULL THEN 'S'
		ELSE 'N'
	END AS RECORRENTE,
	CASE
		WHEN C.CHAVE_CONTRATO IS NULL THEN ''
		WHEN A.CATEGORIA_NPS <> C.CATEGORIA_NPS THEN 'S'

		ELSE 'N'
	END AS MUDANCA_DE_FAIXA,
	C.DT_ENVIO AS DT_RESP_ANT,
	C.NOTA AS NOTA_ANT,
	C.RESPOSTA_QUALITATIVA_CLI AS RESPOSTA_QUALITATIVA_CLI_ANT,
	C.CATEGORIA_NPS AS CATEGORIA_NPS_ANT
FROM ZENVIA_VISAO_UNICA_3 AS A
LEFT JOIN ULT_RESPOSTA_VU AS B
ON A.CHAVE_CONTRATO = B.CHAVE_CONTRATO
AND A.CHAVE_CPF = B.CHAVE_CPF
AND A.DT_ENVIO = B.DT_ENVIO
LEFT JOIN NPS.HIST_ZENVIA_VISAO_UNICA AS C
ON A.CHAVE_CONTRATO = C.CHAVE_CONTRATO
AND A.CHAVE_CPF = C.CHAVE_CPF
AND C.DT_ENVIO = B.MAX_DT_ENVIO
;RUN;

	NPS.ZENVIA_VISAO_UNICA
	ZENVIA_VISAO_UNICA_3
