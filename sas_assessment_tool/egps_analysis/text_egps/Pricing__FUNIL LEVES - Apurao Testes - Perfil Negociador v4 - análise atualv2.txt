    create table limites_vintis as
    select Vintil_Ponderado,
           min(FG_GH) as Limite_Inferior,
           max(FG_GH) as Limite_Superior
    from acumulada
    group by Vintil_Ponderado
    order by Vintil_Ponderado;
quit;


proc sql;
    create table PROD_GH_FINAL as
    select a.*,
           catx(' - ', put(b.Limite_Inferior, 6.4), put(b.Limite_Superior, 6.4)) as Faixa_VINTIL_GH
    from acumulada a
    left join limites_vintis b
    on a.Vintil_Ponderado = b.Vintil_Ponderado;
quit;


proc freq data=PROD_GH_FINAL;
    tables Faixa_VINTIL_GH / nocum;
