options compress=yes;
PROC SQL BUFFERSIZE=256000;
	sysecho "Gerando base Conta-Cartao do IQPRDBI";

	CONNECT TO sasiosiq AS dbcon
	(HOST="iqprdbi"
	SERVER=iqprdbi
	USER='lgettRiscoPrevencaoFraude'
	PASSWORD='1o4ANrTKKe4isVam');

	CREATE TABLE emer.conta_cartao AS
	SELECT *
	FROM CONNECTION TO DBCON (
		SELECT
			A.*
			,mod(F.CdContaCartao, 1000000000) AS Nu_Conta_Cartao
			,C.DtAberturaContaCartao AS Dt_Abertura_Conta
			,C.DtPrimeiroDesbloqueio AS Dt_Desbloqueio
			,C.NmEstadoContaCartao as Status_ContaCartao_Atual
			,C.NmEstadoContaCartaoAnterior as Status_ContaCartao_Anter
			,C.DtAlteracaoEstadoContaCartao as Dt_Alter_Estado_ContaCartao
			,C.DtPrimeiraCompra as Dt_Primeira_Compra
			,E.CdPropostaCartaoCrdtoStmaOrgm AS Num_Proposta
			,D.NuCPFCNPJ

			,convert(CHAR(6),F.NuBINPlastico)+ '******' + RIGHT(trim(convert(CHAR(17),F.CdPlastico)),4) AS nu_plast_cartao
			,G.NuPlastico
  		    ,G.NuBINPlastico
		    
		    ,F.NuOrdemTitularidadePlastico as Ordem_Titular
			,CASE WHEN F.NuOrdemTitularidadePlastico = 0 THEN 'TITULAR'
       		ELSE F.NuOrdemTitularidadePlastico || ' ADICIONAL' end as Desc_Portador
			,G.NmTipoCartao

		    ,G.DtEmissaoPlastico
			,G.DtDesbloqueioPlastico
			,G.NmEstadoPlastico as Status_PlastCartao
			,G.DtAlteracaoEstadoPlastico as Dt_Alter_Status_PlastCartao
			,G.FlVersaoVigente
			,G.DtValidadePlastico
		FROM (
			SELECT *
			FROM (
				SELECT
					RANK() OVER(partition BY AAA.CdDimContaCartao ORDER BY AAA.CdDimContaCartao, AAA.DtReferencia desc) AS RANK
					,AAA.CdDimContaCartao
					,AAA.CdDimContaCartaoVersao
					,AAA.cdpropostacartaocredito
					,AAA.DtReferencia
					,AAA.VrLimiteContaCartaoGeral
					,AAA.VrSaldoAtual
					,AAA.CdPessoaTitularConta
          		FROM DWBI_DMOC.TbFatoContaCartao AS AAA
				 ) AS AA
          	WHERE AA.rank = 1) AS A
			LEFT JOIN DWBI_DIME.TbDimContaCartao AS C
				ON A.CdDimContaCartaoVersao = C.CdDimContaCartaoVersao
			LEFT JOIN DWBI_DIME.TbDimPropostaCartaoCredito AS E
				ON A.cdpropostacartaocredito = E.cdpropostacartaocredito
			LEFT JOIN DWBI_DIME.TbDimPessoa AS D
				ON A.CdPessoaTitularConta = D.CdPessoa


			LEFT JOIN DWBI_DTWH.TbPlastico AS F
				on A.CdDimContaCartao = F.CdContaCartao
			LEFT JOIN DWBI_DIME.TbDimPlastico AS G
				ON F.cdplastico = G.cddimplastico
			WHERE G.FlVersaoVigente = 'S' and C.DtAberturaContaCartao >= '2018-01-01'
	);
disconnect from DBCON;
QUIT;
options compress=no;

data prop_missing;
sysecho 'Filtrando Casos missing';
set emer.conta_cartao(
where=(Dt_Abertura_Conta >= '01jan2021'd and upcase(num_proposta) in ('WS','')));
run;

proc sql;
sysecho 'Cruzamento Base Unica';
	create table bs_unica as
	select
		strip(put(b.nuproposta, 15.)) as proposta
		,b.nucpfcnpj as cpf_bu
		,a.*
	from work.prop_missing as a
	inner join mis.baseunicacartoes as b
		on a.cddimcontacartao = b.cddimcontacartao
	order by a.cddimcontacartao
	;
quit;
proc sort data=bs_unica out=bs_unica_nodup nodupkey; by cddimcontacartao; run;

%let inicio = %sysfunc(time());

proc sql ;
sysecho 'Update casos no Datamart Conta-Cartao';
	update emer.conta_cartao as a
	set
	num_proposta = (select proposta from work.bs_unica_nodup as b where a.cddimcontacartao = b.cddimcontacartao)

	where cddimcontacartao in (select cddimcontacartao from work.bs_unica_nodup)
	;
quit;
