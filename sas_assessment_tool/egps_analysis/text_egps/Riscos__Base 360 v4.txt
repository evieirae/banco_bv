PROC SQL;
   CREATE TABLE FOR_BS_PEP_1 AS
   SELECT DISTINCT
          t1.*,
          (CASE WHEN t2.CdPessoaSerasaTitular IS MISSING THEN T1.CdPessoaSerasa
					 ELSE T2.CdPessoaSerasaTitular END) AS CdPessoaSerasaFinal,
		  (CASE WHEN t2.CdListaPessoaTitular IS MISSING THEN T1.CdLista
					 ELSE T2.CdListaPessoaTitular END) AS CdListaFinal,
          (CASE WHEN t2.DsGrauRelacionamento IS MISSING THEN 'PEP TITULAR'
					ELSE T2.DsGrauRelacionamento END) AS GrauRelacionamento

       FROM BS_PEP t1
       LEFT JOIN BS_RELA t2 ON (t1.CdPessoaSerasa = t2.CdPessoaSerasaRelacionado)
				AND (t1.CdLista =   t2.CdListaPessoaRelacionado);
QUIT;

PROC SQL;
   CREATE TABLE FOR_BS_PEP AS
   SELECT DISTINCT
          t1.CPF_PEP,
          T1.Nmpessoa as Nome_PEP,
		  T1.GrauRelacionamento,
		  t1.SGUF as UF,
		  T1.DSMUNICIPIO AS MUNICIPIO,
   		  input(Compress(t1.CPF_PEP),14.) FORMAT=BEST16. AS DOC_NUM,
          t3.NmCargo,
		  t1.GrauRelacionamento,
		  'SERASA' AS LISTA,
		  t2.CdOrgaoSerasa,
		  t2.CdListaOrgao,
		  (DATEPART(T2.DtNomeacao)) FORMAT=DATE9. AS Dt_Inicio_Exercicio,
		  (T2.DtExoneracao) FORMAT=DATE9. AS Dt_Fim_Exercicio,
          (T2.DtExoneracao + 1825) FORMAT=DATE9. AS Dt_Final_Carencia,
		  1 AS FLAG

      FROM FOR_BS_PEP_1 t1
           LEFT JOIN BS_MANDATO t2
			ON (t1.CdPessoaSerasaFinal = t2.CdPessoaSerasa) AND (t1.CdListaFinal = t2.CdListaPessoaSerasa)
          LEFT JOIN BS_NMCARGO t3
			ON (t2.CdCargoSerasa = t3.CdCargoSerasa) AND (t2.CdListaCargo = t3.CdLista)
		  ;
QUIT;

PROC SORT DATA=FOR_BS_PEP NODUPKEY;
BY _ALL_;
RUN;

PROC SQL;
CREATE TABLE BS_PEP_SCORE AS
	SELECT DISTINCT
			A.*,
			(CASE WHEN A.NmCargo = 'S/INFO' THEN 27
			      WHEN A.NmCargo NOT= 'S/INFO' AND A.NmOrgao IS MISSING THEN 27
                  WHEN B.Risco = 'Altissimo' THEN 27
			      WHEN B.Risco = 'Alto' THEN 9
                  WHEN B.Risco in ( 'Médio','médio','MÉDIO') THEN 3
				  WHEN B.Risco in ('Baixo','baixo','BAIXO') THEN 1
				  WHEN B.Risco in ('Sem definição com o governo') THEN 0
				  WHEN B.Risco in ('Todos','todos')
									THEN (CASE WHEN C.RISCO_1 = 'Altíssimo' THEN 27
						 					  		 WHEN C.RISCO_1 in('ALTO','Alto') THEN 9
						 					   		 WHEN C.RISCO_1 in ( 'Médio','médio','MÉDIO') THEN 3
						 					   		 WHEN C.RISCO_1 in ('Baixo','baixo','BAIXO') THEN 1 ELSE 100 END)
                 ELSE 100 END)  AS SCORE_PEP_INI,
				 b.risco

	FROM BS_PEP_FULL_MUNI A
		LEFT JOIN De_Para_PEP B
 			ON (A.NmCargo = B.NmCargo) AND (A.NmOrgao = B.NmOrgao)
		LEFT JOIN BS_MUNICIPIOS C
 			ON (A.MUNICIPIO = C.MUNICIPIO) AND (A.UF = C.UF);
QUIT;

PROC FREQ DATA=BS_PEP_SCORE ; TABLES SCORE_PEP_INI; RUN ;

PROC SQL;
CREATE TABLE BS_PEP_SCORE AS
	SELECT DISTINCT
			A.*,
			(CASE WHEN A.GrauRelacionamento = 'PEP TITULAR' THEN A.SCORE_PEP_INI
				  ELSE (CASE WHEN SCORE_PEP_INI = 27 THEN 9
				  		     WHEN SCORE_PEP_INI = 9 THEN 3
							 WHEN SCORE_PEP_INI = 3 THEN 1
							 WHEN SCORE_PEP_INI = 0 THEN 0
							 WHEN SCORE_PEP_INI = 100 THEN 100
                             WHEN SCORE_PEP_INI = 1 THEN 1 ELSE 0 END)END) AS SCORE_PEP_FIM

	FROM BS_PEP_SCORE A
		;
QUIT;

PROC FREQ DATA=BS_PEP_SCORE ; TABLES SCORE_PEP_INI*SCORE_PEP_FIM; RUN ;

PROC SQL;
   CREATE TABLE BS_ESTEIRA_AGRUPADO AS
   SELECT DISTINCT
          (CASE WHEN t1.ESTEIRA IN ('PRIVATE','PRIVATE/DTVM','PRIVATE_DTVM') THEN 'PRIVATE_DTVM'
		        WHEN t1.ESTEIRA IN ('PME','SME') THEN 'PME'
				WHEN t1.ESTEIRA IN ('SAUDE') THEN 'CDC_SAUDE'
                WHEN t1.ESTEIRA IN ('SUB_BAAS') THEN 'BAAS' ELSE t1.ESTEIRA END) FORMAT $50. AS ESTEIRA,
		   YEAR(t1.ATUALIZACAO) AS ANO,
          (CASE WHEN t1.ATUALIZACAO = '31MAR2023'D THEN '30MAR2023'D ELSE t1.ATUALIZACAO END) FORMAT DATE9. AS ATUALIZACAO,
          t1.TPPESSOA,
          t1.QTDE
      FROM PLD.BS_ESTEIRA_AGRUPADO t1
	  	WHERE t1.ESTEIRA NOT IN ('PEAC','PRAVALER','SEGUROS','PROCURADOR','PROMOTIVA','CONTA_CORRENTE')

;QUIT;
