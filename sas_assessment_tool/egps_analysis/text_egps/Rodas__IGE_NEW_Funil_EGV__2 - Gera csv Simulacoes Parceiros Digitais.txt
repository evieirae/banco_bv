/* =====================Gera csv parceiros digitais=========================*/
/* ======================================================================================*/

PROC SQL;
   CREATE TABLE PDEC_PARCEIROS AS
   SELECT
		t1.nuanomes,
		t1.NuCpfCnpj,
		t1.DtSimulacaoInicial,
		t1.CdSimulacaoInicial,
		t1.CdParceiroComercial,
		t1.NmParceiroComercial,
		t1.DsTipoCanalProducao,
		t1.FlSimulacao,
		t1.FlSimuValida,
		t1.FlSimuValidaEfet,
		t1.FlPropostaOriginada,
		t1.FlPropostaAprovada,
		t1.FlPropostaEfetivada,
		t1.FlPropostaPaga
      FROM SOLITMP.EGV_PDEC_CREDITO t1
	  WHERE t1.nuanomes >= year(today())*100+month(today())-1
	  AND t1.flparceirodigital = 'S'
 ;
QUIT;

%let dir_csv = /riscosbvf/BA/processos/output;
/* FINANZERO */

DATA TbFinanzero;
	set PDEC_PARCEIROS
	(where=(UPPER(NmParceiroComercial) CONTAINS 'FINANZERO'));
