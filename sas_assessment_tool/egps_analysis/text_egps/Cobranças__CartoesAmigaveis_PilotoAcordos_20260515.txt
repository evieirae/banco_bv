    when FX_ATRASO = 'ATE 30 DIAS DE ATRASO' and modelo_perfil_rolagem = 'Alto'  and GH_Novo_26 >  5  then 2
    when FX_ATRASO = 'ATE 30 DIAS DE ATRASO' and modelo_perfil_rolagem = 'Medio' and GH_Novo_26 <= 6  then 2
    when FX_ATRASO = 'ATE 30 DIAS DE ATRASO' and modelo_perfil_rolagem = 'Medio' and GH_Novo_26 >  6  and GH_Novo_26 < 11 then 3
    when FX_ATRASO = 'ATE 30 DIAS DE ATRASO' and modelo_perfil_rolagem = 'Medio' and GH_Novo_26 >= 11 then 4

    when FX_ATRASO = 'ATE 30 DIAS DE ATRASO' and modelo_perfil_rolagem = 'Baixo' and GH_Novo_26 >= 10 then 5

    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Alto'  and GH_Novo_26 >= 1  then 1
    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Medio'  and GH_Novo_26 <= 9  then 1
	when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Medio' and  GH_Novo_26 > 9  and GH_Novo_26 <= 11  then 2
    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Medio' and GH_Novo_26 > 11  then 3
    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Baixo' and GH_Novo_26 <= 9 then 3
    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Baixo' and GH_Novo_26 >  9 and GH_Novo_26 <= 15 then 4
    when FX_ATRASO = 'FLUXO' and modelo_perfil_rolagem = 'Baixo' and GH_Novo_26 = 16 then 5

	ELSE . END AS CENARIO_NOVO_MODELO_26

PROC SQL;
CREATE TABLE RDG_MODELO2 AS
SELECT *
,CASE
      WHEN  RDG_NOVO_COLLECTION <= 19 THEN 'MODELO NOVO 5-30'
      WHEN  RDG_NOVO_COLLECTION <= 89 THEN 'MODELO ANTIGO 5-30'
	  WHEN  RDG_NOVO_COLLECTION > 89 THEN 'CONTROLE MODELO 5-30'
	  ELSE 'ERRO'
	  END AS GRUPO_MODELO
FROM COLL_CENARIOS_3
;QUIT;





PROC SQL;
CREATE TABLE RDG_MODELO3 AS
SELECT *
,CASE
      WHEN GRUPO_MODELO = 'MODELO NOVO 5-30' THEN GH_NOVO_26
      WHEN GRUPO_MODELO = 'MODELO ANTIGO 5-30' THEN GH_ANTIGO_26
	  WHEN GRUPO_MODELO = 'CONTROLE MODELO 5-30' THEN RAND("INTEGER",1,16)
	  ELSE . END AS GH_CORRETO

,CASE WHEN GRUPO_MODELO = 'MODELO NOVO 5-30' THEN CENARIO_NOVO_MODELO_26
      WHEN GRUPO_MODELO = 'MODELO ANTIGO 5-30' THEN CENARIO_MODELO_ANTIGO_26
	  WHEN GRUPO_MODELO = 'CONTROLE MODELO 5-30' THEN RAND("INTEGER",1,5)
	  ELSE . END AS CENARIO_CORRETO

FROM RDG_MODELO2
;QUIT;

PROC SQL;
CREATE TABLE PARCELADOS_ACEL_ACORDO_TAXAS AS
SELECT
	A.CdDimContaCartao,
	A.NUCONTA,
	A.CDPRODUTO,
	A.CdFatura,
	A.VrSaldoFatura,
	A.DtPostagemTransacao,
	A.QtTotalParcelaCompra,
	A.CdRubricaTransacaoCartao,
	A.DsTransacao,
	A.DtReferencia,
	A.FlParcelado,
	A.VrTransacao,
	A.MaiorParcelaPaga,
	COALESCE(B.VrContratoAdesao,C.VALOR_PRINC) AS VrContrato,
	COALESCE(B.DtAceleracao,C.DtAceleracao) AS DtAceleracao FORMAT DATE9.,
	COALESCE(B.VrParcelaAdesao, C.VALOR_PMT, A.VrTransacao) AS VrPMT,
	COALESCE(B.TxRateAdesao, C.TAXA)/100 AS Taxa
FROM PARCELADOS_ACEL_ACORDO A
LEFT JOIN TbContratoParcelamentoFatura B
	ON A.CdDimContaCartao=B.CdContaCartao AND A.QtTotalParcelaCompra=B.NuParcelaAdesao AND A.DtPostagemTransacao=B.DtAdesao AND A.CdRubricaTransacaoCartao IN (6061, 6212, 6078)
LEFT JOIN ACORDOS_CA_UNICO C
	ON A.CdDimContaCartao=(C.PRODUTO*1e9+C.NUCONTA) AND A.DtPostagemTransacao=C.DATA_EFET AND A.QtTotalParcelaCompra=C.QTDE_PARCELAS AND A.CdRubricaTransacaoCartao IN (6053)
;QUIT;

DATA PARCELADOS_ACEL_ACORDO_2;
	SET PARCELADOS_ACEL_ACORDO_TAXAS;
	ParcelasRestantes = QtTotalParcelaCompra-MaiorParcelaPaga;
	ValorRestante = VrPMT*ParcelasRestantes;
	IF DtAceleracao >= DtAcordo AND NOT MISSING(Taxa) THEN VrAceleracoes=(VrPMT*(1-(1+Taxa)**(MaiorParcelaPaga-QtTotalParcelaCompra)))/(Taxa);
	ELSE IF MISSING(Taxa) THEN VrAceleracoes=ValorRestante;
	ELSE VrAceleracoes=0;
	IF DtReferencia >= DtAcordo THEN VrLancamentos = VrTransacao+VrAceleracoes;
	ELSE VrLancamentos=VrAceleracoes;
RUN;
