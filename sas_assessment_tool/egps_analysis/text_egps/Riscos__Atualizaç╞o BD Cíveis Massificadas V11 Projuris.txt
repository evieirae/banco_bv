PROC SQL;
	create table WORK.PAGAMENTO_ACORDO (index=(Cd_Causa)) as
		select distinct t1.Cd_Causa,
						'2.Acordo'				length=15			as Desfecho,
						
						sum(-VlLcto)			format=commax22.2	as Valor,
					
						min(t1.DtPagamento) 	format=ddmmyy10. 	as DtPrimPagto,
						max(t1.DtPagamento) 	format=ddmmyy10. 	as DtUltPagto
		from BDPORTAL.LCTOS_TOTAL t1
		where t1.TipoLcto like '%ACORDO%' and t1.TpVerba = '1' and t1.NmAmbiente = 'VAC'
		group by t1.Cd_Causa
		having t1.TipoLcto not in ('PAGAMENTO DE ACORDO - PÓS CONDENAÇÃO','PAGAMENTO DE ACORDO - PÓS CONDENAÇÃO - DIRETO',
									'REEMBOLSO - PAGAMENTO DE ACORDO PÓS CONDENAÇÃO','PAGAMENTO ACORDO PÓS CONDENAÇÃO – TRANSF JUDICIAL'
									'PAGAMENTO DE CONDENAÇÃO - AÇÃO CONTRÁRIA - DIRETO');
QUIT;
PROC SQL;
	create table WORK.PAGAMENTO_CONDENACAO_PRIM (compress=yes) as
		select distinct t1.Cd_Causa,
						'3.Condenação'			length=15			as Desfecho,
					
						-VlLcto					format=commax22.2	as Valor,
					
						t1.DtPagamento			format=ddmmyy10.	as DtPagto
		from BDPORTAL.LCTOS_TOTAL t1
		where (t1.TipoLcto like '%CONDENA__O%' or t1.TipoLcto contains 'SUCUMB' or t1.TipoLcto contains 'PAGAMENTO - MULTA DIARIA' or t1.TipoLcto contains 'BLOQUEIO DE TRANSFERÊNCIA')
			and t1.TpVerba = '1' and t1.NmAmbiente = 'VAC'
		order by t1.Cd_Causa, t1.DtPagamento;
QUIT;
PROC SORT 	data = WORK.PAGAMENTO_CONDENACAO_PRIM
			out = WORK.PAGAMENTO_CONDENACAO_PRIM (compress=yes index=(Cd_Causa/unique))
			nodupkey;
		by Cd_Causa;
RUN;
PROC SQL;
	create table WORK.PAGAMENTO_CONDENACAO (index=(Cd_Causa)) as
		select distinct t1.Cd_Causa,
						'3.Condenação'			length=15			as Desfecho,
				
						t2.Valor				format=commax22.2	as ValorPrim,
						(sum(-VlLcto)-t2.Valor)	format=commax22.2	as ValorCompl,
						sum(-VlLcto)			format=commax22.2	as ValorTotal,
				
						min(t1.DtPagamento) 	format=ddmmyy10. 	as DtPrimPagto,
						max(t1.DtPagamento) 	format=ddmmyy10. 	as DtUltPagto

		from BDPORTAL.LCTOS_TOTAL t1
			left join WORK.PAGAMENTO_CONDENACAO_PRIM as t2 on (t1.Cd_Causa = t2.Cd_Causa)
		where (t1.TipoLcto like '%CONDENA__O%' or t1.TipoLcto contains 'SUCUMB' or t1.TipoLcto contains 'PAGAMENTO - MULTA DIARIA' or t1.TipoLcto contains 'BLOQUEIO DE TRANSFERÊNCIA')
			and t1.TpVerba = '1' and t1.NmAmbiente = 'VAC'
		group by t1.Cd_Causa;
QUIT;
DATA WORK.PAGAMENTOS_TOTAL (compress=yes drop=Valor);
	set WORK.PAGAMENTO_ACORDO
		WORK.PAGAMENTO_CONDENACAO;
RUN;
PROC SORT data=WORK.PAGAMENTOS_TOTAL;
	by descending DtUltPagto;
RUN;
PROC SORT data=WORK.PAGAMENTOS_TOTAL (compress=yes index=(Cd_Causa/unique)) nodupkey;
	by Cd_Causa;
RUN;
PROC SQL;
	create table WORK.PERDA_ZERO (compress=yes) as
		select 	t1.NumRegSistContrGest 	length=10	as Cd_Causa,
				('1.Perda Zero')		length=15	as Desfecho,
				(0)									as NumPgtos,
				(.)									as DtPrimOcorr,
				t1.DtStatusAJ						as DtUltOcorr,
				(.)									as DtPrimPagto,
				t1.DtStatusAJ						as DtUltPagto
		from BDPO.BDPORTAL_CONSOLIDADA as t1
		where t1.StatusAJ in ('ENCERRADO-ÊXITO','ENCERRADO-ACORDO','ENCERRADO-CONDENAÇÃO') and
			  t1.VlPrincipal = 0  and
			  t1.AreaReporte = 'Jurídico BV Financeira - Ações Contrárias';
QUIT;

		select 	t1.Cd_Causa,
				(MIN(t1.DtAndamento)) format=ddmmyy10. as DtPrimAndamento,
				(MAX(t1.DtAndamento)) format=ddmmyy10. as DtUltAndamento
		from BDPOR2.ANDAMENTO_PROJURIS t1

		where fc.FasePadrao = 'INICIAL'
		group by t1.Cd_Causa;

PROC SQL;
   CREATE TABLE WORK.GARANTIAS_DEP AS
   SELECT DISTINCT t1.CD_CAUSA,
            (SUM(t1.VlLancamentoGarantiaAtualizacao)) FORMAT=21.2 AS VL_DEP,
			pasta
      FROM dbep.TBLPROCESSOGARANTIA t1
      WHERE t1.CdUnidadeExyon = 'VAC'
		AND t1.CdSistema = 'p'
		AND t1.TipoGarantia IN
           (
           'DEPOSITO JUDICIAL',
           'TRANSFERÊNCIA JUDICIAL'
           )
      GROUP BY t1.CD_CAUSA;
QUIT;

PROC SQL;
   CREATE TABLE WORK.GARANTIAS_SEGURO AS
   SELECT DISTINCT t1.CD_CAUSA,
            (SUM(t1.VlLancamentoGarantiaAtualizacao)) FORMAT=21.2 AS VL_DEP,
			pasta,
			t1.DtGarantia
      FROM dbep.TBLPROCESSOGARANTIA t1
      WHERE t1.CdUnidadeExyon = 'VAC'
		AND t1.CdSistema = 'p'
		AND t1.TipoGarantia IN
           (
		   'SEGURO GARANTIA',
		   'FIANÇA BANCÁRIA'
           )
      GROUP BY t1.CD_CAUSA, t1.DtGarantia
      order by t1.DtGarantia desc;
QUIT;

proc sort data=work.GARANTIAS_SEGURO OUT=work.GARANTIAS_SEGURO NODUPKEY;
BY PASTA;
RUN;

run;

PROC SQL;
	create table WORK.ANDAM_FI (compress=yes index=(Cd_Causa/unique)) as
		select 	t1.Cd_Causa,
				(MIN(t1.DtAndamento)) format=ddmmyy10. as DtPrimAndamento,
				(MAX(t1.DtAndamento)) format=ddmmyy10. as DtUltAndamento
		from BDPOR2.ANDAMENTO_PROJURIS t1
			left join WORK.FASES_CIVEIS as fc on (t1.NmAndamento=fc.Fase)
		where fc.FasePadrao = 'INICIAL'
		group by t1.Cd_Causa;

	create table WORK.ANDAM_FS (compress=yes index=(Cd_Causa/unique)) as
		select 	t1.Cd_Causa,
				(MIN(t1.DtAndamento)) format=ddmmyy10. as DtPrimAndamento,
				(MAX(t1.DtAndamento)) format=ddmmyy10. as DtUltAndamento
		from BDPOR2.ANDAMENTO_PROJURIS t1
			left join WORK.FASES_CIVEIS as fc on (t1.NmAndamento=fc.Fase)
		where fc.Fase = 'SENTENÇA'
		group by t1.Cd_Causa;

	create table WORK.ANDAM_FR (compress=yes index=(Cd_Causa/unique)) as
		select 	t1.Cd_Causa,
				(MIN(t1.DtAndamento)) format=ddmmyy10. as DtPrimAndamento,
				(MAX(t1.DtAndamento)) format=ddmmyy10. as DtUltAndamento
		from BDPOR2.ANDAMENTO_PROJURIS t1
			left join WORK.FASES_CIVEIS as fc on (t1.NmAndamento=fc.Fase)
		where fc.FasePadrao = 'RECURSAL'
		group by t1.Cd_Causa;

PROC SQL;
   CREATE TABLE WORK.SUBSIDIO_PROJURIS AS
   SELECT t1.Cd_Causa AS NumRegSistContrGest,
          t1.NmAndamento AS Fase,
          t1.DtAndamento,
		  t2.Motivo,
		  t2.Submotivo,
          t3.Classific
      FROM WORK.SUBSIDIO_PROJURIS t1
           LEFT JOIN BDCIVEL.MASSIFICADO t2 ON (t2.NumRegSistContrGest = t1.Cd_Causa)
           LEFT JOIN JUR.MOT_SUBMOT_CLASSIF t3 ON (t2.Motivo = t3.Motivo) AND (t2.Submotivo = t3.Submotivo)
      ORDER BY t1.Cd_Causa, t1.DtAndamento desc;
QUIT;

PROC SQL;
   CREATE TABLE WORK.COMPLEXO AS
   SELECT t1.NumRegSistContrGest,
          t1.Fase,
          t1.DtAndamento,
          t1.Classific,
		  CASE
				WHEN  t1.Classific = 'COMPLEXO' AND t1.Fase = 'SUBSÍDIOS CONCLUÍDOS  DESFAVORAVEIS À BVF' THEN 'DESFAVORÁVEL'
				WHEN  t1.Classific = 'COMPLEXO' AND t1.Fase = 'SUBSÍDIOS CONCLUÍDOS  FAVORAVEIS À BVF' THEN 'FAVORÁVEL'
				ELSE '' END AS Subsidios
      FROM WORK.SUBSIDIO_PROJURIS t1
      WHERE t1.Classific = 'COMPLEXO' AND t1.Fase IN ('SUBSÍDIOS CONCLUÍDOS  DESFAVORAVEIS À BVF', 'SUBSÍDIOS CONCLUÍDOS  FAVORAVEIS À BVF');
QUIT;

PROC SORT DATA=WORK.COMPLEXO NODUPKEY OUT=WORK.COMPLEXO;
	BY NumRegSistContrGest;
QUIT;

PROC SQL;
	CREATE TABLE WORK.SIMPLES (COMPRESS=YES) AS
		SELECT t1.NumRegSistContrGest,
			   t1.DtAndamento,
			   t1.Fase,
          	   t1.Classific,
				 CASE
					WHEN  t1.Classific = 'SIMPLES' AND t1.FASE CONTAINS 'DESFAVORÁVEL' THEN 'DESFAVORÁVEL'
					WHEN  t1.Classific = 'SIMPLES' AND t1.FASE CONTAINS 'FAVORÁVEL' THEN 'FAVORÁVEL'
					WHEN  t1.FASE CONTAINS 'DESFAVORÁVEL' OR t1.FASE CONTAINS 'DESFAVORAVEIS' OR t1.FASE CONTAINS 'DESFAVORÁVEIS' OR t1.FASE CONTAINS 'DESFAVORAVEL' THEN 'DESFAVORÁVEL'
					WHEN  t1.FASE CONTAINS 'FAVORÁVEL' OR t1.FASE CONTAINS 'FAVORAVEL' OR t1.FASE CONTAINS 'FAVORAVEIS' OR t1.FASE CONTAINS 'FAVORÁVEIS' THEN 'FAVORÁVEL'
					WHEN  t1.FASE CONTAINS 'Prazo Expirado - Defesa' OR T1.FASE CONTAINS 'DESACORDO COMERCIAL - DESFAVORÁVEL' THEN 'DESFAVORÁVEL'
					ELSE '' END AS Subsidios
			FROM WORK.SUBSIDIO_PROJURIS t1
			LEFT JOIN WORK.COMPLEXO t2 ON (t1.NumRegSistContrGest=t2.NumRegSistContrGest)
			WHERE t2.NumRegSistContrGest = ''
			ORDER BY t1.NumRegSistContrGest, t1.DtAndamento desc, t1.Fase desc;
QUIT;

PROC SORT DATA=WORK.SIMPLES NODUPKEY OUT=WORK.SIMPLES;
	BY NumRegSistContrGest;
QUIT;

DATA WORK.MASSIFICADO_PARECER1  (COMPRESS=YES);
	SET COMPLEXO SIMPLES;
RUN;
