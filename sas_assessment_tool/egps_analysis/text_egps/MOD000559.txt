%LET ANOMES = &sysparm.;

PROC SQL;
   CREATE TABLE BASE_ATRASO AS
   SELECT &ANOMES. as DTBASE,
   		  input(NuCpfCnpj,14.) as NuCpfCnpj,
		  t1.NuContrato,
          t1.SqContratoFinanceiroAtivo,
          t1.Atraso
      FROM _CARTEIR.CARTEIRA_SD2_&ANOMES_AUX. t1;
QUIT;

%MACRO APPEND(REF);

	DATA BHV_AUX;
		SET _BHV.BHS_CONTRATOS_PDDBEHAVIOR_&REF. (KEEP=NuContrato
														NuCpfCnpj
														RU
														Produto);
		IF Produto IN ('Leves','Motos','Pesados');
		DTBASE = &REF.;
	RUN;

	DATA BHV;
		SET BHV BHV_AUX;
		IF NOT MISSING(NuContrato);
	RUN;

	PROC SQL;
		DROP TABLE BHV_AUX;
	QUIT;

%MEND;

%macro monthly(func,first,last);
%do mon=&first %to &last;
    %if %substr(&mon,5)=13 %then %let mon=%eval(%substr(&mon,1,4)+1)01;
    %&func(&mon);
%end;
%mend monthly;

DATA BHV;
RUN;

%monthly(APPEND,&ANOMES.,&ANOMES.);

PROC SQL;
   CREATE TABLE BASE_ATRASO_4 AS
   SELECT A.*,B.RU AS BEHAVIOR,B.Produto

      FROM BASE_ATRASO A
      LEFT JOIN BHV B
	  ON (A.NuContrato = B.NuContrato) AND (A.DTBASE = B.DTBASE)
;
QUIT;

/* 2. Score */

/**************************************************************/
/******** Ajuste devido a mudança do modelo de BS Leves *******/
/**************************************************************/
data BASE_ATRASO_41 (drop=BEHAVIOR);
set BASE_ATRASO_4;
format RU_ajust $char8.;
if Produto = 'Leves' then do;
	IF BEHAVIOR = 'BS5.5' THEN RU_ajust = 'BS5';
	else IF BEHAVIOR = 'BS6.5' THEN RU_ajust = 'BS6';
	else IF BEHAVIOR = 'BS7' THEN RU_ajust = 'BS7.5';
	else IF BEHAVIOR = 'BS8.5' THEN RU_ajust = 'BS8';
	else IF BEHAVIOR = 'BS10' THEN RU_ajust = 'BS9';
    else RU_ajust = BEHAVIOR;
end;
else do;
RU_ajust = BEHAVIOR;
end;
BS_ant = BEHAVIOR;
run;

data BASE_ATRASO_42;
set BASE_ATRASO_41;
BEHAVIOR = RU_ajust;
run;

/* DEFINIÇÃO DOS CLUSTERS */

DATA BASE_ATRASO_10;

	SET BASE_ATRASO_42;

	IF Produto = 'Leves' THEN DO;

		IF BEHAVIOR = 'BS1' THEN Cluster = '1';
		IF BEHAVIOR = 'BS1.5' THEN Cluster = '1';
		IF BEHAVIOR = 'BS2' THEN Cluster = '2';
		IF BEHAVIOR = 'BS3' THEN Cluster = '3';
		IF BEHAVIOR = 'BS3.5' THEN Cluster = '4';
		IF BEHAVIOR = 'BS4.5' THEN Cluster = '6';
		IF BEHAVIOR = 'BS5' THEN Cluster = '7';
		IF BEHAVIOR = 'BS6' THEN Cluster = '7';
		IF BEHAVIOR = 'BS7.5' THEN Cluster = '7';
		IF BEHAVIOR = 'BS8' THEN Cluster = '7';
		IF BEHAVIOR = 'BS9' THEN Cluster = '7';
		IF BEHAVIOR = 'Def' THEN Cluster = '1';
		IF BEHAVIOR = 'SemRelac' THEN Cluster = '5';

	END;

	ELSE IF PRODUTO = 'Motos' THEN DO;

		IF BEHAVIOR = 'BS1' THEN Cluster = '1';
		IF BEHAVIOR = 'BS1.5' THEN Cluster = '1';
		IF BEHAVIOR = 'BS2' THEN Cluster = '2';
		IF BEHAVIOR = 'BS3' THEN Cluster = '3';
		IF BEHAVIOR = 'BS3.5' THEN Cluster = '6';
		IF BEHAVIOR = 'BS4' THEN Cluster = '6';
		IF BEHAVIOR = 'BS4.5' THEN Cluster = '7';
		IF BEHAVIOR = 'BS5' THEN Cluster = '7';
		IF BEHAVIOR = 'BS6' THEN Cluster = '7';
		IF BEHAVIOR = 'BS7.5' THEN Cluster = '7';
		IF BEHAVIOR = 'Def' THEN Cluster = '1';
		IF BEHAVIOR = 'SemRelac' THEN Cluster = '5';

	END;

	ELSE DO;

		IF BEHAVIOR = 'BS1' THEN Cluster = '1';
		IF BEHAVIOR = 'BS2' THEN Cluster = '1';
		IF BEHAVIOR = 'BS3' THEN Cluster = '4';
		IF BEHAVIOR = 'BS4.5' THEN Cluster = '6';
		IF BEHAVIOR = 'BS6' THEN Cluster = '6';
		IF BEHAVIOR = 'BS6.5' THEN Cluster = '5';
		IF BEHAVIOR = 'BS7' THEN Cluster = '7';
		IF BEHAVIOR = 'BS8' THEN Cluster = '7';
		IF BEHAVIOR = 'Def' THEN Cluster = '1';
		IF BEHAVIOR = 'SemRelac' THEN Cluster = '5';

	END;

	IF MISSING(Cluster) THEN Cluster = '1';

RUN;

DATA BASE_ATRASO_12A;

	SET BASE_ATRASO_10;

	IF CLUSTER IN ('3','4') THEN CLUSTER = '3';
	IF CLUSTER IN ('5','6') THEN CLUSTER = '6';

RUN;

/* corrige clusters */
PROC SQL;
   CREATE TABLE clusters AS
   SELECT DISTINCT t1.Cluster
      FROM WORK.BASE_ATRASO_12A t1;
QUIT;

data clusters;
	set clusters;
	Score = _N_;
run;

PROC SQL;
	CREATE TABLE BASE_ATRASO_12B AS
	SELECT A.*,B.Score
	FROM BASE_ATRASO_12A AS A
	LEFT JOIN clusters AS B
	ON A.CLUSTER = B.CLUSTER
	;
QUIT;
