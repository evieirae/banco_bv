/*Gerando a base dos Participantes com o Realizado e Premiação*/

Proc Sql;
Create table Base_Producao_PDVs_G3e4 as
Select Distinct
Participantes.NuAnomes as NuAnomes_Campanha
,DtAtualizacao
,Participantes.Canal
,Participantes.DsLoginN1
,Participantes.Gerente
,Participantes.Grupo
,Participantes.Faixa_Efetivacao
,Producao.NuAnoMes
,Producao.NuCnpjLojaFavorecido
,Producao.NmParceiroComercialFavorecido
,Sum(Producao.VrProducao) as VrProducao
,Count(NuContrato) as Qt_Contratos
,Sum(Producao.FlBvMais) as Qt_SPF
,Sum(Producao.FlCAP) as Qt_AP
,Sum(Producao.FlPlacaSolar) as Qt_PlacaSolar
,Calculated Qt_SPF + Calculated Qt_AP + Calculated Qt_PlacaSolar as Qt_Seguros
,Case
	When Participantes.Faixa_Efetivacao = '1 a 2 meses sem efetivar' and  Calculated Qt_Contratos < 3 Then Calculated Qt_Contratos * 100
	When Participantes.Faixa_Efetivacao = '1 a 2 meses sem efetivar' and Calculated Qt_Contratos >= 3 Then Calculated Qt_Contratos * 150

	When Participantes.Faixa_Efetivacao = '3 a 4 meses sem efetivar' and  Calculated Qt_Contratos < 3 Then Calculated Qt_Contratos * 250
	When Participantes.Faixa_Efetivacao = '3 a 4 meses sem efetivar' and Calculated Qt_Contratos >= 3 Then Calculated Qt_Contratos * 350

	When Participantes.Faixa_Efetivacao = 'Mais de 4 meses sem efetivar' and Calculated Qt_Contratos < 3 Then Calculated Qt_Contratos * 250
	When Participantes.Faixa_Efetivacao = 'Mais de 4 meses sem efetivar' and Calculated Qt_Contratos >= 3 Then Calculated Qt_Contratos * 350

	When Participantes.Faixa_Efetivacao = 'Nunca efetivou' and Calculated Qt_Contratos < 3 Then Calculated Qt_Contratos * 250
	When Participantes.Faixa_Efetivacao = 'Nunca efetivou' and Calculated Qt_Contratos >= 3 Then Calculated Qt_Contratos * 350

	Else 0
End as Premio_Contratos

,Case
	When Participantes.Faixa_Efetivacao = '1 a 2 meses sem efetivar' and Calculated Qt_Seguros > 0 Then Calculated Qt_Seguros * 25
	When Participantes.Faixa_Efetivacao = '3 a 4 meses sem efetivar' and Calculated Qt_Seguros > 0 Then Calculated Qt_Seguros * 50
	When Participantes.Faixa_Efetivacao = 'Mais de 4 meses sem efetivar' and Calculated Qt_Seguros > 0 Then Calculated Qt_Seguros * 50
	When Participantes.Faixa_Efetivacao = 'Nunca efetivou' and Calculated Qt_Seguros > 0 Then Calculated Qt_Seguros * 50
	Else 0
End as Premio_Seguros

From Base_PDV_Participantes as Participantes
Inner join tmp.TBPRODUCAO as Producao
 	   	On Producao.NuCnpjLojaFavorecido = Participantes.CNPJ and Participantes.NuAnomes = Producao.NuAnomes
Where Producao.NmGrupoProdutoMeta = 'Solar' and Participantes.Grupo in ('Grupo 3', 'Grupo 4')
Group By 1,2,3,4,5,6,7,8,9
;Quit;

/*Calculando a pontuação de cada Gr nos devidos Grupos/PDVs*/
Proc Sql;
Create Table Base_Campanha_G3e4_Regras as
Select Distinct
NuAnomes
,DtAtualizacao
,Canal
,Gerente
,DsLoginN1
,Grupo
,Sum(Qt_Contratos) as Qt_Contratos
,Sum(Qt_Seguros) as Qt_Seguros
,Sum(Premio_Contratos) as Premio_Contratos
,Sum(Premio_Seguros) as Premio_Seguros

,Sum(
		Case
		 When Faixa_Efetivacao = '1 a 2 meses sem efetivar' Then Qt_Contratos * 1
		 When Faixa_Efetivacao = '3 a 4 meses sem efetivar' Then Qt_Contratos * 3
		 When Faixa_Efetivacao = 'Mais de 4 meses sem efetivar' Then Qt_Contratos * 3
		 When Faixa_Efetivacao = 'Nunca efetivou' Then Qt_Contratos * 3
		 Else 0

End) as Total_Pontos
From Base_Producao_PDVs_G3e4
Group by 1,2,3,4,5
;Quit;

/*Reralizando o Ranking para calculo das Milhas*/
PROC SORT DATA=Base_Campanha_G3e4_Regras OUT=Base_Campanha_G3e4_Regras_RKG;
BY NuAnoMes Grupo DESCENDING Total_Pontos;
;RUN;

proc rank data=Base_Campanha_G3e4_Regras_RKG descending out=Base_Campanha_G3e4_Regras_RKG2 ties=dense;
    var Total_Pontos;
    ranks RKG_Pontos;
    by NuAnoMes Grupo;
run;

/*Gerando base final com os Pontos, Premiações e Milhas*/
Proc Sql;
Create Table Base_Campanha_G3e4_Final as
Select
NuAnomes
,Canal
,Gerente
,DsLoginN1
,Grupo
,Qt_Contratos
,Qt_Seguros
,Premio_Contratos
,Premio_Seguros
,Total_Pontos
,RKG_Pontos
,Case
	When RKG_Pontos = 1 then 15
	When RKG_Pontos = 2 then 13
	When RKG_Pontos = 3 then 11
	When RKG_Pontos = 4 then 9
	When RKG_Pontos = 5 then 7
	When RKG_Pontos = 6 then 5
	When RKG_Pontos = 7 then 3
	When RKG_Pontos >= 8 then 1
	Else 0
End as Milhas
,DtAtualizacao

From Base_Campanha_G3e4_Regras_RKG2
;Quit;
