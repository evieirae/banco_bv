CREATE TABLE BASE_TICKET_FINAL_M AS
SELECT distinct

strip(a.NmAreaEstruturaComercialC)||a.FxIdade as Chave,
a.NmAreaEstruturaComercialC,
a.ChaveCdParceiro_FxIdade,

COALESCE(SUM(a.VrProducao_ProjSab) / SUM(a.QtdContratos_ProjSab),0) AS Ticket_BV_ProjSab FORMAT COMMAX20.2,
COALESCE(SUM(a.Vrgravamesplayers_ProjSab) / SUM(a.qtgravamesplayers_ProjSab),0) AS Ticket_Mercado_ProjSab FORMAT COMMAX20.2,

case when CALCULATED Ticket_BV_ProjSab <=0 then max(b.Ticket_BV_ProjSab)
			else CALCULATED Ticket_BV_ProjSab end as Passo1,

CAse when CALCULATED Passo1 <=0 then CALCULATED Ticket_Mercado_ProjSab
	 when CALCULATED Ticket_Mercado_ProjSab <=0 then max(b.Ticket_BV_ProjSab)
	 else CALCULATED Passo1 end as V1,  /*valida*/

coalesce( max(C.Ticket_BV_ProjSab),max(c.Ticket_Mercado_ProjSab))  as TicketRegional,

CALCULATED V1 / CALCULATED TicketRegional as Multiplo,

Case when
		CALCULATED Ticket_BV_ProjSab <= 0 AND CALCULATED Multiplo >=3 <=0 then CALCULATED TicketRegional
	  else
		case when CALCULATED Multiplo >=3 and CALCULATED Ticket_BV_ProjSab <= 0 then CALCULATED TicketRegional else CALCULATED V1 end
		End as ValorFinal
