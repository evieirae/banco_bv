proc sql;
	create table contratos as
	select contr.NuContratoText,
		   contr.NmProduto,
		   dicprod.NmProdutoGrupo3,
		   contr.DtInicioVigencia,
		   contr.DtFimVigencia
	from yribeiro.tblcontratosiq as contr
	left join yribeiro.tblproduto as dicprod
		on contr.NmProduto = dicprod.NmProduto
	where dicprod.NmProdutoGrupo3 = "VEICULOS";
quit;


proc sql;
	select min(DtInicioVigencia) as min_inic_date into:inicio
	from contratos;

	select max(DtFimVigencia) as max_fim_date into:fim
	from contratos;
quit;

%put &inicio.;
%put &fim.;


data meses(keep = mes);
	format inicio yymmd.;
	format fim yymmd.;
	format mes yymmd.;

	inicio = intnx('month', min(&inicio.), 0, 'same');
	fim = intnx('month', max(&fim.), 0, 'same');
	n_meses = intck("month", inicio, fim);

	do i = 0 to intck("month", inicio, fim) by 1;
		mes = intnx("month", inicio, i, "e");
		output;
	end;
run;


proc sql;
	create table contratos_ativos_por_mes as
	select m.mes,
		   count(c.NuContratoText) as qtd_contratos_ativos format commax25.
	from meses as m
	left join contratos as c
		on c.DtInicioVigencia <= intnx('month', m.mes, 0, 'same')
			and c.DtFimVigencia >= intnx('month', m.mes, 0, 'same')
	group by m.mes;
quit;
