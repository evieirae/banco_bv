proc sql;
create table TbOriginacao_AgentPDVMotos as
select
 NmSubEstruturaComercialC as Regional
,NmAreaEstruturaComercialC as Filial
,NmColaboradorC 		as GR
,cats(CdParceiroComercial,'-',NmParceiroComercial)	as PDV
, case  	when nuanomes = &SafraM7. then 'M7'
		when nuanomes = &SafraM6. then 'M6'
		when nuanomes = &SafraM5. then 'M5'
		when nuanomes = &SafraM4. then 'M4'
		when nuanomes = &SafraM3. then 'M3'
		when nuanomes = &SafraM2. then 'M2'
		when nuanomes = &SafraM1. then 'M1'
		when nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,sum(case when FlConcorrente =1 then 1 else 0 end) as Qtdconcorrente
,sum(case when FlPropostaAPNP =1  then 1 else 0 end) as QtdAPNP
,sum(case when FlPropostaEfetivada=0 then 1 else 0 end) as QtdEfetivada
from tmp.tboriginacao as a

where  1=1
and nuanomes >= &SafraM7.
AND A.NmGrupoProdutoMeta in ('Motos')
and NmTipoParceiroComercial in ('Loja','Concessionaria')
and NmSubEstruturaComercialC in('Canal Motos')
/*and FxIdade <> 'ZeroKM'*/
/*and NmAreaEstruturaComercialC in ('Canal Goiânia','Canal Cascavel','Canal ABC','Canal Porto Alegre')*/
/*and CdParceiroComercial IN( 3939 )*/
/*and nuanomes = 202411*/
group by 1,2,3,4,5
;run;

proc sql;
create table TbProducao_AgentMotos as
select
 NmSubEstruturaComercialC as Regional
,NmAreaEstruturaComercialC as Filial
,NmColaboradorC 		as GR
,cats(CdParceiroComercial,'-',NmParceiroComercial)	as PDV
,case  	when nuanomes = &SafraM7. then 'M7'
		when nuanomes = &SafraM6. then 'M6'
		when nuanomes = &SafraM5. then 'M5'
		when nuanomes = &SafraM4. then 'M4'
		when nuanomes = &SafraM3. then 'M3'
		when nuanomes = &SafraM2. then 'M2'
		when nuanomes = &SafraM1. then 'M1'
		when nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,DsTipoCanalProducao as Canal
,Count(NuContrato)  as QtdContratos
,Sum(coalesce(VpSpread,0)) / sum (coalesce(Vrproducao,0)) as Spread
,SUM(FlCap) as QtdSegAp
,SUM(FlBVMais) as QtdSegSPF
,SUM(FlCompleto) as QtdSegCompleto
,SUM(QtDemaisSemCompleto) as QtSegDemais
,(coalesce(SUM(CASE WHEN doc.NuContratoGestao IS NOT NULL THEN 1 ELSE 0 END), 0) /
 coalesce(SUM(CASE WHEN a.NuContrato IS NOT NULL THEN 1 ELSE 0 END), 0)) AS PercRG
from tmp.tbproducao as a
		LEFT JOIN TipoDoc as doc
		on doc.NuContratoGestao = a.NuContrato
where  1=1
and nuanomes >= &SafraM7.
AND A.NmGrupoProdutoMeta in ('Motos')
and NmTipoParceiroComercial in ('Loja','Concessionaria')
and NmSubEstruturaComercialC in ('Canal Motos')
/*and FxIdade <> 'ZeroKM'*/
/*and NmAreaEstruturaComercialC in ('Canal Goiânia','Canal Cascavel','Canal ABC','Canal Porto Alegre')*/
/*and CdParceiroComercial IN( 7665 ,469165 )*/
/*and nuanomes = 202411*/
group by 1,2,3,4,5,6
order by 1,2,3,4,5,6
;run;

PROC SORT DATA=TbProducao_AgentMotos ;
BY Regional Filial GR PDV;
RUN;

proc sql;
create table Rito_AgentMotos as
select
 TbRito.NmSubEstruturaComercialC as Regional
,TbRito.NmAreaEstruturaComercialC as Filial
,TbRito.NmColaboradorC 		as GR
,TbRito.NmParceiroComercial	as PDV
,case  	when TbRito.nuanomes = &SafraM7. then 'M7'
		when TbRito.nuanomes = &SafraM6. then 'M6'
		when TbRito.nuanomes = &SafraM5. then 'M5'
		when TbRito.nuanomes = &SafraM4. then 'M4'
		when TbRito.nuanomes = &SafraM3. then 'M3'
		when TbRito.nuanomes = &SafraM2. then 'M2'
		when TbRito.nuanomes = &SafraM1. then 'M1'
		when TbRito.nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,Case when TBFOCO_PDV.Indicado = 1 then 'S' end as FocoPDV
,(Sum (coalesce(Part_APNP,0))+
 		Sum (coalesce(Part_Cancelada,0))+
		Sum (coalesce(Part_Card,0))+
		Sum (coalesce(Part_SemCard,0))+
		Sum (coalesce(Part_ContratosBV,0))+
		Sum (coalesce(Part_Negada,0))) /  Sum(coalesce(Part_Mercado,0)) as PartJogo
, Sum(Part_ContratosBV) /  ( Sum(Part_ContratosBV)+
								Sum(Part_Card)+
								Sum(Part_APNP)) as VirouBV

,Sum(coalesce(Part_ContratosBV,0))  / Sum(coalesce(Part_Mercado,0)) as Share
,Sum(coalesce(Part_Mercado,0)) as GravamesMercado
,Sum(coalesce(Part_ContratosBV,0)) as GravamesBV
,Sum (coalesce(Part_APNP,0)) as QtdApnp
,Sum (coalesce(Part_Card,0)) - Sum(coalesce(Qtd_Sem_Prop_MesmaLjRec,0)) as QtdCardPositivo
,(Sum(coalesce(Qtd_Efetivadas,0)) / 	(Sum(coalesce(Qtd_Efetivadas,0)) +
						Sum (coalesce(Qtd_Sem_Prop_MesmaLj,0)) +
						Sum (coalesce(Qtd_APNP_Conco_MesmaLj,0)))) as ShareEntrada
,sum(coalesce(MedidaInad30,0)) / sum(coalesce(DenominadorInad30,0)) as Inad30
,sum(coalesce(MedidaInad15,0)) / sum(coalesce(DenominadorInad15,0)) as Inad15
from basegrc.TbRitoMotos as TbRito

	left join basegrc.TBFOCO_PDV as TBFOCO_PDV
	on TbRito.CdparceiroComercial = TBFOCO_PDV.CdparceiroComercial
	and TbRito.nuanomes = TBFOCO_PDV.nuanomes
where  1=1
/*and CdParceiroComercial IN(120184)*/
/*and nuanomes = 202411*/
group by 1,2,3,4,5,6
;run;

PROC SORT DATA=Rito_AgentMotos ;
BY Regional Filial GR PDV ;
RUN;

proc sql;
create table BancosRankingPDV as
Select
 case  	when nuanomes = &SafraM7. then 'M7'
		when nuanomes = &SafraM6. then 'M6'
		when nuanomes = &SafraM5. then 'M5'
		when nuanomes = &SafraM4. then 'M4'
		when nuanomes = &SafraM3. then 'M3'
		when nuanomes = &SafraM2. then 'M2'
		when nuanomes = &SafraM1. then 'M1'
		when nuanomes = &SafraM0. then 'M0'
	end as nuanomes
,NmSubEstruturaComercialC as Regional
,NmAreaEstruturaComercialC	as  Filial
,NmColaboradorC 		as GR
,NmParceiroComercial as PDV
/*,FxEntradaMinima	*/
,Banco
,Sum(coalesce(Qtd_APNP_Conco_MesmaLoja,0)) + Sum(coalesce(Qtd_Sem_Prop_MesmaLoja,0))  as TotalConcorrencia
from basegrc.ResumoSimulacoesRito as cb
where 1=1
and banco not in ('N/A','BV')
/*and CdParceiroComercial  IN( 7665 ,469165 )*/
/*and FxEntradaMinima = '01 - até 10%'*/
/*and NuAnoMes = 202502*/
group by 1,2,3,4,5,6
ORDER by 1,2,3,4 ,7 desc
;quit;

PROC SORT DATA=BancosRankingPDV ;
BY NuAnoMes GR PDV ;
RUN;

data BancosRanking_TopPDV;
    set BancosRankingPDV (Where=(TotalConcorrencia > 0));
    by  NuAnoMes GR PDV ;
    retain Top 0;
    if first.PDV then Top = 0;
    Top + 1;
	BancoQtd = strip(Banco)||'('||strip(TotalConcorrencia)||')'; /* Nova coluna concatenada */
	TopBanco = 'Banco'||strip(Top)||'_';
run;

PROC SORT DATA=BancosRanking_TopPDV ;
BY Regional Filial GR PDV Top;
RUN;

/* Transformação dos dados */
proc transpose data=BancosRanking_TopPDV out=BancosRankingTranspose_Top (drop=_NAME_);
    by  Regional Filial GR PDV ;
    id TopBanco NuAnoMes ;
    var Banco ;
	Where Top <=3;
run;

;RUN;

/* Adicionando colunas fixas em branco caso nao tenha M0 */
