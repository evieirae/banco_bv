		  when input(a.dmdays,BEST4.)+1 <= 180 then '0121 A 0180'
		  when input(a.dmdays,BEST4.)+1 <= 360 then '0181 A 0360'
		  when input(a.dmdays,BEST4.)+1 <= 540 then '0361 A 0540'
		  when input(a.dmdays,BEST4.)+1 <= 900 then '0541 A 0900' else '0901 A 9999' end as FX_ATRASO

		  when a.dmagency = '4002' then 'REALE'
		  when a.dmagency = '4003' then 'ZANC'
		  when a.dmagency = '4005' then 'RN ADV'
		  when a.dmagency = '4006' then 'BELLINATI'
		  when a.dmagency = '4007' then 'CRC'
		  when a.dmagency = '4008' then 'BELLINATI II'
		  when a.dmagency = '4009' then 'TOLEDO II'
		  when a.dmagency = '4062' then 'CONCILIG'
		  when a.dmagency = '4010' then 'P_DIGITAL'
										end as ASSESSORIA
	,a.dmque

from connection to perm (
	select distinct
	dmacct
	,dmssnum
	,dmdays
	,dmcurbal
	,dmamtdlq
	,dmagnref
	,dmagency
	,dmacctg
	,dmque
	,dmstatus
	,dmqchgdt
	from bv-pdbd-prd.trusted_rcvry.delqmst) as a

	u3acct
	,u3dtpriatr
	,u3dtenqcont
	,u3score1
	,u3score2
	,u3cdleg
	from bv-pdbd-prd.trusted_rcvry.uda3) as c on a.dmacct = c.u3acct

and a.dmstatus is null

	DROP SEEDED_CONTRATO HASH_INPUT_A HASH_INPUT_B HASH L
