		ParcelaEstimada = Vr_Parcela;
		VlrDevidoTotal = MoraEstimada + MultaEstimada + ParcelaEstimada;
		LiquidacaoAntecipada = 0;

	end; else if Atraso <= 0 then do;

		MoraEstimada = 0;
		MultaEstimada = 0;
		ParcelaEstimada = ROUND(Vr_Parcela/((1+TxInternaRetorno3)**(-Atraso)),0.01);
		VlrDevidoTotal = MoraEstimada + MultaEstimada + ParcelaEstimada;
		LiquidacaoAntecipada = Vr_Parcela - ParcelaEstimada;

	end;

	If VrRecebido > ParcelaEstimada then do;
			DescParcelaEstimado = 0;
			ReceitaEncargosEstimado = VrRecebido - ParcelaEstimada;
	end; else do;

			DescParcelaEstimado = ROUND(VrRecebido - ParcelaEstimada, 0.01);
			ReceitaEncargosEstimado = 0;
	end;

RUN;

proc sql;
  create table CONSULTA_RECEB_FLEX_COM_MORA as
  select
      t.*

    
    , (today() - t.dt_vencimento)                                                    as DIAS_ATRASO                       format=best12.

    
    , floor(t.tx_comissao_permanencia_mes * 10) / 1000                               as TX_MORA_MES_TRUNC                 format=12.6

    
    , floor( t.vr_parcela * calculated TX_MORA_MES_TRUNC * 100 ) / 100              as CP_MENSAL                         format=12.2
    , floor( (calculated CP_MENSAL / 30) * 100 ) / 100                               as CP_DIA                            format=12.2

    
    , case
        when missing(t.tx_tir) then .
        
        when t.tx_tir > 1 then t.tx_tir / 100
        else t.tx_tir
      end                                                                            as TX_TIR_ANUAL_FRAC                 format=12.8

    , case
        when calculated TX_TIR_ANUAL_FRAC > 0
          then ( (1 + calculated TX_TIR_ANUAL_FRAC)**(1/365) - 1 )
        else .
      end                                                                            as TX_TIR_DIA                        format=12.10

    
    , case
        when calculated DIAS_ATRASO > 0
             then floor( (calculated CP_DIA * calculated DIAS_ATRASO) * 100 ) / 100
        else 0
      end                                                                            as VL_MORA_RECALC                    format=12.2 label='Mora (recalc)'

    
    , case
        when calculated DIAS_ATRASO < 0
             and not missing(calculated TX_TIR_DIA)
