/*Cria Base de Colaborados Ativos da Estrutura de Distribuição Varejo*/

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbColaboradoresAtivos as
select *
from connection to dbcon
(SELECT HierarquiaComercial.NmEstruturaComercial
,HierarquiaComercial.NmSubEstruturaComercial
,HierarquiaComercial.NmAreaEstruturaComercial
,Colaborador.CdColaboradorRH as IdMatricula
,HierarquiaComercial.CdColaborador as CdColaboradorBI
,HierarquiaComercial.NmColaborador
,HierarquiaComercial.NmCargoColaborador
,HierarquiaComercial.CdEstadoColaborador
,HierarquiaComercial.NmEstadoColaborador
,HierarquiaComercial.DtNascimento
,HierarquiaComercial.DsLogin as DsLogin
,HierarquiaComercial.DsEmail as DsEmail
,HierarquiaComercial.NmColaboradorImediato
,HierarquiaComercial.DsLoginColaboradorImediato
,HierarquiaComercial.DtAdmissaoColaborador
FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
LEFT JOIN DWBI_DTWH.TbColaborador Colaborador
ON  HierarquiaComercial.CdColaborador = Colaborador.CdPessoa
AND Colaborador.CdSistemaOrigem = 202
AND today() BETWEEN Colaborador.DtInicioVigencia and Colaborador.DtFimVigencia
WHERE today() BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
AND YEAR(HierarquiaComercial.DtInicioVigencia) BETWEEN YEAR(today())-1 AND YEAR(today())
AND HierarquiaComercial.CdEstadoColaborador not in (4,10)
AND HierarquiaComercial.CdEstruturaComercial = 8
order by HierarquiaComercial.NmSubEstruturaComercial,
HierarquiaComercial.NmAreaEstruturaComercial,
HierarquiaComercial.NmColaborador
)
;
run;
quit;

proc sql;
create table TbColaboradoresCodigos as
select * from bases.tbcolaborador
where CdColaboradorBI > 0 and CdGrupoCargoColaborador > 0 and CdOperador > 0
;run;

/*Gera base final de colaboradores*/

proc sql;
create table TbColaboradorDistribuicao as
select
Compress(TbColaboradoresAtivos.NmEstruturaComercial,,"kw") AS NmEstruturaComercial
,Compress(TbColaboradoresAtivos.NmSubEstruturaComercial,,"kw")AS NmSubEstruturaComercial
,TbColaboradoresCodigos.CdFilial
,Compress(TbColaboradoresAtivos.NmAreaEstruturaComercial,,"kw")AS NmAreaEstruturaComercial
,TbColaboradoresAtivos.IdMatricula
,TbColaboradoresCodigos.CdColaborador
,TbColaboradoresCodigos.CdOperador as CdURA
,Compress(TbColaboradoresAtivos.NmColaborador,,"kw")AS NmColaborador
,Compress(TbColaboradoresAtivos.NmCargoColaborador,,"kw")AS NmCargoColaborador
,TbColaboradoresAtivos.CdEstadoColaborador
,Compress(TbColaboradoresAtivos.NmEstadoColaborador,,"kw")AS NmEstadoColaborador
,TbColaboradoresAtivos.DtNascimento
,Compress(TbColaboradoresAtivos.DsLogin,,"kw") as DsLogin
,Compress(TbColaboradoresAtivos.DsEmail,,"kw") as DsEmail
,Compress(TbColaboradoresAtivos.NmColaboradorImediato,,"kw")AS NmColaboradorImediato
,Compress(TbColaboradoresAtivos.DsLoginColaboradorImediato,,"kw")AS DsLoginColaboradorImediato
,TbColaboradoresAtivos.DtAdmissaoColaborador
from TbColaboradoresAtivos
left join TbColaboradoresCodigos
    on TbColaboradoresAtivos.CdColaboradorBI = TbColaboradoresCodigos.CdColaboradorBI
;run;

/*Exporta tabela final*/

proc export
data=TbColaboradorDistribuicao
file='/riscosbvf/BA/processos/tmp/TbColaboradorDistribuicao.csv' replace
dbms=dlm;
delimiter=';';
run;
