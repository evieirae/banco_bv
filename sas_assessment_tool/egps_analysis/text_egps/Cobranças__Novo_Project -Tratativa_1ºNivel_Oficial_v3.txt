CONNECT TO SYBASE (SERVER=PTASYBCTR DATABASE=DBCOR  USER=&IDCTR.  PASSWORD=&SENHACTR.);

	CREATE TABLE BASE_HIST_ENVIO   AS 	SELECT DISTINCT

	    G.NUCONTRATOGESTAO,
	    G.SQCONTRATOFINANCEIRO,
		G.CD_NATUREZA_SOLICITACAO,
		G.COD_ASSESSORIA,
		G.ASSESSORIA,
		G.COD_ACAO,
		G.DT_ENVIO FORMAT DDMMYY10.,
		G.DT_RETIRADA FORMAT DDMMYY10.,
		G.FLAG_STATUS_COB

	from temp A LEFT JOIN CONNECTION TO SYBASE 
	(SELECT

	    C.NUCONTRATOGESTAO,
	    C.SQCONTRATOFINANCEIRO,
		A.CD_NATUREZA_SOLICITACAO,
		BA_ASSES.COD_ASSES AS COD_ASSESSORIA,

	FROM DBSLT..TB_SOLICITACAO A LEFT JOIN DBSLT..TB_NATUREZA_SOLICITACAO B
	ON A.CD_NATUREZA_SOLICITACAO=B.CD_NATUREZA_SOLICITACAO

	LEFT JOIN DBSLT..TB_NTZ_SLT_TB_AREA_TB_FILIAL X

	ON B.CD_NATUREZA_SOLICITACAO=X.CD_NATUREZA_SOLICITACAO

	LEFT JOIN DBBVF..TB_AREA Y
	ON X.CD_AREA=Y.CD_AREA

	LEFT JOIN DBSLT..TB_SLT_CLIENTE C
	ON A.NU_SOLICITACAO=C.NU_SOLICITACAO

	LEFT JOIN SCC_FIN..SAJ_PROCESSO BA_ASSES
	ON
	C.NUCONTRATOGESTAO=BA_ASSES.NUCONTRATOGESTAO 

	LEFT JOIN SCC_FIN..SAJ_AGENTE F
	ON BA_ASSES.COD_ASSES=F.COD_AGENTE

	WHERE y.nm_area in ('Atendimento 4 – Célula Avançada','Controles e Qualidade','TSA')
	AND BA_ASSES.COD_ACAO IN (1,3,4,5,59,60,39,278,279,281)
	AND A.DT_SOLICITACAO BETWEEN &REFDTINICIO. AND &REFDTFIM.) G

	ON 1;

PROC SQL;
CREATE TABLE  SALESFORCE_FINAL_CARTAO AS SELECT DISTINCT

	A.NU_SOLICITACAO AS SOLICITACAO,
	A.DMACCT AS CONTRATO,
	A.NUCPFCNPJ,
	A.NOME_CLIENTE,
	A.DS_NATUREZA_SOLICITACAO AS TAREFA,
	'CONTROLE E QUALIDADE' AS CANAL,

	CASE
		WHEN DS_NATUREZA_SOLICITACAO IN ('Cliente não consegue contato com assessoria','Cobrança Indevida','Cobrança Vexatória'
		'Divergências de acordo','Excesso de cobrança','Negociação','Solicitação gravação cobrança','Suspensão de Cobrança') THEN DATEPART(DTENVIO)

		WHEN DS_NATUREZA_SOLICITACAO IN ('Solicitação gravação cobrança','Excesso de cobrança','Cobrança Vexatória', 'Cobrança Indevida')
		AND COD_ASSESS_ATU="" THEN DATEPART(DTENVIO_ANT)
		END AS ENVIO_EPS FORMAT DDMMYY10.,

	A.EPS FORMAT $60.,

	DATEPART(A.DT_SOLICITACAO) AS DATA_ABERTURA FORMAT DDMMYY10.,
	DATEPART(A.DT_EFETIVA_ENCERRAMENTO) AS DATA_ENCERRAMENTO FORMAT DDMMYY10.,
	A.NOME_SOLICITANTE,
	'CONTROLE E QUALIDADE' AS AREA_RESPONSAVEL ,
	TRIM(A.HISTORICO) AS HISTORICO

FROM BASE_FINAL_CARTAO A;
QUIT;

			WHEN C.NUCONTRATOGESTAO IS NOT NULL THEN C.DT_RETIRADA

			WHEN C.NUCONTRATOGESTAO IS NOT NULL THEN C.FLAG_STATUS_COB

CREATE TABLE BASE_EXTRATO_SAS AS SELECT
	*
FROM BASE_EXTRATO_SAS A
WHERE  NOT EXISTS (SELECT *  FROM TAREFA_DE_BUSCA B WHERE B.NU_SOLICITACAO=A.NU_SOLICITACAO)
AND NOT EXISTS (SELECT *  FROM TAREFA_1416_SEM_ASSES C WHERE C.NU_SOLICITACAO=A.NU_SOLICITACAO)

QUIT;

			HISTORICO='-'||'Por gentileza disponibilizar as ligações, histórico de acionamento, whatsapp e ações massivas realizadas para este contrato no período dos últimos 15 dias.';
	END;

ELSE

	IF DS_NATUREZA_SOLICITACAO IN ('Reclamação Reneg') THEN DO;
			EPS=ASSESSORIA;
			HISTORICO='-'||'Por gentileza encaminhar a gravação ou histórico do WhatsApp que foi ofertado a renegociação do contrato. O contrato está elegível a renegociação, favor contatar o cliente para manter a proposta de reneg nas condições abaixo:';
	END;

ELSE

	IF DS_NATUREZA_SOLICITACAO IN ('Alteração da data de vencimento – clientes em atraso', 'Carta de Anuência- 1ª VIA'
									,'Gravame Bloqueado/Bloqueio Judicial', 'Envio termo entrega amigável', 'Extrato Saldo Remanescente') THEN DO;
			EPS=ASSESSORIA;
			HISTORICO='-'||'Sem historico';
	END;

ELSE

	IF DS_NATUREZA_SOLICITACAO IN ('Desconhece negativação','Ultrapassou Prazo de Baixa (5 dias úteis)') THEN DO;
			EPS='MUTANT';
			HISTORICO='-'||'Sem historico';
	END;

ELSE

	IF DS_NATUREZA_SOLICITACAO IN ('Reclamação contrato cedido') THEN DO;
			EPS=NOME_CESSIONARIA;
			HISTORICO='-'||'Sem historico';
	END;

	RUN;

	DATA BASE_FINAL;
	SET BASE_FINAL_SAS
		BASE_FINAL_SALESfORCE;
RUN;
