PROC SQL;
create table work.total_safra as
select ANO_CTO,
		SUM(QT_CONTRATOS) AS PRODUCAO,
   		CASE WHEN ANO_CTO=2010 THEN INTCK("MONTH",'1JAN11'D,TODAY())
		WHEN ANO_CTO=2011 THEN INTCK("MONTH",'1JAN12'D,TODAY())
		WHEN ANO_CTO=2012 THEN INTCK("MONTH",'1JAN13'D,TODAY())
		WHEN ANO_CTO=2013 THEN INTCK("MONTH",'1JAN14'D,TODAY())
		WHEN ANO_CTO=2014 THEN INTCK("MONTH",'1JAN15'D,TODAY())
		WHEN ANO_CTO=2015 THEN INTCK("MONTH",'1JAN16'D,TODAY())
		WHEN ANO_CTO=2016 THEN INTCK("MONTH",'1JAN17'D,TODAY())
		WHEN ANO_CTO=2017 THEN INTCK("MONTH",'1JAN18'D,TODAY())
		WHEN ANO_CTO=2018 THEN INTCK("MONTH",'1JAN19'D,TODAY())
		WHEN ANO_CTO=2019 THEN INTCK("MONTH",'1JAN20'D,TODAY())
   		WHEN ANO_CTO=2020 THEN INTCK("MONTH",'1JAN21'D,TODAY())
   		WHEN ANO_CTO=2021 THEN INTCK("MONTH",'1JAN22'D,TODAY())
		WHEN ANO_CTO=2022 THEN INTCK("MONTH",'1JAN23'D,TODAY())
		WHEN ANO_CTO=2023 THEN INTCK("MONTH",'1JAN24'D,TODAY())
		WHEN ANO_CTO=2024 THEN INTCK("MONTH",'1JAN25'D,TODAY())
		WHEN ANO_CTO=2025 THEN INTCK("MONTH",'1JAN26'D,TODAY())
		WHEN ANO_CTO=2026 THEN -1
		ELSE . END AS Max_IntMesEntrada_safra
FROM WORK.SUMARIO
GROUP BY ANO_CTO;
QUIT;

		T1.IntMesEnt,
		T1.QT_CONTRATOS,
		T1.QT_CAUSAS,
		T2.PRODUCAO AS Produção,

			when T1.IntMesEnt=. then 0
			when T2.Max_IntMesEntrada_safra < T1.IntMesEnt then 0 else 1 end as Max_IntMesEntrada_safra
		from WORK.SUMARIO T1
		left join	work.total_safra T2
		on T1.ANO_CTO = T2.ANO_CTO;
QUIT;

   SELECT t1.NuContrato,

PROC SQL;
   CREATE TABLE CTOS_ENRIQUECIDO (COMPRESS=YES) AS
   SELECT T1.NUCONTRATO,
          T1.DTCONTRATO,
		  YEAR(DtContrato) AS ANO_CTO,
          T1.DTLIQUIDACAOEFETIVA,
		  T1.UF,
		  T1.NmPessoa,
          T1.DTCONT,
          T1.CD_CAUSA,
          T2.DTDETEC AS DT_CADASTRAMENTO,
		  MDY(MONTH(T2.DTDETEC),1,YEAR(T2.DTDETEC)) FORMAT DDMMYY10. AS SAFRA_CADASTRO,
		  t1.NmModalidadeProduto,
          T2.MOTIVO,
          T2.SUBMOTIVO,
          T2.MOTIVOPADRAO,

		  INTCK("MONTH",T1.DTCONTRATO,T2.DTDETEC) AS IntMesEnt

      FROM WORK.CTOS_ENRIQUECIDO T1
           LEFT JOIN BDPO.BDPORTAL_CONSOLIDADA T2 ON (T1.CD_CAUSA = T2.NUMREGSISTCONTRGEST);
QUIT;
