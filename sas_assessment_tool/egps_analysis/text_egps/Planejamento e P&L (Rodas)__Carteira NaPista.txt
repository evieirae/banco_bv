PROC SQL;
CREATE TABLE WORK.PARCEIROSATIVOS AS
SELECT

A.CdParceiroComercial,
A.NuCnpjLoja,
A.DsAtividadeFinanciamento,
A.NmFilialC,
A.NmTipoParceiroComercial,
A.NmAutoShopping,
A.NmGrupoConcessionaria,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN A.NmGrupoConcessionaria
	WHEN A.NmAutoShopping <> '' THEN A.NmAutoShopping
	WHEN A.NmTipoParceiroComercial <> '' THEN A.NmTipoParceiroComercial
	ELSE 'Outros'
END) AS NmClassificacaoGrupoParceiro,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN 'GRANDES GRUPOS'
	WHEN A.NmAutoShopping <> '' THEN 'AUTO SHOPPING'
	WHEN A.NmTipoParceiroComercial <> '' THEN UPCASE(A.NmTipoParceiroComercial)
	ELSE 'OUTROS'
END) AS NmClassificacaoTipoGrupoParceiro,

A.NmParceiroComercial,
A.CdUf,
A.CdColaboradorC,
A.NmColaboradorC,
A.DsLoginN1C,
B.NmGrupoCargoColaborador,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.DtCadastro,
A.FlAtivo

FROM BASES.TBPARCEIROCOMERCIAL A

LEFT JOIN BASES.TBCOLABORADOR B ON B.DSLOGIN = A.DsLoginN1C AND B.CDCOLABORADOR = A.CdColaboradorC

WHERE
A.FlAtivo = 'S'
AND A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.DsAtividadeFinanciamento = 'VEÍCULO'
AND B.NmGrupoCargoColaborador = 'Gerente Relacionamento Veiculos'

AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV BALCAO%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'RENG SAO JOSE DO RIO PRETO' 
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'CA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJAS RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV MESA%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV REMANESCENTE%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE '%BV PESADOS RENEG%'
AND upcase(CALCULATED NmClassificacaoGrupoParceiro) NOT IN ('PARCEIRO DE INDICAÇÃO - CANAL PRÓPRIO')
AND upcase(NmParceiroComercial) NOT LIKE '%RENEG%'
AND upcase(NmParceiroComercial) NOT LIKE '%BV CP%'
AND upcase(NmParceiroComercial) NOT LIKE '%BV PARTICULAR%'
AND upcase(NmParceiroComercial) NOT LIKE '%RENG%'



ORDER BY 1

;RUN;
