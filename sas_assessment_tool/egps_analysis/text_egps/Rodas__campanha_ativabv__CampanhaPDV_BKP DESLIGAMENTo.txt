proc sql;
create table TbVisaoCampanhaPDV as
select	distinct
	 Inativos.NuCnpjLoja
	,Sum(Case when Inativos.NuAnoMes IN (202502,202501,202412)
			then Inativos.QtGravamesMercado else 0 end) /3 as Gravames3MCampanha
	,CASE WHEN Inativos.TpCnae = 'VEICULOS'  and calculated Gravames3MCampanha <= 3	   THEN 'Apto'
		  WHEN Inativos.TpCnae <> 'VEICULOS'  THEN 'Apto'
		  Else 'Inapto' end as ProspeccaoCN
from tmp.tbpotencialrodas Inativos
where 1=1
/*and Inativos.NuAnoMes IN (202502,202501,202412)*/
and Inativos.FxIdade <> 'ZeroKM'
and Inativos.NmGrupoProduto = 'Leves'
and  Inativos.NuCnpjLoja <> ''
/*and  Inativos.NuCnpjLoja  IN('01192333001528', '49708811000130')*/
group by 1
;run;

proc sql;
create table BaseMcDesempate as
select distinct
NuAnoMes,
DsLoginN1,
sum(Case when (NmGrupoProdutoMeta in ('Leves','Motos','Pesados')) then VrMedidaNI end)
	as VrMedidaNI_RD,

CASE WHEN DsTipoCanalProducao = 'CANAL VEICULOS' THEN
sum(Case when (NmGrupoProdutoMeta in ('Leves','Motos','Pesados')
		 and StVeiculo <> ('Novos')) then VrMetaNIN1 end)
+
sum(Case when StVeiculo in ('Novos') then VrMetaN1 end)
ELSE sum(Case when (NmGrupoProdutoMeta in ('Leves','Motos','Pesados')) then VrMetaN1 end)
END	as VrMetaN1_Rd,

calculated VrMedidaNI_RD /calculated  VrMetaN1_Rd   as  AtingimentoRodas,
/*;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;*/
sum(Case when (NmGrupoProdutoMeta in ('CVG Leves','CVG Pesados')
		 ) then VrMedida end)
	as VrMedidaNI_Egv,
sum(Case when (NmGrupoProdutoMeta in ('CVG Leves','CVG Pesados')
		 and StVeiculo <> ('Novos')) then VrMetaN1 end)
	as VrMetaN1_EGV,
calculated VrMedidaNI_Egv /calculated  VrMetaN1_EGV   as  AtingimentoEGV
from bases.TBMETAPRODUCAOFINAL as a
where 1=1
/*and DsLoginN1 in ('alessandro.melo')*/
/*and NmGrupoProdutoMeta in ('Leves','Motos','Pesados')*/
and NmGrupoProdutoMeta in ('CVG Leves','CVG Pesados','Leves','Motos','Pesados')
and NuAnoMes >= 202503
and TpValor = 'VrMargemContribuicao'
group by 1,2
;run;

proc sql;
create table RankingGRDesempate as
select distinct
a.*,
Case when Filial like '%Canal de Neg%' then Sum(b.AtingimentoRodas,b.AtingimentoEGV)
else Sum(b.AtingimentoRodas) end as Atingimento
/*coalesce (b.AtingimentoRodas, (b.AtingimentoEGV+b.AtingimentoRodas)) as Atingimento*/
from RankingGr as a
	left join BaseMcDesempate as b
	on a.NuAnoMes = b.NuAnoMes
		and a.DsLoginN1 = b.DsLoginN1
where 1=1
/*and a.DsLoginN1 in ('luciano.silva','thiago.lage')*/
group by 1,2,3,4
;run;

PROC SORT data=RankingGRDesempate ;
  BY NuAnoMes
     Filial
	DESCENDING Contratos
	DESCENDING Atingimento
	DsLoginN1;
RUN;

    by  NuAnoMes Filial   ;

	if Top = 1
;run;
