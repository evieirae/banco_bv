PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE TbProposta AS
SELECT * FROM connection to dbcon(
SELECT
	 year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) NuAnoMes
	,DATE(Proposta.DtEntradaProposta) AS DtEntradaProposta
	,DATE(Proposta.DtStatusProposta) AS DtStatusProposta
	,Proposta.StProposta
    ,SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0) CdGrupoSubProduto
	,Proposta.NuProposta
	,Proposta.NuContratoGestao
	,Proposta.CdLoja CdParceiroComercial
	,PessoaSobAnalise.TpPessoa
	,Proposta.CdSubProduto
	,PessoaSobAnalise.NuCpfCnpj

FROM DBA.TbProposta Proposta

LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
	ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
	AND PessoaSobAnalise.CdPapel = 1

LEFT JOIN DBA.TbSubProduto SubProduto
	ON Proposta.CdSubProduto = SubProduto.CdSubProduto

LEFT JOIN DBA.TbLoja Loja
	ON Proposta.CdLoja = Loja.CdLoja

LEFT JOIN DBA.TbRegiao Regiao
	ON  Loja.CdRegiao = Regiao.CdRegiao

LEFT JOIN DBA.TbParceiroComercial ParceiroComercial
	ON Proposta.CdLoja = ParceiroComercial.CdParceiroComercial

WHERE Proposta.TpProposta <> '3'
	AND DATE(Proposta.DtEntradaProposta) >=  DATEADD(DAY, -30, TODAY())
	AND Proposta.NuProposta <> 0
	AND Proposta.StProposta IN ('A','AP')
)
;QUIT;

proc sql;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage);
create table StatusConta as
select * from connection to perm(
WITH
  tb1 AS (
  SELECT
    controle.cdcontrolestatuscadastro,
    controle.dtcontrolestatuscadastro,
    controle.dtalteracao,
    controle.cdstatuspreenchimentocadastro,
    status.dsstatuspreenchimentocadastro,
    pessoa.nucpfcnpj,
    pessoa.nmpessoa,
    pessoa.tppessoa,
    ROW_NUMBER() OVER(PARTITION BY pessoa.nucpfcnpj ORDER BY controle.cdcontrolestatuscadastro DESC) ROW
  FROM
    `bv-pdbd-prd.trusted_dbglobal.tbcontrolestatuscadastro` controle
  INNER JOIN
    `bv-pdbd-prd.trusted_dbglobal.tbpessoa` pessoa
  ON
    controle.cdpessoa = pessoa.cdpessoa
  LEFT JOIN `bv-pdbd-prd.trusted_dbglobal.tbstatuspreenchimentocadastro` status
  ON status.cdstatuspreenchimentocadastro = controle.cdstatuspreenchimentocadastro
  WHERE
    TIMESTAMP_TRUNC(dtcontrolestatuscadastro, MONTH) >= TIMESTAMP_TRUNC(TIMESTAMP_SUB(CURRENT_TIMESTAMP(),INTERVAL 30 DAY),MONTH) )
SELECT
  a.cdcontrolestatuscadastro,
  a.dtcontrolestatuscadastro,
  a.dtalteracao dtalteracaofim,
  a.cdstatuspreenchimentocadastro,
  a.dsstatuspreenchimentocadastro,
  a.nucpfcnpj,
  a.nmpessoa,
  a.tppessoa,
  b.dtcontrolestatuscadastro dtcontrolestatuscadastroini
FROM
  tb1 a
LEFT JOIN (
  SELECT
    nucpfcnpj,
    MIN(dtcontrolestatuscadastro) dtcontrolestatuscadastro
  FROM
    tb1
  GROUP BY
    1) b
ON
  a.nucpfcnpj = b.nucpfcnpj
WHERE
  a.ROW = 1
);
disconnect from perm;
run;

proc sql;
create table StatusContaFlag as
select
	nucpfcnpj,
	catx(" - ",nucpfcnpj, nmpessoa) as NmCliente,
	put(datepart(dtcontrolestatuscadastroini), date9.) || ' ' || put(timepart(dtcontrolestatuscadastroini), time8.) as dtEntrada,
	datepart(dtcontrolestatuscadastroini) format date9. as dtcontrolestatuscadastroini,
	put(datepart(dtalteracaofim), date9.) || ' ' || put(timepart(dtalteracaofim), time8.) as dtAlteracao,
	dsstatuspreenchimentocadastro
from StatusConta
order by 1
;run;

proc sql;

b.*

inner join StatusContaFlag b
	on a.nucpfcnpj = b.nucpfcnpj

;run;

create table contadigital as select * from connection to perm(
 	select
	  cast(cd_proposta_origem as int64) cd_proposta_origem,
	  ts_proposta_conta,
	  dt_proposta_conta_origem,
	  cd_cpf_cnpj,
	  st_conta,
	  ts_ultima_alteracao_esteira_fraude ts_ultima_alteracao
	from bv-pdbd-prd.dvry_clie_ddv.contadigital_origemabertura_fato
  	where  date_trunc(dt_proposta_conta, month) >= date_trunc(date_sub(current_date(),interval 30 day),month)      and    cd_proposta_origem is not null
);
disconnect from perm;
run;

a.*,
b.*
from TbPropostaDetalhe a

	on a.nuproposta = b.cd_proposta_origem

data tbcontasemproposta; set contadigital2 (where=(cd_proposta_origem eq .)); run;

data tbpropostaconta2; set tbpropostaconta1 tbcontasemproposta; run;

proc sql;
create table TbPropostaStatusConta0 as
select
	COALESCE(t2.nuanomes, t1.nuanomes) as nuanomes,
	COALESCE(t2.cd_proposta_origem, t1.NuProposta) as NuProposta,
	COALESCE(t1.DtEntradaProposta,t2.dt_proposta_conta_origem, datepart(t2.ts_proposta_conta)) format date9. as DtEntradaProposta,
	COALESCE(t2.StProposta, t1.StProposta) as StProposta,
	COALESCE(t2.cd_cpf_cnpj, t1.NuCpfCnpj) as NuCpfCnpj,
	COALESCE(t2.TpPessoa, t1.TpPessoa) as TpPessoa,
	COALESCE(t2.CdParceiroComercial, t1.CdParceiroComercial) as CdParceiroComercial,
	COALESCE(t2.NmGrupoProdutoMeta, t1.NmGrupoProdutoMeta) as NmGrupoProdutoMeta,
	COALESCE(t2.cd_cpf_cnpj, t1.NmCliente ) as NmCliente,
	COALESCE(t2.dtEntrada, t1.dtEntrada) as dtEntrada,
	COALESCE(t2.dtAlteracao, t1.dtAlteracao) as dtAlteracao,
	COALESCE(t2.st_conta, t1.dsstatuspreenchimentocadastro) as stconta,
	t1.dtcontrolestatuscadastroini,
	put(datepart(datetime()), date9.) || ' ' || put(timepart(datetime()), time8.) as dtprocessamento
from TbPropostaDetalheFlag t1
full join tbpropostaconta2 t2 on t1.NuProposta = t2.cd_proposta_origem
;run;
