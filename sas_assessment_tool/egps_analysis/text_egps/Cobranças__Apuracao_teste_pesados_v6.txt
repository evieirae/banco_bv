CREATE TABLE RECEBIMENTOS AS
SELECT DISTINCT
A.*,
A.DtProcessamento AS DT_PROC_PAG

FROM CONNECTION TO perm(

		SELECT
		CAST(A.CD_CONTRATO AS INT64) AS NUCONTRATO,
		A.SQ_CONTRATO_FINANCEIRO AS SQCONTRATOFINANCEIRO,
		A.DT_PROCESSAMENTO AS DtProcessamento,
		'GCON' AS SISTEMA,
		Count(A.Nu_Parcela) AS QtdParcelas,
		SUM(A.Vr_Recebido) AS TotalRecebido,
		SUM(A.vr_principal_devido) AS PRINCIPAL_DEVIDO,
		SUM(A.Vr_Principal_Baixado) AS TotalBaixado,
		SUM(A.Vr_Comissao_Permanencia_Baixada) AS TotalMora,
		SUM(A.vr_comissao_permanencia_devida) AS MoraDevida,
		SUM(A.Vr_Multa_Baixada) AS TotalMulta,
		SUM(A.vr_multa_devida) AS MultaDevida
		FROM bv-pdbd-prd.dvry_core.tbparcelarecebimento AS A
		LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B ON A.CD_CONTRATO = B.CD_CONTRATO
			AND A.SQ_CONTRATO_FINANCEIRO = B.SQ_CONTRATO_FINANCEIRO
			AND A.Nu_Parcela = B.Nu_Parcela
		LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS C
			ON A.CD_CONTRATO = C.CD_CONTRATO
		LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS D
			ON A.CD_CONTRATO = D.NUCONTRATOGESTAO
		WHERE A.DT_RECEBIMENTO >= &dt_inicio. AND  A.DT_RECEBIMENTO <= &dt_fim. AND Fl_Estornado = FALSE AND
		((C.CD_PRODUTO = '12') OR (C.CD_PRODUTO = '14'))
		GROUP BY A.CD_CONTRATO, A.SQ_CONTRATO_FINANCEIRO
		, A.Dt_Processamento

     ) A;
QUIT;

PROC SQL;
CREATE TABLE CONSULTA_PAG_VF5 AS
SELECT
A.*,
B.TPACORDO,
B.TPACORDO2,
B.AGDMDAYSMIN AS QTDIAINICIOFAIXAATRASO,
B.AGDMDAYSMAX AS QTDIAFIMFAIXAATRASO,
B.LETRA_ALCADA
FROM CONSULTA_PAG_VF4 AS A
LEFT JOIN BASE_CYBER_ADM_2 AS B
ON A.LETRA_TBSCOR = B.LETRA_ALCADA

AND A.ATRASO_PAG BETWEEN B.AGDMDAYSMIN AND B.AGDMDAYSMAX
ORDER BY A.NuContratoGestao
;QUIT;


PROC SQL;
CREATE TABLE CONSULTA_PAG_VF6 AS
SELECT DISTINCT
A.*,
ROUND(((A.EncargosDevidos-A.EncargosBaixados)/A.EncargosDevidos)*100,2) AS DESCONTO_ENCARGOS,
B.ACMAXDISC1 AS LIMITADORMULTA3,
B.ACMAXDISC1 AS DESC_MM,
C.ACMAXDISC1 AS LIMITADORMORA3,
D.ACMAXDISC1 AS DESC_FAIXA
FROM CONSULTA_PAG_VF5 AS A
LEFT JOIN MULTA AS B ON B.LETRA_ALCADA = A.LETRA_TBSCOR AND B.TPACORDO = A.TPACORDO AND B.AGDMDAYSMIN = A.QTDIAINICIOFAIXAATRASO AND B.AGDMDAYSMAX = A.QTDIAFIMFAIXAATRASO
LEFT JOIN MORA AS C ON C.LETRA_ALCADA = A.LETRA_TBSCOR AND C.TPACORDO = A.TPACORDO AND C.AGDMDAYSMIN = A.QTDIAINICIOFAIXAATRASO AND C.AGDMDAYSMAX = A.QTDIAFIMFAIXAATRASO
LEFT JOIN PARCELA AS D ON D.LETRA_ALCADA = A.LETRA_TBSCOR AND D.TPACORDO = A.TPACORDO AND D.AGDMDAYSMIN = A.QTDIAINICIOFAIXAATRASO AND D.AGDMDAYSMAX = A.QTDIAFIMFAIXAATRASO

GROUP BY
	NuContratoGestao
HAVING
	TPACORDO IS MISSING OR TPACORDO2 = 1

;
QUIT;


PROC SQL;
CREATE TABLE
	CONSULTA_PAG_VF7 AS

	*

FROM
	CONSULTA_PAG_VF6
;
