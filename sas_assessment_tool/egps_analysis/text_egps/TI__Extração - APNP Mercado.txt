libname convsimu '/infogrc/convsimu';






%macro BASE_CONCORRENTE(DtRef_ini, DtRef_fim);

	data base_concorrente;
		set _null_;
	run;

	%let DtRef = &DtRef_ini.;
	%do %while (&DtRef. <= &DtRef_fim.);

		data base_concorrente_&DtRef._prod;

			set convsimu.tbsimulacoes;

			

			keep NuAnoMes NuProposta DtEntradaProposta NuContrato DtContrato FlPropostaPaga VrProducao;

			where NuAnoMes = &DtRef.;

		run;

		

		proc sort data =  base_concorrente_&DtRef._prod nodupkey;
			by NuProposta;
		run;

		
		proc sql;

		CREATE TABLE WORK.base_concorrente_&DtRef. AS
   		SELECT t1.NuProposta,
          t1.NuContrato,
          t1.NuAnoMes,
          t1.DtEntradaProposta FORMAT=DDMMYY10. AS DtEntradaProposta,
          t1.DtContrato FORMAT=DDMMYY10. AS DtContrato,
          t1.FlPropostaPaga,
          t1.VrProducao
      	FROM WORK.BASE_CONCORRENTE_&DtRef._PROD t1;

		QUIT;

		
		%let DtRef = %eval(&DtRef. + 1);
		%if %sysfunc(mod(&DtRef., 100)) = 13 %then %let DtRef = %eval(&DtRef. + 88);

	%end;

%mend();
%BASE_CONCORRENTE(202510, 202510);
