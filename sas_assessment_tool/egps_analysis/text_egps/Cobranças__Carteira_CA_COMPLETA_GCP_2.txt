PROC SQL;
CREATE TABLE CART_CA AS SELECT
A.NUCONTA,
A.PRODUTO,
A.DT_INICIO_ATRASO,
CASE
	WHEN B.ENVIO_MIN =. THEN C.ENVIO_MAX
	ELSE B.ENVIO_MIN
END AS DT_ENVIO FORMAT DDMMYY10.,
CASE
	WHEN C.SALDO_DEVEDOR_VENC =. THEN B.DT_RETIRADA
	ELSE C.DT_RETIRADA
END AS DT_RETIRADA FORMAT DDMMYY10.,
B.COD_ASSESS,
C.COD_ASSESS AS COD_ASS_FIM,
B.FAIXA_CS_0,

	WHEN B.ATRASO_ATU =. THEN C.ATRASO_ATU
	ELSE B.ATRASO_ATU

CASE
	WHEN B.DT_ENQUAD =. THEN C.DT_ENQUAD
	ELSE B.DT_ENQUAD
END AS DT_ENQUAD FORMAT DDMMYY10.,
CASE
	WHEN B.STATUS_CONTA ='' THEN C.STATUS_CONTA
	ELSE B.STATUS_CONTA
END AS STATUS_CONTA,
B.STATUS_CONTA AS STATUS_CONTA_I,
C.STATUS_CONTA AS STATUS_CONTA_F

ON A.NUCONTA = C.NUCONTA AND A.PRODUTO = C.PRODUTO AND A.DT_INICIO_ATRASO = C.DT_INICIO_ATRASO;
QUIT;
