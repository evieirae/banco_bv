CREATE TABLE RODAS_PAGO_&ref_tab. AS
SELECT DISTINCT
A.*
,CASE WHEN B.NUCONTRATO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_PAGO_DEPOIS
,CASE WHEN C.NUCONTRATOGESTAO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_BNDU
,CASE WHEN D.NUCONTRATOGESTAO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_RENEG

FROM (SELECT * FROM RODAS_ENRIQUECIMENTO WHERE DATA = &ref_tab.) A
LEFT JOIN CARTEIRA_PARCELA_PAGAS_&ref_tab. B ON
A.NUCONTRATO = B.NUCONTRATO

LEFT JOIN CONNECTION TO PERM (
WITH QUERY_1 AS (
	SELECT
		CAST(NUCONTRATOGESTAO AS NUMERIC) AS  NUCONTRATOGESTAO
	FROM bv-pdbd-prd.trusted_dbpatio.tbentrada
)

SELECT * FROM QUERY_1

) C ON
A.NUCONTRATO = C.NUCONTRATOGESTAO

LEFT JOIN BASE_RENEG_&ref_tab. D ON
A.NUCONTRATO = D.NUCONTRATOGESTAO
AND A.SQCONTRATOFINANCEIRO = D.SQCONTRATOFINANCEIRO
;DISCONTECT FROM PERM
;QUIT;

%end;

%MEND CICLO_RODAS_JUNCAO;

PROC SQL;
CREATE TABLE HISTORICO_ANUENCIA_RODAS AS
SELECT
B.DS_NOSSO_NUMERO AS ds_nosso_numero

FROM LDUARTE.BASE_RODAS_TOTAL_ANUENCIA A
LEFT JOIN WORK.BASE_RECUPERI B ON
A.IDTITULO = B.ID_TITULO

LEFT JOIN LDUARTE.HIST_ANUENCIA_RODAS C ON
B.DS_NOSSO_NUMERO = C.DS_NOSSO_NUMERO

WHERE FLAG_ANUENCIA = 1
AND A.DS_STATUS = 'Protestado'
AND C.DS_NOSSO_NUMERO IS MISSING

;QUIT;

PROC SQL;
CREATE TABLE HISTORICO_ANUENCIA_SOLAR AS
SELECT DISTINCT
B.DS_NOSSO_NUMERO

FROM LDUARTE.BASE_SOLAR_TOTAL_ANUENCIA A
LEFT JOIN WORK.BASE_RECUPERI B ON
A.NUMERO_TITULO = input(B.DS_NUMERO_TITULO,best10.)

LEFT JOIN LDUARTE.HIST_ANUENCIA_SOLAR C ON
B.DS_NOSSO_NUMERO = C.DS_NOSSO_NUMERO

WHERE FLAG_ANUENCIA = 1
AND A.DS_STATUS = 'Protestado'
AND C.DS_NOSSO_NUMERO IS MISSING

;QUIT;
