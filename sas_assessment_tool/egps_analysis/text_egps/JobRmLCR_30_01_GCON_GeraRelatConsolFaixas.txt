proc sql;
	CREATE TABLE RLTEMP.tmp_SaldoParcela_01 AS
	SELECT DISTINCT
		datepart(sp.DtProcessamento) as DtProcessamento format DDMMYY10.,
		sp.NuContrato,
		sp.CdCarteira,
		sp.CdVeiculoLegal,
		sp.CdProduto,
		sp.CdModalidadeProduto,
		sp.CdRatingAtual       AS CdRating,
		sp.CdSItuacaoContabil,
		sp.CdPessoaCessionaria AS CdCessionario,
		CASE WHEN sp.CdRamoAtividade eq . THEN 0 ELSE sp.CdRamoAtividade END AS CdRamoAtividade,
		fl.DistribuicaoGeografica,
		fx.Faixas,
		CASE WHEN sp.DtBNDU eq . THEN 'Não' ELSE 'Sim' END AS FlagBNDU,
		CASE WHEN (sp.NuDiasAtraso >= 15 and sp.CdRatingAtual not in ('A','AA' )) then 'Anormal'  else 'Normal' END AS FlNormalAnormal,
		CASE WHEN dj.FlDepositoJudicial ne . AND fx.FlVencidoVincendo = 'Vencido' THEN 14 ELSE fx.FaixasFluxo END AS FaixasFluxo,
		CASE WHEN dj.FlDepositoJudicial eq . THEN fx.FlVencidoVincendo ELSE 'Vincendo' END AS FlVencidoVincendo,
		SUM(sp.VpTaxaContrato + CASE WHEN sp.CdCarteira = 1 THEN sp.VrPermanenciaApropriar ELSE 0 END) as VpTaxaContrato,
		SUM(sp.VpTaxaCessao)   as VpTaxaCessao
	FROM GCON.Vw_SaldoParcela sp
		LEFT JOIN RLTEMP.tmp_regiaofilial fl 	ON (sp.CdFilial = fl.CdFilial)
		LEFT JOIN RLTEMP.tmp_relatoriopdd dj	ON (sp.NuContrato = dj.NuContrato)
		LEFT JOIN RLTEMP.tmp_faixas fx		ON (sp.DtVencimento = fx.DtVencimento)
	WHERE 	sp.CdModalidadeProduto NOT IN (1511,2001) AND
			sp.CdCarteira IN (1,2,3,4,5) AND
			sp.DtProcessamento = &DtProcessamento.
	GROUP BY
		sp.DtProcessamento,
		sp.NuContrato,
		sp.CdCarteira,
		sp.CdVeiculoLegal,
		sp.CdProduto,
		sp.CdModalidadeProduto,
		sp.CdRatingAtual,
		sp.CdSItuacaoContabil,
		sp.CdPessoaCessionaria,
		CALCULATED CdRamoAtividade,
		fl.DistribuicaoGeografica,
		fx.Faixas,
		CALCULATED FlagBNDU,
		CALCULATED FlNormalAnormal,
		CALCULATED FaixasFluxo,
		CALCULATED FlVencidoVincendo;
run;

proc sql;
	CREATE TABLE RLTEMP.tmp_anormal_compl AS
	SELECT distinct
			NuContrato,
			"Anormal" as FlAnormalCompl
	FROM GCON.TbContratoContabil
	WHERE 	CdCarteira = 1 AND
			CdRatingAtual not in ('A','AA') AND
			NuDiasAtraso >= 15
	ORDER BY NuContrato;
run;

data RLTEMP.tmp_SaldoParcela_01 (compress=yes);
	merge RLTEMP.tmp_SaldoParcela_01(in=a) RLTEMP.tmp_anormal_compl(in=b);
	by NuContrato;

	if CdCarteira=5 and FlAnormalCompl ne '' then FlNormalAnormal = FlAnormalCompl;
	if a=1 then output;
run;
