PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.Privado_Log AS
	SELECT *
	FROM CONNECTION TO DBCON
				(
			SELECT
				RIGHT(CONVERT(CHAR(10),stproposta.dtstatusproposta,103),7)  AS mesano_log,
					stproposta.stproposta,
					Proposta.SistemaOrigem,
					Proposta.coban,
					Proposta.convenio,
					count(stproposta.nuproposta) as ttl_proposta,
					count(distinct(Pessoasobanalise.NuCpfCnpj)) as ttl_cpf
				FROM DBA.TbLogStatusProposta stproposta
					INNER JOIN
						(select nuproposta,
							cdsubproduto,
						CASE
							WHEN upper(DsLogin) = 'Autosservico' THEN 'AUTO'
							WHEN upper(DsLogin) NOT LIKE 'P%' THEN 'CCOM'
							ELSE 'PPAR'
						END
					AS SistemaOrigem,
						case
							when Pessoa.nmpessoa = 'PAKETA SERVICOS FINANCEIROS SA' then 'Paketa'
							else 'Outros'
						end
					as coban,
						case
							when Parceirocom.NmFantasia  = 'PAKETA SERVICOS FINANCEIROS SA' then 'Paketa'
							else 'Outros'
						end
					as convenio
						from DBA.TbProposta
LEFT JOIN DBA.TbParceiroComercial Parceiropessoa ON Parceiropessoa.CdParceiroComercial = TbProposta.CdPessoa
LEFT JOIN DBA.TbPessoa Pessoa ON Pessoa.CdPessoa = Parceiropessoa.CdPessoa
LEFT JOIN DBA.TbParceiroComercial Parceirocom ON Parceirocom.CdParceiroComercial = TbProposta.cdloja ) Proposta ON stproposta.nuproposta = Proposta.nuproposta
							INNER JOIN DBA.TbPessoaSobAnalise Pessoasobanalise ON Proposta.NuProposta = Pessoasobanalise.NuProposta
								WHERE year(stproposta.dtstatusproposta) >= year(getdate())-1
									and proposta.cdsubproduto = 12
								group by RIGHT(CONVERT(CHAR(10),stproposta.dtstatusproposta,103),7),
									stproposta.stproposta,
									Proposta.SistemaOrigem,
									Proposta.coban,
									Proposta.convenio);
RUN;
