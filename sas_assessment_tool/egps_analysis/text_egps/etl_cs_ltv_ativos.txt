PROC SQL noprint;
	connect to SYBASEIQ (&conex_syb.);
	CREATE TABLE FM_&mes_proces. AS SELECT * FROM connection to SYBASEIQ
		(
		Declare @DtCalculo Varchar(8)
		Set @DtCalculo=&ult_dia.

		SELECT  DISTINCT
			a.NuContrato,
			a.SqContratoFinanceiro as SqContratoFinanceiroAtivo ,
			a.VrPresenteAtual as Saldo,
			/*a.CdRatingAtual as Rating,*/
			Rating=
				CASE
					WHEN a.QtDiasAtraso = 0 THEN 'AA'
					WHEN a.QtDiasAtraso BETWEEN 1 AND 15 THEN 'A'
					WHEN a.QtDiasAtraso BETWEEN 16 AND 30 THEN 'B'
					WHEN a.QtDiasAtraso BETWEEN 31 AND 60 THEN 'C'
					WHEN a.QtDiasAtraso BETWEEN 61 AND 90 THEN 'D'
					WHEN a.QtDiasAtraso BETWEEN 91 AND 120 THEN 'E'
					WHEN a.QtDiasAtraso BETWEEN 121 AND 150 THEN  'F'
					WHEN a.QtDiasAtraso BETWEEN 151 AND 180 THEN 'G'
					WHEN a.QtDiasAtraso > 180 THEN 'H'
				END,
			a.QtDiasAtraso as Atraso,
			Produto=
				CASE
					WHEN D.CdModalidadeProduto in (1,2,3,7,8,25,26,27,38,41,46,48)       THEN 'Leves'
					WHEN D.CdModalidadeProduto in (5,49)                                 THEN 'Motos'
					WHEN D.CdModalidadeProduto in (9,28,47)                              THEN 'Pesados'
					/*
					when D.CdModalidadeProduto in (17)                                   then 'Privado'*/
					/*
					when D.CdModalidadeProduto in (14,18,32,34,37,39,42)                 then 'Publico'*/
					/*

					WHEN D.CdModalidadeProduto in (6,45,56,78,79) and f.cdtipobem in (9,10)    THEN 'Leves'

					WHEN D.CdModalidadeProduto in (6,45,56,78,79) and f.cdtipobem in (1,2,4,8) THEN 'Pesados'

			LEFT JOIN dba.tbproposta as l

			LEFT JOIN CDCCG.TBOBJGARANTIA as m
				ON m.nucontrato =r.nucontrato and m.SqObjGarantia = r.SqObjGarantia
			LEFT JOIN DBA.TBPROPGARANTIAMAQEQUIP as k
				ON l.Nuproposta=k.Nuproposta
		WHERE a.DtProcessamento= @DtCalculo
			and r.CdModelo not like '%&amp%'

PROC SQL noprint;
	connect to SYBASEIQ (&conex_syb.);

	CREATE TABLE FIDIC_&mes_proces. as
		SELECT * FROM connection to SYBASEIQ
			(
			Declare @DtCalculo Varchar(8)
			Set @DtCalculo=&ult_dia.
			SELECT
				NuContrato,
				SqContratoFinanceiro as SqContratoFinanceiroAtivo,
				CdRatingAtual as Rating,
				QtDiasAtraso as Atraso,
				CdModalidadeProduto,
				dtprocessamento,
				carteira =
					CASE
						WHEN
						DtProcessamento >= '20140531' and
						cdpessoacessionaria in (238000000005,245000000012,19000254012,19000427101,19000562240,19000562241)
						and cdgrupoPDD = 3 THEN 'FIDC OFF Ballance'
						WHEN
						DtProcessamento >= '20140531' and
						cdpessoacessionaria IN (238000000005,245000000012,19000254012,19000427101,19000562240,19000562241)
						and cdgrupoPDD = 1
						and CdCarteira in (1,4) THEN 'FIDC ON Ballance'
						WHEN
						DtProcessamento >= '20140531' and
						cdpessoacessionaria not in (238000000005,245000000012,19000254012,19000427101,19000562240,19000562241)
						and cdgrupoPDD = 1
						and CdCarteira in (1,5) THEN 'CessaoCoobrigacao Pos'
						WHEN
						DtProcessamento >= '20140531' and
						cdpessoacessionaria NOT IN (238000000005,245000000012,19000254012,19000427101,19000562240,19000562241)
						and cdgrupoPDD = 1
						and CdCarteira in (1,3) THEN 'CessaoCoobrigacao Pre'
						ELSE 'NÃO_CONTABIL'
					END,
				vrpresenteatual as Saldo2,
				vrpddatual as vrProvisao,
				vrtaxacessao as VpTaxaCessao,
				vrpddcessao as vrPddTaxaCessao,
				origem = 'FIDC'
			INTO #Final
			FROM GESTAOCONTABIL.Vw_RelatorioPDD_Contingencia
			WHERE DtProcessamento = @DtCalculo
				/*and CdRatingAtual not in ('HH')*/
				and Carteira not in ('NÃO_CONTABIL')
				/*Filtros contingência*/

				and CdSituacaoContrato = 'Carteira'
				and CdModalidadeProduto not in (1511, 1512, 2001)
				and CdVeiculoLegal in (1, 4)
				and cdcarteira in (1, 3, 4, 5, 12)
				and DtProcessamentoLiquidacao is null
				and VrPresenteAtual > 0
			/*GROUP BY nucontrato, SqContratoFinanceiro, dtprocessamento, CdModalidadeProduto, carteira, CdRatingAtual,QtDiasAtraso*/

			/*Faz ajustes para tirar contratos de FIDC OFF Ballance lançados incorretamente*/
			SELECT
				NuContrato as NuContrato_bkp
			INTO #bkp

			/* MANTER POR CONTA DO DTPROCESSAMENTO ANTIGO */
			FROM  GESTAOCONTABIL.TBPDDDIARIA
			WHERE DtProcessamento = '20140531' and CdGrupoPdd = 3

			/*TBPDDDIARIA_BKP não existe*/
			SELECT
				a.*,
				b.NuContrato_bkp
			FROM #Final as a LEFT JOIN #bkp as b ON a.NuContrato=b.NuContrato_bkp

			);
QUIT;

				a.*,

                                                       SqContratoFinanceiro=tmpsqcontratofinanceiro)) as b

	/*Tirar as duplicações*/

		BY NuContrato SqContratoFinanceiro;
	RUN;

	PROC DATASETS LIB=WORK;

			   total_parcelas_&dtbase. tot_parc&dtbase.;

	p1_cobertura_cdetran=sum(p1_molicar_defl,-p1_totalcdetran)/saldo;
