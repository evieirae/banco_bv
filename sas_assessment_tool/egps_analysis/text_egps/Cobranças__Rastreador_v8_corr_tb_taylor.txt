SELECT * FROM

 CONNECTION TO dbcon
(SELECT DISTINCT
	d.NuContrato,

	total_atraso_juros = SUM(CASE WHEN d.STPARCELA <>'Q' AND d.DTVENCIMENTO < GETDATE() THEN (CONVERT(NUMERIC(18,2),((isnull(ROUND((isnull(Round(((e.TXCOMISSAOPERMANENCIAMES / 3000 ) * e.VRPRESTACAO), 2),0))*(DATEDIFF(DAY,d.DTVENCIMENTO,GETDATE())),2),0))+(e.VRPRESTACAO*(txmulta/100))) + e.VRPRESTACAO)) ELSE 0 END)

ON d.nucontrato = e.nucontrato AND	 D.sqcontratofinanceiro = E.sqcontratofinanceiro
	where d.Stparcela = 'A' and d.DtVencimento < GETDATE()
GROUP BY
	d.nucontrato,
	d.sqcontratofinanceiro,

	d.nucontrato,
