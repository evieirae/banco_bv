	Create table GCP_BALANCETE AS
	Select * FROM CONNECTION TO perm (
		Select
			*
		From `bv-pdbd-prd.dvry_corp_financas.sfinance_balancete_fato`
		Where
			aa_anomes  = 202510     
			and nu_instituicao in (51, 175, 183, 213, 640, 6556)
)
;disconnect from perm

;Quit;
proc sql;
  create table BALANCETE_UNIAO as
  select
          nu_agencia as 'Agência'n,
          ds_descricao_conta_contabil as 'Descrição'n,
          strip(cd_conta_contabil_nu) as 'Conta contábil'n,
          st_ativo_passivo as 'Identificador Ativo/passivo'n,
          nu_grau as 'Identificador Grau'n,
          vr_saldo_anterior as 'Saldo anterior'n,
		  st_sinal_saldo_anterior as 'Sinal ant. contas alt.'n,
          vr_debito as 'Débitos'n,
          vr_credito as 'Créditos'n,
          vr_saldo_atual as 'Saldo atual'n,
		  st_saldo_atual as 'Sinal'n,
		  nu_prefixo  as 'Prefixo da conta'n,
		  nu_nint_cotabil as 'Nr. Int. c. contábil'n,
		  strip(put(nu_instituicao, 10.)) as INST
  from gcp_balancete
  where ts_alteracao_dvry =
    (select max(ts_alteracao_dvry) from gcp_balancete);

quit;
