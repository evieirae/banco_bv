CREATE TABLE SALDO.SALDO_REMANESCENTE AS

	A.NUCONTRATO,
	A.SQCONTRATOFINANCEIRO,
	A.DTLIQUIDACAOEFETIVA,
	A.DTCANCELAMENTO,
	A.VALORPARCELA,
	A.QTPARCELAS,
	DATEPART(A.DATA) AS DATA FORMAT DDMMYY10.,
	A.STATUS,
	DATEPART(C.DT_ASSES) AS DT_ASSES FORMAT DDMMYY10.,
	DATEPART(C.DT_RET_ASSES) AS DT_RET_ASSES FORMAT DDMMYY10.,
	C.VL_CAUSA,
	C.NUM_PROCESSO,
	C.COD_ASSES,
	C.ASSESSORIA,
	C.ATRASO
FROM SALDO.SALDO_REMANESCENTE A  JOIN CONNECTION TO DBCON
			(SELECT
				P.NUCONTRATOGESTAO,
				P.SQCONTRATOFINANCEIRO,
				P.DT_ASSES,
				P.DT_RET_ASSES,
				P.VL_CAUSA,
				P.NUM_PROCESSO,
				P.COD_ASSES,
				B.RAZAO_NOME AS ASSESSORIA,
				P.QT_DIAS_ATRASO AS ATRASO

				INTO  #BA_AUX

			FROM DBA.SAJ_PROCESSO P JOIN SCC_FIN.SAJ_AGENTE B
			ON P.COD_ASSES = B.COD_AGENTE
			WHERE DATE(P.DT_ASSES) <= &DT_FINAL.
			AND (DATE(P.DT_RET_ASSES) >= &DT_INICIAL. OR DATE(P.DT_RET_ASSES) IS NULL)
			AND P.COD_ACAO IN (5)

			SELECT
				NUCONTRATOGESTAO,
				

				MIN(DT_ASSES) AS DT_ASSES

			INTO #BA_AUX_1

			FROM #BA_AUX
			GROUP BY
				NUCONTRATOGESTAO
				

			SELECT
				B.*

			FROM #BA_AUX_1 A JOIN #BA_AUX B

			ON A.NUCONTRATOGESTAO=B.NUCONTRATOGESTAO 
			AND A.DT_ASSES=B.DT_ASSES

)C
	ON A.NUCONTRATO = C.NUCONTRATOGESTAO
	 ;
