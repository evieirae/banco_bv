options validvarname=any;
options nofmterr;

/* Marcar horário de início da atualização */
%LET W_ini = %SYSFUNC(datetime());

/* Enviar e-mail de início da atualização */
OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_INI EMAIL
    FROM=("inf-gerenciais@bv.com.br")
    TO=("inf-gerenciais@bv.com.br")
SUBJECT="TbMolicar | Início da atualização";

DATA _NULL_;
    FILE MAIL_INI;
        PUT "Este processo gerará as seguintes bases:";
        PUT " - AUXL.TbMolicar";
RUN;

/* Execução dos processos */
%include '/infogrc/tbmolicar/codigos/tbmolicar.sas';

/* Calcular tempo de processamento total */
%LET W_fim = %SYSFUNC(datetime());
%LET W_dur = %SYSFUNC(intck(SECOND, &W_ini, &W_fim), tod8.);

/* Enviar e-mail de finalização da atualização */
PROC SQL;
CREATE TABLE TbAtualizacao AS
  SELECT
    DsCategoria,
    COUNT(DISTINCT CdModelo) AS QtdModelos,
    MAX(DATEPART(DtAlteracao)) AS UltAtualizacao FORMAT=DATE9.
  FROM
    TbMolicar
  GROUP BY 1;
RUN;

OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_FIM EMAIL
    FROM=("inf-gerenciais@bv.com.br")
    TO=("inf-gerenciais@bv.com.br")
    TYPE="text/html"
    SUBJECT="TbMolicar | Fim da atualização";

ODS html body=MAIL_FIM style=noline rs=none;

PROC REPORT DATA=TbAtualizacao nowd HEADLINE HEADSKIP
style (report) = {background = none font_face = "Calibri" font_size = 9pt just=left OUTLINECOLOR = none}
style (column) = {background = none CELLHEIGHT = 2.5% font_face = "Calibri" font_size = 9pt just=center OUTLINECOLOR = none}
style (header) = {foreground = white font_face="Calibri" font_size = 9pt just=center background = #69c3cf OUTLINECOLOR = none} ;

COLUMNS DsCategoria QtdModelos UltAtualizacao;
RUN;

PROC ODSTEXT;
   p "Prezados,";
   p "";
   p "A base TbMolicar foram atualizadas.";
   p "A geração da base demorou &W_dur. para ser concluída.";
   P "";
   p "Atenciosamente,";
   p "Informações Gerenciais Estratégicas";
RUN;

ods _all_ close;

DATA LOGEXE;
  IdProcesso = 'TMOLICAR';
  LENGTH Processo $40.;
  Processo = 'Importação TbMolicar';
  FORMAT Inicio Final DATETIME18.;
  Inicio = &W_ini.;
  Final = &W_fim.;
  Duracao = "&W_dur.";
RUN;

PROC APPEND DATA=LOGEXE BASE=IGELOG.LogExecucaoProcessos; RUN;
