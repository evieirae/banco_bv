proc sql;

create table work.eventos_wpp_1 as

select *
,case when Descricao_Evento in ('Entrada Instagram','Entrada Facebook') then 'Meta'
      when Descricao_Evento in ('EntradaGoogleSearch') then 'Google' else 'verificar' end as campanha
from CONTATOS.AUX_HIST_CTX

;quit;

proc sql;

create table work.eventos_wpp_2 as

select
*
from CONTATOS.AUX_HIST_ACORDO_WHATS

;quit;

proc sql;
create table work.eventos_wpp_3 as

select distinct
a.*
,1 as flag_ctx
,case
	when a.Numero_Protocolo = b.Numero_Protocolo then 1
	else 0
end as flag_acordo
from work.eventos_wpp_1 as a
left join work.eventos_wpp_2 as b on a.Numero_Protocolo = b.Numero_Protocolo

;quit;

proc sql;
create table work.eventos_wpp_4 as

select
Dt_Evento as dt_evento
,Safra as safra
,campanha
,sum(flag_ctx) as qtd_entradas_ctx
,sum(flag_acordo) as qtd_acordos
from work.eventos_wpp_3
group by
Dt_Evento
,Safra
,campanha

;quit;

From work.eventos_wpp_1;
;quit;
