     CREATE TABLE C2C_a AS SELECT * FROM CONNECTION TO Sybase
(
select
t1. NuProposta,
t1. CdLoja,
t1. DtStatusProposta,
t1. StProposta,
t1. CdSubProduto,
t2. NuCpfCnpj as NuCpfCnpjFavorecido,
