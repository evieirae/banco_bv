PROC SQL;
CREATE TABLE TbPDVInad30 AS
SELECT
	NuAnoMes
	,FxIdade
	,CdParceiroComercial
	,coalesce(sum(Case when TpValor = 'Inad30' then Coalesce(VrMedida,0) else 0 End),0) as MedidaInad30 FORMAT numx18.
	,coalesce(sum(Case when TpValor = 'Inad30' then Coalesce(VrDenominador,0) else 0 End),0) as DenominadorInad30 FORMAT numx18.
From BASES.TbInad30_DashMetaComercial
where
	NmGrupoProdutoMeta = 'Leves'
	and NuAnoMes between &safraM6. and &safraM0.
	and FxIdade <> 'ZeroKM'
	and DsTipoCanalProducao in  ('LOJA CANAL NEGOCIOS')
	and NmCanalEstruturaComercial = 'Canal Negocios'
	and TpValor = 'Inad30'
group by 1,2,3
;RUN;

PROC SQL;
CREATE TABLE TbPDVInad15 AS
SELECT
	 NuAnoMes
	,CdParceiroComercial
	,FxIdade
	,coalesce(sum(Case when TpValor = 'Meta (Inad15)' then Coalesce(VrMedida,0) else 0 End),0) as MedidaInad15 FORMAT numx18.
	,coalesce(sum(Case when TpValor = 'Meta (Inad15)' then Coalesce(VrDenominador,0) else 0 End),0) as DenominadorInad15 FORMAT numx18.
FROM BASES.TbInad_DashMetaComercial
where
	NmGrupoProdutoMeta = 'Leves'
	and NuAnoMes between &safraM6. and &safraM0.
	and FxIdade <> 'ZeroKM'
	and DsTipoCanalProducao in  ('LOJA CANAL NEGOCIOS')
	and NmCanalEstruturaComercial = 'Canal Negocios'
	and TpValor = 'Meta (Inad15)'
group by 1,2,3
;RUN;

PROC SQL;
CREATE TABLE TbPDVinad AS
SELECT
*
FROM TbPDVInad30 as a
	left join TbPDVInad15 as b
		on A.CdParceiroComercial = B.CdParceiroComercial
		and A.NuAnoMes = b.NuAnoMes
		and A.FxIdade = B.FxIdade
/*group by 1,2,3*/
;RUN;
