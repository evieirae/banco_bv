PROC SQL;
CREATE TABLE work.DESPESAS AS
SELECT
 t1.Cd_Causa
,t1.PASTA
,t1.VlLcto
,t1.DtOcorrencia
,t1.CdLcto
,t1.TipoLcto
,t1.NmNatureza
,t1.NmNaturezaUnidadeCustom
,t1.LctoGrupo
,t1.LctoGrupoSub1
,t1.LctoGrupoSub2
,t1.DtOcorrencia   AS DtAprovacao
,YEAR(t1.DtOcorrencia) 	AS DT_DtOcorrencia_ANO
,MONTH(t1.DtOcorrencia) 	AS DT_DtOcorrencia_MES
,YEAR(t1.DtOcorrencia)*100+MONTH(t1.DtOcorrencia) AS DT_DtOcorrencia_YM
FROM DBEP.TBLPAGAMENTOS t1
WHERE 1=1
AND t1.CdSistema = 'p'
AND t1.NmNatureza= 'Cível'
AND t1.NmNaturezaUnidadeCustom in ( 'Massificado','Especial','Estratégico')
AND t1.DtOcorrencia BETWEEN &DT_INICIAL. AND &DT_FINAL.
AND t1.LctoGrupo = 'EMOLUMENTOS'
;RUN;



PROC SQL;
CREATE TABLE WORK.RATEIO_EMOLUMENTOS AS
SELECT
 t1.Cd_Causa
,t1.pasta
,t1.VlLcto
,t1.CdLcto
,t1.TipoLcto
,t2.NmProdutoGrupo	AS RATEIO
,t2.NmEmpresa2		AS EMPRESA_RATEIO
,t2.NuContrato1DeN
,IQ.NmProduto
,IQ.NmModalidadeProduto
,t1.DtAprovacao
,'EMOLUMENTOS'      AS Origem
FROM work.DESPESAS t1
LEFT JOIN DBEP.VWTBLCAUSACONS t2 ON t2.CDPASTA = t1.PASTA AND T2.cdSistema = 'p'
LEFT JOIN yribeiro.TBLCONTRATOSIQ IQ ON  IQ.NuContratoText = t2.NuContrato1DeN
;RUN;

PROC SQL;
CREATE TABLE work.BASE_RATEIO_EMOLUMENTOS AS
SELECT
 t1.EMPRESA_RATEIO
,t1.RATEIO
,(SUM(t1.VlLcto)) FORMAT=COMMAX22.2 		AS VLR_EMOLUMENTO
,(count(t1.VlLcto)) 						AS QTD_LCTO
,(count(DISTINCT t1.CD_CAUSA)) 				AS QTD_CAUSA
,(MIN(t1.CD_CAUSA)) 						AS CD_CAUSA_min
,(MAX(t1.CD_CAUSA)) 						AS CD_CAUSA_max
FROM RATEIO_EMOLUMENTOS t1
GROUP BY
 t1.EMPRESA_RATEIO
,t1.RATEIO
;run;

proc sql;

;run;


proc sql;
create table WORK.PRODUTO_RESUMO as

EMPRESA_RATEIO,
RATEIO,
sum(VLR_EMOLUMENTO) as TOTAL_EMOLUMENTO
from WORK.BASE_RATEIO_EMOLUMENTOS

EMPRESA_RATEIO,
RATEIO
;RUN;


proc sql;
create table WORK.EMPRESA_TOTAL as
select
EMPRESA_RATEIO,
sum(TOTAL_EMOLUMENTO) as TOTAL_GERAL_EMOLUMENTO
from WORK.PRODUTO_RESUMO
group by EMPRESA_RATEIO
;RUN;


proc sql;
create table WORK.PRODUTO_PERCENTUAL as
select
a.EMPRESA_RATEIO,
a.RATEIO,
a.TOTAL_EMOLUMENTO,
a.TOTAL_EMOLUMENTO / b.total_geral_EMOLUMENTO as PERCENTUAL
from WORK.PRODUTO_RESUMO as a
inner join work.empresa_total as b on a.EMPRESA_RATEIO = b.EMPRESA_RATEIO
;RUN;


proc sql;
create table WORK.PROPORCIONAL_TOTAL as
select
EMPRESA_RATEIO,
sum(VLR_EMOLUMENTO) as TOTAL_PROPORCIONAL
from WORK.BASE_RATEIO_EMOLUMENTOS
where RATEIO  = 'PROPORCIONAL'
group by EMPRESA_RATEIO
;RUN;


proc sql;
create table WORK.PRODUTO_FINAL_EMOLUMENTOS as
select
a.EMPRESA_RATEIO,
a.RATEIO,
a.TOTAL_EMOLUMENTO + (coalesce(b.total_proporcional, 0) * a.percentual) as  VLR_EMOLUMENTOS
from WORK.PRODUTO_PERCENTUAL as a left join work.proporcional_total as b on a.EMPRESA_RATEIO = b.EMPRESA_RATEIO;
quit;
