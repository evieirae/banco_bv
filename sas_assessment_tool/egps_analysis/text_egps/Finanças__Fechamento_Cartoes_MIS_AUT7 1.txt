Create table TBL_KPI_CANCELAMENTOS as
select SafraRef,
		   NmFamiliaProdutoCartao,

           NmVariante,
		   TipoCancelamento AS TipoInformacao,
		   COUNT(CdDimContaCartao) AS Valor
from connection to perm
 (
   select
			EXTRACT(YEAR FROM a.DtEvento) * 100 + EXTRACT(MONTH FROM a.DtEvento)  AS SafraRef,
				a.CdDimContaCartao,
			       b.NmFamiliaProdutoCartao,
			       b.NmBandeiraCartao,

			       CASE WHEN b.NmVarianteConta = 'GOLD'     AND b.DtAberturaContaCartao >= '2021-09-01' THEN 'LIVRE'
						WHEN b.NmVarianteConta = 'PLATINUM' AND b.DtAberturaContaCartao >= '2021-09-01' THEN 'MAIS'
					    WHEN b.NmTipoCartao IN ('VISA LIVRE','BV MASTERCARD','BV LIVRE','MASTERCARD BV LIVRE','VISA BV LIVRE')  THEN 'LIVRE'
				        WHEN b.NmTipoCartao IN ('VISA MAIS','MASTERCARD MAIS','MASTERCARD BV MAIS','VISA BV MAIS')              THEN 'MAIS'
				        WHEN b.NmTipoCartao IN ('VISA UNICO','MASTERCARD BV UNICO','VISA BV UNICO','MASTERCARD BLACK MULTIPLO') THEN 'UNICO'
						ELSE b.NmVarianteConta
						END AS NmVariante,

			       CASE WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('CREDITO EM LIQUIDACAO')                                    THEN 'Cancelamento CL'
			            WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('CREDITO EM LIQUIDACAO II')                                 THEN 'Cancelamento CL'
			            WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('PREVENTIVO DE COBRANCA')                                   THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('FALECIDO')                                                 THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo like 'CANCELAD% EMISSOR'                                        THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo like 'CANCELAD% CLIENTE'                                        THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('CANCELAME PROVISORIO ABANDONO')                            THEN 'Cancelamento Abandono'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('CANCELAME PROVISORIO A PEDIDO')                            THEN 'Cancelamento Pedido'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('CANCELAME DEFINITIVO AUTO')                                THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('FRAUD APLICATION')                                         THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('FALSIDADE IDEOLOGICA')                                     THEN 'Cancelamento Outros'
			       		WHEN dsatributoanterior NOT IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') AND dsatributonovo IN ('ATIVA','ACORDO - FASE AMIGAVEL','ATRASO','PRE-ACORDO') THEN 'Contas Reabilitadas'
					    END AS TipoCancelamento

		    FROM bv-pdbd-prd.trusted_dwbi_dmoc.tbfatoevento a

		    INNER JOIN bv-pdbd-prd.trusted_dwbi_dime.tbdimcontacartao b
		    ON a.cddimcontacartaoversao = b.cddimcontacartaoversao

		    WHERE (a.DtEvento BETWEEN &Dt_Ref_Inicial. AND &Dt_Ref_Final.)
			AND cast(a.Cddimobjeto as INT64) = 2 AND cast(a.Cddimevento as INT64) IN (16,17,18,19) AND
			      cast(b.VrLimiteContaCartaoGeral as FLOAT64) <> 0

)	GROUP BY TipoCancelamento,
             SafraRef,
		     NmFamiliaProdutoCartao,
		     NmBandeiraCartao,
             NmVariante
	HAVING TipoCancelamento IS NOT MISSING;

	SELECT a.SafraRef,

	       SUM(a.Valor) AS Valor

            CASE WHEN a.NmLinhaDRE IS MISSING THEN 'Verificar'
                 WHEN a.NmLinhaDRE IN ('Custos Diretos','Custos de Cobrança','Fraudes') AND a.ClasseValor IS NOT MISSING THEN a.SubGrupoNmLinhaDRE
	             WHEN a.NmLinhaDRE IN ('Custos Diretos','Custos de Cobrança','Fraudes') AND a.ClasseValor IS MISSING     THEN 'Verificar'

	             WHEN a.NmLinhaDRE IN ('Custos Diretos','Custos de Cobrança','Fraudes') AND a.ClasseValor IS MISSING     THEN 'Verificar'

                 END AS NmLinhaDRE,

	        a.AMORT_DEPREC_OPEX AS TipoDespesa,
	        a.CLASS_INDIR_DIR_CUSTOS AS Classificacao,
			a.Diretoria,
			a.GrupoAcionista,
			a.GrupoConta,
			a.ClasseValor,
			a.NmClasseValor,
  CASE WHEN a.AreaAlocado in ('Cartões BV') then 'Cartões'

SELECT  YEAR(DATEPART(A.DtProcessamento))*100 + MONTH(DATEPART(A.DtProcessamento)) AS SafraRef,
		CASE WHEN a.CdModalidade = 204 THEN 'Carteira Rotativo'
		     WHEN a.CdModalidade = 210 THEN 'Carteira Parcelado'
		     WHEN a.CdModalidade = 218 THEN 'Carteira Rotativo Atraso'
		     WHEN a.CdModalidade = 1304 THEN 'Carteira A Vencer'
		     WHEN a.CdModalidade IN (1901,1904) THEN 'Limite'
		     ELSE 'Verificar'
		     END AS NmGrupoCarteira,

		CASE WHEN A.CdClassificacaoOperacao IN ('A','B','C')
		     THEN 'Carteira Under60'
		     ELSE 'Carteira Over60'
		     END AS FlagCarteiraRotativo,

		CASE WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '01' THEN '01-Rotativo'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '02' THEN '02-Saque'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '03' THEN '03-Parcelamento Emissor'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '04' THEN '04-Pagamento de Contas'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '05' THEN '05-Acordo em Fatura'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '06' THEN '06-Parcelamento de Fatura'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '07' THEN '07-Compras à Vista'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '08' THEN '08-Limite de Crédito'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '09' THEN '09-Parcelamento Total'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0A' THEN '0A-Saque Parcelado'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0B' THEN '0B-Crédito Pessoal'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0C' THEN '0C-Crediário'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0D' THEN '05-Acordo em Fatura CL'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0E' THEN '0E-PagContas Parcelado'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0F' THEN '0F-Compulsório'
			 ELSE 'Verificar'
		     END AS NmCarteiraSegregada,

		 

	   CASE WHEN CALCULATED NmGrupoCarteira = 'Carteira A Vencer'                                                                     THEN 'Carteira À Vista'
	        WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '03-Parcelamento Emissor'     THEN 'Carteira Parcelamento Emissor'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura')      THEN 'Carteira Parcelamento Acordo CA'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura CL')   THEN 'Carteira Parcelamento Acordo CL'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '09-Parcelamento Total'       THEN 'Carteira Parcelamento Especial'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('06-Parcelamento de Fatura','0C-Crediário','0A-Saque Parcelado') THEN 'Carteira Parcelamento Fatura'
            WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0F-Compulsório') THEN 'Carteira Parcelamento Fatura Comp'

		    WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0B-Crédito Pessoal')     THEN 'Carteira Crédito Pessoal'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0E-PagContas Parcelado') THEN 'Carteira Pgto de Contas Parcelado'

            WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Over60'  THEN 'Carteira Rotativo Over60'
	        WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Under60' THEN 'Carteira Rotativo Under60'
            ELSE 'Verificar'
			END AS NmCarteiraDRE,

        a.NuContrato,
		sum(A.VlSaldoCarteira) as VlSaldoCarteira

FROM TbOrigemClassificacaoRisco_2 AS a
WHERE a.CdClassificacaoOperacao <> 'HH'

 GROUP BY

 		NmGrupoCarteira,
		FlagCarteiraRotativo,
		NmCarteiraSegregada,
		NmCarteiraDRE,
        a.NuContrato
;
QUIT;



PROC SQL;

CREATE TABLE TBL_CARTEIRA_POR_FAIXA AS

	SELECT  a.SafraRef,
			CASE WHEN COALESCE(a.Qt_DiasAtraso,0) <= 0                 THEN 'Carteira Em dia'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 1   AND 14  THEN 'Carteira 1 a 14'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 15  AND 30  THEN 'Carteira 15 a 30'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 31  AND 60  THEN 'Carteira 31 a 60'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 61  AND 90  THEN 'Carteira 61 a 90'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 91  AND 120 THEN 'Carteira 91 a 120'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 121 AND 150 THEN 'Carteira 121 a 150'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 151 AND 180 THEN 'Carteira 151 a 180'
				 WHEN COALESCE(a.Qt_DiasAtraso,0) > 180               THEN 'Carteira Acima 180'
				 END AS FaixaAtraso,
			SUM(a.VlSaldoCarteira) AS VrSaldoCarteira

	FROM TBL_TRATAMENTOS AS a
	GROUP BY 1,2;
QUIT;

		   SUM(a.VlSaldoCarteira) AS Valor

	SELECT a.SafraRef,
		   'ND' AS NmFamiliaProdutoCartao,
		   'ND' AS NmBandeiraCartao,
		   'ND' AS NmVariante,

		   a.FaixaAtraso AS TipoInformacao,

		   SUM(a.VrSaldoCarteira) AS Valor
	FROM TBL_CARTEIRA_POR_FAIXA AS a
	GROUP BY 1,2,3,4,5
;
QUIT;

%MACRO DO_LABEL(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;

				CASE WHEN a.Mod = '0204' THEN 'Carteira Rotativo'
				     WHEN a.Mod = '0210' THEN 'Carteira Parcelado'
				     WHEN a.Mod = '0218' THEN 'Carteira Rotativo Atraso'

				     WHEN a.Mod IN ('1901','1904') THEN 'Limite'

				     END AS NmGrupoCarteira,

				     THEN 'Carteira Under60'
				     ELSE 'Carteira Over60'
				     END AS FlagCarteiraRotativo,
				CASE WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '01' THEN '01-Rotativo'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '02' THEN '02-Saque'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '03' THEN '03-Parcelamento Emissor'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '04' THEN '04-Pagamento de Contas'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '05' THEN '05-Acordo em Fatura'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '06' THEN '06-Parcelamento de Fatura'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '07' THEN '07-Compras à Vista'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '08' THEN '08-Limite de Crédito'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '09' THEN '09-Parcelamento Total'
				     WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0A' THEN '0A-Saque Parcelado'
					 WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0B' THEN '0B-Crédito Pessoal'
					 WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0C' THEN '0C-Crediário'
					 WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0D' THEN '05-Acordo em Fatura CL'
					 WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0E' THEN '0E-PagContas Parcelado'
					 WHEN SUBSTRN(COMPRESS(A.Contrt),3,2) = '0F' THEN '0F-Compulsório'
					 ELSE 'Verificar'
				     END AS NmCarteiraSegregada,
			   CASE WHEN CALCULATED NmGrupoCarteira = 'Carteira A Vencer'                                                                     										  THEN 'Carteira À Vista'
			        WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '03-Parcelamento Emissor'     										  THEN 'Carteira Parcelamento Emissor'
					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura')                                              THEN 'Carteira Parcelamento Acordo CA'
					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura CL')   										  THEN 'Carteira Parcelamento Acordo CL'
					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '09-Parcelamento Total'       										  THEN 'Carteira Parcelamento Especial'

					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('06-Parcelamento de Fatura','0C-Crediário','0A-Saque Parcelado')    THEN 'Carteira Parcelamento Fatura'
					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0F-Compulsório') 												  THEN 'Carteira Parcelamento Fatura Comp'

				    WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0B-Crédito Pessoal')     										  THEN 'Carteira Crédito Pessoal'
					WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0E-PagContas Parcelado')                                           THEN 'Carteira Pgto de Contas Parcelado'

		            WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Over60'  THEN 'Carteira Rotativo Over60'
			        WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Under60' THEN 'Carteira Rotativo Under60'
		            ELSE 'Verificar'
					END AS NmCarteiraDRE,
				b.Qt_DiasAtraso,
				SUM(a.Saldo) AS VlSaldoCarteira

		FROM ESTUDOS2.CARTAO3040_&SafraRef. AS a

		LEFT JOIN
		(
			SELECT t.CdProduto,
				   t.Cd,
			       MAX(t.DiaAtraso) AS Qt_DiasAtraso
			FROM ESTUDOS2.CARTAO3040_&SafraRef. AS t
			WHERE t.ClassOp <> 'HH' AND t.Mod NOT IN ('1901','1904')
			GROUP BY 1,2
		) AS b
		ON a.Cd = b.Cd AND
		   a.CdProduto = b.CdProduto

		WHERE a.ClassOp <> 'HH'
		GROUP BY Data_Base,
				 NmGrupoCarteira,
				 FlagCarteiraRotativo,
				 NmCarteiraSegregada,
			     NmCarteiraDRE,
				 b.Qt_DiasAtraso
		;
		QUIT;

		PROC SQL;

		CREATE TABLE TBL_CARTEIRA_2_&I. AS

			SELECT  a.SafraRef,
					a.NmCarteiraDRE,
					CASE WHEN COALESCE(a.Qt_DiasAtraso,0) = 0                 THEN 'Carteira Em dia'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 1   AND 14  THEN 'Carteira 1 a 14'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 15  AND 30  THEN 'Carteira 15 a 30'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 31  AND 60  THEN 'Carteira 31 a 60'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 61  AND 90  THEN 'Carteira 61 a 90'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 91  AND 120 THEN 'Carteira 91 a 120'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 121 AND 150 THEN 'Carteira 121 a 150'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 151 AND 180 THEN 'Carteira 151 a 180'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) BETWEEN 181 AND 360 THEN 'Carteira Acima 180'
						 WHEN COALESCE(a.Qt_DiasAtraso,0) > 360               THEN 'Carteira Acima 180'
						 END AS FaixaAtraso,
					SUM(a.VlSaldoCarteira) AS VrSaldoCarteira
			FROM TBL_CARTEIRA_1_&I. AS a
			GROUP BY 1,2,3;
		QUIT;

		PROC SQL;

		CREATE TABLE TBL_CARTEIRA_&I. AS

			SELECT a.SafraRef,
				   'Cartões BV' AS NmFamiliaProdutoCartao,
				   'ND' AS NmBandeiraCartao,
				   'ND' AS NmVariante,

				   a.NmCarteiraDRE AS TipoInformacao,

				   SUM(a.VlSaldoCarteira) AS Valor
			FROM TBL_CARTEIRA_1_&I. AS a
			WHERE a.NmCarteiraDRE NE 'Verificar'
			GROUP BY 1,2,3,4,5

			UNION ALL

			SELECT a.SafraRef,
				   'Cartões BV' AS NmFamiliaProdutoCartao,
				   'ND' AS NmBandeiraCartao,
				   'ND' AS NmVariante,

				   a.FaixaAtraso AS TipoInformacao,

				   SUM(a.VrSaldoCarteira) AS Valor
			FROM TBL_CARTEIRA_2_&I. AS a
			WHERE a.NmCarteiraDRE NE 'Verificar'
			GROUP BY 1,2,3,4,5
		;
		QUIT;

		SET TBL_CARTEIRA
		    TBL_CARTEIRA_&I.;

%MACRO DO_LABEL(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;
		CREATE TABLE TBL_CARTEIRA_1_&I. AS

				CASE WHEN a.Mod = '0204' THEN 'Carteira Rotativo'
				     WHEN a.Mod = '0210' THEN 'Carteira Parcelado'
				     WHEN a.Mod = '0218' THEN 'Carteira Rotativo Atraso'
