PROC SQL;
CREATE TABLE PropostasAPNP AS
  SELECT
	year(A.DtRetornoCetip)*100+month(A.DtRetornoCetip)AS NuAnoMes,
    A.CdParceiroComercial,
        CASE
		WHEN A.FxIdade = 'ZeroKM' THEN 'Novos'
		WHEN A.FxIdade is not null THEN 'Usados'
		ELSE A.FxIdade
	END AS StatusVeiculo,
    A.FxIdade,
	SUM(CASE WHEN StatusProposta = 'Negada'  THEN QtdPropostas END) AS Part_Negada,
    SUM(CASE WHEN StatusProposta = 'Cancelada'  THEN QtdPropostas END) AS Part_Cancelada,
    SUM(CASE WHEN StatusProposta = 'APNP'  THEN QtdPropostas END) AS Part_APNP
  FROM
    CONVPROP.BASEAPNP A

  Left join auxl.Grupo_Mensageria as M
		on M.Mensageria = A.Mensageria

 LEFT JOIN tmp.tbparceirocomercial C
		ON A.CDPARCEIROCOMERCIAL = C.CDPARCEIROCOMERCIAL
  WHERE 1=1
   	And calculated StatusVeiculo = 'Usados'
    AND A.ProdutoAjuste IN ('Leves')
    AND A.Banco <> 'NÃO CONVERTIDO'
    AND A.FlPropostaValidaPDV = 1
    AND A.PgtoLoja = 'S'
	AND C.FlAtivo = 'S'
	AND A.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS')
    AND A.StatusProposta IN ('Negada','Cancelada','APNP')
    AND A.DtRetornoCetip >= "&m6."d
    AND A.DtRetornoCetip <= &DtAtCetip.
	AND C.NmSubEstruturaComercialC = 'Canal de Negócios'
  GROUP BY
    1,2,3,4
;RUN;

PROC SQL;
CREATE TABLE Card_Simulacoes AS
  SELECT
    year(DtInclusaoGravame)*100+month(DtInclusaoGravame)  AS  NuAnoMes,
    CdParceiroComercial,
    FxIdade,
    COUNT(DISTINCT CASE WHEN  StatusSimulacao ='Card sem proposta originada'
								AND Banco NOT IN ('N/A','BV')
								AND StPagoLoja = 'S'
								AND DtInclusaoGravame <= &DtAtCetip. THEN NuCpfCnpj END) AS Part_Card,

	COUNT(DISTINCT CASE WHEN  StatusSimulacao ='Simulação sem card'
								AND Banco NOT IN ('N/A','BV')
								AND StPagoLoja = 'S'
								AND DtInclusaoGravame <= &DtAtCetip. THEN NuCpfCnpj END) AS Part_SemCard,

	COUNT(DISTINCT CASE WHEN StatusSimulacao= 'Card sem proposta originada'
							 AND DtSimulacaoInicial <= &DtAtCetip. 			THEN NuCpfCnpj End) AS EstoqueTotal,
    COUNT(DISTINCT CASE WHEN StatusSimulacao= 'Simulação sem card'
							 AND DtSimulacaoInicial <= &DtAtCetip. 			THEN NuCpfCnpj End) AS EstoqueSemCardTotal

   ,COUNT(DISTINCT CASE WHEN  StatusSimulacao ='Card sem proposta originada'
								AND Banco NOT IN ('N/A','BV')
								AND StPagoLoja = 'S'
								AND GrupoMsg= 'R'
								AND DtInclusaoGravame <= &DtAtCetip. THEN NuCpfCnpj END) AS Part_CardRec

FROM SimulacoesRito

	where 1=1
	and StatusSimulacao in ('Simulação sem card', 'Card sem proposta originada')
	AND FxIdade <>'ZeroKM'
	AND year(DtInclusaoGravame)*100+month(DtInclusaoGravame) >= &safraM6.
	AND FlAtivo = 'S'
  GROUP BY
    1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbRitoPDV_C001 AS
  SELECT distinct
	a.NuAnoMes,
	a.CdParceiroComercial,
	a.Fxidade,
	coalesce(Sum(B.Part_APNP,0)) as Part_APNP,
	coalesce(Sum(b.Part_Cancelada,0)) as Part_Cancelada,
	coalesce(Sum(b.Part_Negada,0)) as Part_Negada,
	coalesce(Sum(C.Part_Card,0)) as Part_Card,
	coalesce(Sum(C.Part_SemCard,0)) as Part_SemCard,
	coalesce(Sum(C.EstoqueTotal,0)) as EstoqueTotal,
	coalesce(Sum(C.EstoqueSemCardTotal,0)) as EstoqueSemCardTotal,
	coalesce(Sum(C.Part_CardRec,0)) as Part_CardRec

FROM TbRitoPDV_A001 as A
	left join PropostasAPNP as B
	on  A.NuAnoMes = b.NuAnoMes
    and A.CdParceiroComercial = b.CdParceiroComercial
    and A.FxIdade  = b.FxIdade

	left join Card_Simulacoes as C
	on  A.NuAnoMes = C.NuAnoMes
    and A.CdParceiroComercial = C.CdParceiroComercial
    and A.FxIdade  = C.FxIdade
/*where a.CdParceiroComercial = 9695 and a.nuanomes = 202407*/
  GROUP BY
    1,2,3;
RUN;
