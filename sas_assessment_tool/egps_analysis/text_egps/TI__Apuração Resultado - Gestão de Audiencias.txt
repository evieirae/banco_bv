%macro caixa(anomes_ini, anomes_fim, anomes_financas);



data base_contratos_inicial;
set _null_;
run;

%let anomes = &anomes_ini.;
%do %while (&anomes. <= &anomes_fim.);

	data base_contratos_inicial;
	set base_contratos_inicial work.base_anlt_&anomes.;

	run;

	%let anomes = %eval(&anomes. + 1);
	%if %sysfunc(mod(&anomes., 100)) = 13 %then %let anomes = %eval(&anomes. + 88);

%end;



proc sql;

	create table caixa_&anomes_fim. as
		select NUCONTRATOGESTAO,
		sum(VLR_MARGEM_SERVICOS) as VLR_MARGEM_SERVICOS,
		sum(VLR_PDD_BRUTA) as VLR_PDD_BRUTA,
		sum(VLR_RECUPERACAO) as VLR_RECUPERACAO,
		sum(VLR_DESC_PRINCIPAL) as VLR_DESC_PRINCIPAL,
		sum(RECEITA_SERVICO) as RECEITA_SERVICO

		from FINAL.TABELAO_&anomes_financas._AUTOMACAO
		group by 1;

	quit;



proc sql;
	create table base_contratos as
	select
		a.*,
		b.*,
	    sum(b.VLR_MARGEM_SERVICOS, b.VLR_PDD_BRUTA, b.VLR_RECUPERACAO, b.VLR_DESC_PRINCIPAL) as MARGEM_LIQUIDA
	from
		base_contratos_inicial as a
	left join caixa_&anomes_fim. as b
  		on a.nucontratogestao = b.NUCONTRATOGESTAO;

quit;



proc sql;
	create table AGRUP_SAFRA_VP_&anomes_fim. as
		select
		ds_grupo,
		&anomes_fim. as carteira,
		safra,
		sum(RECEITA_SERVICO) as RECEITA_SERVICO,
		sum(MARGEM_LIQUIDA) as MARGEM_LIQUIDA
		from base_contratos
		group by 1,2,3;

quit;

%mend;
%caixa(202506, 202506, jun25);
