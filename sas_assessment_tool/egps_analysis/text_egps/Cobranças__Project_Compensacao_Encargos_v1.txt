PROC SQL;
CONNECT TO SYBASE
	(SERVER = PTASYBCTR
	 DATABASE = master
	 USER=&User_PTASYBCTR.
	 PASSWORD=&pw_PTASYBCTR.);
CREATE TABLE WORK.A_TBL_BOLETOS_GERADOS_CTR AS

SELECT
	  *
FROM CONNECTION TO SYBASE(

SELECT DISTINCT
	  Year(A.DtGeracao)*100+Month(A.DtGeracao) AS Safra
	, A.DtGeracao
	, A.NuContratoGestao
	, A.SqContratoFinanceiro
	, A.TpEntrada
	, A.DtVencimento AS Dt_Venc_Acordo
	, A.DtPagamento
	, A.StNegociacao
	, A.CdOcoConta
	, A.CdIdEmpBanco
	, A.CdCarteiraBanco
	, A.CdNossoNumero
	, A.TpAcordoNegociacao
	, A.VrTotParcela
	, A.VrTotMora
	, A.VrTotMulta
	, A.VrTotHonorario
	, A.VrTotDespesa
	, A.VrBoleto
	, A.DtCancelamento
	, A.VrTotSemDescParcela
	, A.DtStNegociacao
	, SUBSTRING(CAST(A.NuContratoGestao AS VARCHAR(14)),1,2) AS Produto
	, CASE
		WHEN A.NuContratoGestao IS NOT NULL THEN 1
		ELSE 0
	  END AS Flag_Boleto_Gerado
	, CASE
		WHEN A.StNegociacao IN ('LP','PL','PG') THEN 1
		ELSE 0
	  END AS Flag_Boleto_Pago
	, CASE
		WHEN A.StNegociacao NOT IN ('LP','PL','PG') AND A.DtCancelamento IS NOT NULL THEN 1
		ELSE 0
	  END AS Flag_Boleto_Cancelado
	, CASE
		WHEN A.TpAcordoNegociacao = 'Q' THEN 'Quit'
		WHEN A.TpAcordoNegociacao = 'A' THEN 'Atualiz'
		WHEN A.TpAcordoNegociacao = 'P' THEN 'Parcial'
	  	ELSE TpAcordoNegociacao
	  END AS Tipo_Acordo
INTO #Base_Acordos 
FROM dbcontencioso..TbNegociacaoContrato A
WHERE 1=1
AND   A.DtPagamento IS NOT NULL
AND   A.DtGeracao  >= &DT_INI_CTR.
AND   A.DtGeracao  <= &DT_FIM_CTR.
ORDER BY A.DtGeracao ASC, A.NuContratoGestao




SELECT DISTINCT
	  A.*
	, B.DtVencimento AS Dt_Venc_Parc
	, B.DtVencimento_Max
	, B.Pc_HO_Cobrado
	, DATEDIFF(DAY,B.DtVencimento,A.DtGeracao) AS Atraso
INTO #Base_Acordos_Enr 
FROM #Base_Acordos A
INNER JOIN (	SELECT DISTINCT
				  MIN(B.DtVencimento)	AS DtVencimento
				, MAX(B.PcHonorario)	AS Pc_HO_Cobrado
				, MAX(B.DtVencimento)	AS DtVencimento_Max
				, B.CdOcoConta
				, B.CdIdEmpBanco
				, B.CdCarteiraBanco
				, B.CdNossoNumero
				, B.NuContratoGestao
				, B.SqContratoFinanceiro
				FROM DBCONTENCIOSO..TBNEGOCIACAOPARCELA B
				INNER JOIN #Base_Acordos C
				  ON  B.NuContratoGestao 		= C.NuContratoGestao
			      AND B.SqContratoFinanceiro  	= C.SqContratoFinanceiro
				  AND B.CdOcoConta   			= C.CdOcoConta
				  AND B.CdIdEmpBanco 			= C.CdIdEmpBanco
				  AND B.CdCarteiraBanco 		= C.CdCarteiraBanco
				  AND B.CdNossoNumero			= C.CdNossoNumero
				GROUP BY
				  B.CdOcoConta
				, B.CdIdEmpBanco
				, B.CdCarteiraBanco
				, B.CdNossoNumero
				, B.NuContratoGestao
				, B.SqContratoFinanceiro) B
ON  A.NuContratoGestao 		= B.NuContratoGestao
AND A.SqContratoFinanceiro  = B.SqContratoFinanceiro
AND A.CdOcoConta   			= B.CdOcoConta
AND A.CdIdEmpBanco 			= B.CdIdEmpBanco
AND A.CdCarteiraBanco 		= B.CdCarteiraBanco
AND A.CdNossoNumero			= B.CdNossoNumero








SELECT
	 *
	,CASE
		WHEN datediff(dd,DtGeracao,DtVencimento_Max) > 0 THEN 1
		ELSE 0
	END AS Valid_Tratamento_Vincendas
INTO #valid_quitacao_1 
FROM #Base_Acordos_Enr
WHERE 1=1
AND TpEntrada NOT IN ('A','M') 
AND TpAcordoNegociacao = 'Q'




SELECT DISTINCT
	 A.*
	,B.DtGeracao
INTO #valid_quitacao_2 
FROM DBCONTENCIOSO..TBNEGOCIACAOPARCELA AS A
INNER JOIN #valid_quitacao_1 AS B	ON  A.NuContratoGestao 				= B.NuContratoGestao
									AND A.SqContratoFinanceiro  		= B.SqContratoFinanceiro
									AND A.CdOcoConta   					= B.CdOcoConta
									AND A.CdIdEmpBanco 					= B.CdIdEmpBanco
									AND A.CdCarteiraBanco 				= B.CdCarteiraBanco
									AND A.CdNossoNumero					= B.CdNossoNumero
									AND B.Valid_Tratamento_Vincendas	= 1




SELECT
	 *
	,CASE
		WHEN VrRepasse > 0 THEN 1
		ELSE 0
	END AS Fl_Pmt_Vencida
INTO #valid_quitacao_3 
FROM #valid_quitacao_2




SELECT
	  CdOcoConta
	, CdIdEmpBanco
	, CdCarteiraBanco
	, CdNossoNumero
	, NuContratoGestao
	, SqContratoFinanceiro
	, Fl_Pmt_Vencida
	, sum(VrParcela) AS VrParcela_Quit
	, sum(VrMora) AS VrMora_Quit
	, sum(VrMulta) AS VrMulta_Quit
	, sum(VrHonorario) AS VrHonorario_Quit
INTO #valid_quitacao_4 
FROM #valid_quitacao_3
GROUP BY
	  CdOcoConta
	, CdIdEmpBanco
	, CdCarteiraBanco
	, CdNossoNumero
	, NuContratoGestao
	, SqContratoFinanceiro
	, Fl_Pmt_Vencida










SELECT DISTINCT
	  A.*
	, CASE
		WHEN A.TpEntrada IN ('A','M') THEN 0
		WHEN A.Pc_HO_Cobrado < 6 THEN 0
		WHEN A.Pc_HO_Cobrado >= 10 THEN 0
		ELSE 1
	  END Fl_Compensacao
	, B.VrParcela_Quit
	, B.VrMora_Quit
	, B.VrMulta_Quit
	, B.VrHonorario_Quit
	, B.Fl_Pmt_Vencida
INTO #Base_Acordos_Enr_2 
FROM #Base_Acordos_Enr AS A
LEFT JOIN #valid_quitacao_4 AS B	ON  A.NuContratoGestao 		= B.NuContratoGestao
									AND A.SqContratoFinanceiro  = B.SqContratoFinanceiro
									AND A.CdOcoConta   			= B.CdOcoConta
									AND A.CdIdEmpBanco 			= B.CdIdEmpBanco
									AND A.CdCarteiraBanco 		= B.CdCarteiraBanco
									AND A.CdNossoNumero			= B.CdNossoNumero
									AND B.Fl_Pmt_Vencida		= 1




SELECT
	  *
	, 0.1 AS HO_Politica
	, ((VrTotParcela+VrTotMora)+VrTotMulta) AS VrBase
	, ((VrParcela_Quit+VrMora_Quit)+VrMulta_Quit) AS VrBaseQuit
INTO #Base_Acordos_Enr_3 
FROM #Base_Acordos_Enr_2




SELECT
	  *
	, CASE
		WHEN VrBaseQuit IS NULL THEN VrBase
		ELSE VrBaseQuit
	  END AS VrBaseFinal
INTO #Base_Acordos_Enr_4 
FROM #Base_Acordos_Enr_3




SELECT
	   A.*
	, (A.VrBaseFinal*HO_Politica) AS Vlr_Ho_Total
INTO #Base_Acordos_Enr_5 
FROM #Base_Acordos_Enr_4 AS A




SELECT
	  A.*
	, CASE
		WHEN A.Atraso BETWEEN   0 AND    4 THEN '0001 - 0004'
		WHEN A.Atraso BETWEEN   5 AND   15 THEN '0005 - 0015'
		WHEN A.Atraso BETWEEN  16 AND   30 THEN '0016 - 0030'
		WHEN A.Atraso BETWEEN  31 AND   45 THEN '0031 - 0045'
		WHEN A.Atraso BETWEEN  46 AND   70 THEN '0046 - 0070'
		WHEN A.Atraso BETWEEN  71 AND   90 THEN '0071 - 0090'
		WHEN A.Atraso BETWEEN  91 AND  120 THEN '0091 - 0120'
		WHEN A.Atraso BETWEEN 121 AND  150 THEN '0121 - 0150'
		WHEN A.Atraso BETWEEN 151 AND  180 THEN '0151 - 0180'
		WHEN A.Atraso BETWEEN 181 AND  360 THEN '0181 - 0360'
		WHEN A.Atraso BETWEEN 361 AND  540 THEN '0361 - 0540'
		WHEN A.Atraso BETWEEN 541 AND  720 THEN '0541 - 0720'
		WHEN A.Atraso BETWEEN 721 AND 1080 THEN '0721 - 1080'
		WHEN A.Atraso >=1081  			   THEN '1081 - 9999'
		ELSE '99.Verificar'
      END AS Fx_Atraso
    , CASE
    	WHEN A.Atraso BETWEEN   0 AND    15 THEN 'A_Ate_15_Dias'
    	WHEN A.Atraso BETWEEN  16 AND    30 THEN 'B_16_a_30_Dias'
    	WHEN A.Atraso BETWEEN  31 AND    70 THEN 'C_31_a_70_Dias'
    	WHEN A.Atraso BETWEEN  71 AND   360 THEN 'D_71_a_360_Dias'
    	WHEN A.Atraso 				  >=361 THEN 'E_Acima_360_Dias'
    	ELSE '99.Verificar'
      END AS Fx_Atraso_New
	, CAST((A.Vlr_HO_Total-A.VrTotHonorario) AS DECIMAL(18,2)) AS Vlr_Compensado
INTO #Base_Acordos_Enr_6 
FROM #Base_Acordos_Enr_5 AS A




SELECT
	 *
	,CASE
		WHEN TpEntrada = 'A' THEN 'ASSESSORIA'
		WHEN TpEntrada = 'B' THEN 'CHATBOT'
		WHEN TpEntrada = 'C' THEN 'SAC'
		WHEN TpEntrada = 'D' THEN 'MARKETPLACE'
		WHEN TpEntrada = 'E' THEN 'MARKETPLACE'
		WHEN TpEntrada = 'F' THEN 'MARKETPLACE'
		WHEN TpEntrada = 'G' THEN 'MARKETPLACE'
		WHEN TpEntrada = 'H' THEN 'APP'
		WHEN TpEntrada = 'I' THEN 'WHATSAPP COB'
