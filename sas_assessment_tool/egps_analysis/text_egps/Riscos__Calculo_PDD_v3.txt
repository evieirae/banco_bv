%MACRO REFVECM(START,END);
	%LOCAL i;
	%IF &START = &END %THEN &START %RETURN;
	%LET START = %SYSFUNC (INPUTN(&START,YYMMN6.));
	%LET END = %SYSFUNC(INPUTN(&END,YYMMN6.));

%SYSFUNC(PUTN(%SYSFUNC(INTNX(MONTH,&START,&i)),YYMMN6.))
	%END;
%MEND REFVECM;

	%END;

%MACRO SHIFTREFM(REF,N);
	%LOCAL DT;
	%LET DT = %SYSFUNC(MDY(%SUBSTR(&REF,5,2),01,%SUBSTR(&REF,1,4)));
	%SYSFUNC(INTNX(MONTH,&DT,&N),YYMMN6.)
%MEND SHIFTREFM;

%MACRO SHIFTREFM2(VARIAVEL, N);
	INPUT(PUT(INTNX('MONTH',INPUT(PUT(&VARIAVEL.,6.),YYMMN6.),&N.),YYMMN6.),6.)
%MEND SHIFTREFM2;

%MACRO REFVECM(START,END);
	%LOCAL i;
	%IF &START = &END %THEN &START %RETURN;
	%LET START = %SYSFUNC (INPUTN(&START,YYMMN6.));
	%LET END = %SYSFUNC(INPUTN(&END,YYMMN6.));
	%DO i = 0 %TO %SYSFUNC(INTCK(MONTH,&START,&END)) %BY %SYSFUNC(SIGN(&END-&START));
%SYSFUNC(PUTN(%SYSFUNC(INTNX(MONTH,&START,&i)),YYMMN6.))
	%END;
%MEND REFVECM;

%MACRO FOREACH(SEQ,MNAME);
	%LOCAL _i_;
	%GLOBAL CMACRO_FOREACH_PCT;
	%DO _i_ = 1 %TO %SYSFUNC(COUNTW(&SEQ,%STR( )));
		%LET CMACRO_FOREACH_PCT = %SYSFUNC(PUTN(%SYSEVALF((&_i_-1)/%SYSFUNC(COUNTW(&SEQ,%STR( )))),PERCENT8.1));
%&MNAME(%SCAN(&SEQ,&_i_,%STR( )))
	%END;
%MEND FOREACH;

%MACRO CALCULA_ATRASO(ANOMES);

DATA BASE_TRADE_&ANOMES.;
SET _TRADE.ACORDOS_TRADE_&ANOMES.;

ATRASO = data_referencia - 'data de vencimento'N;

IF ATRASO < 0 THEN ATRASO = 0;
ELSE ATRASO = ATRASO;

SAFRA_BASE = YEAR(data_referencia)*100 + MONTH(data_referencia);
SAFRA_VENC = YEAR('data de vencimento'N)*100 + MONTH('data de vencimento'N);

WHERE UPCASE('Marcação Final'n) NOT IN ('ACORDO','CANCELADO','RECOMPRA');

RUN;

PROC SQL;
CREATE TABLE SALDO_CARTEIRA_&ANOMES. AS
SELECT
	&ANOMES. AS SAFRA
	,SUM('valor presente'N) AS SALDO_CART
	,SUM(CASE WHEN ATRASO > 360 THEN 0 ELSE 'valor presente'N END) AS SALDO_CART_SEM_WO
	,SUM(CASE WHEN ATRASO >   6 THEN 0 ELSE 'valor presente'N END) AS SALDO_CART_EM_DIA

FROM BASE_TRADE_&ANOMES.
GROUP BY 1
; RUN;

%MEND;
%FOREACH(%REFVECM(202207,202303),CALCULA_ATRASO);

%MACRO CALCULA_NPL(SAFRA);

PROC SQL;
CREATE TABLE MARCA_NPL_&SAFRA. AS
SELECT
	A.*
	,B.ATRASO AS ATRASO_%SHIFTREFM(&SAFRA., -1)
	,CASE
		WHEN B.ATRASO <= 90
		 AND A.ATRASO  > 90
		 AND A.ATRASO <= 360 THEN 1 ELSE 0 END AS FLAG_NPL

FROM BASE_TRADE_&SAFRA. AS A
LEFT JOIN BASE_TRADE_%SHIFTREFM(&SAFRA., -1) AS B ON A.CONTRATO = B.CONTRATO
; RUN;

PROC SQL;
CREATE TABLE SALDO_NPL_&SAFRA. AS

	&SAFRA. AS SAFRA
	,SUM('valor presente'N) AS SALDO_NPL

FROM MARCA_NPL_&SAFRA.

GROUP BY 1
; RUN;

%MEND;
%FOREACH(%REFVECM(202208,202303),CALCULA_NPL);
