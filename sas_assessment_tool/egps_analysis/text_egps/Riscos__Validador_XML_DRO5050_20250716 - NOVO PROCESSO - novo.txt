proc sql;
create table work.chavesProb as
select distinct
evento_tipoAvaliacao
,evento_codSistemaOrigem
,evento_codigoEventoOrigem
,evento_naturezaContingencia
,probabilidadePerda_probabilidade
from (
select
evento_tipoAvaliacao
,evento_codSistemaOrigem
,evento_codigoEventoOrigem
,evento_naturezaContingencia
,probabilidadePerda_probabilidade

from work.EventoProbAnt

union all

select
evento_tipoAvaliacao
,evento_codSistemaOrigem
,evento_codigoEventoOrigem
,evento_naturezaContingencia
,probabilidadePerda_probabilidade

from work.EventoProbAtual)

;run;

proc sql;
create table work.verificarProb as
select

a.evento_tipoAvaliacao
,a.evento_codSistemaOrigem
,a.evento_codigoEventoOrigem
,a.evento_naturezaContingencia
,a.probabilidadePerda_probabilidade

,sum(case when b.evento_codigoEventoOrigem is null then 0 else 1 end) as RowsInOld
,sum(b.probabilidadePerda_valorRisco) 							  	  as VlProbPerdaRiscoOld

from work.chavesProb as a

left join work.EventoProbAnt as b

on a.evento_tipoAvaliacao = b.evento_tipoAvaliacao
and a.evento_codSistemaOrigem = b.evento_codSistemaOrigem
and a.evento_codigoEventoOrigem = b.evento_codigoEventoOrigem
and a.evento_naturezaContingencia = b.evento_naturezaContingencia
and a.probabilidadePerda_probabilidade = b.probabilidadePerda_probabilidade

group by
a.evento_tipoAvaliacao
,a.evento_codSistemaOrigem
,a.evento_codigoEventoOrigem
,a.evento_naturezaContingencia
,a.probabilidadePerda_probabilidade
;run;

proc sql;
create table work.EventoProbAvaliacao as
select
a.evento_tipoAvaliacao
,a.evento_codSistemaOrigem
,a.evento_codigoEventoOrigem
,a.evento_naturezaContingencia
,a.probabilidadePerda_probabilidade
,a.RowsInOld
,a.VlProbPerdaRiscoOld

,sum(case when b.evento_codigoEventoOrigem is null then 0 else 1 end) as RowsInNew
,sum(b.probabilidadePerda_valorRisco) 							  	  as VlProbPerdaRiscoNew

from work.verificarProb as a

left join work.EventoProbAtual as b
on a.evento_tipoAvaliacao = b.evento_tipoAvaliacao
and a.evento_codSistemaOrigem = b.evento_codSistemaOrigem
and a.evento_codigoEventoOrigem = b.evento_codigoEventoOrigem
and a.evento_naturezaContingencia = b.evento_naturezaContingencia
and a.probabilidadePerda_probabilidade = b.probabilidadePerda_probabilidade

group by
a.evento_tipoAvaliacao
,a.evento_codSistemaOrigem
,a.evento_codigoEventoOrigem
,a.evento_naturezaContingencia
,a.probabilidadePerda_probabilidade
,a.RowsInOld
,a.VlProbPerdaRiscoOld

;run;
