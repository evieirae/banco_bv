proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
   Create table VinculoParceiroArvore as
   select * from connection to perm
		(
SELECT

a.CdGrupoParceiroPrecificacao,
a.NmGrupoParceiroPrecificacao,
b.CdParceiroComercial,
c.CdArvoreDecisao,
c.flativo

FROM `bv-pdbd-prd.trusted_precportalprecificacao.tbgrupoparceiroprecificacao` a

LEFT JOIN `bv-pdbd-prd.trusted_precportalprecificacao.tbvinculoparceirogrupo` b  on a.CdGrupoParceiroPrecificacao=b.CdGrupoParceiroPrecificacao
LEFT JOIN `bv-pdbd-prd.trusted_precportalprecificacao.tbgrupoparceirogrupocategoria` d on d.CdGrupoParceiroPrecificacao =a.CdGrupoParceiroPrecificacao
LEFT JOIN `bv-pdbd-prd.trusted_precportalprecificacao.tbgrupoparceiroarvoredecisao` c on c.CdGrupoParceiroPrecificacao =a.CdGrupoParceiroPrecificacao
LEFT JOIN `bv-pdbd-prd.trusted_precportalprecificacao.tbarvoredecisao` e on e.CdArvoreDecisao =c.CdArvoreDecisao
where e.flativo = 'S'
);
disconnect from perm;
QUIT;

data SOLITMP.VinculoParceiroArvore;
   set VinculoParceiroArvore;
