        WHEN 2 THEN 'Segunda'

    ATRASO,
    COUNT(*) AS TOTAL_ATRASO

GROUP BY ATRASO;
QUIT;




DATA PARAM_MIX_ATRASO;
INPUT ATRASO PERC_C1 PERC_C2 PERC_C3;
DATALINES;
1 0.61 0.24 0.15
2 0.57 0.26 0.17
3 0.49 0.30 0.21
4 0.45 0.32 0.23
;
RUN;

DATA MIX_POR_ATRASO;
MERGE TOTAL_POR_ATRASO (IN=A)
      PARAM_MIX_ATRASO (IN=B);
BY ATRASO;

IF A AND B;


TOTAL_50 = FLOOR(TOTAL_ATRASO * 0.50);


N_CLUSTER1 = FLOOR(TOTAL_50 * PERC_C1);
N_CLUSTER2 = FLOOR(TOTAL_50 * PERC_C2);
N_CLUSTER3 = FLOOR(TOTAL_50 * PERC_C3);


RESTO = TOTAL_50 - (N_CLUSTER1 + N_CLUSTER2 + N_CLUSTER3);
N_CLUSTER1 = N_CLUSTER1 + RESTO;

KEEP ATRASO TOTAL_ATRASO TOTAL_50
     PERC_C1 PERC_C2 PERC_C3
     N_CLUSTER1 N_CLUSTER2 N_CLUSTER3;
RUN;

	U = RANUNI (12345);

BY ATRASO;
RUN;

DATA AMOSTRA_FINAL;
MERGE BASE_RANDOM (IN=A)
      MIX_POR_ATRASO;
BY ATRASO;

IF NOT A THEN DELETE;


RETAIN CONT1 CONT2 CONT3;


IF FIRST.ATRASO THEN DO;
    CONT1 = 0;
    CONT2 = 0;
    CONT3 = 0;
END;


IF CLUSTER = 1 THEN CONT1 + 1;
ELSE IF CLUSTER = 2 THEN CONT2 + 1;
ELSE IF CLUSTER = 3 THEN CONT3 + 1;


IF CLUSTER = 1 AND CONT1 <= N_CLUSTER1 THEN OUTPUT;
ELSE IF CLUSTER = 2 AND CONT2 <= N_CLUSTER2 THEN OUTPUT;
ELSE IF CLUSTER = 3 AND CONT3 <= N_CLUSTER3 THEN OUTPUT;
RUN;

CREATE TABLE PERC_ATINGIU_QTD_IA AS
SELECT A.*
, CASE WHEN CLUSTER = 1 THEN QT_TOTAL * 0.9
WHEN CLUSTER = 2 THEN QT_TOTAL * 0.4
WHEN CLUSTER = 3 THEN QT_TOTAL * 0.2
ELSE 0
END AS QTD_IA_TOTAL
, CASE WHEN CLUSTER = 1 THEN QT_ATINGIU_ATIVA * 0.9
WHEN CLUSTER = 2 THEN QT_ATINGIU_ATIVA * 0.4
WHEN CLUSTER = 3 THEN QT_ATINGIU_ATIVA * 0.2
ELSE 0
END AS QTD_IA_ATIVA
, (QT_TOTAL - calculated QTD_IA_TOTAL) AS QTD_PA_TOTAL
, (QT_ATINGIU_ATIVA - calculated QTD_IA_ATIVA) AS QTD_PA_ATIVA
FROM PERC_ATINGIU_ATIVA_VALIDA A

%LET NOME_BASE = 20260401;
%LET DATA_FECH = '2026-02-28';
%LET DATA_HOJE = '2026-03-01';

PROC SQL;
    CONNECT TO bigquery AS perm
    ( project='bv-sadi-dtm'
      credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd'
      READ_MODE=storage
      ALLOW_LARGE_RESULTS=must
      MAX_CHAR_LEN=100 );

    CREATE TABLE TB_HISTORICO_&NOME_BASE. AS
    SELECT *
    FROM CONNECTION TO perm (
        SELECT
            K.NUCONTRATO,
            K.SQCONTRATOFINANCEIRO,

            DATE_DIFF(
                DATE(&DATA_HOJE.),
                DATE(
                    MIN(
                        CASE
                            WHEN B.ST_PARCELA <> 'Q'
                              OR (B.ST_PARCELA = 'Q'
                                  AND DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO) > DATE(&DATA_HOJE.))
                            THEN DATE(B.DT_VENCIMENTO)
                        END
                    )
                ),
                DAY
            ) AS ATRASO_REF

        FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil K
        LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela B
          ON K.NUCONTRATO = B.CD_CONTRATO
         AND K.SQCONTRATOFINANCEIRO = B.SQ_CONTRATO_FINANCEIRO

        WHERE DATE(K.DTPROCESSAMENTO) = DATE(&DATA_FECH.)

        GROUP BY
            K.NUCONTRATO,
            K.SQCONTRATOFINANCEIRO

        HAVING
            DATE_DIFF(
                DATE(&DATA_HOJE.),
                DATE(
                    MIN(
                        CASE
                            WHEN B.ST_PARCELA <> 'Q'
                              OR (B.ST_PARCELA = 'Q'
                                  AND DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO) > DATE(&DATA_HOJE.))
                            THEN DATE(B.DT_VENCIMENTO)
                        END
                    )
                ),
                DAY
            ) >= 1

AND
DATE_DIFF(
                DATE(&DATA_HOJE.),
                DATE(
                    MIN(
                        CASE
                            WHEN B.ST_PARCELA <> 'Q'
                              OR (B.ST_PARCELA = 'Q'
                                  AND DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO) > DATE(&DATA_HOJE.))
                            THEN DATE(B.DT_VENCIMENTO)
                        END
                    )
                ),
                DAY
            )
<= 4;
    );

    QUIT;
