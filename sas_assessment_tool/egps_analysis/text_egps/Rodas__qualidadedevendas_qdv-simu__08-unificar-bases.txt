options compress=yes;

/* Gerar base para QDV Simu */
PROC SQL;
  CREATE TABLE QDVCompleto01 AS
    SELECT * FROM Consolidado
    OUTER UNION CORRESPONDING
    SELECT * FROM MarketShare
    OUTER UNION CORRESPONDING
    SELECT * FROM SuperTrunfo
    OUTER UNION CORRESPONDING
    SELECT * FROM Inad
    OUTER UNION CORRESPONDING
    SELECT * FROM Simulacoes;
RUN;

PROC SQL;
CREATE TABLE QDVCompleto AS
  SELECT
    &DtProd. FORMAT=mmddyy10. AS DtAtualizacaoProducao,
    &DtOrig. FORMAT=mmddyy10. AS DtAtualizacaoOriginacao,
    &DtSimu. FORMAT=mmddyy10. AS DtAtualizacaoSimulacao,
    &DtMkt. FORMAT=mmddyy10. AS DtAtualizacaoMktShare,
	*
	FROM QDVCompleto01;
RUN;
