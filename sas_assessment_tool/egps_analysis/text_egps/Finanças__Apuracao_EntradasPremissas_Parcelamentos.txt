CREATE TABLE TBL_CARTOES_ACORDO_CL_3 AS

	SELECT DISTINCT b.SafraCtt,
				    b.Dt_AcordoCL FORMAT DATE9.,
				    a.Cd_Acordo,
				    a.NuCpfCnpj,
				    a.CdDimContaCartao,
		            a.PrazoAcordo,
					a.ST_ACORDO,
				    a.SALDO_NEGOCIACAO AS Vr_EntradasCL,
					a.SALDO_NEGOCIACAO-MOD(a.SALDO_NEGOCIACAO,1) AS Chave_Saldo

	FROM TBL_CARTOES_ACORDO_CL_2 AS a
	LEFT JOIN
	(
    	SELECT DISTINCT Cd_Acordo,
		                NuCpfCnpj,
						MAX(SafraCtt)       AS SafraCtt,
						MAX(Dt_AcordoCL)    AS Dt_AcordoCL
		FROM TBL_CARTOES_ACORDO_CL_2
		GROUP BY 1,2
	) AS b
	ON a.Cd_Acordo = b.Cd_Acordo AND
	   a.NuCpfCnpj = b.NuCpfCnpj
	WHERE a.SafraRef = b.SafraCtt AND b.SafraCtt IS NOT MISSING
;

CREATE TABLE TBL_PARCELADOS_1 AS
SELECT * FROM CONNECTION TO PERM
(
	SELECT
	    CAST((FORMAT_DATE('%Y%m', a.DtAdesao)) AS String) AS SafraRef,

	    CASE
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 1 THEN 'Parcelamento Total'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 2 THEN 'Parcelamento Fatura'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 3 THEN 'Parcelamento Compulsório'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 4 THEN 'Saque Parcelado'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 5 THEN 'Crédito Pessoal'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 6 THEN 'Crediário'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 7 THEN 'Pagamento de Contas Parcelado'
	    END AS NmTipoParcelado,

	    CASE
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 1 THEN 'prod_ParcTotal'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 2 THEN 'prod_Conve'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 3 THEN 'prod_compu'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 4 THEN 'prod_Conve'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 5 THEN 'produca_CP'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 6 THEN 'prod_Conve'
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 7 THEN 'prod_PG_Co'
	    END AS NmInputModelo,

	    CASE
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 2 THEN (
	            CASE
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 0 AND 2 THEN '1. 1 a 2'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 3 AND 4 THEN '2. 3 a 4'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 5 AND 6 THEN '3. 5 a 6'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 7 AND 10 THEN '4. 7 a 10'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) >= 11 THEN '5. > 10'
	            END)
	        WHEN CAST(a.CdTipoParcelado AS INT64) IN (1, 3) THEN '1. 1 a 24'
	        WHEN CAST(a.CdTipoParcelado AS INT64) IN (4, 6) THEN (
	            CASE
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 0 AND 2 THEN '1. 1 a 2'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 3 AND 4 THEN '2. 3 a 4'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 5 AND 6 THEN '3. 5 a 6'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 7 AND 10 THEN '4. 7 a 10'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) >= 11 THEN '5. > 10'
	            END)
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 5 THEN (
	            CASE
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 0 AND 15 THEN '1. 1 a 15'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 16 AND 23 THEN '2. 16 a 23'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) >= 24 THEN '3. > 23'
	            END)
	        WHEN CAST(a.CdTipoParcelado AS INT64) = 7 THEN (
	            CASE
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 0 AND 2 THEN '1. 1 a 2'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) BETWEEN 3 AND 5 THEN '2. 3 a 5'
	                WHEN CAST(a.NuParcelaAdesao AS INT64) >= 6 THEN '3. > 5'
	            END)
	        ELSE 'Verificar'
	    END AS FaixasPrazo,

	    CAST(a.NuParcelaAdesao AS INT64) AS Prazo,

	    SUM((CAST(a.TxRateAdesao AS FLOAT64)/100) * CAST(a.VrContratoAdesao AS FLOAT64)) AS TaxaPond,
	    SUM(CAST(a.NuParcelaAdesao AS INT64) * CAST(a.VrContratoAdesao AS FLOAT64)) AS PrazoPond,
	    SUM(CAST(a.VrContratoAdesao AS FLOAT64)) AS Valor

	FROM `bv-pdbd-prd.trusted_dwbi_dtwh.tbcontratoparcelamentofatura` a
	WHERE a.DtAdesao BETWEEN &Dt_Ref_Inicial. AND &Dt_Ref_Final.
	GROUP BY
	    FORMAT_DATE('%Y%m', a.DtAdesao),
	    NmTipoParcelado,
	    NmInputModelo,
	    FaixasPrazo,
	    Prazo
);
DISCONNECT FROM PERM;
