%macro etls_loader;

   %let etls_tableOptions = ;

   /*---- Map the columns  ----*/
   proc datasets lib = work nolist nowarn memtype = (data view);
      delete WCH9VM9E;
   quit;

   %put %str(NOTE: Mapping columns ...);
   proc sql noprint;
      create view work.WCH9VM9E as
         select
            DtProcessamento,
            QtdRegistrosLidos,
            JOBSAS,
            ASSUNTO,
            QtdRegistrosGravados,
            STATUS,
            DESCRICAO,
            DTREFERENCIA
      from &etls_lastTable
      ;
   quit;

   %let SYSLAST = work.WCH9VM9E;

   %let etls_lastTable = &SYSLAST;
   %let etls_tableOptions = ;

   /* Determine if the target table exists  */
   %let etls_tableExist = %eval(%sysfunc(exist(PEPRDMT.TB_LOG_PEPR, DATA)) or
         %sysfunc(exist(PEPRDMT.TB_LOG_PEPR, VIEW)));

   /*---- Create a new table  ----*/
   %if (&etls_tableExist eq 0) %then
   %do;  /* if table does not exist  */

      %put %str(NOTE: Creating table ...);

      data PEPRDMT.TB_LOG_PEPR
              (COMPRESS = YES);
         attrib DtProcessamento length = 8
            format = ddmmyy.
            informat = DATETIME.;
         attrib QtdRegistrosLidos length = 8;

         attrib ASSUNTO length = $40;
         attrib QtdRegistrosGravados length = 8;
         attrib STATUS length = $8;
         attrib DESCRICAO length = $50;
         attrib DTREFERENCIA length = 8;
         call missing(of _all_);
         stop;
      run;

      %rcSet(&syserr);

   %end;  /* if table does not exist  */

   %put %str(NOTE: Appending data ...);

   proc append base = PEPRDMT.TB_LOG_PEPR(COMPRESS = YES)
      data = &etls_lastTable (&etls_tableOptions)  force ;
    run;

   %rcSet(&syserr);

   proc datasets lib = work nolist nowarn memtype = (data view);
      delete WCH9VM9E;
   quit;
