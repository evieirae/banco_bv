    MBV.Onda AS FlOndaMotorBV,

    TXP.TpOrigemTaxa,
    CASE WHEN INPUT(SUBSTR(PRD.NuCpfCnpj,6,2),2.) BETWEEN 00 AND 19 THEN 'Sem PPR'

    PRD.FxPrestacao,
    PRD.FxFinanciamento,
    CASE WHEN PRD.DiasEfetivacao = . THEN ''
         WHEN PRD.DiasEfetivacao < 0 THEN '< 0'
         WHEN PRD.DiasEfetivacao BETWEEN 0 AND 3 THEN '0 - 3'
         WHEN PRD.DiasEfetivacao BETWEEN 4 AND 5 THEN '4 - 5'
         WHEN PRD.DiasEfetivacao BETWEEN 6 AND 7 THEN '6 - 7'

    Led.CompraLeads,
    CASE WHEN FNI.NuProposta <> . THEN 'S' ELSE 'N' END AS FlFNI,

                     WHEN PRD.NMPARCEIROCOMERCIAL CONTAINS 'BV' AND PRD.NMPARCEIROCOMERCIAL CONTAINS 'COCKPIT' THEN 'GR COCKPIT'
					 WHEN PRD.NMPARCEIROCOMERCIAL CONTAINS 'BV'
							AND (PRD.NMPARCEIROCOMERCIAL CONTAINS 'GARAN'
							OR PRD.NMPARCEIROCOMERCIAL CONTAINS 'PARTICULAR')
							AND PRD.NMPARCEIROCOMERCIAL NOT CONTAINS 'COCKPIT'
							AND PRD.NMPARCEIROCOMERCIAL NOT CONTAINS 'CONSULT'
							OR (PRD.NMPARCEIROCOMERCIAL CONTAINS 'CP'
							AND PRD.NMPARCEIROCOMERCIAL CONTAINS 'BV') THEN 'BV BALCÃO'

  LEFT JOIN

	63,64,65,66,67,68,69,70,71,72,73;
