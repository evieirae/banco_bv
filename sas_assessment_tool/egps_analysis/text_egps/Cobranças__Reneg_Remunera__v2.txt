PROC SQL;
	CONNECT TO SYBASEIQ AS DBCON
	(HOST='PTA-IQPRD01'

	DB=PTAIQCONT

	CREATE TABLE RENEG.RENEG_IMP_&SAFRA. AS
		SELECT DISTINCT
			A.*,
			B.DT_ASSES AS DT_ASSES FORMAT DDMMYY10.,
			B.DT_RET_ASSES AS DT_RET_ASSES FORMAT DDMMYY10.,
			B.COD_AGENTE_PRINCIPAL,
			B.COD_ASSES,
			B.RAZAO_NOME
		FROM
			RENEG.RENEG_IMP_&SAFRA. A LEFT JOIN CONNECTION TO DBCON (
		SELECT
			B.NUCONTRATOGESTAO,
			B.SQCONTRATOFINANCEIRO,
			C.COD_AGENTE_PRINCIPAL,
			A.COD_ASSES,
			C.RAZAO_NOME,
			B.NUM_PROCESSO,
			A.COD_ACAO,
			DATE(A.DT_ASSES) AS DT_ASSES,
			DATE(A.DT_RET_ASSES) AS DT_RET_ASSES
		FROM
			DBA.SAJ_PROCESSO A RIGHT JOIN
		(SELECT
			NUCONTRATOGESTAO,
			SQCONTRATOFINANCEIRO,
			NUM_PROCESSO
		FROM
			DBA.SAJ_PROCESSO
		WHERE
			COD_ACAO IN (1,3,39,59,60,4,5,55,56,58,278,279,280,281)
			AND
			DATE(DT_ASSES) <= DATE(&DATAFIM.) AND (DATE(DT_RET_ASSES) >= DATE(&DATAINI.) OR DATE(DT_RET_ASSES) IS NULL)
		GROUP BY
			NUCONTRATOGESTAO,
			SQCONTRATOFINANCEIRO,
			NUM_PROCESSO) B
			ON
			A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
			AND
			A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
			AND
			A.NUM_PROCESSO = B.NUM_PROCESSO

		LEFT JOIN SCC_FIN.SAJ_AGENTE C
			ON
			A.COD_ASSES = C.COD_AGENTE) B
			ON
			A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
			AND
			A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
			AND
			B.DT_ASSES <= A.DT_RENEG AND (B.DT_RET_ASSES >= A.DT_RENEG OR B.DT_RET_ASSES IS MISSING);
QUIT;
RUN;
