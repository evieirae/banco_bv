PROC SQL;
   CREATE TABLE WORK.PARCEIROS AS
   SELECT t1.NmAreaEstruturaComercialC,
		  t1.NmSubEstruturaComercialC,
          t1.CdParceiroComercial,
		  t1.NmParceiroComercial,
		  t1.DsLoginN1C as GR,
         t1.DsLoginN2C as GF,
         t1.DsLoginN3C as GReg

      FROM PROJETOS.TBPARCEIROCOMERCIAL t1
      WHERE t1.FlAtivo = 'S' AND t1.NmCanalEstruturaComercialC IN
           (
           'Canal Veiculos'
           ) AND t1.NmSubEstruturaComercialC NOT IN
           (
           'Canal Concessionárias',
           'Canal de Negócios',
           'Canal Motos',
           'Canal Pesados'
           ) AND t1.NmTipoParceiroComercial = 'Loja' AND t1.NmAutoShopping = "";
