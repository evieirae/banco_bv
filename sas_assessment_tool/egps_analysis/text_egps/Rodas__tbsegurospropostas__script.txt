PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE TbSegurosPropostas001 AS SELECT * FROM connection to dbcon (
  SELECT
    A.NuProposta,
    B.CdTipoSeguro,
    B.VrSeguro
  FROM
    DBA.TbProposta A
  INNER JOIN
    DBCRED.TbPropostaSeguro B
    ON A.NuProposta = B.NuProposta
  WHERE
    A.DtEntradaProposta >= &iqA1.
);
RUN;

PROC SQL;
CREATE TABLE TbSegurosPropostas AS
  SELECT
    NuProposta,
    MAX(CASE WHEN CdTipoSeguro = 1 THEN 1 ELSE 0 END) AS FlAuto,
    MAX(CASE WHEN CdTipoSeguro = 2 THEN 1 ELSE 0 END) AS FlSPF,
    MAX(CASE WHEN CdTipoSeguro = 3 THEN 1 ELSE 0 END) AS FlCap,
    MAX(CASE WHEN CdTipoSeguro = 4 THEN 1 ELSE 0 END) AS FlImobiliario,
    MAX(CASE WHEN CdTipoSeguro = 5 THEN 1 ELSE 0 END) AS FlSeguroVida,
    SUM(CASE WHEN CdTipoSeguro = 1 THEN VrSeguro END) AS VrAuto,
    SUM(CASE WHEN CdTipoSeguro = 2 THEN VrSeguro END) AS VrSPF,
    SUM(CASE WHEN CdTipoSeguro = 3 THEN VrSeguro END) AS VrCap,
    SUM(CASE WHEN CdTipoSeguro = 4 THEN VrSeguro END) AS VrImobiliario,
    SUM(CASE WHEN CdTipoSeguro = 5 THEN VrSeguro END) AS VrSeguroVida
  FROM
    TbSegurosPropostas001
  GROUP BY 1;
RUN;

PROC SQL;
CREATE TABLE TbOriginacaoSeguros AS
  SELECT
    A.NuAnoMes,
    A.NuProposta,
    A.DtEntradaProposta,
    A.NmGrupoProdutoMeta,
    COALESCE(B.FlAuto,0) AS FlAuto,
    COALESCE(B.FlSPF,0) AS FlSPF,
    COALESCE(B.FlCap,0) AS FlCap,
    COALESCE(B.FlImobiliario,0) AS FlImobiliario,
    COALESCE(B.FlSeguroVida,0) AS FlSeguroVida,
    COALESCE(B.VrAuto,0) AS VrAuto,
    COALESCE(B.VrSPF,0) AS VrSPF,
    COALESCE(B.VrCap,0) AS VrCap,
    COALESCE(B.VrImobiliario,0) AS VrImobiliario,
    COALESCE(B.VrSeguroVida,0) AS VrSeguroVida
  FROM
    Originacao A
  LEFT JOIN
    TbSegurosPropostas B
    ON A.NuProposta = B.NuProposta;
RUN;
