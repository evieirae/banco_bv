PROC SQL; CONNECT TO SYBASE (SERVER=MORSYBCTR USER= &U_CTR. PASSWORD= &S_CTR.);
CREATE TABLE TbCMRCTR	AS
SELECT A.*
FROM CONNECTION TO SYBASE (
	
	SELECT CdPessoaMonitora = MC.CdPessoaMonitoravel
		  ,DtClassificacao  = CONVERT(DATE,CG.DtClassificacaoGeral)
          ,CdMonitoria	    = MAX(MC.CdControleMonitoramentoCliente)

	INTO #TbMxClass
	FROM DBMONITORCLIENTE..TbControleMonitoramentoCliente AS MC
        ,DBMONITORCLIENTE..TbClassificacaoGeral 	  	  AS CG
	WHERE MC.CdControleMonitoramentoCliente = CG.CdControleMonitoramentoCliente
 	  AND CG.FlVigente = 'S'
 	GROUP BY MC.CdPessoaMonitoravel
            ,CONVERT(DATE,CG.DtClassificacaoGeral)

	
	SELECT A.*
		  ,CdClassificacao   = CG.CdTipoClassificacao
		  ,TpClassificacao   = CG.TpMotivoClassificacao
          ,FlParecer		 = 0
		  ,DsResponsavel	 = PC.DsLoginResponsavel

		  ,DsParecer		 = PC.DsParecerCliente

	LEFT  JOIN DBMONITORCLIENTE..TbParecerCliente 	  AS PC ON A.CdMonitoria = PC.CdControleMonitoramentoCliente

	WHERE PC.CdControleMonitoramentoCliente IS NULL

	
 	SELECT CdPessoaMonitora  = MC.CdPessoaMonitoravel
		  ,DtClassificacao   = CONVERT(DATE,CG.DtClassificacaoGeral)
		  ,CdMonitoria	     = MAX(MC.CdControleMonitoramentoCliente)
 		  ,CdClassificacao   = CG.CdTipoClassificacao
		  ,TpClassificacao   = CG.TpMotivoClassificacao
		  ,FlParecer		 = 1

			,PC.FlAplicadoAoGrupo

	UNION ALL
	SELECT * FROM #TbCParecer

	
	SELECT CdPessoaMonitora
		  ,DtClassificacao
		  ,CdClassificacao
		  ,FlParecer = MAX(FlParecer)

	INTO #TbControle
	FROM #TbClass
	GROUP BY  CdPessoaMonitora,DtClassificacao,CdClassificacao

	
	SELECT D.CdPessoa
		  ,A.CdPessoaMonitora
		  ,E.NuCpfCnpj
		  ,E.NmPessoa
		  ,E.TpPessoa
		  ,A.DtClassificacao
		  ,A.CdMonitoria
		  ,A.CdClassificacao
		  ,A.TpClassificacao
		  ,DsClassificacao   = C.DsTipoClassificacao
		  ,PesoClassificacao = C.NuPesoTipoClassificacao
		  ,DsResponsavel	 = LTRIM(RTRIM(A.DsResponsavel))
		  ,A.FlAplicaGrupo
		  ,A.FlParecer
		  ,DsParecer		 = LTRIM(RTRIM(A.DsParecer))

	INTO #TbFinal
	FROM #TbClass 	 AS A
		,#TbControle AS B
		,DBMONITORCLIENTE..TbTipoClassificacao AS C
		,DBMONITORCREDITO..TbPessoaMonitoravel AS D

	WHERE A.CdPessoaMonitora = B.CdPessoaMonitora
  	  AND A.DtClassificacao  = B.DtClassificacao
      AND A.CdClassificacao  = B.CdClassificacao
      AND A.FlParecer		 = B.FlParecer
      AND A.CdClassificacao	 = C.CdTipoClassificacao
      AND A.CdPessoaMonitora = D.CdPessoaMonitoravel
      AND D.CdPessoa		 = E.CdPessoa

    SELECT * FROM #TbFinal ORDER BY CdPessoa,DtClassificacao,CdClassificacao,CdMonitoria
	) AS A
;QUIT;RUN;

PROC SQL; CREATE TABLE TbBrutaRating AS
SELECT DISTINCT
 A.CdGpEconomico
,A.CdPessoaFonte
,B.NuClassRating	AS NuRating
,A.DsRatingP		AS DsRating
FROM TbBruta AS A
LEFT JOIN TbDsRating  AS B	ON A.DsRatingP = B.CD_RATING_FINAL

;QUIT;RUN;

;BY CdGpEconomico CdPessoaFonte;WHERE CdGpEconomico IS NOT NULL;RUN;


PROC SQL; CREATE TABLE TbPFonte 	AS
SELECT
 CdPessoaFonte
,MAX(NuRating)		AS NuRatingP
FROM TbBrutaRating
GROUP BY 1
;QUIT;RUN;


PROC SQL; CREATE TABLE TbGpFonte AS
SELECT
 A.CdGpEconomico
,MAX(B.NuRatingP) AS NuRatingGp
FROM TbFonte  AS A
	,TbPFonte AS B
WHERE A.CdGpEconomico NOT IS NULL
  AND A.CdPessoaFonte = B.CdPessoaFonte
GROUP BY 1
HAVING NuRatingGp IS NOT NULL
;QUIT;RUN;

PROC SQL; CREATE TABLE TbFonteFinal AS

 A.CdPessoaFonte
,A.NuRatingP
,C.NuRatingGp
,MAX(A.NuRatingP,C.NuRatingGp)	AS NuRating
FROM  TbPFonte 		AS A
LEFT JOIN TbFonte 	AS B ON A.CdPessoaFonte = B.CdPessoaFonte
LEFT JOIN TbGpFonte	AS C ON B.CdGpEconomico = C.CdGpEconomico
ORDER BY A.CdPessoaFonte
;QUIT;RUN;

PROC SQL; CREATE TABLE TbArrastoP 	AS
SELECT
 CdPessoaFonte
,MAX(NuRating) as NuRatingArrasto

FROM TbFonteFinal
GROUP BY 1
;QUIT;RUN;

PROC SQL; CREATE TABLE TbArrastoP 	AS

,B.CD_RATING_FINAL AS DsRatingArrasto
FROM TbArrastoP		 AS A
LEFT JOIN TbDsRating AS B ON A.NuRatingArrasto = b.NuClassRating

WHERE NuRatingArrasto IS NOT NULL

ORDER BY 1
;QUIT;RUN;

PROC SQL; CREATE TABLE TbArrastoGP 	AS
SELECT
 A.CdGpEconomico
,MAX(B.NuRatingArrasto) as NuRatingArrasto

FROM TbFonte 	AS A
    ,TbArrastoP	AS b
WHERE A.CdGpEconomico IS NOT NULL
  and A.CdPessoaFonte = B.CdPessoaFonte

GROUP BY 1
;QUIT;RUN;

PROC SQL; CREATE TABLE TbArrastoGP 	AS
SELECT A.*
,B.CD_RATING_FINAL AS DsRatingArrasto
FROM TbArrastoGP	 AS A
LEFT JOIN TbDsRating AS B ON A.NuRatingArrasto = b.NuClassRating

;QUIT;RUN;

PROC DELETE
DATA =
		TbBrutaRating

	    WHEN NmDelimitadorRisco = 'Total Específicas' THEN DtVencimentoDL
		ELSE . END)	AS DtVencimentoEspecificaGL FORMAT DDMMYY8.

PROC SQL; CREATE TABLE TbRGL&Tp. AS
SELECT DISTINCT
	   A.DtProcessamento
	  ,A.&Nm.
	  ,A.TpDelimitador
	  ,A.FlSuspenso
	  ,B.*
	  ,C.*
	  ,D.VrAprovadoDL	 AS VrTotalAprovado
	  ,D.VrTomadoDL		 AS VrTotalRisco
	  ,E.VrAprovadoDL	 AS VrEspecificaAprovado
	  ,E.VrTomadoDL		 AS VrEspecificaRisco

	  ,(D.VrAprovadoDL - E.VrAprovadoDL) AS VrRiscoCreditoGLAprovado
	  ,(D.VrTomadoDL   - E.VrTomadoDL)   AS VrRiscoCreditoGLRisco

LEFT JOIN TbDelimitador AS D ON A.&Nm. = D.&Nm.
							AND A.TpDelimitador 	 = D.TpDelimitador
							AND D.NmDelimitadorRisco = 'Exposição Total'
							AND D.FlAtivoDL = 'S'

LEFT JOIN TbDelimitador AS E ON A.&Nm. = E.&Nm.
							AND A.TpDelimitador 	 = E.TpDelimitador
							AND E.NmDelimitadorRisco = 'Total Específicas'
							AND E.FlAtivoDL = 'S'

WHERE A.TpDelimitador = &DL.
  AND A.&Nm. IS NOT NULL

;QUIT;RUN;
%MEND TbRGL;
