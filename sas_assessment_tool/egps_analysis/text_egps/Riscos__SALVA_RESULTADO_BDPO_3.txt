   SELECT t1.Mês,
          t1.'Conta BRGaap'n,

               t1.'Valor do Lançamento'n DESC;

       t6.Diretoria

from TRAB.BASE_CONTABILIDADE_&DSRefModel. (compress=yes pw="&DS_PASSWORD.")          AS T1

        left join TRAB.HISTORICO_&DSRefModel. (pw=&DS_PASSWORD.)                         AS T3 ON (T1.Chave_Exyon=T3.CAUSA)
        left join TRAB.HISTORICA                                                   AS T4 ON (T1.Chave_Exyon=T4.BVT_PASTA)
        left join TRAB.ANTARES_&DSRefAnter.  (pw=&DS_PASSWORD.)                          AS T5 ON (T1.CPF=T5.CPFNum)
        left join TRAB.CC_TRABALHISTA_&DSRefAnter.                                       AS T6 ON (T1.Chave_Exyon=T6.causa)
;
quit;

PROC SQL;

	create table BASE_CONT_iB_&DSRefModel._EVENTO (compress=yes) as
		select

		        case when T1.DespNovasAcoes > 0 then 'EVENTO        '
					 else 'EVENTO        ' end format $CHAR40. as ENTIDADE,

				case when T1.DespNovasAcoes > 0 then 'CRIAR    '
					                            else 'CRIAR    ' end format $CHAR10. as OPERAÇÃO,
                T1.PASTA         as 'EVENTO-NUMERO DE REGISTRO'n,
                'PJUR'           as 'EVENTO-SISTEMA'n,
                'TRABALHISTA'    as 'EVENTO-GRUPO PERDA'n,
                'ORG_OUT_004160' as 'EVENTO-ÁREA REGISTRADORA (IC)'n,
                'Trabalhista'    as 'EVENTO-CLASSIFICAÇÃO'n,

                CASE WHEN T3.DtRescisao NE . THEN min(T3.DtRescisao, T3.DTCADASTRO)
                     ELSE (coalesce(T3.DTCADASTRO,T3.DTINICAUSA)) END FORMAT DDMMYY10. as  'EVENTO-DATA OCORRÊNCIA'n,

                  T3.DTCADASTRO  FORMAT DDMMYY10.  as 'EVENTO-DATA RECONHECIMENTO'n,

                case   when T1.Assunto_Judicial = 'ACORDO EXTRAJUDICIAL' then 	"Acordo trabalhista extrajudicial"
                    	when T1.Assunto_Judicial ne 'ACORDO EXTRAJUDICIAL'  then  "Ação trabalhista"
				 		else "" end format $CHAR30. as 'EVENTO-NOME'n,
                T4.tipo_acao AS 'EVENTO-DESCRIÇÃO'n,
				3	              as 'EVENTO-CATEGORIA BASILEIA N1'n,
                31                as 'EVENTO-CATEGORIA BASILEIA N2'n,
                'Pessoas'         as 'EVENTO-CAUSA DA PERDA'n,
		    	case when T1.Empresa_Bacen = 'BANCO VOTORANTIM S.A.'                       THEN  	'ORG_CPN_000004'
            	     when T1.Empresa_Bacen = 'VOTORANTIM ASSET MANAGEMENT D.T.V.M. LTDA.'  THEN     'ORG_CPN_000006'
					 when T1.Empresa_Bacen = 'VOTORANTIM CORRETORA DE SEGUROS S.A.'        THEN     'ORG_CPN_000020'
					 when T1.Empresa_Bacen = 'PROMOTIVA S/A'                               THEN     'ORG_CPN_000005'
					 when T1.Empresa_Bacen = 'BANCO BV S.A.'                               THEN     'ORG_CPN_000021'
					 when T1.Empresa_Bacen = 'Bankly'                                      THEN     'ORG_CPN_000027'
					 when T1.Empresa_Bacen = 'BVIA'                                        THEN     'ORG_CPN_000023'
					 when T1.Empresa_Bacen = 'BVEP'                                        THEN     'ORG_CPN_000029'
                	 else "" end format $CHAR15. as 'EVENTO-EMPRESA GERADORA (IC)'n,
				case    when T3.DiretoriaPadrao = 'Auditoria Interna'	                      or t6.diretoria in ('Auditoria Interna', 'Riscos') then 'ORG_OUT_008702'
				        when T3.DiretoriaPadrao = 'Comercial Canal Próprio'                   or T6.Diretoria in ('Distribuição Varejo', 'Crédito e Cobrança', 'Crédito, Prevenção a Fraudes e Cobra', 'Crédito, Prevenção a Fraudes e Cobrança') then 'ORG_OUT_009881'
						when T3.DiretoriaPadrao = 'Controles Internos e Riscos Operacionais' or T6.Diretoria in ('Clientes, Produtos e Inovação', 'Finanças e Relações com Investidores', 'Governança e Jurídico', 'Riscos') then 'ORG_OUT_008745'
	                    when T3.DiretoriaPadrao = 'Corporate & Investment Banking' or T6.Diretoria in ('Corporate & Investment Banking', 'Distribuição Varejo', 'Operações', 'Tesouraria')     then	'ORG_OUT_004691'
                        when T3.DiretoriaPadrao = 'Finanças e Relações com Investidores' or T6.Diretoria in ('Finanças e Relações com Investidores', 'Operações')   then	'ORG_OUT_004045'

						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Clientes, Produtos e Inovação')          then 'ORG_OUT_009871'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Corporate & Investment Banking')         then 'ORG_OUT_004691'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Crédito e Cobrança')                     then 'ORG_OUT_010021'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Crédito, Prevenção a Fraudes e Cobra')   then 'ORG_OUT_009684'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Crédito, Prevenção a Fraudes e Cobrança') then 'ORG_OUT_009684'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Despesa Corporativa')                    then 'ORG_OUT_008128'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Distribuição Varejo')                    then 'ORG_OUT_009881'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Finanças e Relações com Investidores')   then 'ORG_OUT_004045'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Governança e Jurídico')                  then 'ORG_OUT_008206'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Negócios Varejo')                        then 'ORG_OUT_009717'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Operações')                              then 'ORG_OUT_004687'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Presidencia')                            then 'ORG_OUT_008269'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Presidência')                            then 'ORG_OUT_008269'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Tecnologia e Dados')                     then 'ORG_OUT_004022'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Tecnologia, Dados e Plataforma')         then 'ORG_OUT_004022'
						when T3.DiretoriaPadrao = '' and T6.Diretoria in ('Varejo')                                 then 'ORG_OUT_009717'

						when T3.DiretoriaPadrao =	'BVIA'	                                  then 'ORG_OUT_009959' 
						when T3.DiretoriaPadrao =	'Clientes, Produtos e Inovação'           then 'ORG_OUT_009871'
						when T3.DiretoriaPadrao =	'Comercial Promotiva'	                  then 'ORG_OUT_010222'
						when T3.DiretoriaPadrao =	'Conselho e Administração'	              then	'ORG_OUT_008269'
						when T3.DiretoriaPadrao =	'Corporate & Investment Banking'	      then	'ORG_OUT_004691'
						when T3.DiretoriaPadrao =	'Crédito e Cobrança'	                  then	'ORG_OUT_010021'
						when T3.DiretoriaPadrao =	'Crédito, Prevenção a Fraudes e Cobrança' then	'ORG_OUT_009684'
						when T3.DiretoriaPadrao =	'Despesa Corporativa'                     then	'ORG_OUT_008128'
						when T3.DiretoriaPadrao =	'Distribuição Varejo'	                  then	'ORG_OUT_009881'
						when T3.DiretoriaPadrao =	'Finanças e Relações com Investidores'	  then	'ORG_OUT_004045'
						when T3.DiretoriaPadrao =	'Governança e Jurídico'	                  then	'ORG_OUT_008206'
						when T3.DiretoriaPadrao =	'Negócios Relacional'	                  then	'ORG_OUT_009881'
						when T3.DiretoriaPadrao =	'Negócios Varejo'	                      then	'ORG_OUT_009717'
						when T3.DiretoriaPadrao =	'Operações'	                              then	'ORG_OUT_004687'
						when T3.DiretoriaPadrao =	'Pessoas e Cultura'	                      then	'ORG_OUT_004257'
						when T3.DiretoriaPadrao =	'Presidencia'	                          then	'ORG_OUT_008269'
						when T3.DiretoriaPadrao =	'Presidência'	                          then	'ORG_OUT_008269'
						when T3.DiretoriaPadrao =	'Private Bank'	                          then	'ORG_OUT_004691'
						when T3.DiretoriaPadrao =	'Riscos'	                              then	'ORG_OUT_008745'
						when T3.DiretoriaPadrao =	'Tecnologia e Dados'	                  then	'ORG_OUT_004022'
						when T3.DiretoriaPadrao =	'Tecnologia, Dados e Plataforma'	      then	'ORG_OUT_004022'
						when T3.DiretoriaPadrao =	'Tesouraria'	                          then	'ORG_OUT_004033'
						when T3.DiretoriaPadrao =	'Varejo'	                              then	'ORG_OUT_009717'
						when T3.DiretoriaPadrao =	'Wealth Management & Services'	          then	'ORG_OUT_004691'
					
					else 'ORG_OUT_008128' end format $CHAR15. as	'EVENTO-ÁREA GERADORA (IC)'n,
                (CALCULATED	'EVENTO-ÁREA GERADORA (IC)'n) format $CHAR15. as 'EVENTO-ÁREA AFETADA (IC)'n,
              	 'iBPRC_000027'        as 'EVENTO-PROCESSOS (ICs)'n,
				 'PRD_000164'          as 'EVENTO-PRODUTO (IC)'n,

     	 	case when  (CALCULATED	'EVENTO-ÁREA GERADORA (IC)'n) eq 'ORG_OUT_008128' then '999999'
                   when trim(put(t6.CCustoPara, best20.))    ne "." or trim(put(t6.CCustoPara, best20.))    ne '0' or trim(put(t6.CCustoPara, best20.))    ne '' then  put(t6.CCustoPara, best20.)
                   when trim(put(T5.CCustoRateio, best20.))  ne "." or trim(put(T5.CCustoRateio, best20.))  ne '0' or trim(put(T5.CCustoRateio, best20.))  ne '' then  put(T5.CCustoRateio, best20.)
		 	       when trim(put(t6.CCustoPara, best20.))    eq "." or  trim(put(t6.CCustoPara, best20.))   eq '0' or  trim(put(t6.CCustoPara, best20.))   eq '' then  '999999'
                   when trim(put(T5.CCustoRateio, best20.))  eq "." or  trim(put(T5.CCustoRateio, best20.)) eq '0' or  trim(put(T5.CCustoRateio, best20.)) eq '' then  '999999'
				   when trim(put(T5.CCustoRateio, best20.))  eq "." or  trim(put(T5.CCustoRateio, best20.)) eq '0' or  trim(put(T5.CCustoRateio, best20.)) eq '' then  '999999'
                                                     else  "999999"  end as 'EVENTO-CENTRO CUSTO (ID)'n,



			case 	when T3.segmento = '1.Varejo'     	                              then 	1
			    	when T3.DiretoriaPadrao = 'Auditoria Interna'                     then  1
					when T3.DiretoriaPadrao = 'Bankly'                                then  5
					when T3.DiretoriaPadrao = 'BVIA'                                  then  1
					when T3.DiretoriaPadrao = 'Clientes'                              then  1
					when T3.DiretoriaPadrao = 'Clientes, Produtos e Inovação'         then  1
					when T3.DiretoriaPadrao = 'Comercial Promotiva'                   then  1
					when T3.DiretoriaPadrao = 'Corporate & Investment Banking'        then  2
					when T3.DiretoriaPadrao = 'Crédito e Cobrança'                    then  1
					when T3.DiretoriaPadrao = 'Crédito, Prevenção a Fraudes e Cobrança' then 1
					when T3.DiretoriaPadrao = 'Distribuição Varejo'                   then  1
					when T3.DiretoriaPadrao = 'Finanças e Relações com Investidores'  then  1
					when T3.DiretoriaPadrao = 'Finanças, RI & Estrategia'             then  1
					when T3.DiretoriaPadrao = 'Finanças, RI, Estratégia & Tesouraria' then  1
					when T3.DiretoriaPadrao = 'Governança e Jurídico'                 then  1
					when T3.DiretoriaPadrao = 'Negócios Relacional'                   then  1
					when T3.DiretoriaPadrao = 'Negócios Varejo'                       then  1
					when T3.DiretoriaPadrao = 'Novos Negócios'                        then  2
					when T3.DiretoriaPadrao = 'Operações'                             then  1
					when T3.DiretoriaPadrao = 'Pessoas e Cultura'                     then  1
					when T3.DiretoriaPadrao = 'Plataforma'                            then  5
					when T3.DiretoriaPadrao = 'Riscos'                                then  1
					when T3.DiretoriaPadrao = 'Seguros e Cartões'                     then  1
					when T3.DiretoriaPadrao = 'Tecnologia'                            then  1
					when T3.DiretoriaPadrao = 'Tesouraria'                            then  4
					when T3.DiretoriaPadrao = 'Varejo'                                then  1
					when T3.DiretoriaPadrao = 'Wealth Management & Services'          then  7
			     else 1 end                     as 'EVENTO-UNIDADE DE NEGÓCIO'n,
                 'NA'                  as 'EVENTO-RISCO MERCADO OU CRÉDITO?'n,
                 'N'                   AS 'EVENTO-RISCO AMBIENTAL?'n,
                 'N'                   AS 'EVENTO-RISCO CIBERNÉTICO?'n,
                 'TRA'                 AS 'EVENTO-NATUREZA CONTINGÊNCIA'n,
                 T1.STATUS             AS 'EVENTO-STATUS CONTINGÊNCIA'n,
				 max(T3.DTCADASTRO, datepart(t4.data_baixa_provisoria), t1.Data_Encerramento ) format DDMMYY10.        AS 'EVENTO-DATA STATUS CONTINGÊNCIA'n,
                'I'                    AS 'EVENTO-AVALIAÇÃO DO RISCO'n,
                case when T1.Assunto_Judicial = 'ACORDO EXTRAJUDICIAL' then 'N'
                     when T1.Empresa_Bacen in ('BANCO BV S.A.', 'BANCO VOTORANTIM S.A.', 'bankly') then 'S'
                     else 'N' end format $CHAR1. as 'EVENTO-ENVIAR BACEN?'n,
				case when (calculated 'EVENTO-ENVIAR BACEN?'n) = 'N' then (cats('Empresa =', T3.Empresa_Bacen, '/ & /', T3.Flag_Bacen, '/ & Assunto =', T3.assunto, '/ & Despesa =', T1.DespTotal ))
                     else 'NA' end format $CHAR500. as 'EVENTO-MOTIVO NÃO REMESSA BACEN?'n,

     			T1.PASTA,
				datepart(t4.data_reabertura) format DDMMYY10. as reabertura,
                INTNX('MONTH', T1.DBProvisao, 0, 'end')   format DDMMYY10.        AS  'DATA PROVISAO'n

	from TRAB.BASE_CONTABILIDADE_&DSRefModel. (compress=yes pw="&DS_PASSWORD.")          AS T1
		left join TRAB.BASE_PWC_TRABALHISTA_&DSRefModel. (pw=&DS_PASSWORD. COMPRESS=YES) AS T2 ON (T1.Chave_Exyon=T2.Causa_Projuris)
        left join TRAB.HISTORICO_&DSRefModel. (pw=&DS_PASSWORD.)                         AS T3 ON (T1.Chave_Exyon=T3.CAUSA)
        left join TRAB.HISTORICA                                                   AS T4 ON (T1.Chave_Exyon=T4.BVT_PASTA)
        left join TRAB.ANTARES_&DSRefAnter.  (pw=&DS_PASSWORD.)                          AS T5 ON (T1.CPF=T5.CPFNum)
        left join TRAB.CC_TRABALHISTA_&DSRefModel.                                       AS T6 ON (T1.Chave_Exyon=T6.causa)
        left join TRAB.BASE_PWC_TRABALHISTA_&DSRefModel. (pw=&DS_PASSWORD.)              AS T7 ON (T1.pasta=T7.pasta)

		where T1.Chave_Exyon = '2124059' or   INTNX('MONTH', T1.DBProvisao, 0, 'end') >= datepart(t4.data_reabertura) >= T1.DBProvisao
;
QUIT;
