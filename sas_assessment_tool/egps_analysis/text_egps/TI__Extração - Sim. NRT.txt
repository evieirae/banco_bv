%macro base_nrt(DtRef);

PROC SQL;
   CREATE TABLE WORK.nrt_&DtRef. AS
   SELECT DISTINCT
		  t1.NuAnoMes,
          t1.NuCpfCnpj,
          t1.CdSimulacaoFinal,
          t1.NuProposta,
          t1.NuContrato,
          t1.Produto,
          t1.FlPropostaOriginada,
          t1.FlPropostaAprovada,
          t1.FlPropostaEfetivada
      FROM CONVSIMU.TBSIMULACOES t1
      WHERE t1.NuAnoMes = &DtRef. AND t1.Canal = 'MULTIMARCAS' AND t1.SistemaFinal IN
           (
           'APAR',
           'PPAR'
           ) AND t1.Mensageria IN
           (
           '0. Aprovado 100%',
           '1. Muita chance >85%',
           '2. Grande chance >70%',
           '3. Boa chance >50%',
           '4. Tem chance >0%'
           );
QUIT;

%mend();
