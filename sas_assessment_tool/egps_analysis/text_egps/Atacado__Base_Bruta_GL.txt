%MACRO TbOfficer ();
PROC SQL;
CONNECT TO SYBASE (SERVER=MORSYBCTR USER= &U_CTR. PASSWORD= &S_CTR.);
CREATE TABLE TbOfficer	AS
SELECT A.*
FROM CONNECTION TO SYBASE (
SELECT	 CdOfficer 			= FC.CdPessoaFuncionario
		,NmOfficer 			= ISNULL(P.NmApelido,P.NmPessoa)
		,CdDivisaoOfficer	= FC.CdDivisaoComercial
		,NmDivisaoOfficer	= DC.NmDivisaoComercial
		,DC.CdFilial
		,F.NmFilial
		,CdSegmentoMercado	= DC.CdSegmentoMercado
		,NmSegmentoMercado	= SM.NmSegmentoMercado
		,CdSegmentoContabil	= SM.CdSegmentoContabil
		,NmSegmentoContabil = SC.NmSegmentoContabil
		,F.CdPraca
		,PC.NmPraca
		,PC.SgPraca

FROM DBGLOBAL..TbFuncionarioAreaComercial FC
    ,DBGLOBAL..TbPessoa P
    ,DBGLOBAL..TbDivisaoComercial DC
	,DBGLOBAL..TbSegmentoMercado  SM
	,DBGLOBAL..TbSegmentoContabil SC
	,DBGLOBAL..TbFilial F
	,DBGLOBAL..TbPraca	PC

WHERE FC.CdPessoaFuncionario = P.CdPessoa
  AND FC.CdDivisaoComercial	*= DC.CdDivisaoComercial
  AND DC.CdFilial			*= F.CdFilial
  AND DC.CdSegmentoMercado	*= SM.CdSegmentoMercado
  AND SM.CdSegmentoContabil *= SC.CdSegmentoContabil
  AND F.CdPraca				*= PC.CdPraca
ORDER BY FC.CdPessoaFuncionario
) AS A
;QUIT;RUN;
%MEND TbOfficer ;

%MACRO TbPessoa ();
PROC SQL; CONNECT TO SYBASE (SERVER=MORSYBCTR USER= &U_CTR. PASSWORD= &S_CTR.);
CREATE TABLE TbPessoaCTR AS
SELECT A.*
FROM CONNECTION TO SYBASE (


SELECT DISTINCT
       M.CdPessoa
      ,PG.CdGrupoEconomico
INTO #TbPessoaMonitora
FROM DBMONITORCREDITO..TbPessoaMonitoravel AS M
    ,DBGLOBAL..TbPessoaGrupoEconomico AS PG
WHERE M.CdPessoa *= PG.CdPessoa
ORDER BY CdPessoa


SELECT
	   C.CdPessoa
      ,PG.CdGrupoEconomico
INTO #TbPessoaCliente
FROM DBGLOBAL..TbCliente AS C
    ,DBGLOBAL..TbPessoaGrupoEconomico AS PG
WHERE C.CdPessoa *= PG.CdPessoa
ORDER BY CdPessoa


SELECT
 A.CdGrupoEconomico
,FlGPCliente   = MAX(A.FlCliente)
,FlGPMonitora  = MAX(A.FlMonitora)

INTO #TbGrupos

FROM ( SELECT DISTINCT CdGrupoEconomico, FlCliente = 1,FlMonitora = 0 FROM #TbPessoaCliente
       UNION
       SELECT DISTINCT CdGrupoEconomico, FlCliente = 0,FlMonitora = 1 FROM #TbPessoaMonitora
       UNION
       SELECT DISTINCT CdGrupoEconomico, FlCliente = 0,FlMonitora = 1 FROM DBMONITORCREDITO..TbPessoaMonitoravel) AS A

WHERE A.CdGrupoEconomico IS NOT NULL
GROUP BY A.CdGrupoEconomico
ORDER BY CdGrupoEconomico

SELECT P.CdPessoa
	  ,DtInclusao       = CONVERT(DATE,P.DtInclusao)
	  ,DtAlteracao      = CONVERT(DATE,P.DtAlteracao)
      ,P.NuCpfCnpj
      ,P.NmPessoa
      ,TpPessoa         = UPPER(P.TpPessoa)
      ,CdGpEconomico    = PG.CdGrupoEconomico
      ,NmGpEconomico    = GP.NmGrupoEconomico
      ,FlGPCliente      = ISNULL(GPP.FlGPCliente	,0)
      ,FlGPMonitora     = ISNULL(GPP.FlGPMonitora	,0)
      ,FlCliente        = CASE WHEN PC.CdPessoa IS NOT NULL			THEN 1 ELSE 0 END
      ,FlMonitora       = CASE WHEN PM.CdPessoa IS NOT NULL			THEN 1 ELSE 0 END
      ,FlPEP            = CASE WHEN P.FlPoliticamenteExposta = 'S'	THEN 1 ELSE 0 END
      ,FlBacen          = CASE WHEN P.FlAutorizaBacen = 'S'			THEN 1 ELSE 0 END
      ,DtInicioOfficer  = CONVERT(DATE,PC.DtInicioOfficeMaster)
      ,CdOfficer        = PC.CdPessoaFuncionario

INTO #TbPessoa
FROM DBGLOBAL..TbPessoa                 AS P
    ,DBGLOBAL..TbCliente                AS PC
    ,DBGLOBAL..TbPessoaGrupoEconomico	AS PG
    ,DBGLOBAL..TbGrupoEconomico         AS GP
    ,#TbPessoaMonitora                  AS PM
    ,#TbGrupos                          AS GPP

WHERE P.CdPessoa 		  *= PC.CdPessoa
  AND P.CdPessoa 		  *= PG.CdPessoa
  AND PG.CdGrupoEconomico *= GP.CdGrupoEconomico
  AND P.CdPessoa 		  *= PM.CdPessoa
  AND PG.CdGrupoEconomico *= GPP.CdGrupoEconomico
SELECT * FROM #TbPessoa WHERE (FlGPCliente + FlCliente + FlGPMonitora + FlMonitora) > 0 ORDER BY CdPessoa) AS A

;QUIT; RUN;

DATA TbPessoaCTR;
 SET TbPessoaCTR;
FORMAT DtInclusao 		DDMMYY8.
 	   DtAlteracao		DDMMYY8.
	   DtInicioOfficer	DDMMYY8.;RUN;

%MEND TbPessoa ;

PROC SQL; CREATE TABLE TbPessoa AS
SELECT
 DATETIME()			AS DtExecucao	FORMAT DATETIME19.
,A.CdPessoa
,A.DtInclusao
,A.DtAlteracao
,A.NuCpfCnpj
,A.TpPessoa
,CASE
 WHEN INPUT(A.NuCpfCnpj,BEST.) IN (0,.) THEN 'CD'|| STRIP(PUT(A.CdPessoa,BEST.))
 WHEN UPCASE(A.TpPessoa) = 'J'			THEN SUBSTR(A.NuCpfCnpj,1,8)
 ELSE SUBSTR(A.NuCpfCnpj,1,9)
 END 				AS NuCpfCnpjRaiz LENGHT = 50
,A.*
,B.NmOfficer
,B.CdDivisaoOfficer
,B.NmDivisaoOfficer
,B.CdFilial
,B.NmFilial
,B.CdSegmentoMercado
,B.NmSegmentoMercado
,B.CdSegmentoContabil
,B.NmSegmentoContabil
,B.CdPraca
,B.NmPraca
,B.SgPraca

,CASE WHEN UPCASE(B.NmSegmentoMercado) = 'PRIVATE' 		THEN 1 ELSE 0 END AS FlPrivateP
,CASE WHEN INPUT(A.NuCpfCnpj,14.) > 0					THEN 1 ELSE 0 END AS R_CpfCnpj
,CASE WHEN C.CdPessoa IS NOT NULL 						THEN 1 ELSE 0 END AS FlComClassPessoa

,CASE WHEN UPCASE(TpPessoa) = 'F'		 				THEN 1
 	  WHEN SUBSTR(NuCpfCnpj,9,4)	= '0001'			THEN 1
 	  WHEN CALCULATED R_CpfCnpj = 0						THEN 1 ELSE 0 END AS FlMatriz

,CASE WHEN FIND (A.NmGpEconomico,"SACADO SME" ,"I") > 0 THEN 1
 	  WHEN FIND (A.NmGpEconomico,"SACADOS SME","I") > 0 THEN 1
 	  WHEN FIND (A.NmGpEconomico,"PORTFOLIO"  ,"I") > 0 THEN 1
	  WHEN FIND (A.NmGpEconomico,"PME"        ,"I") > 0 THEN 1 ELSE 0 END AS FlNMGPSME

,CASE
 WHEN FIND (A.NmGpEconomico,"PEAC"		  ,"I") > 0		THEN 1
 WHEN FIND (A.NmGpEconomico,"WEEL"		  ,"I") > 0
 AND INPUT(COMPRESS(A.NmGpEconomico ,,'KD'),BEST.) > 0  THEN 1 ELSE 0 END AS FlNMGPPEAC

,CASE WHEN FIND (B.NmDivisaoOfficer,"SME"	  ,"I") > 0	THEN 1 ELSE 0 END AS DivSME


,CASE WHEN FIND(B.NmDivisaoOfficer,"PME"	  ,"I") > 0
				OR UPCASE(B.NmDivisaoOfficer) IN (
												  'CORPORATE LESTE / UPPER'
												  'PLATAFORMA BH UPPER'
												  'PLATAFORMA UPPER SUDESTE'
												  'PLATAFORMA UPPER SUL'
												  'UPPER MIDDLE') THEN 1 ELSE 0 END AS DivPME

,CASE WHEN CdGpEconomico IN (3,23444)					THEN 1 ELSE 0 END AS FlGpVotorantim
,CASE WHEN CdGpEconomico = 1 							THEN 1 ELSE 0 END AS FlGPPrivate



FROM TbPessoaCTR 		AS A
LEFT JOIN TbOfficer		AS B ON A.CdOfficer = B.CdOfficer
LEFT JOIN TbPessoaClass AS C ON A.CdPessoa  = C.CdPessoa

;QUIT;RUN;
