%include '/projetos/MISV/sas-misv-base-lib/scripts/ConectaDB.sas';
%include '/projetos/MISV/sas-misv-base-lib/scripts/libname.sas';
LIBNAME SNS '/infogrc/Base_Sinergia_Solar/Bases';

%CRIAVARDB;
/*base com os parceiros autorizados do Solar*/
PROC SQL;
%CONECTADB (server=CTR);
Create Table TBPARCEIROAUTORIZADO_SOLAR1 as
SELECT Distinct NuCpfCnpj, NmTitular, FlAtivo, TpPessoa, datepart(DtCadastro) as DtCadastro format Date9.
from connection to dbcon
(
Select AUTO.NmTitular,
	AUTO.FlAtivo,
	AUTO.TpPessoa,
	AUTO.DtCadastro,
	AUTO.NuCpfCnpj,
	ULTIMO_PGTO.ULTIMO_PAGAMENTO,
	ULTIMO_PGTO.QTD,
	ULTIMO_PGTO.TOTAL_VALOR
	FROM DBCOR..TbFavorecidoAutorizado AUTO
	LEFT JOIN (
		SELECT PEPGTO.NuCpfCnpj,
				MAX(PGTO.DtPagamento) AS ULTIMO_PAGAMENTO,
				count(pgto.CdPagamento) AS QTD,
				sum(pgto.VrLiberado) AS TOTAL_VALOR

		FROM DBPAGAMENTO..TbPagamento PGTO INNER JOIN DBCRED..TbProposta PT
		ON PGTO.NuContratoGestao = PT.NuContratoGestao
			  INNER JOIN DBPAGAMENTO..TbInformacaoBancariaPagamento INFOPGTO ON INFOPGTO.CdInformacaoBancariaPagto = pgto.CdInforBancariaPagtoFavorecido
			  INNER JOIN DBPAGAMENTO..TbPessoaPagamento PEPGTO ON PEPGTO.CdPessoaPagamento = infopgto.CdPessoaPagamento WHERE PT.CdSubProduto = 70
			  AND PT.CdLoja = 378215
		GROUP BY PEPGTO.NuCpfCnpj) ULTIMO_PGTO ON ULTIMO_PGTO.NuCpfCnpj = auto.nuCpfCnpj WHERE AUTO.cdParceiroComercial = 378215
)
;
Quit;

/*CÓDIGO PARA GERAR FAVORECIDOS MEU PARCEIRO SOLAR*/
Proc Sql;
%CONECTADB (server=CTR);
Create Table TBPARCEIROAUTORIZADO as
SELECT NuProposta
		,NuContratoGestao
		,DatePart(DtPagamento) as DtPagamento Format DDMMYY10.
		,CdPagamento
		,VrLiberado
    	,NuCpfCnpj
from connection to dbcon
(
SELECT  P.NuProposta
		, P.NuContratoGestao
		, PGTO.DtPagamento
		, PGTO.CdPagamento
		, PGTO.VrLiberado
    	, PEPGTO.NuCpfCnpj
FROM DBCRED..TbProposta P
	INNER JOIN DBPAGAMENTO..TbPagamento PGTO
		ON PGTO.NuContratoGestao = P.NuContratoGestao
    INNER JOIN DBPAGAMENTO..TbInformacaoBancariaPagamento INFOPGTO
    	ON INFOPGTO.CdInformacaoBancariaPagto = pgto.CdInforBancariaPagtoFavorecido
    INNER JOIN DBPAGAMENTO..TbPessoaPagamento PEPGTO
    	ON PEPGTO.CdPessoaPagamento = infopgto.CdPessoaPagamento
WHERE P.CdSubProduto = 70
  AND P.CdLoja = 418254
  AND P.DtStatusProposta > '09oct2017'
  AND P.NuContratoGestao IS NOT NULL)
  ;
  QUIT;

Proc Sql;
%CONECTADB (server=CTR);
Create Table TbFavorecidoAutorizado as
SELECT
		NuCpfCnpj,
		NmTitular,
		FlAtivo,
		TpPessoa,
		DatePart(DtCadastro) as DtCadastro Format DDMMYY10.
from connection to dbcon
(
SELECT DISTINCT NuCpfCnpj, NmTitular, FlAtivo, TpPessoa, DtCadastro

FROM DBCOR..TbFavorecidoAutorizado
WHERE cdParceiroComercial = 418254)
;
QUIT;

Proc Sql;
Create Table Work.TbParc_Favorecido_Saida as
Select
	  Favorecido.NmTitular
	, Favorecido.FlAtivo
	, Favorecido.TpPessoa
	, Favorecido.DtCadastro
	, Favorecido.NuCpfCnpj
	, Max(ParceriroAutorizado.DtPagamento) as ULTIMO_PAGAMENTO Format DDMMYY10.
	, Count(ParceriroAutorizado.CdPagamento) as QTD
	, Sum(ParceriroAutorizado.VrLiberado) as TOTAL_VALOR

From TbFavorecidoAutorizado as Favorecido
Left Join TBPARCEIROAUTORIZADO as ParceriroAutorizado ON Favorecido.NuCpfCnpj = ParceriroAutorizado.NuCpfCnpj
Group by 1,2,3,4,5;
Quit;

Data Work.Base_Pesquisa_Solar;
Set TBPARCEIROAUTORIZADO_SOLAR1
	TbParc_Favorecido_Saida;
Run;

/*Gerando base saida com parceiro Solar e FDS*/
Proc Sql;
Create Table TBPARCEIROAUTORIZADO_SOLAR as
Select Distinct
	  NmTitular
	, FlAtivo
	, TpPessoa
	, DtCadastro
	, NuCpfCnpj
	, ULTIMO_PAGAMENTO Format DDMMYY10.
	, QTD
	, TOTAL_VALOR

From Base_Pesquisa_Solar;
Quit;

Proc Sort Nodupkey Data = TBPARCEIROAUTORIZADO_SOLAR; by NuCpfCnpj; Quit;

Data SNS.TBPARCEIROAUTORIZADO_SOLAR;
Set TBPARCEIROAUTORIZADO_SOLAR
;Run;
