data indata_ro;
	infile pipedir truncover;
	input line $char1000.;
	length diretorio $1000 database 8;
	retain diretorio;
	if substr(line,length(line),1) = ":" then diretorio = substr(line,1,length(line)-1);
	if substr(line,1,1) = "-" then do;
		call scan(line,9,pos,len,' ','');
			if len > 0 then arquivo1 = substr(line,pos,len);
		call scan(line,10,pos,len,' ','');
			if len > 0 then arquivo2 = substr(line,pos,len);
		arquivo = catx("",arquivo1,arquivo2);
		call scan(line,3,pos,len,' ','');
			if pos ne 0 then usuario = substr(line,pos,len);
		call scan(line,5,pos,len,' ','');
			if substr(line,pos,len) ne "" then tamanho = input(substr(line,pos,len),22.);
		call scan(line,6,pos,len,' ','');
			if substr(line,pos,len) ne "" then data = input(substr(line,pos,len),yymmdd10.);
	end;

	format tamanho commax22.0
		   data    ddmmyy10.
		   database ddmmyy10.
	;

	database = mdy(month(data),1,year(data));

	drop line pos len arquivo1 arquivo2;
	if arquivo ne "";
run;

   by descending tamanho;
