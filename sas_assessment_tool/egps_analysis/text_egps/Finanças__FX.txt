Proc SQL;

create table work.clientes_base as
select input(cpfcnpj, best14.) as cpfcnpj, cdcliente, nmcliente
from planctrl.TB001_SEGMENTACAO;


delete from work.clientes_base
where cdcliente in (1276,181666,540642,1256206);

create table work.agrupada_cliente as
select * from work.base_agrupada as b
left join clientes_base c
on c.cpfcnpj = b.cnpj;

Create table base_operacao as
Select ChaveUnica, CdObjetoNegociacao, CdOperacao, ModalidadeProduto, GrupoConciliacao, max(datainicioop) as datainicioop format date9., max(cdcliente) as cdcliente, sum(vrsaldoprincipalatualme) as vrsaldoprincipalme, Count (*) as Qtd
From PLANCTRL.REPOS_ANALITICO_&DtBase.
Where cdModalidadeProduto in (2, 3, 11, 16, 24, 25, 26, 28, 27, 29, 30, 31) and CdOperacao NE ""
Group by ChaveUnica, CdObjetoNegociacao, CdOperacao, ModalidadeProduto, GrupoConciliacao;

create table work.agru_cli_ope_sap as
select * from work.agrupada_cliente as b
inner join base_operacao o
on input(o.cdoperacao, best14.) = b.codsap
and o.datainicioop = "Data de Fechamento"n;

update work.agrupada_cliente
set codsap = -1
where idnew not in ( select idnew from work.agru_cli_ope_sap);

create table work.agru_cli_ope_nsap as
select * from work.agrupada_cliente as b
inner join base_operacao o
on o.datainicioop = "Data de Fechamento"n
and o.cdcliente = b.cdcliente
and o.vrsaldoprincipalme = "Volume ME"n
where b.codsap = -1
;

update work.agrupada_cliente
set codsap = -2
where codsap not in ( select codsap from work.agru_cli_ope_sap)
and codsap not in ( select input(cdoperacao, best14.) from work.agru_cli_ope_nsap);

delete from work.agru_cli_ope_nsap
where cdobjetonegociacao in (select cdobjetonegociacao from work.agru_cli_ope_sap);

create table work.agru_cli_ope_cldt as
select * from work.agrupada_cliente as b
left join base_operacao o
on o.datainicioop = "Data de Fechamento"n
and o.cdcliente = b.cdcliente
where b.codsap = -2
;

delete from work.agru_cli_ope_cldt
where cdobjetonegociacao in (select cdobjetonegociacao from work.agru_cli_ope_sap);

delete from work.agru_cli_ope_cldt
where cdobjetonegociacao in (select cdobjetonegociacao from work.agru_cli_ope_nsap);

Run;

Proc Sort nodupkey Data=Work.agru_cli_ope_cldt; by cdobjetonegociacao; Run;

	Work.agru_cli_ope_nsap
     work.agru_cli_ope_cldt;

create table work.saida as
select anomes, chave_unica, cdobjetonegociacao, cdmodalidade_produto_sas,
cdveiculo, cdgrupoconciliacao,
sum(carteira_contabil_brl) as carteira_contabil, sum(res_dia_gerencial_sas_brl) as res_contabil,
sum(Resultado_Acionista_BRL) as res_acionista, sum(Resultado_Comercial_BRL) as res_comercial, sum(Resultado_Tesouraria_BRL) as res_tesouraria
from PLANCTRL.BASE_SAIDA_SIC_&DtBase.
where cdobjetonegociacao in (select cdobjetonegociacao from ope_tratadas)
group by anomes,  chave_unica, cdobjetonegociacao, cdmodalidade_produto_sas,
cdveiculo, cdgrupoconciliacao;

create table work.final as
select * from work.ope_tratadas as o
left join work.saida s
on s.cdobjetonegociacao = o.cdobjetonegociacao;

create table work_naoencontrados as
select * from work.agrupada_cliente
where idnew not in ( select idnew from work.final);

Select anomes, chave_unica, cdobjetonegociacao, cdmodalidade_produto_sas,
cdveiculo, cdgrupoconciliacao,
sum(carteira_contabil_brl) as carteira_contabil, sum(res_dia_gerencial_sas_brl) as res_contabil,

from PLANCTRL.BASE_SAIDA_SIC_&DtBase.
where cdModalidade_Produto_SAS in (2, 3, 11, 16, 24, 25, 26, 28, 27, 29, 30, 31)
and chave_unica not in (select chave_unica from work.final)
group by anomes,  chave_unica, cdobjetonegociacao, cdmodalidade_produto_sas,
cdveiculo, cdgrupoconciliacao;
