%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
CREATE TABLE Df_Producao AS
SELECT * FROM connection to dbcon
(
SELECT
NuProposta,
SAFRA,
NUCONTRATOGESTAO,
VRFINANCIAMENTO,
NmLoja,
STATUS_PROPOSTAS,
NmColaborador,
DSSUBPRODUTO,
DSMOTIVORECUSA

FROM(

SELECT DISTINCT
dateadd(DD, -(DAY(A.DTENTRADAPROPOSTA)-1), CAST(A.DTENTRADAPROPOSTA AS DATE)) AS SAFRA,
K.NuProposta,
D.NmLoja,
A.VRFINANCIAMENTO,
A.NUCONTRATOGESTAO,
T.NMColaborador AS NmColaborador,

CASE
    WHEN A.STPROPOSTA IN ('A', 'PA') THEN 'APROVADAS'
    WHEN A.STPROPOSTA = 'AP' THEN 'CONTRATOS'
    WHEN A.STPROPOSTA = 'N' THEN 'NEGADAS'
    ELSE 'OUTROS'
END AS STATUS_PROPOSTAS,

CASE
     WHEN H.CDMOTIVORECUSA = 213 THEN 'CONDICIONADA'
     WHEN H.CDMOTIVORECUSA = 212 THEN 'FRAUDE-COMPROVADA'
     WHEN H.CDMOTIVORECUSA = 144 THEN 'SUSPEITA-DE-FRAUDE'
     WHEN H.CDMOTIVORECUSA = 223 THEN 'FRAUDE-COMPROVADA-MOTOR'
	 WHEN H.CDMOTIVORECUSA IS NULL THEN 'APROVADA'
     ELSE 'OUTROS'
END AS DSMOTIVORECUSA,

B.DSSUBPRODUTO,

ROW_NUMBER() OVER(PARTITION BY K.NuProposta ORDER BY K.DtStatusProposta DESC) AS rn

FROM DBA.TBPROPOSTA AS A
LEFT JOIN DBA.TbLogStatusProposta AS K ON A.NuProposta = K.NuProposta
LEFT JOIN DBA.TBSUBPRODUTO AS B ON A.CDSUBPRODUTO=B.CDSUBPRODUTO
INNER JOIN (SELECT DISTINCT NUPROPOSTA FROM DBA.TBLOGSTATUSPROPOSTA WHERE Stproposta IN ('A','AP') AND NUPROPOSTA IS NOT NULL) AS C ON A.NUPROPOSTA = C.NUPROPOSTA
LEFT JOIN DBA.TBLOJA AS D ON A.CDLOJA = D.CDLOJA
LEFT JOIN DBCRED.TBPROPOSTARECUSA AS G ON A.NUPROPOSTA= G.NUPROPOSTA
LEFT JOIN DBCRED.TBMOTIVORECUSA AS H ON G.CDMOTIVORECUSA= H.CDMOTIVORECUSA
LEFT JOIN DBA.TbColaborador AS T ON  A.CDCOLABORADOR = T.CDCOLABORADOR

WHERE CAST(a.dtENTRADAproposta AS DATE )>= '2022-09-01'
AND B.Cdproduto= 12

)tb_producao

WHERE rn = 1
)
;RUN;

/* ================== DfFraudes ================== */
/* ========================================================= */

%CRIAVARDB;
proc sql;
%CONECTADB(server=CTR);
CREATE TABLE Df_Fraudes AS
SELECT * FROM connection to dbcon
(
SELECT DISTINCT
a.NuContratoGestao,
C.NuProposta,
a.DT_ULT_ALT,
d.DtBaixaTarefa,
b.CD_SITUACAO,
b.DS_SITUACAO,
CAST(c.VRFINANCIAMENTO AS FLOAT) AS VALOR_PERDAS,
g.DsTipoFraude

FROM SCC_FIN..TB_SITUACAO_CONTRATO a
LEFT JOIN SCC_FIN..TB_SITUACAO b ON a.CD_SITUACAO = b.CD_SITUACAO
LEFT JOIN dbcred..tbproposta AS c ON a.NUCONTRATOGESTAO = c.NUCONTRATOGESTAO
LEFT JOIN DbPrevencaoFraude..TbAnaliseTratativaFraude AS d ON c.nucontratogestao = d.nucontrato
LEFT JOIN DBSLT..TB_SOLICITACAO AS f ON d.NUSOLICITACAO = f.Nu_Solicitacao
LEFT JOIN DbPrevencaoFraude..TbTipoFraude AS g ON d.cdtipofraude=g.cdtipofraude
LEFT JOIN dbcor..TBSUBPRODUTO AS h ON c.CDSUBPRODUTO=h.CDSUBPRODUTO

WHERE a.CD_SITUACAO IN (4,11)
AND convert(DATE, a.DT_ULT_ALT)>= '2022-09-01'
AND h.Cdproduto= 12
)
;RUN;

PROC SQL;
CREATE TABLE Df_FraudesFinal AS
SELECT
	NuContratoGestao
	,NuProposta
	,DT_ULT_ALT
	,DtBaixaTarefa
	,CD_SITUACAO
	,DS_SITUACAO
	,VALOR_PERDAS
FROM Df_Fraudes
;RUN;

PROC SORT DATA=Df_FraudesFinal NODUPKEY; BY NuContratoGestao Nuproposta ;RUN;
