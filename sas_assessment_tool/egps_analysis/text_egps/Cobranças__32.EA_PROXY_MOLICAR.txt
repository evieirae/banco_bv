PROC SQL;
CREATE TABLE RESUMO_MOLICAR AS
SELECT
       A.Chave_Concat_2,
       A.Chave_Concat_3,
       A.QTD,

       
       A.MEDIA,
       A.MINIMO,
       A.MAXIMO,
	   A.DESVIO_PADRAO,
       B.MODA,
       B.FREQ_MODA,
       (B.FREQ_MODA / A.QTD) AS PCT_MODA FORMAT=PERCENT8.2,

       
       A.MEDIA_SD,
       A.MINIMO_SD,
       A.MAXIMO_SD,
	   A.DESVIO_PADRAO_SD,
       C.MODA_SD,
       C.FREQ_MODA_SD,
       (C.FREQ_MODA_SD / A.QTD) AS PCT_MODA_SD FORMAT=PERCENT8.2

FROM
(
    SELECT
           UPCASE(
               CATX('|',
                    CdMarca,
                    CdModelo,
                    AaFabricacao,
                    AaModelo
               )
           ) AS Chave_Concat_2 LENGTH=500,

           UPCASE(
               CATX('|',
                    CdMarca,
                    CdModelo,
                    AaModelo
               )
           ) AS Chave_Concat_3 LENGTH=500,

           COUNT(*) AS QTD,

           
           MEAN(molicar_defl) AS MEDIA,
           MIN(molicar_defl)  AS MINIMO,
           MAX(molicar_defl)  AS MAXIMO,
		   STD(molicar_defl)  AS DESVIO_PADRAO,

           
           MEAN(custo_sdetran) AS MEDIA_SD,
           MIN(custo_sdetran)  AS MINIMO_SD,
           MAX(custo_sdetran)  AS MAXIMO_SD,
		   STD(custo_sdetran)  AS DESVIO_PADRAO_SD

    FROM base_c A
    LEFT JOIN Base_inicial_v2 B
        ON A.nucontratogestao = B.nucontrato
    WHERE B.nucontrato IS NOT NULL
    GROUP BY Chave_Concat_2, Chave_Concat_3
) A

LEFT JOIN MODA_MOLICAR B
    ON A.Chave_Concat_2 = B.Chave_Concat_2

LEFT JOIN MODA_SDETRAN C
    ON A.Chave_Concat_2 = C.Chave_Concat_2

ORDER BY A.QTD DESC,
         A.Chave_Concat_2 DESC
;
QUIT;
