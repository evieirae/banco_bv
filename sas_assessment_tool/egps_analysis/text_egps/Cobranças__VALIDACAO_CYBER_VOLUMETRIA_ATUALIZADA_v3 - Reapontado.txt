CREATE TABLE PARCELA AS
SELECT DISTINCT
*
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT
    &REF_M0. AS REF_M0,
    CAST(&DIA_ANTERIOR. AS DATE) AS DIA_ANTERIOR,
    t1.cd_contrato,
    t1.sq_contrato_financeiro,
    DATE(MIN(CASE
        WHEN t1.st_parcela <> 'Q'
        OR (t1.st_parcela = 'Q'
        AND DATE(t1.dt_processamento_ult_recebimento) >= &DIA_ANTERIOR.) THEN t1.dt_vencimento
        ELSE NULL
    END)) AS DtPrimeiroVenc
FROM
    bv-pdbd-prd.dvry_core.tbparcela AS t1
WHERE
    t1.cd_contrato IN (&CONTRATO1.)
GROUP BY
    t1.cd_contrato,
    t1.sq_contrato_financeiro

)
SELECT * FROM QUERY_1

)
		GROUP BY cd_contrato

		HAVING sq_contrato_financeiro = MAX(sq_contrato_financeiro)
;QUIT;

DATA PARCELA2;
SET PARCELA2 PARCELA;
RUN;

CREATE TABLE
	ELEGIVEIS_QUEBRAS AS
SELECT DISTINCT
	A.*,
	CASE
		WHEN A.FLAG_LIQUIDA = 1 THEN 'LIQUIDADO'
		WHEN A.ATRASO_ATUALIZADO = 0 THEN 'EM DIA'
		WHEN A.ATRASO_ATUALIZADO < 6 AND A.ATRASO_ATUALIZADO IS NOT MISSING THEN '<6'
		WHEN A.ATRASO_ATUALIZADO BETWEEN 6 AND 15 THEN 'ATIVA'
		WHEN A.ATRASO_ATUALIZADO BETWEEN 16 AND 70 THEN 'ADM'
		WHEN A.ATRASO_ATUALIZADO > 70 THEN 'CONTENCIOSO'
		WHEN B.DTCESSAO IS NOT NULL THEN 'FIDIC'
		ELSE 'OUTRO'
		END AS ATRASO_RECENTE,
	CASE
		WHEN A.ATRASO_MARCACAO = 0 THEN 'EM DIA'
		WHEN A.ATRASO_MARCACAO < 6 AND A.ATRASO_MARCACAO IS NOT MISSING THEN '<6'
		WHEN A.ATRASO_MARCACAO BETWEEN 6 AND 15 THEN 'ATIVA'
		WHEN A.ATRASO_MARCACAO BETWEEN 16 AND 70 THEN 'ADM'
		WHEN A.ATRASO_MARCACAO > 70 THEN 'CONTENCIOSO'
		ELSE 'OUTRO'
		END AS ATRASO_TBSCORE
FROM
	ELEGIVEIS_ATRASO_MARCACAO A
LEFT JOIN CONNECTION TO PERM (
    SELECT
    A.nucontrato,
    A.sqcontratofinanceiro,
    A.vrpdd AS PDD_M1,
    A.nudiasatraso AS ATRASO_M1,
    (CAST(A.vrpermanenciaapropriado AS NUMERIC) + CAST(A.vrtaxacontrato AS NUMERIC)) AS SALDO_M1,
    A.dtprocessamentoliquidacao,
    A.dtcessao
FROM
    bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A
WHERE
    A.dtprocessamento = &DATA_MENOS_UM.
    AND A.cdcarteira IN (1, 3, 4, 5, 12)




   ) B
   	ON
   		A.NUCONTRATO = INPUT(B.NUCONTRATO, 14.)
	AND
		A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO
;
QUIT;
