%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbMunicipio as
select * from connection to dbcon
(
	SELECT
		 Municipio.CdMunicipioIBGE
		,Municipio.CdMunicipio
		,Municipio.NmMunicipio AS DsMunicipio
		,Municipio.CdUF
		,Filial.CdFilialSistemaOrigem as CdFilial
		,Filial.NmFilial AS DsFilial
	FROM DWBI_DTWH.TbMunicipio Municipio

		LEFT JOIN DWBI_DTWH.TbFilial Filial
			ON  Municipio.CdFilial = Filial.CdFilialSistemaOrigem
			AND Filial.CdSistemaOrigem = 209

)
;
run;
quit;

proc sql;
create table TbMunicipioFiliais as
select
	 MunicipioFilial.CdMunicipioIBGE
	,MunicipioFilial.CdMunicipio
	,MunicipioFilial.DsMunicipio
	,MunicipioFilial.CdUF
	,Uf.DsEstado
	,Uf.DsRegiao
	,EstruturaComercial.CdAreaEstruturaComercial
	,EstruturaComercial.NmAreaEstruturaComercial
	,EstruturaComercial.CdSubEstruturaComercialA
	,EstruturaComercial.NmSubEstruturaComercialA
from TbMunicipio MunicipioFilial

	left join basegrc.TbDimuf Uf
		on  MunicipioFilial.CdUF = Uf.CdUF

	left join bases.TbEstruturaComercial EstruturaComercial
		on  MunicipioFilial.CdFilial = EstruturaComercial.CdFilial
		and today() between EstruturaComercial.DtInicioVigencia and EstruturaComercial.DtFimVigencia
		and EstruturaComercial.NmNivel = 'N2'
;run;

proc export
data=TbMunicipioFiliais
file='/projetos/MISV/sas-misv-base-indicadores-varejo/outbound/TbMunicipioFiliais.csv' replace
dbms=dlm;
delimiter=';';
run;
