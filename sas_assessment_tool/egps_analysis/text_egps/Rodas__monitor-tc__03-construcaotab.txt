/*DATAS DE CADA PRODUTO*/
PROC SQL;
CREATE TABLE BASE_DATA AS SELECT
NMGRUPOPRODUTOMETA AS PRODUTO_REF
,MAX(DTCONTRATO) AS DATA_REF FORMAT DDMMYY10.
FROM tmp.TBPRODUCAO
where NMGRUPOPRODUTOMETA IN('Leves','Motos','Pesados')
GROUP BY 1
;RUN;

PROC SQL;
CREATE TABLE TAB AS SELECT
A.NMGRUPOPRODUTOMETA
,MAX(A.DTCONTRATO)                                    AS DATA FORMAT DDMMYY10.
,MIN(A.VrTarifaAvaliacaoBem)                                     AS Valor_Minimo
,AVG(A.VrTarifaAvaliacaoBem)                                     AS Valor_Medio
,MAX(A.VrTarifaAvaliacaoBem)                                     AS Valor_Maximo
,SUM(CASE WHEN VrTarifaAvaliacaoBem > 0 THEN 1 ELSE 0 END)/
COUNT(NUCONTRATO)                                          AS TAB

FROM TMP.TBPRODUCAO AS A
INNER JOIN BASE_DATA AS B
ON B.PRODUTO_REF = A.NMGRUPOPRODUTOMETA
AND B.DATA_REF = A.DTCONTRATO
WHERE NMGRUPOPRODUTOMETA IN('Leves','Motos','Pesados')
GROUP BY 1
ORDER BY NMGRUPOPRODUTOMETA
;RUN;

/*TAB CALCULADA*/

Proc sql;
Create table TAB_CALC as select

NMGRUPOPRODUTOMETA AS PRODUTO
,DATA
,Valor_Minimo format dollar10.2
,Valor_Medio format dollar10.2
,Valor_Maximo format dollar10.2
,TAB format percent10.2
,CASE WHEN NMGRUPOPRODUTOMETA = 'Leves' and Valor_Maximo NE &TAB_LEVES. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Leves' and Valor_Minimo NE &TAB_LEVES. THEN 'Alerta'
	  WHEN NMGRUPOPRODUTOMETA = 'Leves' and Valor_Medio  NE &TAB_LEVES. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Motos' and Valor_Maximo NE &TAB_MOTOS. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Motos' and Valor_Minimo NE &TAB_MOTOS. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Motos' and Valor_Medio  NE &TAB_MOTOS. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Pesados' and Valor_Maximo  NE &TAB_PESADOS. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Pesados' and Valor_Minimo  NE &TAB_PESADOS. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Pesados' and Valor_Medio   NE &TAB_PESADOS. THEN 'Alerta'
ELSE 'S/Alerta' END AS Alerta_Valor
,CASE WHEN NMGRUPOPRODUTOMETA = 'Leves' and TAB > &TAB_LEVES_MAX. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Leves' and TAB < &TAB_LEVES_MIN. THEN 'Alerta'
	  WHEN NMGRUPOPRODUTOMETA = 'Motos' and TAB > &TAB_MOTOS_MAX. THEN 'Alerta'
	  WHEN NMGRUPOPRODUTOMETA = 'Motos' and TAB < &TAB_MOTOS_MIN. THEN 'Alerta'
	  WHEN NMGRUPOPRODUTOMETA = 'Pesados' and TAB > &TAB_PESADOS_MAX. THEN 'Alerta'
      WHEN NMGRUPOPRODUTOMETA = 'Pesados' and TAB < &TAB_PESADOS_MIN. THEN 'Alerta'
ELSE 'S/Alerta' END AS Alerta_Taxa

FROM TAB

;RUN;

/*BASE PARA ANALISE DE INCOERENCIAS */

PROC SQL;
CREATE TABLE PRODUCAO AS SELECT
A.*
,CASE WHEN A.VRTARIFAAVALIACAOBEM>0 THEN 1 ELSE 0 END AS FL_TAB
,C.CdUf

FROM BASES.TBPRODUCAO AS A

LEFT JOIN BASES.TBPARCEIROCOMERCIAL AS C
ON A.CdParceiroComercial = C.CdParceiroComercial

WHERE A.NUANOMES>=202101 AND A.NMGRUPOPRODUTOMETA IN ('Leves','Motos','Pesados')

;RUN;

PROC SORT DATA= PRODUCAO DUPOUT=DUPLI OUT= PRODUCAO NODUPKEY;
BY
NUCONTRATO
DTCONTRATO
NUPROPOSTA
DTENTRADAPROPOSTA

;RUN;
