%macro ExecutaSomenteSegundaOuDomingo();
    %let dow   = %sysfunc(weekday(%sysfunc(today())));

    %if (&dow = 2) or (&dow = 1) %then
		%do;
        	%put LOG: EXECUTANDO ;
			%include '/infogrc/funil_pricing/leves/vm/code/06-ImportXLSX.sas';
			%include '/infogrc/funil_pricing/leves/vm/code/07-Base Simulacoes.sas';

			%include '/infogrc/funil_pricing/leves/vm/code/09-Tabelao Final.sas';
    	%end;
    %else
		%do;
        	%put LOG: N EXECUTOU ;
    	%end;
%mend;
