PROC SQL;
CONNECT TO BIGQUERY AS PERM (PROJECT='bv-sadi-dtm' CREDFILE='/projetos/acessos/bq/di/sadi-dtm.json'
      SCHEMA='bv-pdbd-prd' READ_MODE=STORAGE);
CREATE TABLE BASE_ORACLE AS SELECT DISTINCT
G.*
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT
cast(t1.dmacct as INT64) AS nucontrato,
cast(t1.dmdays as INT64) AS ATRASO,
t1.dmque3 AS letra,
cast(t2.u2score1 as INT64) AS score,
t1.dmspdesc AS produto
FROM bv-pdbd-prd.trusted_rcvry.delqmst AS T1
LEFT JOIN bv-pdbd-prd.trusted_rcvry.uda2 AS T2 ON t1.dmacct = t2.u2acct
WHERE t1.dmspdesc IN
(
'EQUIPAMENTOS')
)
SELECT * FROM QUERY_1) AS G

;disconnect from perm;
QUIT;

PROC SQL;
CONNECT TO BIGQUERY AS PERM (PROJECT='bv-sadi-dtm' CREDFILE='/projetos/acessos/bq/di/sadi-dtm.json'
	SCHEMA='bv-pdbd-prd' READ_MODE=STORAGE);
CREATE TABLE carga_solar_gcon AS
SELECT
A.F2 AS NUCONTRATO,
A.F5 AS SCORE,
A.F7 AS LETRA,
E.NuDiasAtraso AS ATRASO,
E.LETRA_IQ,
E.SCORE_IQ,
E.DtGravacaoScore

FROM carga_solar_&data_subida. A 

LEFT JOIN CONNECTION TO PERM (SELECT
	A.NUCONTRATO,
	A.NuDiasAtraso,
	B.DsAcaoCobranca AS LETRA_IQ,
	B.nuscore as SCORE_IQ,
	B.DtGravacaoScore

	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil A
    LEFT JOIN
		(SELECT DISTINCT NUCONTRATOGESTAO, MAX(DtGravacaoScore) AS DtGravacaoScore, DsAcaoCobranca, nuscore
		 FROM bv-pdbd-prd.trusted_dbcobranca.tbscorecobranca
		 GROUP BY NUCONTRATOGESTAO, DsAcaoCobranca, nuscore) B ON A.NUCONTRATO = B.NUCONTRATOGESTAO

	WHERE A.DTPROCESSAMENTO = &data_subida_gcp. 
	  AND A.CDPRODUTO IN (13)) AS E ON A.F2 = INPUT(E.NUCONTRATO, 14.)
GROUP BY NUCONTRATO
HAVING E.DtGravacaoScore = MAX(E.DtGravacaoScore)

;disconnect from perm;
QUIT;

PROC SQL;
CREATE TABLE valida_carga_gcon AS
SELECT
A.*,
B.LETRA AS LETRA_CYBER,
B.SCORE AS SCORE_CYBER,
CASE WHEN A.LETRA = B.LETRA THEN 1
	 WHEN A.LETRA <> B.LETRA AND B.ATRASO <= 5 THEN 1
	 ELSE 0 END AS VALIDA_LETRA,

CASE WHEN A.LETRA = LETRA_IQ THEN 1
	 WHEN A.LETRA <> LETRA_IQ AND A.ATRASO <= 5 THEN 1
	 ELSE 0 END AS VALIDA_LETRA_IQ,

CASE WHEN A.SCORE = B.SCORE THEN 1
	 WHEN A.SCORE <> B.SCORE AND B.ATRASO <= 5 THEN 1
	 ELSE 0 END AS VALIDA_SCORE,

CASE WHEN A.SCORE = SCORE_IQ THEN 1
	 WHEN A.SCORE <> SCORE_IQ AND A.ATRASO <= 5 THEN 1
	 ELSE 0 END AS VALIDA_SCORE_IQ

FROM carga_solar_gcon A

LEFT JOIN (SELECT * FROM BASE_ORACLE  WHERE SCORE IS NOT NULL) B ON A.NUCONTRATO = B.NUCONTRATO
;QUIT;

      SCHEMA='bv-pdbd-prd' READ_MODE=STORAGE);
CREATE TABLE BASE_CYBER AS SELECT DISTINCT
G.*,
CASE WHEN ATRASO < 16 THEN 'a. até 15'
	 WHEN ATRASO < 91 THEN 'b. até 90'
	 WHEN ATRASO < 181 THEN 'c. até 180'
	 WHEN ATRASO < 361 THEN 'd. até 360'
	 WHEN ATRASO < 540 THEN 'e. até 540'
	 ELSE 'f. acima 540' END AS FX_ATRASO,
LENGTH(COMPRESS(score)) AS num_caracteres
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT
cast(t1.dmacct as INT64) AS nucontrato,
cast(t1.dmdays as INT64) AS ATRASO,
t1.dmque3 AS letra,
t2.u2score1 AS score,
t1.dmspdesc AS produto
FROM bv-pdbd-prd.trusted_rcvry.delqmst AS T1
LEFT JOIN bv-pdbd-prd.trusted_rcvry.uda2 AS T2 ON t1.dmacct = t2.u2acct
WHERE t1.dmspdesc IN
(
'EQUIPAMENTOS')
)
SELECT * FROM QUERY_1) AS G

;disconnect from perm;
QUIT;
