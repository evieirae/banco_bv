PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE PropostasESG AS
  SELECT * FROM connection TO dbcon
  (
    SELECT distinct
	  PRP.cdloja as cod_loja,
      PRP.NuProposta,
      PRP.DtEntradaProposta,
      PRP.StProposta,
      PRP.NuContrato,
      GRE.CdModelo,
      GRE.CdMolicar,
      GRE.CdMarca,
      GRE.AAModelo,
      GRE.AAFabricacao,
      GRE.DsCategoria,
      MLC.CdModelo AS CdModelo_Molicar,
      MLC.DsCombustivel
    FROM
      DBA.TbProposta PRP
    INNER JOIN
      DBA.TbPropGarantiaMaqEquip GRE
      ON PRP.NuProposta = GRE.NuProposta
    LEFT JOIN
      DBA.TbMolicar MLC
      ON GRE.CdMolicar = MLC.CdMolicar
    WHERE
      (
        MLC.DsCombustivel LIKE '%(EE)%'
        OR MLC.DsCombustivel LIKE '%(GE)%'
        OR MLC.DsCombustivel LIKE '%(DE)%'
	OR MLC.DsCombustivel LIKE '%Gas./Eta./E%'
      )
      AND PRP.DtEntradaProposta >= &DtIniPta.
    ORDER BY PRP.DtEntradaProposta);
RUN;

PROC SORT DATA=PropostasESG DUPOUT=DUPLI OUT=AUXL.PROPOSTAS_ESG NODUPKEY;
BY NUPROPOSTA
   DTENTRADAPROPOSTA;
RUN;

PROC SQL;
CREATE TABLE BaseESG AS
  SELECT DISTINCT
    UPCASE(CdMarca) AS CdMarca,
    UPCASE(CdModelo) AS CdModelo,
    UPCASE(combustivel) AS Combustivel
  FROM AUXL.BASE_MODELOS_ESG_&SafraESG.;
RUN;

PROC SQL;
CREATE TABLE TbModeloMolicar AS
  SELECT
    PRP.*,
    ESG.combustivel
  FROM
    AUXL.PROPOSTAS_ESG PRP
  LEFT JOIN
    BaseESG ESG
    ON ESG.CdModelo = UPCASE(PRP.CdModelo_Molicar)
  WHERE
    PRP.StProposta NOT IN ('C','CE')
  ORDER BY 2,1;
RUN;

PROC SORT DATA=TbModeloMolicar NODUPKEY; BY NuProposta; RUN;

PROC SQL; CREATE INDEX NuProposta ON TbModeloMolicar; QUIT;
