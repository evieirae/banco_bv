	PROC SQL;
	   CREATE TABLE WORK.CASHFLOWG5_AUX AS
	   SELECT t1.DtReferencia,
			  t1.DataFluxo,
			  "01jan1900"d as DataCarencia format ddmmyy10.,
			  t1.GrupoDeConciliacao,
			  t1.Veiculo,
			  catx("#","G5",Modalidade,t1.DataFluxo,t1.ValorFuturo) as ID,

			  "Onshore" as Onshore_Offshore,

			  t1.FluxoAmortizacaoVF,
			  t1.ValorFuturo
		  FROM WORK.CASHFLOWMOR_&dtDataBase. t1
	where GrupoDeConciliacao = 5
	;
	QUIT;

	/*Calcula valor presente*/;
	PROC SQL;
	   CREATE TABLE _Cash.CASHFLOWG5_&dtDataBase. AS
	   SELECT t1.DtReferencia,
			  t1.DataFluxo,
			  t1.DataCarencia,
			  t2.DU,

			  t1.Veiculo,

			  t1.FluxoAmortizacaoVF as FluxoAmortizacaoVP,

			  t1.FluxoJurosVF,
			  t1.FluxoAmortizacaoVF,
			  t1.ValorFuturo
		  FROM WORK.CASHFLOWG5_AUX t1 left join _PRE.TBCURVAPRE t2
		  on t1.DtReferencia = t2.Dt_Referencia and t1.DataFluxo=t2.Dt_Vencimento

/*Cria FluxoDV10 a partir das tabelas SensFluxo e TbPosicao*/;
%macro r02FuxoDV10(SensFluxo, TbPosicao, FluxoDV10, DtReferencia);

	/*Cria a tabela cenÃ¡rio base a partir da tabela SensFluxo onde o fator de risco Ã© o Base*/
	data CenBaseDV10_1;
		set &SensFluxo;
			where Fator_de_Risco eq 'Base Scenario';
		rename Valor_Presente = PresenteBase Valor_Futuro = FuturoBase;
	run;

	data CenBaseDV10_2;
		set &SensFluxo;
			where Fator_de_Risco eq 'Base Scenario Gerencial';
		rename Valor_Presente = PresenteBase Valor_Futuro = FuturoBase;
	run;

	/*Cria tabela FluxoDv10 a partir da Sensfluxo filtrada pelo cenÃ¡rio base adicionando na lateral os demais cenÃ¡rios */
	 proc sql noprint;
		create table FluxoDV10_1 as
			select distinct a.*
						,b.PresenteBase
						,b.FuturoBase
			from &SensFluxo(where=(Fator_de_Risco  not contains 'Base Scenario' AND Fator_de_Risco not contains 'Gerencial')) as a
				left joinNOT
					CenBaseDV10_1 as b
				on (a.Vencimento = b.Vencimento and a.PosID = b.PosID)
			;
		quit;

	proc sql noprint;
		create table FluxoDV10_2 as
			select distinct a.*
						,b.PresenteBase
						,b.FuturoBase
			from &SensFluxo(where=(Fator_de_Risco not contains 'Base Scenario' AND Fator_de_Risco contains 'Gerencial')) as a
				left joinNOT
					CenBaseDV10_2 as b
				on (a.Vencimento = b.Vencimento and a.PosID = b.PosID)
			;
		quit;

	/*Cria tb FluxoDV10 a partir da FluxoDV10 + TbPosicao*/
	proc sql noprint;
		create table FluxoDV10_1 as
			select  b.DtReferencia,
				a.PosID,
				a.Fator_de_Risco,
				b.Modalidade,
				b.TipodeNegociacao,
				b.MesaOperacao,
				b.CarteiraGerencial,
				b.BookGerencial,
				b.ProdutoGerencial,
				b.GrupoDeConciliacao,
				b.TpPNL,
				b.TipoContabil,
				b.Veiculo,
				a.Vencimento,
				a.Valor_Presente,
				a.PresenteBase,
				a.FuturoBase
			from FluxoDV10_1 as a
				inner join &TbPosicao as b
					on a.PosID = b.PosID
				where vencimento > b.DtReferencia
		;
	quit;

		proc sql noprint;
		create table FluxoDV10_2 as
			select  b.DtReferencia,
				a.PosID,
				a.Fator_de_Risco,
				b.Modalidade,
				b.TipodeNegociacao,
				b.MesaOperacao,
				b.CarteiraGerencial,
				b.BookGerencial,
				b.ProdutoGerencial,
				b.GrupoDeConciliacao,
				b.TpPNL,
				b.TipoContabil,
				b.Veiculo,
				a.Vencimento,
				a.Valor_Presente,
				a.PresenteBase,
				a.FuturoBase
			from FluxoDV10_2 as a
				inner join &TbPosicao as b
					on a.PosID = b.PosID
				where vencimento > b.DtReferencia
		;
	quit;

	proc sql noprint;
		create table &FluxoDV10 as
			select * from WORK.FluxoDV10_1
				union all
			select * from WORK.FluxoDV10_2
		;
	quit;

%mend r02FuxoDV10;

/*Processamento Base*/;
%macro r04ProcessamentoBase(FluxoDV10, DtReferencia, DataCorte);

	/*Atualiza valor do DV10, com a FluxoDV10 ou a Base_ajustes se houver*/
	data &FluxoDV10;
		set &FluxoDV10
		%if %sysfunc(exist(work.Base_ajustes)) %then work.Base_ajustes;
		;
		format vrDV10 commax32.2;

		If Fator_de_Risco In ('DvLongo_CDI'
								,'DvLongo_Eurodollar'
								,'DvLongo_Libor'
								,'DvLongo_SOFR'
								,'DvLongo_Selic'
								,'DvLongo_CDI - Gerencial'
								,'DvLongo_Eurodollar - Gerencial'
								,'DvLongo_Libor - Gerencial'
								,'DvLongo_SOFR - Gerencial'
								,'DvLongo_Selic - Gerencial') Then	vrDV10 = Valor_Presente;

		Else If Fator_de_Risco In ('DvCurto_CDI'
								,'DvCurto_Eurodollar'
								,'DvCurto_Libor'
								,'DvCurto_SOFR'
								,'DvCurto_Selic'
								,'DvCurto_CDI - Gerencial'
								,'DvCurto_Eurodollar - Gerencial'
								,'DvCurto_Libor - Gerencial'
								,'DvCurto_SOFR - Gerencial'
								,'DvCurto_Selic - Gerencial') Then
								Do;
									PresenteBase = 0;
									FuturoBase = 0;
									vrDV10 = Valor_Presente;
								End;
		Else vrDV10 = sum(-PresenteBase,Valor_Presente);
	run;

	/*Determina os fatores de risco a serem mantidos e os agrupa em PreFixado, CupomLibor e FatorRisco(nome original)*/
	data work.ListaExclusao;
		set &FluxoDV10
			(keep=Fator_de_Risco
			PosId
			where = (Fator_de_Risco in ('DvCurto_CDI'
										,'DvCurto_Selic'
										,'DvCurto_EuroDollar'
										,'DvCurto_Libor'
										,'DvCurto_SOFR'
										,'DvCurto_CDI - Gerencial'
										,'DvCurto_EuroDollar - Gerencial'
										,'DvCurto_Libor - Gerencial'
										,'DvCurto_SOFR - Gerencial'
										,'DvCurto_Selic - Gerencial'
										,'DvLongo_CDI'
										,'DvLongo_Selic'
										,'DvLongo_EuroDollar'
										,'DvLongo_Libor'
										,'DvLongo_SOFR'
										,'DvLongo_CDI - Gerencial'
										,'DvLongo_EuroDollar - Gerencial'
										,'DvLongo_Libor - Gerencial'
										,'DvLongo_SOFR - Gerencial'
										,'DvLongo_Selic - Gerencial')));

		If Fator_de_Risco in ('DvCurto_CDI'
								,'DvCurto_Selic'
								,'DvLongo_CDI'
								,'DvLongo_Selic') then Fator_de_Risco = 'Pre-Fixado';

		Else If Fator_de_Risco in ('DvCurto_EuroDollar'
									,'DvLongo_EuroDollar'
									,'DvCurto_Libor'
									,'DvLongo_Libor') then Fator_de_Risco = 'Cupom Libor';

		Else If Fator_de_Risco in ('DvCurto_SOFR'
									,'DvLongo_SOFR') then Fator_de_Risco = 'Cupom SOFR';

		Else If Fator_de_Risco in ('DvCurto_SOFR - Gerencial'
									,'DvLongo_SOFR - Gerencial') then Fator_de_Risco = 'Cupom SOFR Gerencial';

		Else If Fator_de_Risco in ('DvCurto_CDI - Gerencial'
								,'DvCurto_Selic - Gerencial'
								,'DvLongo_CDI - Gerencial'
								,'DvLongo_Selic - Gerencial') then Fator_de_Risco = 'Pre-Fixado Gerencial';

		Else If Fator_de_Risco in ('DvCurto_EuroDollar - Gerencial'
									,'DvLongo_EuroDollar - Gerencial'
									,'DvCurto_Libor - Gerencial'
									,'DvLongo_Libor - Gerencial') then Fator_de_Risco = 'Cupom Libor Gerencial';

		Else Fator_de_Risco = Fator_de_Risco;
	run;

	/*Comprime a tabela por grupo de fatores de risco*/
	proc sort data=work.ListaExclusao  nodup;
		by PosId
			Fator_de_Risco;
	run;

	/*Exclui os itens da lista de exlusÃ£o*/
	proc sql;
		create table &FluxoDV10 as
			select * from &FluxoDV10 a
				where not exists (select 1 From work.ListaExclusao b
				where a.PosId = b.PosId and
					a.Fator_de_Risco = b.Fator_de_Risco)
		;
	quit;

 	data &FluxoDV10;
		set &FluxoDV10;
			If Fator_de_Risco in ('DvCurto_CDI'
									,'DvCurto_Selic'
									,'DvLongo_CDI'
									,'DvLongo_Selic') then Fator_de_Risco = 'Pre-Fixado';

			Else If Fator_de_Risco in ('DvCurto_EuroDollar'
										,'DvLongo_EuroDollar'
										,'DvCurto_Libor'
										,'DvLongo_Libor') then Fator_de_Risco = 'Cupom Libor';

			Else If Fator_de_Risco in ('DvCurto_SOFR'
										,'DvLongo_SOFR') then Fator_de_Risco = 'Cupom SOFR';

			Else If Fator_de_Risco in ('DvCurto_SOFR - Gerencial'
									,'DvLongo_SOFR - Gerencial') then Fator_de_Risco = 'Cupom SOFR Gerencial';

			Else If Fator_de_Risco in ('DvCurto_CDI - Gerencial'
										,'DvCurto_Selic - Gerencial'
										,'DvLongo_CDI - Gerencial'
										,'DvLongo_Selic - Gerencial') then Fator_de_Risco = 'Pre-Fixado Gerencial';

			Else If Fator_de_Risco in ('DvCurto_EuroDollar - Gerencial'
										,'DvLongo_EuroDollar - Gerencial'
										,'DvCurto_Libor - Gerencial'
										,'DvLongo_Libor - Gerencial') then Fator_de_Risco = 'Cupom Libor Gerencial';

		Else Fator_de_Risco = Fator_de_Risco;

		if &DtReferencia. < &DataCorte. then
			do;
			if _n_ eq 1 then
				do;
				output;
				end;
			end;
		ELSE OUTPUT;
	run;

	/*Formata dados Alocação */;
	DATA &FluxoDV10;
		SET &FluxoDV10;
		format Valor_Presente PresenteBase FuturoBase vrDV10 BEST32.;
	RUN;

%mend r04ProcessamentoBase;

%macro r05FuxoDV10G5(stServidor, nDataBase, DataBaseDt);

		/*%let nDataBase=20200403;*/
		/*%let DataBaseDt="03apr2020"d;*/

		/*%let stServidor=MOR;*/

		PROC SQL;
		   CREATE TABLE WORK.TBCURVAPRE_&nDataBase. AS
		   SELECT t1.Nm_Curva,
				  t1.Dt_Referencia,
				  t1.Dt_Vencimento,
				  t1.DC,

			  FROM _PRE.TBCURVAPRE t1
			  WHERE t1.Dt_Referencia = &DataBaseDt.;
		QUIT;

		PROC SQL;
		   CREATE TABLE WORK.FLUXODV10G5 AS
		   SELECT DISTINCT t1.DtReferencia,
						   t1.ID as PosId,
						   t1.Indexador as Fator_de_Risco,
						   t1.ModalidadeSIC as  Modalidade,
						   t1.TipoNegociacao as TipoDeNegociacao,

						   t1.Carteira as CarteiraGerencial,

							when t1.MtMFluxo<0 then "P" end as TipoContabil,

						   t1.MtMFluxo as PresenteBase,
						   t1.ValorFuturo as FuturoBase,
						   SUM(calculated Valor_Presente, -t1.MtMFluxo)  as vrDV10
			FROM WORK.CASHFLOWG5_&nDataBase. t1 left join WORK.TBCURVAPRE_&nDataBase. t2
			  on t1.DtReferencia = t2.Dt_Referencia and t1.DataFluxo=t2.Dt_Vencimento

			;
			QUIT;
		%end;
		%else %do;
			PROC SQL;
			   CREATE TABLE WORK.FLUXODV10_AUX AS
			   SELECT t1.*
					  ,t2.GrupoDeConciliacao
				  FROM BaseDV10.FLUXODV10&stServidor._&nDataBase. t1
					left join POSONE.TBPOSICAOMOR_&nDataBase. t2
				  on t1.PosID = t2.PosId
			;
			QUIT;
		%end;

		PROC SQL;
		   CREATE TABLE WORK.FLUXODV10_AUX1 AS
		   SELECT t1.DtReferencia,
				  t1.PosId,
				  t1.Fator_de_Risco,
				  t1.Modalidade,
				  t1.TipoDeNegociacao,
				  t1.MesaOperacao,
				  t1.CarteiraGerencial,
				  t1.BookGerencial,
				  t1.ProdutoGerencial,
				  t1.TpPNL,
				  t1.TipoContabil,
				  t1.Veiculo,
				  t1.Vencimento,
				  t1.Valor_Presente,

		QUIT;

		PROC SQL;
		 CREATE TABLE BaseDV10.FLUXODV10&stServidor._&nDataBase. AS
			   SELECT * from WORK.FLUXODV10_AUX1 t1
				union all
				SELECT * from WORK.FLUXODV10_AUX2 t2

/*Funcao que verifica conteudo da tabela fornecida como parametro*/;
%macro f02checkConteudo(tabela);
	/*	Abre tabela*/
	%let dsid=%sysfunc(open(&tabela.,is));
	/*	Verifica se a tabela foi aberta*/
	%if &dsid. > 0 %then
		%do;
			/*	Se a tabela tiver sido aberta verifica a existÃªncia de linhas*/
			%let num=%sysfunc(attrn(&dsid.,nlobs));
			/*	Se tiver linhas na tabela retorna 1, se nÃ£o retorna 0*/
			%if &num. > 0 %then
				1;
			%else 0;
			/* Fecha tabela*/
			%let rc=%sysfunc(close(&dsid.));
		%end;
	/*	Se a tabela nÃ£o for aberta retorna 0*/
	%else 0;
%mend f02checkConteudo;

/*Rotina com o Fluxo Principal do Programa*/;
%macro r01Main(Servidor, dtDataBase);

	/*Verifica Existencia e conteudo das tabelas SensFluxo e TbPosicao*/
	%f03checkDependencia(Listener.sensfluxo&Servidor._&dtDataBase);
	%f03checkDependencia(Listener.tbposicao&Servidor._&dtDataBase);
	%f03checkDependencia(_CASH.SENSFLUXOMOR_&dtDataBase);
	%f03checkDependencia(POSONE.TBPOSICAOMOR_&dtDataBase);

	%if &dtDataBase. < &dtCorteOne. %then %do;

		/*Cria FluxoDV10 Algo*/
		%r02FuxoDV10(Listener.SensFluxo&Servidor._&dtDataBase
						,Listener.TbPosicao&Servidor._&dtDataBase
						,BaseDV10.FluxoDV10&Servidor._&dtDataBase
						,dtDataBase);

		/*Exclui os itens desnecessarios e organiza a tabela*/
		%r04ProcessamentoBase (BaseDV10.FluxoDV10&Servidor._&dtDataBase, &dtDataBase, &DataCorte);

		%if %f02checkConteudo (_CASH.SENSFLUXOMOR_&nDataBase.) NE 0 AND
			%f02checkConteudo (POSONE.TBPOSICAOMOR_&nDataBase.) NE 0 %then %do;

			/*Ajusta SensibilidadeFluxo e TBPosicao do Algo One*/
			%r01TrataBasesAlgoOne ();

			/*Cria FluxoDV10 Algo One*/
			%r02FuxoDV10(WORK.SENSFLUXOMOR_AlgoOne_&dtDataBase
						,WORK.TBPOSICAOMOR_AlgoOne_&dtDataBase
						,WORK.FluxoDV10A5&Servidor._&dtDataBase
						,dtDataBase);

			/*Exclui os itens desnecessarios e organiza a tabela*/
			%r04ProcessamentoBase (WORK.FluxoDV10A5&Servidor._&dtDataBase, &dtDataBase, &DataCorte);

		%end;

		/*Migra? dos grupos do Algo One*/
		%r03TrocaGrupos (&Servidor,&dtDataBase);

		/*Data de corte do uso do grupo 5 do Algo One*/
		%if &dtDataBase.>= 20200731 AND %f02checkConteudo (_Cash.CASHFLOWMOR_&dtDataBase.) NE 0 %then %do;

			/*Prepara Grupo 5 do Algo One*/
			%r01cashflow ();

			/*Troca grupo 5*/
			%r05FuxoDV10G5 (&Servidor, &dtDataBase, &DataBaseDt);

		%end;

	%end;
	%else %do;

		/*Ajusta SensibilidadeFluxo e TBPosicao do Algo One*/
		%r01TrataBasesAlgoOne ();

		/*Cria FluxoDV10 Algo One*/
		%r02FuxoDV10(WORK.SENSFLUXOMOR_AlgoOne_&dtDataBase
					,WORK.TBPOSICAOMOR_AlgoOne_&dtDataBase
					,WORK.FLUXODV10A5&Servidor._&dtDataBase
					,dtDataBase);

		/*Exclui os itens desnecessarios e organiza a tabela*/
		%r04ProcessamentoBase (WORK.FLUXODV10A5&Servidor._&dtDataBase, &dtDataBase, &DataCorte);

		/*Migra? dos grupos do Algo One*/
		%r03TrocaGrupos (&Servidor,&dtDataBase);

		/*Prepara Grupo 5 do Algo One*/
		%r01cashflow ();

		/*Troca grupo 5*/
		%r05FuxoDV10G5 (&Servidor, &dtDataBase, &DataBaseDt);

	%end;

	/*Chama Programa de Alocação de vertices*/
	%include "/&Modo_Exec./riscosmercado/rmsas/bin/sas/joblist/A002-AlocacaoDV10.sas";

	/*Define ni­vel de acesso ao  diretorios*/
	x "chmod 775 /&Modo_Exec./riscosmercado/rmsas/dados/datamart/rskm/sensibilidade/dv10/*sas7bdat";

%mend r01Main;
