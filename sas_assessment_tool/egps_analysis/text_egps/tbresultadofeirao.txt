/*CREATE TABLE TbMetaFeirao AS*/
CREATE TABLE bases.TbMetaFeirao AS
	SELECT
		Feirao.IdFeirao
		,Feirao.NmFeirao
		,put(Feirao.IdFeirao,15.)||'-'||Feirao.NmFeirao AS NmChaveFeirao
		,Feirao.DtInicioFeirao
		,Feirao.DtInicioFeirao    AS DtInicioApuracao    format date9.
		,Feirao.DtInicioFeirao+29 AS DtFimApuracao       format date9.
		,Feirao.DtInicioFeirao-30 AS DtInicio30DiasAntes format date9.
		,Feirao.DtInicioFeirao+59 AS DtFim30DiasDepois   format date9.
		,Feirao.DtInicioFeirao+44 AS DtFimApuracao45     format date9.
		,Feirao.DtInicioFeirao-45 AS DtInicio45DiasAntes format date9.
		,Feirao.DtInicioFeirao+89 AS DtFim45DiasDepois   format date9.

		,CASE
			WHEN Feirao.NmModeloEvento = 'Hiper_Saldão' THEN 'HIPER SALDÃO'
			WHEN Feirao.NmModeloEvento = 'Eventos_Itinerantes' THEN 'ITINERANTE'
			WHEN Feirao.NmModeloEvento = 'Eventos_em_Auto_Shopping' THEN 'AUTO SHOPPING'
			WHEN Feirao.NmModeloEvento = 'Ação_de_Vendas' THEN 'AÇÃO DE VENDAS'
			WHEN Feirao.NmModeloEvento = 'CVG_Ação_de_Vendas' THEN 'CVG AÇÃO DE VENDAS'
			WHEN Feirao.NmModeloEvento = 'Feirão BV Nacional' THEN 'FEIRÃO BV NACIONAL'
			ELSE Feirao.NmModeloEvento END AS NmModeloEvento

		,Feirao.NmTipoEvento
		,Parceirocomercial.NmParceiroComercial
		,Feirao.CdParceiroComercial
		,EstruturaComercial.NmAreaEstruturaComercial
		,EstruturaComercial.NmSubEstruturaComercialA
		,EstruturaComercial.LstSubstituicao
		,EstruturaComercial.DsLoginN2A
		,EstruturaComercial.DsLoginN3A
		,EstruturaComercial.DsLoginN3H
		,Feirao.VrProducaoHistoricaPDV
		,Feirao.VrProducaoHistoricaFeirao
		,Feirao.VrInvestimentoPDV
		,Feirao.VrInvestimentoBVFeirao
		,Feirao.VrProducaoIncrementalPDV
		,Feirao.VrProducaoMetaPDV
		,Feirao.VrProducaoMetaFeirao
		,CASE WHEN FeiraoCancelado.IdFeirao IS NOT NULL THEN 1 ELSE 0 END AS FlFeiraoCancelado

	FROM basegrc.TbFeiraoArquivos Feirao

		LEFT JOIN tmp.TbParceirocomercial Parceirocomercial
			ON Feirao.CdParceiroComercial = Parceirocomercial.CdParceirocomercial

		LEFT JOIN bases.TbEstruturaComercial EstruturaComercial
			ON Feirao.CdFilial = EstruturaComercial.CdFilial
			AND Feirao.DtInicioFeirao BETWEEN EstruturaComercial.DtInicioVigencia AND EstruturaComercial.DtFimVigencia
			AND EstruturaComercial.CdAreaEstruturaComercial IS NOT NULL
			AND EstruturaComercial.NmNivel = 'N2'

		LEFT JOIN basegrc.TbFeiraoCancelado FeiraoCancelado
			ON Feirao.IdFeirao = FeiraoCancelado.IdFeirao
	WHERE Feirao.DtInicioFeirao >= '01JAN2022'd
;run;
