PROC SQL;
CREATE TABLE TbParcVisao AS
 SELECT
  C.NuContrato
 ,C.NuProcesso
 ,P.NuParcela
 ,P.StParcela
 ,P.VrParcela
 ,P.DtVencimento
 ,C.MesCompetencia
 ,P.DtProcessamentoRecebimento
 ,C.DtRenegVisao
 ,CASE WHEN P.StParcela <> 'Q' 								THEN .
 	   WHEN P.DtProcessamentoRecebimento > C.DtLimite		THEN .

 ON C.NuContrato 	    = P.NuContrato
AND C.SqFinanceiraVisao = P.SqFinanceira

WHERE (P.StParcela <> 'Q' OR P.DtProcessamentoRecebimento >=  C.DtRefenciaVisao)

ORDER BY C.NuContrato , C.NuProcesso , P.NuParcela

;QUIT;RUN;
