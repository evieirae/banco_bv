CREATE TABLE AUXILIAR_ATRASO AS
SELECT DISTINCT
NUCONTA,
PRODUTO,
DT_INICIO_ATRASO,
COUNT(*) AS CONTADOR
FROM(SELECT DISTINCT
	NUCONTA,
	PRODUTO,
	SUM(-DIAS_AT,DT_ENVIO) AS DT_INICIO_ATRASO FORMAT DDMMYY10.
	FROM CARTEIRA_CL_BV3

PROC SQL;

SELECT DISTINCT
NUCONTA, PRODUTO, DIA_VENC, CONT_V, COUNT(*) AS CONTADOR

PROC SQL;

SELECT DISTINCT
A.NUCONTA,
A.PRODUTO,
A.ENVIO_MIN,
B.CD_DT_NOVA_DISPONIB AS DT_RETIRADA FORMAT DDMMYY10.,

A.NUCONTA,
