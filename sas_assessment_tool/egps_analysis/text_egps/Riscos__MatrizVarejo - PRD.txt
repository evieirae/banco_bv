PROC SQL;
   CREATE TABLE BS_PJ_INI AS
   SELECT DISTINCT
        T1.* ,
		(CASE WHEN T1.TP_QUADRO = 'PRINCIPAL' AND T3.DOCUMENTO IS NOT MISSING THEN 0
		      WHEN T3.DOCUMENTO IS NOT MISSING THEN 0
              WHEN T2.DOCUMENTO IS NOT MISSING THEN 0
              ELSE 1 END) AS FLAG

   FROM BS_PJ t1
   LEFT JOIN BS_PROCESSOS_VAREJO T2 ON (T1.DOC_NUM_QUADRO = T2.DOCUMENTO)
   LEFT JOIN BS_PROCESSOS_VAREJO_PJ T3 ON (T1.DOC_NUM_QUADRO = T3.DOCUMENTO and t3.Nome NOT = '')
   WHERE T1.TP_QUADRO NOT = 'S/INFO'

;QUIT;

PROC SQL;
   CREATE TABLE BS_APTOS AS
   SELECT DISTINCT
          t1.NOME,
          t1.NuConsultado,
          
            (SUM(t1.FLAG)) AS SUM_of_FLAG
      FROM WORK.BS_PJ_INI t1
      GROUP BY t1.NOME,
               t1.NuConsultado;
QUIT;

PROC SQL;
   CREATE TABLE BS_PJ_INI AS
   SELECT DISTINCT
        T1.*

   FROM BS_PJ t1
   LEFT JOIN BS_APTOS T2 ON (T1.NuConsultado = T2.NuConsultado)
   WHERE t2.SUM_of_FLAG = 0

;QUIT;

PROC SQL;
   CREATE TABLE LISTA_BLOQUEADOS AS
   		SELECT DISTINCT
		  	t1.DOC_NUM,
		   	t1.DATA_ATUAL,
		  	SUM((CASE WHEN T1.LISTA = 'CEIS' THEN 1 ELSE 0 END)) AS CEIS,
			SUM((CASE WHEN T1.LISTA = 'TRAB_ESCRAVO' THEN 1 ELSE 0 END)) AS TRAB_ESCRAVO,
			SUM((CASE WHEN T1.LISTA IN ('Proprietário preso por envol','Mídia Negativa') THEN 1 ELSE 0 END)) AS MIDIA,
			SUM((CASE WHEN T1.LISTA = 'CNEP' THEN 1 ELSE 0 END)) AS CNEP,
			SUM((CASE WHEN T1.LISTA = 'LENIENCIA' THEN 1 ELSE 0 END)) AS LENIENCIA,
			SUM((CASE WHEN T1.LISTA = 'CEPIM' THEN 1 ELSE 0 END)) AS CEPIM,
			SUM((CASE WHEN T1.LISTA IN ('RECUSADO','Recusados','Recusados Matriz','RECUSADO BD',
                                        'RECUSADO MATRIZ','RECUSADO MELIUZ','RECUSADO ONEPLD','RECUSADO SALES',
                                        'Renovação Contínua','RENOVACAO') THEN 1 ELSE 0 END)) AS RECUSADOS,
			SUM((CASE WHEN T1.LISTA IN ('Monitoramento de PLD','CONTA PONTE','PLD2','MONITORAMENTO TRANSACIONAL',
                                        'MONITORAMENTO_MELIUZ','Monitoramento de PLD / BAAS','Device') THEN 1 ELSE 0 END)) AS MONITORAMENTO_PLD,
			SUM((CASE WHEN T1.LISTA IN ('DENUNCIA','Denúncia') THEN 1 ELSE 0 END)) AS DENUNCIA,
			SUM((CASE WHEN T1.LISTA IN ('Reportados Coaf','Sancionados') THEN 1 ELSE 0 END)) AS OUTROS,
			SUM((CASE WHEN T1.LISTA = 'FRAUDES' THEN 1 ELSE 0 END)) AS LISTA_FRAUDES,
			SUM((CASE WHEN T1.LISTA = 'Favorecido Bloqueado' THEN 1 ELSE 0 END)) AS Fav_Bloqueado,
			SUM((CASE WHEN T1.LISTA IS MISSING THEN 1
                      WHEN T1.LISTA IN ('BLOQUEADO ONEPLD','BLOQUEADOS ONEPLD') THEN 1 ELSE 0 END)) AS BLOQUEADOS,
			SUM((CASE WHEN T1.LISTA = 'EX-COLABORADOR' THEN 1 ELSE 0 END)) AS EX_COLABORADOR,
			1 AS TOTAL

   FROM LISTA_BLOQUEADOS t1
    LEFT JOIN PEP_INTER T3 ON (T1.NOME = T3.NOME)
    LEFT JOIN SANCIONADOS_BS T4 ON (T1.NOME = T4.NOME)
   		WHERE T1.DOC_NUM NOT=. AND T1.DOC_NUM >0 AND t1.TPLISTA NOT= 'RETIRAR'
		GROUP BY T1.DOC_NUM
   ;
QUIT;

PROC SQL;
	CREATE TABLE BASE_VAREJO_FILTRO AS
		SELECT DISTINCT
		        NuConsultado,
				NOME,
				(CASE WHEN SUM(SCORE_PEP) > 0 THEN 1 ELSE 0 END) AS PEP,
                (CASE WHEN (SUM(SCORE_LISTA) - SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM)) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS INTERNAS'N,
				(CASE WHEN SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS EXTERNAS'N,
				(CASE WHEN SUM(SCORE_SANCIONADOS) > 0 THEN 1 ELSE 0 END)  AS 'SANÇÕES'N,
                (CASE WHEN SUM(MIDIA) > 0 THEN 1 ELSE 0 END) AS MIDIA,
			    (CASE WHEN SUM(SCORE_CNAE) > 0 THEN 1 ELSE 0 END) AS CNAE,
				(CASE WHEN SUM(CASE WHEN STATUS_PROCESSO IN ('REPROVADO') THEN 1 ELSE 0 END)> 0 THEN 1 ELSE 0 END) AS Processos

	FROM BS_PJ_INI
	GROUP BY NuConsultado,NOME
;
QUIT;

FORMAT STATUS_FINAL $150.;
STATUS_FINAL = 'Sem classificacao';


IF Processos = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo)';
IF Processos = 1 AND MIDIA = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo) + midia';
IF Processos = 1 AND MIDIA = 1 AND PEP = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo + pep) + midia';
IF Processos = 1 AND MIDIA = 1 AND CNAE = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + PEP) = 0 THEN STATUS_FINAL = 'Com risco(processo + CNAE) + midia';
IF Processos = 1 AND CNAE = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + PEP) = 0 THEN STATUS_FINAL = 'Com risco(processo + CNAE)' ;
IF Processos = 1 AND 'LISTAS INTERNAS'N  = 1 AND ( 'SANÇÕES'N + MIDIA + PEP + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + procesos)' ;
IF Processos = 1 AND PEP = 1 and ('LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(processo + pep)';



IF 'LISTAS INTERNAS'N = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas)';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND (PEP + Processos+ 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas) + midia';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND PEP = 1 AND (Processos + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + pep) + midia';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + PEP) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE) + midia';
IF 'LISTAS INTERNAS'N = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + MIDIA + PEP) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE)' ;
IF 'LISTAS INTERNAS'N = 1 AND  (Processos + 'SANÇÕES'N + MIDIA + PEP + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE)' ;


IF  MIDIA = 1 AND (PEP + Processos + 'SANÇÕES'N + CNAE + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Sem risco + midia';


IF  CNAE = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(CNAE)';


IF  'SANÇÕES'N = 1 AND (PEP + Processos + CNAE + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(sançoes)';



IF PEP = 1 AND ('LISTAS INTERNAS'N + Processos + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep)';
IF PEP = 1 AND MIDIA = 1 AND ('LISTAS INTERNAS'N + Processos+ 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep) + midia';
IF PEP = 1 AND MIDIA = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(pep + CNAE) + midia';
IF PEP = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(pep + CNAE)' ;
IF PEP = 1 AND 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND Processos = 1 and ( 'SANÇÕES'N  + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep + Listas Internas + Processos) + midia' ;


IF (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + Processos  + CNAE) = 0 THEN STATUS_FINAL = 'Sem risco';

PROC SQL;
   CREATE TABLE LISTA_BLOQUEADOS AS
   		SELECT DISTINCT
		  	t1.DOC_NUM,
		   	t1.DATA_ATUAL,
		  	SUM((CASE WHEN T1.LISTA = 'CEIS' THEN 1 ELSE 0 END)) AS CEIS,
			SUM((CASE WHEN T1.LISTA = 'TRAB_ESCRAVO' THEN 1 ELSE 0 END)) AS TRAB_ESCRAVO,
			SUM((CASE WHEN T1.LISTA IN ('Proprietário preso por envol','Mídia Negativa') THEN 1 ELSE 0 END)) AS MIDIA,
			SUM((CASE WHEN T1.LISTA = 'CNEP' THEN 1 ELSE 0 END)) AS CNEP,
			SUM((CASE WHEN T1.LISTA = 'LENIENCIA' THEN 1 ELSE 0 END)) AS LENIENCIA,
			SUM((CASE WHEN T1.LISTA = 'CEPIM' THEN 1 ELSE 0 END)) AS CEPIM,
			SUM((CASE WHEN T1.LISTA IN ('RECUSADO','Recusados','Recusados Matriz','RECUSADO BD',
                                        'RECUSADO MATRIZ','RECUSADO MELIUZ','RECUSADO ONEPLD','RECUSADO SALES',
                                        'Renovação Contínua','RENOVACAO') THEN 1 ELSE 0 END)) AS RECUSADOS,
			SUM((CASE WHEN T1.LISTA IN ('Monitoramento de PLD','CONTA PONTE','PLD2','MONITORAMENTO TRANSACIONAL',
                                        'MONITORAMENTO_MELIUZ','Monitoramento de PLD / BAAS','Device') THEN 1 ELSE 0 END)) AS MONITORAMENTO_PLD,
			SUM((CASE WHEN T1.LISTA IN ('DENUNCIA','Denúncia') THEN 1 ELSE 0 END)) AS DENUNCIA,
			SUM((CASE WHEN T1.LISTA IN ('Reportados Coaf','Sancionados') THEN 1 ELSE 0 END)) AS OUTROS,
			SUM((CASE WHEN T1.LISTA = 'FRAUDES' THEN 1 ELSE 0 END)) AS LISTA_FRAUDES,
			SUM((CASE WHEN T1.LISTA = 'Favorecido Bloqueado' THEN 1 ELSE 0 END)) AS Fav_Bloqueado,
			SUM((CASE WHEN T1.LISTA IS MISSING THEN 1
                      WHEN T1.LISTA IN ('BLOQUEADO ONEPLD','BLOQUEADOS ONEPLD') THEN 1 ELSE 0 END)) AS BLOQUEADOS,
			SUM((CASE WHEN T1.LISTA = 'EX-COLABORADOR' THEN 1 ELSE 0 END)) AS EX_COLABORADOR,
			1 AS TOTAL

   FROM LISTA_BLOQUEADOS t1
    LEFT JOIN PEP_INTER T3 ON (T1.NOME = T3.NOME)
    LEFT JOIN SANCIONADOS_BS T4 ON (T1.NOME = T4.NOME)
   		WHERE T1.DOC_NUM NOT=. AND T1.DOC_NUM >0 AND t1.TPLISTA NOT= 'RETIRAR'
		GROUP BY T1.DOC_NUM
   ;

PROC SQL;
	CREATE TABLE BS_PJ_INI AS
		SELECT DISTINCT
			A.*,
		    (CASE WHEN A.DOC_NUM_QUADRO IS MISSING THEN 'ESTRANGEIRO'
			      WHEN A.DOC_NUM_QUADRO = 0 THEN 'ESTRANGEIRO'
                  WHEN B.DOCUMENTO IS NOT MISSING THEN 'REPROVADO'

                  FORMAT $100. AS STATUS_PROCESSO,
          B.Motivo_da_Reprovacao

LEFT JOIN BS_PROCESSOS B ON (A.DOC_NUM_QUADRO = B.DOCUMENTO)

PROC SQL;
	CREATE TABLE BASE_VAREJO_FILTRO AS
		SELECT DISTINCT
		        NuConsultado,
				NOME,
				(CASE WHEN SUM(SCORE_PEP) > 0 THEN 1 ELSE 0 END) AS PEP,
                (CASE WHEN (SUM(SCORE_LISTA) - SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM)) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS INTERNAS'N,
				(CASE WHEN SUM(LISTA_CEIS + LISTA_TRAB_ESCRAVO + LISTA_CNEP + LISTA_LENIENCIA + LISTA_CEPIM) > 0 THEN 1 ELSE 0 END)  AS 'LISTAS EXTERNAS'N,
				(CASE WHEN SUM(SCORE_SANCIONADOS) > 0 THEN 1 ELSE 0 END)  AS 'SANÇÕES'N,
                (CASE WHEN SUM(MIDIA) > 0 THEN 1 ELSE 0 END) AS MIDIA,
			    (CASE WHEN SUM(SCORE_CNAE) > 0 THEN 1 ELSE 0 END) AS CNAE,
				(CASE WHEN SUM(CASE WHEN STATUS_PROCESSO IN ('REPROVADO') THEN 1 ELSE 0 END)> 0 THEN 1 ELSE 0 END) AS Processos

	FROM BS_PJ_INI
	GROUP BY NuConsultado,NOME
;
QUIT;

DATA BASE_VAREJO_FILTRO;
SET BASE_VAREJO_FILTRO;
FORMAT STATUS_FINAL $150.;
STATUS_FINAL = 'Sem classificacao';


IF Processos = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo)';
IF Processos = 1 AND MIDIA = 1 AND (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo) + midia';
IF Processos = 1 AND MIDIA = 1 AND PEP = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(processo + pep) + midia';
IF Processos = 1 AND MIDIA = 1 AND CNAE = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + PEP) = 0 THEN STATUS_FINAL = 'Com risco(processo + CNAE) + midia';
IF Processos = 1 AND CNAE = 1 AND ('LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + PEP) = 0 THEN STATUS_FINAL = 'Com risco(processo + CNAE)' ;
IF Processos = 1 AND 'LISTAS INTERNAS'N  = 1 AND ( 'SANÇÕES'N + MIDIA + PEP + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + procesos)' ;
IF Processos = 1 AND PEP = 1 and ('LISTAS INTERNAS'N + 'SANÇÕES'N + CNAE + MIDIA) = 0 THEN STATUS_FINAL = 'Com risco(processo + pep)';



IF 'LISTAS INTERNAS'N = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas)';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND (PEP + Processos+ 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas) + midia';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND PEP = 1 AND (Processos + 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + pep) + midia';
IF 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + PEP) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE) + midia';
IF 'LISTAS INTERNAS'N = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + MIDIA + PEP) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE)' ;
IF 'LISTAS INTERNAS'N = 1 AND  (Processos + 'SANÇÕES'N + MIDIA + PEP + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(listas internas + CNAE)' ;


IF  MIDIA = 1 AND (PEP + Processos + 'SANÇÕES'N + CNAE + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Sem risco + midia';


IF  CNAE = 1 AND (PEP + Processos + 'SANÇÕES'N + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(CNAE)';


IF  'SANÇÕES'N = 1 AND (PEP + Processos + CNAE + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(sançoes)';



IF PEP = 1 AND ('LISTAS INTERNAS'N + Processos + 'SANÇÕES'N + MIDIA + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep)';
IF PEP = 1 AND MIDIA = 1 AND ('LISTAS INTERNAS'N + Processos+ 'SANÇÕES'N + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep) + midia';
IF PEP = 1 AND MIDIA = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(pep + CNAE) + midia';
IF PEP = 1 AND CNAE = 1 AND (Processos + 'SANÇÕES'N + MIDIA + 'LISTAS INTERNAS'N) = 0 THEN STATUS_FINAL = 'Com risco(pep + CNAE)' ;
IF PEP = 1 AND 'LISTAS INTERNAS'N = 1 AND MIDIA = 1 AND Processos = 1 and ( 'SANÇÕES'N  + CNAE) = 0 THEN STATUS_FINAL = 'Com risco(pep + Listas Internas + Processos) + midia' ;


IF (PEP + 'LISTAS INTERNAS'N + 'SANÇÕES'N + MIDIA + Processos  + CNAE) = 0 THEN STATUS_FINAL = 'Sem risco';
RUN;
