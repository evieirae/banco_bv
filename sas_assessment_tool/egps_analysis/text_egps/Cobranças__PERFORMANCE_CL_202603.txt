PROC SQL;
CREATE TABLE PERF_CL AS
SELECT
	A.*,
	CASE
		WHEN A.RECEBIDO_PARCELADO > 0 AND A.RECEBIDO_QUITADO > 0 THEN (A.RECEBIDO_PARCELADO + A.RECEBIDO_QUITADO)
		WHEN A.RECEBIDO_PARCELADO =. AND A.RECEBIDO_QUITADO > 0 THEN A.RECEBIDO_QUITADO
		WHEN A.RECEBIDO_PARCELADO >0 AND A.RECEBIDO_QUITADO =. THEN A.RECEBIDO_PARCELADO
		END AS RECEBIMENTO_TOTAL
FROM PERF_CL.CL_TABLEAU AS A
JOIN (SELECT DISTINCT
				DATA_BASE,
				MAX(DT_REF) AS DT_REF FORMAT DDMMYY10.
				FROM PERF_CL.CL_TABLEAU
				WHERE DATA_BASE >= &DAT_EXEC.
				GROUP BY DATA_BASE
				) AS B

PROC SQL;
CREATE TABLE PERF_CL AS
SELECT
	DATA_BASE,
	ASSESSORIA,
	CL_STATUS,
	FX_ATRASO,
	(SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO)) AS PERFORMANCE,
	CASE
		WHEN DATA_BASE < 202111 AND FX_ATRASO = "75 A 120" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.50)
		WHEN DATA_BASE < 202111 AND FX_ATRASO = "121 A 180" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.30)
		WHEN DATA_BASE < 202111 AND FX_ATRASO = "181 A 360" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.20)

		WHEN FX_ATRASO = "75 A 120" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.50)
		WHEN FX_ATRASO = "121 A 180" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.20)
		WHEN FX_ATRASO = "181 A 360" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.30)
		WHEN FX_ATRASO = "361 A 720" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.40)
		WHEN FX_ATRASO = "ACIMA DE 720" THEN ((SUM(RECEBIMENTO_TOTAL)/SUM(SALDO_ENVIADO))*0.60)
	END AS PERFORMANCE_PESO
FROM PERF_CL
GROUP BY
	DATA_BASE,
	ASSESSORIA,
	CL_STATUS,
	FX_ATRASO
;RUN;
