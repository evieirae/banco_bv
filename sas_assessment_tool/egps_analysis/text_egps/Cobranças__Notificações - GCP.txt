CREATE TABLE BASE_NOTIFICACAO AS
SELECT DISTINCT A.*,  H.COD_ASSESSORIA,
CASE WHEN C.ATIVA_PRINC IN ('C') THEN 1 ELSE 0 END AS FLAG_ACAO_CONTRA,
CASE WHEN F.NUCONTRATO IS NULL THEN 0 ELSE 1 END AS FLAG_PROTESTO,
CASE WHEN E.FL_45_135 IS NULL THEN 0 ELSE 1 END AS FLAG_CARTA,
CASE WHEN G.NUCONTRATO IS NULL THEN 0 ELSE 1 END AS FLAG_NOTIFICADO
FROM COBR AS A

LEFT JOIN ASSESSORIA AS C
ON A.NUCONTRATO = C.NUCONTRATOGESTAO
LEFT JOIN DADOS_ASSESSORIA AS H
ON A.NUCONTRATO = H.CONTRATO
LEFT JOIN ENVIADOS_SOL_135 AS E
ON input(A.NUCONTRATO, 14.) = E.NUCONTRATO
LEFT JOIN PROTESTO AS F
ON input(A.NUCONTRATO, 14.) = F.NUCONTRATO
LEFT JOIN NOTIFICADOS AS G
ON input(A.NUCONTRATO, 14.) = G.NUCONTRATO
WHERE A.VRSALDO >= 6000
;RUN;
