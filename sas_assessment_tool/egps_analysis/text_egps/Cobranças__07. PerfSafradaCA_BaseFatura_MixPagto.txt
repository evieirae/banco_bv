,case when b.cdfaturaemissor ne . then 1
	  when a.vrtotalpagofatura > 0 then 1
	  when a.vrtotalpagofatura >= a.vrtotalfatura then 1 else 0 end as Fl_Regularizado 

,case when b.cdfaturaemissor ne . then 'Acordo CA'

	  when calculated Fl_Regularizado=1 and a.vrtotalfatura > a.vrpagamentominimo and a.vrpagamentominimo = a.vrtotalpagofatura and a.vrtotalpagofatura ne 0 then 'Pagamento Mínimo'
	  when calculated Fl_Regularizado=1 and calculated perc_pago_total > 99.99 and calculated perc_pago < 100 and a.vrpagamento2 ne 0 and calculated perc_pago ne calculated perc_pago_total then 'Parcelamento de Fatura'
	  when calculated Fl_Regularizado=1 and a.vrtotalpagofatura >= a.vrtotalfatura then 'Quitação'

	  when calculated Fl_Regularizado=1 and a.vrpagamentominimo < a.vrtotalpagofatura then 'Pagamento Parcial'
	  when calculated Fl_Regularizado=1 and a.vrtotalpagofatura ne 0 and a.vrpagamentominimo > a.vrtotalpagofatura then 'Inferior ao Mínimo'




else '' end as obs

,case when calculated obs = 'Quitação' and a.status_atual_conta = 'CREDITO EM LIQUIDACAO' and a.vrtotalpagofatura = 0 and a.vrtotalfatura = 0 then 'Crédito Liquidado'
	  when calculated obs = 'Quitação' 						then 'Quitação'
	  when calculated obs = 'Pagamento Parcial' 			then 'Pagamento Parcial'
	  when calculated obs = 'Pagamento Mínimo' 				then 'Pagamento Mínimo'
	  when calculated obs = 'Parcelamento de Fatura' 		then 'Parcelamento de Fatura'
	  when calculated obs = 'Acordo CA' 					then 'Acordo CA'
	  when calculated obs = 'Inferior ao Mínimo'			then 'Inferior ao Mínimo'
	  when today() < intnx('day', a.dtvencimentofatura, 3) 	then 'A Vencer'

,a.vrtotalpagofatura as vrtotalpagofatura2

select distinct
a.*

,(a.vrpagamento2/a.vrtotalfatura)*100 as perc_pago format 8.4
,(a.vrtotalpagofatura2/a.vrtotalfatura)*100 as perc_pago_total format 8.4

,case when b.cdfaturaemissor ne . then 1
	  when a.vrtotalpagofatura2 > 0 then 1
	  when a.vrtotalfatura <= a.vrtotalpagofatura2 then 1 else 0 end as Fl_Regularizado 

,case when b.cdfaturaemissor ne . then 'Acordo CA'
	  when calculated Fl_Regularizado=1 and a.vrtotalpagofatura2 >= a.vrtotalfatura and a.vrtotalpagofatura2 = a.vrpagamento2 then 'Quitação'
	  when calculated Fl_Regularizado=1 and a.vrtotalpagofatura2 >= a.vrtotalfatura and calculated perc_pago > 99.99 then 'Quitação'
	  when calculated Fl_Regularizado=1 and calculated perc_pago_total > 99.99 and calculated perc_pago < 100 and a.vrpagamento2 ne 0 and calculated perc_pago ne calculated perc_pago_total then 'Parcelamento de Fatura'
	  when calculated Fl_Regularizado=1 and a.vrtotalfatura > a.vrpagamentominimo and a.vrpagamentominimo = a.vrtotalpagofatura2 = a.vrpagamento2 then 'Pagamento Mínimo'
	  when calculated Fl_Regularizado=1 and a.vrpagamentominimo < a.vrtotalpagofatura2 then 'Pagamento Parcial'
	  when calculated Fl_Regularizado=1 and a.vrpagamento2 = 0 then 'Sem Pagamento' 
	  when calculated Fl_Regularizado=1 and a.vrpagamento2 ne 0 and a.vrpagamentominimo > a.vrtotalpagofatura2 then 'Inferior ao Mínimo'
	  when calculated Fl_Regularizado=1 then 'Quitação'

else '' end as obs

,case when calculated obs = 'Quitação' and a.status_atual_conta = 'CREDITO EM LIQUIDACAO' and a.vrpagamento2 = 0 then 'Crédito Liquidado'
	  when calculated obs = 'Quitação' 						then 'Quitação'
	  when calculated obs = 'Pagamento Parcial' 			then 'Pagamento Parcial'
	  when calculated obs = 'Pagamento Mínimo' 				then 'Pagamento Mínimo'
	  when calculated obs = 'Parcelamento de Fatura' 		then 'Parcelamento de Fatura'
	  when calculated obs = 'Acordo CA' 					then 'Acordo CA'
	  when calculated obs = 'Inferior ao Mínimo'			then 'Inferior ao Mínimo'
	  when today() < intnx('day', a.dtvencimentofatura, 3) 	then 'A Vencer'

	  else 'Sem Regularização' end as TP_PAGAMENTO

,'' as ORIGEM_PAGAMENTO

from base_faturas_pagto_b as a
left join satie.ACORDOS_PISMO as b on a.cdfaturaemissor=b.cdfaturaemissor

where a.dtvencimentofatura > '31oct2025'd
order by a.nucontacartao, a.dtcortefatura
;quit;
