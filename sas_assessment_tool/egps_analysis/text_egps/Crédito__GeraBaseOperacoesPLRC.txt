%macro tratamento(anomes,dataref);

	safraref = &anomes.;

	if dtdesembolso >  &dataref. then nmoperacao   = 'Não existe'; else
                                      nmoperacao   = 'Existe    ';

	if dtvencimento >= &dataref. then nmvencimento = 'A vencer'; else
                                      nmvencimento = 'Vencida ';

	if dtmovimento = .           then nmpagamento  = 'Não paga'; else
	if dtmovimento <= &dataref.  then nmpagamento  = 'Paga    '; else
                                      nmpagamento  = 'Não paga';

	if nmvencimento = 'Vencida ' and nmpagamento = 'Não paga' then nmstatusfinal = 'Em atraso      '; else
	if nmvencimento = 'Vencida ' and nmpagamento = 'Paga    ' then nmstatusfinal = 'Liquidada      ';

	if nmstatusfinal = 'Em atraso      ' then qtdiasatraso =  &dataref. - dtvencimento; else qtdiasatraso = 0;

	if dtmovimento ne . then atraso_historico = dtmovimento - dtvencimento; else atraso_historico = &dataref. - dtvencimento;

	if qtdiasatraso > 5  and floperacaoprejuizo ne 1 then flover05 = 1; else flover05 = 0;

	if qtdiasatraso > 30 and floperacaoprejuizo ne 1 then flover30 = 1; else flover30 = 0;
	if qtdiasatraso > 60 and floperacaoprejuizo ne 1 then flover60 = 1; else flover60 = 0;
	if qtdiasatraso > 90 and floperacaoprejuizo ne 1 then flover90 = 1; else flover90 = 0;

	if atraso_historico > 5  and floperacaoprejuizo ne 1 then flever05 = 1; else flever05 = 0;
	if atraso_historico > 15 and floperacaoprejuizo ne 1 then flever15 = 1; else flever15 = 0;
	if atraso_historico > 30 and floperacaoprejuizo ne 1 then flever30 = 1; else flever30 = 0;
