CREATE TABLE CONSULTA_AJUIZAMENTOb AS SELECT
* from
CONNECTION TO PERM	(

WITH QUERY_1 AS (

SELECT
    t1.nucontratogestao,
    t1.sqcontratofinanceiro,
    DATE(t1.dt_desp) AS Dt_DESP
FROM
    bv-pdbd-prd.trusted_scc_fin.saj_desp_contrato AS t1
WHERE
    t1.dt_desp > '2019-01-01'
    AND t1.cod_desp IN (14, 80)
GROUP BY
    t1.nucontratogestao,
    t1.sqcontratofinanceiro,
    t1.dt_desp

),
QUERY_2 AS (

SELECT
    A.NuContratoGestao,
    A.SqContratoFinanceiro,
    DATE(A.Dt_DESP) AS Dt_DESP,
    DATE_DIFF(A.Dt_DESP, MIN(CASE
        WHEN B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.Dt_DESP) THEN B.dt_vencimento
        ELSE NULL
    END), DAY) AS Atraso,
    SUM(CASE
        WHEN B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.Dt_DESP) THEN B.vr_parcela
        ELSE 0
    END) AS Risco_Aberto,
    COUNT(CASE
        WHEN B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.Dt_DESP) THEN B.vr_parcela
        ELSE NULL
    END) AS Qtd_Parcelas_Abertas,
    SUM(CASE
        WHEN (B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND DATE(B.dt_ultimo_recebimento) >= A.Dt_DESP))
        AND DATE(B.dt_vencimento) < A.Dt_DESP THEN B.vr_parcela
        ELSE 0
    END) AS Risco_Aberto_em_Atraso,
    COUNT(CASE
        WHEN (B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND DATE(B.dt_ultimo_recebimento) >= A.Dt_DESP))
        AND DATE(B.dt_vencimento) < A.Dt_DESP THEN B.vr_parcela
        ELSE NULL
    END) AS Qtd_Parcelas_em_Atraso,
    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.Dt_DESP) THEN B.dt_vencimento
        ELSE NULL
    END) AS DT_VENCIMENTO_NOVO,
    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.Dt_DESP) THEN B.nu_parcela
        ELSE NULL
    END) AS PARCELA,
    C.cod_asses
FROM
    QUERY_1 AS A
LEFT JOIN
    bv-pdbd-prd.dvry_core.tbparcela AS B ON A.NuContratoGestao = B.cd_contrato
    AND A.SqContratoFinanceiro = B.sq_contrato_financeiro
LEFT JOIN
    bv-pdbd-prd.trusted_scc_fin.saj_processo AS C ON A.NuContratoGestao = C.nucontratogestao
    AND A.SqContratoFinanceiro = C.sqcontratofinanceiro
    AND C.cod_acao IN (1, 3, 39, 59, 60)
    AND (DATE(C.dt_asses) <= A.Dt_DESP AND (C.dt_ret_asses IS NULL OR DATE(C.dt_ret_asses) > A.Dt_DESP))
GROUP BY
    A.NuContratoGestao,
    A.SqContratoFinanceiro,
    A.Dt_DESP,
    C.cod_asses

)
SELECT * FROM QUERY_2

)
;QUIT;

PROC SORT DATA=CONSULTA_AJUIZAMENTOb NODUPKEY; BY NuContratoGestao SqContratoFinanceiro PARCELA;RUN;

CREATE TABLE PRODUCAO_TBPARCELA AS SELECT
* from
CONNECTION TO PERM	(
SELECT distinct
a.cd_contrato,
a.dm_contrato,
a.dt_vencimento,
case when a.nm_grupo_subproduto = "LEVES" and a.ds_subproduto = "CP COM GARANTIA LEVES" then "EGV LEVES"
	 when a.nm_grupo_subproduto = "PESADOS" and a.ds_subproduto = "CP COM GARANTIA PESADOS" then "EGV PESADOS"

LEFT JOIN `bv-pdbd-prd.dvry_core.tbparcela` b on a.cd_contrato = b.cd_contrato and b.nu_parcela = 1 and b.sq_contrato_financeiro = 1
LEFT JOIN `bv-pdbd-prd.dvry_core.tbparcela` c on a.cd_contrato = c.cd_contrato and b.nu_parcela = 1 and c.sq_contrato_financeiro = 1
