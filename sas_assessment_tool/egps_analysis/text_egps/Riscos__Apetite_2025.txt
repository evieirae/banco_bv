PROC SQL;
	CREATE TABLE PRAZO_POND_SETOR AS
		SELECT	DsSetorClienteRisco_novo,
				SUM(PRAZO_POND) AS PRAZO_POND,
				SUM(VRSALDO) AS VRSALDO,
				SUM(PRAZO_POND)/SUM(VRSALDO) AS PRAZO_SETOR
	FROM BU_POND
	GROUP BY 1;

	CREATE TABLE PRAZO_POND_MACRO AS
		SELECT	MACRO_SETOR,
				SUM(PRAZO_POND) AS PRAZO_POND,
				SUM(VRSALDO) AS VRSALDO,
				SUM(PRAZO_POND)/SUM(VRSALDO) AS PRAZO_MACRO
	FROM BU_POND
	GROUP BY 1;

QUIT;

PROC SQL;
	CREATE TABLE SALDI_LIQ AS
	SELECT DISTINCT
					"JOIN" AS JOIN,
					SUM('Exposição Líquida'N) AS TOTAL_LIQ FORMAT COMMAX22.2
	FROM BASE_APETITE_2
	;
	CREATE TABLE DISPONIBILIDADE_0_setor AS
		SELECT DISTINCT
						"JOIN" AS JOIN,
						A.'Setor crédito'n,
						MAX(A.LIM_VER_SET) AS LIM_VER_SET format PERCENT8.2,
						MAX(A.LIM_AMR_SET) AS LIM_AMR_SET format PERCENT8.2,
						SUM(A.'Exposição Líquida'N) AS 'Exposição Líquida'N

	FROM BASE_APETITE_2 AS A
	GROUP BY 1,2;

	CREATE TABLE DISPONIBILIDADE_0_macro AS
		SELECT DISTINCT
						"JOIN" AS JOIN,
						A.'Macro Setor'n,
						MAX(A.LIM_VER_MCR) AS LIM_VER_MCR format PERCENT8.2,
						MAX(A.LIM_AMR_MCR) AS LIM_AMR_MCR format PERCENT8.2,
						SUM(A.'Exposição Líquida'N) AS 'Exposição Líquida'N

	FROM BASE_APETITE_2 AS A
	GROUP BY 1,2;

	SELECT DISTINCT

%macro monta_bu(mesref);
    proc sql;
        create table BU_&mesref. as
            select distinct AnoMes_Referencia,
                            DsGrupoEconomicoRisco,
                            TIPO_CARTEIRA,
                            DsGrupoProduto,
                            DsProduto,
                            DsStatus_Monitoramento,
                            DsRatingCliente,
                            sum(VrSaldo)                   as VrSaldo           format=COMMAX15.0,
                            sum(VrProvisao)                as VrProvisao        format=COMMAX15.0,
                            min(DtInicioOperacao)          as ini_operacao      format=MMDDYY10.,
                            max(DtVencimentoParcela)       as fim_operacao      format=MMDDYY10.
            from MISCRED.BASE_UNICA_ATACADO_&mesref.
            where TIPO_CARTEIRA not in ("CORRETORA", "SECURITIES")
              and DsTipoCarteira = "REGULAR"
              and DsGrupoEconomicoRisco not in ("S3 BANK")
            group by 1,2,3,4,5,6,7;
    quit;

    data BU_&mesref.;
        set BU_&mesref.;

        
        if prxmatch('/SACADO|SACADO SME |SACADO SME/', DsGrupoEconomicoRisco) then TIPO_CARTEIRA = "SMES";

        if DsGrupoEconomicoRisco in (
            'DEB SECURITIZADA FARMTECH', 'DEB SECURITIZADA VAMOS',
            'FIAGRO MULTITECNICA', 'REDE SIM') then TIPO_CARTEIRA = 'FIDC';
        if substr(DsGrupoEconomicoRisco,1,4) = 'DEB ' then TIPO_CARTEIRA = 'FIDC';
        if prxmatch('/FIDC|FIDC |CAIENA AGR |CAIENA AGR/', DsGrupoEconomicoRisco) then TIPO_CARTEIRA = "FIDC";
        if DsGrupoEconomicoRisco = "FIDC TRADEMASTER" then TIPO_CARTEIRA = "FIDC";

        if DsGrupoEconomicoRisco = "TRADEMASTER" then TIPO_CARTEIRA = "CORPORATE";
        if DsGrupoEconomicoRisco = "SERVIMED FIDC" then TIPO_CARTEIRA = "FIDC";
        if DsGrupoEconomicoRisco = "SERVIMED" then TIPO_CARTEIRA = "LARGE";

        if (DsProduto = "CESSÃO DE PORTFÓLIO MULTI" or DsProduto = "CCB PEAC") and
           DsGrupoEconomicoRisco in ('AIZ.', 'AGRO GALAXY', 'AMAZON', 'VESTAS', 'VILLEFORT') then TIPO_CARTEIRA = 'CORPORATE';

        if (DsProduto = "CESSÃO DE PORTFÓLIO MULTI" or DsProduto = "CCB PEAC") and
           DsGrupoEconomicoRisco in (
            'MAGAZINE LUIZA', 'CARREFOUR', 'ATACADAO','ATACADAO S.A.','ATACADAO S.A',
            'GERDAU', 'ARCELOR', 'DAIMLER', 'PAO DE ACUCAR', 'DROGASIL', 'AMBEV',
            'VIA VAREJO', 'HAVAN', 'LOCAMERICA', 'LOJAS AMERICANAS', 'LABORATORIO FLEURY') then TIPO_CARTEIRA = 'LARGE';

        
        if year(fim_operacao) < 2100 then fim_operacao = fim_operacao;
        else fim_operacao = .;

        if AnoMes_Referencia = 14823 then AnoMes_Referencia = 202012;

        
        days = sum(fim_operacao, -ini_operacao);
        dif_dias = intck('day', ini_operacao, fim_operacao);

        if AnoMes_Referencia = 202306 and days > 15000 then days = .;
        if AnoMes_Referencia = 202306 and dif_dias > 15000 then days = .;

        
        if DsGrupoEconomicoRisco in (
            "BANCO DO BRASIL", "BANCO BRADESCO", "BRADESCO", "BTG PACTUAL",
            "CAIXA ECONÔMICA FEDERAL", "BANCO ITAU", "ITAU", "BANCO SANTANDER", "SANTANDER") then delete;
    run;

    proc sql;
        create table BU_top_&mesref. as
            select distinct AnoMes_Referencia,
                            DsGrupoEconomicoRisco,
                            sum(VrSaldo)           as VrSaldo         format=COMMAX15.0,
                            mean(VrSaldo)          as VrSaldo_Medio   format=COMMAX15.0,
                            sum(VrProvisao)        as VrProvisao      format=COMMAX15.0,
                            mean(dif_dias)         as PRAZO_MEDIO_DIAS
            from BU_&mesref.
            group by 1,2;
    quit;

    proc sql;
        create table BU_total_&mesref. as
            select distinct AnoMes_Referencia,
                            "TOTAL"                as DsGrupoEconomicoRisco,
                            "TOTAL"                as MARCACAO         format=$10.,
                            sum(VrSaldo)           as VrSaldo          format=COMMAX15.0,
                            mean(VrSaldo)          as VrSaldo_Medio    format=COMMAX15.0,
                            sum(VrProvisao)        as VrProvisao       format=COMMAX15.0,
                            0                      as VrAjustePrudencial format=COMMAX15.0,
                            0                      as VrImpairment     format=COMMAX15.0,
                            mean(dif_dias)         as PRAZO_MEDIO_DIAS
            from BU_&mesref.
            group by 1,2,3;
    quit;

    proc sort data=BU_top_&mesref.; by descending VrSaldo; run;

    %top(1);
    %top(5);
    %top(10);
    %top(20);
	%top(50);
%mend;

PROC SQL;
	CREATE TABLE DEPARA_MACROSETOR AS
		SELECT 	A.*,
				B.LIM_VER_SET_STR,
				B.LIM_AMR_SET_STR,
				B.LIM_VER_SET,
				B.LIM_AMR_SET,
				C.LIM_VER_MCR_STR,
				C.LIM_AMR_MCR_STR,
				C.LIM_VER_MCR,
				C.LIM_AMR_MCR

	FROM	DEPARA_MACROSETOR_0 AS A
	LEFT JOIN LIMITE_SETOR AS B ON(A.SETOR = B.SETOR)
	LEFT JOIN LIMITE_MACROSETOR AS C ON(A.MACRO_SETOR = C.MACRO_SETOR);
