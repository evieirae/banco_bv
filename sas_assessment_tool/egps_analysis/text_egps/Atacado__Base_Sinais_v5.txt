PROC SQL;

	NuCnpjRaiz

PROC SQL;
CREATE TABLE BASE_RESUMO_OCORRENCIA AS
SELECT DISTINCT
	A.NuCnpjRaiz
	,A.DtConsulta
	,A.QdPefin
	,A.VrPefin
	,A.QdProtesto
	,A.VrProtesto
	,A.QdAcoes
	,A.VrAcoes
	,A.QdDivVencida AS QdDividaVencida
	,A.VrDivVencida AS VrDividaVencida
	,A.QdRefin
	,A.VrRefin
	,A.QdFalencia
	,A.QdRR AS QdRJR
	,A.QdRC AS QdRJC
	,A.QdRER AS QdREJR
	,A.QdREH AS QdREJH
	,A.QdRJ
	,B.DtOcorrencia AS DtPefin
	,C.DtOcorrencia AS DtProtesto
	,D.DtOcorrencia AS DtAcao
	,E.DtOcorrencia AS DtDividaVencida
	,F.DtOcorrencia AS DtRefin

FROM GCSERASA.TBGCRESUMOOCORRENCIAS AS A
LEFT JOIN BASE_DATA_OCORRENCIA AS B ON A.NuCnpjRaiz = B.NuCnpjRaiz AND A.DtConsulta = B.DtConsulta AND B.CdOcorrencia = 1
LEFT JOIN BASE_DATA_OCORRENCIA AS C ON A.NuCnpjRaiz = C.NuCnpjRaiz AND A.DtConsulta = C.DtConsulta AND C.CdOcorrencia = 3
LEFT JOIN BASE_DATA_OCORRENCIA AS D ON A.NuCnpjRaiz = D.NuCnpjRaiz AND A.DtConsulta = D.DtConsulta AND D.CdOcorrencia = 4
LEFT JOIN BASE_DATA_OCORRENCIA AS E ON A.NuCnpjRaiz = E.NuCnpjRaiz AND A.DtConsulta = E.DtConsulta AND E.CdOcorrencia = 7
LEFT JOIN BASE_DATA_OCORRENCIA AS F ON A.NuCnpjRaiz = F.NuCnpjRaiz AND A.DtConsulta = F.DtConsulta AND F.CdOcorrencia = 10
WHERE (YEAR(A.DtConsulta)*10000 + MONTH(A.DtConsulta)*100 + DAY(A.DtConsulta)) > &DT_MAX_SERASA.
;RUN;

PROC SQL; CREATE TABLE TbResponsavel AS
SELECT
 CdGpEconomico
,NmGpEconomico
,FlIFP			   AS FlIF
,CdPessoa
,NmPessoa
,NmGerenteCredito  AS NmGerente
,NmAnalistaCredito AS NmAnalista
,NmOfficer
,NmDivisaoOfficer
,CASE
 WHEN FIND (NmDivisaoOfficer,"ATIVOS ESPECIAIS","I") > 0 THEN 1 ELSE 0 END AS FlEspecial

,CASE WHEN FlIF = 1								  THEN 1
	  WHEN FIND (NmDivisaoOfficer,"FI.","I") > 0  THEN 1 ELSE 0 END AS FlFI

,CASE WHEN FIND (NmDivisaoOfficer ,"GPR","I") > 0 THEN 1
	  WHEN FIND (NmAnalistaCredito,"GPR","I") > 0 THEN 1 ELSE 0 END AS FlGPR

,CASE WHEN NmOfficer IN ("","SEM OFFICER")		  THEN 1 ELSE 0 END AS ROfficer

  AND CdGpEconomico IS NOT NULL

PROC SQL ;
    CREATE TABLE TbVisaoGarantias AS SELECT
    'Garantias'                                     AS TpVisao      LENGTH=50
    , DATA_ARQUIVO                                  AS DtConsulta
    , TIPO_GARANTIA                                 AS TpRegistro   LENGTH=100
    , CD_PESSOA 									AS CdPessoa
    , ''                                            AS Sigla1       LENGTH=100
    , NATUREZA                                      AS DsGrupo1     LENGTH=100
	, GARANTIA_DETALHE_OPERACIONAL 					AS DsObs
    , INPUT(DIAS_PENDENTES, 8.)           			AS QdDiasAtraso

    , COUNT(DISTINCT CONTRATO)                      AS QdContratos
    , SUM(VALOR_EXIGIDO)                            AS VrExigido
    , SUM(VALOR_REALIZADO)                          AS VrRealizado
    , SUM(VALOR_REALIZADO)/SUM(VALOR_EXIGIDO)       AS VrPcExigido		FORMAT percent10.2
    , ROUND((1-(SUM(VALOR_REALIZADO)/SUM(VALOR_EXIGIDO)))*100)   AS VrPcInsuficiente
	, SUM(VALOR_EXIGIDO)- SUM(VALOR_REALIZADO)		AS VrInsuficiencia

    FROM INSIGHT.TB_GARANTIAS_ANALITICO
    WHERE FLAGNOTIFICA__O = 'Sim'
    AND GARANTIA_DETALHE_OPERACIONAL NOT IN ('Contrato de Repasse BNDES', 'Garantia Com Consituição Futura - Dentro do Prazo', 'Cesta de Garantias', 'Contrato em Ativos Especiais', 'Garantia Conforto', 'Insuficiência TVM')

%MACRO P_Modelo ();





	 IF FlIFGP = 1				THEN NuPonto = 0;


ELSE IF UPCASE(TpVisao) = "SERASA"			THEN DO;

	 	  IF TpRegistro = "Falencia"	AND TTQdRegistroGp > 0								THEN NuPonto = 1000;
	 ELSE IF TpRegistro = "RJ"			AND TTQdRegistroGp > 0								THEN NuPonto = 1000;

	 ELSE IF TpRegistro = "Refin"		AND TTVrRegistroGp >= VrFaturamentoBrutoGp * 0.005
										AND TTVrRegistroGp <= VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 8;
	 ELSE IF TpRegistro = "Refin"		AND TTVrRegistroGp >  VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 9;

	 ELSE IF TpRegistro = "Comercial" 	AND TTVrRegistroGp >= VrFaturamentoBrutoGp * 0.005
										AND TTVrRegistroGp <= VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 2;
	 ELSE IF TpRegistro = "Comercial" 	AND TTVrRegistroGp >  VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 3;

	 ELSE IF TpRegistro = "Acoes"		AND TTVrRegistroGp >= VrFaturamentoBrutoGp * 0.005
	 									AND TTVrRegistroGp <= VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 2;
	 ELSE IF TpRegistro = "Acoes"		AND TTVrRegistroGp >  VrFaturamentoBrutoGp * 0.01	THEN NuPonto = 3;

	 ELSE IF TpRegistro = "Acoes" 		AND TTQdRegistroGp >= 20 AND TTQdRegistroGp <= 40	THEN NuPonto = 2;
	 ELSE IF TpRegistro = "Acoes" 		AND TTQdRegistroGp >  40							THEN NuPonto = 3;
	 																						ELSE NuPonto = 0;
END;


ELSE IF UPCASE(TpVisao) = 'VNP'				THEN DO;
		
  	     IF FlComissaoFianca = 1 AND QdDiasAtraso >= 30 AND VrPercentual > 0.10				THEN NuPonto = 7;
	ELSE IF FlComissaoFianca = 1															THEN NuPonto = 0;

		
	ELSE IF QdDiasAtraso >=   7  AND QdDiasAtraso < 15	AND VrPercentual > 0.10				THEN NuPonto = 6;
	ELSE IF QdDiasAtraso >=  15  AND QdDiasAtraso < 30	AND VrPercentual > 0.10				THEN NuPonto = 8;
	ELSE IF QdDiasAtraso >=  30  AND QdDiasAtraso < 60	AND VrPercentual > 0.05				THEN NuPonto = 9;
	ELSE IF QdDiasAtraso >=  60  AND QdDiasAtraso < 90										THEN NuPonto = 10;
	ELSE IF QdDiasAtraso >=  90  AND QdDiasAtraso < 120										THEN NuPonto = 15;
	ELSE IF QdDiasAtraso >= 120  AND QdDiasAtraso < 180										THEN NuPonto = 20;
	ELSE IF QdDiasAtraso >=	180																THEN NuPonto = 100;
																							ELSE NuPonto = 0;
END;


ELSE IF UPCASE(TpVisao) = "ILM90D"		THEN DO;
  	     IF VrPercentual <= 0.80	AND VrPercentual > 0.70 								THEN NuPonto = 3;
	ELSE IF VrPercentual <= 0.70	AND VrPercentual > 0.50 								THEN NuPonto = 4;
	ELSE IF VrPercentual <= 0.50	AND VrPercentual > 0.00 								THEN NuPonto = 5;

	 IF DtBaseBruta  = DtULTAlerta										THEN FlDia = 1;

%MACRO R_Alerta ();

IF VrMin IN ( . , 0) OR VrAtual < VrMin	THEN RVrMin = 0; ELSE RVrMin = 1;
IF QdMin IN ( . , 0) OR QdAtual < QdMin	THEN RQdMin = 0; ELSE RQdMin = 1;
RMin = MAX(RVrMin , RQdMin);

VPcVr = COALESCE((VrAtual / VrAntigo),0);
VQd	  = COALESCE((QdAtual - QdAntigo),0);

IF PcVrAumento IN (. , 0) OR VPcVr = 0 		THEN APc = 0; ELSE APc = 1;
IF PcVrReducao IN (. , 0) OR VPcVr = 0		THEN RPc = 0; ELSE RPc = 1;

IF QdAumento   IN (. , 0) OR VQd   = 0		THEN AQd = 0; ELSE AQd = 1;
IF QdReducao   IN (. , 0) OR VQd   = 0		THEN RQd = 0; ELSE RQd = 1;

	 IF DsClassificacao = "CMR7"								THEN RA =  0;

ELSE IF  FlAntigo = 1
	AND (VrAntigo + QdAntigo) > 0
	AND (VrAtual  + QdAtual ) = 0								THEN RA = -1;
ELSE IF FlAntigo = 1 AND RPc = 1 AND VPcVr <= (1 - PcVrReducao)	THEN RA = -2;
ELSE IF FlAntigo = 1 AND RQd = 1 AND VQd   <= -1 * QdReducao	THEN RA = -3;


ELSE IF (VrAntigo + QdAntigo) = 0 AND RMin   = 1					THEN RA =  1;
ELSE IF  FlAntigo = 1 AND APc = 1 AND VPcVr >= (1 + PcVrAumento)	THEN RA =  2;
ELSE IF  FlAntigo = 1 AND AQd = 1 AND VQd   >= QdAumento			THEN RA =  3;
																	ELSE RA =  0;

	 IF RA < 0 THEN FlA = -1;
ELSE IF RA > 0 THEN FlA =  1;
			   ELSE FlA =  0;
%MEND R_Alerta;

PROC SQL;
CREATE TABLE TBBRUTA_RATING AS
	SELECT
		t1.*,

		CASE
			WHEN t1.DsRatingGp_CIB = 'AAA'    THEN 1
			WHEN t1.DsRatingGp_CIB = 'AA+'    THEN 2
			WHEN t1.DsRatingGp_CIB = 'AA'     THEN 3
			WHEN t1.DsRatingGp_CIB = 'AA-'    THEN 4
			WHEN t1.DsRatingGp_CIB = 'A+'     THEN 5
			WHEN t1.DsRatingGp_CIB = 'A'      THEN 6
			WHEN t1.DsRatingGp_CIB = 'A-'     THEN 7
			WHEN t1.DsRatingGp_CIB = 'BBB+'   THEN 8
			WHEN t1.DsRatingGp_CIB = 'BBB'    THEN 9
			WHEN t1.DsRatingGp_CIB = 'BBB-'   THEN 10
			WHEN t1.DsRatingGp_CIB = 'BB+'    THEN 11
			WHEN t1.DsRatingGp_CIB = 'BB'     THEN 12
			WHEN t1.DsRatingGp_CIB = 'BB-'    THEN 13
			WHEN t1.DsRatingGp_CIB = 'B+'     THEN 14
			WHEN t1.DsRatingGp_CIB = 'B'      THEN 15
			WHEN t1.DsRatingGp_CIB = 'B-'     THEN 16
			WHEN t1.DsRatingGp_CIB = 'CCC+'   THEN 17
			WHEN t1.DsRatingGp_CIB = 'CCC'    THEN 18
			WHEN t1.DsRatingGp_CIB = 'CCC-'   THEN 19
			WHEN t1.DsRatingGp_CIB = 'CC+'    THEN 20
			WHEN t1.DsRatingGp_CIB = 'CC'     THEN 21
			WHEN t1.DsRatingGp_CIB = 'CC-'    THEN 22
			WHEN t1.DsRatingGp_CIB = 'C+'     THEN 23
			WHEN t1.DsRatingGp_CIB = 'C'      THEN 24
			WHEN t1.DsRatingGp_CIB = 'C-'     THEN 25
			WHEN t1.DsRatingGp_CIB = 'D'      THEN 26
			WHEN t1.DsRatingGp_CIB IN ('D1', 'Dc1') THEN 27
			WHEN t1.DsRatingGp_CIB IN ('D2', 'Dc2') THEN 28
			WHEN t1.DsRatingGp_CIB IN ('D3', 'Dc3') THEN 29
			WHEN t1.DsRatingGp_CIB = 'D4'     THEN 30
			WHEN t1.DsRatingGp_CIB IN ('E+', 'E', 'E-', 'F', 'G', 'H') THEN 31
			WHEN t1.DsRatingGp_CIB = 'DF'     THEN 32
			WHEN t1.DsRatingGp_CIB IN ('WL', 'WL1') THEN 33
			ELSE 0
		END
	AS NuRatingGP_CIB

	FROM TbBruta AS t1;
QUIT;
