%MACRO CICLO_SALDO_GCON;


%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA., %str(*)));


%do i = 1 %to &NUM_ELEMENTOS.;

    
    %let ref_tab = %scan(&LISTA., &i., *);

    
    DATA _NULL_;
        MES = INPUT(scan("&LISTA.", &i., "*"), yymmdd10.);
        MES_POS = PUT(INTNX("MONTH", MES, 1, "e"), yymmddn8.);
        
        CALL SYMPUT("REF", PUT(MES, yymmdd10.));
		CALL SYMPUT("REF2", PUT(MES, yymmddn8.));
        CALL SYMPUT("REF_POS", PUT(INTNX("MONTH", MES, 1, "e"), yymmdd10.));
        CALL SYMPUT("REF_TAB_POS", MES_POS);
    RUN;

    %put &REF;
    %put &REF_POS;
    %put &REF_TAB_POS;

    
    PROC SQL;
        %CONNECT_PERM;
        CREATE TABLE GCON_&ref_tab. AS
        SELECT
            A.*,
            "&REF2." AS REF
        FROM CONNECTION TO PERM (
            WITH QUERY_1 AS (
                SELECT
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual,
                    MAX(a.nudiasatraso) AS QTATRASO,
                    SUM(CAST(a.vrpermanenciaapropriado AS FLOAT64) + CAST(a.vrtaxacontrato AS FLOAT64)) AS SALDO,
                    SUM(CAST(a.vrpdd AS FLOAT64)) AS PDD_CONTABIL
                FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS a
                LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS c
                    ON a.nucontrato = c.cd_contrato
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS d
                    ON a.nucontrato = d.nucontratogestao AND d.stproposta = 'AP'
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS G
                    ON d.cdsubproduto = G.cdsubproduto
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbgruposubproduto AS H
                    ON G.cdgruposubproduto = H.cdgruposubproduto
                
                WHERE DATE(a.dtprocessamento) = PARSE_DATE('%Y-%m-%d', "&REF.")
                  AND a.dtprocessamentoliquidacao IS NULL
                GROUP BY
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual
            )
            SELECT * FROM QUERY_1
        ) A;
    QUIT;

%end;

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
    A.*,
    D.SALDO AS SALDO_M0,
    D.PDD_CONTABIL AS PDD_M0,
    D.CDRATINGATUAL AS RATING_M0,

    B.SALDO AS SALDO_M1,
    B.PDD_CONTABIL AS PDD_M1,
    B.CDRATINGATUAL AS RATING_M1,

    C.SALDO AS SALDO_M2,
    C.PDD_CONTABIL AS PDD_M2,
    C.CDRATINGATUAL AS RATING_M2,

    F.SALDO AS SALDO_M3,
    F.PDD_CONTABIL AS PDD_M3,
    F.CDRATINGATUAL AS RATING_M3,

    E.SALDO AS SALDO_M4,
    E.PDD_CONTABIL AS PDD_M4,
    E.CDRATINGATUAL AS RATING_M4,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END AS QT_DIAS_REFENVIO,

    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END AS QT_DIAS_REFATRASO,

    
    CASE
        WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 1
        WHEN A.DTULTIMORECEBIMENTO <= A.DATA_VIGENCIA OR A.DTULTIMORECEBIMENTO <= A.DTVENCIMENTO THEN 0
        ELSE 1
    END AS FL_CONSIDERAR,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 10 THEN '01. Até 10 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 20 THEN '02. 11-20 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 60 THEN '04. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 30 THEN '01. Até 30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 60 THEN '02. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 90 THEN '03. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 120 THEN '04. 91-120 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 150 THEN '05. 121-150 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 180 THEN '06. 151-180 dias'
        ELSE '07. 180+ dias'
    END AS FAIXA_REFATRASO,

	PUT(A.DATA_VIGENCIA, yymmn6.) AS SAFRA

FROM BASE_UNICA_FIM A
LEFT JOIN SISTEMA_TOTAL D ON PUT(A.NUCONTRATO, 14.) = D.NUCONTRATO AND A.DATA_M0 = D.REF
LEFT JOIN SISTEMA_TOTAL B ON PUT(A.NUCONTRATO, 14.) = B.NUCONTRATO AND A.DATA_M1 = B.REF
LEFT JOIN SISTEMA_TOTAL C ON PUT(A.NUCONTRATO, 14.) = C.NUCONTRATO AND A.DATA_M2 = C.REF
LEFT JOIN SISTEMA_TOTAL F ON PUT(A.NUCONTRATO, 14.) = F.NUCONTRATO AND A.DATA_M3 = F.REF
LEFT JOIN SISTEMA_TOTAL E ON PUT(A.NUCONTRATO, 14.) = E.NUCONTRATO AND A.DATA_M4 = E.REF;

PROC SQL;

,D.SALDO AS SALDO_M0
,CASE WHEN D.CDRATINGATUAL ='HH' THEN D.SALDO ELSE D.PDD_CONTABIL END AS PDD_M0
,D.CDRATINGATUAL AS RATING_M0

,B.SALDO AS SALDO_M1
,CASE WHEN B.CDRATINGATUAL ='HH' THEN B.SALDO ELSE B.PDD_CONTABIL END AS PDD_M1
,B.CDRATINGATUAL AS RATING_M1

,C.SALDO AS SALDO_M2
,C.PDD_CONTABIL AS PDD_CONTABIL_M2
,CASE WHEN C.CDRATINGATUAL ='HH' THEN C.SALDO ELSE C.PDD_CONTABIL END AS PDD_M2
,C.CDRATINGATUAL AS RATING_M2

,F.SALDO AS SALDO_M3
,F.PDD_CONTABIL AS PDD_CONTABIL_M3
,CASE WHEN F.CDRATINGATUAL ='HH' THEN F.SALDO ELSE F.PDD_CONTABIL END AS PDD_M3
,F.CDRATINGATUAL AS RATING_M3

,E.SALDO AS SALDO_M4
,E.PDD_CONTABIL AS PDD_CONTABIL_M4
,CASE WHEN E.CDRATINGATUAL ='HH' THEN E.SALDO ELSE E.PDD_CONTABIL END AS PDD_M4
,E.CDRATINGATUAL AS RATING_M4

,CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO
,CASE WHEN DATEPART(A.DTULTIMORECEBIMENTO) IS MISSING THEN -100 ELSE INTCK('DAYS', A.DATA_VIGENCIA, DATEPART(A.DTULTIMORECEBIMENTO)) END AS QT_DIAS
,CASE WHEN DATEPART(A.DTULTIMORECEBIMENTO) IS MISSING THEN 1
	  WHEN DATEPART(A.DTULTIMORECEBIMENTO) <= A.DATA_VIGENCIA THEN 0
	  ELSE 1 END AS FL_CONSIDERAR

FROM BASE_UNICA_FIM A
LEFT JOIN SISTEMA_TOTAL D ON
A.NUCONTRATO = D.NUCONTRATO
AND A.DATA_M0 = D.REF

LEFT JOIN SISTEMA_TOTAL B ON
A.NUCONTRATO = B.NUCONTRATO
AND A.DATA_M1 = B.REF

LEFT JOIN SISTEMA_TOTAL C ON
A.NUCONTRATO = C.NUCONTRATO
AND A.DATA_M2 = C.REF

LEFT JOIN SISTEMA_TOTAL F ON
A.NUCONTRATO = F.NUCONTRATO
AND A.DATA_M3 = F.REF

LEFT JOIN SISTEMA_TOTAL E ON
A.NUCONTRATO = E.NUCONTRATO
AND A.DATA_M4 = E.REF

%MACRO CICLO_SALDO_GCON;


%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA., %str(*)));


%do i = 1 %to &NUM_ELEMENTOS.;

    
    %let ref_tab = %scan(&LISTA., &i., *);

    
    DATA _NULL_;
        MES = INPUT(scan("&LISTA.", &i., "*"), yymmdd10.);
        MES_POS = PUT(INTNX("MONTH", MES, 1, "e"), yymmddn8.);
        
        CALL SYMPUT("REF", PUT(MES, yymmdd10.));
		CALL SYMPUT("REF2", PUT(MES, yymmddn8.));
        CALL SYMPUT("REF_POS", PUT(INTNX("MONTH", MES, 1, "e"), yymmdd10.));
        CALL SYMPUT("REF_TAB_POS", MES_POS);
    RUN;

    %put &REF;
    %put &REF_POS;
    %put &REF_TAB_POS;

    
    PROC SQL;
        %CONNECT_PERM;
        CREATE TABLE GCON_&ref_tab. AS
        SELECT
            A.*,
            "&REF2." AS REF
        FROM CONNECTION TO PERM (
            WITH QUERY_1 AS (
                SELECT
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual,
                    MAX(a.nudiasatraso) AS QTATRASO,
                    SUM(CAST(a.vrpermanenciaapropriado AS FLOAT64) + CAST(a.vrtaxacontrato AS FLOAT64)) AS SALDO,
                    SUM(CAST(a.vrpdd AS FLOAT64)) AS PDD_CONTABIL
                FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS a
                LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS c
                    ON a.nucontrato = c.cd_contrato
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS d
                    ON a.nucontrato = d.nucontratogestao AND d.stproposta = 'AP'
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS G
                    ON d.cdsubproduto = G.cdsubproduto
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbgruposubproduto AS H
                    ON G.cdgruposubproduto = H.cdgruposubproduto
                
                WHERE DATE(a.dtprocessamento) = PARSE_DATE('%Y-%m-%d', "&REF.")
                  AND a.dtprocessamentoliquidacao IS NULL
                GROUP BY
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual
            )
            SELECT * FROM QUERY_1
        ) A;
    QUIT;

%end;

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
    A.*,
    D.SALDO AS SALDO_M0,
    D.PDD_CONTABIL AS PDD_M0,
    D.CDRATINGATUAL AS RATING_M0,

    B.SALDO AS SALDO_M1,
    B.PDD_CONTABIL AS PDD_M1,
    B.CDRATINGATUAL AS RATING_M1,

    C.SALDO AS SALDO_M2,
    C.PDD_CONTABIL AS PDD_M2,
    C.CDRATINGATUAL AS RATING_M2,

    F.SALDO AS SALDO_M3,
    F.PDD_CONTABIL AS PDD_M3,
    F.CDRATINGATUAL AS RATING_M3,

    E.SALDO AS SALDO_M4,
    E.PDD_CONTABIL AS PDD_M4,
    E.CDRATINGATUAL AS RATING_M4,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END AS QT_DIAS_REFENVIO,

    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END AS QT_DIAS_REFATRASO,

    
    CASE
        WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 1
        WHEN A.DTULTIMORECEBIMENTO <= A.DATA_VIGENCIA OR A.DTULTIMORECEBIMENTO <= A.DTVENCIMENTO THEN 0
        ELSE 1
    END AS FL_CONSIDERAR,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 10 THEN '01. Até 10 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 20 THEN '02. 11-20 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 60 THEN '04. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 30 THEN '01. Até 30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 60 THEN '02. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 90 THEN '03. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 120 THEN '04. 91-120 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 150 THEN '05. 121-150 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 180 THEN '06. 151-180 dias'
        ELSE '07. 180+ dias'
    END AS FAIXA_REFATRASO,

	PUT(A.DATA_VIGENCIA, yymmn6.) AS SAFRA

FROM BASE_UNICA_FIM A
LEFT JOIN SISTEMA_TOTAL D ON PUT(A.NUCONTRATO, 14.) = D.NUCONTRATO AND A.DATA_M0 = D.REF
LEFT JOIN SISTEMA_TOTAL B ON PUT(A.NUCONTRATO, 14.) = B.NUCONTRATO AND A.DATA_M1 = B.REF
LEFT JOIN SISTEMA_TOTAL C ON PUT(A.NUCONTRATO, 14.) = C.NUCONTRATO AND A.DATA_M2 = C.REF
LEFT JOIN SISTEMA_TOTAL F ON PUT(A.NUCONTRATO, 14.) = F.NUCONTRATO AND A.DATA_M3 = F.REF
LEFT JOIN SISTEMA_TOTAL E ON PUT(A.NUCONTRATO, 14.) = E.NUCONTRATO AND A.DATA_M4 = E.REF;
QUIT;

    E.SALDO AS SALDO_M4,
    E.PDD_CONTABIL AS PDD_M4,
    E.CDRATINGATUAL AS RATING_M4,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END AS QT_DIAS_REFENVIO,

    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END AS QT_DIAS_REFATRASO,

    
    CASE
        WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 1
        WHEN A.DTULTIMORECEBIMENTO <= A.DATA_VIGENCIA OR A.DTULTIMORECEBIMENTO <= A.DTVENCIMENTO THEN 0
        ELSE 1
    END AS FL_CONSIDERAR,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 10 THEN '01. Até 10 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 20 THEN '02. 11-20 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'

        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 30 THEN '01. Até 30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 60 THEN '02. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 90 THEN '03. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 120 THEN '04. 91-120 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 150 THEN '05. 121-150 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 180 THEN '06. 151-180 dias'
        ELSE '07. 180+ dias'
    END AS FAIXA_REFATRASO,

	PUT(A.DATA_VIGENCIA, yymmn6.) AS SAFRA

%MACRO CICLO_SALDO_GCON;


%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA., %str(*)));


%do i = 1 %to &NUM_ELEMENTOS.;

    
    %let ref_tab = %scan(&LISTA., &i., *);

    
    DATA _NULL_;
        MES = INPUT(scan("&LISTA.", &i., "*"), yymmdd10.);
        MES_POS = PUT(INTNX("MONTH", MES, 1, "e"), yymmddn8.);
        
        CALL SYMPUT("REF", PUT(MES, yymmdd10.));
		CALL SYMPUT("REF2", PUT(MES, yymmddn8.));
        CALL SYMPUT("REF_POS", PUT(INTNX("MONTH", MES, 1, "e"), yymmdd10.));
        CALL SYMPUT("REF_TAB_POS", MES_POS);
    RUN;

    %put &REF;
    %put &REF_POS;
    %put &REF_TAB_POS;

    
    PROC SQL;
        %CONNECT_PERM;
        CREATE TABLE GCON_&ref_tab. AS
        SELECT
            A.*,
            "&REF2." AS REF
        FROM CONNECTION TO PERM (
            WITH QUERY_1 AS (
                SELECT
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual,
                    MAX(a.nudiasatraso) AS QTATRASO,
                    SUM(CAST(a.vrpermanenciaapropriado AS FLOAT64) + CAST(a.vrtaxacontrato AS FLOAT64)) AS SALDO,
                    SUM(CAST(a.vrpdd AS FLOAT64)) AS PDD_CONTABIL
                FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS a
                LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS c
                    ON a.nucontrato = c.cd_contrato
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS d
                    ON a.nucontrato = d.nucontratogestao AND d.stproposta = 'AP'
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS G
                    ON d.cdsubproduto = G.cdsubproduto
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbgruposubproduto AS H
                    ON G.cdgruposubproduto = H.cdgruposubproduto
                
                WHERE DATE(a.dtprocessamento) = PARSE_DATE('%Y-%m-%d', "&REF.")
                  AND a.dtprocessamentoliquidacao IS NULL
                GROUP BY
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual
            )
            SELECT * FROM QUERY_1
        ) A;
    QUIT;

%end;

         ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END AS QT_DIAS_REFATRASO,

        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 30 THEN '01. Até 30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 60 THEN '02. 31-60 dias'

        ELSE '07. 180+ dias'
    END AS FAIXA_REFATRASO,

    B.PDD_CONTABIL AS PDD_M1,
    B.CDRATINGATUAL AS RATING_M1,

    C.SALDO AS SALDO_M2,
    C.PDD_CONTABIL AS PDD_M2,
    C.CDRATINGATUAL AS RATING_M2,

    F.SALDO AS SALDO_M3,
    F.PDD_CONTABIL AS PDD_M3,
    F.CDRATINGATUAL AS RATING_M3,

    E.SALDO AS SALDO_M4,
    E.PDD_CONTABIL AS PDD_M4,
    E.CDRATINGATUAL AS RATING_M4,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO,

    
    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END AS QT_DIAS_REFENVIO,

    CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100
         ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END AS QT_DIAS_REFATRASO,

    
    CASE
        WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN 1
        WHEN A.DTULTIMORECEBIMENTO <= A.DATA_VIGENCIA OR A.DTULTIMORECEBIMENTO <= A.DTVENCIMENTO THEN 0
        ELSE 1
    END AS FL_CONSIDERAR,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 10 THEN '01. Até 10 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 20 THEN '02. 11-20 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 60 THEN '04. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.DTULTIMORECEBIMENTO IS MISSING THEN -100 ELSE (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) END) = -100 THEN '00. Sem pagamento'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 30 THEN '01. Até 30 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 60 THEN '02. 31-60 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 90 THEN '03. 61-90 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 120 THEN '04. 91-120 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 150 THEN '05. 121-150 dias'
        WHEN (A.DTULTIMORECEBIMENTO - A.DTVENCIMENTO) <= 180 THEN '06. 151-180 dias'
        ELSE '07. 180+ dias'
    END AS FAIXA_REFATRASO,

	PUT(A.DATA_VIGENCIA, yymmn6.) AS SAFRA

FROM BASE_UNICA_FIM A
LEFT JOIN SISTEMA_TOTAL D ON A.NUCONTRATO = D.NUCONTRATO AND A.DATA_M0 = D.REF
LEFT JOIN SISTEMA_TOTAL B ON A.NUCONTRATO = B.NUCONTRATO AND A.DATA_M1 = B.REF
LEFT JOIN SISTEMA_TOTAL C ON A.NUCONTRATO = C.NUCONTRATO AND A.DATA_M2 = C.REF
