%macro indicadores_2025;
    %do mes=7 %to 12;
        proc sql;
            connect to bigquery as perm (
                project='bv-sadi-dtm'
                credfile='/projetos/acessos/bq/di/sadi-dtm.json'
                schema='bv-pdbd-prd'
                READ_MODE=storage
            );

            create table BASE_ACIO_v2_2025_%sysfunc(putn(&mes,z2.)) as
            select * from connection to perm (
                WITH params AS (
                  SELECT DATE(2025, &mes, 1) AS first_day,
                         DATE_ADD(DATE(2025, &mes, 1), INTERVAL 1 MONTH) AS next_month
                ),
                dias_mes AS (
                  SELECT DATE_ADD(p.first_day, INTERVAL i DAY) AS d
                  FROM params p,
                  UNNEST(GENERATE_ARRAY(0, DATE_DIFF(p.next_month, p.first_day, DAY) - 1)) AS i
                ),
                uteis AS (
                  SELECT d
                  FROM dias_mes
                  WHERE EXTRACT(DAYOFWEEK FROM d) NOT IN (1,7)
                  ORDER BY d
                ),
                limite AS (
                  SELECT d AS dt_limite
                  FROM uteis
                  LIMIT 1 OFFSET 10
                )
                SELECT
                    CAST(dt_acionamento AS DATE) AS DTACIONAMENTO,
                    CD_CONTRATO AS NUCONTRATOGESTAO,
                    MAX(CAST(tp_resultado_acionamento AS INT64)) AS TPRESULTADO,
                    CAST(TP_OCORRENCIA_CHAMADA AS INT64) AS IDCORRENCIA,
                    CD_ASSESSORIA AS CODIGO_ASSES
                FROM `bv-pdbd-prd.dvry_corp_cobranca.assessoria_acionamentos_cobranca_fato`
                WHERE EXTRACT(YEAR FROM dt_acionamento) = 2025
                  AND EXTRACT(MONTH FROM dt_acionamento) = &mes
                  AND CAST(dt_acionamento AS DATE) <= (SELECT dt_limite FROM limite)
                  AND SUBSTR(CAST(CD_CONTRATO AS STRING), 1, 2) IN ('12', '14')
                  AND tp_resultado_acionamento IN ('1','2','3','4','5','6')
                GROUP BY
                    CAST(dt_acionamento AS DATE),
                    CD_CONTRATO,
                    CD_ASSESSORIA,
                    TP_OCORRENCIA_CHAMADA
            );
            disconnect from perm;
        quit;
    %end;
%m%mend;
