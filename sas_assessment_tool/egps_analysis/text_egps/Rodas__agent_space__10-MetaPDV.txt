%macro executar_includesMetaPDV();

    %let dia_semana = %sysfunc(weekday(%sysfunc(today())));
    %let dia_mes = %sysfunc(day(%sysfunc(today())));

    %put dia_semana: &dia_semana;
    %put dia_mes: &dia_mes;

    %if &dia_semana = 2 %then %do; /* Segunda-feira */
        %if &dia_mes < 10 %then %do;

		proc sql;
		create table tbMeta as
		select nuanomes,DsLoginN1,sum(VrProducao) as TBMetaGR from basegrc.TBMETANAOISONOMICA
		where nuanomes=&SafraM1.
		and Produto ='Leves'
		group by 1,2
		;run;

		proc sql;
		create table tbMetaPDV as
		select
		tbisonomica.nuanomes
		,tbisonomica.DsLoginN1
		,tbisonomica.CdParceiroComercial
		,put(tbisonomica.CdParceiroComercial,10.) ||"-"|| tranwrd(NmParceiroComercial,'-','') as CdparceirocomercialNM
		,(VrProducao/TBMetaGR)*100 as TbMetaPDV
		,sum(VrProducao) as VrMeta
		from basegrc.TBMETANAOISONOMICA tbisonomica
		left join tbMeta tbmeta on tbmeta.DsLoginN1 = tbisonomica.DsLoginN1
		left join bases.tbparceirocomercial tbpar on tbpar.cdparceirocomercial = tbisonomica.cdparceirocomercial
		and tbmeta.nuanomes = tbisonomica.nuanomes
		where tbisonomica.nuanomes=&SafraM1.
		and tbisonomica.Produto ='Leves'
		/*and tbisonomica.DsLoginN1='alessandra.domingues'*/
		and tbisonomica.DsLoginN1<>''
		group by 1,2,3,4
		;run;

        %end;
        %else %do;

		proc sql;
		create table tbMeta as
		select nuanomes,DsLoginN1,sum(VrProducao) as TBMetaGR from basegrc.TBMETANAOISONOMICA
		where nuanomes=&SafraM0.
		and Produto ='Leves'
		group by 1,2
		;run;

		proc sql;
		create table tbMetaPDV as
		select
		tbisonomica.nuanomes
		,tbisonomica.DsLoginN1
		,tbisonomica.CdParceiroComercial
		,put(tbisonomica.CdParceiroComercial,10.) ||"-"|| tranwrd(NmParceiroComercial,'-','') as CdparceirocomercialNM
		,(VrProducao/TBMetaGR)*100 as TbMetaPDV
		,sum(VrProducao) as VrMeta
		from basegrc.TBMETANAOISONOMICA tbisonomica
		left join tbMeta tbmeta on tbmeta.DsLoginN1 = tbisonomica.DsLoginN1
		left join bases.tbparceirocomercial tbpar on tbpar.cdparceirocomercial = tbisonomica.cdparceirocomercial
		and tbmeta.nuanomes = tbisonomica.nuanomes
		where tbisonomica.nuanomes=&SafraM0.
		and tbisonomica.Produto ='Leves'
		/*and tbisonomica.DsLoginN1='alessandra.domingues'*/
		and tbisonomica.DsLoginN1<>''
		group by 1,2,3,4
		;run;

        %end;
    %end;
    %else %do;
        %put Hoje não é segunda-feira. Nenhum include será executado.;
    %end;

%mend executar_includesMetaPDV;

%executar_includesMetaPDV;
