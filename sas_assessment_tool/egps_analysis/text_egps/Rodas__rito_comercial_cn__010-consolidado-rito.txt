/*Calcula dia uteis*/

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbDiasUteis AS
SELECT * FROM connection to dbcon
(
  WITH TbTempoData AS (
    SELECT
      NuAnoMes,
      DtTempo,
      CASE WHEN FlDiaUtil = 'S' AND FlFeriadoNacional = 'N' THEN 1 ELSE 0 END AS FlDiaUtil,
      CASE WHEN SgDiaSemana = 'SAB' AND FlFeriadoNacional = 'N' THEN 0.4 ELSE 0 END AS FlSabUtil
    FROM DWBI_DIME.TbDimTempoData
  ),
  TbAggregated AS (
    SELECT
      NuAnoMes,
      SUM(FlDiaUtil) AS SumDiaUtil,
      SUM(FlSabUtil) AS SumSabUtil,
      SUM(CASE WHEN DtTempo < TODAY() THEN FlDiaUtil + FlSabUtil ELSE 0 END) AS SumDiaUtilPrd
    FROM TbTempoData
    WHERE
      NuAnoMes >= &safraM12.
      AND NuAnoMes <= &safraM0.
    GROUP BY NuAnoMes
  )
  SELECT
    NuAnoMes,
    SumDiaUtil + SumSabUtil AS TotalDiaUtil,
    SumDiaUtilPrd AS TotalDiaUtilPrd
  FROM TbAggregated
);
RUN;

PROC SQL;
CREATE TABLE TbPorte AS
  SELECT
    CdParceiroComercial,
    FlPorteA AS Porte,
	NmTipoParceiroComercial as TipoParceiro
  FROM TMP.TBPOTENCIALRODAS
  WHERE
    CdParceiroComercial <> .
    AND FlPorteA IS NOT NULL
  ORDER BY
    CdParceiroComercial,
    NuAnoMes DESC;
RUN;

PROC SORT DATA=TbPorte NODUPKEY;
BY CdParceiroComercial;
RUN;

PROC SQL;
CREATE TABLE TbRitoFinal AS
  SELECT
		A.*

		,cats(B.CdParceiroComercial,'-',B.NmParceiroComercial) as NmParceiroComercial
		,put(today(),date9.) 		as DtAtualizacao
		,&DtAtCetip. format date9.	as DtAtualizacaoCetip
		,Focal.LstColaboradorFocal
		,Sub.LstSubstituicao
		,B.NmColaboradorC
		,B.NmAreaEstruturaComercialC
		,B.NmSubEstruturaComercialC
		,cat(strip(B.DsLoginN1C),'@bv.com.br') as EmailDsLoginN1C
		,cat(strip(B.DsLoginN2C),'@bv.com.br') as EmailDsLoginN2C
		,cat(strip(B.DsLoginN3C),'@bv.com.br') as EmailDsLoginN3C
		,B.DsLoginN1C
		,B.DsLoginN2C
		,B.DsLoginN3C
		,B.NmTipoParceiroComercial as TipoParceiro
		,coalesce(C.Porte,'Sem Porte') as Porte
		,D.TotalDiaUtil as TotalDiaUtil FORMAT=num8.2
		,D.TotalDiaUtilPrd as TotalDiaUtilPrd FORMAT=num8.2
		,coalesce(E.ClassFinal,'Sem Classificação')  as DsClassificacaoRiscos
		,ins.Agrupamento as Insights

FROM TbRitoUnion AS A
  	left join tmp.TBPARCEIROCOMERCIAL as B
		on b.CdParceiroComercial = a.CdParceiroComercial

	LEFT JOIN WORK.TBPORTE C
		ON A.CDPARCEIROCOMERCIAL = C.CDPARCEIROCOMERCIAL

	LEFT JOIN WORK.TbDiasUteis D
		ON A.NuAnoMes = D.NuAnoMes

	LEFT JOIN WORK.TbClassificacaoRiscosPDVs E
		ON A.CDPARCEIROCOMERCIAL = E.CdLoja

	left join basegrc.TbColaboradorFocal as Focal
/*		on B.NmAreaEstruturaComercialC = Focal.NmAreaEstruturaComercial*/
		on B.CdfilialC = Focal.Cdfilial
		and Focal.NmGrupoProdutoMeta = 'Leves'
	/*	and Focal.NmCanalEstruturaComercial = 'Canal Veiculos' */

	left join bases.TbEStruturaComercial as Sub
		on B.NmAreaEstruturaComercialC = Sub.NmAreaEstruturaComercial
		and B.NmColaboradorC = Sub.NmColaborador
/*		and Sub.NmCanalEstruturaComercial = 'Canal Veiculos'*/
		and Sub.NuAnoMes = &safraM0.

	left join TBINSIGHTS as ins
		on a.cdparceirocomercial = ins.PDV

	Where b.NmSubEstruturaComercialC = 'Canal de Negócios'
 ;RUN;
