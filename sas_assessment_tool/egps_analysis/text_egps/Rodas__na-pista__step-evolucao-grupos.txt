PROC SQL;
CREATE TABLE GRUPOSD_01 AS
  SELECT
    GR_Vendedor AS LoginGR,
    COMPRESS(GR_Vendedor,".","") AS LoginGRTxt,
    DtCriacaoGrupo AS Data,
    COUNT(DISTINCT NuCnpjLojaFilial) AS QtdLojas
  FROM
    NAPISTA.TBGRUPOS
  WHERE
    DtAtualizacao = &grupos.
   AND GR_Vendedor IS NOT NULL
   AND GR_Vendedor LIKE "%.%"
  GROUP BY 1,2,3;
RUN;

DATA GRUPOSD_02;
SET GRUPOSD_01;
  RETAIN QtdLojasCadastradas;
  IF FIRST.LoginGR THEN QtdLojasCadastradas = QtdLojas;
  ELSE QtdLojasCadastradas = QtdLojasCadastradas + QtdLojas;
  BY LoginGR;
RUN;

PROC SQL; SELECT "'"||PUT(MIN(Data),yymmddn8.)||"'", "'"||PUT(MAX(Data),yymmddn8.)||"'", MAX(Data) INTO :MinGrp, :MaxGrp, :MaxGrpN FROM GRUPOSD_02;

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbDias AS SELECT * FROM connection to dbcon (
  SELECT DtTempo FROM DWBI_DIME.TbDimTempoData WHERE DtTempo BETWEEN &MinGrp. AND &MaxGrp.
);
RUN;

DATA TbDiasGR;
SET GRUPOSD_02 (DROP=LoginGRTxt QtdLojas);
  STOP;
RUN;

PROC SQL; CREATE TABLE GRS AS SELECT DISTINCT LoginGR,LoginGRTxt FROM GRUPOSD_02;

%Macro TbDiasGR (GR,GR2);
DATA &GR.;
SET GRUPOSD_02;
  WHERE LoginGR = "&GR2.";
RUN;

PROC SQL; SELECT MIN(Data) INTO :MinGR FROM &GR.; RUN;

PROC SQL;
CREATE TABLE Dias_&GR. AS
  SELECT
    "&GR2." AS LoginGR,
    A.DtTempo AS Data,
    B.QtdLojasCadastradas AS QtdLojas
  FROM
    TbDias A
  LEFT JOIN
    &GR. B
   ON A.DtTempo = B.Data
  WHERE
    A.DtTempo >= &MinGR.
  ORDER BY 1,2;
RUN;

DATA Dias_&GR._02 (DROP=QtdLojas);
SET Dias_&GR.;
  RETAIN QtdLojasCadastradas;
  IF FIRST.LoginGR THEN QtdLojasCadastradas = QtdLojas;
  ELSE IF QtdLojas = . THEN QtdLojasCadastradas = QtdLojasCadastradas;
     ELSE QtdLojasCadastradas = QtdLojas;
RUN;

PROC APPEND DATA=Dias_&GR._02 BASE=TbDiasGR;run;

PROC DATASETS LIBRARY=WORK;
  DELETE
    &GR.
    Dias_&GR.
    Dias_&GR._02;
RUN;
%Mend;

DATA _null_;
SET GRS;
  CALL EXECUTE('%TbDiasGR('||LoginGRTxt||','||LoginGR||')');
RUN;

PROC SQL;
CREATE TABLE TbParceirosGR AS
  SELECT
    DsLoginN1C AS LoginGR,
	CdTipoParceiroComercial,
	NmCanalEstruturaComercialC,
	COUNT(DISTINCT NuCnpjLoja) AS QtdLojas
  FROM
    TbParceiroComercial
  WHERE DsLoginN1C IS NOT NULL
    AND FlAtivo = 1
    AND CdTipoParceiroComercial in (5,2)
    AND NmCanalEstruturaComercialC = 'Canal Veiculos'
  GROUP BY 1,2,3;
RUN;

PROC SQL;
CREATE TABLE TbEstruturaGR AS
  SELECT DISTINCT
    NmSubEstruturaComercialC AS Regional,
    NmAreaEstruturaComercialC AS Filial,
    DsLoginN1C AS LoginGR,
    DsLoginN2C AS LoginGF,
    DsLoginN3C,
    LstSubstituicaoC
  FROM
    TbParceiroComercial
  WHERE DsLoginN1C IS NOT NULL
    AND DsLoginN1C LIKE "%.%";
RUN;

PROC SORT DATA=TbEstruturaGR NODUPKEY;
BY LoginGR;
RUN;

PROC SQL;
CREATE TABLE FiltroDatas AS
  SELECT DISTINCT
    CASE WHEN Data >= INTNX('DAY',&MaxGrpN.,-30) THEN Data
        ELSE INTNX('MONTH',Data,0,'END')
         END AS DtFiltro FORMAT=DATE9.
  FROM
    TbDiasGR;
RUN;

PROC SQL;
CREATE TABLE GRUPOS_DIARIO AS
  SELECT
    A.LoginGR,
    CASE WHEN Data >= INTNX('DAY',&MaxGrpN.,-30) THEN Data
        ELSE INTNX('MONTH',Data,0,'BEG')
        END AS Data FORMAT=DATE9.,
    A.QtdLojasCadastradas,
    C.QtdLojas AS QtdLojasTotal,
    D.Regional,
    D.Filial,
    D.LoginGF,
    D.DsLoginN3C,
    D.LstSubstituicaoC
  FROM
    TbDiasGR A
  INNER JOIN
    FiltroDatas B
   ON A.Data = B.DtFiltro
  LEFT JOIN
    TbParceirosGR C
   ON A.LoginGR = C.LoginGR
  LEFT JOIN
    TbEstruturaGR D
   ON A.LoginGR = D.LoginGR
  ORDER BY 1,2;
RUN;
