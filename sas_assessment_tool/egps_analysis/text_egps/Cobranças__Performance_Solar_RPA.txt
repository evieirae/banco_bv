PROC SQL; 
	CONNECT TO SYBASEIQ AS DBCON
		(HOST='PTA-IQPRD01'
		SERVER=PTAIQPRD
		DB=PTAIQCONT
		USER=&USUARIO.
		PASSWORD=&SENHA.);
	CREATE TABLE ATIVA.BASE_SOLAR_&DT_BASE. AS
		SELECT DISTINCT
			A.Referencia,
			A.FL_BASE,
			A.NUCONTRATO,
			A.SQCONTRATOFINANCEIRO,
			A.NR_CPFCNPJ,
			A.ATRASO,
			A.CD_ASSESSORIA,
			A.RISCO_ENVIO,
			A.DTDISPONIBILIZACAO,
			A.DTNOVADISPONIBILIZACAO,
			A.TPCOB,
			A.DT_ENVIO,
			A.DT_ENVIO_NOVO,
			A.DT_VCT,
			A.DMQUE,
			A.SQFLEX,
			CASE WHEN A.FL_BASE EQ "FLEX" THEN C.NUPARCELA
				ELSE B.NUPARCELA END AS NUPARCELA,
			CASE WHEN A.FL_BASE EQ "FLEX" THEN C.VRPARCELA
				ELSE B.VRPARCELA END AS VRPARCELA,
			CASE WHEN A.FL_BASE EQ "FLEX" AND C.NUPARCELA IS NULL THEN "FLEX NAO ENCONTRADA"
				WHEN A.FL_BASE EQ "FLEX" AND C.NUPARCELA IS NOT NULL THEN "FLEX ENCONTRADA"
				WHEN A.FL_BASE EQ "GCON" AND B.NUPARCELA IS NULL THEN "GCON NAO ENCONTRADA"
				WHEN A.FL_BASE EQ "GCON" AND B.NUPARCELA IS NOT NULL THEN "GCON ENCONTRADA"
				ELSE "OUTROS" END AS PARCELA_ENCONTRADA

		FROM ATIVA.BASE_SOLAR_&DT_BASE. A
		LEFT JOIN CONNECTION TO DBCON(
			SELECT
				NUCONTRATO,
				SQCONTRATOFINANCEIRO,
				NUPARCELA,
				VRPARCELA,
				DTVENCIMENTO
			FROM DBA.VWTBPARCELA
			WHERE
				(DTVENCIMENTO BETWEEN &DTVCT_MIN. AND &DTVCT_MAX.)
				AND NUCONTRATO > 13000000000000
		) B
		ON A.NUCONTRATO = B.NUCONTRATO
			AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

			AND	((A.DT_VCT - DATEPART(B.DTVENCIMENTO)) > -5 AND (A.DT_VCT - DATEPART(B.DTVENCIMENTO)) < 13)

		LEFT JOIN CONNECTION TO DBCON(
			SELECT
				NUCONTRATO,
				SQCONTRATOFINANCEIRO,
				NUPARCELA,
				VRPARCELA,
				DTVENCIMENTO
			FROM STGTXT_FLEX.TbParcela
			WHERE
				DTVENCIMENTO BETWEEN &DTVCT_MIN. AND &DTVCT_MAX.
		) C
		ON A.NUCONTRATO = C.NUCONTRATO
		AND A.SQFLEX = C.SQCONTRATOFINANCEIRO  
			AND ((A.DT_VCT - DATEPART(C.DTVENCIMENTO)) > -5 AND (A.DT_VCT - DATEPART(C.DTVENCIMENTO)) < 13);
QUIT;
RUN;
