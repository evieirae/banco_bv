      from bv-pdbd-prd.trusted_dwbi_dmoc.tbfatosolicitacaoautorizacao
	  WHERE DtsolicitacaoAutorizacaoEmsr >= '2025-07-01' AND DtsolicitacaoAutorizacaoEmsr <= '2025-07-31'
