PROC SQL;
CREATE TABLE WORK.ACORDOSNOVABASE AS

SELECT

,A.*

FROM ACORDOS.NOVABASE AS A

PROC SQL;
CREATE TABLE WORK.MIS_NUMREG AS

case when substr(case when a.Cd_causa = '' then a.SerialImport else a.Cd_causa end,1,9) = "projuris-"

,a.Pasta
,a.DtCalculoTarifa as DtEnvio

PROC SQL;
CREATE TABLE WORK.TARIFAMIS AS
SELECT
a.NumRegSistContrGest length = 60
,a.Pasta			  length = 30
,a.DtEnvio
,b.DBCadastro			as DtCadastro
,b.MotivoPadrao
,b.Submotivo
,b.UF
,b.Municipio
,c.Assessoria
,a.'FASE1ALÇADA1'n 				as Fase1Alcada1
,a.'FASE1ALÇADA2'n 				as Fase1Alcada2
,a.'FASE1ALÇADA3'n 				as Fase1Alcada3
,. 								as Contraproposta
,a.'FASE2ALÇADA4'n				as Fase2Alcada4

,. 								as AlcadaEspecial
,b.DtUltPagto
,b.Subsidios
, . 							as Provisao
, . 							as ValorCondenacaoMOD
, -b.VlPrincipal 				as ValorPago
, b.Desfecho_Pagtos				as DesfechoPagtos
,b.StatusAJ 					as StatusAcaoJudicial
,'MIS' length = 10 as Origem
,CASE WHEN (B.Desfecho_Pagtos = "2.Acordo" AND B.Desfecho_Status not = "3.Condenação" ) THEN 'ACORDO SEM CONDENAÇÃO'
	  WHEN (A.Pasta not='' AND A.'FASE1ALÇADA3'n not=. AND (B.Desfecho_Pagtos not= '2.Acordo')) THEN 'ACORDO NAO FECHADO'
     ELSE 'TARIFAS' END AS  NmBase length = 30
,CASE WHEN (B.Desfecho_Pagtos = "2.Acordo" AND B.Desfecho_Status not = "3.Condenação" ) THEN 'Sem Alçada'
	  WHEN (A.Pasta not='' AND A.'FASE1ALÇADA3'n not=. AND (B.Desfecho_Pagtos not= '2.Acordo')) THEN 'Alçada não fechada'
     ELSE 'Tarifas' END AS  AlcadaFechada length = 30

LEFT JOIN WORK.ASSESSORIA_EXCLUSIVO c ON (c.Pasta = a.Pasta)

;run;

ELSE IF ValorPago <= Fase2Alcada4 																		THEN ALCADA_FECHADA2 = 'Alçada 4';
ELSE IF ValorPago <= Fase2Alcada5 																		THEN ALCADA_FECHADA2 = 'Alçada 5';

PROC SQL;
CREATE TABLE WORK.MONITORAMENTO AS
SELECT

 t1.NumRegSistContrGest   FORMAT $CHAR100.
,t1.Pasta 			      FORMAT $CHAR100.
,t2.DtEnvioInicial 				            as DtEnvio
,t1.DBCadastro					            as DtCadastro
,t1.MotivoPadrao
,t1.Submotivo
,t1.UF
,t1.Municipio
,t3.Assessoria
,t2.'FASE1ALÇADA1'n 	                   	as Fase1Alcada1
,t2.'FASE1ALÇADA2'n			                as Fase1Alcada2

,t2.'FASE2ALÇADA4'n			                as Fase2Alcada4

,t2.'Alçada Especial'n	                 	as AlcadaEspecial
,t2.FlagSerasa
,t1.DtUltPagto
,t1.Subsidios
,t2.'Provisão'n				                as Provisao
,t2.'Valor Condenação_MOD'n                 as ValorCondenacaoMOD

,t1.Desfecho_Pagtos				            as DesfechoPagtos

AND t1.Desfecho_Pagtos = '2.Acordo'
AND t1.Desfecho_Status NE '3.Condenação'
;run;

proc sql;
create table Work.tbl1482MONITFINAL  as
select
a.*
,b.NmAndamento
,b.DtAndamento
,d.VlProbCondenacaoModeloRiscos
,c.NmAndamento as AcordoParalelo
,intnx('day' , b.DtAndamento , 21916 ) as DtAndamento2C
,0 as FlPagoAcimaAlcada
,(a.ValorPago -
	max(a.Fase1Alcada1,
		a.Fase1Alcada2,
		a.Fase1Alcada3,
		a.Fase2Alcada4,
		a.Fase2Alcada5,
		a.AlcadaEspecial)
	)
	 as Dif
,max(a.Fase1Alcada1,
		a.Fase1Alcada2,
		a.Fase1Alcada3,
		a.Fase2Alcada4,
		a.Fase2Alcada5,
		a.AlcadaEspecial) + 1000 as VlAlcadaPlus1000
,e.NmEncerramentoBoletim
,e.NmMotivoEncerramento
, case when e.NmMotivoEncerramento in (
'ACORDO COM PAGAMENTO'
,'ACORDO COM RECEBIMENTO'
,'ACORDO SEM PAGAMENTO'
,'PROCEDÊNCIA ACORDO PÓS-CONDENAÇÃO') then 'S' else 'N' end as FlAcordo
,F.VlAcordo AS VlAcordoAtual
,g.FlagSerasa

from Work.tbl1482MONITORAMENTOFINAL as a

left join work.TblAndamentoAptoInapto as b
on a.pasta = b.pasta

left join work.TblAndamentoOrigemPgto as c
on a.pasta = c.pasta

left join work.TLBProbCondModRis as d
on a.pasta = d.CdPasta

left join WORK.TblEncerramento as e
on a.pasta = e.CdPasta

left join WORK.TBLPROCESSOACORDO as F
on a.pasta = F.Pasta

LEFT JOIN WORK.NOVABASE as g
ON a.pasta = g.Pasta

where 1=1
;run;
