CASE
	WHEN NUANOMES >= 202504 and NUANOMES <= 202512 AND NmCanalEstruturaComercial = 'Canal Solar' AND TPVALOR = 'VrProducao' THEN MIN((VrMedidaNIProj / ((VrMetaN1+VrMetaNIN1)*0.8)),&Max_Atingimento.)
	WHEN TPVALOR = 'VrProducao' THEN MIN((VrMedidaNIProj / (VrMetaN1+VrMetaNIN1)),&Max_Atingimento.)
ELSE 0 END AS Pc_AtingMeta_VrProducao,

CASE
	WHEN nuanomes <= 202509 and NmCanalEstruturaComercial in ('Canal Veiculos') and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Rodas+VrMargemDescontos) / (VrMetaN1_MC_Rodas + VrMetaNIN1_MC_Rodas)),&Max_Atingimento.)
	WHEN nuanomes >= 202510 and NmCanalEstruturaComercial in ('Canal Veiculos') and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Rodas+VrMargemDescontos_MC_Rodas) / (VrMetaN1_MC_Rodas + VrMetaNIN1_MC_Rodas)),&Max_Atingimento.)

	WHEN nuanomes <= 202507 and NmCanalEstruturaComercial in ('Canal Negocios') and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Rodas+VrMargemDescontos) / (VrMetaN1_MC_Rodas + VrMetaNIN1_MC_Rodas)),&Max_Atingimento.)
	WHEN nuanomes >= 202508 and NmCanalEstruturaComercial in ('Canal Negocios') and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Rodas+VrMargemDescontos_MC_Rodas) / (VrMetaN1_MC_Rodas + VrMetaNIN1_MC_Rodas)),&Max_Atingimento.)

	WHEN NuAnoMes <= 202403 and NmCanalEstruturaComercial = 'Canal Solar' and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaProj_MC_Solar+VrMargemDescontos) / (VrMetaN1_MC_Solar)),&Max_Atingimento.)
	WHEN NuAnoMes >= 202404 and NuAnoMes <= 202503 and NmCanalEstruturaComercial = 'Canal Solar' and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Solar+VrMargemDescontos) / (VrMetaNIN1_MC_Solar)),&Max_Atingimento.)
	WHEN NuAnoMes >= 202504 and NUANOMES <= 202512 and NmCanalEstruturaComercial = 'Canal Solar' and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaNIProj_MC_Solar+VrMargemDescontos) / (VrMetaNIN1_MC_Solar*0.8)),&Max_Atingimento.)

	WHEN NmCanalEstruturaComercial = 'Canal Privado e FGTS' and TPVALOR = 'VrMargemContribuicao' THEN MIN(((VrMedidaProj_MC_Privado+VrMargemDescontos) / (VrMetaN1_MC_Privado)),&Max_Atingimento.)

PROC SQL;
CREATE TABLE BASE_ACELERADOR_CV_CN AS
SELECT

A.*,
CASE WHEN B.Meta_Acelerador IS NULL THEN 0 else B.Meta_Acelerador end as Meta_Acelerador format commax20.2,

CASE
	WHEN
		A.NuAnoMes >= B.NuAnoMes_Ini AND A.NuAnoMes <= B.NuAnoMes_Fim
		AND A.NmCanalEstruturaComercial = B.NmCanalEstruturaComercial
		AND A.TpValor = 'VrProducao'
/*	then A.AtingProdN1Proj*/
		then a.Pc_AtingMeta_VrProducao
ELSE 0 END AS Ating_Meta_Prod FORMAT COMMAX20.10,

CASE
	WHEN
		A.NuAnoMes >= B.NuAnoMes_Ini AND A.NuAnoMes <= B.NuAnoMes_Fim
		AND A.NmCanalEstruturaComercial = B.NmCanalEstruturaComercial
		AND A.TpValor = 'VrMargemContribuicao'
/*	then A.AtingMcN1Proj*/
		then a.Pc_AtingMeta_VrMC_Rodas
ELSE 0 END AS Ating_Meta_MC_Rodas FORMAT COMMAX20.10,

(
(CASE WHEN A.TpValor = 'VrProducao' AND CALCULATED Ating_Meta_Prod >= 1 THEN CALCULATED Ating_Meta_Prod  ELSE 0 END)
	+
(CASE WHEN A.TpValor = 'VrMargemContribuicao' AND CALCULATED Ating_Meta_MC_Rodas >= 1 THEN CALCULATED Ating_Meta_MC_Rodas ELSE 0 END)
)
AS Ating_Meta_TpValor FORMAT COMMAX20.10,

(
(CASE WHEN A.TpValor = 'VrProducao' AND CALCULATED Ating_Meta_Prod >= 1 THEN 1 ELSE 0 END)
	+
(CASE WHEN A.TpValor = 'VrMargemContribuicao' AND CALCULATED Ating_Meta_MC_Rodas >= 1 THEN 1 ELSE 0 END)
)
AS Ating_Meta_TpValor_Aux FORMAT COMMAX20.10,

0 AS FlElegAcelerador

/*CASE WHEN CALCULATED FlElegAcelerador = 1 THEN 'Apto Acelerador' else 'Sem Acelerador' END AS NmElegAcelerador*/

FROM WORK.BASE_META_PRODUCAO_ATING A

LEFT JOIN WORK.BASE_ACELERADOR_GR B ON
1 = 1
AND A.NuAnoMes >= B.NuAnoMes_Ini
AND A.NuAnoMes <= B.NuAnoMes_Fim
AND A.NmCanalEstruturaComercial = B.NmCanalEstruturaComercial
AND A.TpValor = B.TpValor

where
A.NmCanalEstruturaComercial NOT IN ('Canal Solar')
/*1 = 1*/
/*and A.TpValor = 'VrProducao'*/
/*and A.NmGrupoProdutoMeta = 'Motos'*/
/*and a.CdColaborador = 69950*/

order by
A.DsLoginN1,
A.NUANOMES,
A.TPVALOR

;RUN;

/*A.NmColaborador,*/
A.NmGrupoCargo,
A.DsLoginN1,

B.Perc_Turbo AS Perc_Turbo,
C.Qtd_Eleg_KPI_Total,

/*sum(A.Pontos_Total) as Pontos_Total,*/

CASE WHEN A.NmGrupoCargo in ('Gerente Relacionamento Solar', 'Gerente Relacionamento Solar_3Cobans') AND NuAnoMes >= 202601 and (A.SPF < 0.7 or A.AP_PLACA < 0.7) THEN 1 ELSE 0 END AS Fl_Punicao_Solar, /*INCLUIDO A PARTIR DE JAN-26*/

sum(CASE
		WHEN A.NmGrupoCargo in ('Gerente Relacionamento Solar', 'Gerente Relacionamento Solar_3Cobans') AND Calculated Fl_Punicao_Solar = 1 THEN (A.Pontos_Total - (A.Pontos_Total * 0.30))
		WHEN A.NmGrupoCargo in ('Gerente Relacionamento Solar', 'Gerente Relacionamento Solar_3Cobans') AND Calculated Fl_Punicao_Solar = 0 THEN A.Pontos_Total
		WHEN A.NmGrupoCargo NOT in ('Gerente Relacionamento Solar', 'Gerente Relacionamento Solar_3Cobans') THEN A.Pontos_Total
ELSE 0 END) AS Pontos_Total, /*INCLUIDO A PARTIR DE JAN-26*/

sum(A.Qtd_Metas_Ating) as Qtd_Metas_Ating,

CASE WHEN CALCULATED Qtd_Metas_Ating = C.Qtd_Eleg_KPI_Total THEN 1 ELSE 0 END AS FlAting_PtsTurbo,

CASE WHEN CALCULATED FlAting_PtsTurbo = 1 THEN (CALCULATED Pontos_Total+(Perc_Turbo*CALCULATED Pontos_Total)) ELSE CALCULATED Pontos_Total END AS Pontos_Turbo

FROM WORK.BASE_META_PRODUCAO_PONTOS_TOTAL A

LEFT JOIN WORK.BASE_PESOS_TURBO B ON
1 = 1
AND A.NuAnoMes >= B.NuAnoMes_Ini
AND A.NuAnoMes <= B.NuAnoMes_Fim
AND A.NmCanalEstruturaComercial = B.NmCanalEstruturaComercial
AND A.NmGrupoCargo = B.NmGrupoCargoColaborador

LEFT JOIN WORK.BASE_QTD_METAS_GR C ON
1 = 1
AND A.NuAnoMes >= C.NuAnoMes_Ini
AND A.NuAnoMes <= C.NuAnoMes_Fim
AND A.NmCanalEstruturaComercial = C.NmCanalEstruturaComercial
AND A.NmGrupoCargo = C.NmGrupoCargoColaborador

/*WHERE DSLOGINN1 = 'aarao.souza'*/

GROUP BY 1,2,3,4,5,6,7
ORDER BY
A.DsLoginN1,
A.NUANOMES
