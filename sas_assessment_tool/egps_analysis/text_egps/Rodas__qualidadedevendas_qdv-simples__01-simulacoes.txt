PROC SQL; CREATE TABLE MsgProposta AS
SELECT
MsgPr.NuProposta
,MsgPr.Mensageria
,MsgPr.Mensageria2
,case when GrupoMsg='P' then 'Mens. Positiva'
      when GrupoMsg='R' then 'Mens. Recusada'
	  else 'Mens. Demais'
	  end as MensageriaAgrupda
FROM SOLITMP.MsgPropostaQDV MsgPr
left join auxl.grupo_mensageria Gmens on Gmens.Mensageria=MsgPr.Mensageria
ORDER BY 2 DESC;RUN;

PROC SORT DATA=MsgProposta NODUPKEY;
BY NuProposta;
RUN;

PROC SQL; CREATE INDEX NuProposta ON MsgProposta; QUIT;
