SET ESTUDOS2.BANKLY3040_&SAFRA.;

;RUN;

;RUN;

DATA BASE_TOTAL2_&SAFRA.;
SET  BASE_TOTAL_&SAFRA.;

	IF LENGTH(Contrt) = 14 THEN DO ;
		NU_CONTA_1 = SUBSTR(Contrt,5)*1;
		CD_PRODUTO_1 = SUBSTR(Contrt,1,2)*1;
	END;
	ELSE IF LENGTH(Contrt) < 14 THEN DO;
		NU_CONTA_1 = Contrt*1;
		CD_PRODUTO_1 = .;
	END;

RUN;

PROC SQL;
CREATE TABLE BASE_&SAFRA. AS
SELECT DISTINCT
A.*
,4 AS CD_PRODUTO_2
,CASE WHEN LENGTH(A.Contrt) > 14 THEN C.NU_CONTA_1 ELSE A.NU_CONTA_1 END AS NU_CONTA_2

FROM BASE_TOTAL2_&SAFRA. AS A

LEFT JOIN (SELECT DISTINCT CD,NU_CONTA_1,CD_PRODUTO_1 FROM BASE_TOTAL2_&SAFRA. WHERE CD_PRODUTO_1 NE .) AS B ON A.CD = B.CD AND A.NU_CONTA_1 = B.NU_CONTA_1
LEFT JOIN (SELECT DISTINCT CD,NU_CONTA_1,CD_PRODUTO_1 FROM BASE_TOTAL2_&SAFRA. WHERE CD_PRODUTO_1 NE .) AS C ON A.CD = C.CD

;QUIT;

cd as NUCPFCNPJ
,NU_CONTA_2 as NUCONTA
,CD_PRODUTO_2 as PRODUTO

;QUIT;

;QUIT;
