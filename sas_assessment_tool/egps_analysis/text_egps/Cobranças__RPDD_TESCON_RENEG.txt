PROC SQL;
CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=20);
CREATE TABLE RENEG_RPDD AS
SELECT
	A.*,
	CASE
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.02 		THEN "00,0-02,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.05 		THEN "02,1-05,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.08 		THEN "05,1-08,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.13 		THEN "08,1-13,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.19 		THEN "13,1-19,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.27 		THEN "19,1-27,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.37 		THEN "27,1-37,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.56 		THEN "37,1-56,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.85 		THEN "56,1-85,0%"
		WHEN Parcelas_pagas_ate_30d/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 1 			THEN "85,1-100%"
		ELSE "ERRO"
	END AS PERCENT_PAGO_ATE_30D,
	CASE
		WHEN C.vr_saldo_carteira <= 10000 	THEN "01. ATÉ 10K"
		WHEN C.vr_saldo_carteira <= 20000 	THEN "02. ATÉ 20K"
		WHEN C.vr_saldo_carteira <= 30000 	THEN "03. ATÉ 30K"
		WHEN C.vr_saldo_carteira <= 45000 	THEN "04. ATÉ 45K"
		WHEN C.vr_saldo_carteira >  45000 	THEN "05. ACIMA DE 45K"
	END AS FX_SALDO,
	C.vr_saldo_carteira AS SALDO_M0,
	C.vr_provisao_total AS PROVISAO_M0,
	D.vr_saldo_carteira AS SALDO_MOB,
	D.vr_provisao_total AS PROVISAO_MOB

FROM BASE_ELEG AS A

LEFT JOIN RPDD_M0 AS C
ON A.NuContrato = INPUT(C.cd_contrato, BEST32.)

LEFT JOIN RPDD_MOB AS D
ON A.NuContrato = INPUT(D.cd_contrato, BEST32.)

LEFT JOIN CONNECTION TO PERM(
SELECT DISTINCT cd_contrato,
             	sq_contrato_financeiro,
				SUM(CASE WHEN st_parcela='Q' AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END)        AS Parcelas_Pagas,
                SUM(CASE WHEN st_parcela='A' OR (st_parcela='Q' AND dt_ultimo_recebimento > &fim_mes_m0.) THEN 1 ELSE 0 END)        AS Parcelas_Abertas,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) < 30 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) AS Parcelas_pagas_ate_30d

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G
ON A.NuContrato = INPUT(G.cd_contrato, BEST32.) AND A.SqContratoFinanceiro = G.sq_contrato_financeiro
;QUIT;

PROC SQL;
CREATE TABLE CUBO_EXPORT_&anomes._&mob. AS
SELECT
	&anomes. 	AS SAFRA,
	&MOB. 		AS MOB,
	BASE,
	PRODUTO,
	FX_ATRASO,
	MODELO_FINAL,
	Teste_Controle_Final,
	PERCENT_PAGO_ATE_30D,
	FX_SALDO,
	COUNT(NUCONTRATO) 	AS QNT,
	SUM(SALDO_M0) 		AS SALDO_M0,
	SUM(SALDO_MOB) 		AS SALDO_MOB,
	SUM(PROVISAO_M0) 	AS PROVISAO_M0,
	SUM(PROVISAO_MOB) 	AS PROVISAO_MOB

FROM RENEG_RPDD
GROUP BY
	SAFRA,
	MOB,
	BASE,
	PRODUTO,
	FX_ATRASO,
	MODELO_FINAL,
	Teste_Controle_Final,
	PERCENT_PAGO_ATE_30D,
	FX_SALDO
;QUIT;

%mend;

SELECT *
FROM CONNECTION TO PERM(

SELECT DISTINCT
    A.nucontratogestao,
    A.sqcontratofinanceiro,
	A.dt_baixa,
	MIN(B.dt_vencimento) as dt_vencimento,


    MAX(DATE_DIFF(A.dt_baixa, B.dt_vencimento, DAY)) as atraso

FROM (
  SELECT DISTINCT
	nucontratogestao,
	sqcontratofinanceiro,
	MAX(DATE(dt_baixa)) as dt_baixa,
	MAX(DATE(dt_reneg)) as dt_reneg

FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
WHERE dt_baixa >= &dt_inicio_reneg. AND dt_baixa < &dt_fim_reneg.
	AND (dt_cancelamento IS NULL OR dt_reativacao IS NOT NULL)
	AND st_reneg = 3
group by 1,2
) A

LEFT JOIN (
  SELECT DISTINCT
	cd_contrato,
	sq_contrato_financeiro,
	st_parcela,
	DATE(dt_ultimo_recebimento) as dt_ultimo_recebimento,
	DATE(dt_vencimento) as dt_vencimento

FROM bv-pdbd-prd.dvry_core.tbparcela

) B
ON B.cd_contrato = A.nucontratogestao and B.sq_contrato_financeiro = A.sqcontratofinanceiro
WHERE B.st_parcela = 'A' OR (B.st_parcela = 'Q' and date(B.dt_ultimo_recebimento) >= date(A.dt_baixa))



group by 1,2,3

)
;quit;
