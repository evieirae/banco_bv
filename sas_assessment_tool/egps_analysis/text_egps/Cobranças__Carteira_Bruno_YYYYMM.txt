CREATE TABLE BASE_RENEG_C AS
SELECT A.*,

	   
       B.Qtd_Parcelas_Abertas 						 AS QtdParcelas_Antigo,
	   B.Atraso                                      AS Atraso_Antigo,
	   DATEPART(B.DT_VENCIMENTO_NOVO) FORMAT= date9. AS DtVencimento_Antigo,
	   CASE
		   WHEN B.Atraso = .    THEN 'i. Missing'
		   WHEN B.Atraso <=   0 THEN 'a. Em dia'
		   WHEN B.Atraso <=  30 THEN 'b. 1 a 30'
		   WHEN B.Atraso <=  60 THEN 'c. 31 a 60'
		   WHEN B.Atraso <=  90 THEN 'd. 61 a 90'
		   WHEN B.Atraso <= 120 THEN 'e. 91 a 120'
		   WHEN B.Atraso <= 180 THEN 'f. 120 a 180'
		   WHEN B.Atraso <= 360 THEN 'g. 180 a 360'
		   WHEN B.Atraso  > 360 THEN 'h. WO'
	   END AS Fx_Atraso_Antigo,

	   B.SAFRA_BAIXA,

	   CASE WHEN B.Atraso > 0 THEN 1 ELSE 0 END AS FLAG_ATRASO,
	   CASE WHEN B.Atraso <= 0 THEN 1 ELSE 0 END AS FLAG_EMDIA,

	   C.VrParcela_Novo,
       DATEPART(C.DtVencimento_Novo)  FORMAT= date9. AS DtVencimento_Novo

	SELECT cd_contrato as NuContrato,

     		WHERE Nu_parcela = 1) AS C
		ON A.NuContrato = C.NuContrato AND
			(A.SqContrato_m1 + 1) = C.SqContratoFinanceiro
