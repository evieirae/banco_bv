      %put %str(NOTE: Creating table ...);

      data PEPRDMT.TB_LOG_PEPR
              (COMPRESS = YES);
         attrib DtProcessamento length = 8
            format = ddmmyy.
            informat = DATETIME.;
         attrib QtdRegistrosLidos length = 8;
         attrib JOBSAS length = $40;
         attrib ASSUNTO length = $40;
         attrib QtdRegistrosGravados length = 8;
         attrib STATUS length = $8;
         attrib DESCRICAO length = $50;
         attrib DTREFERENCIA length = 8;
         call missing(of _all_);
         stop;
      run;

      %rcSet(&syserr);

   %end;  /* if table does not exist  */

   /*---- Append  ----*/
   %put %str(NOTE: Appending data ...);

   proc append base = PEPRDMT.TB_LOG_PEPR(COMPRESS = YES)
      data = &etls_lastTable (&etls_tableOptions)  force ;
    run;

   %rcSet(&syserr);

   proc datasets lib = work nolist nowarn memtype = (data view);
      delete W1176J45H;
   quit;

%mend etls_loader;

/* Check return code for job  */
%macro etls_jobRCChk;
   %if (&job_rc ge 5) %then
   %do;
      %macro etls_sendData(libref=, dataset=, message=);

         %local etls_syntaxcheck;
         %let etls_syntaxcheck = %sysfunc(getoption(syntaxcheck));
         /* Turn off syntaxcheck option to perform following steps  */
         options nosyntaxcheck;
		DATA _NULL_;

				CALL SYMPUT('STATUS','ERRO');
				CALL SYMPUT('DESCRICAO','PROCESSO FINALIZADO COM ERRO');

		RUN;

		   data work.etls_tempData;

				LENGTH ASSUNTO $40.
					   JOBSAS $40.
				       STATUS $8.
						DESCRICAO $50.
						DTREFERENCIA 8.;

				        QtdRegistrosGravados = 0;
				 		DtProcessamento =&DtProcessamento.;
						ASSUNTO = &ASSUNTO.;
						JOBSAS = "&etls_jobName.";
				 		QTDRegistrosLidos= 0;
						STATUS = "&STATUS.";
						DESCRICAO = "&DESCRICAO.";
						DTREFERENCIA = &DT_REF.;

			RUN;

         %local etls_obs;
         %let etls_obs = %sysfunc(getoption(obs));
         /* Set obs option to max to perform following steps  */
         options obs = max;

         proc append base=&libref..&dataset. data=work.etls_tempData force;   run;

         /* Reset obs option to previous setting  */
         options obs = &etls_obs;

         proc datasets lib = work nolist;
            delete etls_tempData;
         quit;

         /* Reset syntaxcheck option to previous setting  */
         options &etls_syntaxcheck;
      %mend etls_sendData;
      %etls_sendData
         (libref = PEPRDMT,
          dataset = TB_LOG_PEPR,
          Message = Error);

   %end;
%mend etls_jobRCChk;
%etls_jobRCChk;
