proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd'
READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100
);
create table TagVeicular as
select
*

from connection to perm(
  SELECT
    DISTINCT
    Cast(tdcb.dtinclusao as date) as dtinclusao,
    tcbe.cddadoclientebeneficio,
    treb.cdidentificadorbeneficio,
    tdcb.nucontacobranca,
    tcbe.DsUltimoEventoBeneficioEps
  FROM
    `bv-pdbd-prd.trusted_sgppgestaoparceria.tbclientebeneficioeps` tcbe
  INNER JOIN
    `bv-pdbd-prd.trusted_sgppgestaoparceria.tbdadoclientebeneficio` tdcb
  ON
    tcbe.cddadoclientebeneficio = tdcb.cddadoclientebeneficio
  INNER JOIN
    `bv-pdbd-prd.trusted_sgppgestaoparceria.tbretornoepsbeneficio` treb
  ON
    tcbe.cdclientebeneficioeps = treb.cdclientebeneficioeps /*and treb.TpStatusRetorno = 'ATIVO'*/
    where 1=1
    and Cast(tdcb.dtinclusao as date) < current_date()
    and tcbe.DsUltimoEventoBeneficioEps = 'ATIVO'
  ORDER BY
    2,4
 );
disconnect from perm;
quit;

PROC SQL;
CREATE TABLE TbContaDigitalFinal  AS
SELECT

	Producao.NuAnoMes
	,ContaDigital.SafraProposta
	,ContaDigital.SafraConta
	,ContaDigital.SafraAtivacao
	,ContaDigital.NuContrato
	,Producao.NuProposta
	,ContaDigital.NuCpfCnpj
	,ContaDigital.DtContrato
	,ContaDigital.DtConta
	,ContaDigital.DtAtivacao
	,ContaDigital.DtEncerramentoConta
	,ContaDigital.FlElegContaDigital
	,ContaDigital.FlContaDigital
	,ContaDigital.FlAtivacaoContaDigital
	,ContaDigital.FlElegContaDigitalComissao
	,ContaDigital.DtAtualizacao

	,ContaDigital.FlContaAberta
	,ContaDigital.FlHabilitadoDAVeiculos
	,ContaDigital.DtPrimeiroDARodas FORMAT DATE9.
	,ContaDigital.DtSafraPrimeiroDARodas FORMAT DATE9.
	,ContaDigital.FlDebitoAutomaticoEfetivado
	,ContaDigital.FlElegivelBonus
/*	,case when ContaDigital.FlElegivelBonus = 1 and today() - ContaDigital.DtContrato > 120 then 0 else 1 end as  FlElegivelBonusRevisada*/
	,case when ContaDigital.FlElegivelBonus = 1 and today() - ContaDigital.DtContrato > 120 then 0
		  when ContaDigital.FlElegivelBonus is null then 0
		  when ContaDigital.FlElegivelBonus = 0 	then 0
		else 1 end as  FlElegivelBonusRevisada

	,Producao.DsTipoCanalProducao
	,Producao.NmCanalEstruturaComercial
	,Producao.NmAreaEstruturaComercial
	,Producao.NmSubEstruturaComercialA
	,Producao.CdFilial
	,Producao.DsFilial
	,Producao.CdParceiroComercial
	,Producao.NmParceiroComercial
	,Producao.CdParceiroComercial
	,Producao.NmPessoa
	,Producao.TpPessoa
	,Producao.CdColaborador
	,Producao.NmColaborador
	,Producao.DsLoginN1
	,Producao.DsLoginN2A
	,Producao.DsLoginN3A
	,Producao.LstSubstituicao
	,Producao.NmGrupoProdutoMeta
	,Producao.VrProducao
	,ColaboradorFocal.LstColaboradorFocal
	,'              ' as ContatoNuCpfCnpj
	,'              ' as NuTelefone1
	,'              ' as NuTelefone2
	,tag.DsUltimoEventoBeneficioEps as StatusTag
	,Producao.DtPrimeiroVencimento
	,Producao.QtParcelas
	,ContaDigital.TpOferta
	,Case when tag.DsUltimoEventoBeneficioEps = 'ATIVO' then 1 else 0 end as FlStatusTag

FROM basegrc.TbContaDigital60 ContaDigital

	INNER JOIN TMP.TbProducao Producao
		ON ContaDigital.NuContrato = Producao.NuContrato

	LEFT JOIN basegrc.TbColaboradorFocal ColaboradorFocal
			ON  Producao.CdFilial = ColaboradorFocal.CdFilial
		    AND Producao.NmGrupoProdutoMeta = ColaboradorFocal.NmGrupoProdutoMeta
			/*AND ColaboradorFocal.NmCanalEstruturaComercial = 'Canal Veiculos'*/

	LEFT JOIN  TagVeicular as tag
	ON ContaDigital.nuconta = tag.nucontacobranca

;RUN;

  SELECT MAX(DtAtualizacao) format date9.   INTO :max_DtAtualizacao   FROM TbContaDigitalFinal_Final;

/*Corta a base e Joga o valor maximo pra base por conta do congelamento*/

data TbContaDigitalFinal_Final60 (Drop= NuTelefone1 NuTelefone2);

set	TbContaDigitalFinal_Final (Where=(NuAnomes>=202506 and NuAnomes <= &safraM0.))
	Bases.TbContaDigitalFinal_Final (Where=(NuAnomes >= 202401 and NuAnomes <= 202505));
	DtAtualizacao = input("&max_DtAtualizacao.", date9.);
	format DtAtualizacao date9.;

	run;

/*Atualiza telefones*/

	PROC SORT DATA=bases.tbpessoa (where=(NuTelefone1 <> '' and
									 NuTelefone2 <> '' and

BY NuCpfCnpj;
RUN;

proc sort data=tbpessoa; by NuCpfCnpj; run;

proc sort data=TbContaDigitalFinal_Final60; by NuCpfCnpj; run;

data bases.TbContaDigitalFinal_Final60;
	merge TbContaDigitalFinal_Final60 (in=a)
	tbpessoa (in=b keep=NuCpfCnpj NuTelefone1 NuTelefone2 );
	by NuCpfCnpj;
	if a;
