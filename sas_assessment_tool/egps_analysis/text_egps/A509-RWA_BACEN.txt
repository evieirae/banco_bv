%macro BusinessDay();

PROC SQL;
   CREATE TABLE WORK.BusinessDay AS
   SELECT 	t1.ID,
			tranwrd(t1.BusinessDay, "{", "") as BusinessDay1 length=30000,
			tranwrd(calculated BusinessDay1, "}", "") as BusinessDay length=30000
      FROM WORK.rwa_bacenmor_AUX t1
/*      WHERE t1.ID contains '00009#1#A#211#NULO#375#SPE#8'*/
;QUIT;

data BusinessDay (drop=BusinessDay1);
  set BusinessDay;
  save=BusinessDay ;
  do i=1 to count(save,'|')+1;
     BusinessDay=scan(save,i,'|');
	 output;
  end;
  drop i save;
run;

data BusinessDay;
  n=_N_;
  set BusinessDay;
run;

			  input(tranwrd(tranwrd(tranwrd(t12.Auxiliary_Amount, "BRL", ""),")",""),"(","-"),32.10) as Auxiliary_Amount informat BEST32.10 format BEST32.10

	PROC SQL;
		CREATE TABLE WORK.REG_TBONDS_ADROPTIONS AS
			SELECT
				t1.DtReferencia,
				t1.ID,
				t1.RWA,
				t1.BusinessDay,
				t1.FatorRisco,
				t1.Delta,
				t1.Bacen_Exposure_Amount
			FROM
				WORK.REGULATORIO_FINAL t1
			WHERE
				t1.CodigoModalidade IN (298,328) AND
				t1.RWA = 'PJUR2'
		;
	QUIT;

	proc sql;
		CREATE TABLE WORK.SUM_TBONDS_ADROPTIONS AS
			SELECT
				t1.DtReferencia,
				t1.ID,
				SUM(t1.Bacen_Exposure_Amount) * t1.Delta AS Bacen_Exposure_Amount
			FROM
				WORK.REG_TBONDS_ADROPTIONS t1
			GROUP BY
				t1.DtReferencia,
				t1.ID
		;
	quit;

	proc sql;
		CREATE TABLE WORK.REG_AUX_TBONDS_ADROPTIONS AS
			SELECT
				t1.DtReferencia,
				t1.ID,
				t1.RWA,
				t1.FatorRisco,
				MIN(t1.BusinessDay) AS BusinessDay,
				t2.Bacen_Exposure_Amount
			FROM
				WORK.REG_TBONDS_ADROPTIONS t1
			INNER JOIN
				WORK.SUM_TBONDS_ADROPTIONS t2
				ON t2.DtReferencia = t1.DtReferencia
				AND t2.ID = t1.ID
			WHERE
				t1.BusinessDay NE 1
			GROUP BY
				t1.DtReferencia,
				t1.ID,
				t1.RWA,
				t1.FatorRisco,
				t2.Bacen_Exposure_Amount
		;
	quit;

	proc sql;
		CREATE TABLE WORK.REG_AUX AS
			SELECT
				*
			FROM
				WORK.REGULATORIO_FINAL
			WHERE
				CodigoModalidade NOT IN (298,328) OR CodigoModalidade EQ .
		;
	quit;
