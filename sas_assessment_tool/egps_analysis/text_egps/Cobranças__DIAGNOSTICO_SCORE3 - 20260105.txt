CREATE TABLE RISCO_ABERTO AS
SELECT
	A.*
	,intck('day', DT_VENCIMENTO, TODAY()-1) AS DIFF
FROM
	CONNECTION TO PERM (
	SELECT DISTINCT
                 cd_contrato
				 ,sq_contrato_financeiro
				 ,MIN((CASE WHEN ST_PARCELA <> 'Q' OR
								(ST_PARCELA = 'Q' AND
								 DATE(DT_PROCESSAMENTO_ULT_RECEBIMENTO) > '2025-08-11')  THEN NU_PARCELA END)) AS NU_PARCELA

				 ,MIN(DATE((CASE WHEN ST_PARCELA <> 'Q' OR

								 DATE(DT_PROCESSAMENTO_ULT_RECEBIMENTO) > '2025-08-11')  THEN DT_VENCIMENTO END))) AS DT_VENCIMENTO

                 ,SUM(CASE WHEN  ST_PARCELA <> 'Q' OR
								(ST_PARCELA = 'Q' AND
								 DATE(DT_PROCESSAMENTO_ULT_RECEBIMENTO) > '2025-08-11')  THEN VR_PARCELA ELSE 0 END) AS RISCO_ABERTO

                 FROM bv-pdbd-prd.dvry_core.tbparcela

GROUP BY cd_contrato
HAVING sq_contrato_financeiro = MAX(sq_contrato_financeiro)

CREATE TABLE CRUZA_BASES AS
SELECT DISTINCT
A.*
,B.NU_SCORE3
,B.DT_GRAVACAO_SCORE

,C.GH AS GH_NOVO

FROM TB_HISTORICO A
LEFT JOIN CASOS_ANALITICOS B ON

AND A.SQCONTAT = B.SQ_CONTRATO_FINANCEIRO
AND B.DT_GRAVACAO_SCORE < A.DT_DISPONIBILIZACAO

PROC SQL;
CREATE TABLE CRUZA_BASES_V2 AS
SELECT DISTINCT
A.*
,B.NU_SCORE3 AS NUSCORE_ANT
,B.DT_GRAVACAO_SCORE AS DTGRAVACAOSCORE_ANT
,INTCK('DAY', B.DT_GRAVACAO_SCORE, A.DT_GRAVACAO_SCORE) AS DIF_DIAS
,CASE WHEN A.NU_SCORE3 <> A.SCORE3 THEN 1 ELSE 0 END AS DIFERENTE_SCORE

FROM CRUZA_BASES A
LEFT JOIN CASOS_ANALITICOS B ON
A.NR_CONTRATO = B.CD_CONTRATO
AND A.SQCONTAT = B.SQ_CONTRATO_FINANCEIRO
AND B.DT_GRAVACAO_SCORE < A.DT_DISPONIBILIZACAO
AND B.NU_SCORE3 <> A.NU_SCORE3

GROUP BY A.NR_CONTRATO, A.SQCONTAT
HAVING B.DT_GRAVACAO_SCORE = MAX(B.DT_GRAVACAO_SCORE)

;QUIT;

PROC SQL;
CREATE TABLE CRUZA_BASES_V3 AS
SELECT DISTINCT
A.*
,B.DIAS_DIFF + B.QT_ATRASO AS MAX_DIFF_MARCACAO
,CASE WHEN A.DTGRAVACAOSCORE_ANT IS MISSING THEN '01. Primeira Marcação'
      WHEN A.DIFERENTE_SCORE = 1            THEN (CASE WHEN A.DIF_DIAS <= 50    THEN '02. Última marcação inf 50d' ELSE '03. Outro' END)
	  WHEN A.DIFERENTE_SCORE = 0            THEN (CASE WHEN A.DIF_DIAS >  50    THEN '04. Última marcação maior 50d'
                                                     WHEN CALCULATED MAX_DIFF_MARCACAO > 29 THEN '05. Recorrência +29d' ELSE '06. Outro' END)
ELSE '07. Outro' END AS FLAG

FROM CRUZA_BASES_V2 A
LEFT JOIN CONTRATO_DIFERENCA B ON
A.NR_CONTRATO = B.CD_CONTRATO
AND A.SQCONTAT = B.SQ_CONTRATO_FINANCEIRO
AND B.DT_GRAVACAO_SCORE < A.DT_DISPONIBILIZACAO

GROUP BY A.NR_CONTRATO, A.SQCONTAT
HAVING B.DIAS_DIFF = MAX(B.DIAS_DIFF)

;QUIT;
