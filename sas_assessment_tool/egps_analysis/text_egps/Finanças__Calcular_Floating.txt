proc sql;

Create table work.rural_soma as
select sum(total_acionista) as total_acionista, 1 as agrup
from work.rural;

Create table work.float_sint as
select dtapuracaosaldo, cdbanco, sum(vrsaldototal) as vrsaldototal, 21 as per_compulsorio, 25 as per_rural, max(CDI) as CDI
from work.FLOATING
group by dtapuracaosaldo, cdbanco
order by dtapuracaosaldo, cdbanco;



Create table work.float_perc as
select *, case when vrsaldototal > 500000000 then (vrsaldototal - ((vrsaldototal - 500000000) * per_compulsorio/100) ) * ((1+CDI)**(1/252)-1)
		  else vrsaldototal * ((1+CDI)**(1/252)-1)
		  end as CalcFloatnovo,
		  case when vrsaldototal > 500000000 then (vrsaldototal - ((vrsaldototal - 500000000) * per_compulsorio/100) ) / vrsaldototal
		  else vrsaldototal / vrsaldototal
		  end as perc_dia
from work.float_sint;

quit;

proc sql;
create table work.floating_calc as
select f.DtApuracaoSaldo, f.cdbanco, f.cdpessoa, (perc_dia * f.VrSaldoTotal)*((1+f.CDI)**(1/252)-1) as CalcFloat, 1 as agrup, f.VrSaldoTotal
from work.FLOATING f
inner join work.float_perc p
on p.dtapuracaosaldo = f.dtapuracaosaldo
and p.cdbanco = f.cdbanco;

create table work.floating_soma as
select cdbanco, sum(CalcFloat) as tot_float, 1 as agrup
from floating_calc
group by cdbanco;

create table work.floating_atual as
select DtApuracaoSaldo, f.cdbanco, cdpessoa, CalcFloat, tot_float, CalcFloat/tot_float as percfloat, total_acionista, total_acionista*CalcFloat/tot_float as desconto_rural, CalcFloat - COALESCE((total_acionista*CalcFloat/tot_float),0) as calcfloat_atual, f.VrSaldoTotal
from  work.floating_calc f
inner join work.floating_soma s
on s.agrup = f.agrup
and s.cdbanco = f.cdbanco
left join work.rural_soma r
on r.agrup = s.agrup
and f.cdbanco = 161;

quit;

PROC SORT nodupkey DATA=PLANCTRL.TB001_SEGMENTACAO out=Work.Segmentado;
 BY cdcliente ;
 Run;

proc sql;
Create table work.floating_segmentado as
select distinct f.DtApuracaoSaldo, f.cdbanco, f.cdpessoa, f.calcfloat_atual as CalcFloat , f.VrSaldoTotal, CdSegmentoMercado, nmSegmentoMercado, CdGrupoComercial, nmGrupoComercial,
CdDivisaoComercial, nmDivisaoComercial, cdpessoafuncionario as CdOfficer, nmfuncionario as nmOfficer,
CdGrupoEconomico, nmGrupoEconomico
from work.floating_atual as f
left join Work.Segmentado as s
on s.cdcliente = f.cdpessoa
where f.cdpessoa NE .;

update work.floating_segmentado
set  CdSegmentoMercado = 15,
     nmSegmentoMercado = 'DIGITAL'
where cdpessoa in (-1, -2);

update work.floating_segmentado

update work.floating_segmentado
set  cdbanco = 1
where cdbanco = 161;

update work.floating_segmentado
set  cdbanco = 1455
where cdbanco = 436;

quit;
