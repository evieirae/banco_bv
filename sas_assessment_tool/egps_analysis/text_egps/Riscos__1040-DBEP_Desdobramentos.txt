PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.tblDesdobramentos_Projuris  AS
SELECT

 Cd_Causa as Cd_Causa
,SERIAL_IMPORT as SERIAL_IMPORT
,Pasta as Pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdCausaUnidade as CdCausaUnidade
,CdSistema as CdSistema
,CdUnidadeExyon as CdUnidadeExyon
,NumProcesso as NumProcesso
,NumProcessoOriginal as NumProcessoOriginal
,IdProcessoDesdobramento as IdProcessoDesdobramento
,DtCdastroDesdobramento as DtCdastroDesdobramento
,DtDesdobramento as DtDesdobramento
,UltimaModificacao as UltimaModificacao
,SerialId as SerialId
,Capturar as Capturar
,DataInicio as DataInicio
,UF as UF
,municipio as municipio
,DenVara as DenVara
,OrgaoTribunal as OrgaoTribunal
,Esfera as Esfera
,AreaJud as AreaJud
,AcaoJud as AcaoJud
,ValorPleito as ValorPleito
,ValorPedido as ValorPedido
,ValorProvisao as ValorProvisao
,DataBxCont as DataBxCont
,DataEncerr as DataEncerr
,TipoEncerr as TipoEncerr
,MotivoEncerr as MotivoEncerr
,CdCausaPrin as CdCausaPrin
,NmCausaPrin as NmCausaPrin
,DtInicio as DtInicio
,VlPleito as VlPleito
,VlPedido as VlPedido
,VlProvisao as VlProvisao
,NmStatusProcesso as NmStatusProcesso
,DtBxCont as DtBxCont
,DtEncerr as DtEncerr
,MotEncerr as MotEncerr
,CdTipoAtendimento as CdTipoAtendimento
,AssuntoJudicial as AssuntoJudicial
,Fase as Fase

FROM CONNECTION TO postgres(

 select
case
	when pr.serial_import like ('%INQ%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%BVT%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%CON%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%VAC%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%CTO%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRI%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%TRB%') then (left(pr.SERIAL_IMPORT,10))
		else pr.serial_import end                            as Cd_Causa
,pr.SERIAL_IMPORT 											 as SERIAL_IMPORT
,pr.pasta                               					 as Pasta
,pr.ID_NATUREZA												 as IdNatureza
,pr.NATUREZA_UNIDADE_CUSTOM									 as IdNaturezaUnidadeCustom
,null::numeric						                         as CdCausaUnidade
,'p' 														 as CdSistema
,case
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 2             then 'TRI'
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 1             then 'TBR'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 4             then 'CTO'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 12            then 'CON'
	WHEN pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 13            then 'VAC'
	WHEN pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM in (1 ,7, 11)   then 'INQ'
	WHEN pr.ID_NATUREZA = 27 and pr.NATUREZA_UNIDADE_CUSTOM = 5             then 'BVT'
	WHEN pr.ID_NATUREZA = 32 and pr.NATUREZA_UNIDADE_CUSTOM in (7, 11 , 12) then 'INQ'
	else null end 										      as CdUnidadeExyon

,de.NUMERO					            					  as NumProcesso
,de.NUMERO_ORIGINAL                                           as NumProcessoOriginal
,de.ID_PROCESSO_DESDOBRAMENTO                                 as IdProcessoDesdobramento
,cast(de.Data_Cadastro as date)                               as DtCdastroDesdobramento
,cast(de.Data as date)                                        as DtDesdobramento
,cast(de.ULTIMA_MODIFICACAO as date)                          as UltimaModificacao
,de.SERIAL_ID                                                 as SerialId
,de.capturar                                                  as Capturar
,to_char(cast(pr.data_inicio as DATE),'DD/MM/YYYY')           as DataInicio
,(
select uf.uf from public.D_PROCESSO_DESDOBRAMENTO de
inner join public.LT_UF uf on uf.ID_UF = de.ID_UF
where de.ID_PROCESSO = pr.ID_PROCESSO
and de.ATUAL = 'T'
)                                                             as UF
,lm.municipio
,lj.JURISDICAO                                                as DenVara
,tb.tribunal												  as OrgaoTribunal
,''			                                                  as Esfera
,na.Natureza                                                  as AreaJud
,ta.TIPO_ACAO                                                 as AcaoJud
,cast(pr.VALOR_REFERENCIA as varchar)                         as ValorPleito
,cast(pr.TOTAL_OBJETOS as varchar)                            as ValorPedido
,''                                                           as ValorProvisao
,to_char(cast(pr.Data_Baixa_Provisoria as date),'DD/MM/YYYY') as DataBxCont
,to_char(cast(pr.Data_Encerramento as date),'DD/MM/YYYY')     as DataEncerr
,te.DESCRICAO                                                 as TipoEncerr
,en.Encerramento                                              as MotivoEncerr

,case when de.ATUAL = 'T' then 'SIM'
	  when de.ATUAL = 'F' then 'NAO'
	  else null
	  end	                                 				  as CdCausaPrin

,lo.objeto                                                    as NmCausaPrin
,cast(pr.data_inicio as date)                          		  as DtInicio
,cast(pr.VALOR_REFERENCIA as bigint)                          as VlPleito
,cast(pr.TOTAL_OBJETOS as bigint)                             as VlPedido
,cast(0   as bigint)                                          as VlProvisao


,case when PR.DATA_CADASTRO is null
	   and PR.DATA_PRE_CADASTRO is not null
	   and PR.DATA_ENCERRAMENTO is null 		then 'PRE CADASTRO'
      when PR.DATA_ENCERRAMENTO is not null 	then 'ENCERRADO'
      when PR.STATUS_BAIXA = 'B'
       and PR.DATA_ENCERRAMENTO is null 		then 'BAIXA_PROVISORIA'
  												else 'ATIVA'
  												end 		 as NmStatusProcesso

,case when
	(case when PR.DATA_CADASTRO is null
	   and PR.DATA_PRE_CADASTRO is not null
	   and PR.DATA_ENCERRAMENTO is null 		then 'PRE CADASTRO'
      when PR.DATA_ENCERRAMENTO is not null 	then 'ENCERRADO'
      when PR.STATUS_BAIXA = 'B'
       and PR.DATA_ENCERRAMENTO is null 		then 'BAIXA_PROVISORIA'
  												else 'ATIVA'
  												end 		) = 'ATIVA'
  	  then null
  	  else cast(pr.Data_Baixa_Provisoria as date) end  		 as DtBxCont

,cast(pr.Data_Encerramento as date)                          as DtEncerr
,case
 when pr.Data_Encerramento is null then 'EM CURSO'
 else en.Encerramento end                    				 as MotEncerr
,''															 as CdTipoAtendimento
,AC.ASSUNTO_CUSTOM											 as AssuntoJudicial
,lf.Fase

from       public.M_PROCESSO               pr
left join  public.LT_NATUREZA              na on na.ID_NATUREZA                = pr.ID_NATUREZA
left join  public.D_PROCESSO_DESDOBRAMENTO de on de.ID_PROCESSO                = pr.ID_PROCESSO
left join  public.LT_TIPO_ACAO             ta on ta.ID_TIPO_ACAO               = pr.ID_TIPO_ACAO
inner join public.LT_MUNICIPIO             lm on lm.ID_MUNICIPIO               = de.ID_COMARCA
inner join public.LT_JURISDICAO            lj on lj.ID_JURISDICAO              = de.ID_JURISDICAO
left join  public.LT_TIPO_ENCERRAMENTO     te on te.ID_TIPO_ENCERRAMENTO       = pr.ID_TIPO_ENCERRAMENTO
left join  public.LT_ENCERRAMENTO          en on en.ID_ENCERRAMENTO            = pr.ID_ENCERRAMENTO
left join  public.D_DESDOBRAMENTO_OBJETO  dos on dos.id_processo_desdobramento = de.id_processo_desdobramento
left join  public.LT_TRIBUNAL              tb on tb.id_tribunal                = de.id_tribunal
left join  public.LT_OBJETO                lo on lo.id_objeto                  = dos.id_objeto
left join  public.LT_ASSUNTO_CUSTOM        ac on pr.ID_ASSUNTO_CUSTOM 		   = ac.ID_ASSUNTO_CUSTOM
inner join public.LT_FASE                  lf on lf.id_fase                    = de.id_fase





;

);DISCONNECT FROM postgres;
RUN;
