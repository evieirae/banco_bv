proc sql;
%CONECTADB(server=BI);
create table TbExtTaxaMercado as
select * from connection to dbcon
(
WITH
	TbDataCorte AS
(
	SELECT
		DATE(DtTempo) AS DtReferencia
	FROM DWBI_DIME.TbDimTempoData
	WHERE 1 = 1
		AND DtTempo >= today()-5
		AND DtTempo  < today()
)

,TbParcelas AS
(
	SELECT
		row_number() over (ORDER BY DtTempo) AS QtParcelas
	FROM DWBI_DIME.TbDimTempoData
	WHERE DtTempo BETWEEN '20170101' AND '20170828'
)
,TbTipoTaxaMercadoOrigem AS
(
	SELECT
		NuDiaSemana AS CdTipoTaxaMercadoOrigem
	FROM DWBI_DIME.TbDimTempoData
	WHERE DtTempo BETWEEN '20170101' AND '20170106'
		AND NuDiaSemana IN (1,5,6)
)

,TbTaxaMercadoDia AS
(
	SELECT
		 TipoTaxaMercadoOrigem.CdTipoTaxaMercadoOrigem
		,DataCorte.DtReferencia
		,Parcelas.QtParcelas
		,TaxaMercado.PcPresente
	FROM TbDataCorte DataCorte

		INNER JOIN TbParcelas Parcelas
			ON 1=1

		INNER JOIN TbTipoTaxaMercadoOrigem TipoTaxaMercadoOrigem
			ON 1=1

		LEFT JOIN DWBI_DTWH.TbTaxaMercado TaxaMercado
			ON  DataCorte.DtReferencia BETWEEN TaxaMercado.DtInicioVigencia AND coalesce(TaxaMercado.DtFimVigencia,today())
			AND Parcelas.QtParcelas BETWEEN TaxaMercado.PzContratoInicio AND TaxaMercado.PzContratoFim
			AND TipoTaxaMercadoOrigem.CdTipoTaxaMercadoOrigem = TaxaMercado.CdTipoTaxaMercadoOrigem

	WHERE TipoTaxaMercadoOrigem.CdTipoTaxaMercadoOrigem in (1,5,6)
)
,TbTaxaMercadoMax AS
(
	SELECT
		 CdTipoTaxaMercadoOrigem
		,DtReferencia
		,max(QtParcelas) AS QtParcelas
	FROM TbTaxaMercadoDia
	WHERE PcPresente IS NOT NULL
	GROUP BY
		 CdTipoTaxaMercadoOrigem
		,DtReferencia
)

,TbTaxaMercadoMaxPresente AS
(
	SELECT
		 Dia.CdTipoTaxaMercadoOrigem
		,Dia.DtReferencia
		,Dia.QtParcelas
		,Dia.PcPresente
	FROM TbTaxaMercadoDia Dia

		INNER JOIN TbTaxaMercadoMax Max
			ON  Dia.CdTipoTaxaMercadoOrigem = Max.CdTipoTaxaMercadoOrigem
			AND Dia.DtReferencia = Max.DtReferencia
			AND Dia.QtParcelas = Max.QtParcelas
)

,TbTaxaMercado AS
(
SELECT
	 Dia.CdTipoTaxaMercadoOrigem
	,Dia.DtReferencia
	,Dia.QtParcelas
	,CASE
		WHEN Dia.QtParcelas >= 49 AND Dia.DtReferencia <= '20200313' AND Dia.CdTipoTaxaMercadoOrigem <> 6 THEN 0.06675
		WHEN Dia.QtParcelas >= 37 AND Dia.DtReferencia <= '20200313' AND Dia.CdTipoTaxaMercadoOrigem <> 6 THEN 0.06094
		WHEN Dia.QtParcelas >= 25 AND Dia.DtReferencia <= '20200313' AND Dia.CdTipoTaxaMercadoOrigem <> 6 THEN 0.05982
		WHEN Dia.QtParcelas >= 13 AND Dia.DtReferencia <= '20200313' AND Dia.CdTipoTaxaMercadoOrigem <> 6 THEN 0.06401
	 ELSE 0 END AS TxTirAgravamentoLA
	,coalesce(Dia.PcPresente,Max.PcPresente) AS PcPresente
FROM TbTaxaMercadoDia Dia
	LEFT JOIN TbTaxaMercadoMaxPresente Max
		ON  Dia.CdTipoTaxaMercadoOrigem = Max.CdTipoTaxaMercadoOrigem
		AND Dia.DtReferencia = Max.DtReferencia
		AND Dia.PcPresente IS NULL
ORDER BY
	 Dia.CdTipoTaxaMercadoOrigem
	,Dia.DtReferencia
	,Dia.QtParcelas
)

SELECT
	 TaxaMercado.CdTipoTaxaMercadoOrigem
	,TaxaMercado.DtReferencia
	,TaxaMercado.QtParcelas
	,max(TaxaMercado.PcPresente) AS TxTirDivulgada
	,max(Subsidio.PcPresente) AS Subsidio
	,max(TaxaMercado.TxTirAgravamentoLA) AS TxTirAgravamentoLA

	,max(TaxaMercado.PcPresente)-
	 max(Subsidio.PcPresente)+
	 max(TaxaMercado.TxTirAgravamentoLA) as TxTirContabil

FROM TbTaxaMercado TaxaMercado
	left join TbTaxaMercado Subsidio
		on  TaxaMercado.DtReferencia = Subsidio.DtReferencia
		and TaxaMercado.QtParcelas = Subsidio.QtParcelas
		and Subsidio.CdTipoTaxaMercadoOrigem = 5
where TaxaMercado.CdTipoTaxaMercadoOrigem in (1,6)
GROUP BY
	 TaxaMercado.CdTipoTaxaMercadoOrigem
	,TaxaMercado.DtReferencia
	,TaxaMercado.QtParcelas
)
;run;

data bases.TbDimTaxaMercado;
update
	bases.TbDimTaxaMercado
	TbExtTaxaMercado;
by CdTipoTaxaMercadoOrigem DtReferencia QtParcelas;
run;

proc sql;
create table TbTirContabilPos as
select
	UnPivot.DtTir as DtInicio format date9.
,case when Max.DtTirMax is null then UnPivot.DtTir else today() end as DtFim format date9.
,case
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	6	then	 1
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	12	then	 7
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	18	then	13
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	24	then	19
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	30	then	25
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	36	then	31
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	42	then	37
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	48	then	43
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	60	then	49
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	72	then	61
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	84	then	73
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =   	96	then	85
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =     120	then	97
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =     180	then   121
	when input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) =     240	then   181
	end as PzInicio
,input(compress(tranwrd(UnPivot.TpValor,'Tir',''),'" ',''),10.0) as PzFim
,coalesce(UnPivot.PcPresente,0)*10 as TxTirContabil format 18.4
from TbTirContabilPosUnpivot UnPivot
left join TbTirContabilPosMax Max
	on UnPivot.DtTir = Max.DtTirMax
order by 1,2,3,4
;run;

proc sql;
create table TbTirContabilPosDataPrazo as
select
	DataPrazo.DtReferencia
,DataPrazo.QtParcelas
,TirContabilPos.TxTirContabil
from TbDataPrazo DataPrazo
left join TbTirContabilPos TirContabilPos
	on  DataPrazo.DtReferencia between TirContabilPos.DtInicio and TirContabilPos.DtFim
	and DataPrazo.QtParcelas between TirContabilPos.PzInicio and TirContabilPos.PzFim
where TirContabilPos.TxTirContabil > 0
;run;

proc sql;
create table TbTaxaMercadoPos as
select
	 DimTaxaMercado.CdTipoTaxaMercadoOrigem
	,DimTaxaMercado.DtReferencia
	,DimTaxaMercado.QtParcelas
	,DimTaxaMercado.TxTirDivulgada
	,DimTaxaMercado.Subsidio
	,DimTaxaMercado.TxTirAgravamentoLA
	,coalesce(TirContabilPosDataPrazo.TxTirContabil,DimTaxaMercado.TxTirContabil) as TxTirContabil
from TbTirContabilPosDataPrazo TirContabilPosDataPrazo
left join bases.TbDimTaxaMercado DimTaxaMercado
	on  TirContabilPosDataPrazo.DtReferencia = DimTaxaMercado.DtReferencia
	and TirContabilPosDataPrazo.QtParcelas = DimTaxaMercado.QtParcelas
	and DimTaxaMercado.CdTipoTaxaMercadoOrigem = 6
;run;

data bases.TbDimTaxaMercado;

	bases.TbDimTaxaMercado
	TbTaxaMercadoPos;
by CdTipoTaxaMercadoOrigem DtReferencia QtParcelas;
run;
