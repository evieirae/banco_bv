proc sql;
	create table compara_filtro_can as
	select n,
	emissora_chave1, emissora_chave2,
	tipo_oper1, tipo_oper2,
	coalesce(garantia_firme_vol_prop1,0) as garantia_firme1,
	coalesce(garantia_firme_vol_prop2,0) as garantia_firme2,
	coalesce(partic_bv_op1,0) as partic_bv1,
	coalesce(partic_bv_op2,0) as partic_bv2
	from compara;
	create table compara_filtro_can as
	select sum(partic_bv2-partic_bv1) as 'Emissão'n,
	sum(garantia_firme2-garantia_firme1) as 'Garantia'n,
	sum(case when ((case when (garantia_firme2-garantia_firme1)=0 then 0 else 1 end)
				+(case when (partic_bv2-partic_bv1)=0 then 0 else 1 end))>= 1 then 1 else 0 end) as 'Qtd de Op'n
	from compara_filtro_can;
	quit;

max_n=max(n);

	data title; title = "Campos com alguma diferença entre o histórico e os novos dados,

	%macro loop;

	%do x=1 %to &max_n. %by 1;

		data compara_&x.;

		where n=&x.;
		run;

		proc transpose data=compara_&x. out=compara_&x.t;
		var prazo_anos1 prazo_anos2 prazo_anos
		mes_mandato1 mes_mandato2 mes_mandato
		emissora1 emissora2 emissora
		tipo_oper1 tipo_oper2 tipo_oper
		rating_cli1 rating_cli2 rating_cli
		rating_oper1 rating_oper2 rating_oper
		clausulas_mkt_flex1 clausulas_mkt_flex2 clausulas_mkt_flex
		clausulas_macc1 clausulas_macc2 clausulas_macc
		prev_liquidacao1 prev_liquidacao2 prev_liquidacao
		total_emissao1 total_emissao2 total_emissao
		perct_partic_bv_op1 perct_partic_bv_op2 perct_partic_bv_op
		partic_bv_op1 partic_bv_op2 partic_bv_op
		garantia_firme_vol_prop1 garantia_firme_vol_prop2 garantia_firme_vol_prop
		garantia_firme_vol_aprov1 garantia_firme_vol_aprov2 garantia_firme_vol_aprov
		garantia_firme_vol_naprov1 garantia_firme_vol_naprov2 garantia_firme_vol_naprov;
		run;

		proc sql;
		create table diferente_&x. as
		select monotonic() as m, *
		from compara_&x.t (where=(COL1='DIFERENTE'));
		quit;

		proc sql;
		create table auxiliar_&x. as
		select count(*) as t
		from diferente_&x.;
		quit;

		data _null_;
		set auxiliar_&x.;
		call symput('t',t);
		run;

		%if &t.=0 %then %do;

	data title; title = "Sem diferenças, todos os campos são iguais"; run;
	proc report data=title nowindows noheader nocenter
	  style(column) = { background=no };
	run;

		%end;
		%else %do;

		data _null_;
		set diferente_&x.;
		call symput('max_m',max(m));
		run;

		%macro loop_dentro;

		%do y=1 %to &max_m. %by 1;

			data diferente_&x._&y.;
			set diferente_&x.;
			where m=&y.;
			run;

			data _null_;
			set diferente_&x._&y.;
			call symput('_name_',compress(_name_));
			run;

			proc sql;
			create table output_&x._&y. as
			select 'em andamento' as status,
			emissora2 as emissora,
			&_name_.1, &_name_.2,
			'1 -> historico; 2 -> novo' as legenda
			from compara_&x.;
			quit;

			proc print data=output_&x._&y. noobs;run;

		%end;
		%mend;
		%loop_dentro;

		%end;

	%end;
	%mend;
	%loop;

%end;

proc sql;

	coalesce(garantia_firme_vol_prop1,0) as garantia_firme1,
	coalesce(garantia_firme_vol_prop2,0) as garantia_firme2,
	coalesce(partic_bv_op1,0) as partic_bv1,
	coalesce(partic_bv_op2,0) as partic_bv2
	from compara;
	create table compara_filtro_sus as
	select sum(partic_bv2-partic_bv1) as 'Emissão'n,
	sum(garantia_firme2-garantia_firme1) as 'Garantia'n,
	sum(case when ((case when (garantia_firme2-garantia_firme1)=0 then 0 else 1 end)
				+(case when (partic_bv2-partic_bv1)=0 then 0 else 1 end))>= 1 then 1 else 0 end) as 'Qtd de Op'n
	from compara_filtro_sus;
	quit;

data _null_;

max_n=max(n);
call symput('max_n',max(n));
run;

	data title; title = "Campos com alguma diferença entre o histórico e os novos dados,
	verificar pois serão susbstituidos pelos dados mais recentes na tabela final ao fim desse processo "; run;
	proc report data=title nowindows noheader nocenter
	  style(column) = { background=yellow };
	run;
	data title; title = "*Caso não deseje a substituição altere os dados mais recentes na tabela de input"; run;
	proc report data=title nowindows noheader nocenter
	  style(column) = { background=no };
	run;

	%macro loop;

	%do x=1 %to &max_n. %by 1;

		data compara_&x.;
		set compara;
		where n=&x.;
		run;

		proc transpose data=compara_&x. out=compara_&x.t;
		var prazo_anos1 prazo_anos2 prazo_anos
		mes_mandato1 mes_mandato2 mes_mandato
		emissora1 emissora2 emissora
		tipo_oper1 tipo_oper2 tipo_oper
		rating_cli1 rating_cli2 rating_cli
		rating_oper1 rating_oper2 rating_oper
		clausulas_mkt_flex1 clausulas_mkt_flex2 clausulas_mkt_flex

		total_emissao1 total_emissao2 total_emissao
		perct_partic_bv_op1 perct_partic_bv_op2 perct_partic_bv_op
		partic_bv_op1 partic_bv_op2 partic_bv_op
		garantia_firme_vol_prop1 garantia_firme_vol_prop2 garantia_firme_vol_prop
		garantia_firme_vol_aprov1 garantia_firme_vol_aprov2 garantia_firme_vol_aprov
		garantia_firme_vol_naprov1 garantia_firme_vol_naprov2 garantia_firme_vol_naprov;
		run;

		proc sql;
		create table diferente_&x. as
		select monotonic() as m, *
		from compara_&x.t (where=(COL1='DIFERENTE'));
		quit;

		proc sql;
		create table auxiliar_&x. as
		select count(*) as t
		from diferente_&x.;
		quit;

		data _null_;
		set auxiliar_&x.;
		call symput('t',t);
		run;

		%if &t.=0 %then %do;

	data title; title = "Sem diferenças, todos os campos são iguais"; run;
	proc report data=title nowindows noheader nocenter
	  style(column) = { background=no };
	run;

		%end;
		%else %do;

		data _null_;
		set diferente_&x.;
		call symput('max_m',max(m));
		run;

		%macro loop_dentro;

		%do y=1 %to &max_m. %by 1;

			data diferente_&x._&y.;
			set diferente_&x.;
			where m=&y.;
			run;

			data _null_;
			set diferente_&x._&y.;
			call symput('_name_',compress(_name_));
			run;

			proc sql;
			create table output_&x._&y. as
			select 'em andamento' as status,
			emissora2 as emissora,
			&_name_.1, &_name_.2,
			'1 -> historico; 2 -> novo' as legenda
			from compara_&x.;
			quit;

			proc print data=output_&x._&y. noobs;run;

		%end;
		%mend;
		%loop_dentro;

		%end;

	%end;
	%mend;
	%loop;

%end;

%mend;
