PROC SQL;
   CREATE TABLE WORK.Dedistencias AS
   SELECT t1.DBEncerramento,
          t1.MotivoPadrao,
            (case when t1.MotEncerr in ('ENCERRADO / SEM ÔNUS - DESISTÊNCIA ACORDO BA',
            'ENCERRADO / SEM ÔNUS - DESISTÊNCIA DO(A) AUTOR(A)',
            'ENCERRADO / SEM ÔNUS - EXTINÇÃO', 'ENCERRADO / SEM ÔNUS - AUTOR(A) NÃO COMP À AUDIÊNCIA') then
            '4.Desistência' Else t1.Desfecho_Status end) AS Desfecho,
            (COUNT(t1.NumRegSistContrGest)) AS Num_Acoes
      FROM ACM.MASSIFICADO t1
      WHERE t1.Desfecho_Status NOT = '' AND t1.DBEncerramento >= '1Jan2019'd
      GROUP BY t1.DBEncerramento,
               t1.MotivoPadrao,
               (CALCULATED Desfecho);
QUIT;

%MACRO CalculaData(DataRef=, Limite=, DifMes=);
		%let Mes = %sysfunc(month(&DataRef.));
		%let Ano = %sysfunc(year(&DataRef.));

		
		%if (&Mes. - &DifMes.) > 0 %then %do;
			%let MesOut = (&Mes. - &DifMes.);
			%let AnoOut = &Ano.;
		%end;
		%else %do;
			%let MesOut = %sysevalf(&Mes. - &DifMes. + 12);
			%if &MesOut. = 0 %then %do;
				%let MesOut = 1;
			%end;
			%put &MesOut.;
			%let AnoOut = %sysevalf(&Ano. - 1);
		%end;

		
		%if "&Limite." = "INFERIOR" %then %let DiaOut = 01;
		%else %do;
			%if &MesOut = 01 or
			   &MesOut = 03 or
			   &MesOut = 05 or
			   &MesOut = 07 or
			   &MesOut = 08 or
			   &MesOut = 10 or
			   &MesOut = 12 %then %let DiaOut = 31;
			%else %if &MesOut = 04 or
					&MesOut = 06 or
					&MesOut = 09 or
					&MesOut = 11 %then %let DiaOut = 30;
			%else %if &MesOut = 02 and %sysfunc(mod(&AnoOut,4)) = 0 %then %let DiaOut = 29;
			%else %if &MesOut = 02 and %sysfunc(mod(&AnoOut,4)) ne 0 %then %let DiaOut = 28;
			%else %let DiaOut = .;
		%end;

		%let DtOut = %sysfunc(mdy(&MesOut, &DiaOut, &AnoOut));
		&DtOut.
%MEND;
