CREATE TABLE COLLORI_COMPLETO_V1 AS
SELECT
    A.*,

    

    CASE
        WHEN A.NMPRODUTORISCO = 'LEVES' THEN
            CASE
                WHEN A.CART_2 IN ('01. RENEG_EMERG', '02. RENEG_N_EMERG') THEN '1. Reneg'
                WHEN A.CART_2 = '03. BAD BANK' THEN '2. Bad Bank'
                WHEN A.CLIENTE_50K = 1 THEN '3. Saldo > 50k'
                ELSE '4. Geral'
            END

        WHEN A.NMPRODUTORISCO IN ('CVG_LEVES', 'CVG_PESADOS') THEN
            CASE
                WHEN A.CART_2 IN ('01. RENEG_EMERG', '02. RENEG_N_EMERG') THEN '1. Reneg'
                WHEN A.CLIENTE_50K = 1 THEN '3. Saldo > 50k'
                ELSE '4. Geral'
            END

        ELSE
            CASE
                WHEN A.CART_2 IN ('01. RENEG_EMERG', '02. RENEG_N_EMERG') THEN '1. Reneg'
                WHEN A.CART_2 = '03. BAD BANK' THEN '2. Bad Bank'
                WHEN A.CLIENTE_50K = 1 THEN '3. Saldo > 50k'
                ELSE '4. Geral'
            END
    END AS FLAG_GRUPO

FROM WORK.BASE_ESTUDO_PERFORMANCE_MIX A


;QUIT;
