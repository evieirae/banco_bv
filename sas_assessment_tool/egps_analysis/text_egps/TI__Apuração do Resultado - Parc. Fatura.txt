	%macro exp(anomes_ini, anomes_cart);

			data base_anlt;
			set _null_;

		%do %while (&anomes. <= &anomes_cart.);

		%end;

 	proc sql;

	create table cartao as

           select a.cdcontacartao,

		   sum(a.vrmargemliquida) as margem_liquida
    	from cartao.TBL_RAC_RESUMO_&anomes_cart. a

		group by 1;

	quit;











		proc sql;

			create table base_anlt as

			select a.*,
				   c.margem_liquida as margem_liquida_cartoes
			from base_anlt a
            left join cartao c on a.cdcontacartao = c.cdcontacartao;


		quit;


































	proc sql;

	create table res_contabil_&anomes_cart. as
	select safra,
		   &anomes_cart. as carteira,
		   round(sum(margem_liquida_cartoes)) as margem_liquida
	from base_anlt
	group by 1,2;

	quit;

%mend;
