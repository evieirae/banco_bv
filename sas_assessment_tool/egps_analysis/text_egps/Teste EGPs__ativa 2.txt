CREATE TABLE BASE_CARTEIRA_ATIVA AS
SELECT VWC.*

FROM
    CONNECTION TO DBCON
(Select
	*
From
    DBA.VwTbContrato			AS C
    ,DBA.VwTbContratoFinanceiro AS D
    ,DBA.TbModalidadeProduto	AS MP

WHERE 1=1
AND C.NuContrato   		= D.NuContrato
AND C.SQContratoFinanceiroAtivo	= D.SqContratoFinanceiro
AND C.CdModalidadeProduto = MP.CdModalidadeProduto

AND(
		(c.CdProduto in (12,14)

	   	)
 )
) AS VWC INNER JOIN base_primeira_parcela AS BPP ON BPP.NUCONTRATO = VWC.NUCONTRATO;
QUIT;

CREATE TABLE BASE_CARTEIRA_ATIVA_FINAL AS

FROM base_primeira_parcela AS a

ORDER BY a.DT_BASE
;QUIT;RUN;
