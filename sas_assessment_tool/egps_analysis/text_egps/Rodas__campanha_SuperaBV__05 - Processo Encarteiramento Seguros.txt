PROC SQL;
CREATE TABLE TbEncarteiramentoNI AS
SELECT DISTINCT
A.NuAnoMes,
CdColaborador,
DsLoginN1,
CdParceiroComercial
FROM
BASEGRC.TbMetaNaoisonomica A
WHERE NUANOMES = &Base_Mes.
and DsLoginN1 IS NOT NULL

ORDER BY 1,2,3,4
;RUN;

PROC SQL;
CREATE TABLE BASE_SEGUROS_NI AS
SELECT

A.NUANOMES,
A.NmGrupoCargoColaborador,
a.DsLoginN1,
A.DsLoginN1CartPDV,
B.DsLoginN1 AS DSLoginMetaNI,
COALESCE(B.DsLoginN1, A.DsLoginN1) AS DsLoginN1NovaCart,
a.CdColaborador,
b.CdColaborador as CdColaboradorMetaNI,
coalesce(b.CdColaborador, a.CdColaborador) as CdColaboradorNovaCart,
A.NmGrupoProdutoMeta,
a.cdparceirocomercial,
'VrSPF_VrAP_VrAuto' as TpValor,

nucontrato,
VrSeguros as VrMedida
/*sum(a.VrBVMais) as VrBVMais,*/
/*sum(a.VrCap) as VrCap,*/
/*sum(a.VrDemaisComCompletoMax) as VrAutoTotal*/

FROM TMP.TBPRODUCAO A

LEFT JOIN WORK.TbEncarteiramentoNI B ON B.NUANOMES = A.NUANOMES AND A.CDPARCEIROCOMERCIAL = B.CDPARCEIROCOMERCIAL

WHERE
A.NUANOMES = &Base_Mes.
and A.NmCanalEstruturaComercial = 'Canal Veiculos'
and A.NmGrupoCargoColaborador in ('Gerente Relacionamento Motos', 'Gerente Relacionamento Pesados', 'Gerente Relacionamento Veiculos')
and A.NmGrupoProdutoMeta in ('Leves', 'Motos', 'Pesados', 'CVG Leves', 'CVG Pesados')
/*and a.dsloginn1 = 'arlete.leitao'*/
/*and a.NmAreaEstruturaComercial = 'Canal São Luís'*/
/*AND (a.dsloginn1 in ('arlete.leitao', 'monica.pinheiro') or DsLoginN1CartPDV in ('arlete.leitao', 'monica.pinheiro'))*/

/*group by 1,2,3,4,5*/

order by a.DsLoginN1

;RUN;

PROC SQL;
CREATE TABLE BASE_SEGUROS_NI_01 AS
SELECT

A.NUANOMES,
A.CdColaboradorNovaCart AS CdColaborador,
B.NmGrupoCargo as NmGrupoCargoColaborador,
A.DsLoginN1NovaCart AS DsLoginN1,
A.TpValor,
SUM(A.VrMedida) AS VrMedida

FROM WORK.BASE_SEGUROS_NI A

LEFT JOIN BASES.TBESTRUTURACOMERCIAL B ON B.DSLOGINN1 = A.DSLOGINN1 AND B.NUANOMES = A.NUANOMES

GROUP BY 1,2,3,4,5
order by 1,3,4

;RUN;
