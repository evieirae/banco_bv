PROC SQL;
%ConexaoIQ(server=BI);

CREATE TABLE TBL_GASTOAVISTA AS

SELECT t.*
FROM %IQ
(
    SELECT year(a.DtSolicitacaoAutorizacaoEmsr)*100+month(a.DtSolicitacaoAutorizacaoEmsr)  AS SafraRef,
	       a.CdDimContaCartao,

           CASE WHEN c.NmTipoPlanoVenda LIKE ('%LOJ%')     THEN 'PARCELADO SEM JUROS'
	            WHEN c.NmTipoPlanoVenda LIKE ('%VISTA%')   THEN 'COMPRA A VISTA'
	            WHEN c.NmTipoPlanoVenda LIKE ('%EMISS%')   THEN 'PARCELADO EMISSOR'
	            ELSE 'OUTROS'
            END AS NmTipoPlanoVenda,
           sum(case when a.VrAutorizacao >= 100000000 then 0 else a.VrAutorizacao*c.QtParcelas end) AS QtParcelas_Pond,
           sum(case when a.VrAutorizacao >= 100000000 then 0 else a.VrAutorizacao end)              AS VrAutorizacao

	FROM DWBI_DMOC.TbFatoSolicitacaoAutorizacao a

	INNER JOIN DWBI_DIME.TbDimContaCartao b
	ON a.cddimcontacartaoversao = b.cddimcontacartaoversao

	INNER JOIN DWBI_DTWH.TbTipoPlanoVenda c
	ON a.CdTipoPlanoVenda = c.CdTipoPlanoVenda

	WHERE (a.DtSolicitacaoAutorizacaoEmsr BETWEEN &Dt_Ref_Inicial. AND &Dt_Ref_Final.) AND a.cdrespostaautorizacao = 1

	GROUP BY year(a.DtSolicitacaoAutorizacaoEmsr)*100+month(a.DtSolicitacaoAutorizacaoEmsr),
			 a.CdDimContaCartao,
		     NmTipoPlanoVenda
) AS t
;
QUIT;

PROC SORT DATA = TBL_GASTOAVISTA; BY SafraRef CdDimContaCartao; RUN;
PROC TRANSPOSE DATA=TBL_GASTOAVISTA OUT=TBL_GASTOAVISTA_2 (DROP=_LABEL_ _NAME_);
	BY SafraRef CdDimContaCartao;
	VAR QtParcelas_Pond;
	ID NmTipoPlanoVenda;
RUN;

RUN;
