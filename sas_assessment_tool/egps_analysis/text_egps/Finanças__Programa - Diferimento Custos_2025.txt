proc sql;

CREATE TABLE DesligadoMesesAnteriores AS
SELECT
	NmSubEstruturaComercial
	,NmAreaEstruturaComercial
	,CdColaborador
	,NmColaborador
	,DsLogin
	,DsEmail
	,NmCargoColaborador
	,CdEstadoColaborador
	,NmEstadoColaborador
	,DtDesligamentoColaborador
	,DATEPART(DtDesligamentoColaborador) as DtDesligamentoColaborador2 FORMAT DATE9.

FROM connection to dbcon
	(
		SELECT * FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
		WHERE today() BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
		AND YEAR(HierarquiaComercial.DtInicioVigencia) BETWEEN YEAR(today())-1 AND YEAR(today())
		AND YEAR(DtDesligamentoColaborador) BETWEEN YEAR(today())-1 AND YEAR(today())
		AND HierarquiaComercial.CdEstadoColaborador in (4,10) AND CdEstruturaComercial = 8

		ORDER BY DtDesligamentoColaborador desc
	)
;
run;
quit;

	,YEAR(DtDesligamentoColaborador2)*100 + MONTH(DtDesligamentoColaborador2) AS SafraDesligados

FROM DesligadoMesesAnteriores
;RUN;

create table Base_Margem_GR_Prod_Deslig as
select
NuAnoMes,
IdMatricula,

Group by 1,2,3
;quit;

Proc sql;
create table Base_Ratear_Desligados (Where =(ComissaoRateada NE .)) as
select
a.*,
c.Total_MC as Ratear,
b.Custo_Comissao,
(a.VrMargemContribuicao/c.Total_MC) as Perc,
calculated Perc * b.Custo_Comissao as ComissaoRateada
from TbMCRealizada_Desligados as a
	left join WORK.Custo_Comissao_Desligados as b on a.IdMatricula = b.Id_RH and a.Verba = b.Produto 
	left join Base_Margem_GR_Prod_Deslig as c on a.IdMatricula = c.IdMatricula and a.Verba = c.Produto

;quit;

Proc sql;
create table Validar_Rateio_Ativos as
select
"Ativos" as Base
,a.NuAnoMes
,a.Verba
,a.IdMatricula
,a.Tipo_Operador
,AVG(b.Custo_Comissao) as Comissao_RH Format COMMAX10.2
,Sum(a.ComissaoRateada) as Comissao_Rateada Format COMMAX10.2
from Base_Ratear as a
	left join CUSTO_COMISSAO as b on a.IdMatricula = b.Id_RH and a.Verba = b.Produto
Group by 1,2,3,4,5
;quit;

Proc sql;
create table Validar_Rateio_Desligados as
select
"Desligados" as Base
,a.NuAnoMes
,a.Verba
,a.IdMatricula
,a.Tipo_Operador
,AVG(b.Custo_Comissao) as Comissao_RH Format COMMAX10.2
,Sum(a.ComissaoRateada) as Comissao_Rateada Format COMMAX10.2
from Base_Ratear_Desligados as a
	left join Custo_Comissao_Desligados as b on a.IdMatricula = b.Id_RH and a.Verba = b.Produto
Group by 1,2,3,4,5
;quit;

Data Base_Validar_Rateio;
Set Validar_Rateio_Desligados
	Validar_Rateio_Ativos
;Run;

Proc sql;
create table Base_Validar_Rateio_Saida as
select
Base
,NuAnoMes
,Verba
,Sum(Comissao_RH) as Comissao_RH Format COMMAX10.2
,Sum(Comissao_Rateada) as Comissao_Rateada Format COMMAX10.2
,Calculated Comissao_RH - Calculated Comissao_Rateada as Diferença  Format COMMAX10.2
,Case When (Calculated Diferença > 0.01 or Calculated Diferença < -0.01) Then "Verificar" Else "Batido" End as Status_Diferença
from Base_Validar_Rateio as a
	left join Custo_Comissao_Desligados as b on a.IdMatricula = b.Id_RH and a.Verba = b.Produto
Group by 1,2,3
;quit;
