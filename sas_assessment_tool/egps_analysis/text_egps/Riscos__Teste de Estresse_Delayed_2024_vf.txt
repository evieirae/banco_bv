%macro gera_projecao_modelo(simulacao, dif_resp, var_resp, coef_resp, 	  desv_resp,
							 				     	 var1, 	   coef1, 	  desv1, l1,
							 				     	 var2, 	   coef2, 	  desv2, l2,

									, &&coef&i. * a&i.
								 %end;
								 %end; );

			
			if anomes < &anomes_ini. then proj_delta_maisdesv = .;
			else if anomes = &anomes_ini. then proj_delta_maisdesv = delta_&var_resp.;
			else proj_delta_maisdesv = sum(
											%if %sysevalf(&coef_resp. >= 0 ) %then %do;
											(&coef_resp. + &desv_resp.) * proj_delta_maisdesv
											%end;
											%else %do;
											(&coef_resp. - &desv_resp.) * proj_delta_maisdesv
											%end;

											%do i = 1 %to 4;
											%if %eval( &&l&i. >= 0 ) %then %do;
												%if %sysevalf( &&coef&i. > 0 ) %then %do;
												,(&&coef&i. + &&desv&i.) * a&i.
												%end;
												%else %do;

											(&coef_resp. - &desv_resp.) * proj_delta_menosdesv
											%end;
											%else %do;
											(&coef_resp. + &desv_resp.) * proj_delta_menosdesv

											%do i = 1 %to 4;
											%if %eval( &&l&i. >= 0 ) %then %do;
												%if %sysevalf( &&coef&i. > 0 ) %then %do;
												,(&&coef&i. - &&desv&i.) * a&i.
												%end;
												%else %do;
												,(&&coef&i. + &&desv&i.) * a&i.
												%end;
											%end;
											%end; );

		drop %do i = 1 %to 4;
			 %if %eval( &&l&i. >= 0 ) %then %do; a&i. %end;
			 %end; ;

			if anomes < &anomes_ini. then proj_media = .;
			else if anomes = &anomes_ini. then proj_media = media_&var_resp.;
			else proj_media = %if &dif_resp. = S %then %do; proj_delta; %end;
							  %else %if &dif_resp. = P %then %do; proj_media * (1 + proj_delta); %end;
							  %else %do; proj_media + proj_delta; %end;

			
			if anomes < &anomes_ini. then proj_media_maisdesv = .;
			else if anomes = &anomes_ini. then proj_media_maisdesv = media_&var_resp.;
			else proj_media_maisdesv = %if &dif_resp. = S %then %do; proj_delta_maisdesv; %end;
									   %else %if &dif_resp. = P %then %do; proj_media_maisdesv * (1 + proj_delta_maisdesv); %end;
							  		   %else %do; proj_media_maisdesv + proj_delta_maisdesv; %end;

			
			if anomes < &anomes_ini. then proj_media_menosdesv = .;
			else if anomes = &anomes_ini. then proj_media_menosdesv = media_&var_resp.;
			else proj_media_menosdesv = %if &dif_resp. = S %then %do; proj_delta_menosdesv; %end;
									    %else %if &dif_resp. = P %then %do; proj_media_menosdesv * (1 + proj_delta_menosdesv); %end;
							  		    %else %do; proj_media_menosdesv + proj_delta_menosdesv; %end;

	run;

%mend;











%macro escolhe_projecao_modelo(simulacao, var_resp, media_resp, escolha);

	%let anomes_ini = %eval( &ano. * 100 + 1 );

	data proj_mode;
	set dados_mode;

		
		%if &escolha. = mais %then %do; media = proj_media_maisdesv; %end;
		%else %if &escolha. = menos %then %do; media = proj_media_menosdesv; %end;
		%else %do; media = proj_media; %end;

		if anomes < &anomes_ini. then projecao_&var_resp. = &var_resp.;

		keep anomes &var_resp. media projecao_&var_resp.;

	run;

	data proj_mode;
	set proj_mode;

		a1 = lag1(projecao_&var_resp.);
		a2 = lag2(projecao_&var_resp.);
		a3 = lag3(projecao_&var_resp.);
		a4 = lag4(projecao_&var_resp.);
		a5 = lag5(projecao_&var_resp.);

		
		
		if anomes = &anomes_ini. then do;
			call symput("v1", a1);
			call symput("v2", a2);
			call symput("v3", a3);
			call symput("v4", a4);
			call symput("v5", a5);
		end;

		if anomes >= &anomes_ini. then projecao_&var_resp. = .;

		if anomes >= &anomes_ini.;

		drop a1 a2 a3 a4 a5;

	run;

	data projecao;
	set proj_mode;

		
		retain a1-a5 (&v1., &v2., &v3., &v4., &v5.);

		%if &media_resp. = MM3 %then %do; &simulacao._mode_&var_resp. = 3 * media - sum(a1, a2); %end;
	 	%else %if &media_resp. = MM6 %then %do; &simulacao._mode_&var_resp. = 6 * media - sum(a1, a2, a3, a4, a5); %end;
	 	%else %do; &simulacao._mode_&var_resp. = media; %end;

		
		
		a5 = a4;
		a4 = a3;
		a3 = a2;
		a2 = a1;
		a1 = &simulacao._mode_&var_resp.;

		keep anomes &simulacao._mode_&var_resp.;

	run;

%mend;

%macro sazonalidade(entrada, base, vari,dt_ini);

	                              lineattrs=(color=red);

	                               markerattrs=(symbol='circle')
	                               lineattrs=(color=blue);

	where not missing(anomes);
	run;

	coalesce(b.z_&vari.,0) as &vari.
	from &entrada. as a
	left join &base. as b
	on a.anomes=b.anomes;
	quit;

%mend;

%macro rateio_tvm;

	data rateio_tvm;
	set _NULL_;

	run;

	
	%let data_bu = %eval( &ultima_bu. - 100 + 1 );
	%do %while (&data_bu. <= &ultima_bu.);

		%if ( %sysfunc(substr(&data_bu., 5, 2)) = 13 ) %then %let data_bu = %eval(&data_bu. + 88);
		%put &data_bu.;

		proc sql;

			create table aux_tvm as
			select &data_bu. as anomes,
				   sum(VrAjustePrudencial) as Ajuste_Prudencial format commax30.,
				   sum(VrImpairment) as Impairment format commax30.

			from _miscred.base_unica_atacado_&data_bu.
			where DsGrupoProduto in ('TVM');

		quit;

		data rateio_tvm;
		set rateio_tvm aux_tvm;

			pct_aj_prud = Ajuste_Prudencial / sum( Ajuste_Prudencial, Impairment );
			pct_impairment = Impairment / sum( Ajuste_Prudencial, Impairment );

		run;

		%let data_bu = %eval(&data_bu. + 1);

	%end;

	
	proc sql;

		create table rateio_tvm_pct as
		select mean( pct_aj_prud ) as pct_aj_prud,
			   mean( pct_impairment ) as pct_impairment

		from rateio_tvm;

	quit;

%mend;

	select pct_aj_prud
	into: pct_aj_prud
	from rateio_tvm_pct;

	from rateio_tvm_pct;

AS SELECT a.*,

%macro cria_meses();

			mes = &i.;

		%end;

	run;

set &simulacao._prov_trab;

	
	if mes <= &meses_desl. then do;
		desl_mes = &HC_Tot. * &perc_desl. / &meses_desl.;
		redu_cust_demi = - mes * desl_mes * &desp_colb.; 
		aume_cust_demi = desl_mes * &cres_colb.; 
	end;
	else do;
		desl_mes = 0;
		redu_cust_demi = - &HC_Tot. * &perc_desl. * &desp_colb.;
		aume_cust_demi = 0;
	end;

	delta_demi = redu_cust_demi + aume_cust_demi;

	aume_acao_trab = desl_mes * &perc_trab. * &pu_trab.; 

run;
