  SELECT

    DATE(DtGeracao) AS DtGeracao,

    b.cod_asses,

    CASE
      WHEN B.COD_ASSES IN (220,101,497,1497,1588,1618,1591,1584,1654,1660,204,1662,1581,14,1659,1590,1651,258,1575,270,186,2022,1657,133,1,1336,2041,2040,2021,1625,2005,2064,2065,1652,1656,2066,2067) THEN 'RODAS'

    END AS Fase,

    CASE
      WHEN B.COD_ASSES IN (1625,2064,1652,2066,1618) THEN 'MENDES'
      WHEN B.COD_ASSES IN (2005,2065,1656,2067,204,1657,1336,1581,2036,1599,2049,2048,2031,1592,1593) THEN 'PASCHOALOTTO'
      WHEN B.COD_ASSES IN (220,1651,14,1590,2055,1650,1528,1639,2033,1551,1640,1491,1644,1641,1552) THEN 'BELLINATI'
      WHEN B.COD_ASSES IN (258,1654,1588,1497) THEN 'LM ANGELIS'
      WHEN B.COD_ASSES IN (270,133,1660,2038,1584) THEN 'SCHULZE'
      WHEN B.COD_ASSES IN (186,1,1662,1591,1663,2054,1559,1634,1648,1649,1530,1645,1642,1524) THEN 'TOLEDO'

      WHEN B.COD_ASSES IN (1627) THEN 'BAIXADO'

      WHEN B.COD_ASSES IN (497,1659,1575,101,1583) THEN 'PASQUALI'
      WHEN B.COD_ASSES IN (1666,2016,1532,739) THEN 'RAMOS'
      ELSE 'VERIFICAR'
    END AS Assessoria,

    CASE
      WHEN B.COD_ASSES IN (1625,2064,2005,2065) THEN 'ATIVA'
      WHEN B.COD_ASSES IN (1652,1656,2066,2067) THEN 'ATIVA RENEG'
      WHEN B.COD_ASSES IN (220,258,204,270,186,2021,497) THEN 'ADMINISTRATIVA'
      WHEN B.COD_ASSES IN (1651,1654,1657,1660,1662,2040,1659) THEN 'ADMINISTRATIVA RENEG'

      WHEN B.COD_ASSES IN (2054,2055) THEN 'PRE CONTENCIOSO'

      WHEN B.COD_ASSES IN (1551,1640,1530,1532) THEN 'RENEG CONSIGNADO'

      WHEN B.COD_ASSES IN (1641,1642) THEN 'CONTENCIOSO VU'
      WHEN B.COD_ASSES IN (1552,1524,739) THEN 'DESAVERBADOS'
      WHEN B.COD_ASSES IN (1583) THEN 'RECORRENTE'
      ELSE 'VERIFICAR'
    END AS Carteira,

    CASE
      WHEN a.StNegociacao = 'C' THEN 'Cancelado'
      ELSE 'Negociado'
    END AS StatusNegociacao,

    COUNT(1) AS totalAssessoria,

    SUM(CASE WHEN c.flbolepix IS NULL OR c.flbolepix = '0' OR c.flbolepix = 'N' THEN 1 ELSE 0 END) AS totalBoletoLegado,
    SUM(CASE WHEN c.flbolepix = 'S' THEN 1 ELSE 0 END) AS totalBoletoHibrido

  FROM
    `bv-pdbd-prd.trusted_dbcontencioso.tbnegociacaocontrato` a

  INNER JOIN
    `bv-pdbd-prd.trusted_scc_fin.saj_processo` b
    ON SAFE_CAST(a.NuContratoGestao AS INT64) = SAFE_CAST(b.NuContratoGestao AS INT64)
    AND a.SqContratoFinanceiro = b.SqContratoFinanceiro
    AND b.num_processo = SAFE_CAST(a.nuprocesso AS INT64)

  LEFT JOIN
    `bv-pdbd-prd.trusted_dbgopr_goprregistroparcial.tbsolicitacaoregistroparcial` c
    ON SAFE_CAST(a.NuContratoGestao AS INT64) = SAFE_CAST(c.nucontrato AS INT64)
    AND a.SqContratoFinanceiro = c.sqcontratofinanceiro
    AND a.cdnossonumero = c.cdnossonumero

  WHERE
    StNegociacao NOT IN ('PA','R')

    AND DATE(DtGeracao) BETWEEN '2025-01-01' AND '2026-01-05'

  GROUP BY
    DATE(DtGeracao),
    b.cod_asses,
    Fase,
    Assessoria,
    StatusNegociacao
