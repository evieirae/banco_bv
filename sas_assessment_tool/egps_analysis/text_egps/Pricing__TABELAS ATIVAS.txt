%include '/infogrc/utils/IQ.sas';
%CRIAVARDB;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE DELTA_PPR AS SELECT * FROM CONNECTION TO DbCon(
SELECT
CdTabelaFinanciamento,
CdParametroVariacaoTabelaFnnco AS DELTA_PPR
FROM DbCor..TbTabelaFinanciamentoPrmtoVrco v where v.FlAtivo = 'S'
ORDER BY v.DtInclusao);
RUN;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE TABELAS_PA AS SELECT * FROM CONNECTION TO DbCon(
SELECT
CdTabelaFinanciamento,
CdParametroVariacaoRangeCPF AS CD_LIMITE_PA
FROM dbcor..TbTabelaFinanciamentoRangeCPF v  where v.FlAtivo = 'S'
ORDER BY v.DtInclusao);
RUN;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE VIGENCIAPA AS SELECT * FROM CONNECTION TO DbCon(
SELECT
CdParametroVariacaoRangeCPF,
DtFimVigencia as DtFimVigenciaPA
FROM DBCOR.dbo.TbParametroVariacaoRangeCPF
);
RUN;

PROC SQL;
CREATE TABLE LIMITE_PA AS
SELECT

A.CdTabelaFinanciamento,
A.CD_LIMITE_PA,
b.DtFimVigenciaPA

FROM WORK.TABELAS_PA AS A
LEFT JOIN WORK.VIGENCIAPA AS B ON A.CD_LIMITE_PA = B.CdParametroVariacaoRangeCPF;
RUN;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE TABELAS_ATIVAS AS SELECT * FROM CONNECTION TO DbCon(
SELECT
a.CdTabelaFinanciamento
,a.DsTabela
,a.DsLogin
,a.DtInclusao
,a.DtInicioVigencia
,a.DtExpiracao
,a.DtAlteracao
,CASE a.TpRestricao
WHEN 1 THEN 'FILIAL'
WHEN 3 THEN 'NACIONAL'
WHEN NULL THEN 'PARCEIRO'
WHEN 2 THEN 'PARCEIRO'
WHEN 4 THEN 'NACIONAL/PARCEIRO'
END AS TpRestricao
,f.dsfilial
,CASE a.StTabelaFinanciamento
WHEN 'A' THEN 'APROVADA'
WHEN 'L' THEN 'CANCELADA'
WHEN 'E' THEN 'ENCERRADA'
WHEN 'R' THEN 'RECUSADA'
WHEN 'C' THEN 'CRIACAO'
WHEN 'P' THEN 'PENDENTE'
END AS 'Status'
,CASE a.CdAreaUtilizacao
WHEN 'C' THEN 'CONCESSIONARIA'
WHEN 'L' THEN 'MULTIMARCA'
WHEN 'T' THEN 'TODOS'
WHEN NULL THEN 'TODOS'
END AS 'Canal'
,d.PcTaxaAdicional,
a.FlPermiteAcordo,
a.PcMaximoAcordo
FROM DBCOR..TbTabelaFinanciamento a
INNER JOIN DBCOR..TbFinanciamentoVeiculo d ON a.CdTabelaFinanciamento = d.CdTabelaFinanciamento
LEFT JOIN dbcor..tbtabelafinanciamentofilial e ON a.CdTabelaFinanciamento= e.CdTabelaFinanciamento
LEFT JOIN dbcor..tbfilial f ON e.CdFilial = f.CdFilial WHERE
a.StTabelaFinanciamento IN ('A') );
RUN;

PROC SQL;
CREATE TABLE TABELAS_ATIVAS_FINAL AS
SELECT
*
FROM WORK.TABELAS_ATIVAS AS A
LEFT JOIN WORK.LIMITE_PA AS B ON A.CdTabelaFinanciamento = B.CdTabelaFinanciamento
LEFT JOIN WORK.DELTA_PPR AS C ON A.CdTabelaFinanciamento = C.CdTabelaFinanciamento;
RUN;



PROC SORT DATA= TABELAS_ATIVAS_FINAL OUT= TABELAS_ATIVAS_GERAL NODUPKEY;
BY CdTabelaFinanciamento;
RUN;

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE VIGENCIAPA AS SELECT * FROM CONNECTION TO DbCon(
SELECT
CdParametroVariacaoRangeCPF,
DsParametroVariacaoRangeCPF,
DtFimVigencia as DtFimVigenciaPA
FROM DBCOR.dbo.TbParametroVariacaoRangeCPF
GROUP BY CdParametroVariacaoRangeCPF
);
RUN;
