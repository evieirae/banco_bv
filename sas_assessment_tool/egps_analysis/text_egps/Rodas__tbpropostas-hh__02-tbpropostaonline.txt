%CONECTADB(server=PTA);
CREATE TABLE TbPropostaOnline AS
SELECT * FROM connection to dbcon
(
WITH TbColaborador as
(
	SELECT DISTINCT
		 Colaborador.CdOperador
		,first_value(Colaborador.CdColaborador) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdColaborador
		,first_value(Colaborador.CdFilial) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdFilial

	FROM DBA.TbColaborador Colaborador
	WHERE Colaborador.CdColaborador > 0
	AND Colaborador.CdOperador > 0
)
	SELECT
			 year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) NuAnoMes
			,DATE(Proposta.DtEntradaProposta) DtEntradaProposta
			,DATE(Proposta.DtStatusProposta) DtStatusProposta
			,DATETIME(Proposta.DtStatusProposta) DtStatusProposta_ajustada
			,DATE(CASE WHEN Proposta.StProposta in ('A','PA') THEN today()-1 ELSE Proposta.DtStatusProposta end) as DtTirReferencia
			,CASE WHEN SubProduto.CdModalidadeProduto = 57 THEN 6 else 1 end as CdTipoTaxaMercadoOrigem
			,Proposta.NuProposta
			,Proposta.StProposta
			,Proposta.NuContratoGestao
			,PropostaFinanceiro.CdTipoCobranca
			,TipoCobranca.NmTipoCobranca
			,Colaborador.CdColaborador
			,CASE WHEN Colaborador.CdColaborador <> ColaboradorPromotor.CdColaborador then ColaboradorPromotor.CdColaborador end as CdColaboradorPromotor
			,CASE WHEN SubProduto.CdProduto = 11 or Proposta.CdLoja = 418254 THEN Colaborador.CdFilial ELSE Regiao.CdFilial END AS CdFilial
			,CASE WHEN SubProduto.CdProduto = 11 or Proposta.CdLoja = 418254 THEN FilialColaborador.DsFilial ELSE Filial.DsFilial END AS DsFilial

			,Proposta.CdLoja CdParceiroComercial
			,Proposta.CdPessoa CdParceiroComercialCoban
			,PessoaSobAnalise.CdPessoaSobAnalise CdPessoa
			,PessoaSobAnalise.NmPessoaSobAnalise NmPessoa
			,PessoaSobAnalise.TpPessoa
			,PessoaSobAnalise.NuCpfCnpj
			,EnderecoTbPessoaSobAnalise.CdUf as CdUfPessoa
			,EnderecoTbPessoaSobAnalise.NmCidade as NmCidadePessoa
			,Proposta.PcCoeficiente
			,Proposta.QtDiasCarencia
			,CASE
				WHEN Proposta.QtParcelas >  60 THEN '60+'
				WHEN Proposta.QtParcelas >  48 THEN '60'
				WHEN Proposta.QtParcelas >  36 THEN '48'
				WHEN Proposta.QtParcelas >  24 THEN '36'
				WHEN Proposta.QtParcelas >  18 THEN '24'
				WHEN Proposta.QtParcelas >  12 THEN '18'
				WHEN Proposta.QtParcelas >   0 THEN '12'
			 END AS FxParcela

			,CASE WHEN SubProduto.CdProduto > 11 THEN PropostaFinanceiro.PcEntradaMolicar END PcEntradaMolicar
			,CASE WHEN SubProduto.CdProduto > 11 THEN Proposta.PcEntrada END PcEntrada

			,CASE
				WHEN SubProduto.CdProduto IN (10,11) THEN NULL
				WHEN Proposta.PcEntrada > 50 THEN '07 - Acima 50%'
				WHEN Proposta.PcEntrada > 40 THEN '06 - Ate 50%'
				WHEN Proposta.PcEntrada > 30 THEN '05 - Ate 40%'
				WHEN Proposta.PcEntrada > 20 THEN '04 - Ate 30%'
				WHEN Proposta.PcEntrada > 10 THEN '03 - Ate 20%'
				WHEN Proposta.PcEntrada > 0 THEN '02 - Ate 10%'
				ELSE '01 - Sem entrada'
			 END AS FxEntrada

			,Garantia.AaFabricacao
			,Garantia.AaModelo
			,Garantia.CdMarca
			,Garantia.CdModelo

			,CASE WHEN Garantia.FlZeroKm = ' ' THEN 'N' ELSE FlZeroKm END FlZeroKm
			,CASE
				WHEN Garantia.FlZeroKm = 'S' then 'ZeroKM'
				WHEN Garantia.QtAnosIdade >  10 THEN 'Acima de 10 anos'
				WHEN Garantia.QtAnosIdade >   7 THEN '8 a 10 anos'
				WHEN Garantia.QtAnosIdade >   2 THEN '3 a 7 anos'
				WHEN Garantia.QtAnosIdade >  -1 THEN '0 a 2 anos'
			 END FxIdade

			,PropostaFinanceiro.TpOrigemTaxa
			,case when PropostaFinanceiro.TpOrigemTaxa='A' then 1 else 0 end as FlTxAlterada
			,SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0) CdGrupoSubProduto
			,Proposta.CdSubProduto
			,SubProduto.DsSubProduto
			,PropostaComplemento.CdSistemaOrigem
			,Proposta.CdTabelaFinanciamento
			,case when PropostaCorrespondenteBncro.NuProposta is not null then 1 else 0 end as FlPropostaBipartida
			,case when FlUtilizacaoFgts = 'S' then 1 else 0 end as FlPropostaUtilizacaoFgts
			,Garantia.NuChassis
			,CASE WHEN nullif(Garantia.NuChassis,'') IS NOT NULL THEN 'CCB' END AS FlCCB
			,Proposta.StAverbacao
			,CASE WHEN Proposta.TpProposta <> '3' THEN 'S' ELSE 'N' END AS FlPropostaValida
			,Proposta.VrProposta
			,Proposta.VrEntrada
			,(PropostaFinanceiro.PcEntradaMolicar/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpEntrada

			,(Proposta.VrProposta-Proposta.VrEntrada) as VrLiberado
			,Proposta.PcMargemLiquida
			,(Proposta.PcMargemLiquida * (Proposta.VrProposta - Proposta.VrEntrada) / 100 ) AS VrMargemLiquida
			,Proposta.VrPDD
			,Proposta.VrFinanciamento
			,Proposta.VrPrestacao
			,Proposta.VrRetencao
			,Proposta.VrTac
			,Proposta.VrIofCpmfIss

			,PropostaFinanceiro.TxGerencial
			,(PropostaFinanceiro.TxGerencial/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxGerencial

			,Proposta.TxPactuada
			,(Proposta.TxPactuada/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxPactuada

			,Proposta.TxCetCalculada
			,(Proposta.TxCetCalculada/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxCetCalculada
			,Proposta.TxFinanciamento
			,(Proposta.TxFinanciamento/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxFinanciamento
			,Proposta.TxNet
			,(Proposta.TxNet/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxNet
			,PropostaFinanceiro.PcTaxaAlterada AS TxAlterada
			,(PropostaFinanceiro.PcTaxaAlterada/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxAlterada
			,PropostaFinanceiro.TxOriginalProposta AS TxOriginalProposta
			,(PropostaFinanceiro.TxOriginalProposta/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxOriginalProposta
			,PropostaFinanceiro.PcTaxaRetorno as TxRetorno
			,(PropostaFinanceiro.PcTaxaRetorno/100)*(Proposta.VrProposta-Proposta.VrEntrada) as VpTxRetorno

			,case when PessoaSobAnalise.NuScoreFim = 99 then 1 else PessoaSobAnalise.NuScoreFim end as NuScoreFim
			,(case when PessoaSobAnalise.NuScoreFim = 99 then 1 else PessoaSobAnalise.NuScoreFim end) * (Proposta.VrProposta-Proposta.VrEntrada) as VpScore

			,Garantia.QtAnosIdade
			,Garantia.QtAnosIdade*(Proposta.VrProposta-Proposta.VrEntrada) as VpIdade

			,Proposta.QtParcelas
			,Proposta.QtParcelas*(Proposta.VrProposta-Proposta.VrEntrada) as VpPrazo
			,Proposta.tsreplicacao

	FROM DBA.TbProposta Proposta

		LEFT JOIN DBA.TbSubProduto SubProduto
			ON Proposta.CdSubProduto = SubProduto.CdSubProduto

		LEFT JOIN DBCRED.TbPropostaFinanceiro PropostaFinanceiro
			on Proposta.NuProposta = PropostaFinanceiro.NuProposta

			LEFT JOIN DBCOR.TbTipoCobranca TipoCobranca
				ON  PropostaFinanceiro.CdTipoCobranca = TipoCobranca.CdTipoCobranca

		LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
			ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
			AND PessoaSobAnalise.CdPapel = 1

			LEFT JOIN DBCRED.TbEnderecoTbPessoaSobAnalise EnderecoTbPessoaSobAnalise
				ON  PessoaSobAnalise.CdPessoaSobAnalise = EnderecoTbPessoaSobAnalise.CdPessoaSobAnalise
				AND EnderecoTbPessoaSobAnalise.CdTipoEndereco = 1

		LEFT JOIN DBA.TbPropostaComplemento PropostaComplemento
			ON Proposta.NuProposta = PropostaComplemento.NuProposta

		LEFT JOIN DBCRED.TbPropostaCorrespondenteBncro PropostaCorrespondenteBncro
			on  Proposta.NuProposta = PropostaCorrespondenteBncro.NuProposta

		LEFT JOIN DBCRED.TbPropostaConsignado PropostaConsignado
			on  Proposta.NuProposta = PropostaConsignado.NuProposta

		LEFT JOIN DBA.TbPropGarantiaMaqEquip Garantia
			ON  Proposta.NuProposta = Garantia.NuProposta
			AND Garantia.CdPropostaGarantia = 1

		LEFT JOIN DBA.TbLoja Loja
			ON Proposta.CdLoja = Loja.CdLoja

			LEFT JOIN DBA.TbRegiao Regiao
				ON  Loja.CdRegiao = Regiao.CdRegiao

				LEFT JOIN DBA.TbFilial Filial
					on  Regiao.CdFilial = Filial.CdFilial

		LEFT JOIN DBA.TbParceiroComercial ParceiroComercial
			ON Proposta.CdLoja = ParceiroComercial.CdParceiroComercial

		LEFT JOIN TbColaborador Colaborador
			ON  Proposta.CdOperador = Colaborador.CdOperador

			LEFT JOIN DBA.TbFilial FilialColaborador
				on  Colaborador.CdFilial = FilialColaborador.CdFilial

		LEFT JOIN TbColaborador ColaboradorPromotor
			ON  Proposta.CdPromotor = ColaboradorPromotor.CdOperador

	WHERE 1=1
	and  Proposta.DtEntradaProposta >= today()-30
	ORDER BY Proposta.NuProposta

)
;RUN;

create table infogrc.TBPROPOSTA_ONLINE as
/*create table tbOriginacao_Online as*/
select

		and TbPropostaOnline.CdTipoTaxaMercadoOrigem = TaxaMercado.CdTipoTaxaMercadoOrigem

	left join bases.TbEstruturaComercial EstruturaComercial
		on  TbPropostaOnline.CdFilial = EstruturaComercial.CdFilial
		and TbPropostaOnline.NuAnoMes  = EstruturaComercial.NuAnoMes
		and EstruturaComercial.NmNivel = 'N2'

	left join TbTipoCanalProducaoProposta TipoCanalProducao
		on TbPropostaOnline.NuProposta = TipoCanalProducao.NuProposta

	left join bases.TbColaborador Colaborador
		on  TbPropostaOnline.CdColaborador = Colaborador.CdColaborador

	left join bases.TbColaborador ColaboradorPromotor
		on  TbPropostaOnline.CdColaboradorPromotor = ColaboradorPromotor.CdColaborador

	left join bases.TbEstruturaComercial EstruturaComercialN1
			on  Colaborador.CdColaboradorBI = EstruturaComercialN1.CdColaboradorBI
			and TbPropostaOnline.NuAnoMes  = EstruturaComercialN1.NuAnoMes
			and EstruturaComercialN1.CdColaboradorBI is not null
			and EstruturaComercialN1.NmNivel = 'N1'

	left join tmp.TbParceiroComercial ParceiroComercial
		on  TbPropostaOnline.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

	left join tmp.TbParceiroComercial ParceiroComercialCoban
		on  TbPropostaOnline.CdParceiroComercialCoban = ParceiroComercialCoban.CdParceiroComercial

	left join bases.TbPropostaComplementoParceiro PropostaSinergia
		on  TbPropostaOnline.NuProposta = PropostaSinergia.NuProposta

	left join TbPropostaComplementoParceiro ParceiroSinergia
		on  coalesce(PropostaSinergia.NuCnpjLoja, ParceiroComercial.NuCnpjLoja) = ParceiroSinergia.NuCnpjLoja

	left join tmp.TbIntegracaoProposta IntegracaoProposta
		on  TbPropostaOnline.NuProposta = IntegracaoProposta.NuProposta

		on  TbPropostaOnline.NuProposta = PropostaAprovadaPaga.NuProposta

	left join basegrc.TbTabelasFinanciamento TabelasFinanciamento
		on TbPropostaOnline.CdTabelaFinanciamento = TabelasFinanciamento.CdTabelaFinanciamento

		on  TbPropostaOnline.NuProposta = CetipConversao.NuProposta

	left join bases.TbTempoData TempoData
		on  TbPropostaOnline.DtEntradaProposta = TempoData.DtTempo

	LEFT JOIN BASEGRC.TbLevesNovosBVO BVO
    	ON BVO.CdParceiroComercial = TbPropostaOnline.CdParceiroComercial
		and BVO.nuanomes = TbPropostaOnline.NuAnoMes

	where NmGrupoProdutoMeta in ('Leves','Leves-BVO','Motos','Pesados','CVG Leves','CVG Pesados','Solar')
