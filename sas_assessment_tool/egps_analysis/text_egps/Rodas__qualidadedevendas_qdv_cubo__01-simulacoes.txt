PROC SQL;
CREATE TABLE Leads_V01 AS
  SELECT DISTINCT
    NuProposta as NuProposta,
	NuAnoMes_Simu,
    upcase(DsSistemaInicial) as CompraLeads,
	CASE WHEN FlComissao = 'Sim' AND upcase(DsSistemaInicial) IN ('DRYVE','DRYVE SEM SIMUL.','ESCALE SEM SIMUL.','KARVI','ICARROS','MERCADO_LIVRE','MOBIAUTO','USADOSBR','SO_CARRAO') THEN 1
		 WHEN UPCASE(DSSISTEMAINICIAL) IN ('SITE_BV','MEU_CARRO_NOVO','APP','NA_PISTA') THEN 1
		 ELSE 0
    END AS FlagProducao,
	DsMidiaOrigem as OrigemSiteBV
  FROM
    LEADSDIG.LEADS_DIGITAIS_23
  WHERE
    NuProposta <> .
