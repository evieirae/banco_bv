proc sql;
create table Base_PainelBaaS as
select
	Data,
	Grupo,
	sum(Qtde) as Qtde,
	sum(Receita) as Receita,
	sum(Resultado) as Resultado,
	sum(Custo) as Custo
from
(
	select
		*,
		case when Origem in ('Floating Boleto', 'Floating Emissao') then 'Floating'
			 when Origem in ('Tarifas','PIX','TED','Boleto','CIP','STR') then 'Tarifas'
			 when Produto = 'Fee' then 'Fee'
			 when Origem = 'Razão' and Produto = 'TOTAL CONTABIL TARIFAS' then 'Contabil'

		end as Grupo
	from BasePainelBaaS
)
group by Data, Grupo
;
quit;

proc sql;
create table PainelBaaS as
select
	Data,
	sum(case when Grupo = 'Floating' then Resultado end) as Floating,
	calculated Floating as 'Margem Financeira'n,
	sum(case when Grupo in ('Tarifas','Conta BV','BVF') then Resultado end) as Tarifas,
	sum(case when Grupo = 'Fee' then Resultado end) as 'Estrutur./Assessoria/Outros Fees'n,
	calculated Tarifas + calculated 'Estrutur./Assessoria/Outros Fees'n as 'Receita de Serviços'n,
	calculated 'Margem Financeira'n + calculated 'Receita de Serviços'n as 'Margem + Serviços'n,
	sum(case when Grupo = 'Contabil' then Resultado end)
	- calculated Tarifas
	- sum(case when Grupo in ('Tarifas','Conta BV','BVF') then -Custo end)
	+ sum(case when Grupo in ('Conta BV') then Receita end)
	+ sum(case when Grupo in ('BVF') then Receita end)
	as Assimetria
from Base_PainelBaaS
group by Data
;
quit;

		CREATE TABLE tb_Tarifa_CashBaaS AS
		SELECT
			a.DtCobranca format=ddmmyy10. as Data,
			a.CdTarifa,
			a.NmTarifa,
			a.cdPessoa,
			a.NuConta,

				when upper(b.Produto) IN ('COBRANCA') and c.Flag = 'S' then -Coalesce(a.QtLancamentos,0) * coalesce(b.CobrancaPropria,0)
				when upper(b.Produto) IN ('COBRANCA') then -Coalesce(a.QtLancamentos,0) * coalesce(b.CustoUnitario,0)

		From tbBaseTarifa a

	- sum(case when Grupo in ('Tarifas','Conta BV','BVF') then -Custo end)
	+ sum(case when Grupo in ('Conta BV') then Receita end)
	+ sum(case when Grupo in ('BVF') then Receita end)
	as Assimetria
