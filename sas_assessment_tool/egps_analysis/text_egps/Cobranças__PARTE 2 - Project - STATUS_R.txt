CREATE TABLE OBF_PARA_PROCESSAR_02 AS SELECT DISTINCT

	Y.DATAPROCESAMENTO,
	Y.DEMANDA,
	Y.VAC,
	Y.CONTRATO,
	Y.CPF,
	Y.NOMEPARTE,
	'CONTENCIOSO' AS FASE,
	C.FLAG_SITU_ASSES,
	C.DT_ENVIO,
	C.DT_RETIRADA,
	D.EPS,
	CASE
		WHEN Y.DEMANDA ='OBF - MANUTENÇÃO DE POSSE' AND FLAG_SITU_ASSES='ASSES VIGENTE' THEN 1
		WHEN Y.DEMANDA ='OBF - MANUTENÇÃO DE POSSE'  THEN 1
		WHEN Y.DEMANDA <> 'OBF - MANUTENÇÃO DE POSSE' THEN 1
		ELSE 0 END AS FLAG_FILTRO,

	CASE WHEN C.ASSESSORIA IS NULL THEN 'passível de encerramento'
		 WHEN Y.DEMANDA ='OBF - MANUTENÇÃO DE POSSE' AND FLAG_SITU_ASSES='RETIRADO DE ASSES' THEN 'passível de encerramento'
		 WHEN C.COD_AGENTE_PRINCIPAL IN (1426,1421,1413,1422,1419,1425,1261,1241,1423,1424,834,1414,1420,833,1031,971,130,1241,1263) THEN 'passível de encerramento' 
		 ELSE 'ENVIAR' END AS OBS

FROM OBF_PARA_PROCESSAR_01 Y LEFT JOIN CONNECTION TO DBCON
(SELECT
    B.NUCONTRATOGESTAO,
    B.SQCONTRATOFINANCEIRO,
	E.COD_AGENTE_PRINCIPAL,
	CONVERT(DATE,B.DT_ASSES) AS DT_ENVIO,
    CONVERT(DATE,B.DT_RET_ASSES) AS DT_RETIRADA,
    CASE WHEN CONVERT(DATE,B.DT_RET_ASSES) IS NULL THEN 'ASSES VIGENTE' ELSE 'RETIRADO DE ASSES' END AS FLAG_SITU_ASSES,
	F.RAZAO_NOME AS ASSESSORIA

    FROM DBA.SAJ_PROCESSO B LEFT JOIN SCC_FIN.SAJ_AGENTE E
    ON B.COD_ASSES=E.COD_AGENTE
    LEFT JOIN SCC_FIN.SAJ_AGENTE F
    ON E.COD_AGENTE_PRINCIPAL = F.COD_AGENTE
    WHERE E.COD_AGENTE_PRINCIPAL <> 0
      
    AND B.COD_ACAO IN (1,3,59,60)) C
	ON y.CONTRATO=C.NUCONTRATOGESTAO
LEFT JOIN BASE_DEPARA_ASSESS D
	ON C.ASSESSORIA=D.AGENTE
WHERE  CALCULATED FLAG_FILTRO=1;
