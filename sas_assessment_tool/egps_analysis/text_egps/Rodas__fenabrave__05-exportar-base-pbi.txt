%include '/infogrc/utils/all-includes.sas';
options compress=yes;

DATA _NULL_;
  dtRef = INTNX('MONTH',TODAY(),-1);
  dtInicio = INTNX('YEAR',dtRef,-1);

  safraIni = (YEAR(dtInicio) * 100) + MONTH(dtInicio);
  safraFim = (YEAR(dtRef) * 100) + MONTH(dtRef);

  CALL SYMPUT('safraIni', safraIni);
  CALL SYMPUT('safraFim', safraFim);
RUN;

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbDiasUteis AS SELECT * FROM connection to dbcon
(
WITH
TbTempoData AS (
  SELECT
    NuAnoMes,
    CASE WHEN DtTempo BETWEEN '20211227' AND '20211230' THEN NuDiaUtilMes - 1
         WHEN DtTempo IN ('20211224','20211231') THEN -3
         ELSE NuDiaUtilMes END AS NuDiaUtilMes,
    NuSemanaAno,
    DtTempo,
	DdData,
	NuDiaSemana
  FROM DWBI_DIME.TbDimTempoData
),
QtdMes AS (
  SELECT
    NuAnoMes,
    MAX(NuDiaUtilMes) AS NuDiaUtilMes,
	MIN(DtTempo) AS Dtref
  FROM TbTempoData
  WHERE
    NuAnoMes >= 201801
	and DtTempo <= today()
  GROUP BY NuAnoMes
)
SELECT DISTINCT
  A.NuAnoMes,
  A.NuDiaUtilMes,
  Dtref
  FROM QtdMes A
);
RUN;

PROC SQL;
  SELECT DISTINCT NuAnoMes, QtTotalDiaUtilMes
  FROM bases.TbTempoData
  WHERE NuAnoMes BETWEEN &safraIni. AND &safraFim.
  ORDER BY NuAnoMes DESC;
RUN;

PROC SQL;
CREATE TABLE tbGravames_01 AS
  SELECT
    base as nuAnoMes,
	UF as cdUF,
	stVeiculo,
	tpVeiculo,
	SUM(qtGravames) AS qtGravames
  FROM RELEXC.MarketShareFenabrave
  WHERE base BETWEEN &safraIni. AND &safraFim.
    AND AjusteManual = 'N'
  GROUP BY 1,2,3,4;
RUN;

PROC SQL;
CREATE TABLE tbFenabrave AS
  SELECT
    Fenabrave.base,
    Fenabrave.cdUf,
    Fenabrave.stVeiculo,
    Fenabrave.tpVeiculo,
    SUM(Fenabrave.qtd) - SUM(Gravames.qtGravames) AS qtTotal
  FROM
    mis.tb_fenabrave_mktshare_00 Fenabrave
  INNER JOIN tbGravames_01 Gravames
    ON Fenabrave.base = Gravames.nuAnoMes
    AND Fenabrave.cdUf = Gravames.CdUf
    AND Fenabrave.StVeiculo = Gravames.StVeiculo
    AND Fenabrave.TpVeiculo = Gravames.TpVeiculo
  WHERE
    Fenabrave.Qtd > Gravames.qtGravames
  GROUP BY 1,2,3,4;
QUIT;

PROC SQL;
CREATE TABLE tbFenabrave_02 AS
  SELECT
    base,
    TRIM(UPCASE(TpVeiculo)) AS dsTpVeiculo,
    TRIM(UPCASE(StVeiculo))||'S' AS dsStVeiculo,
	SUM(qtTotal) AS qtd
  FROM tbFenabrave
  WHERE base BETWEEN &safraIni. AND &safraFim.
  GROUP BY 1,2,3;
RUN;

PROC SQL;
CREATE TABLE tbGravames_02 AS
  SELECT
    base,
    grupo,
    subGrupo,
    TRIM(UPCASE(TpVeiculo)) AS dsTpVeiculo,
    TRIM(UPCASE(StVeiculo))||'S' AS dsStVeiculo,
    fxIdade as dsFxIdade,
    COALESCE(tpAgenteProduto,'1') AS tpAgenteProduto,
    SUM(qtGravames) AS qtd
  FROM RELEXC.MarketShareFenabrave
  WHERE base BETWEEN &safraIni. AND &safraFim.
    AND AjusteManual = 'N'
  GROUP BY 1,2,3,4,5,6,7;
RUN;

PROC SQL;
CREATE TABLE tbMktShareMensal_01 AS
  SELECT * FROM tbGravames_02
  OUTER UNION CORRESPONDING
  SELECT * FROM tbFenabrave_02
;RUN;

PROC SQL;
CREATE TABLE tbMktShareMensal AS
  SELECT
	a.*,
	du.NuDiaUtilMes,
	du.Dtref,
	Case when a.base = &safraFim. then 'M0'
		 when a.base = &safram2. then 'M1'
		 end as MesRef

FROM tbMktShareMensal_01 as a
	left join TbDiasUteis as du
	on du.nuanomes = a.base
;RUN;

%ExportaCSV(TbMarketShare_FenabravePBI,tbMktShareMensal);
