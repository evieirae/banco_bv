select * from connection to dbcon

	,CASE WHEN a.NmTipoProdutoCartao LIKE '%MASTER%' THEN 'MASTER'

FROM 	DWBI_DIME.TbDimContaCartao A
	INNER JOIN
		DWBI_DMOC.TbFatoContaCartao	B
		ON A.CdDimContaCartaoVersao = B.CdDimContaCartaoVersao
	INNER JOIN
		DWBI_DIME.TbDimPessoa C
		ON B.CdPessoaTitularConta = C.CdPessoa
WHERE 	B.DtReferencia BETWEEN A.DtInicioVersaoContaCartao AND A.DtFimVersaoContaCartao
	and B.DtReferencia = &Dt_Ref_Final.
)	AS b
WHERE b.RN = 1
;quit;
