PROC SQL;
	CREATE TABLE tempw.Propostas_Originacao AS

	SELECT distinct
		t2.NuProposta,
		t2.CdColaborador,
		t2.NmColaboradorC as NmGr,
		t2.NmGrupoProdutoMeta as Produto,
		coalesce(t2.DsLoginN1C,t3.DsLoginN1C) as GR,
		coalesce(t2.DsLoginN2C,t3.DsLoginN2C) as GF,
		t2.NmFilialC,
		t2.FlPropostaOriginada,
		t2.FlPropostaNegada,
		t2.FlPropostaAprovada,
		t2.FlPropostaAPNP,
		t2.FlPropostaEfetivada,
		t2.VrPropostaOriginada format commax10.2 as	VrPropostaOriginada,
		t2.VrPropostaNegada format commax10.2 as VrPropostaNegada,
		t2.VrPropostaAprovada format commax10.2 as VrPropostaAprovada,
		t2.VrPropostaAPNP format commax10.2 as	VrPropostaAPNP,
		t2.VrPropostaEfetivada format commax10.2 as VrPropostaEfetivada,
		t3.QtSeguros,
		t3.VrSeguros format commax10.2,
		t3.VrBVMais format commax10.2 as VrSeguro_SPF,
		t3.VrCap format commax10.2 as VrSeguro_AP

	FROM TEMP.TBORIGINACAO t2
	left join
		TEMP.TBPRODUCAO t3
		on t2.NuProposta = t3.NuProposta
		WHERE year(t2.DtEntradaProposta) >= year(today())-1
			/*AND t2.PcMargemLiquida < 0*/
	AND t2.NmGrupoProdutoMeta = 'Privado';
RUN;

PROC SQL;
	CREATE TABLE tempw.Depara_colaborador AS

	SELECT DISTINCT
	CASE WHEN NmGr IS NULL THEN "N/I" ELSE NmGr END AS NmGr
	FROM tempw.Propostas_Originacao
/*where NmGr is not null*/
;
RUN;

%let dir_csv = /riscosbvf/BA/processos/output;

proc export
	data=tempw.Depara_colaborador
	dbms=csv
	OUTFILE= "&dir_csv./Privado_Depara_colaborador.csv"
	replace;
run;
