proc sql; select max(dt_atualizacao) format date9. into:dtmax from basegrc.tbsharemissao; run;
data _null_;
	hoje = input(today(), yymmddn8.);
	dtgcp = "'"||put(year("&dtmax."d),4.)||"-"||put(month("&dtmax."d),z2.)||"-"||put(day("&dtmax."d),z2.)||"'";

	call symputx('hoje', put(hoje, yymmddn8.));
	call symputx('dtgcp', dtgcp);
;run;
%put &hoje; %put &dtmax; %put &dtgcp;

proc sql;
create table tb_missao_analitica as
select distinct
NmAreaEstruturaComercialC,

catx("@", dsloginn2c,"bv.com.br") as emailN2,
Propcase(scan(dsloginn1c, 1, '.')) as gr,
propcase(trim(strip(scan(dsloginn2c, 1, '.')))) as gf
from basegrc.tbsharemissao
where dt_atualizacao = "&dtmax."d
;run;
