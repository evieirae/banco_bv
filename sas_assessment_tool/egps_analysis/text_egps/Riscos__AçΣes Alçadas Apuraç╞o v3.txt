PROC SQL;
   CREATE TABLE WORK.ALCADAS_ACORDOS AS
   SELECT t1.NumRegSistContrGest,
          t1.Pasta,
          t1.DtEnvioInicial,
          t1.'FASE1ALÇADA1'n as Fase1Alcada1,
          t1.'FASE1ALÇADA2'n as Fase1Alcada2,
          t1.'FASE1ALÇADA3'n as Fase1Alcada3,
          t1.'FASE2ALÇADA4'n as Fase2Alcada4,
          t1.'FASE2ALÇADA5'n as Fase2Alcada5,
          t1.'Alçada Especial'n as AlcadaEspecial,
          -t2.VlPrincipal as ValorPago,
	      t1.FlagSerasa,
          t2.Desfecho_Pagtos
      FROM WORK.NOVABASE t1
           LEFT JOIN ACM.MASSIFICADO t2 ON (t1.NumRegSistContrGest = t2.NumRegSistContrGest)
      WHERE t1.DtEnvioInicial BETWEEN '1Jan2023'd AND '31Dec2024'd;
QUIT;

DATA WORK.MONITORAMENTO1 (compress=yes);
SET WORK.ALCADAS_ACORDOS;
FORMAT AlcadaFechada $CHAR100.;
IF	(Fase1Alcada1="" OR Fase1Alcada1=0) AND (Fase1Alcada2="" OR Fase1Alcada2=0)
AND (Fase1Alcada3="" OR Fase1Alcada3=0) AND (Fase2Alcada4="" OR Fase2Alcada4=0)
AND (Fase2Alcada5="" OR Fase2Alcada5=0) THEN AlcadaFechada='Sem Alçada';
ELSE IF ValorPago = 0 AND Desfecho_Pagtos in ('1.Perda Zero', '3.Condenação') THEN AlcadaFechada = '';
ELSE IF ValorPago = . AND Desfecho_Pagtos in ('1.Perda Zero', '3.Condenação') THEN AlcadaFechada = '';
ELSE IF ValorPago > 0 AND Desfecho_Pagtos = "" THEN AlcadaFechada = '';
ELSE IF ValorPago = 0 AND Desfecho_Pagtos not in ('1.Perda Zero', '3.Condenação') THEN AlcadaFechada = 'Ação Ativa';
ELSE IF ValorPago = . AND Desfecho_Pagtos not in ('1.Perda Zero', '3.Condenação') THEN AlcadaFechada = 'Ação Ativa';
ELSE IF ValorPago > 0 AND Desfecho_Pagtos in ('1.Perda Zero', '3.Condenação') THEN AlcadaFechada = '';
ELSE IF ValorPago <= Fase1Alcada1 THEN  AlcadaFechada = 'Alçada 1';
ELSE IF ValorPago >  Fase1Alcada1 AND ValorPago <= Fase1Alcada2   THEN  AlcadaFechada = 'Alçada 2';
ELSE IF ValorPago >  Fase1Alcada2 AND ValorPago <= Fase1Alcada3   THEN  AlcadaFechada = 'Alçada 3';
ELSE IF ValorPago >  Fase1Alcada3 AND ValorPago <= Fase2Alcada4   THEN  AlcadaFechada = 'Alçada 4';
ELSE IF ValorPago >  Fase2Alcada4 AND ValorPago <= Fase2Alcada5   THEN  AlcadaFechada = 'Alçada 5';
ELSE IF ValorPago >  Fase2Alcada5 AND ValorPago <= AlcadaEspecial THEN  AlcadaFechada = 'Alçada Especial';
ELSE IF ValorPago >  Fase1Alcada3 AND (Fase2Alcada5 ="" OR Fase2Alcada5 =0 OR AlcadaEspecial ="")  THEN AlcadaFechada = 'Maior que alçada 3';
ELSE IF ValorPago >  Fase2Alcada5 AND Fase2Alcada5 not ="" AND (AlcadaEspecial=. OR AlcadaEspecial=0) THEN AlcadaFechada = 'Maior que alçada 5';
ELSE if ValorPago >  AlcadaEspecial AND AlcadaEspecial not ="" then AlcadaFechada = 'Maior que alçada Especial';
RUN;
