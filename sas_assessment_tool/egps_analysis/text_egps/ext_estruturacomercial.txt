%CONECTADB(server=BI);
create table TbLstSubstituicaoAux as
select * from connection to dbcon
(
		SELECT
			 Filial.CdFilialSistemaOrigem CdFilial
			,HierarquiaComercialSubstituto.DsLogin
		 FROM DWBI_DTWH.TbSubstituicaoHierarquia SubstituicaoHierarquia

		      INNER JOIN DWBI_DTWH.TbSubstituicaoHierarquiaFilial SubstituicaoHierarquiaFilial
		            ON SubstituicaoHierarquia.CdSubstituicaoHierarquia = SubstituicaoHierarquiaFilial.CodigoSubstituicaoHierarquia

		            INNER JOIN DWBI_DTWH.TbFilial Filial
		                ON SubstituicaoHierarquiaFilial.CdFilial = Filial.CdFilial

		      LEFT JOIN DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoaSubstituto
		            ON  SubstituicaoHierarquia.CdPessoaSubstituto = PessoaRelacaoPessoaSubstituto.CdPessoa
		            AND PessoaRelacaoPessoaSubstituto.CdTipoRelacaoPessoa = 7

		             LEFT JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercialSubstituto
		            	ON  PessoaRelacaoPessoaSubstituto.CdPessoaRelacao = HierarquiaComercialSubstituto.CdColaborador
		            	AND today() BETWEEN HierarquiaComercialSubstituto.DtInicioVigencia AND HierarquiaComercialSubstituto.DtFimVigencia

		 WHERE SubstituicaoHierarquia.CdSubstituicaoHierarquia > 0
		   AND today() BETWEEN SubstituicaoHierarquia.DtInicioVigencia AND SubstituicaoHierarquia.DtFimVigencia
		   AND SubstituicaoHierarquia.StSubstituicaoHierarquia IN ('A')

)
;
run;
quit;

proc sort data=TbLstSubstituicaoAux nodupkey;by CdFilial DsLogin;run;

%CONECTADB(server=BI);
create table TbEstruturaComercialSubstituicao as
select * from connection to dbcon
(

	SELECT DISTINCT
		 Filial.CdFilialSistemaOrigem CdFilial
		,Filial.NmFilial
		,HierarquiaComercialSubstituto.DsLogin
		,HierarquiaComercialSubstituto.NmColaborador
		,SubstituicaoHierarquia.DtInicioVigencia
		,SubstituicaoHierarquia.DtFimVigencia
		,SubstituicaoHierarquia.DsSubstituicaoHierarquia
	FROM DWBI_DTWH.TbSubstituicaoHierarquia SubstituicaoHierarquia

		INNER JOIN DWBI_DTWH.TbSubstituicaoHierarquiaFilial SubstituicaoHierarquiaFilial
			ON SubstituicaoHierarquia.CdSubstituicaoHierarquia = SubstituicaoHierarquiaFilial.CodigoSubstituicaoHierarquia

			INNER JOIN DWBI_DTWH.TbFilial Filial
				ON  SubstituicaoHierarquiaFilial.CdFilial = Filial.CdFilial

		LEFT JOIN DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoaSubstituto
			ON  SubstituicaoHierarquia.CdPessoaSubstituto = PessoaRelacaoPessoaSubstituto.CdPessoa
			AND PessoaRelacaoPessoaSubstituto.CdTipoRelacaoPessoa = 7

			LEFT JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercialSubstituto
				ON  PessoaRelacaoPessoaSubstituto.CdPessoaRelacao = HierarquiaComercialSubstituto.CdColaborador
				AND today() BETWEEN HierarquiaComercialSubstituto.DtInicioVigencia AND HierarquiaComercialSubstituto.DtFimVigencia

	INNER JOIN DWBI_DIME.TbDimTempoData TempoData
		ON  TempoData.DtTempo BETWEEN SubstituicaoHierarquia.DtInicioVigencia AND SubstituicaoHierarquia.DtFimVigencia
		AND TempoData.DtTempo >= today()-7

	WHERE SubstituicaoHierarquia.CdSubstituicaoHierarquia > 0
	  AND SubstituicaoHierarquia.StSubstituicaoHierarquia IN ('A')
	ORDER BY HierarquiaComercialSubstituto.DsLogin
)
;

%CONECTADB(server=BI);
create table TbDiasAtividade as
select * from connection to dbcon
(
	SELECT
		 TempoData.NuAnoMes
		,HierarquiaComercial.CdColaborador CdColaboradorBI
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador = 5 THEN 1 ELSE 0 END) QtDiasAtividadeNormal
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador = 8 THEN 1 ELSE 0 END) QtDiasFerias
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador NOT IN (8,5) THEN 1 ELSE 0 END) QtDiasOutros
		,count(*) AS QtDias
	FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
				ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

			LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
				ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

				INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
					ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
					AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador

				INNER JOIN DWBI_DIME.TbDimTempoData TempoData
					ON  TempoData.DtTempo >= case when HierarquiaComercial.DtInicioVigencia = '19000101' then HierarquiaComercial.DtAdmissaoColaborador else HierarquiaComercial.DtInicioVigencia end
					AND TempoData.DtTempo <= HierarquiaComercial.DtFimVigencia
					AND TempoData.DtTempo >= '20180101'
					AND TempoData.DtTempo <= today()
					AND TempoData.FlDiaUtil = 'S'

	WHERE HierarquiaComercial.CdEstadoColaborador IN (5,8,18,9,13,6,14,17)

	GROUP BY
		 TempoData.NuAnoMes
		,HierarquiaComercial.CdColaborador

)
;
run;
quit;

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbColaboradorFilial_ATE2022 as
select * from connection to dbcon
(

	WITH cte AS
	(

		SELECT
			RelacaoPessoa.CdPessoaRelacao CdColaboradorBI
		   ,CAST (Pessoa.CdPessoaOrigem AS INTEGER) CdColaborador
		   ,row_number() over (partition BY RelacaoPessoa.CdPessoaRelacao
		ORDER BY
			CASE
				WHEN Pessoa.NmPessoa LIKE '%DUP%' THEN -2
				WHEN Colaborador.DsLogin LIKE 'neo.%' THEN -1
				WHEN Colaborador.DsLogin LIKE 'parla.%' THEN -1
				WHEN Colaborador.DsLogin LIKE 'tivit.%' THEN -1
				WHEN Pessoa.FLATIVO = 'S' THEN 1 ELSE 0 END desc, Pessoa.NuExecucaoProcessoInclusao desc) AS RN

		FROM DWBI_DTWH.TbPessoaRelacaoPessoa RelacaoPessoa

			LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
				ON RelacaoPessoa.CdPessoa = Pessoa.CdPessoa

				LEFT JOIN DWBI_DTWH.TbColaborador Colaborador
					ON Pessoa.CdPessoa = Colaborador.CdPessoa
					AND today() BETWEEN Colaborador.DtInicioVigencia AND Colaborador.DtFimVigencia

		WHERE RelacaoPessoa.CdTipoRelacaoPessoa = 7
		  and today() between RelacaoPessoa.DtInicioVigencia and RelacaoPessoa.DtFimVigencia

	)
	,TbLstCdColaborador as
	(
		SELECT
			 CdColaboradorBI
			,max(CASE WHEN rn = 1 THEN CdColaborador end) AS CdColaboradorAtivo
			,CAST(list(CASE WHEN rn > 1 THEN CdColaborador end,', ') AS VARCHAR(255)) AS LstCdColaboradorInativo
		FROM cte
		GROUP BY CdColaboradorBI
	)
	,TbDeParaFilial as
	(
		SELECT
			 DeParaFilial.CdAreaEstruturaComercial
			,DeParaFilial.CdFilial
			,Filial.NmFilial
			,CASE
				WHEN Filial.NmFilial LIKE 'CP SOL%' THEN 5
				WHEN Filial.NmFilial LIKE 'CP CONS%' THEN 4
				WHEN DeParaFilial.CdFilial IN (275,276) THEN 3
				WHEN Filial.NmFilial LIKE 'CP %' THEN 2
				WHEN Filial.NmFilial NOT LIKE 'CP ' THEN 1
			 ELSE 0 END AS CdCanalEstruturaComercial

			,first_value(Area.CdAreaEstruturaCmrclStmaOrgm)
				over (partition by DeParaFilial.CdFilial
					order by
						case when DeParaFilial.CdAreaEstruturaComercial in (17166,17282) then 0 else Area.CdAreaEstruturaComercial end desc) as CdAreaEstruturaComercialMax

			,first_value(Area.NmAreaEstruturaComercial)
				over (partition by DeParaFilial.CdFilial
					order by
						case when DeParaFilial.CdAreaEstruturaComercial in (17166,17282) then 0 else Area.CdAreaEstruturaComercial end desc) as NmAreaEstruturaComercialMax
		FROM DWBI_DTWH.TbDeParaFilial DeParaFilial

			LEFT JOIN DWBI_DTWH.TbFilial Filial
				ON  DeParaFilial.CdFilial = Filial.CdFilialSistemaOrigem
				AND Filial.CdSistemaOrigem = 209

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial Area
				ON  DeParaFilial.CdAreaEstruturaComercial = Area.CdAreaEstruturaCmrclStmaOrgm
	)

	SELECT
		 TempoData.NuAnoMes
		,TempoData.AaData AaReferencia
		,TempoData.MesData MmReferencia
		,(TempoData.DtTempo-day(TempoData.DtTempo))+1 DtInicioVigencia
		,dateadd(month,1,(TempoData.DtTempo-day(TempoData.DtTempo))+1)-1 as DtFimVigencia
		,HierarquiaComercial.CdColaborador CdColaboradorBI
		,LstCdColaborador.CdColaboradorAtivo
		,LstCdColaborador.LstCdColaboradorInativo
		,Colaborador.CdColaboradorRH IdMatricula
		,UPPER(HierarquiaComercialCadastro.NmColaborador) NmColaborador
		,UPPER(substring(HierarquiaComercialCadastro.NmColaborador,1,
		 CHARINDEX(' ',HierarquiaComercialCadastro.NmColaborador))) NomeColaborador
		,UPPER(substring(HierarquiaComercialCadastro.NmColaborador,
		 CHARINDEX(' ',HierarquiaComercialCadastro.NmColaborador)+1,60)) SobreNomeColaborador
		,UPPER(HierarquiaComercialCadastro.DsLogin)||'@BVFINANCEIRA.COM.BR' DsEmailFinanceira
		,substring(CAST(HierarquiaComercialCadastro.NuCpfCnpj+100000000000 AS VARCHAR(12)),2,11) NuCpfCnpj
		,lower(HierarquiaComercialCadastro.DsLogin) DsLogin
		,Filial.CdFilialSistemaOrigem CdFilial
		,Filial.NmFilial
		,DeParaFilial.CdAreaEstruturaComercialMax as CdAreaEstruturaComercial
		,DeParaFilial.NmAreaEstruturaComercialMax as NmAreaEstruturaComercial
		,HierarquiaComercial.CdSubEstruturaComercial
		,HierarquiaComercial.NmSubEstruturaComercial
		,DeParaHierarquia.CdCanalEstruturaComercial
		,DeParaHierarquia.NmCanalEstruturaComercial
		,DeParaHierarquia.NmNivel
		,DeParaHierarquia.CdGrupoCargo
		,DeParaHierarquia.NmGrupoCargo
		,'' NmColaboradorN4
		,'' DsLoginN4
		,count(*) QtDias

	FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial

		LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
			ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

		LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
			ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

			INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
				ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
				AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador

			 CdColaboradorBI

		,UPPER(HierarquiaComercialCadastro.DsLogin)||'@BVFINANCEIRA.COM.BR' DsEmailFinanceira

proc sql;
%CONECTADB(server=BI);
create table TbEstruturaProspectores as
select * from connection to dbcon
(
	WITH cte AS
	(

		SELECT
			a.CdPessoaRelacao CdColaboradorBI
		   ,CAST (b.CdPessoaOrigem AS INTEGER) CdColaborador
		   ,row_number() over (partition BY a.CdPessoaRelacao
		ORDER BY
			CASE
				WHEN B.NMPESSOA LIKE '%DUP%' THEN -1
				WHEN b.FLATIVO = 'S' THEN 1 ELSE 0 END desc, b.NuExecucaoProcessoInclusao desc) AS RN

		FROM DWBI_DTWH.TbPessoaRelacaoPessoa a
			LEFT JOIN DWBI_DTWH.TbPessoa b
				ON a.CdPessoa = b.CdPessoa
		WHERE a.CdTipoRelacaoPessoa = 7
		  and today() between a.DtInicioVigencia and a.DtFimVigencia
	)
	,TbLstCdColaborador as
	(
		SELECT
			 CdColaboradorBI
			,max(CASE WHEN rn = 1 THEN CdColaborador end) AS CdColaboradorAtivo
			,CAST(list(CASE WHEN rn > 1 THEN CdColaborador end,', ') AS VARCHAR(255)) AS LstCdColaboradorInativo
		FROM cte
		GROUP BY CdColaboradorBI
	)

	SELECT
		 TempoData.NuAnoMes
		,(TempoData.DtTempo-day(TempoData.DtTempo))+1 DtInicioVigencia
		,dateadd(month,1,TempoData.DtTempo)-day(TempoData.DtTempo) DtFimVigencia
		,HierarquiaCadastro.NmColaborador
		,substring(CAST(HierarquiaCadastro.NuCpfCnpj+100000000000 AS VARCHAR(12)),2,11) NuCpfCnpj
		,Colaborador.CdColaboradorRH IdMatricula
		,HierarquiaComercial.CdColaborador AS CdColaboradorBI
		,LstCdColaborador.CdColaboradorAtivo
		,LstCdColaborador.LstCdColaboradorInativo
		,HierarquiaComercial.NmCargoColaborador
		,HierarquiaCadastro.DsLogin
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador = 5 THEN 1 ELSE 0 END) AS QtDiasAtividadeNormal
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador = 8 THEN 1 ELSE 0 END) AS QtDiasFerias
		,sum(CASE WHEN HierarquiaComercial.CdEstadoColaborador NOT IN (5,8) THEN 1 ELSE 0 END) AS QtDiasOutros

	FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial

		LEFT JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaCadastro
			ON  HierarquiaComercial.CdColaborador = HierarquiaCadastro.CdColaborador
			AND today() BETWEEN HierarquiaCadastro.DtInicioVigencia AND HierarquiaCadastro.DtFimVigencia

		LEFT JOIN DWBI_DTWH.TbColaborador Colaborador
			ON  HierarquiaComercial.CdColaborador = Colaborador.CdPessoa
			AND Colaborador.CdSistemaOrigem = 202
			AND today() BETWEEN Colaborador.DtInicioVigencia and Colaborador.DtFimVigencia

		INNER JOIN DWBI_DIME.TbDimTempoData TempoData
			ON  TempoData.DtTempo BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
			AND TempoData.DtTempo < today()

		LEFT JOIN TbLstCdColaborador LstCdColaborador
			ON  HierarquiaComercial.CdColaborador = LstCdColaborador.CdColaboradorBI

	WHERE HierarquiaComercial.CdCargoColaborador IN (38,209,252)
	  AND HierarquiaComercial.CdEstadoColaborador IN (5,8,18,9,13,6,14,17)

	GROUP BY
		 TempoData.NuAnoMes
		,(TempoData.DtTempo-day(TempoData.DtTempo))+1
		,dateadd(month,1,TempoData.DtTempo)-day(TempoData.DtTempo)
		,HierarquiaCadastro.NmColaborador
		,substring(CAST(HierarquiaCadastro.NuCpfCnpj+100000000000 AS VARCHAR(12)),2,11)
		,Colaborador.CdColaboradorRH
		,HierarquiaComercial.CdColaborador
		,LstCdColaborador.CdColaboradorAtivo
		,LstCdColaborador.LstCdColaboradorInativo
		,HierarquiaComercial.NmCargoColaborador
		,HierarquiaCadastro.DsLogin
	ORDER BY
		 HierarquiaCadastro.NmColaborador
		,TempoData.NuAnoMes
)
;
run;
quit;

	,EstruturaProspectores.CdColaboradorBI
	,EstruturaProspectores.CdColaboradorAtivo
	,EstruturaProspectores.LstCdColaboradorInativo
	,EstruturaProspectores.NmCargoColaborador
	,EstruturaProspectores.DsLogin
	,EstruturaProspectores.QtDiasAtividadeNormal
	,EstruturaProspectores.QtDiasFerias
	,EstruturaProspectores.QtDiasOutros
	,case when Last.DsLogin is not null then 1 else 0 end as FlUltRegistro

from TbEstruturaProspectores EstruturaProspectores

	left join basegrc.TbProspectoresProprietario ProspectoresProprietario
		on  EstruturaProspectores.DsLogin = ProspectoresProprietario.DsLogin

	left join TbEstruturaProspectoresLast Last
		on  EstruturaProspectores.NuAnoMes = Last.NuAnoMes
		and EstruturaProspectores.DsLogin = Last.DsLogin

order by
	 EstruturaProspectores.NmColaborador
	,EstruturaProspectores.NuAnoMes
