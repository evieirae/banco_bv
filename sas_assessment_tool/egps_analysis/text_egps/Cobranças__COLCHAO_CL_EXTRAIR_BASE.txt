                    MAX(CASE WHEN b.apflag = 'C' THEN b.apduedt END) AS MAXVencPgto,

                    AND CAST(b.apdetid AS INT64) = CAST(c.pbnumparc AS INT64)
                WHERE
                    CAST(a.ahprd AS INT64) > 0
                    AND DATE(a.ahdt) <= DATE_SUB(DATE_ADD(&dt_ini_cart., INTERVAL 1 MONTH), INTERVAL 1 DAY)
                GROUP BY
                    a.ahid,
                    a.ahdt,
                    a.ahprd,
                    a.ahdtqb,
                    a.ahtype
            ) AS P1
    ) AS P2
JOIN

    ON ac.ahid = P2.NuAcordo

    ON ac.ahcollid = ag.agagency

    bv-pdbd-prd.trusted_rcvry.agpmtdet_esp AS PC

    AND DATE(pc.apduedt) BETWEEN P2.DtInicio AND P2.DtFim
LEFT JOIN
    bv-pdbd-prd.trusted_rcvry.uda3_esp AS U3
    ON ac.ahacct = u3.u3acct
LEFT JOIN
    bv-pdbd-prd.trusted_rcvry.tb_pagto_boleto AS PB
    ON CAST(pc.apahid AS INT64) = CAST(pb.pbnumacordo AS INT64)
    AND CAST(pc.apdetid AS INT64) = CAST(pb.pbnumparc AS INT64)
WHERE
    P2.QdPMTPgto > 0
    AND (DATE(P2.DtQuebra) >= &dt_ini_cart. OR P2.DtQuebra IS NULL)

    FORMAT_TIMESTAMP('%Y-%m-%d %H:%M:%S', P2_2.DtQuitacao) AS DtQuitacao,
    FORMAT_TIMESTAMP('%Y-%m-%d %H:%M:%S', (
        SELECT
            MIN(A_INNER2.AHDT)
        FROM
            bv-pdbd-prd.trusted_rcvry.agrhdr AS A_INNER2
        WHERE
            ac2.ahacct = A_INNER2.AHACCT
            AND A_INNER2.AHDT >= P2_2.DtQuebra
    )) AS DtPromessa
FROM
    (
        SELECT
            P1_2.DtInicio AS DtInicio,
            P1_2.DtFim AS DtFim,
            P1_2.Tipo AS Tipo,
            P1_2.NuAcordo AS NuAcordo,
            P1_2.DtCriacao AS DtCriacao,
            P1_2.Plano AS Plano,
            P1_2.DtQuebraAcordo AS DtQuebraAcordo,

            P1_2.MAXDtCicloPgto AS MAXDtCicloPgto,
