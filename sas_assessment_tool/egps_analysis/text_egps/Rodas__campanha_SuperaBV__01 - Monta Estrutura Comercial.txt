/*****************************************/

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
CREATE TABLE TbAdmissaoColaborador AS
SELECT
	NmSubEstruturaComercial
	,NmAreaEstruturaComercial
	,CdColaborador
	,NmColaborador
	,DsLogin
	,DsEmail
	,NmCargoColaborador
	,CdEstadoColaborador
	,NmEstadoColaborador
	,DtAdmissaoColaborador
	,DtDesligamentoColaborador
	,DATEPART(DtDesligamentoColaborador) as DtDesligamentoColaborador2 FORMAT DATE9.

FROM connection to dbcon
	(
		SELECT * FROM DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
		WHERE today() BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
		AND YEAR(HierarquiaComercial.DtInicioVigencia) BETWEEN YEAR(today())-1 AND YEAR(today())
/*		AND YEAR(DtDesligamentoColaborador) BETWEEN YEAR(today())-1 AND YEAR(today())*/
		AND CdEstruturaComercial = 8 /*and HierarquiaComercial.CdEstadoColaborador = 5 EM ATIVIDADE NORMAL*/

		ORDER BY DtDesligamentoColaborador desc
	)

/*where dslogin = 'aarao.souza'*/

;run;
quit;

DATA ESTRUTURA_COMERCIAL;
SET BASES.TBESTRUTURACOMERCIAL;
WHERE NUANOMES = &Base_Mes. and NmCanalEstruturaComercial in ('Canal Veiculos', 'Canal Negocios') and dsloginn1 <> ''
;RUN;

DATA BASE_FOCAL;
SET BASEGRC.TBCOLABORADORFOCAL;
WHERE NmGrupoProdutoMeta = 'Leves';
;run;

PROC SQL;
CREATE TABLE ESTRUTURA_COMERCIAL_01 AS
SELECT

A.NUANOMES,
A.NMCANALESTRUTURACOMERCIAL,
A.NmSubEstruturaComercialH,
A.NmSubEstruturaComercialA,
a.NmAreaEstruturaComercial,
A.LstSubstituicao,
b.LstColaboradorFocal,
A.DsLoginN3H,
A.DSLOGINN3A,
A.DsLoginN2H,
A.DSLOGINN2A,
A.DSLOGINN1,
A.NmColaborador,
C.DtAdmissaoColaborador,
C.NmEstadoColaborador,
a.IdMatricula,
a.CdColaboradorAtivo,
a.NmGrupoCargo

FROM WORK.ESTRUTURA_COMERCIAL A

LEFT JOIN work.BASE_FOCAL as b
		ON  a.CdFilial = b.CdFilial

LEFT JOIN WORK.TbAdmissaoColaborador C ON C.DsLogin = A.DSLOGINN1

/*WHERE NUANOMES = &NuAnoMes.*/

ORDER BY A.NmAreaEstruturaComercial

;RUN;
