PROC SQL;
CREATE TABLE BASE_INCENTIVO_FINAL AS
SELECT

A.NuAnoMes,
A.CdColaborador,
A.NmGrupoCargoColaborador,
A.DsLoginN1,

SUM(A.VrComissaoGR) AS VrComissaoGR_Total FORMAT COMMAX20.2,
B.VrComissaoDiretaGR AS VrComissaoDiretaGR_MC FORMAT COMMAX20.2,

SUM(VrAtingimentoPonderado) as ICM FORMAT percent9.2

FROM ICOM.TBGRCOMISSAOPERF A

LEFT JOIN ICOM.TBGRCOMISSAODIRETA B ON B.NUANOMES = A.NUANOMES AND B.DsLoginN1 = A.DsLoginN1 and B.CdColaborador = A.CdColaborador AND B.NmGrupoCargoColaborador = A.NmGrupoCargoColaborador

GROUP BY 1,2,3,4,6

;RUN;

	end;
run;

PROC SQL;
CREATE TABLE BASE_INCENTIVO_FINAL_01 AS
SELECT

A.NuAnoMes,

B.NmCanalEstruturaComercial,
B.NmEstruturaComercialA,
B.NmSubEstruturaComercialA,
B.NmAreaEstruturaComercial,
B.LstSubstituicao,
B.LstColaboradorFocal,
B.DsLoginN3A,
CASE WHEN B.DsLoginN2A = '' THEN 'cadeira.vazia' ELSE B.DsLoginN2A END AS DsLoginN2A,

A.CdColaborador,
A.NmGrupoCargoColaborador,
A.DsLoginN1,

A.VrComissaoGR_Total FORMAT COMMAX20.2,
A.VrComissaoDiretaGR_MC FORMAT COMMAX20.2,
A.VrComissaoGR_Total + A.VrComissaoDiretaGR_MC as VrComissaoFinal FORMAT COMMAX20.2,

CASE WHEN A.ICM > 2 THEN 2 ELSE A.ICM END AS ICM FORMAT percent9.2, /*REGRA DE ICM MÁXIMO DE 200%*/
CASE WHEN CALCULATED ICM * calculated VrComissaoFinal > 14000 THEN 14000 ELSE CALCULATED ICM * calculated VrComissaoFinal END AS VrICM_ComissaoFinal FORMAT COMMAX20.2, /*REGRA DE COMISSÃO MAXIMA DE R$ 14 MIL SEM DSR*/

C.VrComissaoSeguros FORMAT COMMAX20.2,
C.VrComissaoNaPista FORMAT COMMAX20.2,
C.VrComissaoLevesBVO FORMAT COMMAX20.2,
C.VrComissaoProduto FORMAT COMMAX20.2,

CALCULATED VrICM_ComissaoFinal + C.VrComissaoProduto AS VrComissaoTotal FORMAT COMMAX20.2

FROM BASE_INCENTIVO_FINAL A

LEFT JOIN BASEGRC.TBGRINCENTIVO B ON B.DsLoginN1 = A.DsLoginN1 AND B.NUANOMES = A.NUANOMES
LEFT JOIN WORK.BASE_COMISSAO_PRODUTO C ON C.DsLoginN1 = A.DsLoginN1 AND C.NUANOMES = A.NUANOMES

/*GROUP BY 1,2,3,4,6*/
ORDER BY 1,4,5,12

;RUN;
