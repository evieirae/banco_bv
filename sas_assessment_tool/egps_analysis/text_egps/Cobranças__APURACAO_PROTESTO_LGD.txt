PROC SQL;
CREATE TABLE CUBO_FIM AS
SELECT
    SUM(saldo_m0) AS saldo,
    COUNT(DISTINCT nucontrato) AS qtde,
    CASE
        WHEN MISSING(PRODUTO)
             OR PRODUTO IN ('CVG_LEVES','CVG_PESADOS')
        THEN 'EGV'
        ELSE PRODUTO
    END AS PRODUTO,
    FL_TESTE_CONTROLE,
    DATA_VIGENCIA,
    ANOMES,
    FL_PAGAMENTO,
    FAIXA_ATRASO_MARC

FROM BASE_FINAL

GROUP BY
    CALCULATED PRODUTO,
    ANOMES,
    FL_TESTE_CONTROLE,
    FL_PAGAMENTO,
    FAIXA_ATRASO_MARC,
    DATA_VIGENCIA
;QUIT;

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
    A.*,
    D.SALDO AS SALDO_M0,

    B.SALDO AS SALDO_M1,

    C.SALDO AS SALDO_M2,

    
    CASE WHEN A.dt_ultimo_recebimento2 IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO,


	CASE
		WHEN Atraso_Elegibilidade between 1 and 30 THEN '01. 1-30'
		WHEN Atraso_Elegibilidade between 31 and 60 THEN '02. 31-60'
		WHEN Atraso_Elegibilidade between 61 and 90 THEN '03. 61-90'
		WHEN Atraso_Elegibilidade between 91 and 120 THEN '04. 91-120'
		WHEN Atraso_Elegibilidade between 121 and 180 THEN '05. 121-180'
		WHEN Atraso_Elegibilidade between 181 and 360 THEN '06. 181-360'
		WHEN Atraso_Elegibilidade > 360 THEN '07. Over360'
		ELSE 'OUTRA'
	END AS FAIXA_ATRASO_MARC,

	PUT(A.DATA_VIGENCIA, yymmn6.) AS SAFRA

FROM BASE_UNICA_FIM A
LEFT JOIN SISTEMA_TOTAL D ON PUT(A.NUCONTRATO, 14.) = D.NUCONTRATO AND A.DATA_M0 = D.REF
LEFT JOIN SISTEMA_TOTAL B ON PUT(A.NUCONTRATO, 14.) = B.NUCONTRATO AND A.DATA_M1 = B.REF
LEFT JOIN SISTEMA_TOTAL C ON PUT(A.NUCONTRATO, 14.) = C.NUCONTRATO AND A.DATA_M2 = C.REF


;QUIT;

proc sort data=BASE_FINAL;
    by nucontrato nuparcela descending dt_ultimo_recebimento2;
run;

proc sort data=BASE_FINAL nodupkey;
    by nucontrato nuparcela;
run;

%MACRO CICLO_SALDO_GCON;


%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA., %str(*)));


%do i = 1 %to &NUM_ELEMENTOS.;

    
    %let ref_tab = %scan(&LISTA., &i., *);

    
    DATA _NULL_;
        MES = INPUT(scan("&LISTA.", &i., "*"), yymmdd10.);
        MES_POS = PUT(INTNX("MONTH", MES, 1, "e"), yymmddn8.);
        
        CALL SYMPUT("REF", PUT(MES, yymmdd10.));
		CALL SYMPUT("REF2", PUT(MES, yymmddn8.));
        CALL SYMPUT("REF_POS", PUT(INTNX("MONTH", MES, 1, "e"), yymmdd10.));
        CALL SYMPUT("REF_TAB_POS", MES_POS);
    RUN;

    %put &REF;
    %put &REF_POS;
    %put &REF_TAB_POS;

    
    PROC SQL;
        %CONNECT_PERM;
        CREATE TABLE GCON_&ref_tab. AS
        SELECT
            A.*,
            "&REF2." AS REF
        FROM CONNECTION TO PERM (
            WITH QUERY_1 AS (
                SELECT
                    a.nucontrato,
                    c.cd_produto,
                    c.cd_modalidade_produto,
                    c.dt_contrato,
                    c.cd_cpf_cnpj,
                    d.cdsubproduto,
                    H.cdgruposubproduto,
                    a.cdsituacaocontabil,
                    a.cdratingatual,
                    MAX(a.nudiasatraso) AS QTATRASO,
                    SUM(CAST(a.vrpermanenciaapropriado AS FLOAT64) + CAST(a.vrtaxacontrato AS FLOAT64)) AS SALDO,
                    SUM(CAST(a.vrpdd AS FLOAT64)) AS PDD_CONTABIL
                FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS a
                LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS c
                    ON a.nucontrato = c.cd_contrato
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta AS d
                    ON a.nucontrato = d.nucontratogestao AND d.stproposta = 'AP'
                LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbsubprodutogrupo AS G

PROC SQL;
CREATE TABLE BASE_FINAL AS
SELECT
    A.*,
    D.SALDO AS SALDO_M0,
    B.SALDO AS SALDO_M1,
    C.SALDO AS SALDO_M2,

    
	CASE WHEN A.dt_ultimo_receb_pos_protesto IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO_POS_PROTESTO,

	 
    CASE WHEN A.dt_ultimo_receb_parcela IS MISSING THEN 0 ELSE 1 END AS FL_PAGAMENTO_PARCELA,

    
    CASE
        WHEN A.dt_ultimo_receb_pos_protesto IS MISSING THEN -100
        ELSE (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA)
    END AS QT_DIAS_REFENVIO,

    
    CASE
        WHEN (CASE WHEN A.dt_ultimo_receb_pos_protesto IS MISSING THEN -100 ELSE (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA) END) = -100 THEN '00. Sem pagamento'

        WHEN (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA) <= 30 THEN '03. 21-30 dias'
        WHEN (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA) <= 60 THEN '04. 31-60 dias'
        WHEN (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA) <= 90 THEN '05. 61-90 dias'
        WHEN (A.dt_ultimo_receb_pos_protesto - A.DATA_VIGENCIA) <= 120 THEN '06. 91-120 dias'
        ELSE '07. 120+ dias'
    END AS FAIXA_REFENVIO,

 
	CASE
    	WHEN TODAY() - DATA_VIGENCIA >= 30 THEN 1
    	ELSE 0
	END AS FLAG_30_DIAS,


	CASE
		WHEN A.Atraso_Elegibilidade between 1 and 30 THEN '01. 1-30'
		WHEN A.Atraso_Elegibilidade between 31 and 60 THEN '02. 31-60'
		WHEN A.Atraso_Elegibilidade between 61 and 90 THEN '03. 61-90'
		WHEN A.Atraso_Elegibilidade between 91 and 120 THEN '04. 91-120'
		WHEN A.Atraso_Elegibilidade between 121 and 180 THEN '05. 121-180'
		WHEN A.Atraso_Elegibilidade between 181 and 360 THEN '06. 181-360'
		WHEN A.Atraso_Elegibilidade > 360 THEN '07. Over360'
		ELSE 'OUTRA'
	END AS FAIXA_ATRASO_MARC,

    
    CASE
        WHEN TODAY() - A.DATA_VIGENCIA >= 60 THEN 'Maturada'
        ELSE 'Nao maturada'
    END AS FL_MATURACAO,

    PUT(A.DATA_VIGENCIA, YYMMN6.) AS SAFRA,

	E.cd_cpf_cnpj,
	E.PRODUTO AS PRODUTO_2

PROC SQL;
%CONNECT_PERM;

CREATE TABLE HIST_PROTESTO AS
SELECT *
FROM CONNECTION TO perm
(

WITH BASE AS (

    SELECT
         A.nucontratogestao
        ,A.sqcontratofinanceiro
        ,C.cd_cpf_cnpj AS nucpfcnpj

        ,MIN(
            CASE
                WHEN DATE(D.DtProcessamentoLiquidacao) >= DATE(A.dtgravacaoscore)
                THEN D.DtProcessamentoLiquidacao
            END
        ) AS DT_LIQUIDACAO

        ,MIN(
            CASE
                WHEN DATE(R.dt_baixa) >= DATE(A.dtgravacaoscore)
                THEN R.dt_baixa
            END
        ) AS DT_BAIXA_RENEG

        ,A.dtgravacaoscore

        ,EXTRACT(YEAR FROM A.dtgravacaoscore) * 100
        + EXTRACT(MONTH FROM A.dtgravacaoscore) AS SAFRA_PROTESTO

        ,MIN(
            CASE
                WHEN B.St_Parcela <> 'Q'
                  OR (
                        B.St_Parcela = 'Q'
                    AND B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= DATE(A.dtgravacaoscore)
                     )
                THEN B.Nu_parcela
            END
        ) AS PARCELA_REF

        ,MIN(
            CASE
                WHEN B.St_Parcela <> 'Q'
                  OR (
                        B.St_Parcela = 'Q'
                    AND B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= DATE(A.dtgravacaoscore)
                     )
                THEN B.dt_vencimento
            END
        ) AS DT_PARCELA_REF

        
        ,MIN(
            CASE
                WHEN P.DT_RECEBIMENTO >= DATE(A.dtgravacaoscore)
                THEN P.DT_RECEBIMENTO
            END
        ) AS DT_PRIMEIRO_PAGAMENTO

        
        ,CASE
            WHEN MIN(
                    CASE
                        WHEN P.DT_RECEBIMENTO >= DATE(A.dtgravacaoscore)
                        THEN P.DT_RECEBIMENTO
                    END
                 ) IS NOT NULL
            THEN 1
            ELSE 0
         END AS FL_PAGAMENTO

        ,MIN(
            CASE
                WHEN DATE(K.DtInclusao) >= DATE(A.dtgravacaoscore)
                THEN DATE(K.DtInclusao)
            END
        ) AS DT_BNDU

        ,CASE
            WHEN MIN(
                    CASE
                        WHEN DATE(K.DtInclusao) >= DATE(A.dtgravacaoscore)
                        THEN DATE(K.DtInclusao)
                    END
                 ) IS NOT NULL
            THEN 1
            ELSE 0
         END AS FL_BNDU

        ,MAX(
            CASE
                WHEN DATE(K.DtInclusao) >= DATE(A.dtgravacaoscore)
                THEN
                    CASE
                        WHEN TP.TpApreensao IN (1,2) THEN 'BUSCA'
                        WHEN TP.TpApreensao = 3 THEN 'EA_C_QUITACAO'
                        WHEN TP.TpApreensao = 4 THEN 'EA_S_QUITACAO'
                        WHEN TP.TpApreensao = 5 THEN 'EA_CONVER'
                        WHEN TP.TpApreensao = 6 THEN 'ENTREGA_DIRETA'
                        ELSE 'NA'
                    END
            END
        ) AS TP_BNDU

    FROM bv-pdbd-prd.trusted_dbcobranca.tbscorecobranca A

    LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela B
        ON A.nucontratogestao = B.cd_contrato
       AND A.sqcontratofinanceiro = B.Sq_Contrato_Financeiro

    LEFT JOIN bv-pdbd-prd.dvry_core.tbparcelarecebimento P
        ON A.nucontratogestao = P.CD_CONTRATO
       AND A.sqcontratofinanceiro = P.SQ_CONTRATO_FINANCEIRO
       AND CAST(P.VR_RECEBIDO AS FLOAT64) NOT IN (0,0.01)
       AND CAST(P.FL_ESTORNADO AS STRING) <> 'S'

    LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato C
        ON A.nucontratogestao = C.CD_CONTRATO

    LEFT JOIN bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil D
        ON A.nucontratogestao = D.NUCONTRATO
       AND A.sqcontratofinanceiro = D.SQCONTRATOFINANCEIRO

    LEFT JOIN bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador R
        ON A.nucontratogestao = R.nucontratogestao
       AND A.sqcontratofinanceiro = R.SqContratoFinanceiro
       AND R.DT_CANCELAMENTO IS NULL
       AND R.ST_RENEG = 3

    LEFT JOIN bv-pdbd-prd.trusted_dbpatio.tbentrada E
        ON A.nucontratogestao = E.NuContratoGestao
       AND A.sqcontratofinanceiro = E.SqContratoFinanceiro

    LEFT JOIN bv-pdbd-prd.trusted_dbpatio.tbbem K
        ON E.CdBem = K.CdBem

    LEFT JOIN bv-pdbd-prd.trusted_dbcontencioso.tbprocessojuridico TP
        ON E.NuContratoGestao = TP.nucontrato

    WHERE A.CDMODELO = 101

    GROUP BY
         A.nucontratogestao
        ,A.sqcontratofinanceiro
        ,C.cd_cpf_cnpj
        ,A.dtgravacaoscore
        ,SAFRA_PROTESTO

)

SELECT
     B.*

    ,MAX(
        CASE
            WHEN DATE_DIFF(
                    DATE(G.DTPROCESSAMENTO),
                    DATE(B.DT_PRIMEIRO_PAGAMENTO),
                    DAY
                 ) BETWEEN -5 AND -1
            THEN G.NuDiasAtraso
        END
    ) AS NuDiasAtraso_PRE_PAG

    ,MAX(
        CASE
            WHEN DATE_DIFF(
                    DATE(G.DTPROCESSAMENTO),
                    DATE(B.DT_PRIMEIRO_PAGAMENTO),
                    DAY
                 ) = 0
            THEN G.NuDiasAtraso
        END
    ) AS NuDiasAtraso_DIA_PAG

    ,MIN(
        CASE
            WHEN DATE_DIFF(
                    DATE(G.DTPROCESSAMENTO),
                    DATE(B.DT_PRIMEIRO_PAGAMENTO),
                    DAY
                 ) BETWEEN 1 AND 5
            THEN G.NuDiasAtraso
        END
    ) AS NuDiasAtraso_POS_PAG

    ,S.SALDO

FROM BASE B

LEFT JOIN bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil G
    ON B.nucontratogestao = G.nucontrato
   AND B.sqcontratofinanceiro = G.sqcontratofinanceiro
   AND DATE(G.DTPROCESSAMENTO)
       BETWEEN DATE(B.DT_PRIMEIRO_PAGAMENTO) - 5
           AND DATE(B.DT_PRIMEIRO_PAGAMENTO) + 5

LEFT JOIN (
    SELECT
         DATE(DTPROCESSAMENTO) AS DTPROCESSAMENTO
        ,NUCONTRATO
        ,SQCONTRATOFINANCEIRO
        ,SUM(
            CAST(VRTAXACONTRATO AS NUMERIC)
            + CAST(VRPERMANENCIAAPROPRIADO AS NUMERIC)
         ) AS SALDO
    FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
    GROUP BY 1,2,3
) S
    ON B.nucontratogestao = S.NUCONTRATO
   AND B.sqcontratofinanceiro = S.SQCONTRATOFINANCEIRO
   AND S.DTPROCESSAMENTO = DATE(B.dtgravacaoscore)

GROUP BY
     B.nucontratogestao
    ,B.sqcontratofinanceiro
    ,B.nucpfcnpj
    ,S.SALDO
    ,B.DT_LIQUIDACAO
    ,B.DT_BAIXA_RENEG
    ,B.dtgravacaoscore
    ,B.SAFRA_PROTESTO
    ,B.PARCELA_REF
    ,B.DT_PARCELA_REF
    ,B.DT_PRIMEIRO_PAGAMENTO
    ,B.FL_PAGAMENTO
    ,B.DT_BNDU
    ,B.FL_BNDU
    ,B.TP_BNDU

)
;

    A.*,

    min(

    ) format=date9. as DT_PRIMEIRO_EVENTO,

        when calculated DT_PRIMEIRO_EVENTO = . THEN 'Sem Acao'

        when calculated DT_PRIMEIRO_EVENTO = B.DT_BNDU
            then 'BNDU'

        when calculated DT_PRIMEIRO_EVENTO = DT_PRIMEIRO_PAGAMENTO
             and B.NuDiasAtraso_POS_PAG > B.NuDiasAtraso_PRE_PAG
            then 'Pagamento Parcial (Pulada)'

        when calculated DT_PRIMEIRO_EVENTO = DT_PRIMEIRO_PAGAMENTO
             and coalesce(B.NuDiasAtraso_POS_PAG,0) > 0
            then 'Pagamento Parcial'

        when calculated DT_PRIMEIRO_EVENTO = DT_PRIMEIRO_PAGAMENTO
            then 'Atualizacao'

        else 'Sem Acao'

    end as TIPO_RECEBIMENTO length=30,

    
    1 as QTD,

    
    case
        when calculated TIPO_RECEBIMENTO ne 'Sem Acao'
        then 1
        else 0
    end as QTD_I1,

    
    case
        when calculated TIPO_RECEBIMENTO in ('Quitacao','Atualizacao')
        then 1
        else 0
    end as QTD_I2,

    
    case
        when calculated TIPO_RECEBIMENTO ne 'Sem Acao'
        then SALDO
        else 0
    end as SALDO_I1,

    
    case
        when calculated TIPO_RECEBIMENTO in ('Quitacao','Atualizacao')
        then SALDO
        else 0
    end as SALDO_I2

from BASE_FINAL A
LEFT JOIN HIST_PROTESTO B ON
A.NUCONTRATO = INPUT(B.nucontratogestao, 14.) AND
A.SqContratoFinanceiro = B.sqcontratofinanceiro
;quit;
