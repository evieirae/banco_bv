FILENAME MAIL_INI EMAIL
	FROM=("inf-gerenciais@bv.com.br")
/*	TO=("fernanda.scarpelli@bv.com.br" "inf-gerenciais@bv.com.br" "mariana.cardoso@bv.com.br" "andressa.souza@bv.com.br" "hellen.dantas@bv.com.br" "diego.alves@bv.com.br")*/

	FILE MAIL_INI;
		PUT "Processamento da base iniciado.";

/* Executar código de geração da base */
%include '/infogrc/funil_pricing/egv/vm/code/01 - Libs.sas';

/* Calcular tempo de processamento total */
%LET W_fim = %SYSFUNC(datetime());
%LET W_dur = %SYSFUNC(intck(SECOND, &W_ini, &W_fim), tod8.);

/* Enviar e-mail de finalização da atualização */

PROC SQL;
CREATE TABLE STATUSSIM AS
SELECT
	TpDados,
	MAX(DtSimulacaoInicial) FORMAT=ddmmyy10. AS DtSimulacaoInicial
FROM SOLITMP.FUNIL_PRICING_EGV
where TpDados = 'Simu_Leves_Msg'
group by 1
;RUN;

OPTIONS EMAILHOST=smtprelay.bvnet.bv EMAILPORT=25 EMAILAUTHPROTOCOL=NONE;

FILENAME MAIL_FIM EMAIL
	FROM=("inf-gerenciais@bv.com.br")
		TO=("gustavo.santos@bv.com.br")
		TYPE="text/html"
    SUBJECT="Funil Pricing EGV | Fim da atualizacao";

ODS html body=MAIL_FIM style=noline rs=none;

PROC REPORT DATA=STATUSSIM nowd HEADLINE HEADSKIP
style (report) = {background = none font_face = "Calibri" font_size = 9pt just=left OUTLINECOLOR = none}
style (column) = {background = none CELLHEIGHT = 2.5% font_face = "Calibri" font_size = 9pt just=center OUTLINECOLOR = none}
style (header) = {foreground = white font_face="Calibri" font_size = 9pt just=center background = #69c3cf OUTLINECOLOR = none} ;

COLUMNS TpDados DtSimulacaoInicial;
RUN;

PROC ODSTEXT;
   p "Prezados,";
   p "";
   p "A base foi atualizada.";
   p "O processo demorou &W_dur. para ser concluido.";
   P "";
   p "Atenciosamente,";
   p "Informacoes Gerenciais Estrategicas";
RUN;
