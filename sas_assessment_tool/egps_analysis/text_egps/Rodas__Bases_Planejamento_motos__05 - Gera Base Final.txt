proc sql;
create table Share as
select distinct
Chave1,
case when Sum(QtdContratos_ProjSab) / Sum(qtgravamesplayers_ProjSab) > 0.5 then 0.5
else Sum(QtdContratos_ProjSab) / Sum(qtgravamesplayers_ProjSab) end as Share
from POTENCIAL_RODAS_04_M
where  1=1
group by 1
;run;

proc sql;
create table BasePotencialFinal_M as
select
A.Chave1,
A.cdparceirocomercial,
B.Share,
a.qtgravamesplayers_ProjSab,
B.Share *qtgravamesplayers_ProjSab AS ContratosIdeais,
C.ValorFinal AS TktFinal,
C.ValorFinal * calculated ContratosIdeais as ProducaoIdeal
from POTENCIAL_RODAS_04_M as A

	left join Share as B
	on A.Chave1 = b.Chave1

	left join BASE_TICKET_FINAL_M as c
	on A.ChaveCdParceiro_FxIdade = c.ChaveCdParceiro_FxIdade
where  1=1
;run;

proc sql;
create table ProducaoIdealMotos as
select
A.cdparceirocomercial,
Sum(a.qtgravamesplayers_ProjSab) as Gravames,
Sum(a.ContratosIdeais) as Contratos,
Sum(ProducaoIdeal) as Producao
from BasePotencialFinal_M as A
where  1=1
/*and a.NuCnpjLoja = '08330275000103'*/
group by 1
;run;
