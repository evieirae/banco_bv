CREATE TABLE TbMISAcordo AS
SELECT
 *
,CASE

  AND DTPGTO < DTINICIO AND DTCICLOPGTO < DTINICIO	THEN 0

PROC SQL ;
CREATE TABLE TbMISAcordo AS
SELECT
 *
,CASE
 WHEN PERFIL = 'COLCHAO'					THEN 1 ELSE 0 END 	AS FlEInicio

,CASE
 WHEN R_CxMes = 1
  AND DTCICLOPGTO IS NOT NULL
  AND DTCICLOPGTO <= DtLimite				THEN 1 ELSE 0 END 	AS FlPago

,CASE
 WHEN CALCULATED FlPago = 1 AND PLANO = NUPARCELA	THEN 1
 WHEN DTQUITACAO BETWEEN DTINICIO AND DTFIM			THEN 1
 ELSE 0 END 													AS FlQuitacao

,CASE
 WHEN DTQUEBRA BETWEEN DTINICIO AND DTFIM	THEN 1 ELSE 0 END 	AS FlQuebra

,CASE
 WHEN CALCULATED FlQuebra = 1
  AND DTPROMESSA IS NOT NULL
  AND DTPROMESSA  = DTQUEBRA 				THEN 1 ELSE 0 END 	AS FlPromessa

FROM TbMISAcordo

ORDER BY DTCRIACAO , NUACORDO, NUPARCELA
;QUIT;RUN;

PROC SQL ;
CREATE TABLE TbMISAcordo AS
SELECT
 A.*
,B.QdPMTMes
,CASE
 WHEN A.DTVENCIMENTO = B.DTVENCIMENTO 	THEN 1 ELSE 0 END 	AS QdAcordo

,CASE

,CALCULATED QdAcordo * B.QdQuitacao							AS QdQuitacao
,CALCULATED QdAcordo * B.QdQuebra							AS QdQuebra
,CALCULATED QdAcordo * A.VrAcordo							AS VrCarteira
,A.R_CxMes * A.VRPARCELA 									AS VrCaixa
,CASE
 WHEN A.R_CxMes = 1 THEN DAY(A.DTVENCIMENTO) END 			AS DiaEntradaCx

FROM TbMISAcordo   	AS A
LEFT JOIN TbResumo	AS B
  ON A.DtInicio 	= B.DtInicio
 AND A.NUACORDO		= B.NUACORDO
 AND A.CDAGENTE		= B.CDAGENTE
;QUIT;RUN;

PROC SQL;
CREATE TABLE TbR_Acordo AS
SELECT
 DataDU
,Du
,NMAGENTE
,TpLancamento
,DsFase
,FxPlano
,AVG(Fator)								AS Fator
,SUM(QdAcordo)							AS QdAcordo
,SUM(VrAcordo * QdAcordo )				AS VrAcordo

,AG.*
,PL.*

	,(SELECT DISTINCT NMAGENTE 				FROM TbR_Acordo) AS AG
    ,(SELECT DISTINCT DsFase,FxPlano	 	FROM TbR_Acordo) AS PL

PROC SQL;
CREATE TABLE TbE_Demais AS
SELECT
 RF.DataDU
,RF.Du
,RF.NMAGENTE
,"ESTOQUE"	AS TpLancamento	LENGTH = 20
,RF.DsFase
,RF.FxPlano
,1			AS Fator
,SUM(A.QdAcordo * A.Fator) AS QdAcordo
,SUM(A.VrAcordo * A.Fator) AS VrAcordo

  ON A.NMAGENTE = RF.NMAGENTE
 AND A.DsFase   = RF.DsFase

 AND A.Du       < RF.Du
 AND A.TpLancamento NOT IN ("ESTOQUE INICIAL")
GROUP BY  RF.DataDU,RF.Du,RF.NMAGENTE,RF.DsFase,RF.FxPlano
ORDER BY RF.NMAGENTE, RF.Du
;QUIT;RUN;
