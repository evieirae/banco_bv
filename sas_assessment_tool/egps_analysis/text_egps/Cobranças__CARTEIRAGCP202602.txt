CREATE TABLE TbUFContrato	AS
SELECT DISTINCT
Tab1.CdPessoa,
Tab1.NuCpfCnpj,
Tab1.SqEndereco,
Tab1.CdUf,
Tab1.NuContrato as  NuContrato FORMAT BEST14.
FROM CARTEIRA.TbCobranca_&AnoMesCarga.	AS RF
    ,CONNECTION TO PERM ( WITH QUERY_1 AS (

SELECT
    CAST(T1.NuContrato as NUMERIC) AS NuContrato,
    T1.NuCpfCnpj AS NuCpfCnpj,
    T1.CdPessoa AS CdPessoa,
    T1.SqRes AS SqRes,
    T1.SqCorr AS SqCorr,
    T1.SqAtiva AS SqAtiva,
    T1.SqUF AS SqUF,
    CASE
        WHEN T1.SqRes IS NOT NULL THEN T1.SqRes
        WHEN T1.SqCorr IS NOT NULL THEN T1.SqCorr
        WHEN T1.SqAtiva IS NOT NULL THEN T1.SqAtiva
        ELSE T1.SqUF
    END AS SqEndereco
FROM (
    SELECT
        c.cd_contrato AS NuContrato,
        CAST(REPLACE(c.cd_cpf_cnpj, ' ', '') AS FLOAT64) AS NuCpfCnpj,
        c.cd_pessoa AS CdPessoa,
        MAX(CASE WHEN pe.cdtipoendereco = 1 THEN pe.sqendereco END) AS SqRes,
        MAX(CASE WHEN pe.flcorrespondencia = 'S' THEN pe.sqendereco END) AS SqCorr,
        MAX(CASE WHEN pe.flativo = 'S' THEN pe.sqendereco END) AS SqAtiva,
        MAX(pe.sqendereco) AS SqUF
    FROM bv-pdbd-prd.dvry_core.tbcontrato AS C
    LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS PE
      ON c.cd_pessoa = pe.cdpessoa
     AND pe.cduf IN ('RO', 'AC', 'AM', 'RR', 'PA', 'AP', 'TO', 'MA', 'PI', 'CE', 'RN', 'PB' , 'PE', 'AL', 'SE', 'BA', 'MG', 'ES', 'RJ', 'SP', 'PR', 'SC', 'RS', 'MS', 'MT', 'GO', 'DF')
    WHERE c.dt_contrato <= '2026-02-28'
    AND (c.dt_liquidacao_efetiva IS NULL OR c.dt_liquidacao_efetiva >= '2026-02-02')
    GROUP BY
        c.cd_contrato,
        c.cd_cpf_cnpj,
        c.cd_pessoa
) AS T1

),
QUERY_2 AS (

SELECT
    CAST(A.NuContrato AS NUMERIC) AS NuContrato,
    A.CdPessoa AS CdPessoa,
    A.NuCpfCnpj AS NuCpfCnpj,
    A.SqEndereco AS SqEndereco,
    UPPER(pe.cduf) AS CdUf
FROM
    QUERY_1 AS A
INNER JOIN
    bv-pdbd-prd.trusted_dbcor.tbpessoaendereco AS PE
ON
    A.CdPessoa = pe.cdpessoa AND A.SqEndereco = pe.sqendereco

)
SELECT * FROM QUERY_2) AS Tab1

WHERE RF.NuContrato = Tab1.NuContrato
;
DISCONNECT FROM PERM;
QUIT;

PROC SQL;
CREATE TABLE TbSLAGDoc as
SELECT
 RF.NuMonitoria
,RF.NuSolicitacao
,RF.NmAssessoria
,RF.NuContrato
,GDOC.DtSLA
,MAX(CASE WHEN GDOC.Fila = "CANCELADO"	THEN 1 ELSE 0 END ) 		AS FlCancelado

,MIN(CASE WHEN GDOC.Fila = "RESPONDIDA"	THEN GDOC.DtEntrada END )	AS MinDtEntrada	FORMAT  DATETIME22.

,DATEPART(CALCULATED MinDtEntrada )									AS DtResposta	FORMAT  DDMMYY8.
,CASE
  WHEN CALCULATED DtResposta  = .
   AND TODAY() <= GDOC.DtSLA				THEN 0

  WHEN CALCULATED DtResposta  <> .
   AND CALCULATED DtResposta <= GDOC.DtSLA	THEN 0
  ELSE 1 END 														AS FlForaSLA

FROM TbOcorrenciaTSA	AS RF
	 ,TBCARGAGDOC	AS GDOC

WHERE RF.NuSolicitacao = GDOC.NuSolicitacao
  AND RF.NmAssessoria  = GDOC.NmAssessoriaNovo
  AND RF.NuContrato	   = GDOC.NuContrato
  AND RF.FlSLA = 0

GROUP BY  RF.NuMonitoria,RF.NuSolicitacao,RF.NmAssessoria,RF.NuContrato,GDOC.DtSLA
ORDER BY NuSolicitacao, NmAssessoria
;QUIT;RUN;
