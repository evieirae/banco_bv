PROC SQL;

CREATE TABLE DGT_BI_1 AS

SELECT distinct
 		  A.TEL AS TEL,
          datepart(a.DATA) format ddmmyy10. as DATA,
		  a.DATA as DATETIME,
/*          A.Canal, */
 		  input(substr(Trim(left(SCAN(tranwrd(SCAN(A.Message,-5,':'),'"',''),1,","))),1,11),11.) AS timestamp,
		  Trim(left(SCAN(tranwrd(SCAN(A.Message,-5,':'),'"',''),1,","))) AS timestamp2,
		  Trim(left(SCAN(SCAN(tranwrd(SCAN(A.Message,-1,'{'),'"',''),2,':'),-2,'}'))) AS Message,
      	  B.TEL AS TEL2,
		  B.DATA AS DATA2,
		  B.Atendente AS NMGR,
		  B.NUCPF,
		  B.OportunitStage,
		  CASE WHEN C.TEL IS NOT NULL AND C.MotivoPerda IS NULL THEN "SEM MARCAÇÃO" ELSE C.MotivoPerda END AS MotivoPerda
FROM

(
SELECT
CASE WHEN length(compress('Send Time Phone Number'n))=12 THEN
INPUT(substr(compress('Send Time Phone Number'n),length(compress('Send Time Phone Number'n))-9,13),13.)
WHEN length(compress('Send Time Phone Number'n))=13 THEN
INPUT(substr(compress('Send Time Phone Number'n),length(compress('Send Time Phone Number'n))-10,14),13.)
END AS TEL,

		 input(put(input('Engagement Date Time'n,DDMMYY10.),DDMMYY10.)||" "||put(input(substr('Engagement Date Time'n,length('Engagement Date Time'n)-4,5), TIME8.),TIME8.),ANYDTDTM32.) format datetime16. AS DATA ,
          'Sender Display Name'n as Canal,
          Message,
          'Sender Code'n as SenderId

     FROM PRIVADO.DGT_CTA_LEADS_RAW2
Where 'Engagement Channel Action'n = "INBOUND"
and SCAN(tranwrd( SCAN(Message,-1,'{'),'"',''),1,':') contains "body"
and Trim(left(SCAN(SCAN(tranwrd(SCAN(Message,-1,'{'),'"',''),2,':'),-2,'}'))) is not null
) A
LEFT JOIN

(SELECT  TEL,
          'Hora de início'n AS DATA,
		  'Conta: CPF/CNPJ'n AS NUCPF,
		  'Oportunidade: Fase'n AS OportunitStage,
          Atendente
FROM PRIVADO.DGT_CTA_GR_RAW) B
ON A.TEL = B.TEL
AND A.DATA = B.DATA
LEFT JOIN
PRIVADO.DGT_CTA_MOTIVO_STAGE C
ON C.TEL = A.TEL AND
C.DTCRIACAO = datepart(a.DATA)

;quit;

PROC SORT DATA=DGT_BI_1
	OUT=DGT_BI_2;
	BY DATA TEL timestamp ;
RUN;
