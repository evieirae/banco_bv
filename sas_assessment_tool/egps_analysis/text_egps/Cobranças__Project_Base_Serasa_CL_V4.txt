PROC SQL;
CREATE TABLE SERASAFEIRAO_4B AS
SELECT

	A.*,
	COALESCE(B.PERFIL_PROP_2023,C.PERFIL_PROP_2023,'Bom') as GH



FROM SERASAFEIRAO_4 A

	ON A.CPF=B.NUCPFCNPJ

LEFT JOIN RAFAEL.&base1_cl_entrante. C
	ON A.CPF=C.NUCPFCNPJ

;QUIT;
