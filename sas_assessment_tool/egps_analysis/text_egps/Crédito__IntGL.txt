PROC SQL;
CONNECT TO SYBASE (SERVER=MORSYBCTR USER=&U_CTR. PASSWORD=&S_CTR.);
CREATE TABLE TbTransacao AS SELECT

A.DtTransacao,
A.CdTransacaoPlataformaComercial,
A.DsUsuarioTransacao,
(CASE A.tptransacaoplataforma
WHEN 'A' THEN 'PRE_ALOCACAO'
WHEN 'R' THEN 'Reenquadramento'
WHEN 'C' THEN 'Cancelamento Total'
WHEN 'E' THEN 'Efetivação'
WHEN 'P' THEN 'Baixa Parcial'
WHEN 'T' THEN 'Baixa Total'
WHEN 'L' THEN 'Realocar'
WHEN 'S' THEN 'Consulta'
WHEN 'I' THEN 'Cancelamento Parcial'
WHEN 'O' THEN 'Alterar Operação'
WHEN 'V' THEN 'Reativar Operação'
ELSE 'Unknown'
END) as TpTransacaoPlataforma,
A.tptransacaoplataforma as CodTpTransacaoPlataforma,
A.CdPessoaClienteEmitente,
A.CdPessoaClienteOperacao,
A.CdTipoEventoOperacao,
A.CdPessoaClienteRisco,
A.CdOperacaoLimiteCredito,
A.CdLinhaCreditoTransacao,
A.TpResultadoTransacao,
A.TpOrigemTransacao,
A.VrPrincipal,
A.VrEncargos,
A.VrFuturo,
A.CdSistema,
A.NuChamadaServico
FROM CONNECTION TO SYBASE
(SELECT
A.DtTransacao,
A.CdTransacaoPlataformaComercial,
A.DsUsuarioTransacao,
A.TpTransacaoPlataforma,
A.CdPessoaClienteEmitente,
A.CdPessoaClienteOperacao,
A.CdTipoEventoOperacao,
A.CdPessoaClienteRisco,
A.CdOperacaoLimiteCredito,
A.CdLinhaCreditoTransacao,
A.TpResultadoTransacao,
A.TpOrigemTransacao,
A.VrPrincipal,
A.VrEncargos,
A.VrFuturo,
A.CdSistema,
A.NuChamadaServico
