PROC SQL;
	CREATE TABLE WORK.PREVIA_ATUALIZACAO_ESTOQUE AS
		SELECT 	t1.NumRegSistContrGest							AS VAC,
				t2.causa,
				t2.MotivoPadrao,
				t1.Valor,
				t2.Valor										AS ValorAnterior,
				(SUM(t1.Valor,-1*t2.Valor))						AS Diferenca

		FROM MODTEMP.PROVISAO_&BaseAtual._&DtAtual. AS t1 INNER JOIN
				PROVISAO.PROVISAO_&BaseAnterior. AS t2 ON (NumRegSistContrGest=causa);

PROC SQL;
   CREATE TABLE WORK.PREVIA_ATUALIZACAO_ESTOQUE AS
	   SELECT 	(COUNT(t1.MotivoPadrao))						AS Quantidade_Motivo,
	          	t1.MotivoPadrao,
	            (SUM(t1.Valor)) 			FORMAT=COMMAX22.2 	AS Soma_Valor,
	            (SUM(t1.ValorAnterior)) 	FORMAT=COMMAX22.2 	AS Soma_ValorAnterior,
	            (SUM(t1.Diferenca)) 		FORMAT=COMMAX22.2 	AS Soma_Diferenca
	      FROM WORK.PREVIA_ATUALIZACAO_ESTOQUE AS t1
	      GROUP BY t1.MotivoPadrao;
QUIT;

PROC SQL;

		SELECT 	(SUM(t1.Quantidade_Motivo))					AS Quantidade_Motivo,

				(SUM(t1.Soma_Diferenca))					AS Soma_Diferenca
		FROM WORK.PREVIA_ATUALIZACAO_ESTOQUE AS t1;

PROC SQL;
CREATE TABLE WORK.PAGAMENTO_VAC AS
   SELECT t1.NumRegSistContrGest							AS VAC,
            (SUM(t1.Valor)) 		FORMAT=COMMAX22.2 		AS Soma_do_Valor
      FROM BDPO.BDPORTAL_PAGAMENTOS t1
      WHERE t1.NumRegSistContrGest CONTAINS 'VAC' AND
			t1.DtOcorPG BETWEEN intnx("month",TODAY(),0,"B") and intnx("month",TODAY(),0,"E") AND
			t1.GestProc = 'SEMELHANTES' AND
			t1.TpVerba = 1
      GROUP BY t1.NumRegSistContrGest;
QUIT;

PROC SQL;
CREATE TABLE WORK.PAGAMENTO_SOMA AS
	SELECT 	'Total'														AS MotivoPadrao,
			(-1*(SUM(Soma_do_Valor))) 	FORMAT=COMMAX22.2 				AS Valor

	FROM WORK.PAGAMENTO_VAC
	;
QUIT;

PROC SQL;
CREATE TABLE WORK.MOTIVO_VAC AS
	SELECT 	t1.VAC,
			t1.Soma_do_Valor,
			t2.MotivoPadrao                          AS MotivoPadrao
	FROM WORK.PAGAMENTO_VAC AS t1 LEFT JOIN
		 BDCIVEL.MASSIFICADO AS t2
	ON (t1.VAC = t2.NumRegSistContrGest)

	ORDER BY t2.MotivoPadrao;
QUIT;

PROC SQL;
CREATE TABLE WORK.PAGAMENTO_VAC AS
	SELECT 	t1.MotivoPadrao,
			(-1*(SUM(t1.Soma_do_Valor))) FORMAT=COMMAX22.2 AS Valor

	FROM WORK.MOTIVO_VAC AS t1
	GROUP BY t1.MotivoPadrao;
QUIT;

PROC SQL;
CREATE TABLE WORK.TOTAL_VAZIO_PAGAMENTO AS
	SELECT t1.MotivoPadrao,
			t1.Valor
	FROM WORK.PAGAMENTO_VAC AS t1
	WHERE t1.MotivoPadrao = "";
QUIT;

PROC SQL;
CREATE TABLE WORK.PAG_RATEIO_AUX AS
	SELECT 	t2.MotivoPadrao,
			(t2.Valor - t1.Valor)	FORMAT=COMMAX22.2	AS Valor

	FROM WORK.PAGAMENTO_SOMA AS t2,
		WORK.TOTAL_VAZIO_PAGAMENTO AS t1;
QUIT;

PROC SQL;
CREATE TABLE WORK.PAG_RATEIO_DIV AS
	SELECT 	t1.MotivoPadrao,
			(t1.Valor/t2.Valor)			AS Valor
	FROM WORK.PAGAMENTO_VAC AS t1,

	WHERE t1.MotivoPadrao <> "";
QUIT;

PROC SQL;
CREATE TABLE WORK.PAG_RATEIO_MULT AS
	SELECT 	t1.MotivoPadrao,
			(t1.Valor*t2.Valor)		FORMAT=COMMAX22.2	AS Valor
	FROM WORK.PAG_RATEIO_DIV AS t1,
			WORK.TOTAL_VAZIO_PAGAMENTO AS t2;
QUIT;

PROC SQL;
CREATE TABLE WORK.PAG_TOTAL_RATEIO AS
	SELECT t1.MotivoPadrao,
			(t1.Valor+t2.Valor)	FORMAT=COMMAX22.2	AS Valor
	FROM WORK.PAG_RATEIO_MULT AS t1,
			WORK.PAGAMENTO_VAC AS t2
	WHERE t1.MotivoPadrao=t2.MotivoPadrao;
QUIT;

PROC SQL;
CREATE TABLE WORK.MEDIA_PAGAMENTO AS
	SELECT
			t1.MotivoPadrao,
			t1.Valor/t2.DIAS_UTEIS 	AS MEDIA_VALOR

	FROM WORK.PAG_TOTAL_RATEIO AS t1,
			WORK.DIAS_UTEIS_CORRIDO	AS t2;
QUIT;

PROC SQL;
CREATE TABLE WORK.PREVIA_PAGAMENTO AS

PROC SQL;
CREATE TABLE WORK.TOTAL_PAGAMENTO AS

		'Total' 										AS MotivoPadrao,
