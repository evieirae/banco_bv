%MACRO CalculaData(DataRef=, Limite=, DifMes=);
		%let Mes = %sysfunc(month(&DataRef.));
		%let Ano = %sysfunc(year(&DataRef.));

		
		%if (&Mes. - &DifMes.) > 0 %then %do;
			%let MesOut = (&Mes. - &DifMes.);
			%let AnoOut = &Ano.;
		%end;
		%else %do;
			%let MesOut = %sysevalf(&Mes. - &DifMes. + 12);
			%if &MesOut. = 0 %then %do;
				%let MesOut = 1;
			%end;
			%put &MesOut.;
			%let AnoOut = %sysevalf(&Ano. - 1);
		%end;

		
		%if "&Limite." = "INFERIOR" %then %let DiaOut = 01;
		%else %do;
			%if &MesOut = 01 or
			   &MesOut = 03 or
			   &MesOut = 05 or
			   &MesOut = 07 or
			   &MesOut = 08 or
			   &MesOut = 10 or
			   &MesOut = 12 %then %let DiaOut = 31;
			%else %if &MesOut = 04 or
					&MesOut = 06 or
					&MesOut = 09 or
					&MesOut = 11 %then %let DiaOut = 30;
			%else %if &MesOut = 02 and %sysfunc(mod(&AnoOut,4)) = 0 %then %let DiaOut = 29;
			%else %if &MesOut = 02 and %sysfunc(mod(&AnoOut,4)) ne 0 %then %let DiaOut = 28;
			%else %let DiaOut = .;
		%end;

		%let DtOut = %sysfunc(mdy(&MesOut, &DiaOut, &AnoOut));
		&DtOut.
%MEND;

%MACRO CALCULARPROV(DBASE=);

	%let DBAtu = %sysevalf(%sysevalf(%sysfunc(year("&DBASE."d))*100) + %sysfunc(month("&DBASE."d)));
	%let DtProvisao_Ant = %CalculaData(DataRef="&DBASE."d, Limite=INFERIOR, DifMes=1);
	%let DBAnt = %sysevalf(%sysevalf(%sysfunc(year(&DtProvisao_Ant.))*100) + %sysfunc(month(&DtProvisao_Ant.)));

	

	PROC SQL;
	   CREATE TABLE WORK.BDPORTAL_PAGAMENTOS AS
	   SELECT distinct t1.NumRegSistContrGest,
	     -sum(t1.Valor) format = commax22.2 as Valor ,
		 	case when t2.MotivoPadrao IN ('NÃO REPORTAR','') then 'DEMAIS'
			else t2.MotivoPadrao end as MotivoPadrao,
		  
			  mdy(month(t1.DtOcorPG),1,year(t1.DtOcorPG)) format=ddmmyy10. as Database1
    	  FROM BDPO.BDPORTAL_PAGAMENTOS t1
     	      LEFT JOIN ACM.MASSIFICADO t2 ON (t1.NumRegSistContrGest = t2.NumRegSistContrGest)
     	 WHERE t1.NumRegSistContrGest LIKE 'VAC%' AND t1.GestProc NOT = 'RELEVANTES' AND
				t1.DtOcorPG >= '1Jan2016'd AND t1.TpVerba = 1
		GROUP BY t1.NumRegSistContrGest, Database1;

	QUIT;

	

	PROC SQL;
	   CREATE TABLE WORK.BDPORTAL_PAGAMENTOS_&DBAtu. AS
	   SELECT *
    	  FROM work.BDPORTAL_PAGAMENTOS t1
     	   	 WHERE Database1=&DBASE.;

	QUIT;

	
	PROC SQL;
			create table WORK.REVERSAO_ENCERRADA_&DBAtu. (compress=yes index=(Causa/unique)) as
				select 	p.NumRegSistContrGest as causa,
						case when p.MotivoPadrao IN ('NÃO REPORTAR','') then 'DEMAIS'
							else p.MotivoPadrao end as MotivoPadrao,
						
						p.Valor format=commax22.2 as Provisao,

						mdy(input(substr("&DBAtu.",5,2),2.),1,input(substr("&DBAtu.",1,4),4.)) format=ddmmyy10. as Database

				from MODTEMP.PROVISAO_&DBAnt. as p
					left join MODTEMP.PROVISAO_&DBAtu. b on (p.NumRegSistContrGest = b.NumRegSistContrGest)

				where b.NumRegSistContrGest = "" or b.Valor = 0
				having p.Valor <>0;
	QUIT;

	
	PROC SQL;
			create table WORK.AtuaProv_total_&DBAtu. (compress=yes index=(Causa)) as
				select 	p.NumRegSistContrGest as Causa,
						case when c.MotivoPadrao IN ('NÃO REPORTAR','') then 'DEMAIS'
							else c.MotivoPadrao end as MotivoPadrao,
						p.Valor format=commax22.2 as Provisao,
						b.Valor format=commax22.2 as ProvisaoAnt,
						(p.Valor - b.Valor) format=commax22.2 as DifProvisao,
						mdy(input(substr("&DBAtu.",5,2),2.),1,input(substr("&DBAtu.",1,4),4.)) format=ddmmyy10. as Database
				from MODTEMP.PROVISAO_&DBAtu. as p
					inner join MODTEMP.PROVISAO_&DBAnt. b on (p.NumRegSistContrGest = b.NumRegSistContrGest)
					inner join ACM.MASSIFICADO c on (p.NumRegSistContrGest = c.NumRegSistContrGest)
					where p.Valor <>0
			;
	QUIT;

	
	PROC SQL;
			create table WORK.AtuaProv_total_pag_&DBAtu. (compress=yes index=(Causa)) as
				select 	p.Causa,
						p.MotivoPadrao,
						p.Provisao,
						p.provisaoAnt,
						p.DifProvisao,
						mdy(input(substr("&DBAtu.",5,2),2.),1,input(substr("&DBAtu.",1,4),4.)) format=ddmmyy10. as Database,
						b.Valor,
						b.Database1
				from work.AtuaProv_total_&DBAtu. as p
					left join WORK.BDPORTAL_PAGAMENTOS b ON (p.Causa = b.NumRegSistContrGest) AND (p.Database =
          b.Database1);
	QUIT;

		
		PROC SQL;
			create table WORK.REVERSAO_NAO_ENCERRADA_&DBAtu. (compress=yes index=(Causa)) as
				select 	p.Causa,
						case when p.MotivoPadrao IN ('NÃO REPORTAR','') then 'DEMAIS'
							else p.MotivoPadrao end as MotivoPadrao,
						-p.DifProvisao as Provisao,
						p.Database
				from WORK.AtuaProv_total_pag_&DBAtu. as p
				WHERE p.Database=&DBASE.  and
				Valor <>.
			;
	QUIT;

		
			PROC SQL;
			create table work.AtuaProv_&DBAtu. (compress=yes index=(Causa)) as
				select 	p.Causa,
						p.MotivoPadrao,
						p.Provisao,
						p.ProvisaoAnt,
						p.DifProvisao,
						p.Database
				from WORK.AtuaProv_total_pag_&DBAtu. as p
				WHERE Valor =.
			;
	QUIT;

	
	PROC SQL;
			create table WORK.NovasAcoes_&DBAtu. (compress=yes index=(Causa)) as
				select 	p.NumRegSistContrGest as Causa,
						case when t2.MotivoPadrao IN ('NÃO REPORTAR','') then 'DEMAIS'
							else t2.MotivoPadrao end as MotivoPadrao,
						p.Valor format=commax22.2 as Provisao

			   	from MODTEMP.PROVISAO_&DBAtu. as p
					left join MODTEMP.PROVISAO_&DBAnt. b on (p.NumRegSistContrGest = b.NumRegSistContrGest)
					left join ACM.MASSIFICADO t2 on (p.NumRegSistContrGest = t2.NumRegSistContrGest)

				where b.NumRegSistContrGest = "" and p.Valor <>0;
	QUIT;
