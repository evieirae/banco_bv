create table Work.base_gerencial as
select cp.*, aj.NmBook as Book, aj.NmCarteiraGerencial as NmCarteiraGerencial1, aj.NmMesaOperacao as NmMesaOperacao1
from work.base as cp
left join SHIS.AJUSTE_TB040_MESA_CARTEIRA_BOOK as aj
on aj.cdbook = cp.cdbook;

create table Work.base_ger_prod as
select bg.*, mp.DsModalidadecatalogoproduto as ModalidadeProduto
from work.base_gerencial as bg
left join SHIS.Modalidadeproduto as mp
on mp.cdmodalidadecatalogoproduto = bg.CdModalidadeProduto;

create table Work.base_ger_prod_conc as
select bg.*, nmgrupoconciliacao
from work.base_ger_prod as bg
left join PLANCTRL.TBGRUPOSCONCILIACAO as c
on c.cdgrupoconciliacao = bg.cdgrupoconciliacao;

       ResultadoComercial, ResultadoTesouraria, VrMesaCaixa, Book, NmCarteiraGerencial1, NmMesaOperacao1, ModalidadeProduto, nmgrupoconciliacao, cliente
