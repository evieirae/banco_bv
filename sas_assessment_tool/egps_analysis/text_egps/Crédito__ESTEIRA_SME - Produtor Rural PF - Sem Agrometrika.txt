	CREATE TABLE INFOS_SACADOS_PF_V4 AS SELECT

		,CASE WHEN A.COMERCIAL = "Ruim" THEN 0
		 ELSE 100 END AS Peso_Comercial

		,CASE WHEN A.RISCO_PRODUTIVO = "Muito Baixo" THEN 100
			  WHEN A.RISCO_PRODUTIVO = "Baixo" THEN 75

			  WHEN A.AREA_PROPRIA = "50,0% até 65,0%" THEN 50
			  WHEN A.AREA_PROPRIA = "65,0% até 80,0%" THEN 75
			ELSE 100 END AS Peso_Area_Propria

		,CASE WHEN A.TEMPO_CULT_PRINCIPAL = "Menos de 3 anos" THEN 0
			  WHEN  A.TEMPO_CULT_PRINCIPAL = "De 3 a 5 anos"   THEN 50
			  WHEN A.TEMPO_CULT_PRINCIPAL <= "De 5 a 10 anos" THEN 75

		,CASE WHEN A.EVOL_AREA = 'Entre -15% e 15%' THEN 100

			  WHEN A.EVOL_AREA = 'Crescimento superior a 15% não Justificado' THEN 0
			  WHEN A.EVOL_AREA = 'Redução superior a 15% justificado' THEN 80
			  WHEN A.EVOL_AREA = 'Redução superior a 15% não justificado' THEN 0
		 ELSE 0 END AS Peso_Evolucao_Area

		,CASE WHEN A.HIST_CREDITO = 'Pontual' THEN 100
			  WHEN A.HIST_CREDITO = 'Pontual / Atraso até 5 dias' THEN 100
			  WHEN A.HIST_CREDITO = 'Atraso entre 15 e 30 dias' THEN 75
		 ELSE 0 END AS Peso_Historico_Credito

		,CASE WHEN A.'TEMPO_RELAC_SACADO (anos)'n >= 5 THEN 100
			  WHEN A.'TEMPO_RELAC_SACADO (anos)'n >= 2 THEN 70
		 ELSE 0 END AS Peso_Tempo_Cliente

		,CASE WHEN A.LIQUIDEZ >= 1.43 THEN 100
			WHEN A.LIQUIDEZ >= 1.19 THEN 75
			WHEN A.LIQUIDEZ >= 1.00 THEN 50
			WHEN A.LIQUIDEZ >= 0.85 THEN 25

PROC TRANSPOSE

	DATA = BASE_COMPLETA_PF OUT = BASE_COMPLETA_PF_V2 (DROP = _NAME_);

	BY CPF11 NOTSORTED;

	ID PORTFOLIO_ORIGEM;

	VAR LIMITE_GLOBAL;

PROC SQL;

CREATE TABLE TABELA_INFOS_PF_FINAL AS SELECT

     A.*
		,(CASE WHEN F.LIMITE_GLOBAL >= 0 THEN F.LIMITE_GLOBAL
			ELSE 0 END) AS LIMITE_GLOBAL_CONVENIO
		,(CASE WHEN E.LIMITE_GLOBAL_BV >= 0 THEN E.LIMITE_GLOBAL_BV
			ELSE 0 END) AS LIMITE_GLOBAL_BV

	FROM INFOS_SACADOS_PF_V7 AS A

	LEFT JOIN BASE_COMPLETA_PF_V2 AS D ON A.CPF11 = D.CPF11

	LEFT JOIN BASE_COMPLETA_PF_LIMITE AS E ON A.CPF11 = E.CPF11

	LEFT JOIN BASE_COMPLETA_PF AS F ON A.CPF11 = F.CPF11 AND A.PORTFOLIO = F.PORTFOLIO_ORIGEM;

RUN;

PROC SQL;

	CREATE TABLE INFOS_SACADOS_PF_FINAL_1 AS SELECT

		*

		,(LIMITE_GLOBAL_BV - LIMITE_GLOBAL_CONVENIO) AS LIMITE_ALOCADO
		,CASE WHEN (LIMITE_ESTEIRA - CALCULATED LIMITE_ALOCADO) < 0 THEN 0
			ELSE 1000*INT(MIN(LIMITE_SACADO,(LIMITE_ESTEIRA - CALCULATED LIMITE_ALOCADO))/1000) END AS LIMITE_FINAL

		,CASE WHEN

				 FLAG_SERASA = 1 THEN "Restritivos"
			WHEN FLAG_RESTRICAO = 1 THEN "Restritivos"
			WHEN FLAG_COMERCIAL = 1 THEN "Conceito Comercial"
			WHEN FLAG_AREA_PROPRIA = 1 THEN "Proporção de Área Própria"
			WHEN FLAG_TEMPO_CULTURA = 1 THEN "Tempo na Cultura Principal"
			WHEN FLAG_HISTORICO_CREDITO = 1 THEN "Histórico de Crédito"
			WHEN FLAG_TEMPO_CLIENTE = 1 THEN "Tempo como Cliente"
			WHEN FLAG_SCORE_SERASA = 1 THEN "Score Serasa"
			WHEN LIMITE_ESTEIRA = 0 THEN "Score Serasa"
			WHEN CALCULATED LIMITE_FINAL = 0 THEN "Sem Limite Disponivel"
			ELSE "Pré-Aprovado" END AS RETORNO_ABERTO

		,CASE WHEN CALCULATED RETORNO_ABERTO = "Pré-Aprovado" THEN "Pré-Aprovado"
			ELSE "Recusado" END AS RETORNO

		,CALCULATED LIMITE_FINAL AS LIMITE_GL_AUX

	FROM TABELA_INFOS_PF_FINAL;

RUN;
