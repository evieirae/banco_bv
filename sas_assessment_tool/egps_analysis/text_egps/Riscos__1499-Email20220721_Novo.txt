proc sql;
create table work.Regra11n as
select distinct
11						as CdInconsistencia
,'EncerramentoLote'		as TpInconsistencia

,'Pastas com solicitações de encerramento, Motivo RRO falha operacional 1 e 2, com requisição de encerramento procedente/acordo/acordo pós,
exceto:
- Pastas de exibições de documento com contrato aberto ou fechado;
- Pastas com motivo: VENDA CASADA COM CONTRATO FECHADO;
- Pastas com motivo: PROBLEMAS RELACIONADOS A SEGURO PRESTAMISTA / VEICULOS COM CONTRATOLFECHADO;
- Pastas com motivo: COBRANÇA DE SALDO REMANESCENTE, com contrato fechado;
- Pastas com motivo: FALHA OPERACIONAL e submotivos: DIVERGÊNCIA DAS INFORMAÇÕES CONTRATADAS/ AVERBAÇÃO DA MARGEM ACIMA DO LIMITE LEGAL, com contrato fechado;
- Pastas com motivo: FALHA DE ATENDIMENTO e submotivos: SOLICITAÇÃO DE CANCELAMENTO NÃO ATENDIDA/ PEDIDO DE RENEGOCIAÇÃO/DEVOLUÇÃO DO BEM, com contrato fechado;
- Pastas com motivo: PROBLEMAS ADVINDOS DA AÇÃO DE BUSCA E APREENSÃO/LEILÃO e submotivo: AÇÃO AJUIZADA INDEVIDAMENTE / APREENSÃO INDEVIDA, com contrato fechado.'
as DescRegra lenght=1200

,'req.NmStatusRequisicao in (AGUARDANDO EXECUÇÃO, AGUARDANDO AVALIAÇÃO, EM REQUISIÇÃO, AGUARDANDO INFORMAÇÕES)
and req.TipoRequisicao = Solicitação de Encerramento [Cível Massificado]
and A.NmMotivoRRO in (FALHA OPERACIONAL 1, FALHA OPERACIONAL 2)
and TipoAtributoRequisicao =  Motivo de encerramento
and NmAtributoRequisicao in (ENCERRAMENTO ACORDO, ENCERRAMENTO ACORDO PÓS SENTENÇA, ENCERRAMENTO PROCEDÊNCIA PARCIAL/TOTAL
and not A.NmMotivo = EXIBIÇÃO DE DOCUMENTOS/INFORMAÇÕES
and not (A.NmMotivo in (VENDA CASADA, COBRANÇA DE SALDO REMANESCENTE) and iq.NmSituacaoContrato = f)

and not (iq.NmSituacaoContrato = f
and ((A.NmMotivo = FALHA DE ATENDIMENTO and A.NmSubMotivo in (SOLICITAÇÃO DE CANCELAMENTO NÃO ATENDIDA, PEDIDO DE RENEGOCIAÇÃO/DEVOLUÇÃO DO BEM))
or (A.NmMotivo = FALHA OPERACIONAL and A.NmSubMotivo in (DIVERGÊNCIA DAS INFORMAÇÕES CONTRATADAS, AVERBAÇÃO DA MARGEM ACIMA DO LIMITE LEGAL ))
or (A.NmMotivo = PROBLEMAS ADVINDOS DA AÇÃO DE BUSCA E APREENSÃO/LEILÃO and A.NmSubMotivo = AÇÃO AJUIZADA INDEVIDAMENTE / APREENSÃO INDEVIDA)))'  as Regra lenght=1200

from work.TBLCausasRequisicoes AS A

left join work.tblContratosIQ as iq
on iq.NuContratoText = A.NuContrato1DeN

WHERE 1=1
and A.NmMotivoRRO in ('FALHA OPERACIONAL 1', 'FALHA OPERACIONAL 2')

and A.CdPasta in
(SELECT DISTINCT
	Pasta
	FROM DBEP.TBLPROCESSOREQUISICAOATRIBUTO
	where 1=1
	and TipoAtributoRequisicao =  'Motivo de encerramento'
	and NmAtributoRequisicao in ('ENCERRAMENTO ACORDO', 'ENCERRAMENTO ACORDO PÓS SENTENÇA', 'ENCERRAMENTO PROCEDÊNCIA PARCIAL/TOTAL')
)

and A.CdPasta in
(SELECT DISTINCT
	CdPasta
	FROM work.TBL_Requisicao
	where 1=1
)

and not A.NmMotivo = 'EXIBIÇÃO DE DOCUMENTOS/INFORMAÇÕES'

and not (A.NmMotivo in ('VENDA CASADA', 'COBRANÇA DE SALDO REMANESCENTE')
	and iq.NmSituacaoContrato = 'f')

and not (iq.NmSituacaoContrato = 'f'
	and (
	(A.NmMotivo = 'FALHA DE ATENDIMENTO' and A.NmSubMotivo in ('SOLICITAÇÃO DE CANCELAMENTO NÃO ATENDIDA', 'PEDIDO DE RENEGOCIAÇÃO/DEVOLUÇÃO DO BEM'))
	or
	(A.NmMotivo = 'FALHA OPERACIONAL' and A.NmSubMotivo in ('DIVERGÊNCIA DAS INFORMAÇÕES CONTRATADAS', 'AVERBAÇÃO DA MARGEM ACIMA DO LIMITE LEGAL' ))
	or
	(A.NmMotivo = 'PROBLEMAS ADVINDOS DA AÇÃO DE BUSCA E APREENSÃO/LEILÃO' and A.NmSubMotivo = 'AÇÃO AJUIZADA INDEVIDAMENTE / APREENSÃO INDEVIDA')
)
)

;RUN;
