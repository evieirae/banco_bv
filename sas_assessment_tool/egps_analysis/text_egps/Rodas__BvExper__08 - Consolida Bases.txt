PROC SQL;
CREATE TABLE ESTRUTURA_COMERCIAL_GF AS
SELECT

A.NmCanalEstruturaComercial,
A.NmSubEstruturaComercialA,
A.NmAreaEstruturaComercial,
A.DSLOGINN2A,
/*A.DSLOGINN1*/
A.NUANOMES as nuanomes,
count(*) as qtd

FROM WORK.ESTRUTURA_COMERCIAL A

WHERE
1 = 1
AND A.DSLOGINN2A IS NOT NULL
AND A.DSLOGINN1 IS NOT NULL

group by 1,2,3,4,5
ORDER BY 5 DESC, 1,4

;RUN;

PROC SORT DATA=WORK.ESTRUTURA_COMERCIAL_GF OUT=ESTRUTURA_COMERCIAL_GF_2 NODUPKEY;BY DsLoginN2A;RUN;

/***************/
/*HIERARQUIA GR*/
/***************/

PROC SQL;
CREATE TABLE ESTRUTURA_COMERCIAL_GR AS
SELECT

A.NmCanalEstruturaComercial,
A.NmSubEstruturaComercialA,
A.NmAreaEstruturaComercial,
A.DSLOGINN2A,
A.DSLOGINN1,
A.NUANOMES as nuanomes,
count(*) as qtd

FROM WORK.ESTRUTURA_COMERCIAL A

WHERE
1 = 1
AND A.DSLOGINN2A IS NOT NULL
AND A.DSLOGINN1 IS NOT NULL

group by 1,2,3,4,5,6
ORDER BY 6 DESC, 1,5

;RUN;

PROC SORT DATA=WORK.ESTRUTURA_COMERCIAL_GR OUT=ESTRUTURA_COMERCIAL_GR_2 NODUPKEY;BY DsLoginN1;RUN;

PROC SQL;
CREATE TABLE BvExper_BaseUnica_Resumo_Total_3 AS
SELECT

A.*,

CASE
	WHEN A.NmCanalEstruturaComercial = 'Canal Privado e FGTS' THEN 'Canal Privado e FGTS' /*AJUSTE PRIVADO - DIRECIONADO PARA GF DO CANAL TATHIANA DIAS*/
	WHEN A.TpNivel = 'Fi' THEN B.NmCanalEstruturaComercial
	WHEN A.TpNivel = 'GR' THEN C.NmCanalEstruturaComercial
ELSE 'ERRO'
END AS NmCanalEstruturaComercial_Atual,

CASE
	WHEN A.NmCanalEstruturaComercial = 'Canal Privado e FGTS' THEN 'Canal Solar, Privado e FGTS' /*AJUSTE PRIVADO - DIRECIONADO PARA GF DO CANAL TATHIANA DIAS*/
	WHEN A.TpNivel = 'Fi' THEN B.NmSubEstruturaComercialA
	WHEN A.TpNivel = 'GR' THEN C.NmSubEstruturaComercialA
ELSE 'ERRO'
END AS NmSubEstruturaComercialA_Atual,

CASE
	WHEN A.NmCanalEstruturaComercial = 'Canal Privado e FGTS' THEN 'Canal Privado e FGTS' /*AJUSTE PRIVADO - DIRECIONADO PARA GF DO CANAL TATHIANA DIAS*/
	WHEN A.TpNivel = 'Fi' THEN B.NmAreaEstruturaComercial
	WHEN A.TpNivel = 'GR' THEN C.NmAreaEstruturaComercial
ELSE 'ERRO'
END AS NmAreaEstruturaComercial_Atual,

CASE
	WHEN A.NmCanalEstruturaComercial = 'Canal Privado e FGTS' THEN 'thatiana.souza' /*AJUSTE PRIVADO - DIRECIONADO PARA GF DO CANAL TATHIANA DIAS*/
	WHEN A.TpNivel = 'Fi' THEN B.DSLOGINN2A
	WHEN A.TpNivel = 'GR' THEN C.DSLOGINN2A
ELSE 'ERRO'
END AS DSLOGINN2A_Atual

FROM WORK.BvExper_BaseUnica_Resumo_Total_2 A

LEFT JOIN WORK.ESTRUTURA_COMERCIAL_GF_2 B ON B.DSLOGINN2A = A.DSLOGINN2A AND A.TpNivel = 'Fi'
LEFT JOIN WORK.ESTRUTURA_COMERCIAL_GR_2 C ON C.DSLOGINN1  = A.DSLOGINN1 AND A.TpNivel = 'GR'

;RUN;
