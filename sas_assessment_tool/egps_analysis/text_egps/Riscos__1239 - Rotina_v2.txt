PROC SQL;
CREATE TABLE WORK.ENCERRADAS_HISTORICO_FRAUDE AS

SELECT

t1.CdCausa						as	CD_CAUSA
,t1.DtCadastramento				as	DT_CADASTRAMENTO
,t1.NmMotivo					as	MOTIVO
,t1.NmSubMotivo					as	SUBMOTIVO
,t1.DtEncerramento				as	DT_ENCERRAMENTO
,t1.NmTipoEncerramento			as	TIPO_ENCERRAMENTO
,t1.NmMotivoEncerramento		as	MOTIVO_ENCERRAMENTO
,t1.NmOrgaoJudicial				as	DENOMINACAO
,t1.NmAcaoJudicial				as	ACAO_JUDICIAL
,t1.Empresa						as	EMPRESA
,t1.NmMunicipio					as	MUNICIPIO
,t1.NuContrato1DeN				as	NO_CONTRATO
,""								as	DT_INICIO_CTO
,t1.NomeParte					as	AUTOR
,t1.CPFCNPJ						as	CPFCNPJ
,t1.NmProdutoGrupo				as	PRODUTO
,t1.NmAssessoria				as	ASSESSORIA

FROM DBEP.VWTBLCAUSACONS t1

WHERE 1=1
AND cdSistema = 'p'
AND CdUnidadeExyon = 'VAC'
AND FlLegadoJudicialAdmnistrativo = 'J'
AND t1.NmMotivo 			= 'FRAUDE'
AND t1.NmSafraEncerramento	<>.

AND t1.NmSubMotivo IN(
					 'ANALFABETO'
					,'CONTRATO NÃO RECONHECIDO PELO AUTOR'
					,'GRAVAME NÃO RECONHECIDO'
					,'INDÍGENA'
					,'INSCRIÇÃO NO CADIN'
					,'SAQUE DE OP NÃO RECONHECIDO')
AND t1.NmMotivoEncerramento IN(
					'ACORDO COM PAGAMENTO'
					,'ACORDO COM RECEBIMENTO'
					,'ACORDO PÓS CONDENAÇÃO'
					,'ACORDO SEM PAGAMENTO'
					,'ARQUIVADO COM PAGTO CONDENAÇÃO'
					,'COND - PROC TOTAL/PARCIAL COM PAGTO'
					,'COND - PROC TOTAL/PARCIAL SEM PAGTO'
					,'PROCEDÊNCIA ACORDO PÓS-CONDENAÇÃO'
					,'PROCEDÊNCIA INTEGRAL DO PEDIDO DO AUTOR'
					,'PROCEDÊNCIA PARCIAL DO PEDIDO DO AUTOR'
					,'ACORDO - ACORDO COM PAGAMENTO'
					,'ACORDO - ACORDO SEM PAGAMENTO'
					,'SEM ÔNUS - COND PAGA PELA ASSESSORIA')
;RUN;

PROC SORT DATA= WORK.ENCERRADAS_HISTORICO_FRAUDE 			;BY CD_CAUSA NO_CONTRATO; RUN;

PROC SORT DATA= WORK.ENCERRADAS_HISTORICO_FRAUDE NODUPKEY	;BY CD_CAUSA NO_CONTRATO; RUN;

)
