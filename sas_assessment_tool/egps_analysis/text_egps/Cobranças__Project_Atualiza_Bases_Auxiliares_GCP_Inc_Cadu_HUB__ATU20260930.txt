PROC SQL;
CREATE TABLE WORK.MATRIZ_INAD_01 AS
SELECT DISTINCT
  A.DT_REF
, A.NUCPFCNPJ
, A.NUCONTRATOGESTAO
, CASE WHEN A.NUCPFCNPJ = B.NUCPFCNPJ AND B.QTD > 1 THEN 1 ELSE 0 END AS Flag_MultiplosContr
, CASE
WHEN A.GRUPO = 'CP' AND A.CARTEIRA = 'SOLAR' AND A.SISTEMA = 'GCON' THEN 'SOLAR GCON'
WHEN A.GRUPO = 'CP' AND A.CARTEIRA = 'SOLAR' AND A.SISTEMA = 'FLEX' THEN 'SOLAR FLEX'

WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') AND A.CARTEIRA = 'SOLAR' AND A.SISTEMA = 'GCON' THEN 'SOLAR GCON'
WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') AND A.CARTEIRA = 'SOLAR' AND A.SISTEMA = 'FLEX' THEN 'SOLAR FLEX'
ELSE A.GRUPO END AS PRODUTO
, PROPCASE(CASE
WHEN A.GRUPO = 'CARTAO' THEN A.Flag_TipoContr
WHEN A.GRUPO = 'SALDO' THEN 'Saldo Remanescente'
WHEN A.GRUPO = 'CP' AND A.CARTEIRA = 'SOLAR' THEN (COMPRESS(A.Flag_TipoContr) || ' placa Solar')
WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') AND A.CARTEIRA = 'SOLAR' THEN (COMPRESS(A.Flag_TipoContr) || ' placa Solar')
WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') THEN (COMPRESS(A.Flag_TipoContr) || ' placa Solar')
WHEN A.GRUPO = 'RODAS' AND COMPRESS(A.Flag_TipoContr) IS MISSING THEN 'financiamento veículo'
WHEN A.GRUPO = 'CP' AND A.CARTEIRA = 'NEON' THEN 'empréstimo crédito pessoal BV NEON'
WHEN A.GRUPO = 'CP' AND COMPRESS(A.Flag_TipoContr) IS MISSING THEN 'empréstimo crédito pessoal'
WHEN A.GRUPO IN ('CP','RODAS','CONSIG') THEN (COMPRESS(A.Flag_TipoContr) ||' '|| TRIM(A.Flag_TipoProd))
ELSE 'Verificar' END) AS Tipo_Contrato
, CASE WHEN A.GRUPO = 'CARTAO' THEN (COMPRESS(A.BANDEIRA_CARTAO)||' '||COMPRESS(A.VARIANTE_CARTAO))
   WHEN A.GRUPO = 'RODAS' THEN C.PLACA
   WHEN A.GRUPO = 'SALDO' THEN C.PLACA
   WHEN A.GRUPO = 'CP'   THEN 'final '|| SUBSTR(PUT(A.NUCONTRATOGESTAO,z14.),10,5)
   WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') THEN 'final '|| SUBSTR(PUT(A.NUCONTRATOGESTAO,z14.),10,5)
   WHEN A.GRUPO = 'CONSIG' THEN 'final '|| SUBSTR(PUT(A.NUCONTRATOGESTAO,z14.),10,5)
   ELSE '' END AS Detalhe_Contrato
, CASE
WHEN A.ATRASO <= 0 THEN 'MATRIZ_INAD'


WHEN A.GRUPO IN ('RODAS') AND A.ATRASO < 5 THEN 'MATRIZ_INAD'
WHEN A.GRUPO IN ('RODAS') AND A.ATRASO BETWEEN 5 AND 15 THEN 'NEGOCIACAO_FINANCEIRA'
WHEN A.GRUPO IN ('RODAS') AND A.ATRASO >= 16 THEN 'NEGOCIACAO_FINANCEIRA'
WHEN A.CARTEIRA IN ('CARTAO_CA') AND A.ATRASO <= 94 THEN 'MATRIZ_INAD'
WHEN A.CARTEIRA IN ('CARTAO_CA') AND A.ATRASO >= 94 THEN 'NEGOCIACAO_CARTAO_CL'
WHEN A.CARTEIRA IN ('CARTAO_CL') AND A.ATRASO >= 95 THEN 'NEGOCIACAO_CARTAO_CL'
WHEN A.GRUPO IN ('CARTAO') AND A.ATRASO <= 94 THEN 'MATRIZ_INAD' 
WHEN A.GRUPO IN ('CARTAO') AND A.ATRASO >= 95 THEN 'NEGOCIACAO_CARTAO_CL' 

WHEN A.GRUPO IN ('CP') AND A.ATRASO < 5 THEN 'MATRIZ_INAD'
WHEN A.GRUPO IN ('CP') AND A.ATRASO >= 5 THEN 'NEGOCIACAO_FINANCEIRA' 
WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') AND A.ATRASO < 5 THEN 'WHATS3003' 
WHEN A.GRUPO IN ('SOLAR','SOLAR GCON') AND A.ATRASO >= 5 THEN 'NEGOCIACAO_FINANCEIRA' 
WHEN A.GRUPO IN ('CONSIG') AND A.ATRASO <= 15 THEN 'SEM_NEG'
WHEN A.GRUPO IN ('CONSIG') AND A.ATRASO >= 16 THEN 'SEM_NEG'
WHEN A.GRUPO = 'SALDO' THEN 'SALDO'

WHEN A.ATRASO <= 0 THEN 'ADIMPLENTE'
WHEN A.GRUPO IN ('RODAS') AND A.COD_ASSESSORIA IN ('2062','2063','2064','2065','2066','2067','2068') THEN 'NEGOCIACAOFINANCEIRA'

WHEN A.GRUPO IN ('RODAS') AND A.ATRASO BETWEEN 5 AND 15 THEN 'NEGOCIACAOFINANCEIRA'
