CREATE TABLE work.ExyonProjurisRelevantes1 AS
SELECT * FROM CONNECTION TO postgres(

SELECT
case
when p.SERIAL_IMPORT like '%CTO%' then left(p.SERIAL_IMPORT,10)
when p.SERIAL_IMPORT like '%CON%' then left(p.SERIAL_IMPORT,10)
when p.SERIAL_IMPORT like '%VAC%' then left(p.SERIAL_IMPORT,10)
else p.SERIAL_IMPORT  END AS "Cód# Causa"
,entidade.NOME                            AS "Nome Parte"
,entidade.NUMERO_DOCUMENTO_PRINCIPAL      AS "CPF/CNPJ Parte"
,CAST(P.DATA_INICIO AS DATE)              AS "Dt# Cadastramento (Registro)"
,(SELECT NUMERO_ORIGINAL FROM public.D_PROCESSO_DESDOBRAMENTO
WHERE ID_PROCESSO = P.ID_PROCESSO
  AND ATUAL = 'T')                        AS "Nº Processo"
,(SELECT jurisdicao.jurisdicao FROM public.D_PROCESSO_DESDOBRAMENTO desdobramento
INNER JOIN public.LT_JURISDICAO jurisdicao ON jurisdicao.ID_JURISDICAO = desdobramento.ID_JURISDICAO
WHERE desdobramento.ID_PROCESSO = P.ID_PROCESSO
  AND desdobramento.ATUAL = 'T')          AS Denominação
,P.DATA_ENCERRAMENTO                      AS "Dt# Encerramento"
,(SELECT municipio.MUNICIPIO
FROM 	public.D_PROCESSO_DESDOBRAMENTO desdobramento
INNER JOIN 	public.LT_MUNICIPIO municipio ON municipio.ID_MUNICIPIO = desdobramento.ID_COMARCA
WHERE desdobramento.ATUAL = 'T'
  AND ID_PROCESSO = P.ID_PROCESSO)        AS Municipio
,(SELECT uf.uf FROM public.D_PROCESSO_DESDOBRAMENTO desdobramento
INNER JOIN public.LT_UF uf ON uf.ID_UF = desdobramento.ID_UF
WHERE desdobramento.ID_PROCESSO = P.ID_PROCESSO
  AND desdobramento.ATUAL = 'T')          AS UF
,(SELECT TIPO_ACAO FROM public.LT_TIPO_ACAO WHERE ID_TIPO_ACAO = P.ID_TIPO_ACAO) AS "Ação Judicial"
,(SELECT ASSUNTO_CUSTOM FROM public.LT_ASSUNTO_CUSTOM WHERE ID_ASSUNTO_CUSTOM = P.ID_ASSUNTO_CUSTOM) AS "Assunto Judicial"
,P.VALOR_REFERENCIA                       AS "Valor Pleito"
,OA.VALOR_PEDIDO                          AS "Vl# Discutido"
,'PROVÁVEL'					              AS Prognóstico
,OA.VALOR_PERDA_PROVAVEL	              AS "Vl# Prognóstico"
,OA.VALOR_PERDA_PROVAVEL_ATUALIZAD	      AS "Vl# Prognóstico Atualizado"
,OA.DATA_CADASTRO                         AS "Dt# Formalização"
,P.DATA_BAIXA_PROVISORIA                  AS "Dt# Baixa Contábil"
,CC.CARTEIRA_CUSTOM		                  AS "Carteira"
,(SELECT NOME FROM public.LT_CAUSA_RAIZ WHERE ID_CAUSA_RAIZ = P.ID_CAUSA_RAIZ) AS Motivo
,(SELECT NOME FROM public.LT_CAUSA_RAIZ_DOIS WHERE ID_CAUSA_RAIZ_DOIS = P.ID_CAUSA_RAIZ_DOIS) AS "Sub-Motivo"
,(SELECT UE.NOME
  FROM public.MM_PROCESSO_ADVOGADO AE
  INNER JOIN public.M_USUARIO UE ON UE.ID_USUARIO = AE.ID_USUARIO AND UE.LOGIN LIKE 'ESCRITORIO.%'
  WHERE ID_PROCESSO = P.ID_PROCESSO limit 1)      AS Recurso
,CAST(current_date AS DATE)               AS DtBase
,oa.STATUS_PROCESSO                       AS "Status Processo"
,P.pasta                                  AS Pasta
,'PROJURIS'                               AS Sistema
,p.ID_NATUREZA
,p.ID_CARTEIRA_CUSTOM
,p.NATUREZA_UNIDADE_CUSTOM
,DataCalculoProvisaoObjeto
,DataReclassificacaoProvisao
,StatusCausa
FROM (
SELECT
P.PASTA, P.SERIAL_IMPORT,
A.ASSUNTO_CUSTOM AS ASSUNTO,
N.NATUREZA,
NU.NATUREZA_UNIDADE_CUSTOM,
P.NUMERO_DESDOBRAMENTO_ATUAL,

CASE WHEN P.DATA_CADASTRO IS NULL AND P.DATA_PRE_CADASTRO IS NOT NULL AND P.DATA_ENCERRAMENTO IS NULL THEN 'PRE CADASTRO'
	 WHEN P.DATA_ENCERRAMENTO IS NOT NULL THEN 'ENCERRADO'
     WHEN P.STATUS_BAIXA = 'B' AND P.DATA_ENCERRAMENTO IS NULL THEN 'BAIXA_PROVISORIA'
  ELSE 'ATIVA'
END AS STATUS_PROCESSO,

CASE WHEN P.PROVISIONAR = 'T' THEN 'SIM' ELSE 'N?O' END AS DS_PROVISIONAR,
P.PROVISIONAR,

LMA.METODO_ATUALIZACAO,
PC.DATA_BASE_ATUALIZACAO,
PC.DATA_BASE_ATUALIZACAO_JUROS,
LO.OBJETO,O.DATA_CADASTRO,

O.JUROS_INICIAL,
PC.PERCENTUAL_PROVAVEL,

100 - (PC.PERCENTUAL_PROVAVEL + PC.PERCENTUAL_POSSIVEL) AS PERCENTUAL_REMOTO,
