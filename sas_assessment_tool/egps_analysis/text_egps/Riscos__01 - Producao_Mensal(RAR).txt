	create table work.base_ativos_03 as
		select distinct
			CdObjetoNegociacao,
			coalesce(GRUPO_,Grupo_Economico_SAS) as GRUPO1 length 100,
			case
				when FAIXA_FT2 = 'CORP' then 'CORPORATE'
				else FAIXA_FT2
			end as FAIXA_FT2 length 50,


			CdCliente as CdCliente,
			Cliente as Cliente length 100
		from
			BASE_ATIVOS_01
	;
quit;

proc sql;
	create table work.base_ativos_04 as
		select distinct
			CdCliente,
			max(GRUPO1) as GRUPO1 length 100
		from
			work.base_ativos_03
		group by
			CdCliente
	;
quit;

	else if find(upcase(SEGMENTO),'INSTITUICAO FINANCEIRA') > 0 then output;

	;

PROC FCMP OUTLIB = work.myfunc.exemple;

	function Data_Anterior(data_atual);

		date = put(data_atual,best6.);
		if substr(date,5,2) NE '01' then valor = data_atual - 1;
		else valor = data_atual - 89;
		return (valor);

	endsub;

	function Primeiro_Mes(data_atual);

		date = put(data_atual,best6.);
		ano = substr(date,1,4);
		valor = catx("",ano,'01');
		return (valor);

	endsub;

	function Data_Fin(data) $;

		mes = substr(put(data,best6.),5,2);
		novo_ano = substr(put(data,best6.),3,2);

		if mes = "01" then novo_mes = "JAN";
		else if mes = "02" then novo_mes = "FEV";
		else if mes = "03" then novo_mes = "MAR";
		else if mes = "04" then novo_mes = "ABR";
		else if mes = "05" then novo_mes = "MAI";
		else if mes = "06" then novo_mes = "JUN";
		else if mes = "07" then novo_mes = "JUL";
		else if mes = "08" then novo_mes = "AGO";
		else if mes = "09" then novo_mes = "SET";
		else if mes = "10" then novo_mes = "OUT";
		else if mes = "11" then novo_mes = "NOV";
		else if mes = "12" then novo_mes = "DEZ";

		valor = catx("",novo_mes,novo_ano);

		return (valor);

	endsub;

	function Mes(data) $;

		mes = substr(put(data,best6.),5,2);
		return (mes);

	endsub;

	function Ano(data) $;

		novo_ano = catx('','20',substr(put(data,best6.),3,2));
		return (novo_ano);

	endsub;
