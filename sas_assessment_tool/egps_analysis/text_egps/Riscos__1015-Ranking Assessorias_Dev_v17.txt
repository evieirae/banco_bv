create table work.calendario_enriquecido as
select
data
,day(data) 							as dia
,month(data) 						as mes
,year(data) 						as ano
,year(data)-1						as anoanterior
,qtr(data) 							as trimestre
,cats(month(data),year(data)) 		as MesAno
,year(data)*100 + month(data) 		as AnoMes
,year(data)*100 + qtr(data) 		as AnoTri

,cats(year(data),'T',qtr(data) ) 	as TrimestreReferencia
,cats(year(data)-1,'T',qtr(data) ) 	as AnoTriAnterior
from

proc sql;
create table work.calendario_enriquecido2 as
select a.*

,
(select min(b.data)
	from work.calendario_enriquecido as b
		where b.AnoTri = (select max(AnoTri)
	from work.calendario_enriquecido as b
	where b.AnoTri = a.AnoTri ) ) as  PrimeiraDataTri format=date9.
,
(select max(b.data)
	from work.calendario_enriquecido as b
		where b.AnoTri = (select max(AnoTri)
	from work.calendario_enriquecido as b
	where b.AnoTri = a.AnoTri ) ) as  UltimaDataTri format=date9.

,
(select max(AnoTri)
	from work.calendario_enriquecido as b
	where b.AnoTri < a.AnoTri ) as AnoTriAnterior
,
(select min(b.data)
	from work.calendario_enriquecido as b
		where b.AnoTri = (select max(AnoTri)
	from work.calendario_enriquecido as b
	where b.AnoTri < a.AnoTri ) ) as  DtApuracaoTriAnteriorIni format=date9.
,
(select max(b.data)
	from work.calendario_enriquecido as b
		where b.AnoTri = (select max(AnoTri)
	from work.calendario_enriquecido as b
	where b.AnoTri < a.AnoTri ) ) as  DtApuracaoTriAnteriorFim format=date9.
,cats(YEAR(calculated DtApuracaoTriAnteriorIni) , 'T' , qtr(calculated DtApuracaoTriAnteriorIni)) as TrimestreReferenciaAnterior

,intnx('year', Data, -1, 'same') as DtApuracao12meses format=date9.
,intnx('day',  Data, -1, 'same') as DtApuracao12mesesFim format=date9.

, case when a.data = calculated PrimeiraDataTri then 1 else 0 end as FlPrimeiraDataTri

from work.calendario_enriquecido as a
;run;

,avg(PesoCondenacao) as PesoCondenacao
,avg(PesoAcordo) as PesoAcordo
,avg(PesoPerdaZero) as PesoPerdaZero

create table work.Tbl1015BaseConsolidadaH as
select
b.*
,case when b.NmFonte = "Entradas"
	   and b.NmEncerramentoBoletim <> "ADMINISTRATIVO"
	  then 'S'
	  else 'N' end 														as Filtrar_Base_Entrada

,case when b.NmFonte = "03 Meses Atual"
       and b.NmEncerramentoBoletim <> "ADMINISTRATIVO"
	   and b.QtdEstoqueGeralUF = 1
       and b.QtdEncerradasGeral_1 = 1
	  then 'S'
	  else 'N' end 														as Filtrar_Base_Encerradas

,case when b.NmFonte = "03 Meses Atual"
	   and b.NmEncerramentoBoletim <> "ADMINISTRATIVO"
       and b.FlReabertura <> 1
	  then 'S'
	  else 'N' end 														as Filtrar_Base_Estoque

,case when b.NmFonte = "03 Meses Atual"
	   and b.NmEncerramentoBoletim = "ADMINISTRATIVO"
	  then 'S'
	  else 'N' end 														as Filtrar_Base_Administrativo

,case when b.NmFonte = "03 Meses Atual"
       and b.NmEncerramentoBoletim <> "ADMINISTRATIVO"
       and b.FlReabertura = 1
	  then 'S'
	  else 'N' end 														as Filtrar_Base_Reabertura

from work.tblEstoqueAllCons as b

inner join work.AtuacVigente as e
       on b.nmRecurso = e.NmRecurso
	  and b.CDMotivoRROGrupo = e.CDMotivoRROGrupo
	  and b.CdRegiaoGrupoAssessoria	= e.CdRegiaoGrupoAssessoria

where 1=1

and (
	(b.NmFonte = "Entradas" and b.NmEncerramentoBoletim <> "ADMINISTRATIVO")
	or (b.NmFonte = "03 Meses Atual" and b.NmEncerramentoBoletim <> "ADMINISTRATIVO" and b.QtdEstoqueGeralUF = 1 and b.QtdEncerradasGeral_1 = 1)
	or (b.NmFonte = "03 Meses Atual" and b.NmEncerramentoBoletim <> "ADMINISTRATIVO" and b.FlReabertura <> 1 )
	or (b.NmFonte = "03 Meses Atual" and b.NmEncerramentoBoletim = "ADMINISTRATIVO")
	or (b.NmFonte = "03 Meses Atual" and b.NmEncerramentoBoletim <> "ADMINISTRATIVO" and b.FlReabertura = 1 )
	)
