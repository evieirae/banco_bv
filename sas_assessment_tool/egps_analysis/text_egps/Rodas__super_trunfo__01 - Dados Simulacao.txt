PROC SQL;
CREATE TABLE BASE_SIMULACAO_01 AS
  SELECT
    *,
	CASE WHEN LOWER(Produto) LIKE '%leves%' THEN 'Leves'
	     WHEN LOWER(Produto) LIKE '%pesados%' THEN 'Pesados'
		 WHEN Produto = 'Motos' THEN 'Motos'
	END AS ProdutoAjuste,
	CASE WHEN COALESCE(FxIdade) = 'Acima 10 anos' THEN 'Acima de 10 anos' ELSE COALESCE(FxIdade) END AS FaixaIdade,
	CASE WHEN FlPropostaEfetivada = 1 THEN 1
	     WHEN StatusSimulacao = 'Proposta APNP' THEN 2
		 WHEN StatusSimulacao = 'Proposta negada' THEN 3
		 WHEN StatusSimulacao = 'Card sem proposta originada' THEN 4
		 WHEN StatusSimulacao = 'Simulação sem card' THEN 5
    ELSE 6 END AS Prioridade
  FROM CONVSIMU.tbsimulacoes
  WHERE
    calculated ProdutoAjuste IS NOT NULL
    AND SistemaInicial IN ('PPAR', 'ACOM', 'APAR')
	AND FlCard = 1
  ORDER BY
    NuAnoMes,
    NuCpfCnpj,
	CdParceiroComercial,
    calculated Prioridade,
	NuProposta DESC,
	DtSimulacaoFinal DESC;
RUN;

PROC SORT DATA=BASE_SIMULACAO_01 OUT=SCORE_ULTIMASIMU_01 NODUPKEY;
  BY NuAnoMes NuCpfCnpj CdParceiroComercial ProdutoAjuste;
RUN;

PROC SQL;
CREATE TABLE BASE_SIMULACAO AS
  SELECT
    NuAnoMes,
	CdParceiroComercial,
	ProdutoAjuste AS Produto,
	FaixaIdade,
	COUNT(DISTINCT NuCpfCnpj) AS QtdSimulacoes
  FROM SCORE_ULTIMASIMU_01
  GROUP BY 1,2,3,4;
RUN;
