PROC SQL;

CREATE TABLE TBL_CUBO_SERASA_FINAL AS

SELECT
		a.SafraRef,
		a.SafraConta,

	    CASE WHEN (INT(a.SafraConta/100)) < 2017 THEN '< 2017'
             ELSE PUT((INT(a.SafraConta/100)),6.)
             END AS AnoConta,

		CASE WHEN (INT(a.SafraConta/100)) < 2017 THEN '< 2017'
             ELSE COMPRESS(PUT(INT(a.SafraConta/100),4.)||"_"||PUT(CEIL(MOD(a.SafraConta,100)/3),2.)||'T')
             END AS TrimConta,

		a.MOBConta,
		CASE WHEN a.MobConta BETWEEN 0 AND 11  THEN '0 A 11'
		     WHEN a.MobConta BETWEEN 12 AND 23 THEN '12 A 23'
			 WHEN a.MobConta BETWEEN 23 AND 35 THEN '24 A 35'
			 WHEN a.MobConta BETWEEN 36 AND 47 THEN '36 A 47'
			 ELSE '> 48'
			 END AS FaixaMob,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'LIVRE'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'MAIS'
			      ELSE 'LEGADO' END
		     )
			 ELSE b.NmVarianteAgrupadoOrigem
             END AS NmVarianteOrigemCartao,

		CASE WHEN i.Score_Serasa BETWEEN 0 AND 100    THEN '1. 0 a 100'
		     WHEN i.Score_Serasa BETWEEN 101 AND 200  THEN '2. 101 a 200'
			 WHEN i.Score_Serasa BETWEEN 201 AND 300  THEN '3. 201 a 300'
			 WHEN i.Score_Serasa BETWEEN 301 AND 400  THEN '4. 301 a 400'
			 WHEN i.Score_Serasa BETWEEN 401 AND 500  THEN '5. 401 a 500'
			 WHEN i.Score_Serasa BETWEEN 501 AND 600  THEN '6. 501 a 600'
			 WHEN i.Score_Serasa BETWEEN 601 AND 700  THEN '7. 601 a 700'
			 WHEN i.Score_Serasa BETWEEN 701 AND 800  THEN '8. 701 a 800'
			 WHEN i.Score_Serasa BETWEEN 801 AND 900  THEN '9. 801 a 900'
			 WHEN i.Score_Serasa BETWEEN 901 AND 1000 THEN '10. 901 a 1000'
			 ELSE 'ND'
             END AS Faixa_ScoreSerasa,

		SUM(a.SaldoCart_Total)          AS SaldoCart_Total,
		SUM(a.SaldoCart_AVista)         AS SaldoCart_AVista,
		SUM(a.SaldoCart_Financiada)     AS SaldoCart_Financiada,
		SUM(a.SaldoCart_Accrual)        AS SaldoCart_Accrual,
		SUM(a.SaldoCart_Atraso)         AS SaldoCart_Atraso,
		SUM(a.SaldoCart_Parcelado)      AS SaldoCart_Parcelado,
		COALESCE(SUM(a.SaldoCart_CreditoPessoal),0) AS SaldoCart_CreditoPessoal,
		COALESCE(SUM(a.SaldoCart_PgtoContasParc),0) AS SaldoCart_PgtoContasParc,
		SUM(a.SaldoCart_ParcFatura)     AS SaldoCart_ParcFatura,
		SUM(a.SaldoCart_ParcEspecial)   AS SaldoCart_ParcEspecial,
		SUM(a.SaldoCart_ParcAcordo)     AS SaldoCart_ParcAcordo,
		SUM(a.SaldoCart_ParcEmissor)    AS SaldoCart_ParcEmissor,
		SUM(a.SaldoCart_Rotativo)       AS SaldoCart_Rotativo,
		SUM(a.SaldoCart_RotUnder60)     AS SaldoCart_RotUnder60,
		SUM(a.SaldoCart_RotOver60) 	    AS SaldoCart_RotOver60,

		SUM(a.SaldoGastoAutorizado)   AS SaldoGastoAutorizado,
		SUM(a.SaldoFaturamento) 	  AS SaldoFaturamento,
		SUM(a.QtdAutorizacoes) 		  AS QtdAutorizacoes,
		SUM(a.QtdContasAptas) 		  AS QtdContasAptas,
		SUM(a.QtdContasAtivas_30dias) AS QtdContasAtivas_30dias,
		SUM(a.QtdContasCriadas) 	  AS QtdContasCriadas,
		SUM(a.QtdFaturasEmitidas) 	  AS QtdFaturasEmitidas,

		COALESCE(SUM(f.QtdCancelamentos),0) 	 AS QtdCancelamentos,
		COALESCE(SUM(f.QtdContasReabilitadas),0) AS QtdContasReabilitadas,

		SUM(a.VrTarifaAnuidade) 		  AS VrTarifaAnuidade,
		SUM(a.VrTarifaOverlimit) 		  AS VrTarifaOverlimit,
		SUM(a.VrTarifaOutra) 			  AS VrTarifaOutra,
		SUM(a.VrComissaoSeguros) 	      AS VrComissaoSeguros,

		SUM(a.VrTarifaIntercambio) 	  AS VrTarifaIntercambio,
		SUM(a.VrFraudes) 			  AS VrFraudes,

		SUM(a.VrJuroParceladoFatura)    AS VrJuroParceladoFatura,
		COALESCE(SUM(a.VrJuroCreditoPessoal),0)     AS VrJuroCreditoPessoal,
		COALESCE(SUM(a.VrJuroPgtoContasParc),0)     AS VrJuroPgtoContasParc,
		SUM(a.VrJuroParceladoEspecial)  AS VrJuroParceladoEspecial,
		SUM(a.VrJuroParceladoAcordo)    AS VrJuroParceladoAcordo,
		SUM(a.VrJuroParceladoEmissor)   AS VrJuroParceladoEmissor,
		SUM(a.VrJuroRotativo)           AS VrJuroRotativo,
		SUM(a.VrReceitaVariacaoCambial) AS VrReceitaVariacaoCambial,
		SUM(a.VrFloatFunding)           AS VrFloatFunding,
		SUM(a.VrDemaisResultados) 	    AS VrDemaisResultados,

		SUM(a.VrMoraMulta) 			  AS VrMoraMulta,
		SUM(a.VrCustoCobranca) 		  AS VrCustoCobranca,

		SUM(a.VrCustosContasCriadas)    AS VrCustosContasCriadas,
		SUM(a.VrCustosBandeira) 	    AS VrCustosBandeira,
		SUM(a.VrCustosFidelizacao) 	    AS VrCustosFidelizacao,
		SUM(a.VrCustosManutencao) 	    AS VrCustosManutencao,

		SUM(a.VrPDDBruto)           - COALESCE(SUM(g.VrPDDBruto_Cessao),0)			AS VrPDDBruto,
		SUM(a.VrRecuperacaoCredito) - COALESCE(SUM(g.VrRecuperacao_Cessao),0)       AS VrRecuperacaoCredito,
		SUM(a.VrDescontoPrincipal)  - COALESCE(SUM(g.VrDescontoPrincipal_Cessao),0) AS VrDescontoPrincipal,
		COALESCE(SUM(g.VrCustoCredito_Cessao),0)+COALESCE(SUM(h.Vr_Prudencial),0)   AS VrCessaoCarteira,

		SUM(a.VrDespesaAdministrativa)  AS VrDespesaAdministrativa,
		SUM(a.VrDespesaPessoal) 		AS VrDespesaPessoal,
		SUM(a.VrDespesaTributaria) 	    AS VrDespesaTributaria,
		SUM(a.VrOutraDespesaReceita)    AS VrOutraDespesaReceita,
		SUM(a.VrRemuneracaoVariavel)    AS VrRemuneracaoVariavel,
		SUM(a.VrContingencias) 		    AS VrContingencias,

		SUM(a.VrMargemLiquida)          AS VrMargemLiquida,
		SUM(a.VrMargemContribuicao)     AS VrMargemContribuicao,
		SUM(a.VrLAIR)                   AS VrLAIR,
		SUM(a.VrLucroLiquido)           AS VrLucroLiquido,
		SUM(a.VrPatrimonioLiquidoROE)   AS VrPatrimonioLiquidoROE,

		COALESCE(SUM(c.VrEntrada_ParcFatura),0)   AS VrEntrada_ParcFatura,
		COALESCE(SUM(c.VrEntrada_ParcCompul),0)   AS VrEntrada_ParcCompul,
		COALESCE(SUM(c.VrEntrada_ParcAcordo),0)   AS VrEntrada_ParcAcordo,
		COALESCE(SUM(c.VrEntrada_CP),0)           AS VrEntrada_CP,
		COALESCE(SUM(c.VrEntrada_PgtoContas),0)   AS VrEntrada_PgtoContas,
		COALESCE(SUM(c.SaldoFaturamento_M1),0)    AS SaldoFaturamento_M1,

		COALESCE(SUM(d.Cart_EmDia),0)  AS Cart_EmDia,
		COALESCE(SUM(d.Cart_1a30),0)     AS Cart_1a30,
		COALESCE(SUM(d.Cart_31a60),0)    AS Cart_31a60,
		COALESCE(SUM(d.Cart_61a90),0)    AS Cart_61a90,
		COALESCE(SUM(d.Cart_91a120),0)   AS Cart_91a120,
		COALESCE(SUM(d.Cart_121a150),0)  AS Cart_121a150,
		COALESCE(SUM(d.Cart_151a180),0)  AS Cart_151a180,
		COALESCE(SUM(d.Cart_Acima180),0) AS Cart_Acima180,

		SUM(COALESCE(a.VrMargemLiquida,0)) -
		(SUM(COALESCE(a.VrPDDBruto,0))+SUM(COALESCE(a.VrRecuperacaoCredito,0))+SUM(COALESCE(a.VrDescontoPrincipal,0))) AS VrMargemServicos,

		SUM(COALESCE(a.VrPDDBruto,0))  +
	    SUM(COALESCE(a.VrRecuperacaoCredito,0))+
	    SUM(COALESCE(a.VrDescontoPrincipal,0))    AS VrCustoCredito,

		(CALCULATED VrCustoCredito) - (COALESCE(SUM(g.VrCustoCredito_Cessao),0)+COALESCE(SUM(h.Vr_Prudencial),0)) AS VrCustoCredito_Norm

FROM TBL_EMPILHAMENTO_RAC_FINAL as a

LEFT JOIN CARTAOBV.TBL_CUBO_BASECONTAS_CARTOESBV as b
ON a.SafraRef = b.SafraRef AND
   a.CdContaCartao = b.CdDimContaCartao

LEFT JOIN CARTAOBV.TBL_CUBO_PROPENSAO_CARTOESBV as c
ON a.SafraRef = c.SafraRef AND

LEFT JOIN CARTAOBV.TBL_CUBO_CARTATRASO_CARTOESBV as d

LEFT JOIN CARTAOBV.TBL_RAC_CESSAO_CARTOES as g

LEFT JOIN WORK.TBL_CUBO_SCORE_SERASA as i
ON a.CdContaCartao = i.CdContaCartao

GROUP BY 1,2,3,4,5,6,7,8;
