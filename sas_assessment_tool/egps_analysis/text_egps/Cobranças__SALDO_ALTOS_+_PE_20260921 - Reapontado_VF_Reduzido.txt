       DATE(A.DT_ASSES) <= &DATA_HOJE. AND

A.CD_CONTRATO = B.NUCONTRATOGESTAO

SELECT
A.NUCONTRATOGESTAO,
A.DSACAOCOBRANCA,
DATEPART(A.DTGRAVACAOSCORE) AS DT_INI_VIGENCIA FORMAT = DATE9.,
CASE
	WHEN DATEPART(MIN(B.DTGRAVACAOSCORE)) IS NOT NULL THEN DATEPART(MIN(B.DTGRAVACAOSCORE)) -1
		ELSE .
		END AS DT_FIM_VIGENCIA FORMAT = DATE9.,

	WHEN DATEPART(MIN(B.DTGRAVACAOSCORE)) IS NULL THEN 'VIGENTE'

		END AS STATUS_VIGENCIA

ON A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
AND B.DTGRAVACAOSCORE > A.DTGRAVACAOSCORE
WHERE A.DSACAOCOBRANCA IN ('C1', 'TD1')
GROUP BY
	A.NUCONTRATOGESTAO,
	A.DSACAOCOBRANCA,
	A.DTGRAVACAOSCORE
ORDER BY
	A.DTGRAVACAOSCORE DESC
;QUIT;
