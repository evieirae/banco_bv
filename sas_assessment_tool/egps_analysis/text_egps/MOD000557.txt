PROC SQL;
CREATE TABLE score_pop1_&anomes. AS
SELECT
	*,
	/* Quantas vezes atrasou mais do que 15 dias nas últimas 6 faturas*/
	sum(
	case when QtDiaAtrasoFaturaM1 >15 then 1 else 0 end,
	case when QtDiaAtrasoFaturaM2 >15 then 1 else 0 end,
	case when QtDiaAtrasoFaturaM3 >15 then 1 else 0 end,
	case when QtDiaAtrasoFaturaM4 >15 then 1 else 0 end,
	case when QtDiaAtrasoFaturaM5 >15 then 1 else 0 end,
	case when QtDiaAtrasoFaturaM6 >15 then 1 else 0 end) as qtd_atrs_maior15d_6m,

	/*Quantidade de vezes que pagou antes do corte seguinte nas últimas 6 faturas*/
	sum(
	case when DtAtingimentoPgmnoMnmoM1 ne . and DtAtingimentoPgmnoMnmoM1<DtCorteFaturaM1 then 1 else 0 end,
	case when DtAtingimentoPgmnoMnmoM2 ne . and DtAtingimentoPgmnoMnmoM2<DtCorteFaturaM2 then 1 else 0 end,
	case when DtAtingimentoPgmnoMnmoM3 ne . and DtAtingimentoPgmnoMnmoM3<DtCorteFaturaM3 then 1 else 0 end,
	case when DtAtingimentoPgmnoMnmoM4 ne . and DtAtingimentoPgmnoMnmoM4<DtCorteFaturaM4 then 1 else 0 end,
	case when DtAtingimentoPgmnoMnmoM5 ne . and DtAtingimentoPgmnoMnmoM5<DtCorteFaturaM5 then 1 else 0 end,
	case when DtAtingimentoPgmnoMnmoM6 ne . and DtAtingimentoPgmnoMnmoM6<DtCorteFaturaM6 then 1 else 0 end) as qtd_pg_ant_corte_6m,

	/*Quantidade de vezes que pagou pelo menos o mínimo nas últimas 6 faturas*/
