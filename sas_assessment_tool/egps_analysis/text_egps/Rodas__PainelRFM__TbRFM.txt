proc sql;
    create table TbProducaoEmprestimos as
    select
        t1.DtContrato format=DDMMYYS10. as DtContrato,
        t1.DtAtualizacao format=DDMMYYS10. as DtAtualizacao,
        t1.NmGrupoProdutoMeta,
        t1.NuProposta format=BESTX32. as NuProposta,
        t1.NuContrato format=BESTX32. as NuContrato,
		t2.NuCnpjLoja,
		t2.NmParceiroComercial AS NmParceiroComercial,
		t2.CdParceiroComercial,
		t2.NmColaboradorC,
		t2.NmFilialC,
		t2.NmSubEstruturaComercialC,
		t2.NmAreaEstruturaComercialC,
		t2.NmCanalEstruturaComercialC,
		t2.DsLoginN1C,
		t2.DsLoginN2C,
		t2.DsLoginN3C,
		t2.LstSubstituicaoC,
        t1.DsTipoCanalProducao,
        t1.VrProducao format=COMMAX18.2 as VrProducao,
        case
            when t1.NuContrato ne 0 then 1
            else 0
        end as QtContrato,
		CASE
			WHEN t1.NmParceiroComercial  LIKE '%GR CONSUL%' OR t1.NmParceiroComercial LIKE '%COCKPIT%' THEN 'GR CONSULTOR'
			WHEN t1.NmParceiroComercial  LIKE '%CP GARANTIA%'
				OR t1.NmParceiroComercial  LIKE '%BV PARTI%'
				OR t1.NmParceiroComercial  CONTAINS '%BV%' AND t1.NmParceiroComercial CONTAINS '%PARTICULAR%'
				OR t1.NmParceiroComercial  LIKE '%BV BALC%'
				OR t1.NmParceiroComercial  CONTAINS '%BV CP%'
				OR t1.NmParceiroComercial  LIKE '%BV MESA%'
				OR t1.NmParceiroComercial  CONTAINS '%BV %'
				OR t1.CdParceiroComercial  IN (330023, 438407, 438415, 438444, 466647, 307861, 347441, 19474, 416382,
												326679, 342798 ,360518 ,386992 ,414643 ,418607 ,419715 ,419716 ,424584 ,
												435923 ,438183 ,438192 ,438222 ,438638 ,441516 ,452470 ,465399 ,466807 ,
												467954, 468271)
				THEN 'BALCÃO'
			ELSE 'DEMAIS'
		END as Separador
    from
        bases.tbproducao t1
	  LEFT JOIN BASES.TBPARCEIROCOMERCIAL t2
	  ON t1.CdParceiroComercial = t2.CdParceiroComercial
    where
        t1.DtContrato >= '01JAN2021'd
        and t1.NmGrupoProdutoMeta in ('Motos', 'CVG Leves', 'CVG Pesados', 'Leves', 'Pesados')
    order by
        t1.DtContrato desc;
quit;

proc sql;
    create table TbRFM as
    select
        DtContrato format=DDMMYYS10. as DtContrato,
        DtAtualizacao format=DDMMYYS10. as DtAtualizacao,
        NmGrupoProdutoMeta,
        NuProposta format=BESTX32. as NuProposta,
        NuContrato format=BESTX32. as NuContrato,
		NuCnpjLoja,
		NmParceiroComercial AS NmParceiroComercial,
		CdParceiroComercial,
		NmColaboradorC,
		NmFilialC,
		NmSubEstruturaComercialC,
		NmAreaEstruturaComercialC,
		NmCanalEstruturaComercialC,
		DsLoginN1C,
		DsLoginN2C,
		DsLoginN3C,
		LstSubstituicaoC,
        DsTipoCanalProducao,
        VrProducao format=COMMAX18.2 as VrProducao,
        QtContrato
    from
        TbProducaoEmprestimos
    where
        Separador = 'DEMAIS'
    order by
        DtContrato desc;
quit;
