		AND YEAR(A.MAX_DT_VENCIMENTO_PARC_PAGA)*100+MONTH(A.MAX_DT_VENCIMENTO_PARC_PAGA) = YEAR(intnx('MONTH',TODAY(),-1))*100+MONTH(intnx('MONTH',TODAY(),-1))
