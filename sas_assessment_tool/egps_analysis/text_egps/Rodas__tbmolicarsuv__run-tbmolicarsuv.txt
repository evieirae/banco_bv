%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
create table TbMolicar_SUV as
select * from connection to dbcon
(
SELECT DISTINCT
B.NUPROPOSTA
,A.CDMOLICAR
,A.CDMARCA
,B.CdModelo
,B.AaFabricacao
,B.AaModelo
,A.DSCATEGORIA
,A.DSMOLICARCONFIGURACAO

FROM DBA.TbMolicar AS A

LEFT JOIN DBA.TbPropGarantiaMaqEquip AS B
ON A.CDMOLICAR = B.CDMOLICAR

INNER JOIN DBA.TbProposta AS C
ON B.NUPROPOSTA = C.NUPROPOSTA

WHERE dsmolicarconfiguracao LIKE '%SUV%'
)
;RUN;

PROC SORT DATA=TbMolicar_SUV OUT=TbMolicar_SUV NODUPKEY; BY NUPROPOSTA;RUN;

/*%include '/infogrc/utils/verifica/verifica-originacao.sas';*/

DATA Originacao;
SET tmp.TbOriginacao&a3. - tmp.TbOriginacao&a0.;
RUN;

PROC SQL;
CREATE TABLE AUXL.TbMolicar_SUV AS
SELECT

A.NUPROPOSTA,
B.NuContratoGestao,
A.CDMOLICAR,
A.CDMARCA,
A.CdModelo,
A.AaFabricacao,
A.AaModelo,
A.DSCATEGORIA,
A.DSMOLICARCONFIGURACAO,
B.NmGrupoProdutoMeta

FROM TbMolicar_SUV A

LEFT JOIN Originacao B ON B.NUPROPOSTA = A.NUPROPOSTA

where B.NmGrupoProdutoMeta NOT IN ('CVG Pesados','Pesados')

;RUN;
