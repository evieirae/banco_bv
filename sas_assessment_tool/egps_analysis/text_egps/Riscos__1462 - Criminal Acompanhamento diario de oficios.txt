PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.Tbl1462Respostas  AS
SELECT
SERIAL_IMPORT as SERIAL_IMPORT
,Cd_Causa as Cd_Causa
,Pasta as Pasta
,NomeParte as NomeParte
,Status as Status
,id_processo_evento as id_processo_evento
,NmEvento as NmEvento
,detalhe as detalhe
,Descricao as Descricao
,Responsavel as Responsavel
,DtResposta as DtResposta
,DtAndamento as DtAndamento
,DataMaximaDaResposta as DataMaximaDaResposta
,Recurso as Recurso
FROM CONNECTION TO postgres(
SELECT
    CASE
        WHEN pr.serial_import LIKE '%CTO%' THEN LEFT(pr.serial_import, 10)
        WHEN pr.serial_import LIKE '%INQ%' THEN LEFT(pr.serial_import, 10)
        ELSE pr.serial_import
    END AS Cd_Causa
    , pr.serial_import
    , pr.pasta
    , ent.nome AS NomeParte
    , CASE
        WHEN pr.data_encerramento IS NULL THEN 'EM CURSO'
        ELSE 'ENCERRADO'
    END AS Status
    , pe.id_processo_evento
    , ev.evento AS NmEvento
    , CAST(pe.detalhes AS VARCHAR(2000)) AS detalhe
    , CASE
        WHEN POSITION('[' IN pe.detalhes) > 0
        THEN REPLACE(SUBSTRING(pe.detalhes FROM 1 FOR POSITION('[' IN pe.detalhes) - 1), ' [', '')
        ELSE pe.detalhes
    END AS Descricao
    , CASE
        WHEN POSITION(':' IN pe.detalhes) > 0 AND POSITION('[' IN pe.detalhes) > POSITION(':' IN pe.detalhes)
        THEN REPLACE(
                REPLACE(
                    SUBSTRING(pe.detalhes
                        FROM POSITION(':' IN pe.detalhes) + 1
                        FOR POSITION('[' IN pe.detalhes) - POSITION(':' IN pe.detalhes) - 1), ': ', ''), ']', '')
        ELSE pe.detalhes
    END AS Responsavel
    , LEFT(CAST(pe.detalhes AS VARCHAR(2000)), 10) AS DtResposta
    , CAST(pe.data AS DATE) AS DtAndamento
    , (
        SELECT MAX(pe.data)::DATE
        FROM public.d_processo_evento pe
        INNER JOIN public.m_processo pr
        ON pe.id_processo = pr.id_processo
        WHERE pr.natureza_unidade_custom IN (7)
        AND pr.id_natureza = 32
        AND pe.data <= CURRENT_DATE
    ) AS DataMaximaDaResposta
    , (
        SELECT usuario.nome
        FROM public.mm_processo_advogado advogado
        INNER JOIN public.m_usuario usuario
        ON usuario.id_usuario = advogado.id_usuario
        
        WHERE advogado.id_processo = pr.id_processo
        AND advogado.id_usuario = pr.id_advogado_principal
        LIMIT 1
    ) AS Recurso
FROM public.m_processo pr
INNER JOIN public.d_processo_evento 			as pe 		ON pe.id_processo = pr.id_processo
INNER JOIN public.d_processo_desdobramento 		as pd 		ON pd.id_processo = pe.id_processo
INNER JOIN public.lt_natureza 					as na 		ON na.id_natureza = pr.id_natureza
INNER JOIN public.lt_tipo_acao 					as ta 		ON ta.id_tipo_acao = pr.id_tipo_acao
INNER JOIN public.d_processo_parte 				as pp_autor ON pr.id_processo = pp_autor.id_processo
    AND pp_autor.cliente = 'F'
    AND pp_autor.principal = 'T'
INNER JOIN public.m_entidade 					as ent 		ON ent.id_entidade = pp_autor.id_entidade
INNER JOIN public.d_processo_parte 				as pp 		ON pp.id_processo = pr.id_processo
    AND pp.cliente = 'T'
    AND pp.principal = 'T'
INNER JOIN public.m_entidade 					as entemp 	ON entemp.id_entidade = pp.id_entidade
INNER JOIN public.lt_uf 						as lu 		ON lu.id_uf = pd.id_uf
INNER JOIN public.lt_municipio 					as lm 		ON lm.id_municipio = pd.id_comarca
INNER JOIN public.lt_jurisdicao 				as lj 		ON lj.id_jurisdicao = pd.id_jurisdicao
INNER JOIN public.lt_causa_raiz 				as cr 		ON cr.id_causa_raiz = pr.id_causa_raiz
INNER JOIN public.m_usuario 					as us 		ON us.id_usuario = pr.id_advogado_principal
LEFT JOIN public.lt_tipo_encerramento 			as te 		ON te.id_tipo_encerramento = pr.id_tipo_encerramento
LEFT JOIN public.lt_encerramento 				as en 		ON en.id_encerramento = pr.id_encerramento
LEFT JOIN public.lt_motivo_baixa_provisoria 	as bp 		ON bp.id_motivo_baixa_provisoria = pr.id_motivo_baixa_provisoria
LEFT JOIN public.lt_carteira_custom 			as cc 		ON cc.id_carteira_custom = pr.id_carteira_custom
LEFT JOIN public.lt_fase 						as lf 		ON lf.id_fase = pd.id_fase
INNER JOIN public.lt_evento 					as ev 		ON ev.id_evento = pe.id_evento
WHERE
    pr.natureza_unidade_custom IN (7)
    AND pr.id_natureza IN (32)
    AND ev.evento IN (
        'PETIÇÃO PROTOCOLADA',
        'OITIVA CANCELADA',
        'TERMO DE AUDIÊNCIA',
        'OITIVA DE RATIFICAÇÃO',
        'OFICIO RESPONDIDO',
        'TERMO DE COMPARECIMENTO',
        'TERMO DE RATIFICAÇÃO',
        'INTIMAÇÃO RESPONDIDA',
        'OITIVA DE RATIFICAÇÃO',
        'TERMO DE AUDIÊNCIA',
        'TERMO DE COMPARECIMENTO',
        'OITIVA REMARCADA',
        'OFICIO - REITERAÇÃO RESPONDIDO',
		'ATA DE AUDIÊNCIA'
		,'RESPOSTA'

    )
    AND pr.data_encerramento IS null
;
);DISCONNECT FROM postgres;
RUN;

PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.id_processo_evento  AS
SELECT

 Cd_Causa as Cd_Causa
,SERIAL_IMPORT as SERIAL_IMPORT
,Pasta as Pasta
,evento as evento
,id_processo_evento as id_processo_evento
,detalhes as detalhes
,data as data
,detalhe as detalhe
,Descricao as Descricao
,Responsavel as Responsavel
,DtResposta as DtResposta
,DtAndamento as DtAndamento

FROM CONNECTION TO postgres(

   select
       CASE
        WHEN pr.serial_import LIKE '%CTO%' THEN LEFT(pr.serial_import, 10)
        WHEN pr.serial_import LIKE '%INQ%' THEN LEFT(pr.serial_import, 10)
        ELSE pr.serial_import
    END AS Cd_Causa
    , pr.serial_import
    , pr.pasta
    ,le.evento
	, a.id_processo_evento
    , a.detalhes
    , a.data
	, CAST(a.detalhes AS VARCHAR(2000)) AS detalhe
    , CASE
        WHEN POSITION('[' IN a.detalhes) > 0
        THEN REPLACE(SUBSTRING(a.detalhes FROM 1 FOR POSITION('[' IN a.detalhes) - 1), ' [', '')
        ELSE a.detalhes
    END AS Descricao
    , CASE
        WHEN POSITION(':' IN a.detalhes) > 0 AND POSITION('[' IN a.detalhes) > POSITION(':' IN a.detalhes)
        THEN REPLACE(
                REPLACE(
                    SUBSTRING(a.detalhes
                        FROM POSITION(':' IN a.detalhes) + 1
                        FOR POSITION('[' IN a.detalhes) - POSITION(':' IN a.detalhes) - 1), ': ', ''), ']', '')
        ELSE a.detalhes
    END AS Responsavel
    , LEFT(CAST(a.detalhes AS VARCHAR(2000)), 10) AS DtResposta
    , CAST(a.data AS DATE) AS DtAndamento

   from public.d_processo_evento as a
   left join public.lt_evento as le
   on a.id_evento = le.id_evento
   inner join public.m_processo as pr
     on a.id_processo = pr.id_processo
    and pr.natureza_unidade_custom = 7
    and pr.id_natureza = 32
    and pr.data_encerramento IS NULL
;
);DISCONNECT FROM postgres;
RUN;
