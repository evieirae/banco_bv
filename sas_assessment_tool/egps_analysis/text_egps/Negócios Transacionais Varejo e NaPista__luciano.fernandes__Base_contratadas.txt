data base_1;
set base;
keep NuProposta StProposta  TpDados DtPrimeiroVencimento '(%) SpreadPactuado'n VrComissaoTotal
VrAcordoComercial VrRetornoPlus '(%) Comissao'n '(%) Acordo'n '(%) TotalComissao'n ;
run;

proc sql;
create table base_2 as
select NuProposta
		,max(StProposta) as StProposta
		,max(TpDados) as TpDados
		,max(DtPrimeiroVencimento) as DtPrimeiroVencimento format date9.
		,max('(%) SpreadPactuado'n) as '(%) SpreadPactuado'n
		,max(VrComissaoTotal) as VrComissaoTotal
		,max(VrAcordoComercial) as VrAcordoComercial
		,max(VrRetornoPlus) as VrRetornoPlus
		,max('(%) Comissao'n) as '(%) Comissao'n
		,max('(%) Acordo'n) as '(%) Acordo'n
		,max('(%) TotalComissao'n) as '(%) TotalComissao'n
from base_1
where NuProposta>0
group by NuProposta

;
quit;

data base_3;
set base;
drop  TpDados StProposta DtPrimeiroVencimento '(%) SpreadPactuado'n VrComissaoTotal
VrAcordoComercial VrRetornoPlus '(%) Comissao'n '(%) Acordo'n '(%) TotalComissao'n ;
run;

proc sort data=base_3(where=(NuProposta>0)) nodupkey; by NuProposta; run;
