   CREATE TABLE Fiscal.OBJETOS_TRIBUTARIO_GP_OBJ_&DSRefModel. (compress=yes ) AS
   SELECT t1.PASTA,
		  t1.SERIAL_IMPORT,
          t1.NATUREZA,
          t1.OBJETO,
	      t1.NATUREZA_UNIDADE_CUSTOM as NATUREZA_CUSTOM,
	      (COUNT(t1.OBJETO)) FORMAT=18.2 AS QTDE_OBJETO,
          (MAX(t1.DATA_ALTERACAO_OBJETO)) FORMAT=DATETIME22.3 AS DT_ALTERACAO_OBJETO,
          (SUM(t1.VALOR_PEDIDO)) FORMAT=18.2 AS VALOR_PEDIDO_T,

          (SUM(t1.VALOR_PERDA_PROVAVEL_ATUALIZAD)) FORMAT=20.2 AS PERDA_PROVAVEL_ATUALI_T,
          (SUM(t1.VALOR_PERDA_REMOTO_ATUALIZAD))   FORMAT=20.2 AS PERDA_REMOTO_ATUALI_T,
		  (SUM(t1.VALOR_PERDA_POSSIVEL_ATUALIZAD)+
           			SUM(t1.VALOR_PERDA_PROVAVEL_ATUALIZAD)+
           			SUM(t1.VALOR_PERDA_REMOTO_ATUALIZAD) )FORMAT=20.2 AS VALOR_RISCO_T

      FROM OBJETOS_TRIBUTARIO AS t1
						WHERE STATUS_PROCESSO EQ 'ATIVA'
      GROUP BY t1.PASTA,
               t1.SERIAL_IMPORT,
               t1.NATUREZA,
               t1.OBJETO,
               t1.NATUREZA_UNIDADE_CUSTOM;
QUIT;

          t2.Ultima_Descisao as 'Última Descisão'n,
          t2.Fase ,
          t2.Tipo_de_Tese as 'Tipo de Tese'n,
		  t2.status

      FROM Fiscal.OBJETOS_TRIBUTARIO_GP_OBJ_&DSRefModel.  AS t1
	           LEFT JOIN Fiscal.BASE_GERAL_TRIBUTARIO_&DSRefModel.  AS t2 ON t1.PASTA = t2.PASTA
; QUIT;
