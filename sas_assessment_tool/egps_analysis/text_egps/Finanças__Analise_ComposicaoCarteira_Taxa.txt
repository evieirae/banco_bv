PROC SQL;
CREATE TABLE TBL_CARTEIRA_CARTOESBV_V0 AS

SELECT  A.DtProcessamento,
        YEAR(DATEPART(A.DtProcessamento))*100 + MONTH(DATEPART(A.DtProcessamento)) AS SafraRef,
		YEAR(DATEPART(A.DtInicioContrato))*100 + MONTH(DATEPART(A.DtInicioContrato)) AS SafraCtt_Cadoc,
		DATEPART(A.DtInicioContrato) AS DtContrato_Cadoc,
		A.NuContrato,
		INPUT(SUBSTRN(COMPRESS(A.NuContrato),5,LENGTH(COMPRESS(A.NuContrato))-4),20.) AS CdContratoParcelamentoFtra,
		INPUT(SUBSTRN(COMPRESS(A.NuContrato),1,2),20.) AS CdProduto,

		CASE WHEN a.CdModalidade = 204 THEN 'Carteira Rotativo'
		     WHEN a.CdModalidade = 210 THEN 'Carteira Parcelado'
		     WHEN a.CdModalidade = 218 THEN 'Carteira Rotativo Atraso'
		     WHEN a.CdModalidade = 1304 THEN 'Carteira A Vencer'
		     WHEN a.CdModalidade IN (1901,1904) THEN 'Limite'
		     ELSE 'Verificar'
		     END AS NmGrupoCarteira,

		CASE WHEN A.CdClassificacaoOperacao IN ('A','B','C')
		     THEN 'Carteira Under60'
		     ELSE 'Carteira Over60'
		     END AS FlagCarteiraRotativo,

		CASE WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '01' THEN '01-Rotativo'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '02' THEN '02-Saque'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '03' THEN '03-Parcelamento Emissor'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '04' THEN '04-Pagamento de Contas'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '05' THEN '05-Acordo em Fatura'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '06' THEN '06-Parcelamento de Fatura'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '07' THEN '07-Compras à Vista'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '08' THEN '08-Limite de Crédito'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '09' THEN '09-Parcelamento Total'
		     WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0A' THEN '0A-Saque Parcelado'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0B' THEN '0B-Crédito Pessoal'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0C' THEN '0C-Crediário'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0D' THEN '05-Acordo em Fatura CL'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0E' THEN '0E-PagContas Parcelado'
			 WHEN SUBSTRN(COMPRESS(A.NuContrato),3,2) = '0F' THEN '06-Parcelamento de Fatura'
			 ELSE 'Verificar'
		     END AS NmCarteiraSegregada,

	   CASE WHEN CALCULATED NmGrupoCarteira = 'Carteira A Vencer'                                                                     THEN 'Carteira À Vista'
	        WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '03-Parcelamento Emissor'     THEN 'Carteira Parcelamento Emissor'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura')      THEN 'Carteira Parcelamento Acordo'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('05-Acordo em Fatura CL')   THEN 'Carteira Parcelamento Acordo CL'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada = '09-Parcelamento Total'       THEN 'Carteira Parcelamento Especial'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('06-Parcelamento de Fatura','0C-Crediário','0A-Saque Parcelado','0F-Compulsório') THEN 'Carteira Parcelamento Fatura'

		    WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0B-Crédito Pessoal')     THEN 'Carteira Crédito Pessoal'
			WHEN CALCULATED NmGrupoCarteira = 'Carteira Parcelado' AND CALCULATED NmCarteiraSegregada IN ('0E-PagContas Parcelado') THEN 'Pagamento de Contas Parcelado'

            WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Over60'  THEN 'Carteira Rotativo Over60'
	        WHEN (CALCULATED NmGrupoCarteira = 'Carteira Rotativo' OR CALCULATED NmGrupoCarteira = 'Carteira Rotativo Atraso') AND CALCULATED FlagCarteiraRotativo = 'Carteira Under60' THEN 'Carteira Rotativo Under60'
            ELSE 'Verificar'
			END AS NmCarteiraDRE,

		A.NuCpfCnpj,
		b.Qt_DiasAtraso,
		A.VlSaldoCarteira

FROM TbOrigemClassificacaoRisco AS a

LEFT JOIN
(
	SELECT DISTINCT t.DtProcessamento,
                    t.NuCpfCnpj,
	                MAX(t.QtDiaAtraso) AS Qt_DiasAtraso
	FROM TbOrigemClassificacaoRisco AS t
    WHERE t.CdClassificacaoOperacao <> 'HH'
	GROUP BY 1,2
) AS b
ON a.DtProcessamento = b.DtProcessamento AND
   a.NuCpfCnpj = b.NuCpfCnpj

WHERE a.CdClassificacaoOperacao <> 'HH'
ORDER BY 2;
QUIT;
