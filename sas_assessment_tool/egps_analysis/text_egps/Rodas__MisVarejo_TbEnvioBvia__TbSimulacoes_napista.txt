%include '/infogrc/utils/all-includes.sas';

data simulacoes; set convsimu.tbsimulacoes (where=(nuanomes >= &safraM2.)); run;

data _null_; m2_bq = "'"||put(year("&m2."d),4.)||"-"||put(month("&m2."d),z2.)||"-"||put(day("&m2."d),z2.)||"'"; call symputx('m2_bq',m2_bq); ;run; %put &m2_bq;

proc sql;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
create table simulacoes_qtparcela as
select * from connection to perm(
WITH
  tb1 AS (
  SELECT
    CAST(cd_simulacao_proposta AS numeric) cd_simulacao_proposta,
    MAX(cd_simulacao_financiamento) cd_simulacao_financiamento_max,
  FROM
    `bv-pdbd-prd.dvry_core.tbsimulacoesrodas`
  WHERE
    dt_simulacao_proposta >= &m2_bq
  GROUP BY
    1)
SELECT
  a.cd_simulacao_proposta,
  b.qt_parcelas,
  b.vr_prestacao
FROM
  tb1 a
INNER JOIN (
  SELECT
    cd_simulacao_proposta,
    cd_simulacao_financiamento,
    qt_parcelas,
    vr_prestacao
  FROM
    `bv-pdbd-prd.dvry_core.tbsimulacoesrodas`
  WHERE
    dt_simulacao_proposta >= &m2_bq
    ) b
ON
  a.cd_simulacao_financiamento_max = b.cd_simulacao_financiamento
);
disconnect from perm;
quit;

proc sql;
create table tbpessoa as
select
	a.nutelefone1,
	a.nutelefone2,
	a.nutelefone3,
        a.DsEmail,
	a.DsEmail2,
	a.dtnascimento,
	a.nmpessoa,
	a.nucpfcnpj
from bases.tbpessoa a
inner join (select distinct nucpfcnpj from simulacoes where nucpfcnpj ne "") b
	on a.nucpfcnpj = b.nucpfcnpj
;run;

proc sql;
create table tbmolicar as
select distinct
	input(a.cdmolicar, 9.) as cdmolicar,
	a.aamodelo,
	a.cdmarca,
	a.cdmodelo,
	a.dscategoria
from auxl.tbmolicar a
inner join (select distinct cdmolicar, aamodelo from simulacoes where cdmolicar ne .) b
	on 	input(a.cdmolicar, 9.) = b.cdmolicar and a.aamodelo = b.aamodelo
;run;

proc sql;
create table simulacoesnapista as
select distinct
	 simulacao.nuanomes
	,simulacao.dtsimulacaoinicial as dtsimulacao
	,simulacao.cdsimulacaoinicial as cdsimulacao
	,simulacao.nucpfcnpj
	,simulacao.sistemainicial
	,Pessoa.NuTelefone1
	,Pessoa.NuTelefone2
	,Pessoa.NuTelefone3
	,Pessoa.DsEmail
	,Pessoa.DsEmail2
	,Pessoa.DtNascimento
	,Pessoa.NmPessoa
	,simulacao.produto
	,simulacao.cdparceirocomercial
	,tranwrd(simulacao.nmparceirocomercial, " - ", "-") as nmparceirocomercial
	,simulacao.flcard
	,simulacao.flpap
	,coalesce(simulacao.mensageria,'9. Não houve mensageria') as dsmensageria
	,simulacao.fxentrada
	,simulacao.probabilidadeaprovacao
	,simulacao.motivorecusa
	,simulacao.flpropostaoriginada
	,simulacao.flpropostaaprovada
	,simulacao.flpropostaefetivada
	,simulacao.flconversao
	,simulacao.flconcorrente
	,simulacao.dtinclusaogravame
	,simulacao.banco
	,simulacao.stpagoloja
	,simulacao.nuproposta
	,simulacao.regional
	,simulacao.Canal
	,simulacao.filial
	,simulacao.NmColaborador
	,simulacao.DsLoginN1
	,simulacao.DsLoginN2
	,simulacao.DsLoginN3
	,simulacao.cdmolicar
	,simulacao.aamodelo
	,simulacao.aafabricacao
	,simulacao.idadebem
	,molicar.cdmarca
	,molicar.cdmodelo
	,molicar.dscategoria
	,simulacao.fxidade
	,simulacao.NuPlaca
/*	,simulacao.QtParcela*/
	,simulacao.vrveiculomolicar
	,simulacao.vrveiculosimulacao
	,simulacao.vrproposta as vrsimulacao
	,simulacao.VrEntrada
	,case when  simulacao.pcentradaminima > 1 then  simulacao.pcentradaminima /100 else  simulacao.pcentradaminima end as pcentradaminima
	,parcela.qt_parcelas as qtparcelas
	,parcela.vr_prestacao as vrprestacao
	,simulacao.nucontrato
	,simulacao.dtcontrato
	,simulacao.vrproducao
from simulacoes simulacao
left join tbmolicar molicar on simulacao.cdmolicar = molicar.cdmolicar and simulacao.aamodelo = molicar.aamodelo and simulacao.cdmolicar ne .
left join tbpessoa pessoa on simulacao.nucpfcnpj = pessoa.nucpfcnpj
/*left join leads.abandono on simulacao.cdsimulacaoinicial = abandono.cd_simulacao_proposta*/
inner join auxl.grupo_mensageria msg on msg.mensageria = simulacao.mensageria
left join simulacoes_qtparcela parcela on simulacao.cdsimulacaoinicial = parcela.cd_simulacao_proposta and simulacao.qtparcela = parcela.qt_parcelas
where 1=1
and simulacao.sistemainicial in ('PPAR', 'APAR', 'ACOM', 'APP', 'SITE BV', 'NA_PISTA')
and simulacao.produto in ("Leves")
and simulacao.flsimuvalidapdv=1
and (msg.grupomsg = 'P' or simulacao.banco <> "N/A")
;run;
