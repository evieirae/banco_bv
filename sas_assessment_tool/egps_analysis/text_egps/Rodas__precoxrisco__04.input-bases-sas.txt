PROC SQL;
  CREATE TABLE TbProducao AS
    SELECT
	  p.DtContrato,
	  (YEAR(p.DtContrato) * 100) + MONTH(p.DtContrato) AS SafraProposta,
      p.NuProposta,
	  p.NuContrato,
	  p.NmGrupoProdutoMeta AS Produto,
	  p.NuCpfCnpj,
      'AP' AS StProposta,
	  p.FlBVMais AS FlagSPF,
	  p.FlCap AS FlagCAP,
	  p.FlDemais AS FlagAuto,
	  p.VrSeguros,
	  p.DsTipoCanalProducao,
	  p.TpPessoa,
	  p.CdFilial,
	  p.NmAreaEstruturaComercial AS DsFilial,
	  p.VrProducao,
	   /*Inclui os VP para calculo Guta*/
      ((1+(p.VpTxGerencial/VrProducao)) / (1+(((p.TxTirContabil/100) * (p.VrProducao))/p.VrProducao)) -1 ) * p.VrProducao AS VpSpread,
	  p.VpTxGerencial,
	  p.VpTirContabil,
	  p.VpIdade,
	  p.VpPrazo,
	  p.VpEntrada,
	  p.VpScore,
	  p.VrMargem
    FROM
	  TMP.TbProducao as P

    WHERE
	  p.NmGrupoProdutoMeta IN ('Leves','Motos','Pesados','CVG Leves','CVG Pesados','Leves-BVO')
      AND p.DtContrato >= &inicioDate.;
RUN;

PROC SQL;
CREATE TABLE ScoreProposta01 AS
  SELECT DISTINCT
    NuProposta,
	COALESCE(NuRating,NuScoreFim) AS NuRating
  FROM TMP.TBEXTPROPOSTA
  WHERE DtEntradaProposta >= &inicioDate.;
RUN;

PROC SORT DATA=ScoreProposta01 OUT=ScoreProposta NODUPKEY;
BY NuProposta;
RUN;

PROC SQL;
  CREATE TABLE TbOriginacao AS
    SELECT
	  a.DtEntradaProposta,
      (YEAR(a.DtEntradaProposta) * 100) + MONTH(a.DtEntradaProposta) AS SafraProposta,
      a.NuProposta,
      a.NuContratoGestao AS NuContrato,
	  a.NmGrupoProdutoMeta AS Produto,
	  a.NuCpfCnpj,
      a.StProposta,
	  a.TpPessoa,
	  a.TpOrigemTaxa,
      a.CdParceiroComercial AS CdConvenio,
      a.TxGerencial,
      a.FlPropostaOriginada,
      a.FlPropostaAprovada,
      a.FlPropostaEfetivada,
	  a.FlConcorrente AS FlConversao,
	  CASE WHEN a.FlBancoBrasil = 1 THEN 'BB'
           WHEN a.FlBradesco = 1 THEN 'Bradesco'
           WHEN a.FlItau = 1 THEN 'Itau'
           WHEN a.FlPan = 1 THEN 'Pan'
           WHEN a.FlSafra = 1 THEN 'Safra'
           WHEN a.FlSantander = 1 THEN 'Santander'
           WHEN a.FlMontadoras = 1 THEN 'Montadoras'
           WHEN a.FlOutros = 1 THEN 'Outros'
	  END AS BancoConversao,
	  CASE WHEN (a.StProposta IN ('CA', 'A', 'PA', 'VA', 'P', 'CE')) OR a.FlPropostaAPNP = 1 THEN 'APNP'
	       WHEN a.FlPropostaNegada = 1 THEN 'Negada'
		   WHEN a.StProposta = 'C' THEN 'Cancelada'
	  END AS StPropostaConversao,
      a.QtAnosIdade,
      a.FlTxAlterada,
      a.FxIdade,
      b.NuRating AS NuScoreFim,
      a.CdTabelaFinanciamento,
      a.VrPrestacao,
      a.PcEntrada,
      a.QtDiasCarencia,
      a.VrFinanciamento,
      a.QtParcelas,
      a.VrPDD,
      (a.VrPDD / a.VrFinanciamento) AS RUPE,
      CASE WHEN (a.VrPDD / a.VrFinanciamento) BETWEEN  0.0 AND 0.0399 THEN "0-3"
           WHEN (a.VrPDD / a.VrFinanciamento) BETWEEN 0.04 AND 0.0699 THEN "4-6"
           WHEN (a.VrPDD / a.VrFinanciamento) BETWEEN 0.07 AND 0.0999 THEN "7-9"
           WHEN (a.VrPDD / a.VrFinanciamento) BETWEEN  0.1 AND 0.1299 THEN "10-12"
           WHEN (a.VrPDD / a.VrFinanciamento) BETWEEN 0.13 AND 0.1599 THEN "13-15"
           WHEN (a.VrPDD / a.VrFinanciamento) >= 0.16 THEN "Maior que 15"
	  END AS FaixaRUPE,
      a.TxTIRContabil,
      a.CdUFPessoa AS UF,
      CASE WHEN a.CdUfPessoa IN ('AC','AM','AP','PA','RO','RR','TO') THEN 'Norte'
           WHEN a.CdUfPessoa IN ('AL','BA','CE','MA','PB','PE','PI','RN','SE') THEN 'Nordeste'
           WHEN a.CdUfPessoa IN ('DF','GO','MS','MT') THEN 'Centro-oeste'
           WHEN a.CdUfPessoa IN ('SP','RJ','ES','MG') THEN 'Sudeste'
           WHEN a.CdUfPessoa IN ('PR','RS','SC') THEN 'Sul'
           ELSE "Não identificado"
      END AS RegiaoBR,
      a.NmSubEstruturaComercialA AS Regional,
	  a.DsTipoCanalProducao,
	  a.CdFilial,
	  a.NmAreaEstruturaComercial AS DsFilial,
      a.VrProposta,
      a.VrEntrada,
      a.VrLiberado,
	  a.VrPropostaOriginada,
	  /*Inclui os VP para calculo Guta */
	  (((1+(A.VpTxGerencial/A.VrLiberado)) / (1+(((A.TxTirContabil/100) * (A.VrLiberado))/A.VrLiberado)) -1 ) * A.VrLiberado) AS VpSpread,
	  a.VpTxGerencial,
	  a.VpTirContabil,
	  a.VpIdade,
	  a.VpPrazo,
	  a.VpEntrada,
	  a.VpScore,
	  a.tpVeiculo
	FROM TMP.TbOriginacao a
	LEFT JOIN ScoreProposta b
	  ON a.NuProposta = b.NuProposta
	WHERE a.DtEntradaProposta >= &inicioDate.
      AND a.NmGrupoProdutoMeta IN ('Leves','Motos','Pesados','CVG Leves','CVG Pesados','Leves-BVO');
RUN;

PROC SQL;
  CREATE TABLE TbConsolidado_Step01 AS
    SELECT DISTINCT
	  COALESCE(Originacao.SafraProposta,Producao.SafraProposta) AS Safra,
	  COALESCE(Producao.NuProposta, Originacao.NuProposta) AS NuProposta,
	  Producao.NuContrato AS NuContratoProd,
	  COALESCE(Producao.NuContrato, Originacao.NuContrato) FORMAT=15. AS NuContrato,
	  COALESCE(Producao.Produto, Originacao.Produto) AS Produto,
	  COALESCE(Producao.NuCpfCnpj, Originacao.NuCpfCnpj) AS NuCpfCnpj,
	  Originacao.DtEntradaProposta,
	  Originacao.TpOrigemTaxa,
	  Producao.DtContrato,
      COALESCE(Producao.StProposta, Originacao.StProposta) AS StProposta,
	  COALESCE(Producao.FlagSPF,0) AS FlagSPF,
	  COALESCE(Producao.FlagCAP,0) AS FlagCAP,
	  COALESCE(Producao.FlagAuto,0) AS FlagAuto,
	  Producao.VrSeguros,
	  COALESCE(Originacao.DsTipoCanalProducao, Producao.DsTipoCanalProducao) AS DsTipoCanalProducao,
	  Originacao.CdFilial,
	  COALESCE(Originacao.DsFilial, Producao.DsFilial) AS DsFilial,
      COALESCE(Originacao.TpPessoa, Producao.TpPessoa) AS TpPessoa,
      Originacao.CdConvenio,
      Originacao.TxGerencial,
      COALESCE(Originacao.FlPropostaOriginada,0) AS FlPropostaOriginada,
      COALESCE(Originacao.FlPropostaAprovada,0) AS FlPropostaAprovada,
      COALESCE(Originacao.FlPropostaEfetivada,0) AS FlPropostaEfetivada,
	  COALESCE(Originacao.FlConversao,0) AS FlConversao,
	  Originacao.BancoConversao,
	  Originacao.StPropostaConversao,
      Originacao.QtAnosIdade,
      COALESCE(Originacao.FlTxAlterada,0) AS FlTxAlterada,
      Originacao.FxIdade,
      Originacao.NuScoreFim,
      Originacao.CdTabelaFinanciamento,
      Originacao.VrPrestacao,
      Originacao.PcEntrada,
      Originacao.QtDiasCarencia,
      Originacao.VrFinanciamento,
      Originacao.QtParcelas,
      Originacao.VrPDD,
	  Originacao.RUPE,
	  Originacao.FaixaRUPE,
	  Originacao.TxTIRContabil,
	  Originacao.UF,
	  Originacao.RegiaoBR,
	  Originacao.Regional,
      Originacao.VrProposta,
      Originacao.VrEntrada,
      Originacao.VrLiberado,
	  Originacao.VrPropostaOriginada AS VrPropostaOriginada,
	  Producao.VrProducao AS VrProducao,
	  Originacao.Tpveiculo,

	  /*Incluir aqui tambem o Valor ponderado Guta*/
	  COALESCE(Producao.VpSpread, Originacao.VpSpread) FORMAT=15.4 AS VpSpread,
	  COALESCE(Producao.VpTxGerencial, Originacao.VpTxGerencial) FORMAT=15.2 AS VpTxGerencial,
	  COALESCE(Producao.VpTirContabil, Originacao.VpTirContabil) FORMAT=15.2 AS VpTirContabil,
	  COALESCE(Producao.VpIdade, Originacao.VpIdade) FORMAT=15.2 AS VpIdade,
	  COALESCE(Producao.VpPrazo, Originacao.VpPrazo) FORMAT=15.2 AS VpPrazo,
	  COALESCE(Producao.VpEntrada, Originacao.VpEntrada) FORMAT=15.2 AS VpEntrada,
	  COALESCE(Producao.VpScore, Originacao.VpScore) FORMAT=15.2 AS VpScore,
	  Producao.VrMargem FORMAT=15.2
    FROM TbProducao Producao
	FULL OUTER JOIN TbOriginacao Originacao
      ON Producao.NuProposta = Originacao.NuProposta;
RUN;

PROC SQL;

	  MIN(TxGerencial) AS TxGerencial,
	  MIN(QtAnosIdade) AS QtAnosIdade,
	  MIN(FxIdade) AS FxIdade,
	  MIN(CdTabelaFinanciamento) AS CdTabelaFinanciamento,
	  MIN(CdFilial) AS CdFilial,
	  MIN(DsFilial) AS DsFilial,
	  MIN(VrPrestacao) AS VrPrestacao,
	  MIN(PcEntrada) AS PcEntrada,
      MIN(QtDiasCarencia) AS QtDiasCarencia,
      MIN(VrFinanciamento) AS VrFinanciamento,
      MIN(QtParcelas) AS QtParcelas,
      MIN(VrPDD) AS VrPDD,

      MIN(TxTIRContabil) AS TxTIRContabil,
      MIN(UF) AS UF,
      MIN(RegiaoBR) AS RegiaoBR,
      MIN(Regional) AS Regional,
      MIN(VrProposta) AS VrProposta,
      MIN(VrEntrada) AS VrEntrada,
      MIN(VrLiberado) AS VrLiberado,
	  MIN(VrPropostaOriginada) AS VrPropostaOriginada,
	  MIN(VrProducao) AS VrProducao,
	  /* Inclui o valor ponderado Guta*/
	  MIN(VpSpread) as VpSpread,
	  MIN(VpTxGerencial) as VpTxGerencial,
	  MIN(VpTirContabil) as VpTirContabil,
	  MIN(VpIdade) as VpIdade,
	  MIN(VpPrazo) as VpPrazo,
	  MIN(VpEntrada) as VpEntrada,

	  MIN(VrMargem) as VrMargem,
	  MIN(Tpveiculo) as Tpveiculo,
	  CASE WHEN NuContrato = NuContratoProd THEN 'Manter' ELSE 'Excluir' END AS FlExcluir

	WHERE NuContrato <> .
	GROUP BY NuContrato
    HAVING COUNT(1) > 1;

PROC SQL;

    SELECT DISTINCT
	  Consolidado.Safra,
	  Consolidado.NuProposta,
	  Consolidado.NuContrato,
	  Consolidado.NuCpfCnpj,

	  COALESCE(Limpeza.DtContrato, Consolidado.DtContrato) FORMAT=DATE9. AS DtContrato,
      Consolidado.StProposta,
	  Consolidado.TpOrigemTaxa,
	  Consolidado.FlagSPF,
	  Consolidado.FlagCAP,
	  Consolidado.FlagAuto,

      Consolidado.FlTxAlterada,
      COALESCE(Limpeza.FxIdade, Consolidado.FxIdade) AS FxIdade,

      COALESCE(Limpeza.CdTabelaFinanciamento, Consolidado.CdTabelaFinanciamento) AS CdTabelaFinanciamento,

	  COALESCE(Limpeza.FaixaRUPE, Consolidado.FaixaRUPE) AS FaixaRUPE,
	  COALESCE(Limpeza.TxTIRContabil, Consolidado.TxTIRContabil) AS TxTIRContabil,

      /*Inlui o valor ponderado Guta*/
	  COALESCE(Limpeza.VpSpread, Consolidado.VpSpread) AS VpSpread,
	  COALESCE(Limpeza.VpTxGerencial, Consolidado.VpTxGerencial) AS VpTxGerencial,
	  COALESCE(Limpeza.VpTirContabil, Consolidado.VpTirContabil) AS VpTirContabil,
	  COALESCE(Limpeza.VpIdade, Consolidado.VpIdade) AS VpIdade,

	  COALESCE(Limpeza.VrMargem, Consolidado.VrMargem) AS VrMargem

	LEFT JOIN TbConsolidado_Limpeza01 Limpeza
      ON Consolidado.NuProposta = Limpeza.NuProposta
    WHERE FlExcluir <> 'Excluir';

  SELECT *
  FROM DWBI_DTWH.TbAgenteMIS

);

PROC SQL;
CREATE TABLE TbCetipConversaoEnvio AS
  SELECT
    *,
    CASE WHEN DtEntradaProposta < '14JUL2021'd
         THEN SUBSTR(chave,1,30)||PUT(VrFinanciamento,z10.)
         ELSE chave
    END AS chave2 FORMAT=$53.
  FROM BASES.TbCetipConversaoEnvio;
RUN;

PROC SQL;
CREATE TABLE CONVERSAO AS
  SELECT DISTINCT
    A.NuProposta,
    CASE WHEN Agente.NmGrupoAgenteMIS = 'MONTADORAS' THEN 'MONTADORAS'
         ELSE Agente.NmSubgrupoAgenteMis
    END AS Banco,
    INTCK('DAY',A.DtEntradaProposta,B.Dt_Gravame) AS DiasConversao
  FROM TBCETIPCONVERSAOENVIO A
  LEFT JOIN BASES.TBCETIPCONVERSAORETORNO B
    ON A.Chave2 = B.Chave
  INNER JOIN TbAgente Agente
    ON B.cnpj_agfin = Agente.NuCNPJAgente
  WHERE Agente.NmSubgrupoAgenteMis <> 'BV';
RUN;

PROC SORT DATA=CONVERSAO NODUPKEY;
BY NuProposta;
RUN;
