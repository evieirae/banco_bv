CREATE TABLE CARTEIRA_TOTAL_&REF. AS

FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT DISTINCT
    CAST(a.nucontrato AS INT64) AS NUCONTRATO,

      ,A.SQCONTRATOFINANCEIRO

	  ,A.SAFRA
      ,B.DTULTIMORECEBIMENTO

PROC SQL;
CREATE TABLE TIPO_DE_PAGAMENTO AS
SELECT  NUCONTRATO,
		SQCONTRATOFINANCEIRO,
		DATA_PAGAMENTO,
		SAFRA,
		SUM(FL_ATRASO) AS QTD_PARC_ATRASO,
		SUM(FL_PAGO_NO_DIA) AS QTD_PARC_PAGAS,
		COUNT(NUCONTRATO) AS QTD_PARCELAS_ABERTO,
		CASE WHEN DATA_PAGAMENTO IS NULL THEN 'SEM PG'
			 WHEN CALCULATED QTD_PARC_PAGAS < CALCULATED QTD_PARC_ATRASO THEN 'PMT'
			 WHEN CALCULATED QTD_PARC_PAGAS = CALCULATED QTD_PARCELAS_ABERTO THEN 'QUITACAO'
			 ELSE 'ATUALIZACAO' END AS TIPO_PAGAMENTO

FROM BASE_UNICA_LEFT
WHERE DTULTIMORECEBIMENTO >= DATA_PAGAMENTO OR DTULTIMORECEBIMENTO IS NULL
GROUP BY NUCONTRATO, SQCONTRATOFINANCEIRO, DATA_PAGAMENTO

;QUIT;

PROC SQL;
%CONNECT_GCP;
CREATE TABLE CARTEIRA_TOTAL AS
SELECT A.*
,SALDO AS SALDO_2 FORMAT COMMAX17.2
FROM CONNECTION TO PERM (
WITH QUERY_1 AS (

SELECT DISTINCT
    CAST(a.nucontrato AS INT64) AS NUCONTRATO,
	c.cd_cpf_cnpj AS nucpfcnpj,
    a.sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,

        WHEN a.nudiasatraso BETWEEN 0 AND 90 THEN 'aa. Até 90'
        WHEN a.nudiasatraso BETWEEN 91 AND 150 THEN 'a. 91 a 150'
        WHEN a.nudiasatraso BETWEEN 151 AND 180 THEN 'b. 151 a 180'

    a.dtprocessamento = '2025-06-30'

    AND a.nudiasatraso <= 540
    AND a.cdproduto IN (13)
    AND (CAST(c.cd_modalidade_produto AS INT64) = 55 OR d.cdsubproduto = 70)

UNION ALL

SELECT DISTINCT

PROC SQL;
CREATE TABLE BASE_UNICA_LEFT AS
SELECT DISTINCT A.NUCONTRATO
      ,A.SQCONTRATOFINANCEIRO
      ,A.DATA_PAGAMENTO
      ,B.DTULTIMORECEBIMENTO
      ,B.DTVENCIMENTO

      ,CASE WHEN B.DTVENCIMENTO < A.DATA_PAGAMENTO AND (B.DTULTIMORECEBIMENTO >= A.DATA_PAGAMENTO OR
														B.DTULTIMORECEBIMENTO IS NULL) THEN 1 ELSE 0 END AS FL_ATRASO

      ,CASE WHEN B.DTULTIMORECEBIMENTO = A.DATA_PAGAMENTO THEN 1 ELSE 0 END AS FL_PAGO_NO_DIA

FROM RECEBIMENTOS_202507 A
LEFT JOIN TB_PARCELA_FULL B ON
   A.NUCONTRATO = INPUT(B.NUCONTRATO,14.)
   AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

;QUIT;



PROC SQL;
CREATE TABLE TIPO_DE_PAGAMENTO AS
SELECT  NUCONTRATO,
		SQCONTRATOFINANCEIRO,
		DATA_PAGAMENTO,
		SUM(FL_ATRASO) AS QTD_PARC_ATRASO,
		SUM(FL_PAGO_NO_DIA) AS QTD_PARC_PAGAS,
		COUNT(NUCONTRATO) AS QTD_PARCELAS_ABERTO,
		CASE WHEN DATA_PAGAMENTO IS NULL THEN 'SEM PG'
			 WHEN CALCULATED QTD_PARC_PAGAS < CALCULATED QTD_PARC_ATRASO THEN 'PMT'
			 WHEN CALCULATED QTD_PARC_PAGAS = CALCULATED QTD_PARCELAS_ABERTO THEN 'QUITACAO'
			 ELSE 'ATUALIZACAO' END AS TIPO_PAGAMENTO

FROM BASE_UNICA_LEFT
WHERE DTULTIMORECEBIMENTO >= DATA_PAGAMENTO OR DTULTIMORECEBIMENTO IS NULL
GROUP BY NUCONTRATO, SQCONTRATOFINANCEIRO, DATA_PAGAMENTO

;QUIT;

PROC SQL;
CREATE TABLE BASE_UNICA_LEFT AS
SELECT DISTINCT A.NUCONTRATO
      ,A.SQCONTRATOFINANCEIRO
      ,A.DATA_PAGAMENTO
      ,B.DTULTIMORECEBIMENTO
      ,B.DTVENCIMENTO

      ,CASE WHEN B.DTVENCIMENTO < A.DATA_PAGAMENTO AND (B.DTULTIMORECEBIMENTO >= A.DATA_PAGAMENTO OR
														B.DTULTIMORECEBIMENTO IS NULL) THEN 1 ELSE 0 END AS FL_ATRASO

      ,CASE WHEN B.DTULTIMORECEBIMENTO = A.DATA_PAGAMENTO THEN 1 ELSE 0 END AS FL_PAGO_NO_DIA

FROM RECEBIMENTOS_202507 A
LEFT JOIN TB_PARCELA_FULL B ON
   A.NUCONTRATO = INPUT(B.NUCONTRATO,14.)
   AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

;QUIT;



PROC SQL;
CREATE TABLE TIPO_DE_PAGAMENTO AS
SELECT  NUCONTRATO,
		SQCONTRATOFINANCEIRO,
		DATA_PAGAMENTO,
		SUM(FL_ATRASO) AS QTD_PARC_ATRASO,
		SUM(FL_PAGO_NO_DIA) AS QTD_PARC_PAGAS,
		COUNT(NUCONTRATO) AS QTD_PARCELAS_ABERTO,
		CASE WHEN DATA_PAGAMENTO IS NULL THEN 'SEM PG'
			 WHEN CALCULATED QTD_PARC_PAGAS < CALCULATED QTD_PARC_ATRASO THEN 'PMT'
			 WHEN CALCULATED QTD_PARC_PAGAS = CALCULATED QTD_PARCELAS_ABERTO THEN 'QUITACAO'
			 ELSE 'ATUALIZACAO' END AS TIPO_PAGAMENTO

FROM BASE_UNICA_LEFT
WHERE DTULTIMORECEBIMENTO >= DATA_PAGAMENTO OR DTULTIMORECEBIMENTO IS NULL
GROUP BY NUCONTRATO, SQCONTRATOFINANCEIRO, DATA_PAGAMENTO

;QUIT;

%let SEED_RDG_MASCARA_RED = 16; 

PROC SQL;
CREATE TABLE base_analise AS
SELECT A.*
,B.TIPO_PAGAMENTO
,CASE
        WHEN a.atraso_m0 BETWEEN 91 AND 150 THEN 'a. 91 a 150'
        WHEN a.atraso_m0 BETWEEN 151 AND 360 THEN 'b. 151 a 360'
		ELSE 'c. 361 a 540'
    END AS FX_ATRASO2
,put(A.SALDO, FAIXA_SALDO.) AS FAIXA_SALDO
,put(A.SALDO, FAIXA_SALDO_B.) AS FAIXA_SALDO2
,CASE WHEN A.DIA is null then 'S/ PG'
	  WHEN A.DIA < 14 THEN 'S/ CONTROLE FAKE'
	  ELSE 'C/ CONTROLE FAKE' END AS FL_MUDANCA
,MOD(input(hashing('sha1',PUT(A.NUCONTRATO+INPUT(A.NUCPFCNPJ,11.)+&SEED_RDG_MASCARA_RED.,z11.)),hex8.),100) as RDG_MASCARADO_RED
,CASE WHEN CALCULATED RDG_MASCARADO_RED < 50 THEN 'TESTE_RED'
	  WHEN CALCULATED RDG_MASCARADO_RED < 90 THEN 'CONTROLE_FAKE'
	  ELSE 'CONTROLE' END AS TESTE_SOLAR2
   FROM WORK.RECEBIMENTOS_202507 A
LEFT JOIN TIPO_DE_PAGAMENTO B ON A.NUCONTRATO = B.NUCONTRATO AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO

;QUIT;
