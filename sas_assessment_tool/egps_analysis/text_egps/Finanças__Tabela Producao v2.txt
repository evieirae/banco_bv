PROC SQL;
   CREATE TABLE WORK.OUTPUT_Dash_Prod_&anomes. AS
   SELECT t1.Safra,
          t1.PRODUTO,

            coalesce((SUM(t1.VrProducaoLiquida)),0) FORMAT=22.2 AS ProducaoLiquida,

			coalesce((SUM(t1.VpTaxaNet)),0) AS TxNet,
            coalesce((SUM(t1.VpTirDivulgada)),0) AS TirDivulgada,

            coalesce((SUM(t1.QtSeguros)),0) AS QtSeguros,
