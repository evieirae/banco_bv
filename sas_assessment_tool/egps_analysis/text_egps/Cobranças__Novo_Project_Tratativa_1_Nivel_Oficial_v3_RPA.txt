CREATE TABLE WORK.BA_ASSESS_ATIVA AS
SELECT DISTINCT

	INPUT(NR_CONTRATO,BEST14.) AS NUCONTRATOGESTAO FORMAT BEST14.,
	SQCONTAT AS SQCONTRATOFINANCEIRO,
	COD_ACAO,
	COD_ASSESSORIA,
	ASSESSORIA,
	DT_DISPONIBILIZACAO AS DT_ENVIO,
	DT_NOVA_DISPONIB AS DT_RETIRADA,
	CASE WHEN DT_NOVA_DISPONIB IN (.) THEN 'S' ELSE 'N' END AS FLAG_STATUS_COB

FROM CONNECTION TO PERM (

WITH PARAMETRO AS (
SELECT
FORMAT_DATE('%Y%m', CURRENT_DATE()) AS DT_BASE, 
CURRENT_DATE AS DTREF,

),BASE01 AS (
		SELECT
			B.DTREF,
			SUBSTR(NR_CONTRATO,1,14) AS NR_CONTRATO,
		    CAST(SQCONTAT AS NUMERIC) AS SQCONTAT,
		    NR_CPFCNPJ,
		    NR_DIAS_ATRASO AS ATRASO_SELEC_ENVIO,
		    CAST(CD_ASSESSORIA AS NUMERIC) AS COD_ASSESSORIA,
			CASE
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=3000 THEN 'CONTROLE_CYBER'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2020 THEN 'NO_ACTION_PILOTO4'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2019 THEN 'NO_ACTION_PILOTO3'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2018 THEN 'NO_ACTION_PILOTO2'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2017 THEN 'NO_ACTION_PILOTO1'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2010 THEN 'COB_ATIVA - PASCHOALOTTO DIGITAL'
		        WHEN CAST(CD_ASSESSORIA AS NUMERIC)=2009 THEN 'COB_ATIVA - BELLINATI DIGITAL'

		WHEN D.NUCONTRATOGESTAO IS NOT NULL THEN D.COD_ACAO
		WHEN C.NUCONTRATOGESTAO IS NOT NULL THEN C.COD_ACAO
		END AS COD_ACAO,

		WHEN B.NUCONTRATOGESTAO IS NOT NULL THEN B.COD_ASSESSORIA
		WHEN D.NUCONTRATOGESTAO IS NOT NULL THEN D.COD_ASSESSORIA

LEFT JOIN BASE_ENVIO_ASSESSORIA C
ON G.NUCONTRATOGESTAO=C.NUCONTRATOGESTAO
