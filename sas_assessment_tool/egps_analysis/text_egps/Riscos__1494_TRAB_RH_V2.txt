PROC SQL;
       create table BASE_CONTABILIDADE_CQ as
              select 	T1.Chave_Exyon,
                    	T1.Assunto_Judicial,
               			T1.Fase_Judicial,
                    	T1.Data_Cadastro,
					    T1.DBProvisao     as DBProvisao2C,

                        T2.NUMERO_PROCESSO,
                        T2.PERITO AS PERITO_ATUAL,

PROC SQL;
   CREATE TABLE BASE_CQ_RESUMO AS
   SELECT
                                                                       t1.Assunto_Judicial,
                                                                       t1.ANO_MES_CADASTRO_RT,

      FROM BASE_CONTABILIDADE_CQ_T AS t1
                                           WHERE t1.FLAG_CONSIDERAR = 1 and t1.Data_Cadastro >= '01jul2020'd
      GROUP BY t1.Assunto_Judicial,
                                                                            t1.ANO_MES_CADASTRO_RT

      ;
QUIT;
