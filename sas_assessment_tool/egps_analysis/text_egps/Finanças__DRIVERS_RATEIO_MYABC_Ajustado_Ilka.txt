CREATE TABLE CARTEIRA_CART AS
SELECT
	tb1.*
	,CASE WHEN tb1.ClassOp IN ('A','B','C') THEN 'Carteira Under60'
		  ELSE 'Carteira Over60'
		  END AS FlagCarteiraAtraso
	,SUBSTRN(COMPRESS(tb1.Contrt),3,2) as TipoContrato
	,(COMPRESS(tb1.CdProduto)||COMPRESS(tb1.CdCliente)) as ChaveUnica
	,COALESCE(tb2.CdDimContaCartao,tb1.CdDimContaCartao_CADOC) AS CdContaCartao

FROM
 (
	SELECT
		&SAFRA. AS SafraRef
		,Coalesce(A.Mod, B.Mod) AS Mod
		,Coalesce(A.ClassOp, B.ClassOp) AS ClassOp
		,Coalesce(A.Contrt,B.Contrt) AS Contrt
		,Coalesce(A.cd, B.cd) AS CdCliente
		,Coalesce(A.CdDimCtaCartao, B.CdDimCtaCartao) AS CdDimContaCartao_CADOC
		,Coalesce(A.cdproduto, B.cdproduto) AS CdProduto
		,Coalesce(A.FlagCartAccrual, B.FlagCartAccrual) AS FlagCartAccrual
		,CASE WHEN (A.cd = "" or A.ClassOp = "HH") THEN 0
			  ELSE Coalesce(A.Saldo,0) END AS SaldoCarteiraTotal format COMMA18.2
		,CASE WHEN A.FlagCobranca = "AMIGAVEL" THEN Coalesce(calculated SaldoCarteiraTotal,0)
			  ELSE 0 END AS SaldoCarteiraAmigavel format COMMA18.2
		,CASE WHEN A.FlagCobranca = "CONTENCIOSA" THEN Coalesce(calculated SaldoCarteiraTotal,0)
			  ELSE 0 END AS SaldoCarteiraContenciosa format COMMA18.2

	FROM
	(
		SELECT DISTINCT
			a1.Cd
			,((INPUT(a1.CdProduto,8.)*1000000000) + INPUT(a1.NuConta, 9.)) AS CdDimCtaCartao
			,a1.Contrt
			,a1.CdProduto
			,a1.Mod
			,CASE WHEN coalesce(b1.DiaAtraso,0) <= 59 THEN 'ACCRUAL' ELSE 'NAO ACCRUAL' END AS FlagCartAccrual
			,CASE WHEN coalesce(b1.DiaAtraso,0) >= 5 AND coalesce(b1.DiaAtraso,0) <= 74 THEN 'AMIGAVEL'
				  WHEN coalesce(b1.DiaAtraso,0) >= 75 AND coalesce(b1.DiaAtraso,0) <= 360 THEN 'CONTENCIOSA'
				  ELSE 'OUTROS' END AS FlagCobranca
			,a1.ClassOp
			,Sum(COALESCE(a1.Saldo),0) AS Saldo
		FROM ESTUDOS2.CARTAO3040_&SAFRA. AS a1

		LEFT JOIN
		(
			SELECT
				Cd
				,MAX(DiaAtraso) AS DiaAtraso
			FROM ESTUDOS2.CARTAO3040_&SAFRA.
			GROUP BY 1
		) AS b1 ON a1.Cd = b1.Cd

		WHERE a1.Mod NOT IN ('1901','1904')

		GROUP BY
		a1.Cd
		,CALCULATED CdDimCtaCartao
		,a1.Contrt
		,a1.CdProduto
		,a1.Mod
		,CALCULATED FlagCartAccrual
		,CALCULATED FlagCobranca
		,a1.ClassOp
	) AS A

	FULL JOIN
	(
		SELECT DISTINCT
			a2.Cd
			,((INPUT(a2.CdProduto,8.)*1000000000) + INPUT(a2.NuConta, 9.)) AS CdDimCtaCartao
			,a2.Contrt
			,a2.CdProduto
			,a2.Mod
			,CASE WHEN coalesce(b2.DiaAtraso,0) <= 59 THEN 'ACCRUAL' ELSE 'NAO ACCRUAL' END AS FlagCartAccrual
			,CASE WHEN coalesce(b2.DiaAtraso,0) >= 5 AND coalesce(b2.DiaAtraso,0) <= 74 THEN 'AMIGAVEL'
				  WHEN coalesce(b2.DiaAtraso,0) >= 75 AND coalesce(b2.DiaAtraso,0) <= 360 THEN 'CONTENCIOSA'
				  ELSE 'OUTROS' END AS FlagCobranca
			,a2.ClassOp
			,Sum(COALESCE(a2.Saldo),0) AS Saldo
		FROM ESTUDOS2.CARTAO3040_&SafraAnterior. AS a2

		LEFT JOIN
		 (
			SELECT Cd
			,MAX(DiaAtraso) AS DiaAtraso
			FROM ESTUDOS2.CARTAO3040_&SafraAnterior.
			GROUP BY 1
		 ) AS b2 ON a2.Cd = b2.Cd

		WHERE a2.Mod NOT IN ('1901','1904')

		GROUP BY
		a2.Cd
		,CALCULATED CdDimCtaCartao
		,a2.Contrt
		,a2.CdProduto
		,a2.Mod
		,CALCULATED FlagCartAccrual
		,CALCULATED FlagCobranca
		,a2.ClassOp
	) AS B ON A.Cd = B.Cd
		   AND A.Contrt = B.Contrt
		   AND A.CdProduto = B.CdProduto
		   AND A.Mod = B.Mod
		   AND A.CdDimCtaCartao = B.CdDimCtaCartao
 ) AS tb1


left join IQ_CARTAO AS tb2 ON (COMPRESS(tb1.CdProduto)||COMPRESS(tb1.CdCliente)) = tb2.ChaveUnica

PROC SQL;
%ConexaoIQ(server=BI);
CREATE TABLE GASTO_AUTORIZADO AS
SELECT * FROM connection to dbcon
 (
	SELECT
		&SAFRA. AS SafraRef
		,t.CdDimContaCartao AS CdContaCartao
		,t.TpProdutoCartao1 AS TpProdutoCartao
		,SUM(t.QtdAutorizacoes) AS QtdAutorizacoes
		,SUM(t.QtdContasAtivas_30dias) AS QtdContasAtivas_30dias
		,SUM(t.SaldoGastoAutorizado1) AS SaldoGastoAutorizado
		,CASE WHEN TpProdutoCartao NOT IN (1,5,14,16) THEN SaldoGastoAutorizado ELSE 0 END AS SaldoGastoAutorizado_MASTER
		,CASE WHEN TpProdutoCartao IN (1,5,14) THEN SaldoGastoAutorizado ELSE 0 END AS SaldoGastoAutorizado_VISA
		,CASE WHEN TpProdutoCartao IN (16) THEN SaldoGastoAutorizado ELSE 0 END AS SaldoGastoAutorizado_ELO

	FROM
	(
		SELECT
			year(a.DtSolicitacaoAutorizacaoEmsr)*100+month(a.DtSolicitacaoAutorizacaoEmsr) AS SafraRef
			,a.CdDimContaCartao
			,round(a.CdDimContaCartao/1000000000,1) as TpProdutoCartao1
			,CASE WHEN c.nuordemtitularidadeplastico = 0 THEN 'Titular' ELSE 'Adicional' END AS Titularidade
			,count(*) AS QtdAutorizacoes
			,count(DISTINCT a.cddimcontacartao) AS QtdContasAtivas_30dias
			,sum(CASE WHEN a.vrautorizacao >= 1000000000 THEN 0 ELSE a.vrautorizacao END) AS SaldoGastoAutorizado1

		FROM DWBI_DMOC.TbFatoSolicitacaoAutorizacao a

		INNER JOIN DWBI_DIME.TbDimContaCartao b ON a.cddimcontacartaoversao = b.cddimcontacartaoversao

		INNER JOIN DWBI_DIME.tbdimplastico c ON a.CdDimensaoPlasticoVersionado = c.cddimplasticoversao

		WHERE
			datepart(year, a.DtSolicitacaoAutorizacaoEmsr)*100 + datepart(month, a.DtSolicitacaoAutorizacaoEmsr) = &SAFRA.

			AND a.cdrespostaautorizacao = 1
			AND b.VrLimiteContaCartaoGeral <> 0

		GROUP BY
			year(a.DtSolicitacaoAutorizacaoEmsr)*100+month(a.DtSolicitacaoAutorizacaoEmsr)
			,a.CdDimContaCartao
			,CASE WHEN c.nuordemtitularidadeplastico = 0 THEN 'Titular' ELSE 'Adicional' END
			,round(a.CdDimContaCartao/1000000000,1)
	) AS t
	GROUP BY
		SafraRef
		,t.CdDimContaCartao
		,t.TpProdutoCartao1
	ORDER BY 1,2,3
 )
;

CREATE TABLE BASE_RATEIO_CART AS
SELECT
	A.CdContaCartao AS CdContaCartao
	,COALESCE(B1.SaldoCart_Total,0) AS CARTEIRA
	,COALESCE(B.SaldoCarteiraAmigavel,0) AS AMIGAVEL
	,COALESCE(B.SaldoCarteiraContenciosa,0) AS CONTENCIOSA
	,COALESCE(C.QtdContasAptas,0) AS CONTA_APTA
	,COALESCE(D.QtdAutorizacoes,0) AS QTDE_AUTORIZACOES
	,COALESCE(D.QtdContasAtivas_30dias,0) AS CONTA_ATIVA
	,COALESCE(D.SaldoGastoAutorizado,0) AS GASTO_AUTORIZADO
	,COALESCE(D.SaldoGastoAutorizado_MASTER,0) AS GASTO_AUTORIZADO_MASTER
	,COALESCE(D.SaldoGastoAutorizado_VISA,0) AS GASTO_AUTORIZADO_VISA
	,COALESCE(D.SaldoGastoAutorizado_ELO,0) AS GASTO_AUTORIZADO_ELO
	,COALESCE(E.QtdContasCriadas,0) AS CONTAS_CRIADAS
	,COALESCE(F.QtdFaturasEmitidas,0) AS FATURA_EMITIDA
	,COALESCE(F.QtdFaturasCorreios,0) AS CORREIOS

	,COALESCE(G.VrVolumeFraude,0) AS RUBRICAS_FATURAS
	,COALESCE(H.QtPontosResgatados,0) AS MILHAGEM
	,COALESCE(I.Qtd_OutrasTarifa,0) AS SAQUE

	,COALESCE(J.VrLancado,0) AS CASHBACK
	,COALESCE(K.VrVolumeSeguros,0) AS RUBRICA_SEGUROS
	,COALESCE(L.VrTelemarketingDFIC,0) AS TELEMARKETING
	,COALESCE(L.VrEmbossingDFIC,0) AS PLASTICO

FROM CDCONTACARTAO AS A

LEFT JOIN (SELECT DISTINCT
			CdContaCartao
			,sum(coalesce(SaldoCarteiraAmigavel,0)) AS SaldoCarteiraAmigavel
			,sum(coalesce(SaldoCarteiraContenciosa,0)) AS SaldoCarteiraContenciosa

			FROM CARTEIRA_CART
			GROUP BY CdContaCartao) AS B ON (A.CdContaCartao = B.CdContaCartao)

LEFT JOIN (SELECT DISTINCT
			CdContaCartao
			,sum(coalesce(SaldoCart_Total,0)) AS SaldoCart_Total

			FROM SALDO_CARTEIRA_TOTAL
			GROUP BY CdContaCartao) AS B1 ON (A.CdContaCartao = B1.CdContaCartao)

LEFT JOIN BASE_CONTAS_QV AS C ON (A.CdContaCartao = C.CdContaCartao)

LEFT JOIN (SELECT DISTINCT
			CdContaCartao
			,sum(coalesce(QtdAutorizacoes,0)) AS QtdAutorizacoes
			,sum(coalesce(QtdContasAtivas_30dias,0)) AS QtdContasAtivas_30dias
			,sum(coalesce(SaldoGastoAutorizado,0)) AS SaldoGastoAutorizado
			,sum(coalesce(SaldoGastoAutorizado_MASTER,0)) AS SaldoGastoAutorizado_MASTER
			,sum(coalesce(SaldoGastoAutorizado_VISA,0)) AS SaldoGastoAutorizado_VISA
			,sum(coalesce(SaldoGastoAutorizado_ELO,0)) AS SaldoGastoAutorizado_ELO

			FROM GASTO_AUTORIZADO
			GROUP BY CdContaCartao) AS D ON (A.CdContaCartao = D.CdContaCartao)

LEFT JOIN CONTAS_CRIADAS AS E ON (A.CdContaCartao = E.CdContaCartao)

LEFT JOIN FATURA_EMITIDA AS F ON (A.CdContaCartao = F.CdContaCartao)

LEFT JOIN FRAUDES AS G ON (A.CdContaCartao = G.CdContaCartao)

LEFT JOIN PONTOS_FIDELIDADE AS H ON (A.CdContaCartao = H.CdContaCartao)

LEFT JOIN SAQUE AS I ON (A.CdContaCartao = I.CdContaCartao)

LEFT JOIN TBL_CASHBACK AS J ON (A.CdContaCartao = J.CdContaCartao)

LEFT JOIN SEGUROS AS K ON (A.CdContaCartao = K.CdContaCartao)

LEFT JOIN DFIC AS L ON (A.CdContaCartao = L.CdContaCartao)

;

PROC SQL;
CREATE TABLE CARTAO_RATEIO_01 AS
SELECT DISTINCT
	&SAFRA. as SAFRA
	,A.CdContaCartao
	,C.GRUPO_DESPESA
	,C.DRE
	,C.DRIVER
	,C.VALOR AS VL_MYABC
	,CASE WHEN C.DRIVER = 'TABELA DE LANÇAMENTO DE CSK' AND COALESCE(A.CASHBACK,0) <> 0
		   THEN COALESCE(A.CASHBACK,0) / COALESCE(B.CASHBACK_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'CONTA_APTA' AND COALESCE(A.CONTA_APTA,0) <> 0
		   THEN COALESCE(A.CONTA_APTA,0) / COALESCE(B.CONTA_APTA_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'GASTO_AUTORIZADO' AND C.DRE LIKE '%Fraude%' AND COALESCE(A.GASTO_AUTORIZADO,0) <> 0
		   THEN COALESCE(A.GASTO_AUTORIZADO,0) / COALESCE(B.GASTO_AUTORIZADO_TOTAL,0) *  COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'GASTO_AUTORIZADO' AND C.DRE = 'Fee Master' AND COALESCE(A.GASTO_AUTORIZADO_MASTER,0) <> 0
		   THEN COALESCE(A.GASTO_AUTORIZADO_MASTER,0) / COALESCE(B.GASTO_AUTORIZADO_MASTER_TOTAL,0) *  COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'GASTO_AUTORIZADO' AND C.DRE = 'Fee Visa' AND COALESCE(A.GASTO_AUTORIZADO_VISA,0) <> 0
		   THEN COALESCE(A.GASTO_AUTORIZADO_VISA,0) / COALESCE(B.GASTO_AUTORIZADO_VISA_TOTAL,0) *  COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'GASTO_AUTORIZADO' AND C.DRE = 'Fee Elo' AND COALESCE(A.GASTO_AUTORIZADO_ELO,0) <> 0
		   THEN COALESCE(A.GASTO_AUTORIZADO_ELO,0) / COALESCE(B.GASTO_AUTORIZADO_ELO_TOTAL,0) *  COALESCE(C.VALOR,0)

		   WHEN C.DRIVER IN ('CONTA_ATIVA', 'CONTA ATIVA') AND COALESCE(A.CONTA_ATIVA,0) <> 0
		   THEN COALESCE(A.CONTA_ATIVA,0) / COALESCE(B.CONTA_ATIVA_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'CONTAS_CRIADAS' AND COALESCE(A.CONTAS_CRIADAS,0) <> 0
		   THEN COALESCE(A.CONTAS_CRIADAS,0) / COALESCE(B.CONTAS_CRIADAS_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'FATURA_EMITIDA' AND COALESCE(A.FATURA_EMITIDA,0) <> 0
		   THEN COALESCE(A.FATURA_EMITIDA,0) / COALESCE(B.FATURA_EMITIDA_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'FATURAS VIA CORREIO' AND COALESCE(A.CORREIOS,0) <> 0
		   THEN COALESCE(A.CORREIOS,0) / COALESCE(B.CORREIOS_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'QTDE_AUTORIZACOES' AND COALESCE(A.QTDE_AUTORIZACOES,0) <> 0
		   THEN COALESCE(A.QTDE_AUTORIZACOES,0) / COALESCE(B.QTDE_AUTORIZACOES_TOTAL,0) * COALESCE(C.VALOR,0)

	 	   WHEN C.DRIVER = 'SALDO_CARTEIRA' AND COALESCE(A.CARTEIRA,0) <> 0
		   THEN COALESCE(A.CARTEIRA,0) / COALESCE(B.CARTEIRA_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'RUBRICA_SAQUE' AND COALESCE(A.SAQUE) <> 0
		   THEN COALESCE(A.SAQUE,0) / COALESCE(B.SAQUE_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'RUBRICAS FATURAS' AND COALESCE(A.RUBRICAS_FATURAS) <> 0
		   THEN COALESCE(A.RUBRICAS_FATURAS,0) / COALESCE(B.RUBRICAS_FATURAS_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'SALDO CARTEIRA 5 A 74' AND COALESCE(A.AMIGAVEL) <> 0
		   THEN COALESCE(A.AMIGAVEL,0) / COALESCE(B.AMIGAVEL_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'SALDO CARTEIRA 75 A 360' AND COALESCE(A.CONTENCIOSA) <> 0
		   THEN COALESCE(A.CONTENCIOSA,0) / COALESCE(B.CONTENCIOSA_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'TABELA DE RESGATE DE PONTOS' AND COALESCE(A.MILHAGEM) <> 0
		   THEN COALESCE(A.MILHAGEM,0) / COALESCE(B.MILHAGEM_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'RUBRICA_SEGUROS' AND COALESCE(A.RUBRICA_SEGUROS) <> 0
		   THEN COALESCE(A.RUBRICA_SEGUROS,0) / COALESCE(B.RUBRICA_SEGUROS_TOTAL,0) * COALESCE(C.VALOR,0)

		   WHEN C.DRIVER = 'DIFC' AND UPPER(C.DRE) = 'TELEMARKETING' AND COALESCE(A.TELEMARKETING) <> 0
		   THEN COALESCE(A.TELEMARKETING,0) + (COALESCE(A.TELEMARKETING,0) / COALESCE(B.TELEMARKETING_TOTAL,0)) * (COALESCE(C.VALOR,0) - COALESCE(B.TELEMARKETING_TOTAL,0))

		   WHEN C.DRIVER = 'DIFC' AND UPPER(C.DRE) LIKE '%PLÁSTICO%' AND COALESCE(A.PLASTICO) <> 0
		   THEN COALESCE(A.PLASTICO,0) + (COALESCE(A.PLASTICO,0) / COALESCE(B.PLASTICO_TOTAL,0)) * (COALESCE(C.VALOR,0) - COALESCE(B.PLASTICO_TOTAL,0))

		   ELSE 0 END AS VL_RATEADO

FROM BASE_RATEIO_CART AS A
LEFT JOIN CDCONTACARTAO_TOTAL AS B
ON 1 = 1
LEFT JOIN MYABC_DRIVERS_CART03 AS C
ON 1 = 1

proc sql;
create table clientes_conta as
select distinct
	 input(a.nucpfcnpj,20.) as NUCPFCNPJ
	,input(A.nucontacontrato,20.) as NUCONTACORRENTE
	,A.flcontacontratoativo as CONTA_ATIVA
	,case
		when A.flcontacontratoativo = 0 then 1
		else 0
	 end as CONTA_NAO_ATIVA
	,case
		when datepart(A.dtiniciocontacontrato) >= &DATA_REF. then 1
		else 0
	 end as CONTAS_NOVAS
	,A.flcontacontratorelacionado as CONTAS
	,coalesce(C.QTD_TRANSACOES,0) AS QTDE_SAQUE
	,coalesce(D.QTDE_CUSTO_BAAS,0) AS QTDE_CUSTO_BAAS
	,coalesce(E.VRTRANSACAO_DEBITO,0) AS GASTO_AUTORIZADO
	,case
		when upper(F.origemagrupada) like '%CORE%' or upper(F.origemagrupada) is null then 'Core'
		else 'Não Cliente'
	 end as ORIGEM
	,A.flcontacontratoapto as CONTA_APTA
from TABELA_CLIENTES as A






left join SAQUE_BD as C
	on A.nucpfcnpj = C.nucpfcnpj
	and A.nucontacontrato = C.nucontacorrente

left join CUSTO_BAAS_BD as D
	on A.nucpfcnpj = D.nucpfcnpj
	and A.nucontacontrato = D.nucontacorrente

left join TBL_TRANSACAO_DEBITO as E
	on A.nucpfcnpj = E.nucpfcnpj
	and A.nucontacontrato = E.nucontacorrente

left join BANCO.CONTAS_ABERTAS_ORIGEM as F
	ON input(A.nucpfcnpj,20.) = F.nucpfcnpj

where A.nmtipoproduto = 'Conta Digital'
	  and datepart(A.dtiniciocontacontrato) <= intnx('month',&DATA_REF.,0,'e')
	  and (datepart(A.dtfimcontacontrato) >= intnx('month',&DATA_REF.,0,'e') or datepart(A.dtfimcontacontrato) is null)
;
quit;

options compress=binary reuse=yes replace symbolgen;
proc sql;
create table CLIENTES_RATEIO_01 as
select
 A.NUCPFCNPJ
,A.NUCONTACORRENTE
,A.ORIGEM
,A.CONTA_ATIVA
,A.CONTAS_NOVAS
,A.CONTAS
,A.QTDE_SAQUE
,A.CONTA_APTA
,&contas_ativas_total as CONTAS_ATIVAS_TOTAL
,&contas_novas_total as CONTAS_NOVAS_TOTAL
,&qtde_contas_total as QTDE_CONTAS_TOTAL
,&qtde_saque_total as QTDE_SAQUE_TOTAL
,&gasto_aut_total as GASTO_AUTORIZADO_TOTAL
,&conta_apta_total as CONTA_APTA_TOTAL
,C.GRUPO_DESPESA
,C.DRE
,C.DRIVER
,C.VALOR as VL_MYABC
,case when C.DRIVER = 'QTDE_CONTAS'
		then coalesce(C.VALOR,0) / &qtde_contas_total

	  when C.DRIVER = 'QTDE_SAQUES' and coalesce(A.QTDE_SAQUE,0) <> 0
		then coalesce(A.QTDE_SAQUE,0) / &qtde_saque_total * coalesce(C.VALOR,0)

	  when C.DRIVER = 'GASTO_AUTORIZADO' and coalesce(A.GASTO_AUTORIZADO,0) <> 0
		then coalesce(A.GASTO_AUTORIZADO,0) / &gasto_aut_total *  coalesce(C.VALOR,0)

	  when C.DRIVER = 'CONTAS_ATIVAS' and coalesce(A.CONTA_ATIVA,0) <> 0
		then coalesce(C.VALOR,0) / &contas_ativas_total * A.CONTA_ATIVA

	  when C.DRIVER = 'CONTA_APTA' and coalesce(A.CONTA_APTA,0) <> 0
		then coalesce(C.VALOR,0) / &conta_apta_total * A.CONTA_APTA

	  when C.DRIVER = 'CONTAS_NAO_ATIVAS' and coalesce(A.CONTA_NAO_ATIVA,0) <> 0
		then coalesce(C.VALOR,0) / &contas_nao_atv_total

	  when C.DRIVER = 'NOVAS_CONTAS' and coalesce(A.CONTAS_NOVAS,0) <> 0
		then coalesce(C.VALOR,0) / &contas_novas_total

	  when C.DRIVER = 'NOVAS_CONTAS_ORIGEM' and coalesce(A.CONTAS_NOVAS,0) <> 0 and upper(A.ORIGEM) = 'CORE'
		then &CASH_CORE. * coalesce(C.VALOR,0) / &contas_novas_core

	  when C.DRIVER = 'NOVAS_CONTAS_ORIGEM' and coalesce(A.CONTAS_NOVAS,0) <> 0 and upper(A.ORIGEM) = 'NÃO CLIENTE'
		then &CASH_MAR. * coalesce(C.VALOR,0) / &contas_novas_nao_cliente

	  when C.DRIVER = 'NOVAS_CONTAS_MAR_ABERTO' and coalesce(A.CONTAS_NOVAS,0) <> 0 and upper(A.ORIGEM) = 'NÃO CLIENTE'
		then coalesce(C.VALOR,0) / &contas_novas_nao_cliente

	  when C.DRIVER = 'TRANSACOES_BAAS' and coalesce(A.QTDE_CUSTO_BAAS) <> 0
		then coalesce(A.QTDE_CUSTO_BAAS,0) / &qtde_custo_baas_total * coalesce(C.VALOR,0)

 	  else 0
 end as VL_RATEADO
from CLIENTES_CONTA as A

left join MYABC_DRIVERS_BD03 as C on 1 = 1
;
quit;

create table BASE_RATEIO_VAR_COM_BVO as
select

,coalesce(coalesce(B.CDMODALIDADE, C.CDMODALIDADE), H.CDMODALIDADE) as CDMODALIDADE
,coalesce(F.Produto, H.Produto) as Produto
,coalesce(I.Diretoria,coalesce(B.Canal, C.Canal)) as Canal
,case when coalesce(B.QtDiasAtraso,0) > 90 then coalesce(VrSaldo,0) else 0 end as CARTEIRA_ACIMA_90
,coalesce(B.VrSaldo,0) as VrSaldo
,coalesce(C.FLAG_JUST, 'OUTROS') as FLAG_JUST
,coalesce(C.VrMargemContribuicao, 0) as VrMargemContribuicao
,case when C.NuContratoGestao IS NOT NULL then 'MES ATUAL' else 'OUTROS MESES' end as FLAG_MES
,case when (CALCULATED FLAG_MES) = 'MES ATUAL' then 1 else 0 end as NOVOS_CONTRATOS
,case when UPPER(C.ESTADOBEM) = 'USADO' then 'USADO' else 'OUTROS' end as ESTADO_BEM
,case when (CALCULATED ESTADO_BEM) = 'USADO' then 1 else 0 end as QTD_VEICULOS_USADOS
,case when coalesce(B.NUCONTRATO,C.NuContratoGestao) IS NOT NULL then 1 else 0 end as QTDE
,COALESCE(G.CONTRATOS_DIFERIMENTO,0) AS CONTRATOS_DIFERIMENTO
,COALESCE(H.CONT_JURIDICO,0) AS CONT_JURIDICO
from CONTRATO_COM_BVO as A

left join CARTEIRA_COM_BVO as B on (A.NUCONTRATO = B.NUCONTRATO)

left join SUMARIO_COM_BVO as C on (A.NUCONTRATO = C.NuContratoGestao)

left join CIVEIS_VAR_03_COM_BVO as H on (A.NUCONTRATO = H.NuContrato)

left join DE_PARA_PRODUTO as F on (coalesce(coalesce(B.CDMODALIDADE, C.CDMODALIDADE), H.CDMODALIDADE) = F.CD_MODALIDADE_PRODUTO)

left join ( select distinct
					NuContrato
					,Produto
					,SUM(VrCusto) AS CONTRATOS_DIFERIMENTO
				from CONTRATOS_DIFERIMENTO
				where NuContrato in (select distinct NuContrato from CARTEIRA_COM_BVO)
				group by
				NuContrato
				,Produto) as G on (A.NUCONTRATO = G.NuContrato and F.Produto = G.Produto)

left join FINAL.TAB_REGISTRO_CONTRATO as I on A.NUCONTRATO = I.NUCONTRATOGESTAO

				Produto

				Produto,Gestao) as F on coalesce(C.Produto, A.Produto) = F.Produto
