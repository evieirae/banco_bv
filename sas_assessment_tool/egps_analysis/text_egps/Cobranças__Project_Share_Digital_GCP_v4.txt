CREATE TABLE WORK.TBIDADE AS
SELECT
    A.NUCONTRATO,
    A.NUCPFCNPJ,
    A.CDPESSOA,
    CASE
        WHEN A.DTNASCIMENTO IS NULL THEN .
        ELSE (YEAR(TODAY()) - YEAR(DATEPART(A.DTNASCIMENTO)))
    END AS IDADE,
    CASE
        WHEN A.DTNASCIMENTO IS NULL THEN 'Y.SEM_DTNASCIMENTO'
        WHEN (YEAR(TODAY()) - YEAR(DATEPART(A.DTNASCIMENTO))) < 18 THEN 'Z.VERIFICAR_IDADE'
        ELSE L.FX_IDADE_T
    END AS FX_IDADE
FROM CONNECTION TO DBCON (
    SELECT DISTINCT
        A.NUCONTRATO,
        A.NUCPFCNPJ,
        A.CDPESSOA,
        B.DTNASCIMENTO
    FROM DBA.VWTBCONTRATO A
    LEFT JOIN DBA.TbPessoaFisica B ON A.CDPESSOA = B.CDPESSOA

    UNION ALL

    SELECT DISTINCT
        A.NUCONTRATO,
        A.NUCPFCNPJ,
        A.CDPESSOA,
        B.DTNASCIMENTO
    FROM STGTXT_FLEX.TBCONTRATO A
    LEFT JOIN STGTXT_FLEX.TbPessoaFisica B ON A.CDPESSOA = B.CDPESSOA
) AS A
LEFT JOIN AUXSHARE.BASE_AUX_FX_IDADE L
    ON (YEAR(TODAY()) - YEAR(DATEPART(A.DTNASCIMENTO))) = L.IDADE;

QUIT;

CREATE TABLE WORK.TBL_BOLETOS AS

SELECT DISTINCT
      A.*
    , B.DU AS DU_GERACAO
    , C.DU AS DU_VCTO
    , D.DU AS DU_PGTO
    , TCON.CANAL_MACRO AS CANAL_MACRO
    , CASE
        WHEN e.fl_cliente_heavy_user IS MISSING THEN 0
        WHEN e.fl_cliente_heavy_user = 1 THEN 1
        WHEN e.fl_cliente_heavy_user = 0 THEN 0
      END AS HU
    , CASE
        WHEN F.Flag_Prop_Dig IS MISSING THEN 0
        WHEN F.Flag_Prop_Dig = 1 THEN 1
        WHEN F.Flag_Prop_Dig = 0 THEN 0
      END AS PROP_DIGITAL
    , G.FX_IDADE
    , H.PRODUTO_MACRO
FROM (
    SELECT DISTINCT
          *
        , INPUT(CATS(PUT(YEAR(DATEPART(DTGERACAO)),Z4.),PUT(MONTH(DATEPART(DTGERACAO)),Z2.)),Z6.) AS ANO_MES
        , DATEPART(DTGERACAO)      AS DT_GERACAO      FORMAT= DDMMYY10.
        , DATEPART(DTVENCIMENTO)   AS DT_VENCIMENTO   FORMAT= DDMMYY10.
        , DATEPART(DTPAGAMENTO)    AS DT_PAGAMENTO    FORMAT= DDMMYY10.
        , DATEPART(DT_VCTO_MINIMO) AS DT_VCTO_MIN     FORMAT= DDMMYY10.
        
        , INPUT(SUBSTRN(PUT(NUCONTRATOGESTAO, Z14.), 1, 2), Z2.) AS COD_PRODUTO
    FROM CONNECTION TO SYBASE (
                                SELECT DISTINCT
                                      CASE
                                        WHEN CON.NUCONTRATOGESTAO IS NOT NULL THEN 1
                                        ELSE 0
                                      END AS FLAG_BOLETO_GERADO
                                    , CASE
                                        WHEN CON.NUCONTRATOGESTAO IS NOT NULL AND CON.DTPAGAMENTO IS NOT NULL THEN 1
                                        ELSE 0
                                      END AS FLAG_BOLETO_PAGO
                                    , CASE
                                        WHEN CON.TPENTRADA = 'A' THEN 'ASSESSORIA'
                                        WHEN CON.TPENTRADA = 'B' THEN 'CHATBOT'
                                        WHEN CON.TPENTRADA = 'C' THEN 'SAC'
                                        WHEN CON.TPENTRADA = 'D' THEN 'MARKETPLACE'
                                        WHEN CON.TPENTRADA = 'E' THEN 'MARKETPLACE'
                                        WHEN CON.TPENTRADA = 'F' THEN 'MARKETPLACE'
                                        WHEN CON.TPENTRADA = 'G' THEN 'MARKETPLACE'
                                        WHEN CON.TPENTRADA = 'H' THEN 'APP'
                                        WHEN CON.TPENTRADA = 'I' THEN 'WhatsApp Cob'
                                        WHEN CON.TPENTRADA = 'J' THEN 'MARKETPLACE'
                                        WHEN CON.TPENTRADA = 'L' THEN 'WHATSAPP SF SAC'
                                        WHEN CON.TPENTRADA = 'M' THEN 'MANUAL ASSESSORIA'
                                        WHEN CON.TPENTRADA = 'Q' THEN 'Whatsapp Bot SF'
                                        WHEN CON.TPENTRADA = 'R' THEN 'ROBO'
                                        WHEN CON.TPENTRADA = 'S' THEN 'SITE'
                                        WHEN CON.TPENTRADA = 'T' THEN 'WHATSAPP SF COB'
                                        WHEN CON.TPENTRADA = 'U' THEN 'URA'
                                        WHEN CON.TPENTRADA = 'P' THEN 'PASCHOALOTTO'
										WHEN CON.TPENTRADA = 'V' THEN 'JABUTI'
										WHEN CON.TPENTRADA = 'O' THEN 'GOOGLE'
                                        ELSE '99.VERIFICAR'
                                      END AS CANAL_BOLETO
                                    , CON.NUCONTRATOGESTAO
                                    , CON.SQCONTRATOFINANCEIRO
                                    , CON.NUPROCESSO
                                    , CON.DTGERACAO
                                    , CON.DTVENCIMENTO
                                    , CON.DTPAGAMENTO
                                    , CON.TPENTRADA
                                    , CON.TPACORDONEGOCIACAO
                                    , CON.CDOCOCONTA
                                    , CON.CDIDEMPBANCO
                                    , CON.CDCARTEIRABANCO
                                    , CON.CDNOSSONUMERO
                                    , CON.VRBOLETO
                                INTO #TMP_NEG_CONTR
                                FROM DBCONTENCIOSO..TBNEGOCIACAOCONTRATO AS CON
                                WHERE 1=1
                                AND CON.DTGERACAO BETWEEN &DT_INI. AND &DT_FIM.
                                AND CON.TPENTRADA IN ('A','B','C','D','E','F','G','H','I','J','L','M','Q','R','S','T','U','P', 'V', 'O')

                                SELECT DISTINCT
                                      A.NUCONTRATOGESTAO
                                    , A.CDOCOCONTA
                                    , A.CDIDEMPBANCO
                                    , A.CDCARTEIRABANCO
                                    , A.CDNOSSONUMERO
                                    , MIN(A.CDPARCELA) AS NUPARCELA_MINIMA
                                    , MIN(A.DtVencimento) AS DT_VCTO_MINIMO
                                INTO #TMP_NEG_CONTR_2
                                FROM DBCONTENCIOSO..TBNEGOCIACAOPARCELA AS A
                                INNER JOIN #TMP_NEG_CONTR AS B ON A.NUCONTRATOGESTAO      = B.NUCONTRATOGESTAO
                                                             AND A.CDOCOCONTA        = B.CDOCOCONTA
                                                             AND A.CDIDEMPBANCO      = B.CDIDEMPBANCO
                                                             AND A.CDCARTEIRABANCO   = B.CDCARTEIRABANCO
                                                             AND A.CDNOSSONUMERO     = B.CDNOSSONUMERO
                                GROUP BY
                                      A.NUCONTRATOGESTAO
                                    , A.CDOCOCONTA
                                    , A.CDIDEMPBANCO
                                    , A.CDCARTEIRABANCO
                                    , A.CDNOSSONUMERO

                                SELECT DISTINCT
                                      A.*
                                    , B.NUPARCELA_MINIMA
                                    , B.DT_VCTO_MINIMO
                                    , C.PCHONORARIO
                                INTO #TMP_NEG_CONTR_3
                                FROM #TMP_NEG_CONTR AS A
                                LEFT JOIN #TMP_NEG_CONTR_2 AS B                     ON  A.NUCONTRATOGESTAO = B.NUCONTRATOGESTAO
                                                                                   AND A.CDOCOCONTA       = B.CDOCOCONTA
                                                                                   AND A.CDIDEMPBANCO     = B.CDIDEMPBANCO
                                                                                   AND A.CDCARTEIRABANCO  = B.CDCARTEIRABANCO
                                                                                   AND A.CDNOSSONUMERO    = B.CDNOSSONUMERO
                                LEFT JOIN DBCONTENCIOSO..TBNEGOCIACAOPARCELA AS C   ON  A.NUCONTRATOGESTAO = C.NUCONTRATOGESTAO
                                                                                   AND B.DT_VCTO_MINIMO   = C.DtVencimento
                                                                                   AND A.CDOCOCONTA       = C.CDOCOCONTA
                                                                                   AND A.CDIDEMPBANCO     = C.CDIDEMPBANCO
                                                                                   AND A.CDCARTEIRABANCO  = C.CDCARTEIRABANCO
                                                                                   AND A.CDNOSSONUMERO    = C.CDNOSSONUMERO

                                SELECT DISTINCT
                                    *
                                    , DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) AS ATRASO
                                    , CASE
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 1 AND 15 THEN 'ATIVA'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 16 AND 70 THEN 'ADM'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 71 AND 360 THEN 'CONTENCIOSO'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) >360 THEN 'WO'
                                        ELSE 'VERIFICAR'
                                      END FX_ATRASO
                                    , CASE
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 5 AND 15 THEN '5 a 15 dias'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 16 AND 30 THEN '16 a 30 dias'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 31 AND 70 THEN '31 a 70 dias'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 71 AND 90 THEN '71 a 90 dias'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 91 AND 120 THEN '91 a 120 dias'
                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) BETWEEN 121 AND 150 THEN '121 a 150 dias'

                                        WHEN DATEDIFF(DD,DT_VCTO_MINIMO,CAST(DTGERACAO AS DATE)) >360 THEN 'acima de 360 dias'
                                        ELSE 'Verificar'
                                      END FX_ATRASO_2
                                FROM #TMP_NEG_CONTR_3
                              )
) AS A
