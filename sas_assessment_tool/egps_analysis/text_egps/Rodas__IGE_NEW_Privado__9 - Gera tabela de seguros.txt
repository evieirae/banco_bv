PROC SQL;
%CONECTADB(server=PTA);

  CREATE TABLE tempw.PRIVADO_TBSEGUROS AS
	SELECT *
	FROM CONNECTION TO DBCON
				(

SELECT a.NUPROPOSTA,
		CASE
			WHEN b.SeguroSPF IS NULL THEN 0
			ELSE b.SeguroSPF
		END
	AS VrSeguro_SPF,
		CASE
			WHEN c.SeguroAP IS NULL THEN 0
			ELSE c.SeguroAP
		END
	AS VrSeguro_AP,
		CASE
			WHEN b.SeguroSPF IS NULL THEN 0
			ELSE b.SeguroSPF END+
		CASE
			WHEN c.SeguroAP IS NULL THEN 0
			ELSE c.SeguroAP
		END
	AS VrSeguros,
		CASE
			WHEN b.SeguroSPF IS NOT NULL THEN 1
			ELSE 0
		END
	AS TtlSeguro_SPF,
		CASE
			WHEN c.SeguroAP IS NOT NULL THEN 1
			ELSE 0
		END
	AS TtlrSeguro_AP,
		CASE
			WHEN b.SeguroSPF IS NOT NULL THEN 1
			ELSE 0 END+
		CASE
			WHEN c.SeguroAP IS NOT NULL THEN 1
			ELSE 0
		END
	AS QtSeguros

		from

			(SELECT DISTINCT Proposta.NUPROPOSTA FROM dba.TbProposta Proposta

			WHERE Proposta.CdSubProduto = 12) a

			LEFT join

			(SELECT tbseg.NUPROPOSTA,
				CASE
					WHEN tbseg.CDTIPOSEGURO = 2 THEN tbseg.vrseguro
					ELSE 0
				END
			AS SeguroSPF,
				tbseg.vrseguro AS TotalSeguro FROM dbcred.TbPropostaSeguro tbseg
			LEFT join
				dba.TbProposta Proposta ON tbseg.nuproposta = Proposta.NuProposta
			WHERE tbseg.CDTIPOSEGURO = 2 AND
				Proposta.CdSubProduto = 12) b
				ON a.NUPROPOSTA = b.NUPROPOSTA

			LEFT JOIN
				(SELECT tbseg.NUPROPOSTA,
					CASE
						WHEN tbseg.CDTIPOSEGURO = 5 THEN tbseg.vrseguro
						ELSE 0
					END
				AS SeguroAP,
					tbseg.vrseguro AS TotalSeguro
					FROM dbcred.TbPropostaSeguro tbseg
				LEFT join
					dba.TbProposta Proposta
					ON tbseg.nuproposta = Proposta.NuProposta
				WHERE tbseg.CDTIPOSEGURO = 5
					AND Proposta.CdSubProduto = 12) c
					ON a.NUPROPOSTA = c.NUPROPOSTA
		);
RUN;
