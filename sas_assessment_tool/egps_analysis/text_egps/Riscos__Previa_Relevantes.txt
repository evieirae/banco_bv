CREATE TABLE Cust_II AS
SELECT
t1.*,

FROM Cust t1
LEFT JOIN dbep.VWTBLCAUSACONS   t2 ON (t1.Cd_Causa = t2.CdCausa and t2.cdSistema ='p')
;RUN;

PROC SQL;
CREATE TABLE Cust_III AS
SELECT
t1.Cd_Causa,
