CREATE TABLE DBA AS
SELECT
	*
FROM CONNECTION TO PERM(
	WITH LISTA_CONTRATOS AS (

		SELECT DISTINCT
			NuContratoGestao
		FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
		WHERE
			dt_baixa >= &dt_inicio. AND dt_baixa < &dt_fim. AND
			(dt_cancelamento is null or dt_reativacao is not null) AND
			ST_RENEG = 3
)

	SELECT DISTINCT
		A.NuContratoGestao,
		B.sq_contrato_financeiro,
		B.vr_parcela,
		B.st_parcela,
		B.dt_ultimo_recebimento,
		B.dt_vencimento,
		C.tx_juros as Tx_Contrato,
		CASE
			WHEN D.fl_restruturacao_contrato = FALSE THEN "N"
			WHEN D.fl_restruturacao_contrato = TRUE THEN "S"
			ELSE "N"
		END AS Fl_Reest_Anterior,

	FROM LISTA_CONTRATOS AS A

	LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
	ON A.NuContratoGestao = B.cd_contrato

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS C
	ON A.NuContratoGestao = C.cd_contrato AND C.sq_contrato_financeiro = 1

	LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro AS D
	ON B.cd_contrato = D.cd_contrato and B.sq_contrato_financeiro = D.sq_contrato_financeiro
)
;QUIT;

PROC SQL;
CREATE TABLE RENEG_SIMULADOR_&anomes. AS
SELECT DISTINCT
    A.NuContratoGestao AS NuContrato,
    A.SqContratoFinanceiro,
    A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo,
    A.dt_baixa,
    A.dt_reneg,
    A.vr_presente,
    A.vr_financiado,
    A.qt_parcela,
    A.vr_prestacao,
    A.vr_desconto_mora,
    A.vr_desconto_multa,
    A.vr_desconto_principal,
    A.Tx_Pactuada,
    B.Tx_Contrato,
    B.Fl_Reest_Anterior,
    A.FlRestruturacaoContrato,

    MIN(B.Vr_Parcela) AS VrParcela_Anterior,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DtMin,

    COUNT(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_Abertas,

    COUNT(CASE
        WHEN (B.st_parcela <> 'Q' OR
              (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)) AND
             B.dt_vencimento < A.dt_reneg
        THEN B.Vr_Parcela
        ELSE .
    END) AS Qtd_Parcelas_em_Atraso,

    MIN(CASE
        WHEN B.st_parcela <> 'Q' OR
             (B.st_parcela = 'Q' AND B.dt_ultimo_recebimento >= A.dt_reneg)
        THEN B.dt_vencimento
        ELSE .
    END) AS DT_VENCIMENTO_NOVO

FROM RENEG_REALIZADA_&anomes. A

LEFT JOIN DBA B
  ON A.NuContratoGestao = B.NuContratoGestao
 AND A.SqContratoFinanceiro = B.sq_contrato_financeiro

GROUP BY
    A.NuContratoGestao, A.SqContratoFinanceiro, A.NuContratoGestaoNovo,
    A.SqContratoFinanceiroNovo, A.dt_baixa, A.dt_reneg, A.vr_presente,
    A.vr_financiado, A.qt_parcela, A.vr_prestacao, A.vr_desconto_mora,
    A.vr_desconto_multa, A.vr_desconto_principal, A.Tx_Pactuada,
    A.FlRestruturacaoContrato,
    B.Tx_Contrato, B.Fl_Reest_Anterior
;
QUIT;
