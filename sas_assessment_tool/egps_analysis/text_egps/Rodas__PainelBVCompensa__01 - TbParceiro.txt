/* Criação de tabela auxiliar com dados dos Parceiros */
PROC SQL;
   CREATE TABLE TbParceiro AS
   SELECT t5.CdParceiroComercial,
   		  t5.NmParceiroComercial,
		  t5.NuCnpjLoja,
		  t5.NmColaboradorC,
          t5.NmFilialC,
		  t5.NmSubEstruturaComercialC,
          t5.NmAreaEstruturaComercialC,
          t5.NmCanalEstruturaComercialC,
          t5.DsLoginN1C,
          t5.DsLoginN2C,
          t5.DsLoginN3C,
          t5.LstSubstituicaoC
      FROM BASES.TBPARCEIROCOMERCIAL t5
      WHERE t5.FlAtivo = 'S'
      ORDER BY t5.CdParceiroComercial ASC;
QUIT;
