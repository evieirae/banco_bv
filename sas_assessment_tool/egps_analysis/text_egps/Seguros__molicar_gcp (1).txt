proc sql;
create table nacionalidade_stg as
select distinct
input(AaModelo,best.) as ano,
input(cdmolicar,best.) as molicar,
Nacionalidade_Veiculo,
VrMercado,
count(1) as qtd
from dataseg.tbseguroscompleta
where Nacionalidade_Veiculo is not null
group by
input(AaModelo,best.),
input(cdmolicar,best.),
Nacionalidade_Veiculo,
VrMercado
;quit;

proc sql;
create table nacionalidade as
select distinct
q2.ano,
q2.molicar,
q2.Nacionalidade_Veiculo,
max_vr
from nacionalidade_stg q2
inner join
	(select distinct
	ano,
	molicar,
	max(qtd) as max_qtd,
	max(VrMercado) as max_vr
	from nacionalidade_stg
	group by ano,
	molicar
	) q1
on q1.ano=q2.ano
and q1.molicar=q2.molicar
and q1.max_qtd=q2.qtd
;quit;

proc sql;
create table molicar2 as
select distinct q1.*,
q2.DsCategoria,
q2.CdMarca,
q3.Nacionalidade_Veiculo,
q3.max_vr,
nuchassis,
Q2.CdModelo,
sum(q2.flgarantiamecanica) as flgm,
sum(q2.flcasco) as flcasco,
sum(q2.flrcf) as flRcf
from
work.molicar q1
left join dataseg.tbseguroscompleta q2
	on q1.aa_modelo_veiculo=input(q2.AaModelo,best.)
	and q1.cd_molicar=input(q2.cdmolicar,best.)
left join nacionalidade q3
		on q1.aa_modelo_veiculo=q3.ano
	and q1.cd_molicar=q3.molicar
group by
aa_modelo_veiculo,
cd_molicar,
GM,
Casco,
RCF,
qtd_simulacoes,
q2.DsCategoria,
q2.CdMarca,
q3.Nacionalidade_Veiculo,
q3.max_vr,
nuchassis,
Q2.CdModelo
;quit;

proc sql;
create table molicar2 as
select * from molicar2
where nuchassis is not null
;quit;

proc sort data = molicar2;
  by cd_molicar aa_modelo_veiculo;
run;

data molicar2;
  set molicar2;
  count + 1;
  by  cd_molicar aa_modelo_veiculo;
  if first.cd_molicar  then count = 1;
run;



proc sql;
create table GM as
select distinct
'Garantia Mecanica' as produto,
qtd_simulacoes,cd_molicar,aa_modelo_veiculo,
DsCategoria,CdMarca,Nacionalidade_Veiculo,CdModelo,NuChassis,max_vr AS ValorMolicar
from
work.molicar2
where GM=0 AND flGM=0 AND aa_modelo_veiculo>2015
and DsCategoria in ('AUTOMOVEL','UTILITARIO')
and Nacionalidade_Veiculo='Nacional'
and max_vr<150000
and count<10
order by qtd_simulacoes desc,cd_molicar,aa_modelo_veiculo
;quit;



proc sql;
create table CASCO as
select distinct
'Casco' as produto,
qtd_simulacoes,cd_molicar,aa_modelo_veiculo,
DsCategoria,CdMarca,Nacionalidade_Veiculo,CdModelo,NuChassis,max_vr AS ValorMolicar
from
work.molicar2
where CASCO=0 AND flCasco=0 AND aa_modelo_veiculo>2000
and  Nacionalidade_Veiculo='Nacional'
and DsCategoria in ('AUTOMOVEL','UTILITARIO')
and count<10
and max_vr<300000
order by qtd_simulacoes desc,cd_molicar,aa_modelo_veiculo
;quit;






proc sql;
create table RCF as
select distinct
'RCF' as produto,
qtd_simulacoes,cd_molicar,aa_modelo_veiculo,
DsCategoria,CdMarca,Nacionalidade_Veiculo,CdModelo,NuChassis,max_vr AS ValorMolicar

from (
	select distinct * from
	work.molicar2
	where RCF=0 AND flRCF=0 AND  aa_modelo_veiculo>2000
	and  Nacionalidade_Veiculo='Nacional'
	and count<10 and DsCategoria in ('AUTOMOVEL','UTILITARIO')
	and max_vr<300000
	union
	select distinct * from
	work.molicar2
	where RCF=0 AND flRCF=0 AND aa_modelo_veiculo>2000
	and  Nacionalidade_Veiculo='Nacional'
	and count<10 and DsCategoria ='MOTO'
	and max_vr<150000
	union
	select distinct * from
	work.molicar2
	where RCF=0 AND flRCF=0 AND aa_modelo_veiculo>2000
	and  Nacionalidade_Veiculo='Nacional'
	and count<10 and DsCategoria ='CAMINHAO'
	and max_vr<500000)q1
order by qtd_simulacoes desc,cd_molicar,aa_modelo_veiculo
;quit;
