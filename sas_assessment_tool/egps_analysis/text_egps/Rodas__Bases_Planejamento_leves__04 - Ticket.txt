CREATE TABLE BASE_TICKET_0 AS
SELECT

ChaveCdParceiro_FxIdade,
CdParceiroComercial,
FxIdade,
NmAreaEstruturaComercialC,

SUM(QtdContratos_Proj) AS QtdContratos_Proj FORMAT COMMAX20.6,
SUM(VrProducao_Proj) AS VrProducao_Proj FORMAT COMMAX20.2,
COALESCE(CALCULATED VrProducao_Proj / CALCULATED QtdContratos_Proj,0) AS Ticket_BV_Proj FORMAT COMMAX20.2,

SUM(QtdContratos_ProjSab) AS QtdContratos_ProjSab FORMAT COMMAX20.6,
SUM(VrProducao_ProjSab) AS VrProducao_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED VrProducao_ProjSab / CALCULATED QtdContratos_ProjSab,0) AS Ticket_BV_ProjSab FORMAT COMMAX20.2,

SUM(qtgravamesplayers_Proj) AS qtgravamesplayers_Proj FORMAT COMMAX20.6,
SUM(Vrgravamesplayers_Proj) AS Vrgravamesplayers_Proj FORMAT COMMAX20.2,
COALESCE(CALCULATED Vrgravamesplayers_Proj / CALCULATED qtgravamesplayers_Proj,0) AS Ticket_Mercado_Proj FORMAT COMMAX20.2,

SUM(qtgravamesplayers_ProjSab) AS qtgravamesplayers_ProjSab FORMAT COMMAX20.6,
SUM(Vrgravamesplayers_ProjSab) AS Vrgravamesplayers_ProjSab FORMAT COMMAX20.2,
COALESCE(CALCULATED Vrgravamesplayers_ProjSab / CALCULATED qtgravamesplayers_ProjSab,0) AS Ticket_Mercado_ProjSab FORMAT COMMAX20.2,

case when CALCULATED Ticket_BV_ProjSab <= 0  then CALCULATED Ticket_Mercado_ProjSab else
	CALCULATED Ticket_BV_ProjSab end as TicketFinal

FROM WORK.POTENCIAL_RODAS_03 A
