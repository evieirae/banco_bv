PROC SQL;
CONNECT TO SYBASE AS DBCON
	
(	SERVER   = "PTASYBCTR"
	DATABASE = "DBCONTENCIOSO"
	USER     = &User_PTASYBCTR.
	PASSWORD = &pw_PTASYBCTR.   );
CREATE TABLE WORK.BASE_ACORDO_QUEBRAS AS
SELECT DISTINCT A.*
FROM CONNECTION TO DBCON(
SELECT DISTINCT
  Year(A.DtGeracao)*100+Month(A.DtGeracao) AS SAFRA
, A.DtVencimento
, A.VrBoleto
, A.DtCancelamento
, A.StNegociacao
, A.DtGeracao
, A.VrPagamento
, A.DtPagamento
, A.TpEntrada
, A.NuContratoGestao
, A.SqContratoFinanceiro
, CASE
				When A.StNegociacao = 'AN' Then 'A Negociar'
				When A.StNegociacao = 'N'  Then 'Negociado'
				When A.StNegociacao = 'PA' Then 'Pendente De Aprovacao'
				When A.StNegociacao = 'A'  Then 'Aprovado Pela Bv'
				When A.StNegociacao = 'R'  Then 'Recusado Pela Bv'
				When A.StNegociacao = 'NA' Then 'Negociacao Em Andamento'
				When A.StNegociacao = 'C'  Then 'Negociacao Cancelada'
				When A.StNegociacao = 'LP' Then 'Liberado Pagamento'
				When A.StNegociacao = 'PG' Then 'Pago'
				When A.StNegociacao = 'PL' Then 'Pagamento Realizado'
				When A.StNegociacao = 'DA' Then 'Devolvido Assessoria'
				When A.StNegociacao = 'CD' Then 'Cheque Devolvido'
				When A.StNegociacao = 'AC' Then 'Cheque Aguardando Compensacao'
             ELSE '99.Verificar' END Status_Boleto
, CASE WHEN A.DtGeracao IS NOT NULL THEN 1 ELSE 0 END AS Flag_Boleto_Gerado
, CASE WHEN A.StNegociacao IN ('PL','LP') THEN 1 ELSE 0 END AS Flag_Boleto_Pago
FROM dbcontencioso..TbNegociacaoContrato A
WHERE A.DtGeracao  >= DATEADD(DD, -10, GETDATE())
AND A.DtVencimento <= DATEADD(DD,  10, GETDATE())
AND A.DtVencimento >= DATEADD(DD,  -4, GETDATE())

AND A.StNegociacao NOT IN ('C','R')
AND A.DtPagamento IS NULL
ORDER BY A.DtGeracao ASC, A.NuContratoGestao) AS A
;QUIT;


PROC SQL;
CREATE TABLE WORK.BASE_QUEBRAS_TRAT AS
SELECT DISTINCT A.*
, D.Canal As Canal_Boleto
, D.Canal_Macro
, intck('DAY', DATEPART(A.DtGeracao), DATEPART(A.DtVencimento), 'C')  AS Intervalo_GerVenc_Bol
, DATEPART(A.DtGeracao) Format yymmdd10. AS Dt_Geracao_Boleto
, DATEPART(A.DtVencimento) Format yymmdd10. AS Dt_Vencimento_Boleto
, HOUR(A.DtGeracao) AS Hr_Geracao
, B.NuCPFCNPJ
, B.ATRASO AS ATRASO_ATUAL
, B.PRI_NOME
, intck('DAY', DATEPART(A.DtGeracao), TODAY(), 'C')  AS Dias_Boleto_Gerado
, CASE WHEN A.NuContratoGestao = B.NuContratoGestao THEN 1 ELSE 0 END AS Flag_Base_Cob
, Case
	WHEN D.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) <=  7 THEN 1
	WHEN D.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) >= 20 THEN 1
	ELSE 0 END AS Flag_ForaExped
FROM WORK.BASE_ACORDO_QUEBRAS AS A
INNER JOIN CONTATOS.AUX_BASE_CYBER AS B ON A.NuContratoGestao = B.NuContratoGestao
INNER JOIN (
SELECT DISTINCT
  MAX(DtGeracao) AS DtGeracao
, NuContratoGestao
, SqContratoFinanceiro
FROM WORK.BASE_ACORDO_QUEBRAS
GROUP BY NuContratoGestao, SqContratoFinanceiro) AS C
		On  A.NuContratoGestao = C.NuContratoGestao
		AND A.SqContratoFinanceiro = C.SqContratoFinanceiro
		AND A.DtGeracao = C.DtGeracao
LEFT JOIN TCONFIG.CONFIG_TPENTRADA AS D ON A.TpEntrada = D.TpEntrada

Where A.StNegociacao NOT IN ('C','LP','PL','R','PA')
And Datepart(A.DtVencimento) < Today()
Having Flag_ForaExped = 0
;QUIT;
