%CRIAVARDB;
PROC SQL;
%CONECTADB(server=CRE);
CREATE TABLE TbBaseLeilao AS
SELECT * FROM connection to dbcon
		(
			SELECT distinct
				  TbRetornoBaseLeilao.DsChassi
				 ,TbRetornoBaseLeilao.DtRetorno
				 ,TbRetornoBaseLeilao.DtLeilao as DtLeilao
				 ,TbRetornoBaseLeilao.TpSucateamento as TpSucateamento
			FROM DbConsultaInformacaoBem..TbRetornoBaseLeilao TbRetornoBaseLeilao
/*			where DsChassi = '9BWAA05ZXA4120896'*/
						order by TbRetornoBaseLeilao.DtLeilao desc , TbRetornoBaseLeilao.TpSucateamento
	)
	;RUN;

	PROC SORT DATA=TbBaseLeilao NODUPKEY;
BY DsChassi;
RUN;

PROC SQL;
CREATE TABLE ScoreProposta01 AS
  SELECT DISTINCT
    A.NuProposta,
    A.FxIdade,
    A.VrFinanciamento,
	CASE WHEN B.DsChassi IS NULL THEN 'N' ELSE PUT(B.TpSucateamento,1.) END AS FlLeilao
  FROM TMP.TBEXTPROPOSTA A
	  LEFT JOIN TbBaseLeilao B
    ON A.DtEntradaProposta BETWEEN DATEPART(B.DtRetorno) AND INTNX('DAY', DATEPART(B.DtRetorno), 30)
   AND A.NuChassis = B.DsChassi
  WHERE A.NuAnoMes >= &SafraXM.;
RUN;

PROC SORT DATA=ScoreProposta01 OUT=ScoreProposta NODUPKEY;
BY NuProposta;
RUN;

proc sql;create index NuProposta on ScoreProposta;quit;

PROC SQL;
CREATE TABLE Originacao AS
  SELECT
    A.*,
	CASE WHEN A.VrPrestacao = . THEN ''
	     WHEN A.VrPrestacao BETWEEN 0 AND 100 THEN 'Até R$ 100'
		 WHEN A.VrPrestacao BETWEEN 100 AND 250 THEN 'Até R$ 250'
		 WHEN A.VrPrestacao BETWEEN 250 AND 500 THEN 'Até R$ 500'
		 WHEN A.VrPrestacao BETWEEN 500 AND 750 THEN 'Até R$ 750'
		 WHEN A.VrPrestacao BETWEEN 750 AND 1000 THEN 'Até R$ 1000'
		 WHEN A.VrPrestacao BETWEEN 1000 AND 1500 THEN 'Até R$ 1500'
		 WHEN A.VrPrestacao BETWEEN 1500 AND 2000 THEN 'Até R$ 2000'
         WHEN A.VrPrestacao > 2000 THEN 'Acima de R$ 2000'
	END AS FxPrestacao,
	CASE WHEN A.VrFinanciamento = . THEN ''
	     WHEN A.VrFinanciamento BETWEEN 0 AND 15000 THEN 'Até R$ 15.000'
		 WHEN A.VrFinanciamento BETWEEN 15000 AND 20000 THEN 'Até R$ 20.000'
		 WHEN A.VrFinanciamento BETWEEN 20000 AND 30000 THEN 'Até R$ 30.000'
		 WHEN A.VrFinanciamento BETWEEN 30000 AND 40000 THEN 'Até R$ 40.000'
		 WHEN A.VrFinanciamento BETWEEN 40000 AND 50000 THEN 'Até R$ 50.000'
         WHEN A.VrFinanciamento > 50000 THEN 'Acima de R$ 50.000'
	END AS FxFinanciamento,
	B.FxIdade,
	CASE WHEN A.NuContratoGestao <> . THEN INTCK('DAY', A.DtEntradaProposta, A.DtStatusProposta) END AS DiasEfetivacao,
	((1+(A.VpTxGerencial/A.VrLiberado)) / (1+(((A.TxTirContabil/100) * (A.VrLiberado))/A.VrLiberado)) -1 ) * A.VrLiberado AS VpSpread2,
	((1+((case when A.VrRetencao > 1 then (Finance('RATE', A.QtParcelas, A.VrPrestacao, -((A.VrProposta-A.VrRetencao)+A.VrIofCpmfIss+A.VrTac))*case when A.StProposta IN ('AP') then A.VrLiberado else 0 end) ELSE A.VpTxGerencial END)/A.VrLiberado)) / (1+(((A.TxTirDivulgada/100) * (A.VrLiberado))/A.VrLiberado)) -1 ) * A.VrLiberado AS VpSpreadRebate2,
	B.FlLeilao
  FROM Originacao01 A
  LEFT JOIN ScoreProposta B
    ON A.NuProposta = B.NuProposta;
RUN;

proc sql;create index NuAnoMes on Originacao;quit;
proc sql;create index CdFilial on Originacao;quit;
proc sql;create index NuProposta on Originacao;quit;
proc sql;create index NmGrupoProdutoMeta on Originacao;quit;
proc sql;create index CdParceiroComercial on Originacao;quit;
proc sql;create index DtEntradaProposta on Originacao;quit;

PROC SQL;
SELECT MAX(DtAtualizacao) INTO :DtOrig FROM Originacao;
RUN;

PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE VrMolicar AS SELECT * FROM connection to dbcon (
  SELECT
    A.NuContrato,
    B.CdModelo,
    A.VrMercado
  FROM
    CDCCG.TbContrato A
  LEFT JOIN
    CDCCG.TbObjGarantiaBemVeiculo B
    ON A.NuContrato = B.NuContrato
  WHERE
    A.DtContrato >= &dtIniPta.
);
RUN;

PROC SORT DATA=VrMolicar NODUPKEY;
BY NuContrato;
RUN;

PROC SQL;
CREATE TABLE Producao AS
  SELECT
    A.*,
	CASE WHEN B.VrPrestacao = . THEN ''
	     WHEN B.VrPrestacao BETWEEN 0 AND 100 THEN 'Até R$ 100'
		 WHEN B.VrPrestacao BETWEEN 100 AND 250 THEN 'Até R$ 250'
		 WHEN B.VrPrestacao BETWEEN 250 AND 500 THEN 'Até R$ 500'
		 WHEN B.VrPrestacao BETWEEN 500 AND 750 THEN 'Até R$ 750'
		 WHEN B.VrPrestacao BETWEEN 750 AND 1000 THEN 'Até R$ 1000'
		 WHEN B.VrPrestacao BETWEEN 1000 AND 1500 THEN 'Até R$ 1500'
		 WHEN B.VrPrestacao BETWEEN 1500 AND 2000 THEN 'Até R$ 2000'
         WHEN B.VrPrestacao > 2000 THEN 'Acima de R$ 2000'
	END AS FxPrestacao,
        C.VrFinanciamento,
	CASE WHEN C.VrFinanciamento = . THEN ''
	     WHEN C.VrFinanciamento BETWEEN 0 AND 15000 THEN 'Até R$ 15.000'
		 WHEN C.VrFinanciamento BETWEEN 15000 AND 20000 THEN 'Até R$ 20.000'
		 WHEN C.VrFinanciamento BETWEEN 20000 AND 30000 THEN 'Até R$ 30.000'
		 WHEN C.VrFinanciamento BETWEEN 30000 AND 40000 THEN 'Até R$ 40.000'
		 WHEN C.VrFinanciamento BETWEEN 40000 AND 50000 THEN 'Até R$ 50.000'
         WHEN C.VrFinanciamento > 50000 THEN 'Acima de R$ 50.000'
	END AS FxFinanciamento,
	INTCK('DAY', A.DtEntradaProposta, A.DtContrato) AS DiasEfetivacao,
	((1+(A.VpTxGerencial/A.VrProducao)) / (1+(((A.TxTirContabil/100) * (A.VrProducao))/A.VrProducao)) -1 ) * A.VrProducao AS VpSpread2,
	((1+((case when B.VrRetencao > 1 then (Finance('RATE', A.QtParcelas, B.VrPrestacao, -((A.VrContrato-B.VrRetencao)+B.VrIof+coalesce(A.VrTarifaCadastro,0)))*A.VrProducao) ELSE A.VpTxGerencial END)/A.VrProducao)) / (1+(((A.TxTirContabil/100) * (A.VrProducao))/A.VrProducao)) -1 ) * A.VrProducao AS VpSpreadRebate2,
	C.FlLeilao,
	D.VrMercado,
	D.CdModelo
  FROM Producao01 A
  LEFT JOIN ScoreContrato B
    ON A.NuContrato = B.NuContrato
  LEFT JOIN ScoreProposta C
    ON A.NuProposta = C.NuProposta
  LEFT JOIN VrMolicar D
    ON A.NuContrato = D.NuContrato;
RUN;

proc sql;create index NuAnoMes on Producao;quit;
