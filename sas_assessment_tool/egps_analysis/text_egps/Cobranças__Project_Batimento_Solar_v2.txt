PROC SQL;
CREATE TABLE CARTEIRA_SOLAR_CONSOL AS SELECT DISTINCT
	A.*,
	CASE WHEN B.UF='' THEN A.CDUF ELSE B.UF END AS UFLOJA,
	CASE WHEN C.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FLAG_SUSP,
	CASE WHEN D.NuContratoGestao IS NOT NULL THEN 1 ELSE 0 END AS FLAG_AJUIZ,
	CASE WHEN E.NuContratoGestao IS NOT NULL THEN 1 ELSE 0 END AS FLAG_ACAO_CONTRA

FROM CARTEIRA_SOLAR_CONSOL A LEFT JOIN TB_REGIAOLOJA B
ON A.CDREGIAO=B.CDREGIAO
LEFT JOIN CARTEIRA_SUSPESAO C
ON a.nucontrato=c.NUCONTRATO

left join AJUIZADOS_SOLAR D
ON a.nucontrato=D.NuContratoGestao AND A.SQCONTRATOFINANCEIROATIVO=D.SQCONTRATOFINANCEIRO

left join ACAO_CONTRA_SOLAR E
ON a.nucontrato=D.NuContratoGestao AND A.SQCONTRATOFINANCEIROATIVO=E.SQCONTRATOFINANCEIRO;

CREATE TABLE CARTEIRA_SOLAR AS
SELECT	DISTINCT

	C.*,
	PUT(ATRASO,FX_ATRASO.) AS FX_ATRASO

 FROM	CONNECTION TO dbcon
		(
		SELECT
	*
	INTO #01
FROM STGTXT_FLEX.TBCONTRATO
WHERE CDPRODUTO = 13
AND CDMODALIDADEPRODUTO = 55

SELECT
	A.NUCONTRATO,
	A.NUCPFCNPJ,
	A.CDPESSOA,
	A.CDREGIAO,
	MAX(B.SQCONTRATOFINANCEIRO) AS SQCONTRATOFINANCEIRO
INTO #02
FROM #01 A LEFT JOIN STGTXT_FLEX.TBPARCELA B
ON A.NUCONTRATO = B.NUCONTRATO
GROUP BY
	A.NUCONTRATO,
	A.NUCPFCNPJ,
	A.CDREGIAO,
	A.CDPESSOA

Select
   a.nucontrato,
   a.SqContratoFinanceiro,
   MIN(CASE WHEN   StParcela <> 'Q' OR
                                               (StParcela = 'Q' AND
                                               DATE(DtUltimoRecebimento) >= &dtcorte.) THEN NuParcela ELSE NULL END) AS parc_atrasada,
   DATEDIFF(DAY, MIN(CASE WHEN    StParcela <> 'Q' OR
                                                               (StParcela = 'Q' AND
                                                              DATE(DtUltimoRecebimento) >= &dtcorte.) THEN DtVencimento ELSE NULL END), &dtcorte.) AS Atraso,
	SUM(CASE WHEN	StParcela <> 'Q' OR
								(StParcela = 'Q' AND
								DATE(DtUltimoRecebimento) >= &dtcorte.)  THEN VrParcela ELSE 0 END) AS Risco_Aberto

        INTO #BASE01
    from STGTXT_FLEX.tbparcela a
    GROUP BY a.nucontrato,a.SqContratoFinanceiro

    Select
        a.nucontrato,
        a.SqContratoFinanceiro,
        a.parc_atrasada,
		b.stparcela,
        b.dtVencimento,
        b.DtUltimoRecebimento,
        a.atraso,
		a.Risco_Aberto

          into #BASE02
    from #BASE01 as a left join STGTXT_FLEX.tbparcela b
    on a.nucontrato=b.nucontrato
    and a.SqContratoFinanceiro=b.SqContratoFinanceiro
    and a.parc_atrasada=b.NuParcela

    Select
        a.nucontrato,
        a.SqContratoFinanceiro,
		A.NUCPFCNPJ,
        b.parc_atrasada as NUPARCELA,
        b.stparcela,
        b.dtVencimento,
        b.DtUltimoRecebimento,
        b.atraso,
		b.Risco_Aberto,
		A.CDREGIAO,
		PS.TPPESSOA,
		ED.CDUF AS CDUF

          into #BASE03
    from #02 as a left join #BASE02 b
    on a.nucontrato=b.nucontrato
    and a.SqContratoFinanceiro=b.SqContratoFinanceiro
	LEFT JOIN STGTXT_FLEX.TbPessoaendereco ED
	ON a.CDPESSOA=ED.CDPESSOA
	LEFT JOIN STGTXT_FLEX.TbPessoa PS
	ON a.CDPESSOA=PS.CDPESSOA

    SELECT  * from #BASE03) C;

QUIT;
