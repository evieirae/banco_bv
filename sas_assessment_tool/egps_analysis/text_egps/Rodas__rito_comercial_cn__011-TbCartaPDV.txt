/* =============== Perfil para comparacao ================= */
/* ==== Mesma Filial, Porte, Tipo de Loja e se Shopping === */
/* ======================================================== */

PROC SQL;
CREATE TABLE TbPerfil AS
SELECT A.NmAreaEstruturaComercial AS P_NmAreaEstruturaComercial
	,CASE WHEN B.PredominanciaProduto = 'Motos' THEN  PortePDV.FlPorteNovoMotos ELSE PortePDV.FlPorteA END as P_FlPorteA
	,B.NmTipoParceiroComercial AS P_NmTipoParceiroComercial
	,CASE WHEN B.NmAutoShopping is null THEN 'N' ELSE 'S' END AS P_FlAutoShopping
	,SUM(Rito.VrProducao) AS P_VrProducao
	,SUM(Rito.VrGravamesMercado) AS P_VrGravamesMercado
	,SUM(Rito.Part_ContratosBV) AS P_Part_ContratosBV
	,SUM(Rito.Part_Mercado) AS P_Part_Mercado
	,CALCULATED P_Part_ContratosBV / CALCULATED P_Part_Mercado AS P_MktShareBV
	,SUM(Rito.QtdProducaoRG) AS P_QtdProducaoRG
	,(CALCULATED P_Part_ContratosBV - CALCULATED P_QtdProducaoRG) / CALCULATED P_Part_Mercado AS P_MktShareCNHBV
	,SUM(Rito.Over30) AS P_Over30
	,Sum(Part_ContratosBV) /  ( Sum(Part_ContratosBV)+Sum(Part_Card)+Sum(Part_Negada)+Sum(Part_APNP)+Sum(Part_Cancelada)+Sum(Part_SemCard)) as P_VirouBV3M
	,SUM(Rito.VRSALDO) AS P_VRSALDO
	,CALCULATED P_Over30 / CALCULATED P_VRSALDO AS P_Over30_Perc
	,SUM(Rito.Part_SemCard)+SUM(Rito.Part_Card)+SUM(Rito.Part_APNP)+SUM(Rito.Part_Negada)+SUM(Rito.Part_Cancelada)+SUM(Rito.Part_ContratosBV) AS P_PartJogo_BV
	,SUM(Rito.Part_Mercado) AS P_PartJogo_Mercado
	,CALCULATED P_PartJogo_BV / CALCULATED P_PartJogo_Mercado AS P_PartJogo3M
FROM BASEGRC.TbMetaNaoIsonomica A
	LEFT JOIN BASES.TbParceiroComercial B ON A.CdParceiroComercial = B.CdParceiroComercial
	LEFT JOIN BASEGRC.TbPortepdv PortePDV ON B.NuCnpjLoja = PortePDV.NuCnpjLoja AND PortePDV.NuAno = year(today())
	LEFT JOIN TbRitoFinal Rito ON Rito.nuanomes = A.NuAnoMes and Rito.CdParceiroComercial = A.CdParceiroComercial
WHERE A.Produto = 'Leves' AND
	A.NuAnoMes between
    	(year(intnx('month', today(), -3, 'b'))*100 + month(intnx('month', today(), -3, 'b'))) and
    	(year(intnx('month', today(), -1, 'b'))*100 + month(intnx('month', today(), -1, 'b'))) and
		B.NmCanalEstruturaComercialC = 'Canal Negocios'

GROUP BY 1,2,3,4
ORDER BY 1,2,3,4
;RUN;

/* ============== Dados do NaPista do PDV ================= */
/* ======================================================== */

PROC SQL;
CREATE TABLE TbParceiroNaPista AS
SELECT CdParceiroComercial
, MIN(data_inicio_plano) AS DtNaPistaInicioRelac Format DDMMYY10.
, MAX(data_fim_plano) AS DtNaPistaFimPlano Format DDMMYY10.
, CASE WHEN CALCULATED DtNaPistaFimPlano >= TODAY() THEN 'ATIVO' ELSE 'INATIVO' END AS StatusNaPista
FROM basegrc.TbNaPistaPagamentosResultado
group by 1
order by 1
;run;

/* ====EXPORTA BASE DE CAMPANHA PARA FILTRAR NO DASH======= */

proc sql;
create table BaseFeirao as
select distinct
AnoMes as NuAnomes,
CdParceiroComercial,
NmModeloEvento as NmEvento,
'Campanha RC' as NmTipoEvento
from basegrc.TbFeiraoArquivos

;run;

proc sql;
create table TBBVRELACIONAR as
select distinct
    NuAnoMes,
    CdParceiroComercial,
    NmCampanha as NmEvento,
	NmTipoEvento
from basegrc.TBBVRELACIONAR
where 1=1
and NmGrupo = 'Grupo Ação'
and NuAnoMes >= &safraM3.
and upcase(NmCampanha) not like '%MOTOS%'
and upcase(NmCampanha) not like '%TESTE%'
;quit;

Data BaseCampanhas;
set TBBVRELACIONAR
	BaseFeirao
;run;

proc sql;
create table TbBaseCampanhaPDV as
select
    NuAnoMes,
    CdParceiroComercial,
	SUM(CASE WHEN NmTipoEvento = 'BV Relacionar' THEN 1 ELSE 0 END) AS BV_Relacionar,
	SUM(CASE WHEN NmTipoEvento = 'BV Compensa' THEN 1 ELSE 0 END) AS BV_Compensa,
	SUM(CASE WHEN NmTipoEvento = 'Campanha RC' THEN 1 ELSE 0 END) AS Campanha_RC,
	SUM(CASE WHEN NmTipoEvento = 'Trade Marketing' THEN 1 ELSE 0 END) AS Trade_Marketing
FROM BaseCampanhas
GROUP BY 1,2
ORDER BY 1,2
;run;

proc sql;
create table LeadsNaPista as
select
a.NuAnoMes_Simu as NuAnoMes
,a.CdParceiroComercialSimulacao as CdParceiroComercial
,count (distinct a.CdSimulacao) as QtVisualizacoes
,count (distinct a.NuClienteId) as QtClientes
,count (distinct b.NuClienteId) as QtContratoBV_leadNP
,count (distinct c.NuClienteId) as QtConversaoConc_leadNP
from leadsdig.leads_digitais_25 a
left join leadsdig.leads_digitais_25 b on a.CdSimulacao = b.CdSimulacao and b.FlPropostaEfetivada=1 AND b.CdParceiroComercialSimulacao = b.CdParceiroComercial
left join leadsdig.leads_digitais_25 c on a.CdSimulacao = c.CdSimulacao and c.StPagoLoja="S" AND c.FlConcorrente=1 AND c.FlPropostaEfetivada<>1
where a.DsSistemaInicial='NaPista' and a.flcards=1
and a.NuAnoMes_Simu >= &safraM3.
and a.CdParceiroComercialSimulacao <> .
/*and a.CdParceiroComercialSimulacao in (9200,31884,238622,289700,451012,106466,463788)*/
GROUP BY 1,2
ORDER BY 1,2
;run;
