proc sql;
  create table WORK.JOIN_PORTAL as
  select
      s.DT_SIMULACAO
    , s.NR_CONTRATO
    , s.SQ_FINANCEIRA
    , s.PRODUTO
    , s.VLR_RISCO
    , s.ATRASO
    , s.FX_ATRASO
    , s.NR_PARC

    
    , divide(s.QTD_PARC_PAGAS
            , (s.QTD_PARC_PAGAS + s.QTD_PARC_N_PAGAS)
            ) as PERC_PARC_PAGAS
        format=percent8.2
        label='Percentual de parcelas pagas'

    , s.COD_ASSESSORIA

    
    , case
        when upcase(coalescec(s.ROUBO,''))                      = 'S'
          or upcase(coalescec(s.ALIENADO_BV,''))                = 'N'
          or upcase(coalescec(s.VENDA_INTENCAO_GRAVAME,''))     = 'S'
          or upcase(coalescec(s.OUTRAS_RESTRICOES,''))          = 'S'
          then 'FILTROS LEAF'
        else 'ELEGIVEL AJUIZAMENTO'
      end as FILTRO_LEAF length=20

    , s.SCORE_AJUIZAMENTO

    
    , put( case
             when s.SCORE_AJUIZAMENTO is null
               or s.SCORE_AJUIZAMENTO < 0
               or s.SCORE_AJUIZAMENTO > 1
               then .
             else s.SCORE_AJUIZAMENTO
           end
          , fxscore.
          ) as FX_SCORE_AJUIZAMENTO length=14
            label='Faixa Score Ajuizamento'

    , case
        when c.NR_CONTRATO is null then 'NAO ENVIADO ASSESSORIA'
        else 'ENVIADO ASSESSORIA'
      end as FILTRO_BASE_CONTROLE length=24

  from WORK.PORTAL_ESTRATEGIA_BASE_SIMULACAO as s
  left join WORK.PORTAL_AJUIZAMENTO_CONTROLE as c
    on  s.NR_CONTRATO   = c.NR_CONTRATO
    and s.SQ_FINANCEIRA = c.SQ_FINANCEIRA
  ;
quit;

proc sql;
  create table WORK.RESUMO_FX_SCORE as
  select
      FX_SCORE_AJUIZAMENTO

    , count(*) as TOTAL_FAIXA

    , sum(case when FILTRO_BASE_CONTROLE = 'NAO ENVIADO ASSESSORIA'
               then 1 else 0 end) as QT_NAO_ENVIADO_TOTAL

    , sum(case when FILTRO_BASE_CONTROLE = 'NAO ENVIADO ASSESSORIA'
                and FILTRO_LEAF = 'FILTROS LEAF'
               then 1 else 0 end) as QT_NAO_ENVIADO_FILTROS_LEAF

    , sum(case when FILTRO_BASE_CONTROLE = 'NAO ENVIADO ASSESSORIA'
                and FILTRO_LEAF = 'ELEGIVEL AJUIZAMENTO'
               then 1 else 0 end) as QT_NAO_ENVIADO_ELEGIVEL

  from WORK.JOIN_PORTAL
  group by FX_SCORE_AJUIZAMENTO
  order by FX_SCORE_AJUIZAMENTO;
quit;




data WORK.RESUMO_FX_SCORE;
  set WORK.RESUMO_FX_SCORE;

  
  PCT_LEAF_FAIXA = divide(QT_NAO_ENVIADO_FILTROS_LEAF, TOTAL_FAIXA);
  PCT_ELEG_FAIXA = divide(QT_NAO_ENVIADO_ELEGIVEL,     TOTAL_FAIXA);

proc sql;
  create table WORK.BASE_ENRIQUECIDA as
  select
      ELGF.*,

      
      UF.cd_regional as UF length=5,

      
      case
        
        when prxmatch('/TOLEDO/i',
              upcase(tranwrd(coalescec(LTV.nome_assessoria,''), '_', ' ')))
          then 'TOLEDO'

        
        when prxmatch('/(?:^|[^A-Z0-9Á-ÚÁ-Ú])RN(?:$|[^A-Z0-9Á-ÚÁ-Ú])/i',
              upcase(tranwrd(coalescec(LTV.nome_assessoria,''), '_', ' ')))

              upcase(tranwrd(coalescec(LTV.nome_assessoria,''), '_', ' ')))

	end as REVISIONAL

  left join (
      select distinct nr_contrato, cd_regional
      from WORK.BASE_UF

	AND R.NuContratoText IS NULL

quit;

proc sql;
  create table work.uf_cotas as
  select
    d.uf,
    d.pct_uf,
    (&META_AJPARC_MES. * d.pct_uf) as meta_uf,
    coalesce(a.qtd,0)              as ajuizados_uf,
    coalesce(p.qtd,0)              as ajparc_prev_uf,
    coalesce(s.share_paschoa_uf,0) as share_paschoa_uf,
    (&P_AJ_PARCIAL. * (1 - &P_BAP_PASCHOA. * coalesce(s.share_paschoa_uf,0))) as p_aj_eff_uf,
    max( (&META_AJPARC_MES. * d.pct_uf) - (coalesce(a.qtd,0) + coalesce(p.qtd,0)), 0 ) as need_uf,
    ceil( max( (&META_AJPARC_MES. * d.pct_uf) - (coalesce(a.qtd,0) + coalesce(p.qtd,0)), 0 )
          / case when (&P_AJ_PARCIAL. * (1 - &P_BAP_PASCHOA. * coalesce(s.share_paschoa_uf,0)))>0
                 then (&P_AJ_PARCIAL. * (1 - &P_BAP_PASCHOA. * coalesce(s.share_paschoa_uf,0)))
                 else 1e-6 end
          * &FATOR_OVER.
        ) as quota_uf
  from work.dist_uf d
  left join work.ajuizados_mes_uf a
    on d.uf = a.uf
  left join work.ajparc_prev_uf p
    on d.uf = p.uf
  left join work._share_paschoa s
    on d.uf = s.uf
  ;
quit;
