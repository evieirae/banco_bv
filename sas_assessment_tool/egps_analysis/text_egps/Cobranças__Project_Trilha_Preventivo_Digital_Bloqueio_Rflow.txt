CREATE TABLE WORK.BASE_ACORDO_01 AS
SELECT DISTINCT A.*
FROM CONNECTION TO DBCON(
SELECT DISTINCT
  Year(A.DtGeracao)*100+Month(A.DtGeracao) AS SAFRA
, A.DtGeracao
, A.NuContratoGestao
, A.SqContratoFinanceiro
, A.TpEntrada
, A.DtVencimento
, A.StNegociacao
, A.TpAcordoNegociacao
, A.VrTotParcela
, A.VrTotMora
, A.VrTotMulta
, A.VrTotHonorario
, A.VrTotDespesa
, A.VrBoleto
, A.DtCancelamento
, A.DtStNegociacao
FROM dbcontencioso..TbNegociacaoContrato A
WHERE A.DtGeracao  >= DATEADD(DD, -20, GETDATE())
AND A.DtVencimento <= DATEADD(DD,  20, GETDATE())

AND A.StNegociacao NOT IN ('C','PL','LP','R')
AND A.DtVencimento > DATEADD(DD, -1, GETDATE())
ORDER BY A.DtGeracao ASC, A.NuContratoGestao) AS A
;QUIT;

PROC SQL;
CREATE TABLE WORK.BASE_ACORDO_02 AS
SELECT DISTINCT A.*
, intck('DAY', DATEPART(A.DtGeracao), DATEPART(A.DtVencimento), 'C')  AS Intervalo_GerVenc_Bol
, DATEPART(A.DtGeracao) Format yymmdd10. AS Dt_Geracao_Boleto
, DATEPART(A.DtVencimento) Format yymmdd10. AS Dt_Vencimento_Boleto
, HOUR(A.DtGeracao) AS Hr_Geracao
, CASE
				When A.StNegociacao = 'AN' Then 'A Negociar'
				When A.StNegociacao = 'N'  Then 'Negociado'
				When A.StNegociacao = 'PA' Then 'Pendente De Aprovacao'
				When A.StNegociacao = 'A'  Then 'Aprovado Pela Bv'
				When A.StNegociacao = 'R'  Then 'Recusado Pela Bv'
				When A.StNegociacao = 'NA' Then 'Negociacao Em Andamento'
				When A.StNegociacao = 'C'  Then 'Negociacao Cancelada'
				When A.StNegociacao = 'LP' Then 'Liberado Pagamento'
				When A.StNegociacao = 'PG' Then 'Pago'
				When A.StNegociacao = 'PL' Then 'Pagamento Realizado'
				When A.StNegociacao = 'DA' Then 'Devolvido Assessoria'
				When A.StNegociacao = 'CD' Then 'Cheque Devolvido'
				When A.StNegociacao = 'AC' Then 'Cheque Aguardando Compensacao'
ELSE '99.Verificar' END AS Status_Boleto
, B.NuCPFCNPJ
, B.ATRASO AS ATRASO_ATUAL
, B.PRI_NOME
, intck('DAY', DATEPART(A.DtGeracao), TODAY(), 'C')  AS Dias_Boleto_Gerado
, CASE WHEN A.NuContratoGestao = B.NuContratoGestao THEN 1 ELSE 0 END AS Flag_Base_Cob
, CASE WHEN A.DtGeracao IS NOT NULL THEN 1 ELSE 0 END AS Flag_Boleto_Gerado
, CASE WHEN A.StNegociacao IN ('PL','LP') THEN 1 ELSE 0 END AS Flag_Boleto_Pago
, SUBSTRN(PUT(A.NUCONTRATOGESTAO,Z14.),1,2) AS COD_PRODUTO
, CASE
	WHEN SUBSTRN(PUT(A.NUCONTRATOGESTAO,Z14.),1,2) IN ('12','14') THEN 'RODAS'
	WHEN SUBSTRN(PUT(A.NUCONTRATOGESTAO,Z14.),1,2) IN ('10','11','13') THEN 'EMPRESTIMOS'
	ELSE 'OUTROS' END AS PRODUTO_MACRO
, C.CANAL AS CANAL_BOLETO
, C.CANAL_MACRO
, CASE WHEN C.Flag_Prev_BV = 1 THEN 'S' ELSE 'N' END AS Flag_Prev_BV
, CASE WHEN B.NUCPFCNPJ = D.NUCPFCNPJ THEN 'S' ELSE 'N' END AS Flag_Mult_Contr
, CASE WHEN A.NUCONTRATOGESTAO = E.NUCONTRATOGESTAO AND A.DtGeracao = E.ULT_BOLETO THEN 'S' ELSE 'N' END AS Flag_Ult_Boleto
FROM WORK.BASE_ACORDO_01 AS A
LEFT JOIN CONTATOS.AUX_BASE_CYBER AS B ON A.NuContratoGestao = B.NuContratoGestao

LEFT JOIN TCONFIG.CONFIG_TPENTRADA AS C ON A.TpEntrada = C.TpEntrada
LEFT JOIN (SELECT DISTINCT A.NUCPFCNPJ, COUNT(DISTINCT A.NuContratoGestao) AS Qtd_Contr FROM CONTATOS.AUX_BASE_CYBER AS A GROUP BY A.NUCPFCNPJ Having Qtd_Contr > 1) AS D ON B.NUCPFCNPJ = D.NUCPFCNPJ
LEFT JOIN (SELECT DISTINCT A.NUCONTRATOGESTAO, MAX(A.DtGeracao) AS ULT_BOLETO FROM WORK.BASE_ACORDO_01 AS A GROUP BY A.NUCONTRATOGESTAO) AS E ON A.NUCONTRATOGESTAO = E.NUCONTRATOGESTAO AND A.DtGeracao = E.ULT_BOLETO
ORDER BY A.DtVencimento ASC
;QUIT;
