/*-------- Consulta Ass. Eletrônica no PTASYBCTR --------*/
%let IQUser = lgettbasasm;
%let IQPassword = {SAS002}2CA25D1B221200514F0483281B352D1E21E9EC37458D4F7F;

proc sql;
	connect to sybaseiq as

		dbcon (host='pta-iqprd01' server=ptaiqprd user="&IQUser." password="&IQPassword.");/**/
	create table work.ASSINAT as
		select
			*
		from connection to dbcon
(

 SELECT
    CAST(proposta.Dtentradaproposta AS DATE) AS dtentradaproposta,
     proposta.nuproposta,
     Proposta.QtDiasCarencia,
       CASE
        WHEN PropostaComplemento.CdTipoMeioVenda = 10 THEN 'Eletronica'
        WHEN PropostaComplemento.CdTipoMeioVenda = 0 THEN 'Convencional'
        ELSE 'N/I'
    END AS TipoAssinatura,
    tbpessoafisica.CdSexo AS Genero,
    CASE
        WHEN tbpessoafisica.CdTipoResidencia = 1 THEN 'PROPRIO'
        WHEN tbpessoafisica.CdTipoResidencia = 2 THEN 'FAMILIAR'
        WHEN tbpessoafisica.CdTipoResidencia = 3 THEN 'ALUGADO'
        WHEN tbpessoafisica.CdTipoResidencia = 4 THEN 'NÃO INFORMADO'
        WHEN tbpessoafisica.CdTipoResidencia = 5 THEN 'PROPRIO FINANCIADO'
        WHEN tbpessoafisica.CdTipoResidencia = 6 THEN 'FUNCIONAL'
        WHEN tbpessoafisica.CdTipoResidencia = 7 THEN 'OUTROS'
        WHEN tbpessoafisica.CdTipoResidencia = 8 THEN 'CEDIDO'
    END AS DsTipoResidencia,

    CASE
        WHEN tbpessoafisica.CdTipoProfissional = 0 then 'NAO DEFINIDO'
WHEN tbpessoafisica.CdTipoProfissional = 1 then  'ASSALARIADO'
WHEN tbpessoafisica.CdTipoProfissional = 2 then  'EMPRESARIO'
WHEN tbpessoafisica.CdTipoProfissional = 3 then  'AUTONOMO'
WHEN tbpessoafisica.CdTipoProfissional = 4 then  'LIBERAL'
WHEN tbpessoafisica.CdTipoProfissional = 5 then  'OUTROS'
WHEN tbpessoafisica.CdTipoProfissional = 6 then  'FUNCIONARIO PUBLICO'
WHEN tbpessoafisica.CdTipoProfissional = 7 then  'APOSENTADO OU PENSIONISTA'
WHEN tbpessoafisica.CdTipoProfissional = 8 then  'INVESTIDOR'
    END as DsTipoProfissional,
    PROF.DsProfissao,
     CASE
        WHEN CdEstadoCivil = 1 then 'DESQUITADO'
WHEN CdEstadoCivil = 2 then 'DIVORCIADO'
WHEN CdEstadoCivil = 3 then 'SEPARADO'
WHEN CdEstadoCivil = 4 then 'SOLTEIRO'
WHEN CdEstadoCivil = 5 then 'CASADO'
WHEN CdEstadoCivil = 6 then 'VIUVO'
WHEN CdEstadoCivil = 7 then 'OUTROS'
    END AS DsEstadoCivil,
    tbpessoafisica.VrTotalReceitas AS Renda

FROM
    DBA.TbProposta Proposta
LEFT JOIN DBA.TbPessoaSobAnalise Pessoasobanalise
    ON Proposta.NuProposta = Pessoasobanalise.NuProposta

LEFT JOIN DBA.TbPessoaFisicaSobAnalise tbpessoafisica
    ON PessoaSobAnalise.CdPessoaSobAnalise = tbpessoafisica.CdPessoaSobAnalise
        JOIN DBA.TbProfissao PROF
    ON  PROF.CDPROFISSAO = tbpessoafisica.CDPROFISSAO
LEFT JOIN DBA.TbPessoaSobAnaliseTipoTelefone TelRes
    ON TelRes.CdPessoaSobAnalise = Pessoasobanalise.CdPessoaSobAnalise
    AND TelRes.CdTipoTelefone = 1
LEFT JOIN DBA.TbPessoaSobAnaliseTipoTelefone TelCom
    ON TelCom.CdPessoaSobAnalise = Pessoasobanalise.CdPessoaSobAnalise
    AND TelCom.CdTipoTelefone = 2
LEFT JOIN DBA.TbPessoaSobAnaliseTipoTelefone TelCel
    ON TelCel.CdPessoaSobAnalise = Pessoasobanalise.CdPessoaSobAnalise
    AND TelCel.CdTipoTelefone = 3
LEFT JOIN CPC.TbContrato TbContrato
    ON Proposta.NuContratoGestao = TbContrato.NuContrato
LEFT JOIN DBA.TbPropostaComplemento PropostaComplemento
    ON PropostaComplemento.NuProposta = Proposta.NuProposta
LEFT JOIN DBA.TbParceiroComercial Parceiropessoa
    ON Parceiropessoa.CdParceiroComercial = Proposta.CdPessoa
LEFT JOIN DBA.TbParceiroComercial Parceirocom
    ON Parceirocom.CdParceiroComercial = Proposta.CdLoja
LEFT JOIN DBA.TbParceiroComercial Coban
    ON Coban.CdParceiroComercial = Proposta.CdPessoa
LEFT JOIN DBA.TbPessoa Pessoa
    ON Pessoa.CdPessoa = Parceiropessoa.CdPessoa
LEFT JOIN DBA.TbPessoa PessoaConvenio
    ON PessoaConvenio.CdPessoa = Parceirocom.CdPessoa
LEFT JOIN DBCRED.TbPropostaConsignado Tbsimu
    ON Tbsimu.NuProposta = Proposta.NuProposta
LEFT JOIN DBCOR.TbParceiroComercialColaborador ParceiroColab
    ON Parceirocom.CdParceiroComercial = ParceiroColab.CdParceiroComercial
LEFT JOIN (
    SELECT *
    FROM DBCOR.TbEmpresaConveniada
    WHERE FlAverbacaoAutomatica = 'N'
      AND FlPermiteAverbacaoSite = 'S'
) conveniada
    ON conveniada.CdLoja = Parceirocom.CdParceiroComercial
WHERE
    Proposta.CdSubProduto IN (12,18,22,23,24,25,26,70)
/*,19,20,75, 100, 8, 13, 59, 60, 63, 62, 38, 54,55, 56, 93, 67, 68, 71, 73, 72, 77, 82, 86, 85, 87, 88, 89, 96, 98, 97, 101*/
    AND CAST(Proposta.Dtentradaproposta AS DATE) >= '2024-01-01'
);
QUIT;
