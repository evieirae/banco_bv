	Create Table TBL_TRANSACAO_DEBITO_1 AS
		Select a.*,
		       Case When b.NmBandeira = 'MASTERCARD' Then b.NmBandeira
		            else 'VISA'
				End as NmBandeira

		From TBL_TRANSACAO_DEBITO a
		Left Join TBL_BANDEIRA_CONTA b ON (a.NuContaCorrente= b.Nuconta)

;Quit;



PROC SQL;
	Create Table TBL_TRANSACAO_DEBITO_2 as
		Select
			input(SafraRef, yymmn6.) as Data format ddmmyy10.,

			a.NUCONTACORRENTE as Conta,
			a.Qtd_Transacoes as Qtde_Transacoes,
			a.nmBandeira,
			a.VRTRANSACAO_DEBITO as VRTRANSACAO_DEBITO,
			Case When Upper(a.nmBandeira) = 'VISA' then a.VRTRANSACAO_DEBITO / Sum(a.VRTRANSACAO_DEBITO) else 0 End  as DriverVisa,
			Case When Upper(a.nmBandeira) = 'MASTERCARD' then a.VRTRANSACAO_DEBITO / Sum(a.VRTRANSACAO_DEBITO) else 0 End  as DriverMaster,
			Case When Upper(a.nmBandeira) = 'VISA' then a.VRTRANSACAO_DEBITO / Sum(a.VRTRANSACAO_DEBITO) else 0 End * &Visa. as Receita_Visa,
			Case When Upper(a.nmBandeira) = 'MASTERCARD' then a.VRTRANSACAO_DEBITO / Sum(a.VRTRANSACAO_DEBITO) else 0 End * &Master. as Receita_Master,
			Calculated Receita_Visa + Calculated Receita_Master as Receita_Intercambio

		From TBL_TRANSACAO_DEBITO_1 AS A
		Group by a.nmBandeira
;QUIT;

Proc sql;
	Connect to bigquery as perm (project='bv-safn-dtm' credfile='/projetos/acessos/bq/fn/safn-dtm.json' schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
	Create table baseMyabc AS
		Select * FROM CONNECTION TO perm (
		select
			'Real' as Cenario,
			dt_periodo as data,
			EXTRACT(YEAR FROM dt_periodo) * 100 + EXTRACT(MONTH FROM dt_periodo) AS AnoMes,
			
			
		Case
			When nm_grupo_despesa in ('Contingências') then nm_grupo_conta
			else nm_grupo_despesa
		end
	AS Grupo_Despesa,
		nm_grupo_acionista as Grupo_acionista,
		nm_grupo_conta as Grupo_conta,
		cast(REGEXP_REPLACE(nm_classe_valor, r'[^0-9]', '') as int64) AS ClasseValor,
		REGEXP_REPLACE(nm_classe_valor, r'[^A-Za-zÀ-ÿ ]', '') AS NmClasseValor,
				tp_classificacao as Tipo_Classificacao,
		tp_classificacao_contabil as class_contabil,

		sum(Vr_Gasto) as Valor
	from bv-pdbd-prd.dvry_corp_financas.myabc_custosdespesasmis_fato
		where Nm_Produto in('Banco Digital','BANCO DIGITAL','PLATAFORMA CORE CONTA','Plataforma Core Conta','VCS BD')
			
			and Nm_canal not in ('Méliuz')
			and Ds_modelo in("R'26")
			and nm_grupo_acionista not in ( 'ELIMINAÇÕES', 'Head Count', 'HeadCount')
			AND dt_periodo BETWEEN &DATA_I. AND &DATA_F.
		GROUP BY all
			)
	;
	disconnect from perm
	;
Quit;

		a.class_contabil
	from work.baseMyabc as a left join work.DeparaDRE_myabc as b on (a.ClasseValor = b.ClasseValor);
