CREATE TABLE work.tblProcessoAndamento_Projuris  AS
SELECT

 Cd_Causa as Cd_Causa
,SERIAL_IMPORT as SERIAL_IMPORT
,pasta as pasta
,IdNatureza as IdNatureza
,IdNaturezaUnidadeCustom as IdNaturezaUnidadeCustom
,CdCausaUnidade as CdCausaUnidade
,CdSistema as CdSistema
,CdUnidadeExyon as CdUnidadeExyon
,DataAndamento as DataAndamento
,CdFaseAndamento as CdFaseAndamento

,DtRefEvento as DtRefEvento
,IdProcessoAndamento as IdProcessoAndamento
,DtCumprimento as DtCumprimento
,CadastradoPor as CadastradoPor

FROM CONNECTION TO postgres(

select
case
 	when pr.serial_import like ('%INQ%') then (left(pr.SERIAL_IMPORT,10))
    when pr.serial_import like ('%BVT%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%CON%') then (left(pr.SERIAL_IMPORT,10))
	when pr.serial_import like ('%VAC%') then (left(pr.SERIAL_IMPORT,10))
  	when pr.serial_import like ('%CTO%') then (left(pr.SERIAL_IMPORT,10))
  	when pr.serial_import like ('%TRB%') then (left(pr.SERIAL_IMPORT,10))
  	when pr.serial_import like ('%TRI%') then (left(pr.SERIAL_IMPORT,10))
		else pr.serial_import end                      as Cd_Causa

,pr.SERIAL_IMPORT
,pr.pasta
,pr.ID_NATUREZA 							           as IdNatureza
,pr.NATUREZA_UNIDADE_CUSTOM 				           as IdNaturezaUnidadeCustom
,null::numeric						                   as CdCausaUnidade
,'p' 										           as CdSistema

,case
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 2 			then 'TRI'
	when pr.ID_NATUREZA = 28 and pr.NATUREZA_UNIDADE_CUSTOM = 1  			then 'TBR'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 4 			then 'CTO'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 12 			then 'CON'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM = 13 			then 'VAC'
	when pr.ID_NATUREZA = 21 and pr.NATUREZA_UNIDADE_CUSTOM IN (1 ,7, 11)	then 'INQ'
	when pr.ID_NATUREZA = 27 and pr.NATUREZA_UNIDADE_CUSTOM = 5             then 'BVT'
	when pr.ID_NATUREZA = 32 and pr.NATUREZA_UNIDADE_CUSTOM IN (7, 11 , 12) then 'INQ'
		else null end 						           as CdUnidadeExyon

 ,to_char(cast(pe.data_cadastro as date),'DD/MM/YYYY') as DataAndamento
 ,ev.ID_EVENTO										   as CdFaseAndamento
 ,cast(pe.data_cadastro as date) 					   as DtAndamento
 ,upper(ev.EVENTO) 									   as NmAndamento
 ,us1.nome 											   as NmRecurso
 ,cast(pe.DATA  as date) 						       as DtRefEvento
 ,pe.ID_PROCESSO_EVENTO            				   	   as IdProcessoAndamento
 ,cast(pe.DATA_CUMPRIMENTO as date)                    as DtCumprimento
 ,u.NOME                                               as CadastradoPor

from         public.M_PROCESSO                pr
  inner join public.D_PROCESSO_DESDOBRAMENTO  pd on pd.ID_PROCESSO               = pr.ID_PROCESSO
  inner join public.D_PROCESSO_EVENTO         pe on pe.ID_PROCESSO_DESDOBRAMENTO = pd.ID_PROCESSO_DESDOBRAMENTO
  left join  public.MM_EVENTO_ADVOGADO        ea on ea.ID_PROCESSO_EVENTO        = pe.ID_PROCESSO_EVENTO
  left join  public.M_USUARIO                  u on u.ID_USUARIO                 = pe.ID_USUARIO_CADASTRO
  left join  public.M_USUARIO                us1 on us1.ID_USUARIO               = ea.ID_USUARIO
  inner join public.LT_EVENTO                 ev on ev.ID_EVENTO                 = pe.ID_EVENTO

;

);DISCONNECT FROM postgres;
RUN;
