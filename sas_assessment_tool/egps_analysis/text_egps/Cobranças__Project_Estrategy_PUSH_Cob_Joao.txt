    %local i ds n exist_list sep;
    %let n = 1;
    %let ds = %scan(&in, &n, %str( %sysfunc(byte(10)) %sysfunc(byte(13)) )); 
    %let exist_list = ;
    %let sep = ;

    %do %while(%length(&ds));
        %if %sysfunc(exist(&ds)) %then %do;
            %let exist_list = &exist_list.&sep.&ds;
            %let sep = %str( );
        %end;
        %else %do;
            %put WARNING: Dataset &ds não encontrado. Ignorando no passo &out.;
        %end;
        %let n = %eval(&n + 1);
        %let ds = %scan(&in, &n, %str( %sysfunc(byte(10)) %sysfunc(byte(13)) ));
    %end;

    %if %length(&exist_list) %then %do;
        data &out;
            set &exist_list;
            %if %superq(where) ne %then %do;
                where &where;
            %end;
        run;
    %end;
    %else %do;
        %put WARNING: Nenhum dataset de entrada existe para &out.. Saída não será criada.;
    %end;
%mend append_existing;
