%macro EXTRAI_PDD(SAFRA);
	proc sql;
		create table work.aux_pdd_&SAFRA. as
			select
				&SAFRA. as SAFRA,
				case
					when DsGrupoProduto in ('BNDES','Convênio Flex','Crédito Rural','Créditos do Agronegócio','Empréstimos','External Loans','External Trade Finance','Leasing','Operações de Garantias','Trade Finance') then 'PDD'
					when DsGrupoProduto in ('External Credit Derivatives','External Hedge','Hedge') then 'DERIVATIVO'
					when DsGrupoProduto in ('TVM') then 'AJUSTE PRUDENCIAL E IMPAIRMENT'
					else DsGrupoProduto
				end as Segmentacao length 50,
				sum(VrSaldo) as Saldo format commax30.,
				sum(VrProvisao) as PDD format commax30.,
				sum(VrAjustePrudencial) as Ajuste_Prudencial format commax30.,
				sum(VrImpairment) as Impairment format commax30.
			from _miscred.base_unica_atacado_&SAFRA.
			WHERE AnoMes_Referencia = &SAFRA.
				AND VrProvisao > 0
				AND DtInicioOperacao NE .
				AND DsGrupoProduto NE ''
			group by 1,2;
	quit;
%mend;
%FOREACH(%REFVECM(202301,202412),EXTRAI_PDD);

DATA BASE_PDD_202301_202412;
	SET WORK.aux_pdd_20:;
RUN;

proc sql;
create table SAIDA as
select safra,
PDD/Saldo as SDPDD
from BASE_PDD_202301_202412 (where=(Segmentacao="PDD"));
quit;
