		WHEN a.nudiasatraso BETWEEN 1 AND 4 THEN 'b. 1 a 4'
		WHEN a.nudiasatraso BETWEEN 5 AND 30 THEN 'c. 5 a 30'

        WHEN a.nudiasatraso BETWEEN 211 AND 270 THEN 'j. 211 a 270'
		ELSE 'k. 271 a 540'

	M.CDUF,
    (CAST(a.vrpermanenciaapropriado AS FLOAT64) + CAST(a.vrtaxacontrato AS FLOAT64)) AS SALDO,
	e.SALDO as SALDO_M1

LEFT JOIN
	(SELECT NUCONTRATO,
	 		sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,
	 		(CAST(vrpermanenciaapropriado AS FLOAT64) + CAST(vrtaxacontrato AS FLOAT64)) AS SALDO
	 FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil

	 										 a.sqcontratofinanceiro = e.sqcontratofinanceiro

    a.dtprocessamento = &DATA.
    AND a.nudiasatraso <= 540

	AND a.dtprocessamentoliquidacao is null
	AND a.dtprimeirovencimento <= &DATA.


UNION ALL

SELECT DISTINCT
    CAST(X.nucontrato AS INT64) AS NUCONTRATO,
	y.cd_cpf_cnpj AS nucpfcnpj,
    x.sqcontratofinanceiro AS SQCONTRATOFINANCEIRO,

        WHEN x.nudiasatraso BETWEEN 181 AND 210 THEN 'i. 181 a 210'
        WHEN x.nudiasatraso BETWEEN 211 AND 270 THEN 'j. 211 a 270'
		ELSE 'k. 271 a 540'
    END AS FX_ATRASO_M0,
	V.CDUF,
    (CAST(x.vrpermanenciaapropriado AS FLOAT64) + CAST(x.vrtaxacontrato AS FLOAT64)) AS SALDO,
	z.SALDO as SALDO_M1

    bv-pdbd-prd.trusted_stgtxt_flex.tbcontratocontabil AS X

CREATE TABLE BASE_FPD AS
SELECT DISTINCT

*
,CASE WHEN DIAS_REG <= 0 THEN 0 ELSE DIAS_REG END AS DIAS_REG2

,CASE WHEN DIAS_REG <= 0 THEN 0 ELSE 1 END AS FL_ATRASO

,YEAR(dt_vencimento)*100+MONTH(dt_vencimento) AS safra_venc

,YEAR(dt_vencimento+5)*100+MONTH(dt_vencimento+5) AS safra_competencia

,CASE WHEN CALCULATED DIAS_REG2 >= 5 THEN 1 ELSE 0 END AS FL_ENTRADA_COB

FROM CONNECTION TO PERM(

SELECT DISTINCT
    a.cd_contrato,
    a.sq_contrato_financeiro,
    a.dt_vencimento,
    a.dt_ultimo_recebimento,
    a.st_parcela,
    a.vr_parcela,
	a.nu_parcela,
	c.saldo,
	CASE
        WHEN (a.st_parcela = 'Q' OR a.dt_ultimo_recebimento IS NOT NULL) THEN (DATE_DIFF(a.dt_ultimo_recebimento, a.dt_vencimento, DAY))
		ELSE 9999 END AS DIAS_REG,

	CASE
		 WHEN (DATE_DIFF(CURRENT_DATE()-1, e.min_dt_vencimento, DAY))>=5 THEN 1
	ELSE 0 END AS FL_MATUROU

FROM
    bv-pdbd-prd.dvry_core.tbparcela A

LEFT JOIN
	bv-pdbd-prd.dvry_core.tbcontrato AS B ON a.cd_contrato = b.cd_contrato
LEFT JOIN
    bv-pdbd-prd.trusted_dbcred.tbproposta AS D ON a.cd_contrato = d.nucontratogestao

	(select cd_contrato, sq_contrato_financeiro, sum(vr_parcela) as saldo
		from bv-pdbd-prd.dvry_core.tbparcela
	 group by cd_contrato, sq_contrato_financeiro) AS C ON a.cd_contrato = c.cd_contrato and a.sq_contrato_financeiro = c.sq_contrato_financeiro

	(select cd_contrato, sq_contrato_financeiro, min(dt_vencimento) as min_dt_vencimento
		from bv-pdbd-prd.dvry_core.tbparcela
		group by cd_contrato, sq_contrato_financeiro) AS E ON a.cd_contrato = e.cd_contrato and a.sq_contrato_financeiro = e.sq_contrato_financeiro

WHERE (CAST(b.cd_produto AS INT64) IN (12,14)
	  )
	  and extract (year from dt_vencimento) >= 2026

)
group by
	cd_contrato,
    sq_contrato_financeiro

;quit;
