PROC SQL;
CREATE TABLE WORK.ENVIO_HSM_01 AS
SELECT DISTINCT
  CASE WHEN LENGTH(COMPRESS(NuCPFCNPJ)) <=11 THEN PUT(INPUT(NuCPFCNPJ,best14.),z11.)
  	   WHEN LENGTH(COMPRESS(NuCPFCNPJ)) <=14 THEN PUT(INPUT(NuCPFCNPJ,best14.),z14.)
	   ELSE COMPRESS(NuCPFCNPJ) END AS NuCPFCNPJ
, NuContrato
, Hashtag
, Template
, WhatsApp_Cliente
, Solicitacao_Envio
, Envio
, Confirmacao
, Leitura_Se_Habilitado
, Falha_Entrega
, Motivo_Falha
, WhatsApp_Saida
, Usuario_Solicitante
, CASE
		WHEN CARTEIRA IS MISSING AND UPCASE(TEMPLATE) LIKE ('%PESQUISA_PRESENCIAL%') 	THEN 'Pesquisa_Presencial'
		WHEN CARTEIRA IS MISSING AND UPCASE(TEMPLATE) LIKE ('%TROCA%GARANT%') 			THEN 'Subst_de_Garantia'
		WHEN CARTEIRA IS MISSING AND UPCASE(TEMPLATE) LIKE ('%TIME%QUALIDADE%') 		THEN 'Closetheloop'
		ELSE CARTEIRA END AS Carteira
, CASE
		WHEN Teve_Retorno = 'S'   THEN 'SIM'
		WHEN Teve_Retorno = 'SIM' THEN 'SIM'
		WHEN Teve_Retorno = 'N'   THEN 'NAO'
		WHEN Teve_Retorno = 'NÃO' THEN 'NAO'
		WHEN Teve_Retorno = 'NAO' THEN 'NAO'
END AS Teve_Retorno
, INPUT(COMPRESS(CATS(SUBSTR(Solicitacao_Envio,7,4),SUBSTR(Solicitacao_Envio,4,2))),best6.) AS Safra
, INPUT(COMPRESS(CATS(SUBSTR(Solicitacao_Envio,7,4),'-',SUBSTR(Solicitacao_Envio,4,2),'-',SUBSTR(Solicitacao_Envio,1,2))),yymmdd10.) Format yymmdd10.   AS Dt_Solicitacao_Envio
, CASE WHEN Envio IS NOT MISSING THEN 1 ELSE 0 END AS Flag_Enviado
, CASE WHEN Confirmacao IS NOT MISSING THEN 1 ELSE 0 END AS Flag_Confirmado
, CASE WHEN Leitura_Se_Habilitado IS NOT MISSING THEN 1 ELSE 0 END AS Flag_Leitura
, CASE WHEN Falha_Entrega IS NOT MISSING THEN 1 ELSE 0 END AS Flag_Falha
, CASE WHEN Teve_Retorno = 'S' THEN 1 WHEN Teve_Retorno = 'SIM' THEN 1 ELSE 0 END AS Flag_Interacao
, 'WHATS' AS Canal
FROM WORK.ENVIOS_ATIVOS_HSM
WHERE Solicitacao_Envio NOT IN ('Solicitação envio')
;QUIT;

PROC SQL;
CREATE TABLE WORK.RESUMO_INTERACAO AS
SELECT DISTINCT
  A.NuCPFCNPJ
, A.NuContrato
, A.WhatsApp_Cliente
, COUNT(A.Dt_Solicitacao_Envio) AS Qtd_Registros
, SUM(A.Flag_Enviado) AS Qtd_Enviados
, MIN(A.Dt_Solicitacao_Envio) Format = yymmdd10. AS Dt_MinEnvio
, MAX(A.Dt_Solicitacao_Envio) Format = yymmdd10. AS Dt_MaxEnvio
, CASE WHEN B.Flag_Interacao = 1 THEN 1 ELSE 0 END AS Interacao
, SUM(A.Flag_Interacao) AS Qtd_Interacoes
, B.Dt_MaxInteracao Format = yymmdd10. AS Dt_MaxInteracao
FROM WORK.ENVIO_HSM_FINAL AS A
LEFT JOIN (SELECT DISTINCT A.NuCPFCNPJ, A.NuContrato, A.WhatsApp_Cliente, A.Flag_Interacao, MAX(A.Dt_Solicitacao_Envio) AS Dt_MaxInteracao
FROM WORK.ENVIO_HSM_FINAL AS A
WHERE A.Flag_Interacao = 1
GROUP BY A.NuCPFCNPJ, A.NuContrato, A.WhatsApp_Cliente, A.Flag_Interacao) AS B ON A.NuCPFCNPJ = B.NuCPFCNPJ AND A.NuContrato = B.NuContrato AND A.WhatsApp_Cliente = B.WhatsApp_Cliente
WHERE A.SAFRA IS NOT MISSING
GROUP BY A.NuCPFCNPJ
, A.NuContrato
, A.WhatsApp_Cliente
, B.Dt_MaxInteracao
, CASE WHEN B.Flag_Interacao = 1 THEN 1 ELSE 0 END
;QUIT;

CREATE TABLE WORK.WHATS_INTERACAO_MANUAL AS
SELECT DISTINCT A.NuContrato, MAX(A.Interacao) AS Flag_Interacao, MAX(A.Dt_MaxInteracao) Format = yymmdd10. AS Dt_MaxInteracao
FROM CONTATOS.AUX_HIST_INTERACAO_WHATS  AS A
INNER JOIN CONTATOS.AUX_BASE_CYBER 		AS B ON A.NuContrato = B.NuContratoGestao
Where A.NuContrato IS NOT MISSING
GROUP BY A.NuContrato
Having Dt_MaxInteracao > INTNX('DAY',TODAY(),-65)
;QUIT;
