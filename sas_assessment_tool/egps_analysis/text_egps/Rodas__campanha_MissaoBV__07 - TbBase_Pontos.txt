PROC SQL;
CREATE TABLE TbPts_Crescimento_1 AS
SELECT

A.NUANOMES,
A.DSLOGINN1,
sum(CASE WHEN A.Pontuacao <= 0 THEN 0 ELSE A.Pontuacao END) AS VrPTS

FROM WORK.TbTabelaPontos A

/*where NUANOMES = &Base_Mes.*/

group by 1,2
order by 2,1

;RUN;

/*SUBSTITUI TUDO QUE FOI "MISSING OU '.' POR '0'*/
/*ALGUNS CASOS ESTÃO "VAZIOS" AO FINALIZAR O SCRIPT*/

data work.TbPts_Crescimento_2;
	set work.TbPts_Crescimento_1;

	array num_array _numeric_;
	do over num_array;
		if missing(num_array) then num_array = 0;
	end;
run;
