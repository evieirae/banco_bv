proc sql;
	create table tempw.Privado_Log_1 as
	SELECT
		put(stproposta.dtstatusproposta_ddmmyyy,mmyys10.) AS mesano_log,
		stproposta.stproposta,
		stproposta.nuproposta,
		Proposta.NuCpfCnpj_SobAnalise AS NuCpfCnpj,
		Proposta.SistemaOrigem,
		case when proposta.nmpessoa = 'PAKETA SERVICOS FINANCEIROS SA' then 'Paketa' else 'Outros' end as coban,
		case when proposta.convenio  = 'PAKETA SERVICOS FINANCEIROS SA' then 'Paketa' else 'Outros' end as convenio
	from
		(select * FROM tempw.PRIVADO_LEADTIME
			WHERE stproposta = 'AV') stproposta
				INNER JOIN tempw.CONSIGNADO_PRIVADO Proposta
					ON stproposta.nuproposta = Proposta.nuproposta
/*and stproposta.dtsituacaoaverbacao between  &database_ini. and &database_fim.*/;
RUN;

proc sql;
	create table tempw.Privado_Log_2 as

	SELECT
		mesano_log,
		stproposta,
		SistemaOrigem,
		coban,
		convenio,
		count(nuproposta) as ttl_proposta,
		count(distinct(NuCpfCnpj)) as ttl_cpf
	from tempw.Privado_Log_1
		group by mesano_log, stproposta, SistemaOrigem,	coban, convenio;
RUN;

PROC SQL;
	CREATE TABLE tempw.Privado_Log_3 AS
		SELECT mesano_log AS mesano_log,
			stproposta,
			SistemaOrigem,
			coban,
			convenio,
			ttl_proposta,
			ttl_cpf
		FROM tempw.PRIVADO_LOG

			UNION

		SELECT t1.mesano_log as mesano_log,
			t1.stproposta,
			t1.SistemaOrigem,
			t1.coban,
			t1.convenio,
			/* SUM_of_ttl_proposta */
	(SUM(t1.ttl_proposta)) FORMAT=11. AS ttl_proposta,
	/* SUM_of_ttl_cpf */
	(SUM(t1.ttl_cpf)) FORMAT=11. AS ttl_cpf
FROM tempw.PRIVADO_LOG_2 t1
GROUP BY t1.mesano_log, t1.stproposta, t1.SistemaOrigem, t1.coban, t1.convenio;
RUN;

%let dir_csv = /riscosbvf/BA/processos/output;
