create table CARTEIRA_CL_MELIUZ as
    select distinct
	input(a.nr_cpfcnpj,BEST22.) 					as nr_cpfcnpj

	,input(a.nr_contrato,BEST32.) 					as nr_contrato format=best32.
	,a.cd_regional
	,input(a.nr_grupo,BEST22.) 						as nr_grupo
	,a.cd_novo_status
	,input(a.vl_divida,BEST9.) 						as vl_divida
	,input(a.vl_nova_divida,BEST9.) 				as vl_nova_divida
	,datepart(a.dt_disponibilizacao) format date9. 	as dt_disponibilizacao
	,datepart(a.dt_nova_disponib) format date9. 	as dt_nova_disponib
	,mdy(input(SUBSTR(a.dtenquad,5,2),best8.),input(SUBSTR(a.dtenquad,7,2),best8.),input(SUBSTR(a.dtenquad,1,4),best8.)) format date9. as dtenquad
	,input(a.nr_dias_atraso,BEST9.) 				as nr_dias_atraso
	,a.dmque
	,input(a.cd_assessoria,BEST9.) 					as cd_assessoria
	,input(a.cd_nova_assessoria,BEST9.) 			as cd_nova_assessoria

from connection to perm (
	select distinct
	    nr_contrato
        ,nr_cpfcnpj
        ,cd_regional
        ,nr_grupo
		,cd_novo_status
        ,vl_divida
        ,vl_nova_divida
        ,dt_disponibilizacao
        ,dt_nova_disponib
        ,dtenquad
        ,nr_dias_atraso
        ,dmque
        ,cd_assessoria
        ,cd_nova_assessoria

	from bv-pdbd-prd.trusted_rcvry.tb_historico_disponibilizacao
	where dmque in ('CL','CL1','CL2','CL3','CL4','CL5','CL6')

) as a

where length(a.nr_contrato) > 13


order by nr_contrato, dt_disponibilizacao
;disconnect from perm;

quit;

data confere;
set CARTEIRA_CL_MELIUZ;
where dtenquad = .;
run;
