PROC SQL;
CREATE TABLE DEPARA_PDV_FILIAL AS
SELECT
A.CdParceiroComercial,
NmAutoShopping,
COALESCE(NmAreaEstruturaComercialC,'Desconhecido') as Filial,
CASE WHEN A.NmSubEstruturaComercialC = "" THEN 'Canal Rodas e EGV - Regional II' else A.NmSubEstruturaComercialC end as Regional2,
COALESCE(AGRUPAMENTO_REGIAO,'Desconhecido') as Regional,
COALESCE(FlPorteA,'Desconhecido') as PortePDV,
COALESCE(C.DV,'Desconhecido') as DV,
DsLoginN1C as GR

FROM PROJETOS.TBPARCEIROCOMERCIAL AS A
LEFT JOIN BASEGRC.TBPORTEPDV AS B ON A.CdParceiroComercial = B.CdParceiroComercial
LEFT JOIN PRICING.AGRUPAMENTO_REGIAO_PRICING AS C ON A.NmAreaEstruturaComercialC = C.Filial
WHERE A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.FlAtivo = 'S' and B.NuAno = 2026
group by 1,2,3,4,5,6,7,8;
quit;

'Canal Rodas e EGV - Regional III', 'Canal Rodas e EGV - Regional IV', 'Canal Rodas e EGV - Regional V','Canal Rodas e EGV - Regional VI')

PROC SQL;
CREATE TABLE MSG AS SELECT
CdSimulacaoInicial,
NuProposta,

CASE WHEN Mensageria in ('0. Aprovado','0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%')
	      and Mensageria2 not in ('3. Recusado', '4. Recusado Reversível') then 'Mensageria Positiva'
	 WHEN Mensageria in ('3. Recusado','6. Recusado') or
	      Mensageria in ('0. Aprovado','0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%') and Mensageria2 in ('3. Recusado', '4. Recusado Reversível') then 'Recusado'
	 else 'Demais Msgs'
end as AGRUPAMENTO_MSG,

CASE WHEN A.ProbabilidadeAprovacao <= 0.5 THEN 'ZERO'
     WHEN A.ProbabilidadeAprovacao <= 4.5 THEN 'ATÉ 4,5'
	  WHEN A.ProbabilidadeAprovacao <= 5.5 THEN '4,75-5,5'
	  WHEN A.ProbabilidadeAprovacao <= 7.5 THEN '5,75-7,5'
	  WHEN A.ProbabilidadeAprovacao <= 8.5 THEN '7,75-8,5'
	  WHEN A.ProbabilidadeAprovacao <= 9.8 THEN '9,25-9,75'
	  WHEN A.ProbabilidadeAprovacao <= 10 THEN '10'
	  ELSE 'ZERO' END AS AGRUPAMENTO_RTG_SIMPLES_V2

from INFCVS.TBSIMULACOES AS A
WHERE NuAnoMes >= 202502 and Produto in ('Leves')  and Canal = 'MULTIMARCAS' and SistemaInicial = 'PPAR' and TpPessoa = 'PF'
group by 1,2,3,4;
quit;
