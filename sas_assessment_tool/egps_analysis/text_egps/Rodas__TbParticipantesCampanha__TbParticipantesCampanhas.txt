%include '/infogrc/utils/all-includes.sas';
LIBNAME Porte '/infogrc/Porte_PDV';
LIBNAME BASESGER '/riscosbvf/BA/bases_gerenciais';

/* CONSULTA O PORTE ATUAL */
PROC SQL;
SELECT DISTINCT Max(NuAnoMesAtu) AS NuAnoMesAtu INTO: NuAnoMesAtu
FROM Porte.Base_PortePDV_Hist;
QUIT;

/* CONSULTA PARCEIRO COMERCIAL */
PROC SQL;
CREATE TABLE Base_ParceiroComercial AS
SELECT DISTINCT A.*, PORTE.FlPorteNovo AS Porte
FROM Bases.TbParceiroComercial AS A
LEFT JOIN Porte.Base_PortePDV_Hist AS Porte
  ON A.CdParceiroComercial = Porte.CdParceiroComercial
  AND Porte.NuAnoMesAtu = &NuAnoMesAtu.
WHERE
  A.CdTipoParceiroComercial in (2,5,12) AND
  A.NmCanalEstruturaComercialC in ('Canal Veiculos', 'Negocios Centralizados', 'Canal Negocios') AND
  A.NmParceiroComercial NOT CONTAINS ('%BV BALCA%') AND
  A.NmParceiroComercial NOT CONTAINS ('%BV MES%') AND
  A.NmParceiroComercial NOT CONTAINS ('%LOJA RENEG%') AND
  A.NmColaboradorC NOT CONTAINS ('%OPERADO%');
QUIT;

/* CRIA A BASE DE PARCEIROS PARTICIPANTES */
PROC SQL;
CREATE TABLE TbParticipantesCampanhas AS
SELECT
  B.NuAnoMes AS AnoMes,
  B.CdParceiroComercial AS Codigo,
  A.NmParceiroComercial AS Parceiro,
  strip(put(B.CdParceiroComercial,10.))||'-'||strip(A.NmParceiroComercial) AS Loja,
  B.NmCampanha AS Campanha,
  B.NmGrupo AS Grupo,
  B.NmGrupoProdutoCampanha AS Produto,
  B.CdProduto AS CdProduto,
  A.NmCanalEstruturaComercialC AS Canal,
  A.NmSubEstruturaComercialC AS Regional,
  A.NmAreaEstruturaComercialC AS Filial,
  A.DsLoginN1C AS LoginN1,
  A.NmColaboradorC AS Colaborador,
  A.NuCnpjLoja AS CNPJ,
  C.FlPorteNovo AS Porte,
  A.NmTipoParceiroComercial,
  A.DsLoginN1C,
  A.DsLoginN2C,
  A.DsLoginN3C,
  A.LstSubstituicaoC
FROM Base_ParceiroComercial AS A
LEFT JOIN BASESGER.TbBvRelacionar AS B
  ON B.CdParceiroComercial = A.CdParceiroComercial
LEFT JOIN Porte.Base_PortePDV_Hist AS C
  ON A.CdParceiroComercial = C.CdParceiroComercial
  AND C.NuAnoMesAtu = &NuAnoMesAtu.
WHERE
  B.NmGrupo = 'Grupo Ação' AND
  B.NmTipoEvento = 'BV Relacionar' AND
  B.NmCampanha <> '' AND
  B.NuAnoMes > 202312
ORDER BY B.NuAnoMes;
QUIT;

/* EXPORTA A TABELA FINAL */
PROC EXPORT
  data=TbParticipantesCampanhas
  file='/riscosbvf/BA/processos/output/TbParticipantesCampanhas.csv'
  replace
  dbms=dlm;
  delimiter=';';
RUN;
