SELECT

CASE WHEN A.NmAreaEstruturaComercialC in ('Canal ABC','Canal Freguesia','Canal Guarulhos','Canal Ipiranga','Canal Morumbi','Canal Osasco','Canal Santos','Canal Tatuapé') THEN 'C1R1'

	 ELSE 'DEMAIS' END AS Regional,

CASE WHEN A.DsTipoCanalProducao in ('MULTIMARCAS', 'CONCESSIONARIAS') THEN 'MM e CC'
     ELSE 'DEMAIS CANAIS' END AS FG_CANAL,
SUM(A.VpTxGerencial) as TaxaNet,
SUM(A.VpTirContabil) as TirC,
SUM(A.VpSpreadContabil) as SpreadNet,
