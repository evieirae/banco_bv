	%do i = 1 %to 84;

%macro bases(DtInicioContrato);
	DATA aprovacao (drop=old); SET BaseFinal (rename=(GH = old)); GH = old; RUN;
	DATA aprovacao_2 (drop=old); SET BaseFinal (RENAME=(contratos=contratos_ref Inad30=Inad30_ref GH = old));
		GH = old; IF carteira = "&DtInicioContrato."; RUN;

	proc sql;
	create table aprovacao_3 as
	select distinct a.*,
		b.contratos_ref,
		abs(b.contratos_ref - a.contratos) as contratos_ks_1,
		(a.contratos - b.contratos_ref)* log(a.contratos/b.contratos_ref) as contratos_psi,
		b.Inad30_ref,
		abs(b.Inad30_ref - a.Inad30) as Inad30_ks_1,
		case when b.Inad30_ref <= 0.1 then abs(((log(a.Inad30) - log(b.Inad30_ref))/log(b.Inad30_ref))*(a.total_gh/a.total))
		else abs(((a.Inad30 - b.Inad30_ref)/b.Inad30_ref)*(a.total_gh/a.total)) end as mape,
		abs(((a.Inad30 - b.Inad30_ref)/b.Inad30_ref)*(a.total_gh/a.total))  as mape_semlog
	from aprovacao a
		left join aprovacao_2 b on a.GH = b.GH and a.fl_50k = b.fl_50k
	;quit;

	proc sql;
	create table aprovacao_4 as
	select distinct
		ProdutoAgrupado
		,carteira
		,fl_50k
		,max(contratos_ks_1) as max_contratos_ks_1
		,sum(contratos_psi) as sum_contratos_psi
		,max(Inad30_ks_1) as max_Inad30_ks_1
		,sum(mape) as sum_mape
		,sum(mape_semlog) as sum_mape_semlog
	from aprovacao_3 group by 1,2,3
	;quit;

	proc sql;
	create table aprovacao_5 as
	select distinct a.*,
		b.max_contratos_ks_1,
		b.sum_contratos_psi,
		b.max_Inad30_ks_1,
		b.sum_mape
	from aprovacao_3 a left join aprovacao_4 b on a.carteira = b.carteira and a.fl_50k = b.fl_50k
	;quit;

	proc sql;
	create table aprovacao_5_1 as select *, &DtInicioContrato. AS carteira_REF from aprovacao_5; quit;

	proc sql;
	create table aprovacao_6_1 as
	SELECT distinct ProdutoAgrupado ,carteira ,ks_contratos_ref
	FROM (
		select ProdutoAgrupado
			,carteira
			,GH as ks_contratos_ref
			,case 
				  when carteira = carteira_REF THEN 0
				  when max_contratos_ks_1 = contratos_ks_1 and max_contratos_ks_1 is not null then 1
				  else 0 end as fl_ks_contratos
		from aprovacao_5_1
		) WHERE fl_ks_contratos = 1
	;quit;

	proc sql;
	create table aprovacao_6_2 as
	SELECT distinct ProdutoAgrupado,carteira ,ks_inad_ref
	FROM (
		select
			ProdutoAgrupado
			,carteira
			,GH as ks_inad_ref
			,case 
				  when carteira = carteira_REF THEN 0
				  when max_Inad30_ks_1 = Inad30_ks_1 and max_Inad30_ks_1 is not null then 1 else 0 end as fl_ks_inad
		from aprovacao_5_1
	) WHERE fl_ks_inad = 1
	;quit;

	proc sql;
	create table aprovacao_6 as
		SELECT distinct a.ProdutoAgrupado, a.carteira, a.ks_contratos_ref, b.ks_inad_ref
		from aprovacao_6_1 a inner join aprovacao_6_2 b
		on (a.carteira = b.carteira)
	;quit;

	proc sql;
	create table aprovacao_7 as
		select  distinct a.*, b.ks_contratos_ref, b.ks_inad_ref
		from aprovacao_5 a left join aprovacao_6 b
		on a.carteira = b.carteira
	;quit;
%mend;

%macro variaveis(PRODUTO, PROD, BASE, REF);

proc sql;
    create table variaveisCarteira as select distinct
    &PRODUTO. as produto,
	DtCorte,
    Fl_50K,
    variavel,
    quebras,
    qtd_bons,
    qtd_maus,
    sum(qtd_contas) as total,
    sum(Qtd_Bons) as total_Bom,
    sum(Qtd_Maus) as total_Mau ,
    calculated total_Mau / sum( calculated total_Bom, calculated total_Mau) as ever90m12
    from &BASE.
    group by 1,2,3
    ;quit;

    proc sql;
    create table variaveis
    as
    select
    distinct
    a.*,
    b.total,
    a.qtd_contas/b.total as porc,
    b.Qtd_Bons/b.total_Bom as porcBom,
    b.Qtd_Maus/b.total_Mau as porcMau
    from
    &BASE. a
    left join
    variaveiscarteira b
    on
    a.variavel = b.variavel
    and
    a.DtCorte = b.DtCorte
    and a.quebras = b.quebras
    and a.Fl_50K = b.Fl_50K
    ;quit;

    create table variaveisRef
    as
    select
    distinct
    &PRODUTO. as produto,
    variavel,
    DtCorte,
    Fl_50K,
    quebras,
    porc as porcRef
    from variaveis where DtCorte = &REF.
    ;quit;

    a.*,
    b.porcRef
    from variaveis a

    as

    *,
    case when porcMau = . then . else

    from variaveis2
    group by
     DtCorte, Fl_50K, variavel

    variavel,

    ;quit;

		abs(b.Inad30_ref - a.Inad30) as Inad30_ks_1,
		case when b.Inad30_ref <= 0.1 then abs(((log(a.Inad30) - log(b.Inad30_ref))/log(b.Inad30_ref))*(a.total_gh/a.total))
		else abs(((a.Inad30 - b.Inad30_ref)/b.Inad30_ref)*(a.total_gh/a.total)) end as mape

		left join aprovacao_2_2 b on a.GH = b.GH and a.Fl_50K = b.Fl_50K
	;quit;
