/* Buscar dados de mensageria */
PROC SQL;
CREATE TABLE MsgProposta AS
  SELECT DISTINCT
    Num_Proposta AS NuProposta,
    CASE WHEN Msg_Dash = '9. Sem CdSimuPrazofnnco' THEN '9. Não houve mensageria' ELSE Msg_Dash END AS Mensageria,
    Msg_2 AS Mensageria2
  FROM CRM.SIMU_PROP_AQUISICAO
  WHERE
    Num_Proposta <> .
    AND (calculated Mensageria || Msg_2) IS NOT NULL;
RUN;

PROC SORT DATA=MsgProposta NODUPKEY;
BY NuProposta;
RUN;

/* Buscar informações de leads  */
PROC SQL;
CREATE TABLE Leads AS
  SELECT DISTINCT
    num_proposta as NuProposta,
    Sistema_Origem as CompraLeads,
	CASE WHEN flag_comissao = 1 AND Sistema_Origem IN ('ICARROS','MERCADO_LIVRE','MOBIAUTO','USADOSBR') THEN 1
		 WHEN Sistema_Origem IN ('SITE_BV','MEU_CARRO_NOVO','APP','NA_PISTA') THEN 1
		 ELSE 0
    END AS FlagProducao,
	campanha_agrup as OrigemSiteBV
  FROM
    CRM.SIMU_PROP_LEADS
  WHERE
    num_proposta <> .
    AND Sistema_Origem IS NOT NULL;
RUN;

PROC SORT DATA=Leads NODUPKEY;
BY NuProposta;
RUN;

PROC SQL; CREATE INDEX NuProposta ON Leads; QUIT;
