create table tb_gestao_contabil as
select distinct nucontrato,

			d.dtentradaproposta,
			g.TPPESSOA,

CREATE TABLE BASE_ASSESSORIA AS
SELECT

A.*
,input(sqcontat, 2.) as SQCONTRATOFINANCEIRO format 2.
,DATEPART(A.DT_ASSES) FORMAT DATE9. AS DATA_ENTRADA
,DATEPART(A.DT_RET_ASSES) FORMAT DATE9. AS DATA_SAIDA
,CASE WHEN COD_ASSES IS MISSING THEN 'VERIFICAR'
            WHEN COD_ASSES IN (1, 1530,1524,1634,1648,1649) THEN 'Toledo'
	        WHEN COD_ASSES IN (1532,739,1666,1443) THEN 'Oliveira'

			WHEN COD_ASSES IN (1563) THEN 'LM Angelis'
			WHEN COD_ASSES IN (971, 1336, 1593,1592,1595, 2048, 2049) THEN 'Paschoalotto' ELSE 'Outras' END AS ASSESSORIA

FROM CONNECTION TO PERM (

WITH QUERY_1 AS (

SELECT DISTINCT
    b.dt_disponibilizacao AS DT_ASSES,
    b.dt_nova_disponib AS DT_RET_ASSES,
    cast(b.nrpaanat AS INT64) AS PARC_ENVIADA,
    CAST(b.nr_contrato AS INT64) AS NUCONTRATOGESTAO,
    b.sqcontat,
    cast(b.cd_assessoria as int64) AS COD_ASSES
FROM bv-pdbd-prd.trusted_rcvry.tb_historico_disponibilizacao AS B
LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato AS C
ON b.NR_CONTRATO = c.cd_contrato
WHERE CAST(c.cd_produto AS INT64) IN (13)

)
SELECT * FROM QUERY_1) A

GROUP BY NUCONTRATOGESTAO
HAVING DT_ASSES = MIN(DT_ASSES)

;DISCONNECT FROM PERM;
