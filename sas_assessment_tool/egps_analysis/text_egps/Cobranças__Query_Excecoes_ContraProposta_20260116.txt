PROC SQL;
   CREATE TABLE WORK.QUERY_FOR_BASE_FINAL_0000_0000 AS
   SELECT t1.Publico,
          
            (case
            when t1.ATRASO between 75 and 120  then '75 a 120'
            when t1.ATRASO between 121 and 180  then '121 a 180'
            when t1.ATRASO between 181 and 360  then '181 a 360'
            when t1.ATRASO between 361 and 720  then '361 a 720'
            when t1.ATRASO > 720  then 'Acima 720'
            else 'validar'
            end) AS FAIXA_ATRASO,
          t1.'APROVAÇÃO_STATUS_FINAL'n,
          
            (COUNT(t1.CONTRATO)) AS COUNT_of_CONTRATO
      FROM WORK.BASE_FINAL t1
      GROUP BY t1.Publico,
               (CALCULATED FAIXA_ATRASO),
               t1.'APROVAÇÃO_STATUS_FINAL'n
      ORDER BY t1.'APROVAÇÃO_STATUS_FINAL'n,
               FAIXA_ATRASO;
QUIT;
