Proc Sql;
Create table Base_Producao_PDVs_NI as
Select Distinct
Producao.NuAnoMes
,Producao.NuCnpjLojaFavorecido
,Producao.NmParceiroComercialFavorecido
,Sum(Producao.VrProducao) as VrProducao
,Sum(Producao.VrBvMais) as Premio_SPF
,Sum(Producao.VrCAP) as Premio_CAP
,Sum(Producao.VrPlacaSolar) as Premio_PlacaSolar
,(Calculated Premio_SPF + Calculated Premio_CAP + Calculated Premio_PlacaSolar) as VrPremio_Seguros
,Participantes.NuAnomes as NuAnomes_Campanha
,Participantes.DsLoginN1
,Participantes.Grupo
,Case
	When Producao.NuAnoMes Between Participantes.Anomes_Ini_Prod and Participantes.Anomes_Fim_Prod Then 1
	Else 0
 End as Producao_Trim

,Case
	When Producao.NuAnoMes = Participantes.NuAnomes Then Participantes.NuAnomes
	Else 0
 End as Real_Mes_Camp
From tmp.TBPRODUCAO as Producao
Inner join
		(Select Distinct CNPJ, NuAnomes, Grupo, DsLoginn1, Anomes_Ini_Prod, Anomes_Fim_Prod From Base_PDV_Participantes) as Participantes
 	   	On Producao.NuCnpjLojaFavorecido = Participantes.CNPJ
Where Producao.NuAnomes >= &Anomes. and Producao.NmGrupoProdutoMeta = 'Solar' and Producao.NmAreaEstruturaComercialC NE ''
Group By 1,2,3,4,5
;Quit;


Proc Sql;
Create Table Base_Meta_Campanha_Solar as
Select
NuAnomes_Campanha
,Grupo
,DsLoginN1
,Sum(VrProducao) as VrProducao_Trim Format Commax12.2
,Sum(VrProducao)/3 as Media_Producao_Trim Format Commax12.2
,Case
	When Calculated Media_Producao_Trim < 400000 Then 400000 Else Calculated Media_Producao_Trim
End as Meta_Producao_Final Format Commax12.2
,Sum(VrPremio_Seguros) as VrPremio_Seguros_Trim
,Sum(VrPremio_Seguros) / Sum(VrProducao) as PenFin_Seguros Format Percent10.2
,Case
	When Sum(VrPremio_Seguros) / Sum(VrProducao) < 0.03 Then 0.03 Else Sum(VrPremio_Seguros) / Sum(VrProducao)
End as PenFin_Seguros_Final Format Percent10.2

From Base_Producao_PDVs_NI
Where Producao_Trim = 1 and Grupo in ('Grupo 1', 'Grupo 2')
Group by 1,2,3
;Quit;
