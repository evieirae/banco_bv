CREATE TABLE BASE_FPD AS
SELECT *

FROM CONNECTION TO PERM(

    cd_contrato,
    sq_contrato_financeiro,
    dt_vencimento,

    st_parcela,

        WHEN (DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY)) > 5
        OR (st_parcela = 'A' AND dt_ultimo_recebimento IS NULL) THEN 1
        ELSE 0
    END AS fl_inad5,
    CASE
        WHEN (DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY)) > 7
        OR (st_parcela = 'A' AND dt_ultimo_recebimento IS NULL) THEN 1
        ELSE 0
    END AS fl_inad7,
    CASE
        WHEN (DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY)) > 15
        OR (st_parcela = 'A' AND dt_ultimo_recebimento IS NULL) THEN 1
        ELSE 0
    END AS fl_inad15,
    CASE
        WHEN (DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY)) > 30
        OR (st_parcela = 'A' AND dt_ultimo_recebimento IS NULL) THEN 1
        ELSE 0
    END AS fl_inad30,
    CASE
        WHEN DATE_ADD(dt_vencimento, INTERVAL 30 DAY) < CURRENT_DATE() THEN 1
        ELSE 0
    END AS Fl_vencido30,
	CASE
        WHEN DATE_ADD(dt_vencimento, INTERVAL 5 DAY) < CURRENT_DATE() THEN 1
        ELSE 0
    END AS Fl_vencido5

    bv-pdbd-prd.dvry_core.tbparcela
WHERE
    nu_parcela = 1

);
