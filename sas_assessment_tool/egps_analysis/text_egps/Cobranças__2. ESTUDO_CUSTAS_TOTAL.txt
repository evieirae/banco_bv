%macro SAFRAS_ESTOQUE_AJU(SAFRA,dt_fim,dt_gestao_cob);
PROC SQL;
%CONNECT_GCP;
CREATE TABLE DESPESAS_TODAS_&SAFRA. AS
SELECT * FROM CONNECTION TO perm (
    SELECT
        A.NuContratoGestao,
        A.SqContratoFinanceiro,
		A.num_processo,
        A.cod_desp,
        A.obs_desp,
        A.cod_oper,
        A.VALOR_DESP,
        DATE(A.Dt_DESP) AS Dt_DESP,

        DATE_DIFF(
            DATE(A.Dt_DESP),
            MIN(CASE WHEN (B.St_Parcela <> 'Q' OR
                           (B.St_Parcela = 'Q' AND DATE(A.Dt_DESP) <= DATE_ADD(DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO), INTERVAL 1 DAY)))
                     THEN DATE(B.Dt_Vencimento) ELSE NULL END),
            DAY
        ) AS Atraso,
        MIN(CASE WHEN (B.St_Parcela <> 'Q' OR
                       (B.St_Parcela = 'Q' AND DATE(A.Dt_DESP) <= DATE_ADD(DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO), INTERVAL 1 DAY)))
                 THEN B.Nu_parcela ELSE NULL END) AS Parcela

    FROM (
        SELECT
            A.NuContratoGestao,
            A.SqContratoFinanceiro,
            DATE(A.Dt_DESP) AS Dt_DESP,
            A.cod_desp,
            A.obs_desp,
            A.cod_oper,
			A.num_processo,
            SUM(SAFE_CAST(A.valor_desp AS FLOAT64)) AS VALOR_DESP
        FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil D
		INNER JOIN bv-pdbd-prd.trusted_scc_fin.saj_desp_contrato A
		ON A.NuContratoGestao = D.NUCONTRATO
		AND A.SqContratoFinanceiro = D.SQCONTRATOFINANCEIRO
		WHERE D.DTPROCESSAMENTO = &dt_gestao_cob.
        AND A.dt_desp <= &dt_fim.
        GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP, A.cod_desp, A.obs_desp, A.cod_oper, A.num_processo
    ) AS A
    LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B
        ON A.NuContratoGestao = B.cd_contrato
       AND A.SqContratoFinanceiro = B.Sq_Contrato_Financeiro
    GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP, A.VALOR_DESP, A.cod_desp, A.obs_desp, A.cod_oper, A.num_processo
);

CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_2 AS
SELECT
    NUCONTRATO,
    sqcontratofinanceiro,
    QTD_PARCELAS,
    QTD_PARCELAS_PAGAS,
    
    ROUND(QTD_PARCELAS_PAGAS / QTD_PARCELAS, 0.01) AS pc_amort_parc,

    
    CASE
      WHEN CALCULATED pc_amort_parc = 0                                  THEN 'a. FPD'
      WHEN CALCULATED pc_amort_parc  > 0  AND CALCULATED pc_amort_parc <= 0.10 THEN 'b. até 10%'
      WHEN CALCULATED pc_amort_parc  > 0.10 AND CALCULATED pc_amort_parc <= 0.30 THEN 'c. 10% a 30%'
      WHEN CALCULATED pc_amort_parc >= 0.31 AND CALCULATED pc_amort_parc <= 0.70 THEN 'd. 31% a 70%'
      WHEN CALCULATED pc_amort_parc  > 0.70                               THEN 'e. >70%'
      ELSE 'N/A'
    END AS fx_pc_amort_parc
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_1
;
QUIT;

%macro SAFRAS_HIST_DISTR(SAFRA,dt_fim,dt_fim2);
	PROC SQL;
connect to bigquery as perm (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage);
	CREATE TABLE CONSULTA_HIST_DISTR_&SAFRA. AS SELECT
	NuContratoGestao
	,SqContratoFinanceiro
	,Dt_DESP
	,PARCELA
	,dmque3
	,nrproc
	,DT_DISPONIBILIZACAO
	,cd_nova_assessoria
	,&SAFRA. AS SAFRA
FROM
	CONNECTION TO perm	(

	SELECT * FROM (
			SELECT	A.NuContratoGestao,
					A.SqContratoFinanceiro,
					A.VRAJUIZ,
					A.num_processo,
					DATE(A.Dt_DESP) AS Dt_DESP,
					DATE_DIFF(Dt_DESP, MIN(CASE WHEN	B.St_Parcela <> 'Q' OR
												(B.St_Parcela = 'Q' AND
												B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN Dt_Vencimento ELSE NULL END), DAY) AS Atraso,
					SUM(CASE WHEN	B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN Vr_Parcela ELSE 0 END) AS Risco_Aberto,
					COUNT(CASE WHEN	B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN Vr_Parcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
					SUM(CASE WHEN	(B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO) >= Dt_DESP)) AND
									DATE(B.Dt_Vencimento) < A.Dt_DESP THEN Vr_Parcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
					COUNT(CASE WHEN	(B.St_Parcela <> 'Q' OR
									(B.St_Parcela = 'Q' AND
									DATE(B.DT_PROCESSAMENTO_ULT_RECEBIMENTO) >= Dt_DESP)) AND
									DATE(B.Dt_Vencimento) < A.Dt_DESP THEN Vr_Parcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
					MIN(CASE WHEN	B.St_Parcela <> 'Q' OR
												(B.St_Parcela = 'Q' AND
												B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN Dt_Vencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,
					MIN(CASE WHEN	B.St_Parcela <> 'Q' OR
												(B.St_Parcela = 'Q' AND
												B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN Nu_parcela ELSE NULL END) as PARCELA,
					MIN(CASE WHEN	B.St_Parcela <> 'Q' OR
												(B.St_Parcela = 'Q' AND
												B.DT_PROCESSAMENTO_ULT_RECEBIMENTO >= Dt_DESP)  THEN DT_PROCESSAMENTO_ULT_RECEBIMENTO ELSE NULL END) as DT_PROCESSAMENTO_ULT_RECEBIMENTO
			FROM	(

SELECT NuContratoGestao,
					SqContratoFinanceiro,
					Num_processo,
					DATE(Dt_DESP) AS Dt_DESP,
					sum(SAFE_CAST(valor_desp AS FLOAT64)) AS VRAJUIZ
			FROM	bv-pdbd-prd.trusted_scc_fin.saj_desp_contrato
			WHERE	dt_desp > '2021-01-01 00:00:00' and dt_desp < &dt_fim. and cod_desp in (14,80)
			GROUP BY NuContratoGestao, SqContratoFinanceiro, Dt_DESP, Num_processo

			) AS A
					LEFT JOIN bv-pdbd-prd.dvry_core.tbparcela AS B ON	A.NuContratoGestao = B.cd_contrato AND
														A.SqContratoFinanceiro = B.Sq_Contrato_Financeiro
			GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP, A.VRAJUIZ, A.num_processo
	) AS A
LEFT JOIN (SELECT nr_contrato, sqcontat, dmque3, nrproc, dt_nova_disponib, cd_nova_assessoria, DT_DISPONIBILIZACAO, nrpaanat
FROM bv-pdbd-prd.trusted_rcvry.tb_historico_disponibilizacao) B
ON	A.NuContratoGestao = B.nr_contrato
AND A.SqContratoFinanceiro = SAFE_CAST(B.sqcontat AS INT64)
AND A.PARCELA = SAFE_CAST(B.nrpaanat AS INT64)
)
	;QUIT;

;%MEND;

	   WHEN DATEPART(B.DT_DISPONIBILIZACAO) between inputn(&dt_fim1., 'YYMMDD10.') and inputn(&dt_fim3., 'YYMMDD10.') THEN 1

		    WHEN Recebimento > Desp_Ajuizamento THEN 0
			WHEN Desp_Ajuizamento > 0 THEN ROUND( ((Desp_Ajuizamento - Recebimento) / Desp_Ajuizamento) * 100 , 1)
