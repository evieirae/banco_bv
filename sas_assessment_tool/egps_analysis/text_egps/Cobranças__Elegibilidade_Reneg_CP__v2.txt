create table CARTEIRA_BASE as
select f.*, g.*, h.*
from connection to perm (
    SELECT
        a.dtprocessamento,
        a.nucontrato,
        a.sqcontratofinanceiro,
        b.cd_cpf_cnpj,
        b.cd_produto,
        c.CdSubProduto,
        a.CdModalidadeProduto,
        d.tppessoa,
        a.nudiasatraso,
        a.vrpdd,
        a.CdRatingAtual,
        a.vrprincipal,
        202512 as safra_cob,
        a.VrPermanenciaApropriado,
        a.VrTaxaContrato,
        a.CdSituacaoContabil,
        e.tx_juros
    FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil a
    JOIN bv-pdbd-prd.dvry_core.tbcontrato b ON a.nucontrato = b.cd_contrato
    LEFT JOIN bv-pdbd-prd.trusted_dbcred.tbproposta c ON a.nucontrato = c.nucontratogestao
    LEFT JOIN bv-pdbd-prd.trusted_dbcor.tbpessoa d ON b.cd_cpf_cnpj = d.nucpfcnpj
    LEFT JOIN bv-pdbd-prd.dvry_core.tbcontratofinanceiro e
        ON a.nucontrato = e.cd_contrato
        AND a.sqcontratofinanceiro = e.sq_contrato_financeiro
    WHERE a.dtprocessamento = (
        SELECT MAX(dtprocessamento)
        FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
    )
    AND b.cd_produto IN ('10')
) as f

left join connection to perm (
    SELECT
        a.cd_contrato,
        a.sq_contrato_financeiro,
        COUNT(nu_parcela) AS Qtde_parcelas,
        SUM(CASE
            WHEN st_parcela = 'A' OR DATE(dt_ultimo_recebimento) >= &dia. THEN 1
            ELSE 0
        END) AS Qtde_Parc_aberto,
        SUM(CASE
            WHEN st_parcela = 'Q' AND DATE(dt_ultimo_recebimento) < &dia. THEN 1
            ELSE 0
        END) AS Qtde_Parc_pagas,
        MIN(CASE
            WHEN (st_parcela = 'A' OR DATE(dt_ultimo_recebimento) >= &dia.)
                 AND Dt_Vencimento < &dia.
            THEN nu_parcela
        END) AS parc_enviada,
        SUM(CASE
            WHEN st_parcela = 'A' OR DATE(dt_ultimo_recebimento) >= &dia.
            THEN Vr_parcela ELSE 0
        END) AS risco,
        SUM(CASE
            WHEN st_parcela = 'Q' OR DATE(dt_ultimo_recebimento) < &dia.
            THEN Vr_parcela ELSE 0
        END) AS risco_quitado,
        SUM(CASE
            WHEN (st_parcela = 'A' OR DATE(dt_ultimo_recebimento) >= &dia.)
                 AND Dt_Vencimento < &dia.
            THEN 1 ELSE 0
        END) AS Qtde_Parc_atraso,
        MAX(Vr_parcela) AS Vr_parcela
    FROM bv-pdbd-prd.dvry_core.tbparcela a
    LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato b ON a.cd_contrato = b.cd_contrato
    WHERE b.Cd_Produto IN ('10')
    GROUP BY a.cd_contrato, a.sq_contrato_financeiro
) as g
on f.nucontrato = g.cd_contrato
and f.sqcontratofinanceiro = g.sq_contrato_financeiro

left join connection to perm (
    SELECT
        a.NuContratoGestao,
        a.SqContratoFinanceiro,
        a.dt_asses,
        a.dt_ret_asses,
        a.cod_asses,
        b.razao_nome,
        a.ATIVA_PRINC
    FROM bv-pdbd-prd.trusted_scc_fin.saj_processo a
    LEFT JOIN bv-pdbd-prd.trusted_scc_fin.saj_agente b ON a.cod_asses = b.cod_agente
    LEFT JOIN bv-pdbd-prd.dvry_core.tbcontrato c ON a.nucontratogestao = c.cd_contrato
    WHERE c.Cd_Produto IN ('10')
    AND a.dt_asses < (
        SELECT MAX(dtprocessamento)
        FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
    )
    AND (a.dt_ret_asses IS NULL OR a.dt_ret_asses > (
        SELECT MAX(dtprocessamento)
        FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
    ))
) as h
on f.nucontrato = h.NuContratoGestao
and f.sqcontratofinanceiro = h.SqContratoFinanceiro;

disconnect from perm;
