create table FATO_CONTABIL as
select
"Fato Contabil" as Base
,A.DtReferencia
,A.NuObjetoNegociacao
,A.NuObjetoNegociacaoOriginal
,A.TpContabil
,A.CdGrupoConciliacao
,A.CdModalidadeProduto
,A.NmModalidadeProduto
,A.CdVeiculoLegal
,A.CdChaveNegocio
,A.DtVencimentoOperacao
,A.CdClienteRisco
,sum(A.VrSaldoContaContabil) as Res_Total
from connection to dbcon
 (select
 		fc.DtReferencia
		,case when fc.CdGrupoConciliacao in (9, 10) and fc.CdChaveNegocio like '%|%'
			   then substring(fc.CdChaveNegocio, len(fc.CdChaveNegocio) - charindex('|',reverse(fc.CdChaveNegocio))+2)
				else trim(fc.NuObjetoNegociacao)
 		 end as NuObjetoNegociacao
		,fc.NuObjetoNegociacao as NuObjetoNegociacaoOriginal
		,fc.TpContabil
		,fc.CdGrupoConciliacao
		,ep.CdModalidadeProduto
		,ep.NmModalidadeProduto
		,fc.CdVeiculoLegal
		,fc.CdChaveNegocio
		,fc.VrSaldoContaContabil
		,fc.DtVencimentoOperacao
		,fc.CdClienteRisco

 from posicaoconsolidada.TbFatoResultadoContabil fc

 left join posicaoconsolidada.TbEstruturaProduto ep on fc.CdCnsldrEstruturaProduto = ep.CdCnsldrEstruturaProduto

 where fc.DtReferencia = &Dt_Data.
		 and fc.NuVersaoContabil = ( select max(NuVersaoContabil)
		                             from posicaoconsolidada.TbFatoResultadoContabil
		                             where DtReferencia = fc.DtReferencia
		                             		  and NuObjetoNegociacao = fc.NuObjetoNegociacao)

		 and (substr(fc.NuContaContabil, 1, 1) = '7' or substr(fc.NuContaContabil, 1, 1) = '8')
		 and fc.NuObjetoNegociacao <> '0'
		 and fc.VrSaldoContaContabil <> 0
	) A

group by
Base
,A.DtReferencia
,A.NuObjetoNegociacao
,A.NuObjetoNegociacaoOriginal
,A.TpContabil
,A.CdGrupoConciliacao
,A.CdModalidadeProduto
,A.NmModalidadeProduto
,A.CdVeiculoLegal
,A.CdChaveNegocio
,A.DtVencimentoOperacao
,A.CdClienteRisco
;
quit;

				else trim(fc.NuObjetoNegociacao)
 		 end as NuObjetoNegociacao

		,fc.CdGrupoConciliacao
		,ep.CdModalidadeProduto
		,ep.NmModalidadeProduto
		,fc.CdVeiculoLegal
		,fc.VrSaldoContaContabil

 from posicaoconsolidada.TbFatoResultadoContabil fc

 left join posicaoconsolidada.TbEstruturaProduto ep on fc.CdCnsldrEstruturaProduto = ep.CdCnsldrEstruturaProduto

 left join posicaoconsolidada.TbGrupo g on fc.cdgrupoconciliacao = g.CdGrupo

		 and fc.NuVersaoContabil = ( select max(NuVersaoContabil)
		                             from posicaoconsolidada.TbFatoResultadoContabil
		                             where DtReferencia = fc.DtReferencia
		                             		  and NuObjetoNegociacao = fc.NuObjetoNegociacao)

		 and substr(fc.NuContaContabil, 1, 1) = '1'
		 and fc.NuObjetoNegociacao <> '0'
		 and fc.VrSaldoContaContabil <> 0
		 and ep.CdModalidadeProduto = 214
		 and fc.CdVeiculoLegal not in (980, 316430, 758222) 
		 and fc.CdGrupoConciliacao not in (17) 
	)
;

options compress=binary reuse=yes replace symbolgen;
proc sql;
create table UNIAO_FC_FG as
select
	 Base
	,DtReferencia
	,NuObjetoNegociacao
	,TpContabil
	,CdGrupoConciliacao
	,CdModalidadeProduto
	,NmModalidadeProduto
	,CdVeiculoLegal

	,Flag_TIR_Taxa
	,case when Base = 'Fato Gerencial' then Res_Total else 0 end as Res_Gerencial
	,case when Base = 'Fato Contabil' then Res_Total else 0 end as Res_Contabil
from
(
	select
		 Base
		,DtReferencia
		,NuObjetoNegociacao
		,case when TpContabil = 'V' then 'P'
				when TpContabil = 'C' then 'A'
				else TpContabil
		 end as TpContabil
		,CdGrupoConciliacao
		,CdModalidadeProduto
		,NmModalidadeProduto
		,CdVeiculoLegal

		,Res_Total
		,0 as Flag_TIR_Taxa
	from FATO_CONTABIL
	where CdModalidadeProduto not in (625) 
		 	and CdVeiculoLegal not in (980,  758222) 
		 	and CdGrupoConciliacao not in (17) 

	union all
	select
		 Base
		,DtReferencia
		,NuObjetoNegociacao
		,TpContabil
		,CdGrupoConciliacao
		,CdModalidadeProduto
		,NmModalidadeProduto
		,CdVeiculoLegal

		,Res_Total
		,case when TxPreSpread = TxPreSpreadTIR then 1 else 0 end as Flag_TIR_Taxa
	from FATO_GERENCIAL
	where CdModalidadeProduto not in (625) 
			and CdVeiculoLegal not in (980) 
			and CdGrupoConciliacao not in (17) 
)
;
quit;

options compress=binary reuse=yes replace symbolgen;
proc sql;
create table FATO_CONTABIL_VS_FATO_GERENCIAL as
select
	 DtReferencia
	,NuObjetoNegociacao
	,TpContabil
	,CdGrupoConciliacao
	,CdModalidadeProduto
	,NmModalidadeProduto
	,CdVeiculoLegal

	,case when CdGrupoConciliacao = 11 then
		 case when NuObjetoNegociacao like '%-%'
				then cats(put(CdVeiculoLegal, best8.), '||', substr(NuObjetoNegociacao, 1,index(NuObjetoNegociacao,'-')-1))
			   else cats(put(CdVeiculoLegal, best8.), '||', NuObjetoNegociacao)
		 end
	 else
		 case when NuObjetoNegociacao like '%-%' and CdGrupoConciliacao not in (7, 15)
			   then cats(put(CdVeiculoLegal, best8.), '||', put(CdModalidadeProduto, best3.), '||', substr(NuObjetoNegociacao, 1,index(NuObjetoNegociacao,'-')-1))
			   else cats(put(CdVeiculoLegal, best8.), '||', put(CdModalidadeProduto, best3.), '||', NuObjetoNegociacao)
		 end
	 end as CdChaveOperacao
	,cats(DtReferencia,'|', NuObjetoNegociacao,'|', CdGrupoConciliacao,'|', CdModalidadeProduto,'|', NmModalidadeProduto,'|', CdVeiculoLegal,'|', calculated CdChaveOperacao) as chave
	,sum(Res_Contabil) as Res_Contabil format commax22.2
	,sum(Res_Gerencial) as Res_Gerencial format commax22.2
	,calculated Res_Contabil - calculated Res_Gerencial as DIF format commax22.2
	,sum(Flag_TIR_Taxa) as Flag_TIR_Taxa
from UNIAO_FC_FG

group by
	 DtReferencia
	,NuObjetoNegociacao
	,TpContabil
	,CdGrupoConciliacao
	,CdModalidadeProduto
	,NmModalidadeProduto
	,CdVeiculoLegal

	,CdChaveOperacao
	,chave

having abs(DIF) > 1
;
quit;


options compress=binary reuse=yes replace symbolgen;
proc sql;
create table DIF_NUOBJNEG as
select distinct
	 chave
	,sum(Res_Contabil) as Res_Contabil format commax22.2
	,sum(Res_Gerencial) as Res_Gerencial format commax22.2
	,calculated Res_Contabil - calculated Res_Gerencial as DIF format commax22.2
from FATO_CONTABIL_VS_FATO_GERENCIAL

group by
chave

having abs(DIF) > 1
;
quit;

proc sql;
create table FCvsFG as
select
	 2 as CdTipoAjuste
	,33 as CdTipoRealocacao
	,case when coalesce(t1.CdGrupoConciliacao, t4.cdGrupoConciliacao) = 1
			then 'Opcoes Vencidas'
			else 'Diferença FC vs FG'
	 end as DsJustificativaRealocacao length 99
	,coalesce(t1.CdModalidadeProduto, t4.CdModalidade_Produto_SAS) as CdModalidade_Produto_SAS
	,. as CdTipoComponente
	,. as CdSegmentoMercado
	,coalesce(coalesce(coalesce(t1.TpContabil, t3.TpContabil), t4.Tipo_Contabil), t5.Tipo_Contabil) as CdTipoContabil
	,t1.DtReferencia as DtReferencia
	,0 as VrResultadoTesourariaNotDif
	,0 as VrResultadoComercial
	,0 as VrResultadoAcionista
	,0 as VrResPrivate
	,0 as VrCustoCaixa
	,0 as VrMora
	,0 as VrMulta
	,0 as VrRiscoAlocado
	,t1.CdChaveOperacao
	,coalesce(coalesce(t3.cdCliente, t4.cdCliente), t5.cdCliente) as CdClienteContraParte
	,. as CdOfficer
	,. as CdDivisaoComercial
	,. as CdGrupoComercial
	,. as CdGrupoEconomico
	,. as CdRegional
	,coalesce(t1.CdVeiculoLegal, t4.cdVeiculo) as cdVeiculo
	,. as NuContaContabil
	,coalesce(t6.CdBook, t7.CdBook) as CdBook
	,coalesce(t1.CdGrupoConciliacao, t4.cdGrupoConciliacao) as cdGrupoConciliacao
	,'' as SEGMENTAR
	,coalesce(t1.NuObjetoNegociacao, t4.NuObjetoNegociacao) as NuObjetoNegociacao
	,t1.DIF
	,t1.Flag_TIR_Taxa
from FATO_CONTABIL_VS_FATO_GERENCIAL t1

left join DIF_NUOBJNEG t2 on t1.chave = t2.chave

left join FATO_GERENCIAL_04_ t3
			on t1.NuObjetoNegociacao = t3.NuObjetoNegociacao

			and t1.TpContabil = t3.TpContabil
			and t1.CdModalidadeProduto = t3.CdModalidadeProduto

left join SIC_SAIDA_02_ t4
			 on t1.cdGrupoConciliacao = t4.cdGrupoConciliacao
				 and t1.CdChaveOperacao = t4.Chave_Unica
				 and t1.TpContabil = t4.Tipo_Contabil

left join SIC_SAIDA_M02 t5 on t1.cdGrupoConciliacao = t5.cdGrupoConciliacao and t1.CdChaveOperacao = t5.Chave_Unica

left join FATO_GERENCIAL_04 t6
	on t1.NuObjetoNegociacao = t6.NuObjetoNegociacao
	and coalesce(t1.CdModalidadeProduto, t4.CdModalidade_Produto_SAS) = t6.CdModalidadeProduto


left join SIC_SAIDA_02 t7
	on t1.cdGrupoConciliacao = t7.cdGrupoConciliacao
	and t1.CdChaveOperacao = t7.Chave_Unica

where t2.chave is not null

;
quit;

create table CALCULAR_CDI_SPREAD as
select
a.DtReferencia
,a.NuObjetoNegociacao
,a.CdModalidadeProduto
,a.CdGrupoConciliacao
,a.CdVeiculoLegal
,c.cdCliente
,a.TpContabil
,a.CdBook
,c.Segmento_Mercado_SAS
,case when c.Segmento_Mercado_SAS = "PRIVATE" Then 3
		when c.Segmento_Mercado_SAS = "CORRETORA" Then 11
		when c.Segmento_Mercado_SAS = "CORPORATE" Then 3
		when c.Segmento_Mercado_SAS = "MIDDLE" Then 2
		when c.Segmento_Mercado_SAS = "ATIVOS ESPECIAIS" Then 2
		when c.Segmento_Mercado_SAS = "INSTITUICAO FINANCEIRA" Then 6
		when c.Segmento_Mercado_SAS = "LARGE" Then 10
		else 3
 end as CdSegmentoMercado
,case when a.NuObjetoNegociacao ne "" then Cats (a.CdVeiculoLegal, '||', a.CdModalidadeProduto, '||', a.NuObjetoNegociacao)
		else "Verificar"
 end as ChaveUnica
,case when a.CdSituacaoPosicaoContrato = 60 then "Não Calcula"
		when b.Excecao = "Não Calcula" then "Não Calcula"
		when (a.CdGrupoConciliacao = 2 and a.CdModalidadeProduto in (2, 3, 16, 25, 26, 27, 28, 29, 30, 31)) then "Calcula"
		when (a.CdGrupoConciliacao = 8 and a.CdModalidadeProduto in (94, 182, 185, 186, 190, 194, 196, 197, 198, 208, 209,
																						 211, 213, 214, 221, 224, 265, 266, 268, 277, 400, 409)) then "Calcula"
		when (a.CdGrupoConciliacao = 3 and a.CdModalidadeProduto not in (96, 98, 101, 103, 188, 210, 259, 533, 536, 495, 568)) Then "Calcula"
		else "Não Calcula"
 end as Calcular
,sum(coalesce(a.VrAccrualTxOpeGerDm1,0)) as AccrualTxOpeDm1 format COMMA18.2
,sum(coalesce(a.VrAccrualTxTirDm1,0)) as AccrualTxTirDm1 format COMMA18.2
,sum(coalesce(a.VrAccrualTxOpeGerDm1,0) - coalesce(a.VrAccrualTxTirDm1,0)) as Spread format COMMA18.2
,case when Calculated Calcular = "Calcula"
			  then sum(coalesce(a.VrAccrualTxOpeGerDm1,0)*(c.PctRiskSharingResultado/100) - coalesce(a.VrAccrualTxTirDm1,0)*(c.PctRiskSharingResultado/100)) * &CDI_DIA.
		else 0
 end as CDI_DIA format COMMA18.2
,max(c.PctRiskSharingResultado) as PctRiskSharingResultado
from FATO_GERENCIAL as a

left join Tabela_Excecoes as b on a.NuObjetoNegociacao = b.NuObjetoNegociacao

left join Segmentacao_OK as c on a.NuObjetoNegociacao = c.CdObjetoNegociacao

where a.CdGrupoConciliacao in (2, 3, 8)
		and a.CdBook not in (232, 623, 625)
		and a.DtReferencia = &DtAjuste.
		and c.CdObjetoNegociacao is not null

group by
a.DtReferencia
,ChaveUnica
,a.NuObjetoNegociacao
,a.CdModalidadeProduto
,a.CdGrupoConciliacao
,a.CdVeiculoLegal
,c.cdCliente
,a.TpContabil
,a.CdBook
,c.Segmento_Mercado_SAS
,Calcular

having Calcular = "Calcula"
		 and coalesce(Spread,0) ne 0
;
quit;

				  end
			else 0
	 end as VrResultadoTesourariaNotDif format commax22.2
	,case when t1.CdModalidade_Produto_SAS = t2.CdModalidadeProduto then
				  case when Perc_Risk_Sharing is not missing then
								t1.DIF * coalesce(t2.VrResultadoComercial,0) * t1.Perc_Risk_Sharing/100
						 else t1.DIF * coalesce(t2.VrResultadoComercial,0)
				  end

						 else t1.DIF * coalesce(t2.VrResultadoAcionista,0)
				  end

	,t1.SEGMENTAR
	,t1.NuObjetoNegociacao

	,'Regra1' as Regra length=15

Proc Sql;
Create Table Work.Agrup_analitico as
select max(DataRef) as DataRef format ddmmyy10., GrupoConciliacao, sum(ResultadoDia) as ResultadoDia, sum(ResultadoTesouraria) as ResultadoTesouraria,
Sum(ResultadoComercial) as ResultadoComercial, Sum(ResultadoAcionista) as ResultadoAcionista, sum(VrNaoExplicado) as VrNaoExplicado,

from PLANCTRL.REPOS_ANALITICO&Base.
where GrupoConciliacao not in ('Financeira', 'Swap Interareas')

select DataRef, GrupoConciliacao, sum(ResultadoDia) as ResultadoDia, sum(ResultadoTesouraria) as ResultadoTesouraria,

count(*) as quantidade
from PLANCTRL.REPOS_CONSOLIDADO&Base.
where GrupoConciliacao not in ('Financeira', 'Swap Interareas')
group by DataRef, GrupoConciliacao
order by GrupoConciliacao;

Create table Work.Validacao_Base_gravada as
select  ag.DataRef, ag.DataInicio,  ag.GrupoConciliacao,
Case When ABS((ABS(ag.ResultadoTesouraria)+ABS(ag.ResultadoComercial)+ABS(ag.ResultadoAcionista))
- coalesce((ABS(ac.ResultadoTesouraria)+ABS(ac.ResultadoComercial)+ABS(ac.ResultadoAcionista)),0)) > 100
then 'Diferença na base' else 'OK' end as Validacao
from Work.Agrup_analitico as ag
Left join Work.Agrup_consolidado ac
on ac.DataRef = ag.DataRef
and ac.GrupoConciliacao = ag.GrupoConciliacao
;

quit;
