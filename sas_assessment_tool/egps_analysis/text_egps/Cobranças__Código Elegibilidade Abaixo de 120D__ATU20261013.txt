%MACRO CONTRATOS_INFO (OBSERV=MAX);

%LET _TIMER_START = %SYSFUNC(DATETIME());

OPTIONS OBS= &OBSERV;

DATA _NULL_;

	CALL SYMPUT('DIA_REF_TEXT', CAT("'", PUT(TODAY()-1, YYMMDDN8.), "'"));
	CALL SYMPUT('DIA_REF_TEXT_M0', CAT("'", PUT(INTNX("MONTH", TODAY(), -1, "e"), YYMMDDN8.), "'"));
	CALL SYMPUT('DIA_REF_NUM', PUT(TODAY()-1, DATE9.));
	CALL SYMPUT('SAFRA_REF', PUT(TODAY()-1, YYMMN6.));
	CALL SYMPUT('SAFRA_REF_M0', PUT(INTNX("MONTH", TODAY(), -1, "e"), YYMMN6.));
	CALL SYMPUT('SAFRA_REF_M1', PUT(INTNX("MONTH", TODAY(), -2, "e"), YYMMN6.));
	CALL SYMPUT('SAFRA_REF_M2', PUT(INTNX("MONTH", TODAY(), -3, "e"), YYMMN6.));

RUN;

%PUT _USER_;

PROC SQL;
CREATE TABLE DESC1_&SAFRA_REF. AS
SELECT DISTINCT
A.*
,%POLITICA_ATIVA_ESTRUTURANTE(A.PRODUTO, A.QTATRASO, A.CLUSTER_FINAL, A.COLLECTION_MODELO, LETRA_ATIVA_CAMPEA)

,%POLITICA_ATIVA_DESAF(A.PRODUTO, A.QTATRASO, A.CLUSTER_FINAL, A.COLLECTION_MODELO, LETRA_ATIVA_DESAF1)

,%POLITICA_ATIVA_CIENCIA_DE_DADOS(A.PRODUTO, A.QTATRASO, A.SALDO, A.GH_MODELO_CIENCIA_DADOS, LETRA_ATIVA_DESAF2)

,CASE WHEN A.FLAG_CENARIO_ELEGIBILIDADE = 'POLITICA_CIENCIA_DE_DADOS' THEN CALCULATED LETRA_ATIVA_DESAF2
      WHEN A.FLAG_CENARIO_ELEGIBILIDADE = 'POLITICA_ESTRUTURANTE_BAU' THEN CALCULATED LETRA_ATIVA_CAMPEA
      ELSE CALCULATED LETRA_ATIVA_DESAF1 END AS LETRA_ATIVA_BAU

,%POLITICA_WO_ESTRUTURANTE(A.PRODUTO, A.QTATRASO, A.CLUSTER_FINAL, A.COLLECTION_MODELO, LETRA_WO_BAU)

,CASE WHEN A.FL_BASE = 'ATIVO' THEN CALCULATED LETRA_ATIVA_BAU ELSE CALCULATED LETRA_WO_BAU END AS LETRA_BAU

,%POLITICA_DESCONTO_OTIMO(A.QTATRASO, A.DESCONTO_OTIMO, LETRA_OTIMO)

,%POLITICA_VULNERAVEIS(CALCULATED LETRA_ATIVA_BAU, CALCULATED LETRA_WO_BAU, FL_BASE, LETRA_VULNERAVEL)

,'O2' AS LETRA_SEM_DESCONTO

FROM BASE_ORIGEM_&SAFRA_REF. A

;QUIT;

PROC SQL;
connect to bigquery as perm (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
CREATE TABLE DESC2_&SAFRA_REF. AS
SELECT DISTINCT

A.*
,CASE WHEN INPUT(B.ACMAXDISC1,2.) IS MISSING THEN 0 ELSE INPUT(B.ACMAXDISC1,2.) END AS DESCONTO_HISTORICO
,CASE WHEN INPUT(C.ACMAXDISC1,2.) IS MISSING THEN 0 ELSE INPUT(C.ACMAXDISC1,2.) END AS DESCONTO_BAU
,CASE WHEN INPUT(D.ACMAXDISC1,2.) IS MISSING THEN 0 ELSE INPUT(D.ACMAXDISC1,2.) END AS DESCONTO_OTIMO
,CASE WHEN INPUT(E.ACMAXDISC1,2.) IS MISSING THEN 0 ELSE INPUT(E.ACMAXDISC1,2.) END AS DESCONTO_ELASTICIDADE
,CASE WHEN INPUT(F.ACMAXDISC1,2.) IS MISSING THEN 0 ELSE INPUT(F.ACMAXDISC1,2.) END AS DESCONTO_FINAL_OTIMO

FROM DESC1_&SAFRA_REF. A
LEFT JOIN BASE_CYBER B ON
A.LETRA_HISTORICO = B.AQQUE_TRATADA
AND A.QTATRASO >= INPUT(B.AGDMDAYSMIN,6.)
AND A.QTATRASO <= INPUT(B.AGDMDAYSMAX,6.)
AND B.ACLABEL = 'PARCELA'
AND B.AGPLANO = '1'

LEFT JOIN BASE_CYBER C ON
A.LETRA_BAU = C.AQQUE_TRATADA
AND A.QTATRASO >= INPUT(C.AGDMDAYSMIN,6.)
AND A.QTATRASO <= INPUT(C.AGDMDAYSMAX,6.)
AND C.ACLABEL = 'PARCELA'
AND C.AGPLANO = '1'

LEFT JOIN BASE_CYBER D ON
A.LETRA_OTIMO = D.AQQUE_TRATADA
AND A.QTATRASO >= INPUT(D.AGDMDAYSMIN,6.)
AND A.QTATRASO <= INPUT(D.AGDMDAYSMAX,6.)
AND D.ACLABEL = 'PARCELA'
AND D.AGPLANO = '1'

LEFT JOIN BASE_CYBER E ON
A.LETRA_ELASTICIDADE = E.AQQUE_TRATADA
AND A.QTATRASO >= INPUT(E.AGDMDAYSMIN,6.)
AND A.QTATRASO <= INPUT(E.AGDMDAYSMAX,6.)
AND E.ACLABEL = 'PARCELA'
AND E.AGPLANO = '1'

LEFT JOIN BASE_CYBER F ON
A.LETRA_OTIMO = F.AQQUE_TRATADA
AND A.QTATRASO >= INPUT(F.AGDMDAYSMIN,6.)
AND A.QTATRASO <= INPUT(F.AGDMDAYSMAX,6.)
AND F.ACLABEL = 'PARCELA'
AND F.AGPLANO = '1'

;QUIT;

%MEND CONTRATOS_INFO;
