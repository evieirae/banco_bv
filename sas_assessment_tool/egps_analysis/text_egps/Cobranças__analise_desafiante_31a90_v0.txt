		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas)  > 0.9 		THEN "91,1-100%"

CREATE TABLE apuracao_recorrente AS
SELECT
	A.*,
	B.atraso_m0,
	B.sqcontratofinanceiro_m0,

	CASE
		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN C.DtProcessamentoLiquidacao
		WHEN D.dt_baixa IS NOT NULL 					THEN D.dt_baixa
		WHEN F.dt_ultimo_recebimento IS NOT NULL 		THEN F.dt_ultimo_recebimento
	END FORMAT=DDMMYY10. AS DT_PAGAMENTO,

	CASE
		WHEN C.DtProcessamentoLiquidacao IS NOT NULL 	THEN "Quitacao"
		WHEN D.dt_baixa IS NOT NULL 					THEN "Reneg"
		WHEN F.dt_ultimo_recebimento IS NOT NULL 		THEN "Pgmt"
		ELSE "Sem Acao"
	END AS TP_PAGAMENTO

FROM BASE_ELEG_SEGMENTACAO AS A

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			A.Nucontrato,
			MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro_m0,
			MAX(A.nudiasatraso) as atraso_m0,
			CASE
				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
			END AS SALDO

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

		LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
		ON A.NUCONTRATO = J.NUCONTRATO
		AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

		WHERE DATE(A.dtprocessamento) = &fim_mes_m0.
		GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato
) AS B
ON A.nucontrato = B.NUCONTRATO

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			Nucontrato,
			MIN(Date(DtProcessamentoLiquidacao)) as DtProcessamentoLiquidacao

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) >= &inicio_mes. and DATE(dtprocessamento) <= &fim_mes_m1.
			and DtProcessamentoLiquidacao IS NOT NULL
		group by nucontrato
) AS C

ON A.nucontrato = C.NUCONTRATO and A.Data_Execucao+1 < C.DtProcessamentoLiquidacao


LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			NuContratoGestao,
			MAX(SqContratoFinanceiro)      		AS SqContratoFinanceiro,
			NuContratoGestaoNovo,
			MAX(sqcontratofinanceironovo)     	AS sqcontratofinanceironovo,
			MAX(DATE(dt_baixa))         		AS dt_baixa,
			MAX(dt_reneg)         				AS dt_reneg,
			MAX(SAFE_CAST(vr_presente AS FLOAT64))   	AS vr_presente,
			MAX(SAFE_CAST(vr_financiado AS FLOAT64))  	AS vr_financiado,
			MAX(SAFE_CAST(Tx_Pactuada AS FLOAT64))   	AS Tx_Pactuada,
			MAX(qt_parcela)        				AS qt_parcela,
			MAX(vr_prestacao)        			AS vr_prestacao,
			MAX(FlRestruturacaoContrato)     	AS FlRestruturacaoContrato

		FROM bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador
		WHERE 	DATE(dt_baixa) > &fim_mes_m0. AND DATE(dt_baixa) <= &fim_mes_m1. and
				(dt_cancelamento is null or dt_reativacao is not null) and
				ST_RENEG = 3
		GROUP BY
			NuContratoGestao,
		    NuContratoGestaoNovo
) AS D
ON A.nucontrato = D.NuContratoGestao AND A.sqcontratofinanceiro = D.SqContratoFinanceiro and A.Data_Execucao+1 < D.dt_baixa

LEFT JOIN CONNECTION TO PERM (
	SELECT
		cd_Contrato,
		sq_contrato_financeiro,

		MAX(DATE(dt_ultimo_recebimento)) AS dt_ultimo_recebimento

	FROM bv-pdbd-prd.dvry_core.tbparcela
	WHERE st_parcela = 'Q' AND DATE(dt_ultimo_recebimento) > &fim_mes_m0. AND DATE(dt_ultimo_recebimento) <= &fim_mes_m1.
	GROUP BY 1,2
) as F
ON A.nucontrato = F.cd_Contrato and A.sqcontratofinanceiro = F.sq_contrato_financeiro and A.Data_Execucao+1 < F.dt_ultimo_recebimento

;QUIT;

				WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
				ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))

) AS B

ON A.NUCONTRATO = INPUT(B.NuContrato, BEST32.)

LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			Nucontrato,
			MIN(Date(DtProcessamentoLiquidacao)) as DtProcessamentoLiquidacao

		FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
		where DATE(dtprocessamento) >= &inicio_mes. and DATE(dtprocessamento) <= &fim_mes_m1.
			and DtProcessamentoLiquidacao IS NOT NULL
		group by nucontrato
) AS C
ON A.nucontrato = INPUT(C.NUCONTRATO,BEST32.) and A.Data_Execucao+1 < C.DtProcessamentoLiquidacao



LEFT JOIN CONNECTION TO PERM (
		SELECT DISTINCT
			NuContratoGestao,

	FROM bv-pdbd-prd.dvry_core.tbparcela
	WHERE st_parcela = 'Q' AND DATE(dt_ultimo_recebimento) > &fim_mes_m0. AND DATE(dt_ultimo_recebimento) <= &fim_mes_m1.
	GROUP BY 1,2

CONNECT TO BIGQUERY AS PERM (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=20);
CREATE TABLE BASE_ELEG_SEGMENTACAO AS
SELECT
	A.*,
	&MOB. AS MOB,

	CASE
		WHEN MODQUEBRA2.nucontrato IS NOT NULL 		THEN MODQUEBRA2.GH
		ELSE 20
	END AS GH_QUEBRA_V2,

	CASE
		WHEN COLLECTION_60.nucontrato IS NOT NULL 	THEN COLLECTION_60.GH
		WHEN COLLECTION_90.nucontrato IS NOT NULL 	THEN COLLECTION_90.GH
		ELSE 20
	END AS GH_COLLECTION,

	B.SALDO,
	G.parcela_atraso,
	G.fl_atraso_primeira_parc,
	G.Parcelas_pagas_ate_30d,
	CASE
		WHEN G.Parcelas_pagas_ate_30d <= 2  THEN '1. 0 a 2'
		WHEN G.Parcelas_pagas_ate_30d <= 4  THEN '2. 3 a 4'
		WHEN G.Parcelas_pagas_ate_30d <= 7  THEN '3. 5 a 7'
		WHEN G.Parcelas_pagas_ate_30d <= 14 THEN '4. 8 a 14'
		WHEN G.Parcelas_pagas_ate_30d >  14 THEN '5. 15+'
	END AS GROUP_Parc_Pag_30d,

	CASE
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.05 		THEN "00,0-05,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.1 		THEN "05,1-10,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.2 		THEN "10,1-20,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.3 		THEN "20,1-30,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.4 		THEN "30,1-40,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.5 		THEN "40,1-50,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.6 		THEN "50,1-60,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.7 		THEN "60,1-70,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.8 		THEN "70,1-80,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas) <= 0.9 		THEN "80,1-90,0%"
		WHEN G.Parcelas_Pagas/(G.Parcelas_Abertas + G.Parcelas_Pagas)  > 0.9 		THEN "91,1-100%"
		ELSE "ERRO"
	END AS PERCENT_PAGO,

	G.Parcelas_pagas_ate_90d/(G.Parcelas_Abertas + G.Parcelas_Pagas) as PCT_PG_ATE_90D,

	CASE
		WHEN B.SALDO <= 10000 	THEN "01. ATÉ 10K"
		WHEN B.SALDO <= 20000 	THEN "02. ATÉ 20K"
		WHEN B.SALDO <= 30000 	THEN "03. ATÉ 30K"
		WHEN B.SALDO <= 45000 	THEN "04. ATÉ 45K"
		WHEN B.SALDO  > 45000 	THEN "05. ACIMA DE 45K"
	END AS FX_SALDO,

	CASE WHEN F.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END 		AS FL_RECORRENTE

FROM BASE_ELEG AS A


LEFT JOIN CONNECTION TO PERM(
	SELECT DISTINCT
		A.NuContrato,
		MAX(A.sqcontratofinanceiro) as sqcontratofinanceiro,
		Date(A.DtProcessamento)		AS Data_Execucao,
		A.CdMotivoContratoFinanceiro,
		MAX(A.nudiasatraso) 		AS NuDiasAtraso,
		A.CdSituacaoContabil,
		CASE
			WHEN J.FlRestruturacaoContrato = 'S' THEN SUM(CAST(A.vrpermanenciaapropriado AS FLOAT64)+CAST(A.VrTaxaContratoRestruturacao AS FLOAT64))
			ELSE SUM(CAST(A.VrTaxaContrato AS FLOAT64) + CAST(A.VrPermanenciaApropriado AS FLOAT64))
		END AS SALDO

	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil AS A

	LEFT JOIN bv-pdbd-prd.trusted_dbgestaocdccg.tbcontratofinanceiro AS J
	ON A.NUCONTRATO = J.NUCONTRATO
	AND A.SqContratoFinanceiro = J.SqContratoFinanceiro

	WHERE DATE(A.DtProcessamento) = &fim_mes_m0.
	AND A.CdProduto IN(12, 14)
	GROUP BY A.nucontrato, A.CdSituacaoContabil, J.FlRestruturacaoContrato, A.CdMotivoContratoFinanceiro, Data_Execucao

) AS B
ON A.NUCONTRATO = B.NuContrato



LEFT JOIN BASE_RECORRENTE AS F
ON A.NUCONTRATO = F.NUCONTRATO


LEFT JOIN CONNECTION TO PERM(
SELECT DISTINCT cd_contrato,
             	sq_contrato_financeiro,
				SUM(CASE WHEN nu_parcela = 1 and dt_ultimo_recebimento > dt_vencimento THEN 1 ELSE 0 END)								AS fl_atraso_primeira_parc,
				MIN(CASE WHEN dt_ultimo_recebimento > dt_vencimento THEN nu_parcela ELSE 100 END) 										AS parcela_atraso,
				SUM(CASE WHEN st_parcela='Q' AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END)        						AS Parcelas_Pagas,
                SUM(CASE WHEN st_parcela='A' OR (st_parcela='Q' AND dt_ultimo_recebimento > &fim_mes_m0.) THEN 1 ELSE 0 END)        								AS Parcelas_Abertas,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) < 30 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) 				AS Parcelas_pagas_ate_30d,
				SUM(CASE WHEN DATE_DIFF(dt_ultimo_recebimento, dt_vencimento, DAY) BETWEEN 1 AND 90 AND dt_ultimo_recebimento <= &fim_mes_m0. THEN 1 ELSE 0 END) 	AS Parcelas_pagas_ate_90d,

FROM bv-pdbd-prd.dvry_core.tbparcela
GROUP BY cd_contrato,
         sq_contrato_financeiro) AS G
ON A.NuContrato = G.cd_contrato AND A.SqContratoFinanceiro = G.sq_contrato_financeiro



LEFT JOIN MODELO_QUEBRA_V2_NODUP as MODQUEBRA2
ON INPUT(A.NuContrato,BEST32.) = MODQUEBRA2.NuContrato


LEFT JOIN TBCOLLECTIONRODAS_30_60 as COLLECTION_60
ON INPUT(A.NuContrato,BEST32.) = COLLECTION_60.NuContrato


LEFT JOIN TBCOLLECTIONRODAS_60_90 as COLLECTION_90
ON INPUT(A.NuContrato,BEST32.) = COLLECTION_90.NuContrato


;QUIT;

CREATE TABLE BASE_OVER AS

	,CASE WHEN B.atraso = 0 THEN "1"
		WHEN B.atraso = . THEN "SI"
		WHEN B.atraso > 0 THEN "0"
	END AS FL_EM_DIA
	,CASE WHEN B.atraso > 30 THEN "1"
		WHEN B.atraso = . THEN "SI"
		WHEN B.atraso <= 30 THEN "0"
	END AS FL_OVER30
	,CASE WHEN B.atraso > 90 THEN "1"
		WHEN B.atraso = . THEN "SI"
		WHEN B.atraso <= 90 THEN "0"
	END AS FL_OVER90

FROM BASE_ELEG_SEGMENTACAO AS A


LEFT JOIN CONNECTION TO PERM(

	SELECT NUCONTRATO, MAX(NuDiasAtraso) AS ATRASO
	FROM bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil
	WHERE DATE(DTPROCESSAMENTO) = &dt_MOB.
	GROUP BY NUCONTRATO

	) as B

	ON A.NUCONTRATO = B.NUCONTRATO

;QUIT;

BY nucontrato sqcontratofinanceiro;
RUN;
