    	  when a.ahcollid in ('4032') 		 then 'TOLEDO_2'

    ,(input(p.apamtvlj,BEST9.) + input(p.apamtvlm,BEST9.)) 											  as VRJUROSPARCELASACORDO

	,AC.acclcollid 																					  as ACCLCOLLID

	ahid
	,ahcollid

	,ahautcollid
	,ahssnum

	,ahpamt
	,ahdescpamt

    ,ahstatus

	dmacct
