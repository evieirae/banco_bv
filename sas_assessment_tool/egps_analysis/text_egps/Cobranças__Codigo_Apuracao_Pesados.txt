CREATE TABLE WORK.AUX_BASE_ENVIOS_2 AS
SELECT DISTINCT A.*
FROM CONNECTION TO PERM
(
select * from bv-pdbd-prd.trusted_rcvry.tb_historico_disponibilizacao
where     SAFE_CAST(dt_disponibilizacao AS DATE) >= '2025-07-01'
    AND

DMQUE IN ('AD05','RD05')

) AS A
;DISCONNECT FROM PERM
;QUIT;

PROC SQL;
CREATE TABLE AUX_BASE_ENVIOS_3 AS
SELECT
A.*,
CASE a.fasecob
    WHEN 'AD05' THEN '1 - ADM_TRAD'
    WHEN 'RD05' THEN '2 - ADM_RENEG'

    ELSE 'Código não mapeado'
END AS NM_faseativa_PRODUTO,
SUBSTR(nr_cpfcnpj,6,2) AS RDG,
datepart(a.DT_DISPONIBILIZACAO) AS DT_DISPONIBILIZACAO2,
    datepart(a.DT_NOVA_DISPONIB) AS DT_NOVA_DISPONIB2
from AUX_BASE_ENVIOS_2 as a
WHERE modprod IN ('9','28','47')
AND nr_dias_atraso <= '30'
;quit;

data AUX_BASE_ENVIOS_4;
set AUX_BASE_ENVIOS_3;
	format DT_DISPONIBILIZACAO2 ddmmyy10.;
	format DT_NOVA_DISPONIB2 ddmmyy10.;
	risco = vl_divida*1;


;run;


data calendario;
    format data_ref date9.;
    do data_ref = '01jul2025'd to '30SEP2025'd;
        output;
    end;
run;


proc sql;
    create table registros_ativos_por_dia as
    select
		t.NM_faseativa_PRODUTO,
		t.RDG,
        c.data_ref,
        count(*) as registros_ativos
    from
        calendario c,
        AUX_BASE_ENVIOS_4 t
    where
        t.DT_DISPONIBILIZACAO2 <= c.data_ref and (t.DT_NOVA_DISPONIB2 > c.data_ref or t.DT_NOVA_DISPONIB2 is null)
    group by
		t.NM_faseativa_PRODUTO,
		t.RDG,
        c.data_ref
    order by
        c.data_ref;
quit;
