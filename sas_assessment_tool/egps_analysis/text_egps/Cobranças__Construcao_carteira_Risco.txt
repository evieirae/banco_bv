CREATE TABLE emp_2 AS
SELECT
    A.*,
    b.atraso as atraso_m1,
	b.Qtd_Parcelas_Abertas,
	b.Qtd_Parcelas_em_Atraso,
	b.PARCELA,
	b.Qtd_Parcelas,
	c.nucontrato as contrato_liquidacao,
	d.nucontrato as contrato_cessao,
	G.dt_baixa,
	case when G.nucontratogestao = . then 'sem reneg' else 'fez reneg' end as renegociacao
FROM base_final2 as a
left join
    CONNECTION TO DBCON (

SELECT	 k.NuContrato,
		 k.SqContratoFinanceiro

		INTO	#BASE1
		FROM gestaocontabil.Vw_ContratoContabil k
		where k.dtprocessamento = &dt_aber.
		GROUP BY k.NuContrato,k.SqContratoFinanceiro

SELECT	 		a.NuContrato,
				a.SqContratoFinanceiro,
				DATEDIFF(DAY, MIN(CASE WHEN	 StParcela <> 'Q' OR
											( StParcela = 'Q' AND
											 DtProcessamentoUltRecebimento > &datam1.)  THEN DtVencimento ELSE NULL END), &datam1.) AS Atraso,
				SUM(CASE WHEN	 StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								 DtProcessamentoUltRecebimento > &datam1.)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
				COUNT(CASE WHEN	 StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								 DtProcessamentoUltRecebimento > &datam1.)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
				COUNT(CASE WHEN	( StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								DATE( DtProcessamentoUltRecebimento) > &datam1.)) AND
								DATE( DtVencimento) < &datam1. THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
				MIN(CASE WHEN	StParcela <> 'Q' OR
											( StParcela = 'Q' AND
											 DtProcessamentoUltRecebimento > &datam1.)  THEN Nuparcela ELSE NULL END) as PARCELA,
				COUNT (VrParcela) as Qtd_Parcelas

		FROM	#BASE1 AS A
				LEFT JOIN DBA.VwTbParcela AS B ON	A.NuContrato = B.NuContrato AND
													A.SqContratoFinanceiro = B.SqContratoFinanceiro
		GROUP BY A.NuContrato, A.SqContratoFinanceiro

) as b
on a.nucontratogestao = b.nucontrato

left join
    CONNECTION TO DBCON (

    SELECT nucontrato,SQCONTRATOFINANCEIRO
	FROM gestaocontabil.Vw_ContratoContabil
	WHERE dtprocessamentoliquidacao >= &datam0. and dtprocessamentoliquidacao <= &datam1.
	GROUP BY nucontrato, SQCONTRATOFINANCEIRO

) as c

CREATE TABLE emp_2 AS
SELECT
    A.*,
    b.atraso as atraso_m1,
	b.Qtd_Parcelas_Abertas,
	b.Qtd_Parcelas_em_Atraso,
	b.PARCELA,
	b.Qtd_Parcelas,
	c.nucontrato as contrato_liquidacao,
	d.nucontrato as contrato_cessao,
	G.dt_baixa,
	case when G.nucontratogestao = . then 'sem reneg' else 'fez reneg' end as renegociacao
FROM base_final2 as a
left join
    CONNECTION TO DBCON (

SELECT	 k.NuContrato,
		 k.SqContratoFinanceiro

		INTO	#BASE1
		FROM gestaocontabil.Vw_ContratoContabil k
		where k.dtprocessamento = &dt_aber.
		GROUP BY k.NuContrato,k.SqContratoFinanceiro

SELECT	 		a.NuContrato,
				a.SqContratoFinanceiro,
				DATEDIFF(DAY, MIN(CASE WHEN	 StParcela <> 'Q' OR
											( StParcela = 'Q' AND
											 DtProcessamentoUltRecebimento > &datam1.)  THEN DtVencimento ELSE NULL END), &datam1.) AS Atraso,
				SUM(CASE WHEN	 StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								 DtProcessamentoUltRecebimento > &datam1.)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
				COUNT(CASE WHEN	 StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								 DtProcessamentoUltRecebimento > &datam1.)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
				COUNT(CASE WHEN	( StParcela <> 'Q' OR
								( StParcela = 'Q' AND
								DATE( DtProcessamentoUltRecebimento) > &datam1.)) AND
								DATE( DtVencimento) < &datam1. THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,

											( StParcela = 'Q' AND
											 DtProcessamentoUltRecebimento > &datam1.)  THEN Nuparcela ELSE NULL END) as PARCELA,
				COUNT (VrParcela) as Qtd_Parcelas

		FROM	#BASE1 AS A
				LEFT JOIN DBA.VwTbParcela AS B ON	A.NuContrato = B.NuContrato AND
													A.SqContratoFinanceiro = B.SqContratoFinanceiro
		GROUP BY A.NuContrato, A.SqContratoFinanceiro

) as b
on a.nucontratogestao = b.nucontrato

    CONNECTION TO DBCON (

    SELECT nucontrato,SQCONTRATOFINANCEIRO

	WHERE dtprocessamentoliquidacao >= &datam0. and dtprocessamentoliquidacao <= &datam1.
	GROUP BY nucontrato, SQCONTRATOFINANCEIRO

on a.nucontratogestao = c.nucontrato and a.SQCONTRATOFINANCEIRO = c.SQCONTRATOFINANCEIRO
