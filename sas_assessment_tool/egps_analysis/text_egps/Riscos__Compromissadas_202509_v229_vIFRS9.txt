%Macro utl_etl_extracao(_fase_,_base_,_distinct_,_chaves_,_variaveis_,_filtros_);

	%if %SYSFUNC(EXIST(EXTRACAO.tb_extracao_rwa_&_data_referencia_._&_fase_.)) %then
		%do;

			data _null_;
				put "ERROR: TABELA tb_extracao_rwa_&_data_referencia_._&_fase_. JÁ EXISTE, VERIFIQUE NECESSIDADE DE SOBREPOSIÇÃO ";
			run;

		%end;
	%else
		%do;
			%utl_etl_informacao(_informacao_= Início da extração,_etapa_=&_fase_.)

			proc sql _method _tree;
			create table EXTRACAO.tb_extracao_rwa_&_data_referencia_._&_fase_.(compress=CHAR) as
				select %if %length(&_distinct_.) ne 0 %then distinct;

				%do contador_chaves = 1 %to %sysfunc(countw(&_chaves_.));
					%scan(&_chaves_.,&contador_chaves.),
				%end;

			%do contador_vars = 1 %to %sysfunc(countw(&_variaveis_.));
				%if &contador_vars. < %sysfunc(countw(&_variaveis_.)) %then
					%scan(&_variaveis_.,&contador_vars.),;
				%else %scan(&_variaveis_.,&contador_vars.);
			%end;

			from &_base_. %if %length(&_filtros_.)>0  %then where   &_filtros_.
				%if %length(&_filtros_.)>0 %then

				%do;
					order by

						%do contador_chaves = 1 %to %sysfunc(countw(&_chaves_.));
							%if  &contador_chaves. < %sysfunc(countw(&_chaves_.)) %then
								%scan(&_chaves_.,&contador_chaves.),;
							%else %scan(&_chaves_.,&contador_chaves.);
						%end;
				%end;
			;
			;
			%do contador_chaves = 1 %to %sysfunc(countw(&_chaves_.));
				create index %scan(&_chaves_.,&contador_chaves.) on EXTRACAO.tb_extracao_RWA_&_data_referencia_._&_fase_.(%scan(&_chaves_.,&contador_chaves.));
			%end;

			%if %sysfunc(countw(&_chaves_.)) > 1 %then
				%do;
					create index _index_key_ on EXTRACAO.tb_extracao_RWA_&_data_referencia_._&_fase_.(
						%do contador_chaves = 1 %to %sysfunc(countw(&_chaves_.));
							%if &contador_chaves.< %sysfunc(countw(&_chaves_.)) %then
							%scan(&_chaves_.,&contador_chaves.),;
							%else %scan(&_chaves_.,&contador_chaves.) );
				%end;

		%end;;
	quit;

	%utl_etl_informacao(_informacao_= Término da extração,_etapa_=&_fase_.)
	%end;
%mend utl_etl_extracao;

%macro utl_etl_extracao_csv(_fase_,_filename_,_variaveis_,_length_,_format_,_informat_,_input_,_filtros_);

%put &_fase_.;
%put &_filename_.;
%put &_variaveis_.;
%put &_length_.;
%put &_format_.;
%put &_informat_.;
%put &_input_.;
%put &_filtros_.;
%put %length(&_filtros_.);

	%if %SYSFUNC(EXIST(EXTRACAO.tb_extracao_rwa_&_data_referencia_._&_fase_.)) %then
		%do;

			data _null_;
				put "ERROR: TABELA tb_extracao_rwa_&_data_referencia_._&_fase_. JÁ EXISTE, VERIFIQUE NECESSIDADE DE SOBREPOSIÇÃO ";
			run;

		%end;
	%else
		%do;
			%utl_etl_informacao(_informacao_= Início da extração,_etapa_=&_fase_.)

		FILENAME dir_orig " &_filename_." ;
				data EXTRACAO.TB_EXTRACAO_RWA_&_data_referencia_._&_fase_.%if %length(&_filtros_.) > 0  %then(where=(&_filtros_.));;
					LENGTH
						%do _contador_= 1 %to %sysfunc(countw(&_variaveis_.));
							%scan(&_variaveis_.,&_contador_.,"|") %scan(&_length_.,&_contador_.,"|")
						%end;
							;
					FORMAT
						%do _contador_= 1 %to %sysfunc(countw(&_variaveis_.));
							%scan(&_variaveis_.,&_contador_.,"|") %scan(&_format_.,&_contador_.,"|")
						%end;
						;
					INFORMAT
						%do _contador_= 1 %to %sysfunc(countw(&_variaveis_.));
							%scan(&_variaveis_.,&_contador_.,"|") %scan(&_informat_.,&_contador_.,"|")
						%end;
						;

	            INFILE dir_orig
                DELIMITER = ';'
                MISSOVER
                DSD
				FIRSTOBS=2;
				input
						%do _contador_= 1 %to %sysfunc(countw(&_variaveis_.));
							%scan(&_variaveis_.,&_contador_.,"|") %str(:) %scan(&_input_.,&_contador_.,"|")
						%end;
				;

				run;
			%utl_etl_informacao(_informacao_= Término da extração,_etapa_=&_fase_.)
		%end;
%mend utl_etl_extracao_csv;
