create table TbProspectoresNegocios as
select
	 ProspectoresNegocios.DtAtualizacao
	,ProspectoresNegocios.Id
	,ProspectoresNegocios.NmOrganizacao
	,ProspectoresNegocios.NmColaboradorProspeccao

	,ProspectoresNegocios.DsFunil
	,ProspectoresNegocios.NmEtiqueta
	,ProspectoresNegocios.NmEtapa
	,ProspectoresNegocios.DsMotivoPerda
	,ProspectoresNegocios.DtNegocioCriado
	,ProspectoresNegocios.FlLeadQualificado

	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Ativação'
						,'Reunião Agendada'
						,'Reunião Realizada'
						,'Proposta Enviada'
						,'Documentações'
						,'Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlAtivacao
	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Reunião Agendada'
						,'Reunião Realizada'
						,'Proposta Enviada'
						,'Documentações'
						,'Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlReuniaoAgendada
	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Reunião Realizada'
						,'Proposta Enviada'
						,'Documentações'
						,'Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlReuniaoRealizada
	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Proposta Enviada'
						,'Documentações'
						,'Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlPropostaEnviada
	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Documentações'
						,'Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlDocumentacoes
	,CASE
		WHEN ProspectoresNegocios.NmEtapa in ('Inclusão Sharepoint (cadastro BV)'
						,'On Hold')
			then 1 else 0 end AS FlInclusaoSharepoint
	,CASE
		WHEN ProspectoresNegocios.NmEtapa = 'On Hold'
			then 1 else 0 end AS FlOnHold
	,CASE
		WHEN ProspectoresNegocios.Status = 'Ganho'
			then 1 else 0 end AS FlGanho

from basegrc.TbProspectoresNegocios ProspectoresNegocios

	left join basegrc.TbProspectoresProprietario ProspectoresProprietario
		on  ProspectoresNegocios.NmColaboradorProspeccao = ProspectoresProprietario.NmColaboradorProspeccao

		inner join (select distinct DsLogin, IdMatricula from bases.TbEstruturaProspectores) EstruturaProspectores
			on  ProspectoresProprietario.DsLogin = EstruturaProspectores.DsLogin

where ProspectoresNegocios.DtArquivo = &DtAtualizacao.
;run;

proc sql;
create table TbExtContratoConsolidado as
select
	 Contrato.NuAnoMes
	,ParceiroComercial.CdClassificacaoLimiteConvenio
	,ParceiroComercial.NmClassificacaoLimiteConvenio
	,min(datepart(Contrato.DtContrato)) as DtContratoMin format yymmdd10.
	,max(datepart(Contrato.DtContrato)) as DtContratoMax format yymmdd10.
from bases.TbExtContrato Contrato

	inner join basegrc.TbGrupoProdutoMeta GrupoProdutoMeta
		 on Contrato.CdGrupoSubProduto = GrupoProdutoMeta.CdGrupoSubProduto
		and GrupoProdutoMeta.NmGrupoProdutoMeta = 'Privado'

	left join tmp.TbParceiroComercial ParceiroComercial
		on Contrato.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

where Contrato.StProducao = 'S'
group by 1,2,3
;run;

proc sql;
create table TbConvenioProducaoMensal as
select
	 CdClassificacaoLimiteConvenio
	,NmClassificacaoLimiteConvenio
	,NuAnoMes
	,max(DtContratoMin) as DtContratoMin format yymmdd10.
	,max(DtContratoMax) as DtContratoMax format yymmdd10.

where CdClassificacaoLimiteConvenio is not null

proc sql;
create table TbProspectoresAtividades as
select

	 ProspectoresAtividades.DtAtualizacao
	,ProspectoresAtividades.DtVencimento
	,ProspectoresAtividades.DtAdicionada
	,ProspectoresAtividades.DtAtualizada
	,ProspectoresAtividades.DtRealizado
	,ProspectoresAtividades.ID
	,ProspectoresAtividades.NmOrganizacao
	,ProspectoresAtividades.Status
	,ProspectoresAtividades.NmEtapa
	,ProspectoresAtividades.FlConcluido
	,ProspectoresAtividades.DsAssunto
	,ProspectoresAtividades.TipoAtividade
	,ProspectoresAtividades.NmColaboradorProspeccao
	,ProspectoresProprietario.DsLogin
	,ProspectoresAtividades.StLivreOcupado
	,ProspectoresAtividades.NuCnpjLoja
	,ProspectoresAtividades.DsFunil

from basegrc.TbProspectoresAtividades ProspectoresAtividades

	left join basegrc.TbProspectoresProprietario ProspectoresProprietario
		on  ProspectoresAtividades.NmColaboradorProspeccao = ProspectoresProprietario.NmColaboradorProspeccao

		inner join (select distinct DsLogin from bases.TbEstruturaProspectores) EstruturaProspectores
			on  ProspectoresProprietario.DsLogin = EstruturaProspectores.DsLogin

;run;

proc export
data=TbProspectoresAtividades
file='/projetos/MISV/sas-misv-base-indicadores-varejo/outbound/TbProspectoresAtividades.csv' replace
dbms=dlm;
delimiter=';';
run;

X 'mv /projetos/MISV/sas-misv-base-indicadores-varejo/outbound/TbProspectoresAtividades.csv /projetos/MISV/sas-misv-base-indicadores-varejo/outbound/processados/';
