PROC SQL;
CREATE TABLE TbParceiroComercial AS
  SELECT
    NuCnpjLoja,
    NmParceiroComercial,
    NmAreaEstruturaComercialC,
    NmSubEstruturaComercialC,
    DsLoginN1C,
    NmTipoParceiroComercial,
    CASE WHEN FlAtivo = 'S' THEN 1 ELSE 0 END AS FlAtivo,
    DtUltimaAtualizacao
  FROM
    BASES.TbParceiroComercial
  WHERE NuCnpjLoja IS NOT NULL

  OUTER UNION CORRESPONDING

  SELECT
    NuCnpjLoja,
    Grupo AS NmParceiroComercial,
    Grupo AS NmAreaEstruturaComercialC,
    Grupo AS NmSubEstruturaComercialC,
    Grupo AS DsLoginN1C,
    Grupo AS NmTipoParceiroComercial,
    0 AS FlAtivo,
    . AS DtUltimaAtualizacao
  FROM
    NAPISTA.GRUPOSAUXILIARES
  WHERE NuCnpjLoja IS NOT NULL;
RUN;

PROC SORT DATA=TbParceiroComercial;
BY DESCENDING FlAtivo
   DESCENDING NmSubEstruturaComercialC
   DESCENDING DtUltimaAtualizacao;
RUN;

PROC SORT DATA=TbParceiroComercial NODUPKEY;
BY NuCnpjLoja;
RUN;

  SELECT
    VendedorGrupo,

    INPUT(
      PUT(INPUT(SCAN(OfertaDtInicio,1," "),2.),z2.)
        ||CASE WHEN LOWER(OfertaDtInicio) LIKE '%fev%' THEN 'FEB'
               WHEN LOWER(OfertaDtInicio) LIKE '%abr%' THEN 'APR'
               WHEN LOWER(OfertaDtInicio) LIKE '%mai%' THEN 'MAY'
               WHEN LOWER(OfertaDtInicio) LIKE '%ago%' THEN 'AUG'

               WHEN LOWER(OfertaDtInicio) LIKE '%out%' THEN 'OCT'

PROC SQL;
CREATE TABLE VENDAS AS
  SELECT
    A.VendedorGrupo,
    A.VendedorFilial,
    A.VendedorNuCnpjLoja,
    A.VendedorCidade,
    A.VendedorEstado,
    A.PlacaVeiculo,
    A.OfertaTipo,
    A.OfertaSituacao,
    A.OfertaDtInicio,
    A.OfertaDtFim,
    A.VencedorFilial,
    A.VencedorNuCnpjLoja,
    A.VencedorCidade,
    A.VencedorEstado,
	A.StatusPagamento,
    A.DataPagamento,
    B.NmParceiroComercial AS NmParceiro_Vendedor,
    B.NmAreaEstruturaComercialC AS Filial_Vendedor,
    B.NmSubEstruturaComercialC AS Canal_Vendedor,
    B.DsLoginN1C AS GR_Vendedor,
    B.NmTipoParceiroComercial AS TipoLoja_Vendedor,
    C.NmParceiroComercial AS NmParceiro_Vencedor,
    C.NmAreaEstruturaComercialC AS Filial_Vencedor,
    C.NmSubEstruturaComercialC AS Canal_Vencedor,
    C.DsLoginN1C AS GR_Vencedor,
    C.NmTipoParceiroComercial AS TipoLoja_Vencedor,
    TODAY() AS DtAtualizacao FORMAT=DATE9.
  FROM
    VENDAS002 A
  LEFT JOIN
    TbParceiroComercial B
    ON A.VendedorNuCnpjLoja = B.NuCnpjLoja
  LEFT JOIN
    TbParceiroComercial C
    ON A.VencedorNuCnpjLoja = C.NuCnpjLoja;
RUN;

DATA TbVendas_Aux;
SET NAPISTA.TbVendas;
  WHERE DtAtualizacao <> TODAY();
RUN;

DATA NAPISTA.TbVendas;
SET TbVendas_Aux VENDAS;
RUN;
