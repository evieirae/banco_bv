PROC SQL;
CONNECT TO postgres
(&PROJURIS_CLOUD.);
CREATE TABLE work.TblReclassificacoes AS
SELECT

 CdCausa
,Pasta
,DtimeReclassificacao
,Dtreclassificacao
,id_usuario_cadastro
,ID_PROCESSO
,NmUsuario
,CdSistema
,FlDtReclassificacaoUlt

FROM CONNECTION TO postgres(

SELECT
case
	when p.SERIAL_IMPORT like '%CTO%' then left(p.SERIAL_IMPORT,10)
	when p.SERIAL_IMPORT like '%CON%' then left(p.SERIAL_IMPORT,10)
	when p.SERIAL_IMPORT like '%VAC%' then left(p.SERIAL_IMPORT,10)
	else p.SERIAL_IMPORT  END AS CdCausa
,P.pasta                                  AS Pasta
,dr.data_reclassificacao as DtimeReclassificacao
,cast (dr.data_reclassificacao as date) as Dtreclassificacao
,dr.id_usuario_cadastro
,P.ID_PROCESSO
,U.Nome as NmUsuario
,'p' as CdSistema
,case when dr.data_reclassificacao = dr2.DtReclassificacaoUlt then 1 else 0 end as FlDtReclassificacaoUlt
FROM       public.M_PROCESSO P
LEFT JOIN  public.D_PROCESSO_RECLASSIFICACAO dr ON dr.ID_PROCESSO = P.ID_PROCESSO
LEFT JOIN (SELECT ID_PROCESSO, max(data_reclassificacao) AS DtReclassificacaoUlt FROM public.D_PROCESSO_RECLASSIFICACAO GROUP BY ID_PROCESSO) dr2 ON dr2.ID_PROCESSO = P.ID_PROCESSO
left join public.M_USUARIO U on U.id_usuario = dr.id_usuario_cadastro
WHERE 1=1
AND P.DATA_ENCERRAMENTO IS null
AND dr.data_reclassificacao is not null





);DISCONNECT FROM postgres;

RUN;
