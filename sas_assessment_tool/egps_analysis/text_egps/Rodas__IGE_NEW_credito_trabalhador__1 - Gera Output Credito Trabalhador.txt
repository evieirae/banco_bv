data _null_;
  length max_dt_bq $19 mxdtsimu $32;

  if symexist('mxdtsimu') then mxdtsimu = symget('mxdtsimu');
  else mxdtsimu = '';

  if missing(mxdtsimu) then do;
    max_dt_bq = '2026-02-01 00:00:00';
  end;
  else do;
    d = input(mxdtsimu, date9.);          /* lê como DATE */
    dt = dhms(d,0,0,0);                   /* converte para DATETIME (00:00:00) */
    max_dt_bq = tranwrd(put(dt, e8601dt19.), 'T', ' '); /* yyyy-mm-dd hh:mm:ss */
  end;

  call symputx('max_dt_bq', max_dt_bq, 'G');
run;

    length status_final $20 cd_status_final $20;
    set funil_credito_trab02;
    by cdjornadacreditotrabalhador dticlusao_simu_datetime;

    retain status_final cd_status_final;

    if first.cdjornadacreditotrabalhador then status_final = sgstatusjornadacreditotrblhdr;
    else do;
        if sgstatusjornadacreditotrblhdr = "CAN" then
            status_final = catx("-", status_final, "CAN");
        else
            status_final = sgstatusjornadacreditotrblhdr;
    end;

    if first.cdjornadacreditotrabalhador then cd_status_final = cdstatusjornadacreditotrblhdr;
    else do;
        if sgstatusjornadacreditotrblhdr = "CAN" then
            cd_status_final = catx(".",cd_status_final, cdstatusjornadacreditotrblhdr);
        else
            cd_status_final = cdstatusjornadacreditotrblhdr;
    end;
run;

proc sql;
  create table TbLeadtime as
  select
  	  dt_penele format datetime25.6,
      cdjornadacreditotrabalhador,
      intck('second', dt_penele, dt_procad) as Leadtime_Penele_Procad,
	  intck('second', dt_procad, dt_pag) as Leadtime_Procad_Pag
  from
  (
    select
	    cdjornadacreditotrabalhador,
        min(case when sgstatusjornadacreditotrblhdr='PENELE'
                 then dticlusao_simu_datetime end) as dt_penele,
        min(case when sgstatusjornadacreditotrblhdr='PROCAD'
                 then dticlusao_simu_datetime end) as dt_procad,
	   	min(case when sgstatusjornadacreditotrblhdr='PAG'
     			 then dticlusao_simu_datetime end) as dt_pag
    from work.funil_credito_trab03
    group by cdjornadacreditotrabalhador
  );
quit;
