create table ENQUADRAMENTOS_CL as
select distinct
a.u3acct 					as U3ACCT
,a.u3dtenqcont				as DT_ENQUAD
,input(u3cdleg,best4.)		as U3CDLEG
,b.saldo_enquadramento 		as SALDO_ENQUADRAMENTO

from (
	select distinct u3acct, datepart(u3dtenqcont) format ddmmyy10. as u3dtenqcont, u3cdleg
	from connection to perm (
	select distinct
	u3acct
	,u3dtenqcont
	,u3cdleg
	from bv-pdbd-prd.trusted_rcvry.uda3
	where u3dtenqcont is not null)
	union
	select distinct u3acct, u3dtenqcont format ddmmyy10. as u3dtenqcont, u3cdleg
	from connection to perm (
	select distinct
	u3acct
	,u3dtenqcont
	,u3cdleg
	from bv-pdbd-prd.trusted_rcvry.uda3_esp
	where u3dtenqcont is not null)
) as a

left join (select distinct
			nr_contrato
			,max(input(vl_divida,best9.)) as saldo_enquadramento
			from connection to perm (
					select distinct
					nr_contrato
					,vl_divida
					from bv-pdbd-prd.trusted_rcvry.tb_historico_disponibilizacao
					where dmque in ('CL'))
			group by nr_contrato) as b on a.u3acct = b.nr_contrato

where calculated U3CDLEG ne 2


;disconnect from perm;
quit;

DATA ENQUADRAMENTOS_CL;
SET ENQUADRAMENTOS_CL;
CONTRATO = U3ACCT*1;
