%MACRO AMORTIZACAO(dia_processamento_0,atraso1,SAFRA);
PROC SQL;
  %CONNECT_GCP;

  CREATE TABLE BASE_JOIN AS
  SELECT *
  FROM CONNECTION TO perm
  (
    SELECT DISTINCT
      Z.NUCONTRATO,
      Z.SQCONTRATOFINANCEIRO,
      
      B.NU_PARCELA           AS NUPARCELA,
      B.ST_PARCELA           AS STPARCELA,
	  B.VR_PRINCIPAL          AS VRPRINCPAL_PARCELA,
      B.DT_ULTIMO_RECEBIMENTO,
      B.DT_PROCESSAMENTO_ULT_RECEBIMENTO,
	  B.DT_VENCIMENTO
    FROM  `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` AS Z
    LEFT JOIN `bv-pdbd-prd.dvry_core.tbparcela` AS B
      ON B.CD_CONTRATO   = Z.NUCONTRATO
     AND B.SQ_CONTRATO_FINANCEIRO = Z.SQCONTRATOFINANCEIRO
    WHERE
      Z.DTPROCESSAMENTO = &dia_processamento_0.
      AND Z.NUDIASATRASO >= &atraso1.
	  AND Z.CdSItuacaoContabil IN ('2')
    ORDER BY Z.NUCONTRATO, Z.SQCONTRATOFINANCEIRO, B.NU_PARCELA
  );
  disconnect from perm;
  QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO AS
SELECT A.*
,CASE
WHEN input(&dia_processamento_0., yymmdd10.) < DT_PROCESSAMENTO_ULT_RECEBIMENTO THEN 0
WHEN STPARCELA = 'A' THEN 0
WHEN NUPARCELA IS NOT NULL THEN 1
ELSE 0 END AS FL_PAGAS

,CASE
WHEN input(&dia_processamento_0., yymmdd10.) < DT_PROCESSAMENTO_ULT_RECEBIMENTO THEN VRPRINCPAL_PARCELA
WHEN STPARCELA = 'A' THEN VRPRINCPAL_PARCELA
WHEN NUPARCELA IS NOT NULL THEN 0
ELSE 0 END AS VRPRINCPAL_PARCELA_ABERTAS

,CASE
WHEN input(&dia_processamento_0., yymmdd10.) > DT_VENCIMENTO AND STPARCELA = 'A' THEN VRPRINCPAL_PARCELA
WHEN STPARCELA = 'Q' THEN 0
WHEN NUPARCELA IS NOT NULL THEN 0
ELSE 0 END AS VRPRINCPAL_PARCELA_ATRASADAS

FROM BASE_JOIN A
;QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_1 AS
SELECT COUNT(NUPARCELA) as QTD_PARCELAS,
SUM(FL_PAGAS) AS QTD_PARCELAS_PAGAS,
NUCONTRATO,
sqcontratofinanceiro,
SUM(VRPRINCPAL_PARCELA_ABERTAS) AS VRPRINCPAL_PARCELA_ABERTAS,
SUM(VRPRINCPAL_PARCELA_ATRASADAS) as VRPRINCPAL_PARCELA_ATRASADAS
FROM ENRIQUECIMENTO_COM_AMORTIZACAO A
GROUP BY NUCONTRATO,
sqcontratofinanceiro
;QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_2 AS
SELECT
    NUCONTRATO,
    sqcontratofinanceiro,
	VRPRINCPAL_PARCELA_ABERTAS,
	VRPRINCPAL_PARCELA_ATRASADAS,
    QTD_PARCELAS,
    QTD_PARCELAS_PAGAS,
    
    ROUND(QTD_PARCELAS_PAGAS / QTD_PARCELAS, 0.01) AS pc_amort_parc,

    
    CASE
      WHEN CALCULATED pc_amort_parc = 0                                  THEN 'a. FPD'
      WHEN CALCULATED pc_amort_parc  > 0  AND CALCULATED pc_amort_parc <= 0.10 THEN 'b. até 10%'
      WHEN CALCULATED pc_amort_parc  > 0.10 AND CALCULATED pc_amort_parc <= 0.30 THEN 'c. 10% a 30%'
      WHEN CALCULATED pc_amort_parc >= 0.31 AND CALCULATED pc_amort_parc <= 0.70 THEN 'd. 31% a 70%'
      WHEN CALCULATED pc_amort_parc  > 0.70                               THEN 'e. >70%'
      ELSE 'N/A'
    END AS fx_pc_amort_parc
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_1
;
QUIT;

CREATE TABLE AMORTIZACAO_&SAFRA. AS

    NUCONTRATO,

	MAX(QTD_PARCELAS) AS QTD_PARCELAS,
	MAX(QTD_PARCELAS_PAGAS) AS QTD_PARCELAS_PAGAS,
	MAX(pc_amort_parc) AS pc_amort_parc,
    MAX(fx_pc_amort_parc) AS fx_pc_amort_parc,
	MAX(VRPRINCPAL_PARCELA_ABERTAS) AS VRPRINCPAL_PARCELA_ABERTAS,
	MAX(VRPRINCPAL_PARCELA_ATRASADAS) as VRPRINCPAL_PARCELA_ATRASADAS,
	&SAFRA. AS SAFRA
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_2
GROUP BY NUCONTRATO,sqcontratofinanceiro
;QUIT;
%MEND;

PROC SQL;

CREATE TABLE BASE_PAGAMENTOS AS
SELECT *
FROM CONNECTION TO perm
(SELECT CAST(z.CD_CONTRATO AS INT64) AS NUCONTRATO,
        z.SQ_CONTRATO_FINANCEIRO AS SQCONTRATOFINANCEIRO,
		z.Dt_Recebimento AS DtRecebimento,
		z.Vr_Principal_Devido AS VR_PRINCIPAL_DEVIDO,
		z.Vr_Principal_Baixado AS VR_PRINCIPAL_BAIXADO,
		EXTRACT(YEAR FROM DT_RECEBIMENTO) * 100 + EXTRACT(MONTH FROM DT_RECEBIMENTO) AS SafraAjuste
FROM  bv-pdbd-prd.dvry_core.tbparcelarecebimento AS z
WHERE DATE(z.Dt_Recebimento) >= &dt_inicio.
	  AND DATE(z.Dt_Recebimento) <= &dt_fim.
	  AND CAST(z.Vr_Recebido AS FLOAT64) NOT IN (0, 0.01)
      AND CAST(z.Fl_Estornado AS STRING) NOT IN ('S')

GROUP BY z.CD_CONTRATO, z.SQ_CONTRATO_FINANCEIRO, z.Nu_Parcela, z.Dt_Recebimento
, z.Vr_Principal_Devido
, z.Vr_Principal_Baixado
);
QUIT;

    %let origem   = %sysfunc(pathname(&libname.));
    %let destino  = %sysfunc(pathname(work));
    %let in_gz    = &destino./&arquivo..sas7bdat.gz;
    %let in_sas   = &destino./&arquivo..sas7bdat;

    
    %if %length(&out)=0 %then %let out = &arquivo._final;
    %let outds = work.&out.;

    %put >>> Copiando &arquivo. de &origem. para &destino.;
    
    x cp "&origem./&arquivo..sas7bdat.gz" "&destino./";

    
    %if %sysfunc(fileexist(&in_gz.)) = 0 %then %do;
        %put ERROR: Arquivo nao encontrado apos copia: &in_gz.;
        %return;
    %end;

    
    x gunzip -f "&in_gz.";

    
    %if %sysfunc(fileexist(&in_sas.)) = 0 %then %do;
        %put ERROR: Falha na descompactacao. Nao existe: &in_sas.;
        %return;
    %end;

    %put >>> Arquivo &arquivo. descompactado na WORK!;
%mend;

proc sql;
  create table SALDO_AJUIZAMENTO as
  select
    input(b.nucontrato_str, best32.)         as nucontrato format=14.,
	b.nudiasatraso,
	b.saldo,
    b.vrtaxacontrato,
    b.vrpermanenciaapropriado,
	b.sqcontratofinanceiro,
	b.dtprocessamento,
    a.NuScore,
    a.'Valor Score'n,
    a.'Data da Simulação'n,
    a.RESULTADO
  from process_gcon as b
  left join analitico_score as a
    on a.'Número Contrato'n = input(b.nucontrato_str, best32.)
	and a.'Sequencia Financeira'n = b.sqcontratofinanceiro
	where b.nudiasatraso > 0
	and a.'Data da Simulação'n = B.dt_modelo
  ;
quit;
