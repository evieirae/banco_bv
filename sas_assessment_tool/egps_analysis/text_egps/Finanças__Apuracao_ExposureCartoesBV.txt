				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 391 AND 420 THEN '16. Carteira 391 a 420'

				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 511 AND 540 THEN '20. Carteira 511 a 540'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) > 540               THEN '21. Carteira Acima 540'
				 END AS FaixaAtraso_Atual,
          sum(vr_saldo_carteira) as VrSaldoCarteira,
		  sum(vr_limite) as VrLimiteDisponivel,
		  sum(vr_saldo_carteira)+sum(vr_limite) AS VrExposureTotal,
		  sum(vr_provisao) as VrProvisaoSaldo,
		  sum(vr_provisao_limite) as VrProvisaoLimite,
		  sum(vr_provisao_total) as VrProvisaoTotal
   FROM bv-pdbd-prd.dvry_corp_riscocredat.provisao_varejo_fato as a
   LEFT JOIN
   (
   		SELECT dt_referencia,
		       cd_cpf_cnpj,
			   cd_produto_risco,
			   MAX(qt_dias_atraso) AS Qt_DiasAtraso
		FROM bv-pdbd-prd.dvry_corp_riscocredat.provisao_varejo_fato
        GROUP BY 1,2,3
   ) AS b
   ON a.dt_referencia = b.dt_referencia AND
      a.cd_cpf_cnpj = b.cd_cpf_cnpj     AND
      a.cd_produto_risco = b.cd_produto_risco
   WHERE cd_sistema_origem in ('113','114','115')
   GROUP BY 1,2,3
   ORDER BY 1 DESC
);
