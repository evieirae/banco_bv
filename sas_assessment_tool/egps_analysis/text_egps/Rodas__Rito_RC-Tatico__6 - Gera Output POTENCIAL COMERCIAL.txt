PROC SQL;
   CREATE TABLE WORK.BASE_POTENCIAL_COMERCIAL AS
   SELECT t1.NuAnoMes,
		  t1.Dtatualizacao,
          t1.TpBase,
          t1.FlAtivo,
          t1.NmCanalEstruturaComercialC,
          t1.NmGrupoProduto,
          t1.FxIdade,
		  t1.cdparceirocomercial,
          t1.NmTipoParceiroComercial,
          t1.NmSubEstruturaComercialC,
          t1.NmAreaEstruturaComercialC,
		  t1.NmColaboradorC,
		  t1.DsLoginN1C,
          t1.DsLoginN2C,
          t1.DsLoginN3C,
          t1.LstColaboradorFocal,
          t1.LstSubstituicaoC,
		  t1.NmAutoShopping,
		  t1.FlPorteA,
          t1.QtContratoBV,
          t1.QtGravamesMercado
      FROM tmp.TbPotencialComercial t1
      WHERE t1.TpBase = 1
      and nuanomes <= &mindata.;
QUIT;

data BASE_POTENCIAL_COMERCIAL_1;
    set BASE_POTENCIAL_COMERCIAL;
    /* Ajuste conforme sua variável de data */
   	data = input(put((nuAnoMes*100)+1, 8.), yymmdd8.);
    format data date9.;
    ano = year(data);
    trimestre = qtr(data);
    mes = month(data);
run;

proc sql print;
    select distinct cats(ano, trimestre)
    into :trimestres SEPARATED BY ' '
    from BASE_POTENCIAL_COMERCIAL_1
    order by ano desc, trimestre desc;
quit;

%let trim1 = %scan(&trimestres, 1);
%let trim2 = %scan(&trimestres, 2);
%let trim3 = %scan(&trimestres, 3);
%let trim4= %scan(&trimestres, 4);
%let trim_atual = %scan(&trimestres, 1);
%let trim_anterior = %scan(&trimestres, 2);

proc sql;
    create table RESUMO_TRI_POTENC_COMERC as
    select
		"BASE_TRI" AS TP_BASE,
        ano,
		. as NuAnoMes,
		ano*100+trimestre as NuAnoTri,
/*		put(ano*100+trimestre,best.)||"t" as NuAnoMes_t,*/
        trimestre,
		TpBase,
		FlAtivo,
		NmCanalEstruturaComercialC,
		NmGrupoProduto,
		FxIdade,
		NmTipoParceiroComercial,
		NmSubEstruturaComercialC,
		NmAreaEstruturaComercialC,
		NmColaboradorC,
		DsLoginN1C as DsLoginN1,
		DsLoginN2C as DsLoginN2,
		DsLoginN3C as DsLoginN3,
		LstColaboradorFocal,
		LstSubstituicaoC as LstSubstituicao,
		NmAutoShopping,
		FlPorteA,
		max(DtAtualizacao) as DtAtualizacao Format ddmmyy10.,
		count(distinct(cdparceirocomercial)) as Ttl_pdv,
		SUM(QtContratoBV) AS QtContratoBV FORMAT COMMAX12.2,
        SUM(QtGravamesMercado) AS QtGravamesMercado FORMAT COMMAX12.2
      FROM BASE_POTENCIAL_COMERCIAL_1
	   where cats(ano, trimestre) in ("&trim1", "&trim2","&trim3")
GROUP BY
        ano,
		ano*100+trimestre,
/*		put(ano*100+trimestre,best.)||"t" as NuAnoMes_t,*/
		trimestre,
		TpBase,
		FlAtivo,
		NmCanalEstruturaComercialC,
		DsLoginN1C,
		DsLoginN2C,
		DsLoginN3C,
		LstColaboradorFocal,
		LstSubstituicaoC,
		NmGrupoProduto,
		FxIdade,
		NmTipoParceiroComercial,
		NmSubEstruturaComercialC,
		NmAreaEstruturaComercialC,
		NmColaboradorC,
		NmAutoShopping,
		FlPorteA;
quit;
