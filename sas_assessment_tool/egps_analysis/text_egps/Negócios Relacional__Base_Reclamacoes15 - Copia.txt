PROC SQL;
CREATE TABLE CONTRATO_SEGURO_MAPFRE AS
	SELECT A.*,
	B.NUAPOLICESEGURADORA,
	B.DSTIPOSEGURO,
B.NmParceiroComercial AS SEGURADORA
	FROM TB_CONTRATO  A
	INNER JOIN TB_MAPFRE  B
  	ON A.NUCONTRATO = B.NUCONTRATO
   WHERE b.NUAPOLICESEGURADORA IS NOT NULL;
quit;



PROC SQL;
CREATE TABLE RECLAMACOES_MAPFRESAC_CONTRATO AS
	SELECT DISTINCT
			"MAPFRE" AS SEGURADORA,
			A.CPF_RECLAMANTE,
			C.IDPESSOA AS CPF_SEGURADO,
			A.APOLICE,
			NUCONTRATO AS NUCONTRATO,
		    DATA_DE_RECLAMACAO,
			
			CASE
				WHEN A.Ramo='GARANTIA MECÂNICA' THEN 'GARANTIA MECANICA'
				WHEN A.Ramo='AUTOMÓVEL' THEN 'SEGURO AUTO'
				WHEN A.Ramo='CARTÃO DE CRÉDITO' THEN 'CARTÃO PROTEGIDO'
				WHEN A.Ramo='GARANTIA MECANICA' THEN 'GARANTIA MECANICA'
				ELSE 'VERIFICAR'
			END AS PRODUTO,
			
			"SEGURADORA" AS CANAL,
			"SAC" AS SUBCANAL,
			"SAC" AS NÍVEL,
			B.'Motivo BV'N AS MOTIVO_DA_RECLAMACAO,
			B.'Submotivo BV'N AS SUBMOTIVO_DA_RECLAMACAO,
			A.CLASSIFICACAO_2 AS CAUSA_RAIZ
			
	FROM (SELECT * FROM RECLAMACOES_MAPFRESAC WHERE CLASSIFICACAO_3 IS NOT NULL) A
			LEFT JOIN (SELECT * FROM SEGUROS.DePara_Reclamacoes WHERE SEGURADORA='MAPFRE') B
					ON COMPRESS(UPCASE(A.CLASSIFICACAO_2)) = COMPRESS(UPCASE(B.'MOTIVO SEGURADORA'N))
					AND COMPRESS(UPCASE(A.CLASSIFICACAO_3)) = COMPRESS(UPCASE(B.'SUBMOTIVO SEGURADORA'N))
			LEFT JOIN CONTRATO_SEGURO_MAPFRE C
				ON C.NUAPOLICESEGURADORA=A.APOLICE
;QUIT;

PROC SQL;
CREATE TABLE RECLAMACOES_MAPFRE2INST_CONTRATO AS
	SELECT DISTINCT
			"MAPFRE" AS SEGURADORA,
			A.'CPF Reclamante'N AS CPF_RECLAMANTE,
			C.IDPESSOA AS CPF_SEGURADO,
			A.APOLICE,
			C.NUCONTRATO AS NUCONTRATO,
			DATEPART(A.'Data Abertura'N) AS DATA_DE_RECLAMACAO FORMAT=ddmmyy10.,
			CASE
				WHEN A.Produto IN('Automais BV', 'INDIVIDUAL', 'Automais Online', 'Mapfre Auto')
									THEN 'SEGURO AUTO'
				WHEN A.Produto in ('MASSIFICADOS ESPECIAIS GARANTIA MECÂNICA',
							    	'Garantia Mecânica e Elétrica',
									'Garantia Mecânica de Auto')
									THEN 'GARANTIA MECANICA'
				WHEN A.Produto IN ('MASSIFICADOS ESPECIAIS CARTÃO DE CRÉDITO',
									'Cartões',
									'Cartão de Crédito') THEN 'CARTÃO PROTEGIDO'

CREATE TABLE TEMPO_CANCELAMENTO AS
SELECT DISTINCT
	A.CPF_SEGURADO,
	B.DsImpressaoCet,
	B.DTEVENTOSEGURO,
	INTCK ('DAY', A.DATA_DE_RECLAMACAO, DATEPART(B.DTEVENTOSEGURO)) AS DIAS_CANC
FROM (SELECT * FROM RECLAMACOES WHERE CPF_SEGURADO IS NOT NULL) A
LEFT JOIN (SELECT * FROM SEGUROS.TBCANCELAMENTOS WHERE NMMOTIVOEVENTO IN
('CANCELAMENTO SOLICITADO PELO SEGURADO', 'CANCELAMENTO FEITO PELO CLIENTE VIA APP OU CENTRAL. VALOR DO SEGURO REMOVIDO DO CONTRATO')) B
	ON A.CPF_SEGURADO=INPUT(B.NuCpfCnpj,14.)
WHERE B.DTEVENTOSEGURO > A.DATA_DE_RECLAMACAO;

PROC SQL;
CREATE TABLE RECLAMACOES_CARDIF1_CONTRATO23 AS
	SELECT DISTINCT
			"CARDIF" AS SEGURADORA,
			A.CPF AS CPF_RECLAMANTE,
			INPUT(C.IDPESSOA,14.) AS CPF_SEGURADO,
			C.NUAPOLICESEGURADORA,
			A.APOLICE1,
			A.APOLICE2,
			A.APOLICE3,
			C.NUCONTRATO AS NUCONTRATO,
			Concatenado2 AS DATA_DE_RECLAMACAO,
			"SEGURO PRESTAMISTA" AS PRODUTO,
			"SEGURADORA" AS CANAL,
			CASE WHEN A.CANAL2='Consumidor.gov' THEN 'CONSUMIDOR.GOV'
			 	  WHEN (A.CANAL='Ouvidoria' AND A.CANAL2='e-mail Ouvidoria') THEN 'OUVIDORIA'
				  WHEN (A.CANAL='Ouvidoria' AND A.CANAL2='Notificação Extrajudicial') THEN 'OUVIDORIA'
				  WHEN (CANAL='Ouvidoria' AND A.CANAL2='Ouvidoria 0800') THEN 'OUVIDORIA'
			 	  WHEN (A.CANAL='Ouvidoria' AND A.CANAL2='Ouvidoria BNP') THEN 'OUVIDORIA'
			 	  WHEN (A.CANAL='Ouvidoria' AND A.CANAL2='Página de Ouvidoria') THEN 'OUVIDORIA'
			 	  WHEN (A.CANAL='Ouvidoria' AND A.CANAL2='Reclame Aqui') THEN 'OUVIDORIA'
			 	  WHEN A.CANAL='Procon' THEN 'PROCON'
			 	  WHEN A.CANAL='Reclame aqui' THEN 'RECLAME AQUI'
				  WHEN A.CANAL='Reclame Aqui' THEN 'RECLAME AQUI'
			 	  WHEN A.CANAL='Ação Judicial' THEN 'JURÍDICO'
			 	  WHEN A.CANAL='Call Center' THEN 'SAC'
			ELSE 'VERIFICAR'
			END AS SUBCANAL,
		    B.'Motivo BV'N AS MOTIVO_DA_RECLAMACAO,
			B.'Submotivo BV'N AS SUBMOTIVO_DA_RECLAMACAO,
			A.'Classificação Reclamação'N AS CAUSA_RAIZ
	FROM RECLAMACOES_CARDIF23 A
			LEFT JOIN (SELECT * FROM SEGUROS.DePara_Reclamacoes WHERE SEGURADORA='CARDIF') B
					ON COMPRESS(UPCASE(A.'Motivo Ajustado'N)) = COMPRESS(UPCASE(B.'MOTIVO SEGURADORA'N))
					AND COMPRESS(UPCASE(A.'Classificação Reclamação'N)) = COMPRESS(UPCASE(B.'SUBMOTIVO SEGURADORA'N))
			LEFT JOIN CONTRATO_SEGURO_CARDIF C
				ON INPUT(C.NUAPOLICESEGURADORA,20.)=INPUT(A.APOLICE1,20.)
				OR INPUT(C.NUAPOLICESEGURADORA,20.)=INPUT(A.APOLICE2,20.)
				OR INPUT(C.NUAPOLICESEGURADORA,20.)=INPUT(A.APOLICE3,20.)
;QUIT;
