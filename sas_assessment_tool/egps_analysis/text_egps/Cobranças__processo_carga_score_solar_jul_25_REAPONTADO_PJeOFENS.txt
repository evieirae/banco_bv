CREATE TABLE SOLAR_FLEX_4 AS
SELECT
	A.*,
	C.PERC_PAGO,


	C.PARCELA/E.RENDABV_2_0 AS COMP_RENDA,
	H.SAFRA,
CASE WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 120 AND (H.GH2 IS NULL OR H.GH2 >= 7) THEN "Bom"
	 WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 120 AND  H.GH2 >= 4 THEN "Regular"
	 WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 120 AND  H.GH2 <= 3 THEN "Ruim"
	 WHEN NUDIASATRASO > 120 AND NUDIASATRASO <= 180 AND (H.GH2 IS NULL OR H.GH2 >= 7) THEN "Bom"
	 WHEN NUDIASATRASO > 120 AND NUDIASATRASO <= 180 AND  H.GH2 >= 5 THEN "Regular"
	 WHEN NUDIASATRASO > 120 AND NUDIASATRASO <= 180 AND  H.GH2 <= 4 THEN "Ruim"
	 WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND (H.GH2 IS NULL OR H.GH2 >= 6) THEN "Bom"
	 WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND  H.GH2 >= 4 THEN "Regular"
	 WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND  H.GH2 <= 3 THEN "Ruim" END AS FAIXA_SCORE,
	G.VRFINANCIAMENTO,
	G.TIPO_PES,
	G.REGIAO_LOJA,
	G.UF_CLIENTE,
	CASE WHEN G.UF_CLIENTE = 'RS' AND NUDIASATRASO < 90 THEN 1 ELSE 0 END AS FLAG_ACAO_RS,
	H.GH2,
	M.GH2 AS GHCURTO,
	N.GH AS GHLONGO_NEW,
	CASE
        WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 180 AND (N.GH IS NULL OR N.GH >= 7) THEN 'Bom'
        WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 180 AND N.GH >= 5 THEN 'Regular'
        WHEN NUDIASATRASO > 59 AND NUDIASATRASO <= 180 AND N.GH <= 4 THEN 'Ruim'
        WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND (N.GH IS NULL OR N.GH >= 6) THEN 'Bom'
        WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND N.GH >= 4 THEN 'Regular'
        WHEN NUDIASATRASO > 180 AND NUDIASATRASO <= 360 AND N.GH <= 3 THEN 'Ruim'
    END AS FAIXA_SCORE2,
	I.GH2 AS GH_WO,
	CASE WHEN G.VRFINANCIAMENTO < 25000 THEN '1. < 25K'
		 WHEN G.VRFINANCIAMENTO < 50000 THEN '2. < 50K'
		 WHEN G.VRFINANCIAMENTO >= 50000 THEN '3. >= 50K'
		 END AS FX_FINANCIAMENTO,

	G.SAFRA_PROPOSTA_N AS SAFRA_CONCESSAO,

	CASE WHEN J.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FL_EARLY,
	CASE WHEN K.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FL_COBAN,
	CASE WHEN L.NUCONTRATO IS NOT NULL THEN 1 ELSE 0 END AS FL_BAD_BANK,
	CASE WHEN M.GH2 IS NULL THEN 'Nulo'

		 WHEN M.GH2 >= 6 THEN 'Bom'

O.STATUSINTEGRADOR

PROC SQL;
CREATE TABLE SOLAR_FLEX_9_AUX0 AS
	SELECT A.*
		,CASE WHEN B.DESC_PMT IS NOT NULL THEN B.DESC_PMT
		ELSE 0 END AS ULT_DESC_PMT
		, CASE WHEN B.DESC_ATUAL IS NOT NULL THEN B.DESC_ATUAL
		ELSE 0 END AS ULT_DESC_ATU
		, CASE WHEN B.DESC_QUIT IS NOT NULL THEN B.DESC_QUIT
		ELSE 0 END AS ULT_DESC_QUIT
		,CASE WHEN C.DESC_PMT IS NOT NULL THEN C.DESC_PMT
		ELSE 0 END AS ULT_MM_PMT
		, CASE WHEN C.DESC_ATUAL IS NOT NULL THEN C.DESC_ATUAL
		ELSE 0 END AS ULT_MM_ATU
		, CASE WHEN C.DESC_QUIT IS NOT NULL THEN C.DESC_QUIT
		ELSE 0 END AS ULT_MM_QUIT

	FROM SOLAR_FLEX_8 A
	LEFT JOIN ALCADAS_SOLAR_1 B
		ON A.ULT_ALCADA = B.ALCADA
		AND A.ATRASO_HJ >= B.ATRASO_MIN
		AND A.ATRASO_HJ <= B.ATRASO_MAX

	LEFT JOIN ALCADAS_SOLAR_MULTA_MORA C
		ON A.ULT_ALCADA = C.ALCADA
		AND A.ATRASO_HJ >= C.ATRASO_MIN
		AND A.ATRASO_HJ <= C.ATRASO_MAX;
RUN;

PROC SQL;
CREATE TABLE DESC_FLEX_3 AS
	SELECT *
		, CASE WHEN ELEG_DESC_PMT < ULT_DESC_PMT THEN -1
				WHEN ELEG_DESC_PMT = ULT_DESC_PMT THEN 0
				WHEN ELEG_DESC_PMT > ULT_DESC_PMT THEN 1 END AS FL_PMT
		, CASE WHEN ELEG_DESC_ATU < ULT_DESC_ATU THEN -1
				WHEN ELEG_DESC_ATU = ULT_DESC_ATU THEN 0
				WHEN ELEG_DESC_ATU > ULT_DESC_ATU THEN 1 END AS FL_ATUALIZACAO
		, CASE WHEN ELEG_DESC_QUIT < ULT_DESC_QUIT THEN -1
				WHEN ELEG_DESC_QUIT = ULT_DESC_QUIT THEN 0
				WHEN ELEG_DESC_QUIT > ULT_DESC_QUIT THEN 1 END AS FL_QUITACAO
		, CASE WHEN ELEG_MM_PMT < ULT_MM_PMT THEN -1
				WHEN ELEG_MM_PMT = ULT_MM_PMT THEN 0
				WHEN ELEG_MM_PMT > ULT_MM_PMT THEN 1 END AS FL_PMT_MM
		, CASE WHEN ELEG_MM_ATU < ULT_MM_ATU THEN -1

				WHEN ELEG_MM_ATU > ULT_MM_ATU THEN 1 END AS FL_ATUALIZACAO_MM
		, CASE WHEN ELEG_MM_QUIT < ULT_MM_QUIT THEN -1
				WHEN ELEG_MM_QUIT = ULT_MM_QUIT THEN 0
				WHEN ELEG_MM_QUIT > ULT_MM_QUIT THEN 1 END AS FL_QUITACAO_MM
	FROM SOLAR_FLEX_9_AUX1

proc sql;
    create table Percent_Missing as
    select
        DTPROCESSAMENTO,
        mean(missing(SAFRA_CONCESSAO)) as Perc_Missing_SAFRA_CONCESSAO,
        mean(missing(REGIAO_LOJA)) as Perc_Missing_REGIAO_LOJA,
        mean(missing(PERC_PAGO)) as Perc_Missing_PERC_PAGO,
        mean(missing(NUDIASATRASO)) as Perc_Missing_NUDIASATRASO,
        mean(missing(ATRASO_HJ)) as Perc_Missing_ATRASO_HJ,
        mean(missing(UF_CLIENTE)) as Perc_Missing_UF_CLIENTE,
        mean(missing(COMP_RENDA)) as Perc_Missing_COMP_RENDA
    from EMP_MARCACOES
    group by DTPROCESSAMENTO;
quit;

proc sql;

    select
        DTPROCESSAMENTO,
        mean(missing(GH2)) as Missing_GH2
    from EMP_MARCACOES

    from EMP_MARCACOES
	where NUDIASATRASO >= 360
    group by DTPROCESSAMENTO;

	where NUDIASATRASO < 120

LEFT JOIN Missing_GH_WO AS C

PROC SQL;
CREATE TABLE INFO_CYBER AS
SELECT
A.*,
B.CPFCNPJ,
B.ATRASO,
B.COD_ASSESSORIA,
B.RISCO,
B.ALCADA AS ALCADA_SCORE,
B.SCORE AS SCORE_CYBER,
E.NUDIASATRASO AS ATRASO_ATUALIZADO,
C.ACMAXDISC1 AS DESCONTO_POL,
D.ACMAXDISC1 AS DESCONTO_REAL,
C.ACMAXDISC1 * A.RISCO_ENVIO AS DESC_POL_RISCO,
D.ACMAXDISC1 * A.RISCO_ENVIO AS DESC_CYB_RISCO,
CASE WHEN A.ATRASO_HJ <= 15 THEN '0001 A 0015'
	 WHEN A.ATRASO_HJ > 15 AND ATRASO_HJ <= 90 THEN '0016 A 0090'
	 WHEN A.ATRASO_HJ > 90 AND ATRASO_HJ <= 420 THEN '0091 A 0420'
	 WHEN A.ATRASO_HJ > 420 THEN '0421 A 9999'
	 ELSE '' END AS FAIXA_ATRASO,

CASE WHEN B.COD_ASSESSORIA IN (1648, 1649) THEN 'TOLEDO'
	 WHEN B.COD_ASSESSORIA IN (2031, 2033) THEN 'BELLINATI'
	 WHEN B.COD_ASSESSORIA IN (1592, 1593) THEN 'RN' ELSE 'OUTRO' END AS ASSESSORIA,
CASE WHEN E.NUDIASATRASO < 15 THEN 2
	 WHEN B.CONTRATO IS NULL THEN 1
	 ELSE 0 END AS FORA_CYBER,
CASE
	 WHEN C.ACMAXDISC1 > D.ACMAXDISC1 THEN 'DESCONTO_ABAIXO'
	 WHEN C.ACMAXDISC1 < D.ACMAXDISC1 THEN 'DESCONTO_ACIMA'
	 WHEN C.ACMAXDISC1 = D.ACMAXDISC1 THEN 'DESCONTO_IDEAL'
	 ELSE 'ERRO' END AS SITUACAO_DESCONTO
FROM MARCACAR_&REFNAME. AS A
LEFT JOIN CARTEIRA_CYBER_&dt_atualizacao. AS B ON
	A.NUCONTRATO = B.CONTRATO
LEFT JOIN (SELECT * FROM LISTA_ALCADAS_20250212 WHERE ACLABEL = 'PARCELA' AND AGPLANO = 1) AS C
	ON A.FINALPOLITICA = C.AQQUE
	AND A.ATRASO_HJ >= C.AGDMDAYSMIN
	AND A.ATRASO_HJ <= C.AGDMDAYSMAX
LEFT JOIN (SELECT * FROM LISTA_ALCADAS_20250212 WHERE ACLABEL = 'PARCELA' AND AGPLANO = 1) AS D
	ON B.ALCADA = D.AQQUE
	AND A.ATRASO_HJ >= D.AGDMDAYSMIN
	AND A.ATRASO_HJ <= D.AGDMDAYSMAX
LEFT JOIN ATRASO_ATUAL AS E
	ON A.NUCONTRATO = E.NUCONTRATO;
QUIT;

BY NUCONTRATO ATRASO_HJ FL_BASE;
