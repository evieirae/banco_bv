PROC SQL;
CREATE TABLE TbPagamentoEmExcecao AS
SELECT
	 Producao.DtContrato
	,Producao.DtContrato AS DtIncrementalPBI
	,Producao.NuAnoMes
	,Producao.NuProposta
	,Producao.NuContrato
	,Producao.DsTipoCanalProducao
	,Producao.NmCanalEstruturaComercial
	,Producao.CdFilial
	,Producao.DsFilial
	,Producao.NmAreaEstruturaComercial
	,Producao.NmSubEstruturaComercialA
	,Producao.NmGrupoProdutoMeta
	,Producao.CdParceiroComercial
	,Producao.NmParceiroComercial
	,Producao.CdColaborador
	,Producao.NmColaborador
	,Producao.DsLoginN1
	,Producao.DsLoginN2A
	,Producao.DsLoginN3A
	,Producao.FlContratoExcecao
	,Producao.CdDocumentoAlienacao
	,Producao.StDocumentoAlienacao
	,Producao.VrProducao

	,TbBaseExcecao.NuContratoGestao
	,TbBaseExcecao.NuProposta AS NuPropostaExcecao
	,TbBaseExcecao.DtContrato AS DtContratoExcecao
	,TbBaseExcecao.CdDocumentoAlienacao
	,TbBaseExcecao.Motivo_Alien
	,TbBaseExcecao.SitPrazo
	,TbBaseExcecao.Abono
	,TbBaseExcecao.SituacaoB3
	,TbBaseExcecao.SituacaoB3Pend
	,TbBaseExcecao.SituacaoB3Dt

	,ExtContrato.NmPessoa
	,ExtContrato.AaFabricacao
	,ExtContrato.AaModelo
	,ExtContrato.CdMarca
	,ExtContrato.CdModelo
	,ExtContrato.DsPlaca

	,ColaboradorFocal.LstColaboradorFocal

FROM Bases.TbProducao Producao
	LEFT JOIN Basegrc.TbBaseExcecao TbBaseExcecao
		ON Producao.NuContrato = TbBaseExcecao.NuContratoGestao

	LEFT JOIN basegrc.TbColaboradorFocal ColaboradorFocal
		on  Producao.CdFilial = ColaboradorFocal.CdFilial
		and Producao.NmGrupoProdutoMeta = ColaboradorFocal.NmGrupoProdutoMeta

	LEFT JOIN bases.TbExtContrato ExtContrato
		ON Producao.NuContrato = ExtContrato.NuContrato

WHERE Producao.NuAnoMes >= &safraM24.
/*AND Producao.NuContrato IN (SELECT NuContratoGestao FROM Basegrc.TbBaseExcecao)*/
AND Producao.NmGrupoProdutoMeta IN ('CVG Leves','CVG Pesados','Leves','Leves-Multibanco','Motos','Pesados')
AND ExtContrato.StProducao = 'S'
;run;

proc export
data=TbPagamentoEmExcecao
file='/riscosbvf/BA/processos/tmp/TbPagamentoEmExcecao.csv' replace
dbms=dlm;
delimiter=';';
run;
