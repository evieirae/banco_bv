PROC SQL;

CREATE TABLE work.REVELIAS AS

SELECT

 r.Cd_Causa 									as CD_CAUSA
,r.CdSistema									AS CD_SISTEMA
,c.DtCadastramento 								as DT_CADASTRAMENTO
,UPPER(c.NmRecurso) 							as RECURSO
,c.NMMOTIVORRO									as NM_MOTIVO_RRO
,c.NmMotivo 									as MOTIVO
,c.NmSubMotivo									as SUBMOTIVO
,c.NmAcaoJudicial 								as ACAO_JUDICIAL
,UPPER(c.NmMunicipio) 							as MUNICIPIO
,c.CdUF 										as UF
,CASE
	   WHEN c.NMMOTIVORRO  IN ("CET" , "NOVA CET" , "TARIDAS") THEN "TARIFAS"
 	   WHEN c.NMMOTIVORRO  IN ("FALHA OPERACIONAL 1" , "FALHA OPERACIONAL 2" , "INDENIZATÓRIAS","FALHAS OPERACIONAIS","DIVERSOS" ,"DEMAIS" , "FRAUDES" ,"NEGATIVAÇÃO INDEVIDA" , "") THEN "INDENIZATÓRIA"
  	   WHEN c.NMMOTIVORRO  IN ("REVISIONAIS") AND c.FLASSESSORIACOBRANCA = 1  THEN "REVISIONAL COBRANÇA"
       WHEN c.NMMOTIVORRO  IN ("REVISIONAIS") AND c.FLASSESSORIACOBRANCA <> 1 THEN "REVISIONAL JURIDICO"
       WHEN c.NmMotivo     IN ('DESCUMPRIMENTO DE DETERMINAÇÃO JUDICIAL')  AND c.NMMOTIVORRO = "" THEN "INDENIZATÓRIA"
       END
												as TIPO_MOTIVO_FINAL
,CASE
	   WHEN c.CdUF IN ('SP') 								THEN 'SP'
       WHEN c.CdUF IN ('CE') 								THEN 'CE'
       WHEN c.CdUF IN ('PE','RN') 							THEN 'PE-RN'
       WHEN c.CdUF IN ('BA') 								THEN 'BA'
       WHEN c.CdUF IN ('SE') 								THEN 'SE'
       WHEN c.CdUF IN ('MA','PB','PI') 						THEN 'MA-PB-PI'
       WHEN c.CdUF IN ('AL') 								THEN 'AL'
       WHEN c.CdUF IN ('RO','AP','RR','TO','PA','AC','AM') 	THEN 'NORTE'
       WHEN c.CdUF IN ('MG') 								THEN 'MG'
       WHEN c.CdUF IN ('MS') 								THEN 'MS'
       WHEN c.CdUF IN ('MT') 								THEN 'MT'
       WHEN c.CdUF IN ('GO','DF') 							THEN 'GO-DF'
       WHEN c.CdUF IN ('ES') 								THEN 'ES'
       WHEN c.CdUF IN ('RJ') 								THEN 'RJ'
       WHEN c.CdUF IN ('RS') 								THEN 'RS'
       WHEN c.CdUF IN ('PR','SC') 							THEN 'PR-SC'
       				END
												as REGIAO
,c.DtEncerramento 								as DT_ENCERRAMENTO
,c.NmTipoEncerramento							as TIPO_ENCERRAMENTO
,c.NmMotivoEncerramento							as MOTIVO_ENCERRAMENTO
,r.Fase											as FASE
,r.DtAndamento 									as DT_ANDAMENTO
,r.SAFRA_CADASTRO								as SAFRA_CADASTRO
,c.NmAssessoria 								as ASSESSORIA
,c.VrIndenizacao								as VR_INDENIZACAO
,CASE
		WHEN c.NMMOTIVORRO IN ("FALHA OPERACIONAL 1" , "FALHA OPERACIONAL 2" , "INDENIZATÓRIAS","FALHAS OPERACIONAIS","DIVERSOS" ,"DEMAIS" , "FRAUDES" ,"NEGATIVAÇÃO INDEVIDA", "", " ") THEN 'INDENIZATÓRIA'
		WHEN c.NMMOTIVORRO IS NULL THEN 'REVISIONAIS'
		WHEN c.NMMOTIVORRO IN ("CET" , "NOVA CET" , "TARIDAS" , "TARIFAS") THEN 'TARIFAS'
		WHEN c.NMMOTIVORRO IN ('REVISIONAIS') THEN 'REVISIONAIS'
		END
												as TIPO_MOTIVO_GRUPO
,today() FORMAT=DDMMYY10. 						as DT_PROCESSAMENTO
,c.FLACAOATIVA									AS FL_ACAO_ATIVA
,case
	when c.FlAcaoAtiva =1 then 'ATIVA'
	when c.FlAcaoAtiva =0 then 'ENCERRADA'
	else 'N/D' end
						  						as DS_ACAO_ATIVA

FROM      work.REVELIAS          as r
LEFT JOIN DBEP.VWTBLCAUSACONS as c
ON c.CdCausa = r.CD_CAUSA
and c.CdSistema = r.CdSistema

WHERE r.CD_CAUSA IS NOT NULL

;QUIT;

PROC SORT DATA= work.REVELIAS 		  ; BY CD_CAUSA CD_SISTEMA ;RUN;
PROC SORT DATA= work.REVELIAS NODUPKEY; BY CD_CAUSA CD_SISTEMA ;RUN;
