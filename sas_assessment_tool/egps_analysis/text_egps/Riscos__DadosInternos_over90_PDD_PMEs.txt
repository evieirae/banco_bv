%macro teste(anomes);

proc sql;
create table tboperacaoifrs9_final2_&anomes. as
select *,
b.Modelo as ModeloIF,
c.Modelo as ModeloPME,
d.Modelo as ModeloCIB,

(case when a.Modelo = 'Missing' then

(case when &anomes.<202001 then coalescec(b.Modelo,d.Modelo,a.Modelo)
		else coalescec(c.Modelo,b.Modelo,d.Modelo,a.Modelo) end)

else a.Modelo end) as Modelo2

from tboperacaoifrs9_final_&anomes. as a
left join (select distinct * from marcaCIBIFPME where Modelo='IF') as b
on a.cdClienteRisco=b.cdClienteRisco
left join (select distinct * from marcaCIBIFPME where Modelo='PME') as c
on a.cdClienteRisco=c.cdClienteRisco
left join (select distinct * from marcaCIBIFPME where Modelo='CIB') as d
on a.cdClienteRisco=d.cdClienteRisco;
quit;
