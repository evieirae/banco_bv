PROC SQL;
%CONNECT_GCP;
CREATE TABLE BASE_FPD AS
SELECT DISTINCT

*
,CASE WHEN DIAS_REG <= 0 THEN 0 ELSE DIAS_REG END AS DIAS_REG2

,day(dt_vencimento+5) as eleg_entrada

,CASE WHEN DIAS_REG <= 0 THEN 0 ELSE 1 END AS FL_ATRASO

,YEAR(dt_vencimento)*100+MONTH(dt_vencimento) AS safra_venc

,YEAR(dt_vencimento+5)*100+MONTH(dt_vencimento+5) AS safra_competencia

,CASE WHEN CALCULATED DIAS_REG2 >= 5 THEN 1 ELSE 0 END AS FL_ENTRADA_COB

FROM CONNECTION TO PERM(

SELECT DISTINCT
    a.cd_contrato,
    a.sq_contrato_financeiro,
    a.dt_vencimento,
    a.dt_ultimo_recebimento,
    a.st_parcela,
    a.vr_parcela,
	a.nu_parcela,
	c.saldo,
	CASE
        WHEN (a.st_parcela = 'Q' OR a.dt_ultimo_recebimento IS NOT NULL) THEN (DATE_DIFF(a.dt_ultimo_recebimento, a.dt_vencimento, DAY))
		ELSE 9999 END AS DIAS_REG,

	CASE
		 WHEN (DATE_DIFF(CURRENT_DATE()-1, e.min_dt_vencimento, DAY))>=5 THEN 1
	ELSE 0 END AS FL_MATUROU

FROM
    bv-pdbd-prd.dvry_core.tbparcela A

LEFT JOIN
	bv-pdbd-prd.dvry_core.tbcontrato AS B ON a.cd_contrato = b.cd_contrato
LEFT JOIN
    bv-pdbd-prd.trusted_dbcred.tbproposta AS D ON a.cd_contrato = d.nucontratogestao
LEFT JOIN
	(select cd_contrato, sq_contrato_financeiro, sum(vr_parcela) as saldo
		from bv-pdbd-prd.dvry_core.tbparcela
	 group by cd_contrato, sq_contrato_financeiro) AS C ON a.cd_contrato = c.cd_contrato and a.sq_contrato_financeiro = c.sq_contrato_financeiro
LEFT JOIN
	(select cd_contrato, sq_contrato_financeiro, min(dt_vencimento) as min_dt_vencimento
		from bv-pdbd-prd.dvry_core.tbparcela
		group by cd_contrato, sq_contrato_financeiro) AS E ON a.cd_contrato = e.cd_contrato and a.sq_contrato_financeiro = e.sq_contrato_financeiro

WHERE (CAST(b.cd_produto AS INT64) IN (13)
	  AND (CAST(b.cd_modalidade_produto AS INT64) = 55 OR d.cdsubproduto = 70))

)
group by
	cd_contrato,
    sq_contrato_financeiro

;quit;

PROC SORT DATA= BASE_FPD;
BY cd_contrato sq_contrato_financeiro safra_competencia DESCENDING FL_ENTRADA_COB;
RUN;

PROC SORT DATA= BASE_FPD NODUPKEY;
BY cd_contrato sq_contrato_financeiro safra_competencia;
RUN;

weekday(b.dt_vencimento) as cd_dia_da_semana,
eleg_entrada,
day(b.dt_vencimento) as DIA_VENC,
a.SALDO_2,
put(a.SALDO, FAIXA_SALDO_B.) AS FAIXA_SALDO2,
put(C.vrfinanciamento, FAIXA_SALDO_B.) AS FAIXA_VALOR_FINANCIADO2,
put(C.vrfinanciamento, FAIXA_SALDO.) AS FAIXA_VALOR_FINANCIADO,

CASE
		WHEN ATRASO_M0 <= 0 THEN 0
		WHEN ATRASO_M0 BETWEEN 1 AND 30 THEN 1
		WHEN ATRASO_M0 BETWEEN 31 AND 60 THEN 2
		WHEN ATRASO_M0 BETWEEN 61 AND 90 THEN 3
		WHEN ATRASO_M0 BETWEEN 91 AND 150 THEN 4
        WHEN ATRASO_M0 BETWEEN 91 AND 150 THEN 5
        WHEN ATRASO_M0 BETWEEN 151 AND 180 THEN 6
        WHEN ATRASO_M0 BETWEEN 181 AND 210 THEN 7
        WHEN ATRASO_M0 BETWEEN 211 AND 270 THEN 8
		ELSE 9
    END AS cd_faixa_m0,

CASE
		WHEN ATRASO_M1 <= 0 THEN 0
		WHEN ATRASO_M1 BETWEEN 1 AND 30 THEN 1
		WHEN ATRASO_M1 BETWEEN 31 AND 60 THEN 2
		WHEN ATRASO_M1 BETWEEN 61 AND 90 THEN 3
		WHEN ATRASO_M1 BETWEEN 91 AND 150 THEN 4
        WHEN ATRASO_M1 BETWEEN 91 AND 150 THEN 5
        WHEN ATRASO_M1 BETWEEN 151 AND 180 THEN 6
        WHEN ATRASO_M1 BETWEEN 181 AND 210 THEN 7
        WHEN ATRASO_M1 BETWEEN 211 AND 270 THEN 8
		ELSE 9
    END AS cd_faixa_M1,

CASE WHEN CALCULATED cd_faixa_M1 < CALCULATED cd_faixa_M0 THEN 'ROLAGEM'
	 WHEN CALCULATED cd_faixa_M1 = CALCULATED cd_faixa_M0 THEN 'MANUTENÇÃO'
	 WHEN CALCULATED cd_faixa_M1 > CALCULATED cd_faixa_M0 THEN 'RETROCESSO'
	 ELSE 'AVERIGUAR' END AS TIPO_ROLAGEM,

B.DIAS_REG2,

CASE WHEN B.NU_PARCELA = 1 THEN 1 ELSE 0 END AS FL_1PMT,
CASE WHEN B.DIAS_REG2 >= 5 THEN 1 ELSE 0 END AS FL_ENTRADA_COB,

C.SAFRA_PROPOSTA,
C.CanalProducaoSolar

FROM carteira_emp A

LEFT JOIN BASE_FPD B ON A.NUCONTRATO = INPUT(B.CD_CONTRATO,14.) AND
						A.SQCONTRATOFINANCEIRO = B.SQ_CONTRATO_FINANCEIRO AND
						A.SAFRA = B.safra_competencia

LEFT JOIN base_proposta c on a.nucontrato = c.nucontrato

;QUIT;
