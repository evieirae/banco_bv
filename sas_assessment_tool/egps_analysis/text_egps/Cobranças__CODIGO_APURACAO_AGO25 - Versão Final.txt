%MACRO CICLO_M0;

%let NUM_ELEMENTOS = %sysfunc(COUNTW(&LISTA_MES.), %str(*));

%do i = 1 %to &NUM_ELEMENTOS.;

%let REF_TAB = %scan(&LISTA_MES., &i., *);

DATA _NULL_ ;
    MES_M0 = INPUT(scan("&LISTA_MES.", &i., "*"), YYMMDD10.);
    MES_M0_PLUS_1 = INTNX("MONTH", MES_M0, 1, "E");
    SAFRA_M0 = INTNX("MONTH", MES_M0, 0, "E");

    IF MES_M0_PLUS_1 >= TODAY() THEN
        MES_M1 = TODAY()-1;
    ELSE
        MES_M1 = MES_M0_PLUS_1;

    CALL SYMPUTX("REF_M0", "'" || PUT(MES_M0, YYMMDD10.) || "'" );  
    CALL SYMPUTX("REF_M1", "'" || PUT(MES_M1, YYMMDD10.) || "'" );  
    CALL SYMPUTX("SAFRA_M0", PUT(SAFRA_M0, YYMMN6.));
    CALL SYMPUTX("SAFRA_M1", PUT(MES_M1, YYMMN6.));

RUN;

CREATE TABLE BASE_CONTABIL_&REF_TAB. AS
SELECT DISTINCT

A.*
,MDY(INPUT(SUBSTR(A.REF_M0,5,2),2.),INPUT(SUBSTR(A.REF_M0,7,2),2.),INPUT(SUBSTR(A.REF_M0,1,4),4.)) FORMAT DATE9. AS DATA_INICIO
,MDY(INPUT(SUBSTR(A.REF_M1,5,2),2.),INPUT(SUBSTR(A.REF_M1,7,2),2.),INPUT(SUBSTR(A.REF_M1,1,4),4.)) FORMAT DATE9. AS DATA_FIM
,INTCK('DAY',CALCULATED DATA_INICIO, CALCULATED DATA_FIM) AS DIF_DIAS
,B.PDD_M1
,B.ATRASO_M1
,B.SALDO_M1

,DATEPART(B.DTPROCESSAMENTOLIQUIDACAO) FORMAT DATE9. AS DTPROCESSAMENTOLIQUIDACAO
,DATEPART(B.DTCESSAO) FORMAT DATE9. AS DTCESSAO_M1
,DATEPART(D.DT_BAIXA) FORMAT DATE9. AS DTRENEG

,DATEPART(C.DTPROCESSAMENTO) FORMAT DATE9. AS DTRECEBIMENTOPROCESSAMENTO
,DATEPART(C.DTRECEBIMENTO) FORMAT DATE9. AS DTRECEBIMENTO
,INTCK('DAY', CALCULATED DTRECEBIMENTOPROCESSAMENTO, CALCULATED DTRECEBIMENTO) AS DIFF_DIAS_PAGAMENTO

,C.VRULTRECEBIMENTO
,C.VRPRINCIPALDEVIDO
,CASE WHEN C1.FLFORAALCADA = 'S' THEN 1 ELSE 0 END AS FLFORAALCADA

,CASE WHEN (B.ATRASO_M1 - A.QTATRASO) < 0 THEN 0 ELSE B.ATRASO_M1 - A.QTATRASO END AS DIF_DIAS_M1_M0

,CASE WHEN G1.NUCONTRATO IS NOT MISSING THEN 1 ELSE 0 END AS FLAG_VENDA

,CASE WHEN E.TOTALDEBITO  IS MISSING THEN 0 ELSE E.TOTALDEBITO END AS TOTALDEBITO
,CASE WHEN E.TOTALCREDITO IS MISSING THEN 0 ELSE E.TOTALCREDITO END AS TOTALCREDITO
,CALCULATED TOTALDEBITO - CALCULATED TOTALCREDITO AS DESCONTO_REALIZADO

,F.CART_2
,G.CLIENTE_50K


,G4.GRUPO_TEST
,G4.ACAO
,G4.TESTE_CONTROLE
,G4.DTPROCESSAMENTO AS DTPROCESSAMENTO_ACAO
,G4.ATRASO AS ATRASO_TESTE

,CASE WHEN G4.GRUPO_TEST <> '' THEN (CASE WHEN B.ATRASO_M1 <= G4.ATRASO THEN 1 ELSE 0 END) ELSE 1 END AS FLAG_ELEGIVEL_PAGAMENTO

,CASE WHEN CALCULATED DTPROCESSAMENTOLIQUIDACAO >= CALCULATED DATA_INICIO
            AND CALCULATED DTPROCESSAMENTOLIQUIDACAO <=  CALCULATED DATA_FIM                                                                                   THEN 'QUITACAO'

	  WHEN B.DTCESSAO >= CALCULATED DATA_INICIO
            AND B.DTCESSAO <= CALCULATED DATA_FIM                                                                                                              THEN 'CESSAO'

	  WHEN D.DT_BAIXA IS NOT MISSING                                                                                                                           THEN 'RENEGOCIACAO'


	  WHEN A.QTATRASO <= 30 THEN (
      CASE WHEN (B.ATRASO_M1 <= CALCULATED DIF_DIAS AND CALCULATED DIF_DIAS_M1_M0 < CALCULATED DIF_DIAS)
                 AND CALCULATED FLAG_ELEGIVEL_PAGAMENTO = 1 THEN 'ATUALIZACAO'
           WHEN B.ATRASO_M1 > CALCULATED DIF_DIAS AND CALCULATED DIF_DIAS_M1_M0 < CALCULATED DIF_DIAS AND CALCULATED FLAG_ELEGIVEL_PAGAMENTO = 1 THEN 'PAGAMENTO PARCIAL' ELSE 'SEM ACAO' END)

      WHEN A.QTATRASO > 30 THEN (
      CASE WHEN (B.ATRASO_M1 <= 30 AND CALCULATED DIF_DIAS_M1_M0 < CALCULATED DIF_DIAS)
                 AND CALCULATED FLAG_ELEGIVEL_PAGAMENTO = 1  THEN 'ATUALIZACAO'
           WHEN B.ATRASO_M1 > 30  AND CALCULATED DIF_DIAS_M1_M0 < CALCULATED DIF_DIAS AND CALCULATED FLAG_ELEGIVEL_PAGAMENTO = 1 THEN 'PAGAMENTO PARCIAL' ELSE 'SEM ACAO' END)

      ELSE 'SEM ACAO' END AS TIPO_PAGAMENTO

FROM BASE_M0_&REF_TAB. A
LEFT JOIN connection to perm(
    SELECT DISTINCT
        A.NUCONTRATO,
        A.SQCONTRATOFINANCEIRO,
        CAST(A.VRPDD AS FLOAT64) AS PDD_M1,
        CAST(A.NUDIASATRASO AS INT64) AS ATRASO_M1,
        (SAFE_CAST(A.VRPERMANENCIAAPROPRIADO AS FLOAT64) + SAFE_CAST(A.VRTAXACONTRATO AS FLOAT64)) AS SALDO_M1,
        A.DTPROCESSAMENTOLIQUIDACAO,
        A.DTCESSAO
    FROM `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` AS A
    WHERE DATE(A.DTPROCESSAMENTO) = DATE (&REF_M1.)
    AND A.CDCARTEIRA IN (1,3,4,5,12)

   ) B ON
A.NUCONTRATO = INPUT(B.NUCONTRATO,14.)

LEFT JOIN connection to perm(
    SELECT DISTINCT
        A.CD_CONTRATO,
        A.SQ_CONTRATO_FINANCEIRO,
        A.DT_PROCESSAMENTO AS DTPROCESSAMENTO,
        A.DT_RECEBIMENTO   AS DTRECEBIMENTO,
        SUM(A.VR_RECEBIDO) AS VRULTRECEBIMENTO,
        SUM(A.VR_PRINCIPAL_DEVIDO) AS VRPRINCIPALDEVIDO

    FROM `bv-pdbd-prd.dvry_core.tbparcelarecebimento` AS A
    WHERE DATE(A.DT_PROCESSAMENTO) > DATE (&REF_M0.)
      AND DATE(A.DT_PROCESSAMENTO) <= DATE (&REF_M1.)

    GROUP BY A.CD_CONTRATO, A.SQ_CONTRATO_FINANCEIRO, A.DT_PROCESSAMENTO, A.DT_RECEBIMENTO) C ON

A.NUCONTRATO = INPUT(C.CD_CONTRATO,14.)
AND A.SQCONTRATOFINANCEIRO = C.SQ_CONTRATO_FINANCEIRO

LEFT JOIN connection to perm(
    SELECT DISTINCT
	C.NUCONTRATOGESTAO
	,C.SQCONTRATOFINANCEIRO
	,C.VRBOLETO
	,C.DTGERACAO
	,C.DTPAGAMENTO
	,C.STNEGOCIACAO
	,C.FLFORAALCADA

    FROM bv-pdbd-prd.trusted_dbcontencioso.tbnegociacaocontrato C
	WHERE C.DTPAGAMENTO > &REF_M0. AND C.DTPAGAMENTO <= &REF_M1.
	AND C.STNEGOCIACAO IN('LP', 'PL')) C1 ON

C.CD_CONTRATO = C1.NUCONTRATOGESTAO
AND C.SQ_CONTRATO_FINANCEIRO = C1.SQCONTRATOFINANCEIRO
AND C.DTRECEBIMENTO = C1.DTPAGAMENTO

LEFT JOIN connection to perm(
    SELECT DISTINCT
        NUCONTRATOGESTAO,
        SQCONTRATOFINANCEIRO,
        DT_BAIXA,
        COUNT(SQCONTRATOFINANCEIRO) AS QTDE
    FROM `bv-pdbd-prd.trusted_dbreneg.tb_reneg_simulador`
    WHERE DATE(DT_BAIXA) > DATE &REF_M0.
      AND DATE(DT_BAIXA) <= DATE &REF_M1.
      AND DT_CANCELAMENTO IS NULL
      AND ST_RENEG = 3
    GROUP BY NUCONTRATOGESTAO, SQCONTRATOFINANCEIRO, DT_BAIXA) D ON
A.NUCONTRATO = INPUT(D.NUCONTRATOGESTAO,14.)
AND A.SQCONTRATOFINANCEIRO = D.SQCONTRATOFINANCEIRO

LEFT JOIN BASE_DESCONTO_&REF_TAB. E ON
A.NUCONTRATO = E.NUCONTRATO
AND E.DTPROCESSAMENTO = C.DTPROCESSAMENTO

LEFT JOIN JOAO.CARTEIRA_202504_BRUNO F ON
A.NUCONTRATO = F.NUCONTRATO

LEFT JOIN PRETER.BASE_COBRANCA_202503 G ON
A.NUCONTRATO = INPUT(G.NUCONTRATO,14.)

LEFT JOIN LDUARTE.BASE_VENDA_RODAS_2024 G1 ON
A.NUCONTRATO = G1.NUCONTRATO
AND A.REF_M1 = G1.SAFRA

LEFT JOIN BASE_CONSOLIDA_CAMPS G4 ON
A.NUCONTRATO = G4.NUCONTRATO

GROUP BY A.NUCONTRATO
HAVING B.ATRASO_M1 = MAX(B.ATRASO_M1)

;
disconnect from perm;
