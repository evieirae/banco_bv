PROC SQL;
CREATE TABLE PERFORMANCE_FASE_ATIVA_&AAAA_MM. AS
SELECT DISTINCT
A.*
,CASE WHEN INPUT(A.NR_DIAS_ATRASO,3.) <= 4  THEN '01. Até 4 dias'
      WHEN INPUT(A.NR_DIAS_ATRASO,3.) <= 10 THEN '02. 5 a 10 dias'
	  WHEN INPUT(A.NR_DIAS_ATRASO,3.) <= 15 THEN '03. 10 a 15 dias' ELSE '04. Maior que 15 dias' END AS FX_ATRASO_MARCACAO

,TODAY() FORMAT DATE9. AS DATA_HOJE
,INTNX('DAY', A.DT_DISPONIBILIZACAO, 15) FORMAT DATE9. AS DATA_15DIAS
,INTNX('DAY', A.DT_DISPONIBILIZACAO, 30) FORMAT DATE9. AS DATA_30DIAS

,B.DT_VENCIMENTO
,B.DT_PROCESSAMENTO_ULT_RECEBIMENTO
,B.DIAS_ATRASO_PAG

,CASE WHEN CALCULATED DATA_15DIAS < CALCULATED DATA_HOJE THEN 1 ELSE 0 END AS FL_MATURADO_15D
,CASE WHEN CALCULATED DATA_30DIAS < CALCULATED DATA_HOJE THEN 1 ELSE 0 END AS FL_MATURADO_30D

,CASE WHEN B.DIAS_ATRASO_PAG IS MISSING THEN 'N/A'
      WHEN B.DIAS_ATRASO_PAG <= 30      THEN PUT(B.DIAS_ATRASO_PAG,3.) ELSE 'Maior 30 dias' END AS FX_PAGAMENTO

FROM (SELECT DISTINCT *, INPUT(PUT(DT_DISPONIBILIZACAO, YYMMN6.),6.) AS SAFRA_DISP FROM EMP_PUBLICOS_&AAAA_MM.) A
LEFT JOIN PARCELAS_&AAAA_MM. B ON
A.NR_CONTRATO = B.CD_CONTRATO
AND A.SQCONTAT = B.SQ_CONTRATO_FINANCEIRO
AND B.DT_VENCIMENTO <= A.DT_DISPONIBILIZACAO

GROUP BY A.NR_CONTRATO, A.SQCONTAT, A.DT_DISPONIBILIZACAO

HAVING B.DT_VENCIMENTO = MAX(B.DT_VENCIMENTO)

;QUIT;
