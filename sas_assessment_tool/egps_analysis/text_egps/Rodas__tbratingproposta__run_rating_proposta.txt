DATA work.TBRATINGPROPOSTA;
    LENGTH
        DtStatusProposta   8
        CdSubProduto       8
        NuProposta         8
        NuContrato       $ 14
        RUPED            $ 12 ;
    FORMAT
        DtStatusProposta YYMMDD10.
        CdSubProduto     BEST2.
        NuProposta       BEST9.
        NuContrato       $CHAR14.
        RUPED            $CHAR12. ;
    INFORMAT
        DtStatusProposta YYMMDD10.
        CdSubProduto     BEST2.
        NuProposta       BEST9.
        NuContrato       $CHAR14.
        RUPED            $CHAR12. ;
    INFILE '/infogrc/TbRatingProposta/csv_rating_BQ/TbRatingProposta.csv'
		FIRSTOBS=2
        LRECL=20000
        ENCODING="LATIN1"
        /*TERMSTR=CRLF*/
        DLM=','
        MISSOVER
        DSD ;
    INPUT
        DtStatusProposta : ?? YYMMDD10.
        CdSubProduto     : ?? BEST2.
        NuProposta       : ?? BEST9.
        NuContrato       : $CHAR14.
        RUPED            : $CHAR12. ;
RUN;
