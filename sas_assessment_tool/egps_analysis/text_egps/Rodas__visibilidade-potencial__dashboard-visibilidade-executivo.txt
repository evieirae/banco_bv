/* Busca dados de visibilidade */
PROC SQL;
CREATE TABLE TbVisibilidade AS
  SELECT
    Safra AS NuAnoMes,
	'Visibilidade' as Tipo,
	INPUT(COMPRESS(DtAtualizacao, ,'kad'), DDMMYY10.) FORMAT DATE9. AS DtAtualizacao,
    TRANSTRN(FxIdade,'de ',TRIMN('')) AS FxIdade,
    CASE WHEN Regional LIKE 'Canal Rodas%' THEN Regional ELSE 'Outros' END AS Regional,
	Filial,
	FlCarteira,
	Produto,
    SUM(Quantidade) AS Mercado,
    SUM(CASE WHEN Status = 'NaoPassouBV' THEN Quantidade END) AS QtdForaBV,
    SUM(CASE WHEN Status = 'ContratoBV' THEN Quantidade END) AS QtdContratoBV,
    SUM(CASE WHEN Status LIKE 'APNP%' THEN Quantidade END) AS QtdAPNP,
    SUM(CASE WHEN Status LIKE 'Negada%' THEN Quantidade END) AS QtdNegada,
    SUM(CASE WHEN Status LIKE 'Cancelada%' THEN Quantidade END) AS QtdCancelada,
    SUM(CASE WHEN Status LIKE 'CardP' THEN Quantidade END) AS QtdCardP,
    SUM(CASE WHEN Status LIKE 'CardR' THEN Quantidade END) AS QtdCardR,
    SUM(CASE WHEN Status LIKE 'CardD' THEN Quantidade END) AS QtdCardD,
	SUM(CASE WHEN Status LIKE 'CardPA' THEN Quantidade END) AS QtdCardPA,
	SUM(CASE WHEN Status LIKE 'CardPR' THEN Quantidade END) AS QtdCardPR
  FROM
    SOLITMP.TbVisibilidadePotencial
  WHERE
    Produto in ('Leves','Motos','Pesados')
    AND Safra >= &safraM12.
    AND FlCadastroBV = 'S'
    AND FlAtivo = 'S'
    AND FxIdade <> 'Desconhecido'
  GROUP BY
    1,2,3,4,5,6,7,8;
RUN;

PROC SQL;
CREATE TABLE TbShareMercado AS
  SELECT
    NuAnoMes,
	DtAtualizacao,
	'Share' as Tipo,
    CASE WHEN StVeiculo = 'Novos' THEN 'ZeroKM'
         WHEN FxIdade IN ('11 a 14 anos','15 a 19 anos', '20 anos e acima') THEN 'Acima 10 anos'
         ELSE FxIdade
         END AS FxIdade,
    CASE WHEN Regional LIKE 'Canal Rodas%' THEN Regional ELSE 'Outros' END AS Regional,
	Filial,
	'N/A' AS FlCarteira,
	TpVeiculo as Produto,
    SUM(CASE WHEN Bancos = 'BV' THEN GravTotal END) as QtdGravamesBV,
    SUM(CASE WHEN Bancos = 'SANTANDER' THEN GravTotal END) as QtdGravamesStd,
    SUM(CASE WHEN Bancos = 'BRADESCO' THEN GravTotal END) as QtdGravamesBrad,
    SUM(CASE WHEN Bancos = 'ITAU' THEN GravTotal END) as QtdGravamesItau,
    SUM(CASE WHEN Bancos = 'PANAMERICANO' THEN GravTotal END) as QtdGravamesPan,
    SUM(CASE WHEN Bancos = 'C6' THEN GravTotal END) as QtdGravamesC6,
    SUM(CASE WHEN Bancos = 'SAFRA' THEN GravTotal END) as QtdGravamesSafra,
    SUM(GravTotal) AS GravTotal
  FROM
    RELEXC.TbMarketShareDiariaAberta
  WHERE
    NuAnoMes >= &safraM12.
    AND TpVeiculo in ('Leves','Motos','Pesados')
    AND TpArquivo = 'mf'
    AND AjusteManual = 'N'
    AND TpAgenteProdutoMIS <> '3'
	AND StatusGravame = 'Ativo'
  GROUP BY 1,2,3,4,5,6,7,8;
RUN;

PROC SQL;
CREATE TABLE TbConvsimu AS
  SELECT DISTINCT
    TRANSTRN(COALESCE(FxIdade,'Desconhecido'),"de ",TRIMN('')) AS FxIdade,
  CASE WHEN Mensageria IN ('0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%',
							 '0. Aprovado','1. Aprovado com rastreador','2. Reanálise psicométrico','0. Aprovado','1. Alta chance','2. Tem chance')
              AND Mensageria2 NOT IN ('3. Recusado', '4. Recusado Reversível','6. Recusado')
              THEN 'P'
         WHEN Mensageria in ('3. Recusado', '4. Recusado Reversível','6. Recusado')
              OR (Mensageria IN ('0. Aprovado 100%','1. Muita chance >85%','2. Grande chance >70%','3. Boa chance >50%',
								 '0. Aprovado','1. Aprovado com rastreador','2. Reanálise psicométrico','0. Aprovado','1. Alta chance','2. Tem chance')
                  AND Mensageria2 IN ('3. Recusado', '4. Recusado Reversível','6. Recusado')) THEN 'R'
		WHEN Mensageria IN ('7. Plano Aprovado') THEN 'PA'
		WHEN Mensageria IN ('6. Plano Recusado') THEN 'PR'
         ELSE 'D'
         END AS GrupoMensageria,
		 'N/A' AS FlCarteira,
    *
  FROM
    TBSIMULACOES

;
RUN;

PROC SQL;

PROC SQL;
%CONECTADB(server=BI);
CREATE TABLE TbDiasUteis AS SELECT * FROM connection to dbcon
(
WITH
TbTempoData AS (
  SELECT
    NuAnoMes,
    CASE WHEN DtTempo BETWEEN '20211227' AND '20211230' THEN NuDiaUtilMes - 1
         WHEN DtTempo IN ('20211224','20211231') THEN -3
         ELSE NuDiaUtilMes END AS NuDiaUtilMes,
    NuSemanaAno,
    DtTempo,
	DdData,
	NuDiaSemana
  FROM DWBI_DIME.TbDimTempoData
),
QtdMes AS (
  SELECT
    NuAnoMes,
    MAX(NuDiaUtilMes) AS NuDiaUtilMes
  FROM TbTempoData
  WHERE
    NuAnoMes >= 202401
    AND DtTempo < &LstGrav.
  GROUP BY NuAnoMes
)
SELECT DISTINCT
  A.NuAnoMes,
  A.NuDiaUtilMes
  FROM QtdMes A
);
RUN;

proc sql;
create table TbVisibilidadeMercado2 as
select
A.*,
B.NuDiaUtilMes
from TbVisibilidadeMercado as A
	left join  TbDiasUteis as B
	on A.NuAnoMes = B.NuAnoMes
;run;
