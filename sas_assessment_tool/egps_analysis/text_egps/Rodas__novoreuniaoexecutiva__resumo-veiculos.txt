/* Buscar dados de produção */
PROC SQL;
CREATE TABLE SpreadContrato01 AS
  SELECT DISTINCT
    NuContrato,
	VrRetencao,
    VrPrestacao,
    VrIof
  FROM BASES.TBEXTCONTRATO
  WHERE NuAnoMes >= &safra13M.;
RUN;

PROC SORT DATA=SpreadContrato01 OUT=SpreadContrato NODUPKEY;
BY NuContrato;
RUN;

/* Buscar dados de produção */
PROC SQL;
  CREATE TABLE PRODUCAO01 AS
    SELECT
      DtAtualizacao AS DtBaseProducao FORMAT=mmddyy10.,
      SUBSTR(PUT(a.NuAnoMes,6.),5,2)||"/01/"||SUBSTR(PUT(a.NuAnoMes,6.),1,4) AS Safra,
      CASE WHEN NmGrupoProdutoMeta = 'Leves-BVO' THEN 'Leves' else NmGrupoProdutoMeta end AS Produto,
      CASE WHEN FxIdade = 'ZeroKM' THEN 'Novos' ELSE 'Usados' END AS StVeiculo,
      NmSubEstruturaComercialA AS Regional,
      CASE WHEN NmGrupoProdutoMeta = 'FGTS' THEN 'FGTS'
		   WHEN NmGrupoProdutoMeta = 'Leves-BVO' THEN 'BVO'
			ELSE DsTipoCanalProducao END AS Canal,
      FxIdade,
	  QtCorridosDiaUtilMesSab,
      QtTotalDiaUtilMesSab,
	  CASE WHEN A.NuProposta = C2C.NuProposta THEN 'S' ELSE 'N' END AS FlC2C,
/*	  S.CompraLeads as Origem,*/
	  'N' as Origem,
      COUNT(A.NuContrato) AS QtdContrato,
      SUM(VrProducao) AS VrProducao FORMAT=num18.2,
      SUM(VrProducaoProjetado) AS VrProducaoProjetado FORMAT=num18.2,
      SUM(VrProducaoDiaUtil) AS VrProducaoDiaUtil FORMAT=num18.2,
      SUM(VrFinanciamento) AS VrFinanciamento FORMAT=num18.2,
      SUM(VrMargemContribuicao) AS VrMC FORMAT=num18.2,
      SUM(VrMargemContribuicao * VrProducao) AS VpMC FORMAT=num18.2,
      SUM(VpTxGerencial) AS VpTaxa FORMAT=num18.2,
      SUM(VpTirContabil) AS VpTir FORMAT=num18.2,
      SUM(((1+(VpTxGerencial/VrProducao)) / (1+(((TxTirContabil/100) * (VrProducao))/VrProducao)) -1 ) * VrProducao) AS VpSpread FORMAT=num18.2,
      SUM(((1+((case when B.VrRetencao > 1 then (Finance('RATE', A.QtParcelas, B.VrPrestacao, -((A.VrContrato-B.VrRetencao)+B.VrIof+coalesce(A.VrTarifaCadastro,0)))*A.VrProducao) ELSE A.VpTxGerencial END)/A.VrProducao)) / (1+(((A.TxTirContabil/100) * (A.VrProducao))/A.VrProducao)) -1 ) * A.VrProducao) AS VpSpreadRebate FORMAT=num18.2,
      SUM(VrComissaoParcelada) AS VrComissaoParcelada FORMAT=num18.2,
      SUM(VrComissaoTotal) AS VrComissaoTotal FORMAT=num18.2,
      SUM(VrCapProjetado) AS VrCap FORMAT=num18.2,
      SUM(VrDemaisSemCompletoMaxProjetado) AS VrAuto FORMAT=num18.2,
      SUM(VrBVMaisProjetado) AS VrSPF FORMAT=num18.2,
      SUM(VrCompletoProjetado) AS VrCompleto FORMAT=num18.2,
      COALESCE(SUM(CASE WHEN VrTarifaAvaliacaoBem > 0 THEN 1 END),0) AS QtdTAB,
      COALESCE(SUM(CASE WHEN VrTarifaCadastro > 0 THEN 1 END),0) AS QtdTC,
      SUM(VrPerdaEstimada) AS VrPerdaEstimada FORMAT=num18.2
    FROM TMP.TBPRODUCAO A
	LEFT JOIN SpreadContrato B
	  	ON A.NuContrato = B.NuContrato
	LEFT JOIN CONTRATO.CONTRATACAO_C2C C2C
    	ON A.NuProposta = C2C.NuProposta
/*	LEFT JOIN Tbsimulacoes s*/
/*    	ON s.NuProposta = C2C.NuProposta*/
    WHERE a.NuAnoMes >= &safra13M.
      AND NmGrupoProdutoMeta IN ('Leves','Motos','Pesados','CVG Leves','CVG Pesados','Privado','CP','CIG','FGTS','Estudantil','Saude','Solar','Leves-BVO','Consignado')
    GROUP BY 1,a.NuAnoMes,3,4,5,6,7,8,9,10,11
    ORDER BY 1,a.NuAnoMes,3,4,5,6,7,8,9,10,11;
RUN;

PROC SQL;
CREATE TABLE PRODUCAO AS
  SELECT
    DtBaseProducao,
    Safra,
    Produto,
    StVeiculo,
    Regional,
    Canal,
    FxIdade,
    QtdContrato,
	FlC2C,
	Origem,
    VrProducao,
    VrProducaoProjetado,
    VrProducaoDiaUtil,
    VrFinanciamento,
    VrMC,
    (VrMC / QtCorridosDiaUtilMesSab) * QtTotalDiaUtilMesSab AS VrMCProjetado,
    VpMC,
    VpTaxa,
    VpTir,
    VpSpread,
    VpSpreadRebate,
    VrComissaoParcelada,
    VrComissaoTotal,
    VrCap,
    VrAuto,
    VrSPF,
    VrCompleto,
    QtdTAB,
    QtdTC,
    VrPerdaEstimada
  FROM PRODUCAO01;
RUN;
