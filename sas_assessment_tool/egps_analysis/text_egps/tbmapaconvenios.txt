proc sql;
create table basegrc.TbMapaConveniosClassificacao as
select
	 ParceiroComercial.CdParceiroComercial as Id
	,ParceiroComercial.NmParceiroComercial as Nm
	,1 as TpBase
	,ParceiroComercial.CdClassificacaoLimiteConvenio
	,ParceiroComercial.NmClassificacaoLimiteConvenio
	,case
		when ParceiroComercial.DsClassificacaoRiscos = 'DIAMANTE' then 5
		when ParceiroComercial.DsClassificacaoRiscos in ('OURO','OURO sem classificacao') then 4
		when ParceiroComercial.DsClassificacaoRiscos = 'PRATA' then 3
		when ParceiroComercial.DsClassificacaoRiscos = 'BRONZE' then 2
		when ParceiroComercial.DsClassificacaoRiscos = 'LATAO' then 1
	    end as DsClassificacao

from tmp.TbParceiroComercial ParceiroComercial
where ParceiroComercial.CdTipoParceiroComercial = 3

union all

select
	 ParceiroComercial.CdClassificacaoLimiteConvenio as Id
	,ParceiroComercial.NmClassificacaoLimiteConvenio as Nm
	,2 as TpBase
	,ParceiroComercial.CdClassificacaoLimiteConvenio
	,ParceiroComercial.NmClassificacaoLimiteConvenio
	,max(case
		when ParceiroComercial.DsClassificacaoRiscos = 'DIAMANTE' then 5
		when ParceiroComercial.DsClassificacaoRiscos in ('OURO','OURO sem classificacao') then 4
		when ParceiroComercial.DsClassificacaoRiscos = 'PRATA' then 3
		when ParceiroComercial.DsClassificacaoRiscos = 'BRONZE' then 2
		when ParceiroComercial.DsClassificacaoRiscos = 'LATAO' then 1
	 	end) as DsClassificacao
from tmp.TbParceiroComercial ParceiroComercial
where ParceiroComercial.CdTipoParceiroComercial = 3
group by 1,2,3,4,5
;run;

proc sql;
%CONECTADB(server=PTA);
CREATE TABLE extracao_pta_carteira AS
SELECT * FROM connection to dbcon
(

WITH TbCorteData AS
(

	SELECT
		 year(ContratoContabil.DtProcessamento)*100+month(ContratoContabil.DtProcessamento) AS NuAnoMes
		,max(ContratoContabil.DtProcessamento) AS DtProcessamento
	FROM gestaocontabil.Vw_ContratoContabil ContratoContabil
	WHERE ContratoContabil.DtProcessamento > '20200101'
	  AND ContratoContabil.DtProcessamento < today()-1
	GROUP BY
		year(ContratoContabil.DtProcessamento)*100+month(ContratoContabil.DtProcessamento)

)

SELECT
	 CorteData.NuAnoMes
	,CorteData.DtProcessamento
	,Contrato.CdParceiroComercial
	,Proposta.CdPessoa CdParceiroComercialCoban
	,Contrato.CdOperador CdColaborador
	,coalesce(Colaborador.CdFilial,Contrato.CdFilial) as CdFilial
	,CASE
		WHEN ContratoContabil.NuDiasAtraso > 360 then 'write-off'
		WHEN ContratoContabil.NuDiasAtraso >  90 then 'over90'
		WHEN ContratoContabil.NuDiasAtraso >  60 then 'stop accrual'
	 ELSE 'ativo' END as TpCarteira
	,TipoCobranca.CdTipoCobranca
	,count(DISTINCT Pessoa.NuCpfCnpj) AS QtCliente
	,sum(CASE WHEN ContratoContabil.CdCarteira = 1 THEN (ContratoContabil.VrTaxaContrato + ContratoContabil.VrPermanenciaApropriado) ELSE (ContratoContabil.VrTaxaContrato) END) VrSaldoContabil
FROM gestaocontabil.Vw_ContratoContabil ContratoContabil

	INNER JOIN TbCorteData CorteData
		ON  ContratoContabil.DtProcessamento = CorteData.DtProcessamento

	INNER JOIN CPC.TbContrato Contrato
		ON  ContratoContabil.NuContrato = Contrato.NuContrato

		LEFT JOIN CPC.TbContratoFinanceiro ContratoFinanceiro
			ON  Contrato.NuContrato = ContratoFinanceiro.NuContrato
			AND Contrato.SqContratoFinanceiroAtivo = ContratoFinanceiro.SqContratoFinanceiro

			LEFT JOIN DBCOR.TbTipoCobranca TipoCobranca
				ON  ContratoFinanceiro.CdTipoCobranca = TipoCobranca.CdTipoCobranca

		LEFT JOIN DBA.TbColaborador Colaborador
			ON  Contrato.CdOperador = Colaborador.CdColaborador
			AND Colaborador.CdColaborador NOT IN (503,9777,11046,11210,18512,57936)
			AND Colaborador.CdOperador <> 0

		LEFT JOIN DBA.TbPessoa Pessoa
			ON  Contrato.CdPessoa = Pessoa.CdPessoa

		LEFT JOIN DBCPC.TbControleLimiteOperacional ControleLimiteOperacional
			ON  Contrato.CdParceiroComercial = ControleLimiteOperacional.CdLoja

	LEFT JOIN DBA.TbProposta Proposta
		ON  ContratoContabil.NuContrato = Proposta.NuContratoGestao

		INNER JOIN (SELECT NuContratoGestao,max(NuProposta) NuProposta FROM DBA.TbProposta GROUP BY NuContratoGestao) PropostaMax
			ON  Proposta.NuProposta = PropostaMax.NuProposta

WHERE Contrato.CdProduto = 11
  AND Contrato.CdModalidadeProduto IN (17,75)

GROUP BY
	 CorteData.NuAnoMes
	,CorteData.DtProcessamento
	,Contrato.CdParceiroComercial
	,Proposta.CdPessoa
	,Contrato.CdOperador
	,coalesce(Colaborador.CdFilial,Contrato.CdFilial)
	,CASE
		WHEN ContratoContabil.NuDiasAtraso > 360 then 'write-off'
		WHEN ContratoContabil.NuDiasAtraso >  90 then 'over90'
		WHEN ContratoContabil.NuDiasAtraso >  60 then 'stop accrual'
	 ELSE 'ativo' END
	,TipoCobranca.CdTipoCobranca

)
;RUN;
