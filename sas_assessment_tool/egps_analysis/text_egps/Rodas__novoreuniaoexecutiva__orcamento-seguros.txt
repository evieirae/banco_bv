PROC SQL;
  CREATE TABLE TpSeguros AS SELECT STRIP(seguro) AS seguro FROM TpSeguros;
RUN;

%macro GeraOrcamento(tipo);
PROC SQL;
  CREATE TABLE orc AS
    SELECT
	  "01/"||SUBSTR(PUT(NuAnoMes,6.),5,2)||"/"||SUBSTR(PUT(NuAnoMes,6.),1,4) AS Data,
	  NmGrupoProdutoMeta AS Produto,
	  UPPER(COALESCE(DsTipoCanalProducao,'DESCONHECIDO')) AS Canal,
	  "&tipo." AS Seguro,
	  SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = "QtContrato&tipo.Dn" AND NmGrupoProdutoMeta LIKE 'CVG%' AND NuAnoMes <= 202012 THEN QtDenominador END) AS QtdeContrato FORMAT=NUMX18.0,
	  SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = "QtContrato&tipo.Nm" AND NmGrupoProdutoMeta LIKE 'CVG%' AND NuAnoMes <= 202012 THEN QtNumerador END) AS Qtde FORMAT=NUMX18.0,
	  SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = "Vr&tipo." THEN VrMedida ELSE 0 END) AS Premio FORMAT=NUMX18.2
    FROM RELEXC.TbMeta
	WHERE NmGrupoProdutoMeta IN ('CVG Leves','CVG Pesados','Privado','CP','CIG','Privado','Solar')
	  AND NmGrupoProdutoMeta IS NOT NULL
	  AND NuAnoMes >= 202007
	GROUP BY 1,2,3,4
	HAVING (calculated Premio) > 0;
RUN;

proc append data=orc BASE=orcamento;run;
%mend;

DATA _null_;
SET TpSeguros;
call execute ('%GeraOrcamento('||seguro||')');
RUN;

PROC SQL;
  CREATE TABLE orc2_01 AS
    SELECT
	  '01/'||PUT(Mes,z2.)||'/'||PUT(Ano,4.) AS Data,
	  NmGrupoSubProduto AS Produto,
	  Canal,
	  SUM(Quantidade) AS QtdElegivel
    FROM RELEXC2.Orcamentoseguros
    WHERE Ano >= 2020
      AND NmGrupoSubProduto IN ('CVG Leves','CVG Pesados','Privado','CP','CIG','Privado','Solar')
	  AND Metrica = 'Contrato'
    GROUP BY 1,2,3;
RUN;

PROC SQL;
  CREATE TABLE orc2_02 AS
    SELECT
	  '01/'||PUT(Mes,z2.)||'/'||PUT(Ano,4.) AS Data,
	  NmGrupoSubProduto AS Produto,
	  Canal,
	  CASE WHEN Metrica NOT IN ('SPF','CAP','Completo') THEN 'Demais'
           WHEN Metrica = 'CAP' THEN 'Cap'
		   WHEN Metrica = 'Completo' THEN 'AutoTotal'
           ELSE Metrica END AS Metrica,
	  SUM(Quantidade) AS Qtde
    FROM RELEXC2.Orcamentoseguros
    WHERE Ano >= 2020
      AND NmGrupoSubProduto IN ('CVG Leves','CVG Pesados','Privado','CP','CIG','Privado','Solar')
	  AND Metrica <> 'Contrato'
    GROUP BY 1,2,3,4;
RUN;

PROC SQL;
  CREATE TABLE orc_qtde AS
    SELECT
	  A.Data,
	  A.Produto,
	  UPPER(A.CANAL) AS Canal,
	  A.Metrica,
	  B.QtdElegivel FORMAT=numx12.2,
	  A.Qtde FORMAT=numx12.2
	FROM orc2_02 A
	LEFT JOIN orc2_01 B
	  ON A.Data = B.Data
	 AND A.Produto = B.Produto
	 AND A.Canal = B.Canal;

PROC SQL;
  CREATE TABLE ORC_COMPLETO AS
    SELECT
	  A.Data,
	  A.Produto,
	  A.Canal,
	  A.Seguro,
	  COALESCE(B.QtdElegivel,A.QtdeContrato,0) AS QtdElegivel FORMAT=numx18.2,
	  COALESCE(B.Qtde,A.Qtde,0) AS Qtde FORMAT=numx18.2,
	  A.Premio FORMAT=numx18.2
	FROM orcamento A
	LEFT JOIN ORC_QTDE B
	  ON A.Data = B.Data
	  AND A.Produto = B.Produto
	  AND A.Canal = B.Canal
	  AND A.Seguro = B.Metrica;
RUN;
