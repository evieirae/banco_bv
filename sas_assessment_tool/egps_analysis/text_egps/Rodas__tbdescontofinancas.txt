%macro Importar;
	%let id = %sysfunc(open(Arquivos));
		%syscall set(id);
		%do %while(%sysfunc(fetch(&id.)) eq 0);
			/* Identifica o delimitador do arquivo */
			data _null_;
				infile "&Diretorio." dsd missover obs=1;
				input Cabecalho	:$255.;
				Cabecalho = translate(Cabecalho, '_', ' ');
				call symputx('Delimitador', put(substr(Cabecalho, prxmatch('/[^\w]/',Cabecalho), 1), $hex.));
			run;

			/* Importa o arquivo em uma base temporária */
			data tmp;
				infile "&Diretorio." dsd missover firstobs=2 dlm="&Delimitador."x;
				format NmArquivo $50.
					   mes DDMMYY10.
					   NuContratoGestao	18.;
				NmArquivo = "&Arquivo";
				input
				   mes              : DDMMYY10.
				    Cod_Operador     : BEST5.
				    nome_operador    : $CHAR37.
				    Cod_Promotor     : BEST5.
				    nome_promotor    : $CHAR37.
				    Diretoria        : $CHAR17.
				    Status           : $CHAR21.
				    Agrupamento      : $CHAR27.
				    Modalidade       : $CHAR27.
				    Qtde_de_Contratos :BEST3.
				    Valor_Liberado   : COMMAX18.2
				    Mar_Liq_Ger      : COMMAX18.2
				    NuContratoGestao : BEST20.
				    SqContratoGestao : BEST1.
				    Cod_Produto      : BEST2.
				    Tp_Producao      : $CHAR13.
				    Filial           : $CHAR30.;

				if NuContratoGestao;
				;run;

			/* Consolida a base temporária em uma base final */
			proc append base=BaseFinal data=tmp force;run;

			/* Remove base temporária */
			proc datasets lib=work nolist;
				delete tmp;
			run;
		%end;
	%let id = %sysfunc(close(&id.));
%mend;

PROC SQL;
CREATE TABLE TbDescontoFinancas2019 AS
SELECT
 	year(mes)*100+month(mes) AS NuAnoMes
	,Status
	,Cod_Operador
	,Valor_Liberado
	,Mar_Liq_Ger AS VrMargemDescontos
	,NuContratoGestao
FROM basegrc.TbDescontoFinancas
WHERE Status <> 'Ativo'
	AND year(mes)*100+month(mes) <= 201912
;RUN;

PROC SQL;
CREATE TABLE TbDescontoFinancas2020 AS
SELECT
	year(intnx('month',mes,1))*100+month(intnx('month',mes,1)) as NuAnoMes
	,Status
	,Cod_Operador
	,Valor_Liberado
	,Mar_Liq_Ger AS VrMargemDescontos
	,NuContratoGestao

FROM basegrc.TbDescontoFinancas
WHERE Status NOT IN ('Liquidação Antecipada', 'Inad - 1 PMT', 'Ativo')
	AND year(mes)*100+month(mes) >= 202001
;RUN;

DATA TbDescontoFinancas;
	SET TbDescontoFinancas2019 TbDescontoFinancas2020
;RUN;

PROC SQL;
CREATE TABLE basegrc.TbMargemDescontosAnalitico AS
SELECT
	Descontos.NuAnoMes
	,Contrato.CdColaborador
	,Colaborador.NmColaborador
	,Colaborador.NmGrupoCargoColaborador
	,Colaborador.DsLogin AS DsLoginN1
	,Colaborador.IdMatricula
	,GrupoProdutoMeta.NmGrupoProdutoMeta
	,TipoCanalProducao.DsTipoCanalProducao
	,ParceiroComercial.CdParceiroComercial
	,ParceiroComercial.NmParceiroComercial
	,Contrato.CdFilial
	,Contrato.DtContrato
	,Contrato.NuContrato
	,Descontos.Status
	,Descontos.VrMargemDescontos

FROM TbDescontoFinancas Descontos
	LEFT JOIN bases.TbExtContrato Contrato
		ON Descontos.NuContratoGestao = Contrato.NuContrato

	LEFT JOIN bases.TbTipoCanalProducaoContrato TipoCanalProducao
		ON Descontos.NuContratoGestao = TipoCanalProducao.NuContrato

		LEFT JOIN bases.TbColaborador Colaborador
			ON Contrato.CdColaborador =  Colaborador.CdColaborador

		LEFT JOIN bases.TbColaborador ColaboradorPromotor
			ON  Contrato.CdColaboradorPromotor = ColaboradorPromotor.CdColaborador

		LEFT JOIN basegrc.TbGrupoProdutoMeta GrupoProdutoMeta
			ON  Contrato.CdGrupoSubProduto = GrupoProdutoMeta.CdGrupoSubProduto

		LEFT JOIN tmp.TbParceiroComercial ParceiroComercial
			ON  Contrato.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

		LEFT JOIN bases.TbEstruturaComercial EstruturaComercial
			ON  Contrato.CdFilial = EstruturaComercial.CdFilial
			AND Contrato.NuAnoMes  = EstruturaComercial.NuAnoMes
			AND EstruturaComercial.NmNivel = 'N2'

WHERE Contrato.StProducao = 'S'

;RUN;
