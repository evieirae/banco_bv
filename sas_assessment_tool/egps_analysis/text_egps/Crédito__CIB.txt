		  WHEN VISAO_CLIENTE = 'Preocupante' AND VISAO_SETOR IN ('Fortemente Positivo', 'Levemente Positivo', 'Neutro') THEN 'Auxílio'
		  WHEN VISAO_CLIENTE = 'Atenção' AND VISAO_SETOR IN ('Fortemente Negativo', 'Levemente Negativo') THEN 'Auxílio'

		  WHEN VISAO_CLIENTE = 'Normal' AND VISAO_SETOR IN ('Fortemente Positivo', 'Levemente Positivo', 'Neutro') THEN 'Beneficiado'
