libname emer '/riscosbvf/fraude/PESSOAL/EMERSON';


%let domingo = %sysfunc(intnx(week, %sysfunc(today()), 0));
%let domingo_n = %sysfunc(putn(&domingo. ,ddmmyy10.));
%put >>>>> &domingo_n.;



proc sql;
	create table base_dash as
	select
		a.PROPOSTA
		,a.MES_ANO
		,a.DT_ENTRADA_PROPOSTA
		,a.LIMITE_CONCEDIDO
		,case
			when a.REGRA like ('%APA HP%') or a.REGRA like ('%Aprova%') or a.REGRA = 'Whitelist'
			then 'Aprova'
				when a.REGRA like ('%óbito%') or a.REGRA like ('%menor de%') or a.REGRA like ('%Score%') or a.REGRA = 'Blacklist'
				then 'Reprova'
					else 'Deriva'
		end as TipoRegra
		,case
			when a.REGRA = 'Grupo Controle - Fila geral' then 'FILA GERAL'
			when a.REGRA = 'Aprovacao Automática' then 'Aprovação Automática'
			when a.REGRA = 'Blacklist' then 'Badlist'
			when a.REGRA = 'Whitelist' then 'Goodlist'
			when a.REGRA in ('Cliente VIP 4', 'Cliente VIP 5') then 'Outros'
			else a.REGRA
		end as Regra_Funil
		,case when a.sts_retorno_clearsale in ('APA','RPA') then 0 else 1 end as flg_deriv_mesa
		,case
			when a.sts_retorno_clearsale = 'AMA' and status_proposta in ('A','AE') then 'APM'
			when  a.sts_retorno_clearsale = 'AMA' and status_proposta = 'N' then 'RPM'
			else a.sts_retorno_clearsale
		end as status_clearsale
		,case when a.proposta = b.proposta then 1 else 0 end as FlgFrdEfetiva
		,b.ValorPerda
	from emer.ct_prop_analitico as a
	left join (
		select distinct
			proposta
			,sum(valor_final) format commax20.2 as ValorPerda
		from emer.base_perdas
		where proposta not is missing
		group by 1) as b on a.proposta = b.proposta
	where a.DT_ENTRADA_PROPOSTA >= today()-365 and a.DT_ENTRADA_PROPOSTA <= &domingo.
	and a.id_clearsale not is missing
	order by a.DT_ENTRADA_PROPOSTA
	;
quit;

proc sql;
	create table base_dash_regras as
	select
		count(proposta) format commax10. as Qtde
		,cat(substr(MES_ANO,1,4), '-', substr(mes_ano,5,2)) as SafraProposta
		,DT_ENTRADA_PROPOSTA
		,sum(LIMITE_CONCEDIDO) format commax20.2 as ValorLimite
		,TipoRegra
		,Regra_Funil
		,flg_deriv_mesa
		,status_clearsale
		,FlgFrdEfetiva
		,sum(ValorPerda) format commax20.2 as ValorPerdas
	from base_dash
	group by 2,3,5,6,7,8,9
	order by DT_ENTRADA_PROPOSTA
	;
quit;
