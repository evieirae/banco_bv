%macro BaseMediaInad(group_vars, NomeMes , DataMes, DeltaMes);
PROC SQL;
CREATE TABLE BaseMEDIA_Inad_&NomeMes AS
SELECT DISTINCT
&DataMes AS DataContrato FORMAT DDMMYY10.,
&group_vars.,
Produto,

SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30) AS Inad30Media,
SUM(VrParcelaEver30_Mob3)/SUM(VrParcelaElegivelEver30_Mob3) AS Ever30Medio,

CASE WHEN Produto = "Leves" THEN
((SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30)*1.3) * Beta) + Alpha
ELSE ((SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30)) * Beta) + Alpha END AS PerdaCalcINADMedia,

CASE WHEN Produto = "Leves" THEN
((SUM(VrParcelaEver30_Mob3)/SUM(VrParcelaElegivelEver30_Mob3)) * Beta) + Alpha
ELSE 0 END AS PerdaCalcEverMedio,

SUM(VrParcelaInad30) AS VrParcelaInad30Media,
SUM(VrParcelaElegivelInad30) AS VrParcelaElegivelInad30Media,
SUM(VrParcelaEver30_Mob3) AS VrParcelaEver30Medio,
SUM(VrParcelaElegivelEver30_Mob3) AS VrParcelaElegivelEver30Medio

FROM PRODUCAO  as A
WHERE DataContrato >= INTNX ('MONTH',&DataMes,&DeltaMes) AND DataContrato < &m3
GROUP BY 1,&group_vars.,Produto;RUN;

%macro join_bases(group_vars3);
    %let n = %sysfunc(countw(&group_vars3, %str(,)));
    %let join_clause = ;
    %let select_clause = ;
    %let group_by_clause = ;

    %do i = 1 %to &n;
        %let var = %scan(&group_vars3, &i, %str(,));
        %if &i = 1 %then %do;
            %let join_clause = A.&var = B.&var;
            %let select_clause = A.&var;
            %let group_by_clause = A.&var;
        %end;
        %else %do;
            %let join_clause = &join_clause and A.&var = B.&var;
            %let select_clause = &select_clause, A.&var;
            %let group_by_clause = &group_by_clause, A.&var;
        %end;
    %end;

    PROC SQL;
    CREATE TABLE PROD_ROE AS
    SELECT DISTINCT
    SUM(A.RatingPnd) AS RatingPnd,
    SUM(A.IdadePnd) AS IdadePnd,
    SUM(A.EntradaPnd) AS EntradaPnd,
    SUM(A.PrazoPnd) AS PrazoPnd,
    SUM(A.VrProducao) AS VrProducao,
    SUM(A.QntdContratos) AS QntdContratos,
    SUM(A.PrazoPnd)/SUM(A.VrProducao) AS Prazo,
    SUM(A.VpTxPactuada)/SUM(A.VrProducao) AS PcTaxa,
    SUM(A.VpTirContabil)/SUM(A.VrProducaoB) AS PcTIR,
    SUM(A.VpSpreadContabil)/SUM(A.VrProducaoB) AS PcSpread,
    SUM(A.VrMargemContribuicao) AS MargemSistema,
    SUM(A.VrPerdaEstimada)/SUM(A.VrFinanciamento) AS PcPerda,
    SUM(A.VrFinanciamento) AS VrFinanciamento,
    SUM(A.VrComissaoTotal)/SUM(A.VrProducaoB) AS PcComissao,
    SUM(A.VrProducao)/SUM(A.QntdContratos) AS VrTicket,
    SUM(A.VrBVMais)/SUM(A.FlBVMais) AS VrTicketSPF,
    SUM(A.VrCap)/SUM(A.FlCap) AS VrTicketCAP_AP,
    SUM(A.VrCompleto)/SUM(A.FlCompleto) AS VrTicketCompleto,
    SUM(A.VrDemaisSeguros)/SUM(A.FlDemaisSeguros) AS VrTicketDemais,
    SUM(A.FlBVMais)/SUM(A.FlBVMaisEleg) AS PcSPF,
    SUM(A.FlCap)/SUM(A.FlCapEleg) AS PcCAP_AP,
    SUM(A.FlCompleto)/SUM(A.FlDemaisEleg) AS PcCompleto,
    SUM(A.FlDemaisSeguros)/SUM(A.FlDemaisEleg) AS PcDemais,
    SUM(A.VrTarifaCadastro)/(COALESCE(SUM(CASE WHEN A.VrTarifaCadastro > 0 THEN 1 END),0)) AS VrTC,
    SUM(A.VrTarifaAvaliacaoBem)/(COALESCE(SUM(CASE WHEN A.VrTarifaAvaliacaoBem > 0 THEN 1 END),0)) AS VrTAB,
    (COALESCE(SUM(CASE WHEN A.VrTarifaCadastro > 0 THEN 1 END),0))/SUM(A.QntdContratos) AS PcTC,
    (COALESCE(SUM(CASE WHEN A.VrTarifaAvaliacaoBem > 0 THEN 1 END),0))/SUM(A.QntdContratos) AS PcTAB,

    CASE WHEN A.DataContrato IN (&m3,&m2,&m1) THEN B.Inad30Media ELSE
    SUM(VrParcelaInad30)/SUM(VrParcelaElegivelInad30)
    END AS Inad30,

    CASE WHEN A.DataContrato IN (&m3,&m2,&m1) THEN B.Ever30Medio ELSE
    SUM(VrParcelaEver30_Mob3)/SUM(VrParcelaElegivelEver30_Mob3)
    END AS Ever30,

    CASE WHEN A.Produto = "Leves" THEN

CREATE TABLE PRODUCAO AS

	 WHEN A.Produto IN ('CVG Leves','CVG Pesados') THEN 'EGV'
	 ELSE A.Produto END AS Produto,
A.Regional,
A.DsFilial AS Filial,
CASE WHEN   A.Produto IN ('Leves','Leves-BVO') THEN
			(CASE WHEN A.PcProbabilidadeAprovacao*10 <= 0.5 THEN 'Zero'
			WHEN A.PcProbabilidadeAprovacao*10 <= 6.0 THEN 'Até 6,0'
			WHEN A.PcProbabilidadeAprovacao*10 <= 7.4 THEN '6,25-7,25'
			WHEN A.PcProbabilidadeAprovacao*10 <= 7.9 THEN '7,5-7,75'
			WHEN A.PcProbabilidadeAprovacao*10 <= 8.5 THEN '8,25-8,5'
			WHEN A.PcProbabilidadeAprovacao*10 <= 9.8 THEN '8,75-9,75'
			WHEN A.PcProbabilidadeAprovacao*10 <= 10 THEN '10'
			ELSE 'ZERO' END)
	 WHEN A.Produto = 'Motos' THEN
			(CASE WHEN A.PcProbabilidadeAprovacao*10 <= 5.25 THEN 'Até 5,25'
	 		WHEN A.PcProbabilidadeAprovacao*10 <= 7.25 THEN '5,5-7,25'
	 		WHEN A.PcProbabilidadeAprovacao*10 <= 7.75 THEN '7,5-7,75'
	 		WHEN A.PcProbabilidadeAprovacao*10 <= 8.0 THEN '8,0'
	 		WHEN A.PcProbabilidadeAprovacao*10 <= 8.75 THEN '8,25-8,75'
	 		WHEN A.PcProbabilidadeAprovacao*10 <= 10 THEN '9-10'
	 		ELSE 'ZERO' END)
	 ELSE '-' END AS FxRating,

CASE WHEN A.PcProbabilidadeAprovacao*10 <= 0.5 THEN 'ZERO'
     WHEN A.PcProbabilidadeAprovacao*10 <= 4.5 THEN 'ATÉ 4,5'
	  WHEN A.PcProbabilidadeAprovacao*10 <= 5.5 THEN '4,75-5,5'
	  WHEN A.PcProbabilidadeAprovacao*10 <= 7.5 THEN '5,75-7,5'
	  WHEN A.PcProbabilidadeAprovacao*10 <= 8.5 THEN '7,75-8,5'
	  WHEN A.PcProbabilidadeAprovacao*10 <= 9.8 THEN '9,25-9,75'
	  WHEN A.PcProbabilidadeAprovacao*10 <= 10 THEN '10'
	  ELSE 'ZERO' END AS FxRating_V2,
CASE WHEN A.Produto = 'Motos' THEN
			(CASE WHEN A.QtAnosIdade between 0 and 1 then 'ZeroKM a 1 ano'
     		WHEN A.QtAnosIdade between 2 and 3 then '2 a 3 anos'
     		WHEN A.QtAnosIdade between 4 and 10  then '4 a 10 anos'
	 		ELSE 'ERRO' END)
	 ELSE A.FxIdade END AS FxIdade,
CASE WHEN A.Produto = 'Motos' THEN (CASE WHEN A.FxIdade = 'ZeroKM' THEN 'ZeroKM' ELSE 'Usado' END)
	 ELSE '-' END AS NovoUsadoMotos,
A.FaixaEntrada AS FxEntrada,
A.PrazoFinanciamento AS FxPrazo,
CASE WHEN A.CdConvenio = 451012 THEN 'ESCALE'
	 WHEN A.Produto IN ('CVG Leves','CVG Pesados') THEN CanaisEGV
	 ELSE A.DsTipoCanalProducao END AS Canal,
CASE WHEN A.Produto IN ('Leves','Leves-BVO') THEN F.FlPorteA  ELSE '-' END AS Porte,
CASE WHEN A.Produto = 'Pesados' THEN A.TpPessoa ELSE '-' END AS TpPessoa,
CASE WHEN A.Produto = 'Pesados' THEN
			(CASE WHEN A.VrProducao BETWEEN 0 AND 100000 THEN '0 A 100 Mil'
			 WHEN A.VrProducao BETWEEN 100000 AND 200000 THEN '100 A 200 Mil'
			 WHEN A.VrProducao BETWEEN 200000 AND 300000 THEN '200 A 300 Mil'
			 WHEN A.VrProducao BETWEEN 300000 AND 400000 THEN '300 A 400 Mil'
			 WHEN A.VrProducao > 400000 THEN 'Maior 400 Mil'
			 ELSE '-' END)
	  ELSE '-' END AS FxTKT,
CASE WHEN A.Produto = 'Leves-BVO' THEN 'BVO' ELSE '-' END AS BVO,
CASE WHEN A.Produto IN ('Leves','CVG Leves','CVG Pesados') THEN C.Subcanal_CN  ELSE '-' END AS Subcanal_CN,

CASE WHEN E.CdParceiroComercialtxt IS NOT NULL THEN "Autoshop" ELSE "DEMAIS" END AS AutoShop,
CASE WHEN A.Produto = 'Motos' THEN
			(CASE WHEN G.QtCilindradaVeiculo BETWEEN 0 AND 124 THEN 'Baixa'
	 		WHEN G.QtCilindradaVeiculo BETWEEN 125 AND 349 THEN 'Média'
	 		WHEN G.QtCilindradaVeiculo BETWEEN 350 AND 699 THEN 'Alta'

	 		ELSE 'Não Encontrado' END)

CASE WHEN A.Produto = 'Pesados' THEN G.DsCategoria ELSE '-' END AS Categoria,

INTNX ('MONTH',A.DtContrato,0) as DataContrato FORMAT DDMMYY10.,
INTNX ('DAY',INTNX ('MONTH',A.DtContrato,0),33) as DataVecto FORMAT DDMMYY10.,
I.Alpha,
I.Beta,

SUM(B.VrBVMais) AS VrBVMais,
SUM(B.FlBVMais) AS FlBVMais,
SUM(B.VrCap) AS VrCap,
SUM(B.FlCap) AS FlCap,
SUM(B.VrCompleto) AS VrCompleto,

SUM(B.FlGap)+SUM(B.FlGarantiaMecanica)+SUM(B.FlRCF)+SUM(B.FlCasco) AS FlDemaisSeguros,

SUM(B.FlCapEleg) AS FlCapEleg,
SUM(B.FlDemaisEleg) AS FlDemaisEleg,
SUM(B.VrTarifaCadastro) AS VrTarifaCadastro,
SUM(B.VrTarifaAvaliacaoBem) AS VrTarifaAvaliacaoBem,
SUM(B.VrComissaoTotal) AS VrComissaoTotal,

SUM(A.VrFinanciamento) AS VrFinanciamento,
SUM(B.VrProducao) AS VrProducaoB,
COALESCE(SUM(H.VrParcelaInad30),0) AS VrParcelaInad30,
COALESCE(SUM(H.VrParcelaElegivelInad30),0) AS VrParcelaElegivelInad30,
