%CONECTADB(server=PTA);
CREATE TABLE TbPropostaSeguro AS
SELECT 	 TbPropostaSeguro.NuProposta
		,Max(TbPropostaSeguro.FlBVMais) as FlBVMais
		,Max(TbPropostaSeguro.FlAp) as FlAp
		,Max(TbPropostaSeguro.FlDemais) as FlDemais
		,Max(TbPropostaSeguro.FlPlaca) as FlPlaca
		,Sum(TbPropostaSeguro.VrSeguroBVMais) as VrSeguroBVMais
		,Sum(TbPropostaSeguro.VrSeguroAp) as VrSeguroAp
		,Sum(TbPropostaSeguro.VrSeguroDemais) as VrSeguroDemais
		,Sum(TbPropostaSeguro.VrSeguroPlaca) as VrSeguroPlaca
		,max(case when GrupoProdutoSeguro.CdTipoSeguro = 1 and GrupoProdutoSeguro.CdGrupoProdutoSeguro <> 11 then 1 else 0 end) as FlDemaisSemCompleto
		,max(case when GrupoProdutoSeguro.CdTipoSeguro = 1 and GrupoProdutoSeguro.CdGrupoProdutoSeguro  = 11 then 1 else 0 end) as FlAutoCompleto
		,Sum(case when GrupoProdutoSeguro.CdTipoSeguro = 1 and GrupoProdutoSeguro.CdGrupoProdutoSeguro <> 11 then TbPropostaSeguro.VrSeguroDemais else 0 end) as VrSeguroDemaisSemCompleto
		,Sum(case when GrupoProdutoSeguro.CdTipoSeguro = 1 and GrupoProdutoSeguro.CdGrupoProdutoSeguro  = 11 then TbPropostaSeguro.VrSeguroDemais else 0 end) as VrSeguroAutoCompleto

FROM connection to dbcon
(
	SELECT
		 PropostaSeguro.NuProposta
		,PropostaSeguro.CdProdutoSeguro
		,PropostaSeguro.CdTipoSeguro

		,max(CASE WHEN PropostaSeguro.CdTipoSeguro = 2 THEN 1 ELSE 0 END) FlBVMais
		,max(CASE WHEN PropostaSeguro.CdTipoSeguro in (3,5) THEN 1 ELSE 0 END) FlAp
		,max(CASE WHEN PropostaSeguro.CdTipoSeguro = 1 THEN 1 ELSE 0 END) FlDemais
		,max(CASE WHEN PropostaSeguro.CdTipoSeguro = 6 THEN 1 ELSE 0 END) FlPlaca

		,Sum(CASE WHEN PropostaSeguro.CdTipoSeguro = 2 THEN PropostaSeguro.VrSeguro ELSE 0 END) VrSeguroBVMais
		,Sum(CASE WHEN PropostaSeguro.CdTipoSeguro in (3,5) THEN PropostaSeguro.VrSeguro ELSE 0 END) VrSeguroAp
		,Sum(CASE WHEN PropostaSeguro.CdTipoSeguro = 1 THEN PropostaSeguro.VrSeguro ELSE 0 END) VrSeguroDemais
		,Sum(CASE WHEN PropostaSeguro.CdTipoSeguro = 6 THEN PropostaSeguro.VrSeguro ELSE 0 END) VrSeguroPlaca

	FROM DBCRED.TbPropostaSeguro PropostaSeguro
		 WHERE DATE(PropostaSeguro.TsReplicacao) >= DATEADD (MONTH, -4, TODAY())
	GROUP BY
		  PropostaSeguro.NuProposta
		 ,PropostaSeguro.CdProdutoSeguro
		 ,PropostaSeguro.CdTipoSeguro
) as TbPropostaSeguro

		left join Automati.TbSeg_ProdutoSeguro GrupoProdutoSeguro
		on  TbPropostaSeguro.CdProdutoSeguro = GrupoProdutoSeguro.CdProdutoSeguro
Group by 1
;Quit;

Proc Sql;
Create Table Base_Originacao_Solar as
Select 	Originacao.*

		,TbPessoa.DsEstadoCivil
		,TbPessoa.DsTipoResidencia
		,TbPessoa.CdSexo
		,TbPessoa.DsProfissao
		,intck('year', DatePart(TbPessoa.DtNascimento), Originacao.DtEntradaProposta) as QtAnosIdadeCliente
		,TbPessoa.VrTotalReceitas

		,case when SegurosContratos.FlBVMais > 0 then SegurosContratos.FlBVMais else  Coalesce(SeguroProposta.FlBVMais,SegurosContratos.FlBVMais,0) end  as FlBVMais
		,case when SegurosContratos.FlAp > 0 then SegurosContratos.FlAp else  Coalesce(SeguroProposta.FlAp,SegurosContratos.FlAp,0) end  as FlAp
		,case when SegurosContratos.FlPlaca > 0 then SegurosContratos.FlPlaca else  Coalesce(SeguroProposta.FlPlaca,SegurosContratos.FlPlaca,0) end  as FlPlaca

		,case when SegurosContratos.VrSeguroBVMais > 0 then SegurosContratos.VrSeguroBVMais else  Coalesce(SeguroProposta.VrSeguroBVMais,SegurosContratos.VrSeguroBVMais,0) end  as VrSeguroBVMais
		,case when SegurosContratos.VrSeguroAP > 0 then SegurosContratos.VrSeguroAP else  Coalesce(SeguroProposta.VrSeguroAP,SegurosContratos.VrSeguroAP,0) end  as VrSeguroAP
		,case when SegurosContratos.VrSeguroPlaca > 0 then SegurosContratos.VrSeguroPlaca else  Coalesce(SeguroProposta.VrSeguroPlaca,SegurosContratos.VrSeguroPlaca,0) end  as VrSeguroPlaca

		,tbRefPrecoExterno.dsmarcainversor
		,tbRefPrecoExterno.dsmarcamodulo
		,tbRefPrecoExterno.dtinclusao
        ,tbRefPrecoExterno.flativo
		,tbRefPrecoExterno.vrprecomaximo
		,tbRefPrecoExterno.vrprecominimo
		,tbRefPrecoExterno.vrprecomedio
		,tbRefPrecoExterno.vrpotenciaprojeto
		,tbRefPrecoExterno.vrcontaenergiaeletrica

From tmp.TbOriginacao
	(Where = ((NmCanalEstruturaComercial = 'Canal Negocios' or NmGrupoProdutoMeta = 'Solar') and NuANomes >= &Base_m1.)) as Originacao
	 Left Join TbPropostaSeguro as SeguroProposta

create table bases.TbIndicadoresSolar as
select a.*, b.DtSimulacaoAux,
max(0,case
 when TpDados = 'Producao' then intck('WEEKDAY', DtEntradaProposta, DtContrato)
else intck('WEEKDAY', DtSimulacaoAux, DtEntradaProposta)
end

case when VrLiberado = . then VrOriginado else VrLiberado end as VrLiberadoPrd

;quit;
