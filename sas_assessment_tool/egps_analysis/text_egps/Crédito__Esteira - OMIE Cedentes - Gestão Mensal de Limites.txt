proc sql;
   create table TbGestaoLimites as select
	*
   ,case when LimiteM1 = . then LimiteAtual else LimiteM1 end as LimiteVigente
   ,case when LimiteRaizM1 = . then LimiteRaizAtual else LimiteRaizM1 end as LimiteRaizVigente
   ,case when DsPipe in ('Gelado','Frio') then 'Inativo' else 'Ativo  ' end as DsStatusAtivo
   ,case when FlSuspensoPLCA = 1                                               then 'Já está bloqueado '
         when (PercentualCanceladas >= 10 or ScoreHPJ8 <= 100 or FlOver90 = 1) then 'Deve ser bloqueado'
                                                                               else 'Desbloqueado      '
    end as DsStatus
   ,case when calculated DsStatus = 'Desbloqueado      ' then 'Desbloqueado'
                                                         else 'Bloqueado   ' end as DsStatusFinal
   ,case when LimiteM1 ne LimiteM2 then 'Alteração na última revisão '
                                   else 'Manutenção na última revisão' end as DsStatusUltRevisao
   ,case when calculated DsStatus = 'Desbloqueado      ' and
         (ScoreHPJ8 <= 200 or DsClassificacao in ('PCMR2','PCMR3','PCMR4') or PercentualCanceladas >= 5) and
         (calculated LimiteVigente >= 3*LimiteMajoracaoProposto or calculated LimiteVigente >= 3*CARMediaUlt6meses) and
		 calculated DsStatusUltRevisao = 'Manutenção na última revisão' and
         DsPipe ne 'Quente'
         then 'Redução   '
	     when calculated DsStatus = 'Desbloqueado      ' and
              VrRestritivos <= 5000 and
              ScoreHPJ8 > 200 and
              DsClassificacao not in ('PCMR2','PCMR3','PCMR4') and
              Ever5 ne 1 and
			  PercentualCanceladas <= 5 and
              QuantidadeSacadosUlt6Meses >= 10 and
              PcILM = 1 and
			  DsPipe = 'Quente' and
			  calculated DsStatusUltRevisao = 'Manutenção na última revisão'
		 then 'Aumento   '
	else 'Manutenção' end as DsStatusLimite
   ,case when calculated DsStatusAtivo = 'Ativo  ' and calculated DsStatusLimite = 'Aumento   '
         then max(int(min(1500000,
		              2*(calculated LimiteVigente),
					  calculated LimiteVigente+200000,
                      LimiteMajoracaoProposto)/1000)*1000,calculated LimiteVigente)
         when calculated DsStatusAtivo = 'Ativo  ' and calculated DsStatusLimite = 'Redução   '
		 then max(int(LimiteMajoracaoProposto/1000)*1000,5000)
    else calculated LimiteVigente end as NovoLimite
	,100*(calculated NovoLimite/calculated LimiteVigente-1) as PcDeltaLimite format=f8.2
	,(calculated LimiteRaizVigente-calculated LimiteVigente+calculated NovoLimite) as LimiteRaiz
   from TbConsolidada
;run;
