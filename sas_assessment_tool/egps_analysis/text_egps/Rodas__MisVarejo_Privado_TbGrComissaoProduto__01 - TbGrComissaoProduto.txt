/*Gerando a base de Seguros Placa Solar do GCP - Adesão*/
proc sql ;
connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage);
   Create table Contratos_Seguro_GCP as
   select Distinct
		  Input(nucontrato, 14.) as Nucontrato Format Best32.
		  ,DHMS(DtContrato, 0,0,0) as DtContrato Format DateTime25.6
		  ,SqContratoFinanceiro
		  ,Max(dtiniciovigencia) as DtInicioVigencia Format DateTime25.6
		  ,Max(DtFimVigencia) as DtFimVigencia Format DateTime25.6
		  ,CdTipoSeguro
		  ,CdProdutoSeguro
		  ,UpCase(NmProdutoSeguro) as NmProdutoSeguro
		  ,InPut(VrPremioSeguro, 20.) as VrPremioSeguro Format COMMAX20.2
	from connection to perm
     (
		SELECT DISTINCT
		  core.cd_contrato as nucontrato,
		  PROPSEG.nuproposta,
		  CFPS.sqcontratofinanceiro,
		  PROPSEG.dtiniciovigenciaseguro as dtiniciovigencia,
		  PROP.dtfimvigenciaseguro as DtFimVigencia,
		  CORE.dt_contrato AS DtContrato,
		  PROPSEG.cdtiposeguro,
		  PROPSEG.CdProdutoSeguro,
		  PS.nmprodutosegurointerno as NmProdutoSeguro,
		  PROPSEG.vrtotalseguro as VrPremioSeguro,
		  EPS.cdtipoeventoseguro,
		  PROPSEG.cdpropostaseguro,
		  PS.cdseguradora,
		  EPS.dtprocessamentoeventoseguro as DtProcessamento
		FROM `bv-pdbd-prd.trusted_psegcontratacaoseguro.tbpropostaseguro` PROPSEG
			INNER JOIN `bv-pdbd-prd.trusted_psegseguroprocessamento.tbeventopropostaseguro` EPS
				ON EPS.cdpropostaseguro = PROPSEG.cdpropostaseguro
			INNER JOIN `bv-pdbd-prd.trusted_psegcontratacaoseguro.tbcontratofinanceiroprdtosgro` CFPS
				ON	PROPSEG.nuproposta = CFPS.nuproposta
		  INNER JOIN `bv-pdbd-prd.trusted_psegprodutoseguro.tbprodutoseguro` PS
				ON PROPSEG.cdprodutoseguro = PS.cdprodutoseguro
		  LEFT JOIN `bv-pdbd-prd.trusted_psegcontratacaoseguro.tbpropostaseguro` PROP
		  	  ON PROPSEG.nuproposta = PROP.nuproposta
		  LEFT JOIN `bv-pdbd-prd.dvry_core.tbcontrato` as CORE
			  ON core.cd_contrato = CFPS.nucontrato
		/*	  ON to_base64(sha256(core.cd_contrato)) = CFPS.nucontrato   --- Esta função serve para fazer o Join quando um campo esta mascarado*/
		WHERE
				/*Dados do Seguro Placa Solar*/
				(CORE.cd_contrato > '0' AND
				 EPS.cdtipoeventoseguro IN (100) AND
				 PS.cdprodutoseguro IN (595,596,597) AND /*Código dos produtos Placa Solar*/
				 PROPSEG.FlAtivo = 'S' AND
				 PROPSEG.dtiniciovigenciaseguro > '2025-01-01') AND/*Início em Janeiro-25*/
				 CFPS.flativo='S'
				OR
				/*Dados do Seguro AP*/
				(CORE.cd_contrato > '0' AND
				 EPS.cdtipoeventoseguro IN (100) AND
				 PS.cdseguradora = 396148 AND /*Código da Seguradora que vende somente AP*/
				 PROPSEG.FlAtivo = 'S' AND
				 PROPSEG.dtiniciovigenciaseguro > '2025-05-31') AND/*Início em Junho-25*/
				 CFPS.flativo='S'
				OR
				/*Auto Completo da Liberty e Aliro*/
				(CORE.cd_contrato > '0' AND
				 EPS.cdtipoeventoseguro IN (100) AND
				 PROPSEG.FlAtivo = 'S' AND
				 PS.cdseguradora In (411096, 461144) AND /*Código da Seguradora que vende somente Completo*/
				 PROPSEG.dtiniciovigenciaseguro > '2025-07-01') AND
				 CFPS.flativo='S'
				OR
				/*Auto Completo da Zurich - Entrou a partir do dia 25-10*/
				(CORE.cd_contrato > '0' AND
				 EPS.cdtipoeventoseguro IN (100) AND
				 PROPSEG.FlAtivo = 'S' AND
				 PS.cdseguradora In (411877) AND /*Código da Seguradora que vende somente Completo*/
				 PROPSEG.dtiniciovigenciaseguro > '2025-10-25') AND
				 CFPS.flativo='S'
				OR
				/*Auto Completo da Mapfre - Entrou a partir do dia 15-12*/
				(CORE.cd_contrato > '0' AND
				 EPS.cdtipoeventoseguro IN (100) AND
				 PROPSEG.FlAtivo = 'S' AND
				 PS.cdseguradora In (313602) AND /*Código da Seguradora que vende somente Completo*/
				 PROPSEG.dtiniciovigenciaseguro > '2025-12-12') AND
				 CFPS.flativo='S'
	 	);
   disconnect from perm;
quit;

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
	CREATE TABLE analitico_seguros AS
	SELECT * FROM connection to dbcon
	(
		select distinct
			 Seguro.NuContrato
			,Contrato.DtContrato
			,Seguro.SqContratoFinanceiro
		    ,Seguro.DtInicioVigencia
		    ,Seguro.DtFimVigencia
			/*,ProdutoSeguro.SgProdutoSeguradora*/
			,ProdutoSeguro.CdTipoSeguro
			,ProdutoSeguro.CdProdutoSeguro
			,ProdutoSeguro.NmProdutoSeguro
			,Seguro.VrPremioSeguro

		from DBCONTRATOSEGURO.TbSeguroContratoFinanceiro Seguro

			left join DBCONTRATOSEGURO.TbProdutoSeguro ProdutoSeguro
				on Seguro.CdProdutoSeguro = produtoseguro.CdProdutoSeguro

			left join DBA.VwTbContrato Contrato
				on Seguro.NuContrato = Contrato.NuContrato

		where
		 	(Seguro.NuContrato > 0
		  	and Seguro.DtProcessamento  < today()
		  	and produtoseguro.cdprodutoseguro <> 525
		 	)
		  	OR

		  (Seguro.NuContrato > 0
		  and Seguro.DtProcessamento  < today()
		  and produtoseguro.cdprodutoseguro = 525
		  and Contrato.DtContrato < '20231017'
		  and Seguro.DtInicioVigencia < '20231221'
		  and (Seguro.DtInicioVigencia - dateadd(day,14,dateadd(day,day(Contrato.DtContrato)*-1,dateadd(month,1,Contrato.DtContrato))) < 1)
			))
;QUIT;

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
	CREATE TABLE analitico_fraudados AS
	SELECT * FROM connection to dbcon
	(
		/*Base de contratos Gestão de Contratos*/
		select distinct
			a.NuContrato as NuContratoGestao
			,CONVERT(VARCHAR, a.dtProcessamento - DAY(a.dtProcessamento) + 1, 103) AS dtProcessamento
			,LEFT(CONVERT(VARCHAR, a.dtProcessamento,112),6) AS NuAnoMes
			,a.CdAgenteRecebedor
			,b.NuContaDebito
			,CASE
				WHEN A.CdAgenteRecebedor = 198 THEN 'FRAUDE CONFIRMADA'
				WHEN A.CdAgenteRecebedor = 218 THEN 'FRAUDE ERRO OPERACIONAL'
		 		WHEN A.CdAgenteRecebedor = 219 THEN 'FRAUDE CONDENATORIA'
		 		WHEN A.CdAgenteRecebedor = 248 THEN 'FRAUDE ELETRONICA'
			 END AS DsTipoFraude
			,sum(a.VrLancamentoContabil) AS VrLancamentoContabil
			,CASE
				WHEN C.CdProduto = 10 THEN 'CRÉDITO PESSOAL'
				WHEN C.CdProduto = 11 THEN 'CONSIGNADO'
		 		WHEN C.CdProduto = 12 THEN 'CDC - Crédito Direto'
		 		WHEN C.CdProduto = 13 THEN 'CDC - Crédito Direto'
		 		WHEN C.CdProduto = 14 THEN 'CDC - Crédito Direto'
			END AS Produto
		from gestaocontabil.TbLancamentoContabil AS a
    		LEFT JOIN gestaocontabil.TbRoteiroContabil AS b
				ON a.CdRoteiroContabil = b.CdRoteiroContabil
			LEFT JOIN DBA.VwTbContrato AS c
				on a.Nucontrato = c.Nucontrato
			LEFT JOIN DBA.TbProposta AS e
				ON a.Nucontrato = e.Nucontratogestao
				and e.Stproposta = 'AP'
		where
			a.CdAgenteRecebedor IN (198,219,218)
			and not a.Nucontrato in (12035000320728,12040000154450,12053000231655,12065000181678,12078000141055,12085000107639
				,12116000006624,12258000049153,12272000029540,12272000040389,12272000054659,12272000066425)
    		and b.NuContaDebito IN (81999008405008,81999008400243,81999008400251,81999008400260,81999008403005,81952100100009)
			and LEFT(CONVERT(VARCHAR, a.dtProcessamento,112),6) >= &mes_apuracao_char.
			/*Alterei between  &mes_cancelados_char. and &mes_apuracao_char.*/
		group by
			a.nucontrato,
			a.dtProcessamento,
			A.CdAgenteRecebedor,
			b.NuContaDebito,
			c.DtContrato,
			c.CdProduto

		union

		/*Base de contratos FLEX*/
		select distinct
			a.NuContrato as NuContratoGestao
			,CONVERT(VARCHAR, a.DtMovimento - DAY(a.DtMovimento) + 1, 103) AS dtProcessamento
			,LEFT(CONVERT(VARCHAR, a.DtMovimento,112),6) AS NuAnoMes
			,a.NuContaContabil AS CdAgenteRecebedor
			,a.NuContaContabilFlexCube AS NuContaDebito
			,CASE
				WHEN A.NuContaContabil = 198 THEN 'FRAUDE CONFIRMADA'
		 		WHEN A.NuContaContabil = 218 THEN 'FRAUDE ERRO OPERACIONAL'
		 		WHEN A.NuContaContabil = 219 THEN 'FRAUDE CONDENATORIA'
			END AS DsTipoFraude
			,sum(a.VrLancamentoContabil) AS VrLancamentoContabil
			,CASE
				WHEN A.CdProduto = 10 THEN 'CRÉDITO PESSOAL'
		 		WHEN A.CdProduto = 11 THEN 'CONSIGNADO'
		 		WHEN A.CdProduto = 12 THEN 'CDC - Crédito Direto'
		 		WHEN A.CdProduto = 13 THEN 'CDC - Crédito Direto'
		 		WHEN A.CdProduto = 14 THEN 'CDC - Crédito Direto'
			END AS Produto
		from STGTXT_FLEX.TbLancamentoContabilFlexCube AS a
    	where a.NuContaContabil IN (198,219,218)
     		AND a.NuContaContabilFlexCube IN (81999008405008,81999008400243,81999008400251,81999008400260,81999008403005,81952100100009)
			AND a.DtMovimento =
				(
					SELECT MAX(aa.DtMovimento)
    				FROM STGTXT_FLEX.TbLancamentoContabilFlexCube AS aa
    				WHERE aa.Nucontrato = a.Nucontrato AND aa.DtMovimento = a.DtMovimento
				)
			and LEFT(CONVERT(VARCHAR, a.DtMovimento,112),6) >= &mes_apuracao_char.
			/*Alterei between  &mes_cancelados_char. and &mes_apuracao_char.*/
		group by
			a.nucontrato,
			a.DtMovimento,
			a.NuContaContabil,
			a.NuContaContabilFlexCube,
			a.CdProduto
		order by 2
	)
;QUIT;

PROC SORT DATA=analitico_fraudados NODUPKEY;
BY NuContratoGestao;
RUN;

		WHEN Descricao = "BV Mais Motos" 	THEN 0.0067
		WHEN Descricao = "BV Mais Pesados" 	THEN 0.0067
		WHEN Descricao = "Prêmio SPF Solar" THEN 0.0067

		/* ============== Regra AP  ============= */

		WHEN Descricao = "Prêmio AP Premiável" THEN 0.0058

		/* ============== Regra Seguro Auto  ============= */
		WHEN Descricao = "Seguros Casco Leves" THEN 0.0043
		WHEN Descricao = "Seguros Completo Leves" THEN 0.0056
		WHEN Descricao = "Seguros GE" THEN 0.0080
		WHEN Descricao = "Seguros RCF Leves" THEN 0.0052
		WHEN Descricao = "Seguros RCF Motos" THEN 0.0035
		WHEN Descricao = "Seguros RCF Pesados" THEN 0.0033

		/* ============== Regra Seguro Placa  ============= */
		WHEN Descricao = "Prêmio Placa Solar" THEN 0.0035

	ELSE 0.000 END AS PercComissao

FROM TmpDataTratada
CROSS JOIN DescricaoProdutoSeguro
WHERE DtIniMes >= DtIniMesPassado AND DtIniMes <= intnx('month',TODAY(),0)
;Run;

	basegrc.DescricaoProdutoComissaoSeguro
	DescricaoProdutoComissaoSeguro
;run;
