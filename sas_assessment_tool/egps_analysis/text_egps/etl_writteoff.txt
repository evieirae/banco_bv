PROC SQL NOPRINT;
	connect to SYBASEIQ (&conex_syb.);
CREATE TABLE Leasing AS
SELECT * FROM connection to SYBASEIQ

(
Declare @DtCalculo Varchar(8)
Set @DtCalculo=&ult_dia.

SELECT DISTINCT
	(Nucontrato),
	FlagTroca=1
INTO #ER
FROM lsg.tbobjgarantia
WHERE DtSubstituicao is not null

SELECT
	a.*,
	c.NuContrato,
	c.SqContratoFinanceiroAtivo,
	Produto=Case when c.CdModalidadeProduto in (1,2,3,7,8,25,26,27,38,41,46,48)     	then 'Leves'
				 when c.CdModalidadeProduto in (5,49)   	                            then 'Motos'
				 when c.CdModalidadeProduto in (9,28,47)   		                        then 'Pesados'
				 when c.CdModalidadeProduto in (17)		                           		then 'Privado'
				 when c.CdModalidadeProduto in (14,18,32,34,37,39,42) 	                then 'Publico'
				 when c.CdModalidadeProduto in (16,33,43)                               then 'INSS'
				 when c.CdModalidadeProduto in (4,11,12,13,14,15,50,51,52,53,54,55)     then 'MC'
				 when c.CdModalidadeProduto in (10,19,20,21,22,23,24,29,30,31,44,57)    then 'CP'
				 when c.CdModalidadeProduto in (6,45,56) and f.cdtipobem in (9,10) 		then 'Leves'
				 when c.CdModalidadeProduto in (6,45,56) and f.cdtipobem in (5,6,7)		then 'Motos'
				 when c.CdModalidadeProduto in (6,45,56) and f.cdtipobem in (1,2,4,8)	then 'Pesados'
				 end,
	Origem = 'Leasing'
FROM
	dbcont.lsg_risco AS a INNER JOIN lsg.tbcontrato AS c
				ON a.contrato=c.NuContratoLegado
		LEFT JOIN lsg.TbObjGarantiaBemVeiculo AS b
				ON c.nucontrato=b.nucontrato
		LEFT JOIN lsg.tbobjgarantia AS e
				ON e.nucontrato=b.nucontrato AND e.SqObjGarantia=b.SqObjGarantia
		LEFT JOIN lsg.tbobjgarantiabem AS f ON f.nucontrato=b.nucontrato and
				f.SqObjGarantia=b.SqObjGarantia
		LEFT JOIN #ER AS y ON b.NuContrato=y.NuContrato

		LEFT JOIN dba.tbproposta AS l
				ON b.Nucontrato=l.Nucontratogestao
		LEFT JOIN dba.tbpropgarantiamaqequip AS k
				ON l.Nuproposta=k.Nuproposta
WHERE
	a.dt_calculo = @DtCalculo
