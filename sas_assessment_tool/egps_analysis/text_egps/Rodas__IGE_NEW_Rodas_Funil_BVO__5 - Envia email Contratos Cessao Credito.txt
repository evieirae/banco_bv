%include '/infogrc/utils/all-includes.sas';

%let dir_csv = /riscosbvf/BA/processos/output;
LIBNAME TRIBO '/riscosbvf/coaching/tribo_veiculos/SAS';

/* ======================================================================================*/
/* =====================Gera csv contratos cessao de credito bvo=========================*/
/* ======================================================================================*/

/* Verifica se o dia atual é 21 */
%let hoje = %sysfunc(day(%sysfunc(today())));

/* Define macro para toda a lógica */
%macro envia_email();

    %if &hoje = 21 %then %do;

        /* Cria a tabela */
        proc sql;
            create table BVO_PLAN AS
                SELECT
                    NuAnoMes,
                    Produto,
                    TpPessoa,
                    NuProposta,
                    NuContrato,
                    case
                        when FlEsteiraBB = 1 then "Fase 2"
                        else "Fase 1"
                    end as FASE,
                    FlEsteiraBB,
                    FlPropostaPaga,
                    FlPropostaEfetivada
                from SOLITMP.CUBOSIMULACAORODASBVO_ANALISE
                where NuContrato is not null
                order by NuAnoMes desc;
        quit;

        /* Exporta o CSV */
        %let dir_csv = /riscosbvf/BA/processos/output;
        proc export
            data=BVO_PLAN
            file="&dir_csv./BVO_PLAN.csv"
            replace
            dbms=dlm;
            delimiter=';';
        run;
/* ======================================================================================*/
/* =====================Envia email=========================*/
/* ======================================================================================*/

        /* Dispara o e-mail */
        filename mymail email
            to=('flavio.luis@bv.com.br','rodof.oliveira@bv.com.br')
            subject="Contratos Cessão Crédito BVO"
            attach="&dir_csv./BVO_PLAN.csv";

        data _null_;
            file mymail;
            put "Segue em anexo os contratos BVO Fases 1 e 2.";
        run;

    %end;

%mend;

/* Executa a macro */
%envia_email();
