%MACRO AMORTIZACAO(dia_processamento_0,atraso1,SAFRA);
PROC SQL;
  %CONNECT_GCP;

  CREATE TABLE BASE_JOIN AS
  SELECT *
  FROM CONNECTION TO perm
  (
    SELECT DISTINCT
      Z.NUCONTRATO,
      Z.SQCONTRATOFINANCEIRO,
      
      B.NU_PARCELA           AS NUPARCELA,
      B.ST_PARCELA           AS STPARCELA,
      B.DT_ULTIMO_RECEBIMENTO,
      B.DT_PROCESSAMENTO_ULT_RECEBIMENTO
    FROM  `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` AS Z
    LEFT JOIN `bv-pdbd-prd.dvry_core.tbparcela` AS B
      ON B.CD_CONTRATO   = Z.NUCONTRATO
     AND B.SQ_CONTRATO_FINANCEIRO = Z.SQCONTRATOFINANCEIRO
    WHERE
      Z.DTPROCESSAMENTO = &dia_processamento_0.
      AND Z.NUDIASATRASO >= &atraso1.
    ORDER BY Z.NUCONTRATO, Z.SQCONTRATOFINANCEIRO, B.NU_PARCELA
  );
  disconnect from perm;
  QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO AS
SELECT A.*
,CASE
WHEN input(&dia_processamento_0., yymmdd10.) < DT_PROCESSAMENTO_ULT_RECEBIMENTO THEN 0
WHEN STPARCELA = 'A' THEN 0
WHEN NUPARCELA IS NOT NULL THEN 1
ELSE 0 END AS FL_PAGAS
FROM BASE_JOIN A
;QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_1 AS
SELECT COUNT(NUPARCELA) as QTD_PARCELAS,
SUM(FL_PAGAS) AS QTD_PARCELAS_PAGAS,
NUCONTRATO,
sqcontratofinanceiro
FROM ENRIQUECIMENTO_COM_AMORTIZACAO A
GROUP BY NUCONTRATO,
sqcontratofinanceiro
;QUIT;

PROC SQL;
CREATE TABLE ENRIQUECIMENTO_COM_AMORTIZACAO_2 AS
SELECT
    NUCONTRATO,
    sqcontratofinanceiro,
    QTD_PARCELAS,
    QTD_PARCELAS_PAGAS,
    
    ROUND(QTD_PARCELAS_PAGAS / QTD_PARCELAS, 0.01) AS pc_amort_parc,

    
    CASE
      WHEN CALCULATED pc_amort_parc = 0                                  THEN 'a. FPD'
      WHEN CALCULATED pc_amort_parc  > 0  AND CALCULATED pc_amort_parc <= 0.10 THEN 'b. até 10%'
      WHEN CALCULATED pc_amort_parc  > 0.10 AND CALCULATED pc_amort_parc <= 0.30 THEN 'c. 10% a 30%'
      WHEN CALCULATED pc_amort_parc >= 0.31 AND CALCULATED pc_amort_parc <= 0.70 THEN 'd. 31% a 70%'
      WHEN CALCULATED pc_amort_parc  > 0.70                               THEN 'e. >70%'
      ELSE 'N/A'
    END AS fx_pc_amort_parc
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_1
;
QUIT;

PROC SQL;
CREATE TABLE AMORTIZACAO_&SAFRA. AS
SELECT
    NUCONTRATO,
	sqcontratofinanceiro,
	MAX(QTD_PARCELAS) AS QTD_PARCELAS,
	MAX(QTD_PARCELAS_PAGAS) AS QTD_PARCELAS_PAGAS,
	MAX(pc_amort_parc) AS pc_amort_parc,
    MAX(fx_pc_amort_parc) AS fx_pc_amort_parc,
	&SAFRA. AS SAFRA
FROM ENRIQUECIMENTO_COM_AMORTIZACAO_2
GROUP BY NUCONTRATO,sqcontratofinanceiro
;QUIT;

		when a.nucontratogestao = d.nucontratogestao then 'S'
		else 'N'
	 end as Flag_Acordo_New

		when e.flag_bndu is missing then 0
		else e.flag_bndu
	  end as fl_bndu
	 ,case
		when e.flag_Rev is missing then 0
		else e.flag_Rev
	  end as fl_rev
	 
	 ,case
		when f.fl_protesto is missing then 0
		else f.fl_protesto
	  end as fl_protesto
	 ,case
		when e.Flag_Ajuiz is missing then 0
		else e.Flag_Ajuiz
	  end as fl_ajuiz
	  ,case
		when revi.contrato is not null then 1
		else 0
	  end as fl_rev_adicional
	  ,case
		when aju_cessao.contrato is not null then 1
		else 0
	  end as fl_aju_cessao
from contatos.aux_base_cyber as a
left join (	select distinct a.nucontrato
			from zbloq.bloqueio_reguas_intersec as a
			where 1=1
			and A.DT_REF = Today()
			and A.Flag_Bloqueio = 1
			and Motivo in ('Bloq_Cessao_Rodas','Bloq_Cessao_Cartao','Regra_Base_D0')) as b on a.nucontratogestao = b.nucontrato
left join work.base_mandado_expedido	as c on a.nucontratogestao = c.nucontrato

left join work.base_protesto_final		as f on	a.nucontratogestao = input(f.nucontratogestao,z14.)
												and a.SEQ_FINANCEIRA = f.sqcontratofinanceiro
left join work.FILTRO_ADICIONAL_REVI as revi on a.nucontratogestao = revi.contrato
left join work.FILTRO_AJU_CESSAO as aju_cessao on a.nucontratogestao = aju_cessao.contrato

and a.grupo = 'RODAS'

PROC SQL;
CREATE TABLE CRUZAMENTO1
AS SELECT
A.*
, B.COLLECTION_MODELO
, C.CdRegiao
, C.cdfilial
, C.Estabelecimento
, C.safra_Concessao
, C.CdLoja
, C.PcCompromentRenda
, C.PcEntrada
, C.Ano_Modelo
, C.dstipoprofissional
, C.Tipo_Cliente
, C.FlZeroKm
, C.idade_veiculo
, C.idade_cliente
, C.tppessoa
, C.marca_veiculo
, C.modelo_veiculo
, C.ds_subproduto
, C.ds_categoria
, C.qtparcelas AS qt_parcelas_originacao
, D.pc_amort_parc
, D.fx_pc_amort_parc
, D.QTD_PARCELAS
, D.QTD_PARCELAS_PAGAS
, D.QTD_PARCELAS - D.QTD_PARCELAS_PAGAS AS QTD_PARCELAS_ABERTAS

,CASE
    WHEN A.SALDO < 0 THEN "Valor inválido"
    WHEN A.SALDO <= 10000 THEN "A. Até 10k"
    WHEN A.SALDO <= 20000 THEN "B. 10k a 20k"
    WHEN A.SALDO <= 50000 THEN "C. 20k a 50k"
    WHEN A.SALDO <= 75000 THEN "D. 50k a 75k"
    WHEN A.SALDO <= 100000 THEN "E. 75k a 100k"
    WHEN A.SALDO > 100000 THEN "F. Acima de 100k"
    ELSE "Sem informação"
END AS FX_SALDO
,CASE
        WHEN  A.QTATRASO IS NULL THEN 'z. Missing'
        WHEN  A.QTATRASO < 0     THEN 'y. Negativo'
        WHEN  A.QTATRASO = 0                 THEN 'a. Em dia'
        WHEN  A.QTATRASO BETWEEN 1   AND 30  THEN 'b. 1 a 30'
        WHEN  A.QTATRASO BETWEEN 31  AND 60  THEN 'c. 31 a 60'
        WHEN  A.QTATRASO BETWEEN 61  AND 90  THEN 'd. 61 a 90'
        WHEN  A.QTATRASO BETWEEN 91  AND 120 THEN 'e. 91 a 120'
        WHEN  A.QTATRASO BETWEEN 121 AND 150 THEN 'f. 121 a 150'
        WHEN  A.QTATRASO BETWEEN 151 AND 180 THEN 'g. 151 a 180'
        WHEN  A.QTATRASO BETWEEN 181 AND 210 THEN 'h. 181 a 210'
        WHEN  A.QTATRASO BETWEEN 211 AND 240 THEN 'i. 211 a 240'
        WHEN  A.QTATRASO BETWEEN 241 AND 270 THEN 'j. 241 a 270'
        WHEN  A.QTATRASO BETWEEN 271 AND 300 THEN 'k. 271 a 300'
        WHEN  A.QTATRASO BETWEEN 301 AND 330 THEN 'l. 301 a 330'
        WHEN  A.QTATRASO BETWEEN 331 AND 360 THEN 'm. 331 a 360'
        WHEN  A.QTATRASO BETWEEN 361 AND 390 THEN 'n. 361 a 390'
        WHEN  A.QTATRASO BETWEEN 391 AND 420 THEN 'o. 391 a 420'
        WHEN  A.QTATRASO BETWEEN 421 AND 450 THEN 'p. 421 a 450'
        WHEN  A.QTATRASO BETWEEN 451 AND 480 THEN 'q. 451 a 480'
        WHEN  A.QTATRASO BETWEEN 481 AND 510 THEN 'r. 481 a 510'
        WHEN  A.QTATRASO BETWEEN 511 AND 540 THEN 's. 511 a 540'
        WHEN  A.QTATRASO BETWEEN 541 AND 720 THEN 't. 541 a 720'
		WHEN  A.QTATRASO BETWEEN 720 AND 1080 THEN 'u. 721 a 1080'
        WHEN  A.QTATRASO > 1080               THEN 'v. Over1080'
        ELSE 'z. Missing'
	END as FX_ATRASO_M0
	, CASE WHEN G.molicar_defl_prop = . THEN "Z. SEM INFO"
WHEN A.SALDO = . THEN "Z. SEM INFO"
WHEN G.molicar_defl_prop / A.SALDO = 0 THEN "A. 0%"
WHEN G.molicar_defl_prop / A.SALDO <=0.1 THEN "B. 0% a 10%"
WHEN G.molicar_defl_prop / A.SALDO <=0.25 THEN "C. 10% a 25%"
WHEN G.molicar_defl_prop / A.SALDO <=0.50 THEN "D. 25% a 50%"
WHEN G.molicar_defl_prop / A.SALDO <=0.75 THEN "E. 50% a 75%"
WHEN G.molicar_defl_prop / A.SALDO <=0.75 THEN "F. 75% a 100%"
WHEN G.molicar_defl_prop / A.SALDO <=1.5 THEN "G. 100% a 150%"
WHEN G.molicar_defl_prop / A.SALDO >1.5 THEN "H. > 150%"
ELSE "ZZ.INVESTIGAR" END AS FX_COBERTURA
FROM base_carteira A
LEFT JOIN ltv_novo_emp B
ON A.NUCONTRATO = B.nucontrato
AND A.SQCONTRATOFINANCEIRO = B.sqcontratofinanceiro
AND A.safra_consulta = B.anomes
LEFT JOIN BU2 C

create table work.analise_wo as

select distinct
	a.*
	,case
		when a.nucontratogestao = b.nucontrato  then 'S'
		else 'N'
	 end as Flag_Filtro_List_Rest_Int
	,case
		when a.nucontratogestao = c.nucontrato then 'S'
		else 'N'
	end as Flag_Mand_Exp
	,case
		when a.nucontratogestao = d.nucontratogestao then 'S'
		else 'N'
	 end as Flag_Acordo_New
	 ,case
		when e.flag_bndu is missing then 0
		else e.flag_bndu
	  end as fl_bndu
	 ,case
		when e.flag_Rev is missing then 0
		else e.flag_Rev
	  end as fl_rev
	 
	 ,case
		when f.fl_protesto is missing then 0
		else f.fl_protesto
	  end as fl_protesto
	 ,case
		when e.Flag_Ajuiz is missing then 0
		else e.Flag_Ajuiz
	  end as fl_ajuiz
	  ,case
		when revi.contrato is not null then 1
		else 0
	  end as fl_rev_adicional
	  ,case
		when aju_cessao.contrato is not null then 1
		else 0
	  end as fl_aju_cessao
from contatos.aux_base_cyber as a
left join (	select distinct a.nucontrato
			from zbloq.bloqueio_reguas_intersec as a
			where 1=1
			and A.DT_REF = Today()
			and A.Flag_Bloqueio = 1
			and Motivo in ('Bloq_Cessao_Rodas','Bloq_Cessao_Cartao','Regra_Base_D0')) as b on a.nucontratogestao = b.nucontrato
left join work.base_mandado_expedido	as c on a.nucontratogestao = c.nucontrato
left join work.base_acordos_01			as d on a.nucontratogestao = d.nucontratogestao
left join work.base_juridico			as e on a.NuContratoGestao = e.NuContratoGestao
left join work.base_protesto_final		as f on	a.nucontratogestao = input(f.nucontratogestao,z14.)
												and a.SEQ_FINANCEIRA = f.sqcontratofinanceiro
left join work.FILTRO_ADICIONAL_REVI as revi on a.nucontratogestao = revi.contrato
left join work.FILTRO_AJU_CESSAO as aju_cessao on a.nucontratogestao = aju_cessao.contrato

where 1=1
and a.grupo = 'RODAS'

order by a.cod_assessoria asc

;quit;

create table validacao_pct as
select
    case
        when a.cod_assessoria_novo='2022' then 'CONCILIG'
        when a.cod_assessoria_novo='1618' then 'MENDES E CUNHA'
        when a.cod_assessoria_novo='133'  then 'SCHULZE'
        when a.cod_assessoria_novo='1'    then 'TOLEDO'
        when a.cod_assessoria_novo='1336' then 'PASCHOALOTTO'
        when a.cod_assessoria_novo='14'   then 'BELLINATI'
    end as assessoria length=30,

    a.faixa,
    count(*) as qtd,
    round(count(*)/b.total*100,0.01) as pct

from base_redist_final a

left join (
    select faixa,
           count(*) as total
    from base_redist_final
    group by faixa
) b
on a.faixa=b.faixa

group by assessoria,
         a.faixa,
         b.total;
quit;
