	connect to SYBASEIQ (&conex_syb.);

	create table cdccg001 as
		SELECT * from connection to SYBASEIQ
		(	declare @data_base date
			SELECT @data_base = &ult_dia.
			SELECT @data_base as DtBase,
       a.NuContrato,
       a.SqContratoFinanceiro,
       a.NuParcela,
       a.StParcela,
       a.DtVencimento,
       a.DtUltimoRecebimento,
	   c.NuCpfCnpj,
       c.CdModalidadeProduto,

       case when a.DtVencimento > @data_base then 0

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is not NULL then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and DtUltimoRecebimentoAj is NULL then datediff(day,a.DtVencimento,@data_base)

                 then datediff(day,a.DtVencimento,DtUltimoRecebimentoAj)

            when a.StParcela in ('A','E') and DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj > @data_base

proc sql noprint;
	connect to SYBASEIQ (&conex_syb.);

	create table lsg01 as

			(
declare @data_base date

       a.NuContrato,
       a.SqContratoFinanceiro,
       a.NuParcela,
       a.StParcela,
       a.DtVencimento,
       a.DtUltimoRecebimento,
	   c.NuCpfCnpj,
	   c.CdModalidadeProduto,
	   c.DtLiquidacaoEfetiva,
       case when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is not NULL and b.DtContratoFinanceiro <= @data_base then b.DtContratoFinanceiro
            when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is NULL and c.DtLiquidacaoEfetiva is not NULL and c.DtLiquidacaoEfetiva <= @data_base then c.DtLiquidacaoEfetiva
            when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is NULL and c.DtLiquidacaoEfetiva is NULL and c.DtCancelamento is not NULL and c.DtCancelamento <= @data_base then c.DtCancelamento
            else a.DtUltimoRecebimento
       end as DtUltimoRecebimentoAj,

       case when a.DtVencimento > @data_base then 0
            when a.StParcela in ('Q') and a.DtUltimoRecebimento < a.DtVencimento then 0
            when a.StParcela in ('Q') and a.DtUltimoRecebimento between a. DtVencimento and @data_base then datediff(day,a.DtVencimento,a.DtUltimoRecebimento)
            when a.StParcela in ('Q') and a.DtUltimoRecebimento > @data_base then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is not NULL then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and DtUltimoRecebimentoAj is NULL then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj < a.DtVencimento  then 0

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj between a.DtVencimento and @data_base
                 then datediff(day,a.DtVencimento,DtUltimoRecebimentoAj)

            when a.StParcela in ('A','E') and DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj > @data_base
                 then datediff(day,a.DtVencimento,@data_base)

            else -99
       end as AtrasoMaximo

from lsg.TbParcela as a left join lsg.TbContratoFinanceiro as b on a.NuContrato=b.NuContrato and a.SqContratoFinanceiro=(b.SqContratoFinanceiro-1)

proc sql noprint;
	connect to SYBASEIQ (&conex_syb.);
		create table cdcsg1 as
			select * from connection to SYBASEIQ
			(
Declare @data_base Varchar(8)
Set @data_base=&ult_dia.

select @data_base as DtBase,
       a.NuContrato,
       a.SqContratoFinanceiro,
       a.NuParcela,
       a.StParcela,
       a.DtVencimento,
       a.DtUltimoRecebimento,
	   c.NuCpfCnpj,
       c.CdModalidadeProduto,
	   c.DtLiquidacaoEfetiva,
       case when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is not NULL and b.DtContratoFinanceiro <= @data_base then b.DtContratoFinanceiro
            when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is NULL and c.DtLiquidacaoEfetiva is not NULL and c.DtLiquidacaoEfetiva <= @data_base then c.DtLiquidacaoEfetiva
            when a.DtUltimoRecebimento is NULL and b.DtContratoFinanceiro is NULL and c.DtLiquidacaoEfetiva is NULL and c.DtCancelamento is not NULL and c.DtCancelamento <= @data_base then c.DtCancelamento
            else a.DtUltimoRecebimento
       end as DtUltimoRecebimentoAj,

       case when a.DtVencimento > @data_base then 0
            when a.StParcela in ('Q') and a.DtUltimoRecebimento < a.DtVencimento then 0
            when a.StParcela in ('Q') and a.DtUltimoRecebimento between a. DtVencimento and @data_base then datediff(day,a.DtVencimento,a.DtUltimoRecebimento)
            when a.StParcela in ('Q') and a.DtUltimoRecebimento > @data_base then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is not NULL then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and DtUltimoRecebimentoAj is NULL then datediff(day,a.DtVencimento,@data_base)

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj < a.DtVencimento  then 0

            when a.StParcela in ('A','E') and a.DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj between a.DtVencimento and @data_base
                 then datediff(day,a.DtVencimento,DtUltimoRecebimentoAj)

            when a.StParcela in ('A','E') and DtUltimoRecebimento is NULL and DtUltimoRecebimentoAj > @data_base
                 then datediff(day,a.DtVencimento,@data_base)

            else -99
       end as AtrasoMaximo

from cdcsg.TbParcela as a left join cdcsg.TbContratoFinanceiro as b on a.NuContrato=b.NuContrato and a.SqContratoFinanceiro=(b.SqContratoFinanceiro-1)
                          left join cdcsg.TbContrato           as c on a.NuContrato=c.NuContrato
where a.DtVencimento <= @data_base and AtrasoMaximo > 180
);
	QUIT;

/*Formatando a data*/
DATA cdcsg2;
    SET cdcsg1;
    FORMAT DtEvento date9.;
DtEvento=datepart(DtVencimento);
Motivo = 'Atraso';
IF DtLiquidacaoEfetiva NE '' Then ContratoAtivo = 'N'; else ContratoAtivo = 'S';
KEEP NuContrato NuCpfCnpj SqContratoFinanceiro Motivo DtEvento CdModalidadeProduto AtrasoMaximo DtLiquidacaoEfetiva ContratoAtivo;
RUN;

	IF ContratoAtivoCSG = 'S' And PossuiCtroLiqAtrasoCSG = '' THEN PossuiCtroLiqAtrasoCSG = 'N';

	DtEvento = max (DtEventoRodas, DtEventoBNDUSaldoRemanesc, DtEventoCP,
	DtEventoCSG, DtEventoCDCSG, DtEventoACRF);

	IF DtEvento = DtEventoRodas THEN DO;
			Produto = ProdutoRodas;
			NuContrato = NuContratoRodas;
			Motivo = 'Atraso'; END; ELSE

	IF DtEvento = DtEventoBNDUSaldoRemanesc THEN DO;
			Produto = ProdutoBNDUSaldoRemanesc;
			NuContrato = NuContratoBNDUSaldoRemanesc;
			Motivo = BNDUSaldoRemanesc; END; ELSE

	IF DtEvento = DtEventoCP THEN DO;
			Produto = ProdutoCP;
			NuContrato = NuContratoCP;
			Motivo = 'Atraso'; END; ELSE

	IF DtEvento = DtEventoCSG THEN DO;
			Produto = ProdutoCSG;
			NuContrato = NuContratoCSG;
			Motivo = 'Atraso'; END; ELSE

	IF DtEvento = DtEventoCDCSG THEN DO;
			Produto = ProdutoCDCSG;
			NuContrato = NuContratoCDCSG;
			Motivo = 'Atraso'; END; ELSE

	IF DtEvento = DtEventoACRF THEN DO;
			Produto = ACRF;
			NuContrato = NuContratoACRF;
			Motivo = 'ACRF';
	END;

	FORMAT DtEventoAtraso180 date9.;

	DtEventoAtraso180 = max (DtEventoRodas, DtEventoCP, DtEventoCSG, DtEventoCDCSG);

	QtCtrosAtraso180 = sum(QtCtrosAtraso180Rodas, QtCtrosAtraso180CSG,
		QtCtrosAtraso180CP, QtCtrosAtraso180CDCSG);

	AtrasoMaximo = max(AtrasoRodas, AtrasoCtroLiquidadoRodas, AtrasoCP,
		AtrasoCtroLiquidadoCP, AtrasoCSG, AtrasoCtroLiquidadoCSG, AtrasoCDCSG);

	IF Produto = 'xxx' Then Produto = 'Leves';

RUN;

         SELECT distinct NuContrato, NuCpfCnpj, DtContrato, DtLiquidacaoEfetiva, DtCancelamento

         SELECT distinct NuContrato, NuCpfCnpj, DtContrato, DtLiquidacaoEfetiva, DtCancelamento

			coalesce(a.NuContrato,b.NuContrato,c.NuContrato,d.NuContrato,e.NuContrato) AS NuContrato,

		 FULL JOIN #CtrtCP c on a.NuContrato = c.NuContrato
		 FULL JOIN #CtrtCPC d on a.NuContrato = d.NuContrato
		 FULL JOIN #CtrtCDCSG e on a.NuContrato = e.NuContrato

		 SELECT *

		 FROM #Ctrts
		 WHERE DtCancelamento is null and Dtcontrato <= @data_base
     );

%MACRO APPEND_DRE(DATA, LIMITE);

					RUN;

					RUN;
				%END;
			%LET DATA = %EVAL(&DATA. - 1);
		%END;
	%ELSE %DO;
		%LET DATA = %EVAL(&DATA. - 1);
	%END;
%END;
%MEND;

			PLD-VEICULOS - 2 ou mais contratos quitados até 90 dias no periodo de 12 meses
		A regra contempla apenas o Produto Rodas e contratos originados a partir de 2014

***************************************************************************/

/*Contratos Liquidados em Rodas*/
PROC SQL noprint;
CONNECT TO SYBASEIQ (&conex_syb.);
CREATE TABLE Rodas_Liquidados&Safra. AS SELECT * FROM connection to SYBASEIQ
(

declare @data_base date
SELECT @data_base = &ult_dia.

SELECT
	NuContrato,
	DtContrato AS DtContrato1,
	DtLiquidacaoEfetiva AS DtLiquidacaoEfetiva1,
	NuCpfCnpj,
	CdModalidadeProduto,
	@data_base AS DtBase
INTO #msbbc
FROM CDCCG.TbContrato
WHERE DtContrato >= '20140101'
	  AND DtLiquidacaoEfetiva IS NOT NULL
	  AND DtLiquidacaoEfetiva <= @data_base
	  AND DtCancelamento IS NULL

INSERT INTO #msbbc
SELECT
	NuContrato,
	DtContrato AS DtContrato1,
	DtLiquidacaoEfetiva AS DtLiquidacaoEfetiva1,
	NuCpfCnpj,
	CdModalidadeProduto,
	@data_base AS DtBase

FROM LSG.TbContrato
WHERE DtContrato >= '20140101'
	  AND DtLiquidacaoEfetiva IS NOT NULL
	  AND DtLiquidacaoEfetiva <= @data_base
	  AND DtCancelamento IS NULL

SELECT * FROM #msbbc

);
	QUIT;

DATA BaseLiquidados;
	SET Rodas_Liquidados&Safra. ;

	FORMAT DtContrato DATE9. DtLiquidacao DATE9.;
	DtContrato = DATEPART(DtContrato1);
	DtLiquidacao = DATEPART(DtLiquidacaoEfetiva1);

	NuMOBQuitacao = INTCK("month",DtContrato,DtLiquidacao);
	NuDiasAntecipacao = INTCK("day",DtContrato,DtLiquidacao);

	LENGTH Produto $11.;
	IF CdModalidadeProduto in (1,2,3,6,7,8,25,26,27,38,41,46,48)	THEN Produto = 'Leves'; ELSE
    IF CdModalidadeProduto in (5,49)                            	THEN Produto = 'Motos'; ELSE
    IF CdModalidadeProduto in (4,9,28,47)                       	THEN Produto = 'Pesados'; ELSE Produto = 'Leves';

	DROP
	DtContrato1
	DtLiquidacaoEfetiva1
	;

RUN;

/*Mantem na base apenas os contratos liquidados até 90 dias (MOB 3)*/
DATA BaseLiquidados1;
	SET BaseLiquidados;
WHERE NuDiasAntecipacao <= 90;
RUN;

/*Marca na base os Clientes (NuCpfCnpj) e as quantidades de contratos liquidados no periodo de 90  dias*/
PROC SQL;
CREATE TABLE BaseLiquidados2 AS
SELECT
	*,
	COUNT(NuContrato) AS QtdCtrosLiquidados
FROM BaseLiquidados1
GROUP BY NuCpfCnpj;
	QUIT;

/*Mantem na base apenas clientes que possuem mais que 1 (um) contrato liquidado
	em até 90 dias da data de inicio do contrato*/
DATA BaseLiquidados3;
	SET BaseLiquidados2;
WHERE QtdCtrosLiquidados > 1;
RUN;

/*Ordena a base de dados pela data de liquidação*/
PROC SORT DATA=BaseLiquidados3;
BY NuCpfCnpj DESCENDING DtLiquidacao;
RUN;

/*Cria INDEX para calcular se as quitações antecipadas foram dentro do prazo de 12 meses*/
DATA BaseLiquidados4;
SET BaseLiquidados3;
    INDEX = _N_;
RUN;

PROC SQL NOPRINT;
CREATE TABLE Minimo AS
SELECT
	NuCpfCnpj,
	min(INDEX) as Minimo,
count(*) as QTDE
FROM BaseLiquidados4
GROUP BY NuCpfCnpj;
QUIT;

PROC SQL NOPRINT;
CREATE TABLE BaseLiquidados4 AS
SELECT
	a.*,
	b.Minimo,
	(INDEX - b.Minimo + 1) AS Ranking
FROM BaseLiquidados4 AS A LEFT JOIN Minimo as B ON A.NuCpfCnpj = B.NuCpfCnpj;
	QUIT;

PROC SORT DATA=BaseLiquidados4;
BY DESCENDING index;
RUN;

PROC SQL NOPRINT;
CREATE TABLE BaseLiquidados5 AS
SELECT
	a.*,
	b.DtLiquidacao AS DtLiquidacaoPosterior
FROM BaseLiquidados4 AS A left join BaseLiquidados4 AS B ON A.NuCpfCnpj = B.NuCpfCnpj and b.Ranking +1= a.Ranking
ORDER BY index DESC;
QUIT;

DATA BaseLiquidados6;
SET BaseLiquidados5;

TempoQuitacao = INTCK("month",DtLiquidacao,DtLiquidacaoPosterior);
IF TempoQuitacao = . THEN TempoQuitacao = 99; /*99 apenas para deixar o registro no fim*/

RUN;

/*Mantem na base quitações antecipadas dentro do prazo de 12 meses*/
PROC SORT DATA=BaseLiquidados6 OUT= BaseLiquidados7;
	BY NuCpfCnpj TempoQuitacao;
WHERE TempoQuitacao <= 12;
RUN;

/*Mantem na base apenas o CPF/CNPJ com mais  que um contratos quitado antecipadamente
até 90 dias da data de inicio no periodo de até 12 meses*/
PROC SORT DATA=BaseLiquidados7 OUT=BaseLiquidados8 NODUPKEY;
	BY NuCpfCnpj;
RUN;

/*Gera a base final de PLD-Veiculos*/
DATA BEH.PLDVEICULOS_&Safra.;
	SET BaseLiquidados8;

	PLD = 'S';
	LENGTH RU_PLD $12.;
	RU_PLD = 'PLD-Veiculos';

	DROP
	INDEX
	Minimo
	Ranking
	NuContrato
	;

RUN;

/*Limpa a Work*/
PROC DATASETS LIB=WORK
	KILL;
RUN;

/************************************************************************************************************

     Título:      Código para marcacao de CPF/CNPJ de Lojistas.
     Responsável:  Riscos - Estratégia de Crédito
     Autor:       Márcio Bezerra
     Criação:     julho/2014
     Ultima Atualização: 31 de agosto de 2018 Márcio Bezerra

     Descrição:  Programa para gerar a base com CNPJ/CPF de lojistas e parceiros comerciais
    do produto Rodas para Black / White List do Motor de Crédito

     O QUE FALTA:  Nada

************************************************************************************************************/

PROC SQL noprint;
CONNECT TO SYBASEIQ (&conex_syb.);
CREATE TABLE BaseFullLojas_&Safra. AS SELECT * FROM connection to SYBASEIQ
(

Declare @DtInicio Varchar(8)
Declare @DtFim Varchar(8)
Set @DtInicio=&PrimDia.
Set @DtFim=&ult_dia.

SELECT
	a.CdLoja,
	a.NmLoja,
	a.FlAtivo,
	b.DtCadastro,
	b.DtUltimaAtualizacao,
	a.CdRegiao,
	e.DsRegiao,
	convert(char(14),a.NuCpfCnpj) AS NuCpfCnpjLoja,
	a.TpAtividade, /* depara vide:dbcor..TbAtividadeFinanciamento  -- 1 = Veiculos*/
	b.CdTipoParceiroComercial,
	c.NmTipoParceiroComercial,
	d.SqRepresentanteLegal,
	d.PcParticipacao,
	TipoLoja= Case When (e.DsRegiao like '%(CP)%') or
	                    (e.DsRegiao like '%SP - Sao Bento%') or
	                    (e.DsRegiao like '%Paulista%') or
	                    (e.DsRegiao like '%SP - Renegociação (CA)%') or
	                    (a.NmLoja like '%BV PART%') or
	                    (a.NmLoja like '%BV BALC%') or
	                    (a.NmLoja like '%CP GARANTIA%') or
	                    (a.NmLoja like '%LOJA RENEG%') or
	                    (a.NmLoja like '%LOJA VIRTUAL%') or
	                    (a.NuCpfCnpj like '%01149953%') or
	                    (a.NuCpfCnpj like '%44990901%') or
	                    (a.NuCpfCnpj like '%01219624%')
	                            THEN 'CP'
	                        else 'Outros'
	                 END

FROM Dba.TbLoja AS A LEFT JOIN dba.TbParceiroComercial AS b         ON a.CdLoja=b.CdParceiroComercial
                     LEFT JOIN DBCOR.TbTipoParceiroComercial AS c   ON b.CdTipoParceiroComercial=c.CdTipoParceiroComercial
                     LEFT JOIN DBCOR.TbSocio AS d                   ON a.CdLoja=d.CdParceiroComercial
                     LEFT JOIN Dba.TbRegiao AS e                    ON a.cdregiao=e.cdregiao

WHERE a.TpAtividade = '01'

ORDER BY a.CdLoja, d.SqRepresentanteLegal, b.DtCadastro

);
    QUIT;

PROC SQL noprint;
CONNECT TO SYBASEIQ (&conex_syb.);
CREATE TABLE TbRepresentanteLegal AS SELECT * FROM connection to SYBASEIQ
	(

SELECT
    CdParceiroComercial,
    SqRepresentanteLegal,
    DtUltimaAtualizacao,
    DtCadastro,
    NmRepresentanteLegal,
    NuCpf,
    DtNascimento
FROM DBCOR.TbRepresentanteLegal
WHERE NmRepresentanteLegal <> 'Teste'
ORDER BY CdParceiroComercial, SqRepresentanteLegal, DtCadastro

	);
QUIT;

PROC SQL;
CREATE TABLE BaseFinalLojas_&Safra. AS
SELECT
    a.*,
    b.NuCpf AS NuCpfCnpjSocio,
    b.NmRepresentanteLegal,
    b.DtNascimento
FROM BaseFullLojas_&Safra. AS a left join TbRepresentanteLegal AS b on a.CdLoja=b.CdParceiroComercial
    AND a.SqRepresentanteLegal=b.SqRepresentanteLegal
WHERE TipoLoja ne 'CP';
    QUIT;

DATA BaseFinalLojas1_&Safra.;
SET BaseFinalLojas_&Safra.;

	IF  NuCpfCnpjLoja = NuCpfCnpjSocio Then ID = 1;  else ID = 0;

	FORMAT DtCadastro2 DATE9. DtUltimaAtualizacao2 DATE9.;
	DtCadastro2 = DATEPART(DtCadastro);
	DtUltimaAtualizacao2 = DATEPART(DtUltimaAtualizacao);

	DROP
	DtCadastro
	DtUltimaAtualizacao
	;

	RENAME
	DtCadastro2 = DtCadastro
	DtUltimaAtualizacao2 = DtUltimaAtualizacao
	;

	FORMAT DtProcessamento DATE9.;
	DtProcessamento = "&dtbasefim"d;

RUN;

/*Gera base final de lojistas  - todo mes será realizado update*/
DATA BEH.TbLojistas_BWL
	(KEEP=CdLoja NmLoja FlAtivo DtCadastro DtUltimaAtualizacao CdRegiao
	DsRegiao NuCpfCnpjLoja NuCpfCnpjSocio PcParticipacao DtProcessamento)
	;
	SET BaseFinalLojas1_&Safra.;
RUN;

/*Inicia geração do arquivo para Black White List*/
/* O arquivo final conterá apenas as atualizações ou inserções de lojas ocorridas
no ultimo  mes, os meses anteriores foram enviados em suas respectivas datas
por isso a utilizacao do filtro de dtbaseini  dtbaseim abaixo*/

PROC SQL;
CREATE TABLE base_cpfsocio_&Safra. AS
SELECT DISTINCT
NuCpfCnpjSocio AS I,
11 AS BWL_PRD
FROM BaseFinalLojas1_&Safra.
WHERE ID = 0 AND I NOT IS NULL AND DtUltimaAtualizacao BETWEEN "&dtbaseini"D AND "&dtbasefim"D;
    QUIT;

PROC SQL;
CREATE TABLE base_cnpjloja_&Safra. AS
SELECT DISTINCT
NuCpfCnpjLoja AS I,
11 AS BWL_PRD
FROM BaseFinalLojas1_&Safra.
WHERE ID = 0 AND I NOT IS NULL AND DtUltimaAtualizacao BETWEEN "&dtbaseini"D AND "&dtbasefim"D;
    QUIT;

/* Empilha o CNPJ da Loja com o CPF dos socios*/
DATA ArquivoTbLojistas_BWL_&Safra.;
set base_cpfsocio_&Safra. base_cnpjloja_&Safra.;
RUN;

/*retira duplicidade se houver*/
PROC SORT DATA=ArquivoTbLojistas_BWL_&Safra. NODUPKEY ;
BY I;
RUN;
