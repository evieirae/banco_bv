	VALIDA_SUBIDA_ASS AS

	A.*
	,B.cod_asses

			DATE(dt_asses) <= '2025-12-08' AND dt_ret_asses IS null) B
ON
	A.NUCONTRATO = INPUT(B.NUCONTRATOGESTAO, 17.)
GROUP BY
	A.NUCONTRATO
HAVING
	dt_asses = MAX(dt_asses)

;QUIT;
