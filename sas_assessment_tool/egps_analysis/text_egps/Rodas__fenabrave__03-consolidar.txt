option compress = binary;

libname mis '/riscosbvf/CRM/SAS/DM_MIS';
LIBNAME automati '/riscosbvf/CRM/SAS/AddIn';

proc sql;
create table mis.tb_fenabrave_mktshare_00 as
select

	 input(compress(TRANWRD(TRANWRD(TRANWRD(nmarquivo,'Emplacamentos_',''),'.txt',''),'_','')),6.) as base
	,Estado as CdUf
	,'Novo' as StVeiculo
	,case
		when Segmento_Codigo = 1 then 'Leves'
		when Segmento_Codigo = 2 then 'Leves'
		when Segmento_Codigo = 6 then 'Motos'
		when Segmento_Codigo = 3 then 'Pesados'
		when Segmento_Codigo = 4 then 'Pesados'
	 end as TpVeiculo
	,count(*) as Qtd
from tb_fenabrave_emplacamentos
where Segmento_Codigo in (1,2,3,4,6)
  and Estado is not null
group by 1,2,3,4

union all

select
	 input(compress(TRANWRD(TRANWRD(TRANWRD(nmarquivo,'Transacoes',''),'.txt',''),'_','')),6.)
	,Estado as CdUf
	,'Usado' as StVeiculo
	,case
		when Segmento_Codigo = 1 then 'Leves'
		when Segmento_Codigo = 2 then 'Leves'
		when Segmento_Codigo = 6 then 'Motos'
		when Segmento_Codigo = 3 then 'Pesados'
		when Segmento_Codigo = 4 then 'Pesados'
	 end as TpVeiculo
	,count(*) as Qtd
from tb_fenabrave_transacoes
where Segmento_Codigo in (1,2,3,4,6)
  and Estado is not null
group by 1,2,3,4

;run;

proc sql;
insert into mis.tb_fenabrave_mktshare_00
select
	 year(base)*100+month(base)
	,CdUF
	,case
		when StVeiculo = 1 then 'Novo'
		when StVeiculo = 2 then 'Usado'
	 end as StVeiculo
	,case
		when TpVeiculo = 1 then 'Leves'
		when TpVeiculo = 1 then 'Leves'
		when TpVeiculo = 3 then 'Motos'
		when TpVeiculo = 4 then 'Pesados'
		when TpVeiculo = 4 then 'Pesados'
	 end as TpVeiculo
	,Qtd
from automati.tb_integmktshare_fenabrave
;run;

;proc sql;
create table TbModeloCombustivel as
select distinct
	 Modelo_Codigo
	,coalesce(Combustivel_Codigo,0) as Combustivel_Codigo
	,coalesce(Combustivel,'Indeterminado') as Combustivel
from TbFenabrave
where Modelo_Codigo > 0
order by
	case
		when Combustivel_Codigo = 99 then 1
		when Combustivel_Codigo is null then 1
		when Combustivel_Codigo > 0  then 0
	else 1 end
;run;

proc sort data=TbModeloCombustivel out=automati.TbDimModeloCombustivel nodupkey;by Modelo_Codigo;run;

/*GFM - Tabela com os dados do Fabricante de Motos para atualizar o Mkt Share do COmercial*/
/******************************************************************************************/
proc sql;
create table tbGrupoModeloVeiculo as
select
	 Fenabrave.Fabricante
	,Fenabrave.Fabricante_Codigo
	,Fenabrave.GrupoModeloVeiculo
	,Fenabrave.GrupoModeloVeiculo_Codigo
	,Fenabrave.Modelo
	,Fenabrave.Modelo_Codigo
	,Fenabrave.Segmento
	,Fenabrave.Segmento_Codigo
	,Fenabrave.SubSegmento
	,Fenabrave.SubSegmento_Codigo
	,count(*) as QtOcorrencias
	,coalesce(Cilindrada_Complemento.Cilindrada,MotoCilindrada.Cilindrada) as Cilindrada
	,Cilindrada_Complemento.FxCilindrada
from TbFenabrave Fenabrave

	left join MKTM.tbGrupoModeloVeiculo MotoCilindrada
			on Fenabrave.Modelo_Codigo = MotoCilindrada.modelo_codigo

	left join basegrc.Cilindrada_Complemento as Cilindrada_Complemento
			on Fenabrave.Modelo_Codigo = Cilindrada_Complemento.modelo_codigo

where Fenabrave.Segmento_Codigo = 6

group by 	 Fenabrave.Fabricante
	,Fenabrave.Fabricante_Codigo
	,Fenabrave.GrupoModeloVeiculo
	,Fenabrave.GrupoModeloVeiculo_Codigo
	,Fenabrave.Modelo
	,Fenabrave.Modelo_Codigo
	,Fenabrave.Segmento
	,Fenabrave.Segmento_Codigo
	,Fenabrave.SubSegmento
	,Fenabrave.SubSegmento_Codigo
	,coalesce(Cilindrada_Complemento.Cilindrada,MotoCilindrada.Cilindrada)
	,Cilindrada_Complemento.FxCilindrada

order by QtOcorrencias desc

;run;

proc sort data=tbGrupoModeloVeiculo nodupkey; by modelo_codigo;run;
