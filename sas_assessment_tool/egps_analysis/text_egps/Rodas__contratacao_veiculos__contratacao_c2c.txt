PROC SQL; SELECT MAX(DtContrato) format DATE9. INTO: maxContrato FROM CONTRATO.CONTRATACAO_C2C; QUIT;

DATA _NULL_;

dt_contrato		= intnx('Month',"&maxContrato"d,-1,'b' );
dt_contrato_sql = "'"||put(year(dt_contrato),4.)||put(month(dt_contrato),z2.)||put(day(dt_contrato),z2.)||"'";

CALL symputx('dt_contrato_sas', put(dt_contrato,date9.));
CALL symputx('dt_contrato_sql', dt_contrato_sql);

RUN;

/*%let dt_contrato_sas >= '01JUL2020'd;*/
/*%let dt_contrato_sql >= '20200101';*/

%put &maxContrato;
%put &dt_contrato_sas;
%put &dt_contrato_sql;

/*IDENTIFICA AS PROPOSTAS DE PESSOA FÍSICA PAGAS A PESSOA FÍSICA - C2C*/

PROC SQL;
%CONECTADB(server=CTR);
CREATE TABLE CONTRATOS_PAGAMENTO AS
SELECT * FROM CONNECTION TO DBCON(
	SELECT
		  T1.NuProposta
		, T1.NuContratoGestao AS NuContrato
		, CAST(T9.DtContrato AS DATE) AS DtContrato
		, CAST(T1.DtEntradaProposta AS DATE) AS DtEntradaProposta
		, T1.CdLoja
		, T4.NmGrupoSubProduto
		, T5.NuCpfCnpj
	FROM
		DBCRED.dbo.TbProposta T1
	INNER JOIN
		DBCOR.dbo.TbSubProduto T2
		ON T1.CdSubProduto = T2.CdSubProduto
		AND T2.CdProduto = 12 /*PRODUTO 12 = VEÍCULOS*/
	INNER JOIN
		DBCRED.dbo.TbSubProdutoGrupo T3
		ON T2.CdSubProduto = T3.CdSubProduto
	INNER JOIN
		DBCRED.dbo.TbGrupoSubProduto T4
		ON T3.CdGrupoSubProduto = T4.CdGrupoSubProduto
	INNER JOIN
		DBCRED.dbo.TbPessoaSobAnalise T5
		ON T1.NuProposta = T5.NuProposta
		AND T5.CdPapel = 1 /* 1 = PROPONENTE*/
		AND T5.TpPessoa = 'F' /* PESSOA FÍSICA*/
	INNER JOIN
		DBPAGAMENTO.dbo.TbPagamento T6
		ON T6.NuContratoGestao = T1.NuContratoGestao
	INNER JOIN
		DBPAGAMENTO.dbo.TbInformacaoBancariaPagamento T7
		ON T7.CdInformacaoBancariaPagto = T6.CdInforBancariaPagtoFavorecido
	INNER JOIN
		DBPAGAMENTO.dbo.TbPessoaPagamento T8
		ON T8.CdPessoaPagamento = T7.CdPessoaPagamento
		AND T8.TpPessoa = 'F'
	INNER JOIN
		DBGESTAOCDCCG.dbo.TbContrato T9
		ON T1.NuContratoGestao = T9.NuContrato
	WHERE T9.DtContrato >= &dt_contrato_sql
/*	AND T1.DtEntradaProposta <= '20220930'*/
	AND T1.StProposta = 'AP' 		/*SOMENTE PROPOSTAS COM STATUS APROVADA E PAGA*/
	AND T8.NuCpfCnpj IS NOT NULL 	/*DOCUMENTO DO FAVORECIDO PRECISAR ESTAR IDENTIFICADO*/
	AND T5.NuCpfCnpj <> T8.NuCpfCnpj /*RECEBEDOR É O PRÓPRIO PROPONENTE - NESTE CASO É CVG - B2C*/

);QUIT;

PROC SORT DATA=CONTRATOS_PAGAMENTO NODUPKEY; BY NuContrato; RUN;

PROC SQL; DELETE FROM CONTRATO.CONTRATACAO_C2C WHERE DtContrato >= "&dt_contrato_sas."d; QUIT;
PROC SQL; INSERT INTO CONTRATO.CONTRATACAO_C2C SELECT * FROM CONTRATOS_PAGAMENTO; QUIT;
