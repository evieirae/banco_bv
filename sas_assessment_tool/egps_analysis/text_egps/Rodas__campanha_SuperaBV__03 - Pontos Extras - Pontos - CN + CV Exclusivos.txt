PROC SQL;
CREATE TABLE TbPts_Crescimento_1 AS
SELECT

A.NUANOMES,
A.DSLOGINN1,
sum(CASE WHEN A.Pontuacao <= 0 THEN 0 ELSE A.Pontuacao END) AS VrPTS

FROM WORK.TbTabelaPontos A

where NUANOMES = &Base_Mes.

group by 1,2
order by 2,1

;RUN;

/*SUBSTITUI TUDO QUE FOI "MISSING OU '.' POR '0'*/
/*ALGUNS CASOS ESTÃO "VAZIOS" AO FINALIZAR O SCRIPT*/

data work.TbPts_Crescimento_2;
	set work.TbPts_Crescimento_1;

	array num_array _numeric_;
	do over num_array;
		if missing(num_array) then num_array = 0;
	end;
run;

PROC SQL;
CREATE TABLE BASE_CALCULO_PTS_1 AS
SELECT

&Base_Mes. AS NuAnoMes,
c.DtAdmissaoColaborador,
c.NmEstadoColaborador,
A.DsLoginN1,
C.NmGrupoCargo AS NmGrupoCargoColaborador,
C.CdColaboradorAtivo AS CdColaborador,
'PTS' as PTS,
A.VrPTS,

/*CASE WHEN (B.media_PTS) IN (., 0) THEN 50 ELSE (B.media_PTS) END as media_PTS,*/

/*FAZER O AJUSTE ABAIXO PARA ESTES PONTOS SEREM OS VALORES MÍNIMOS.*/
CASE
	WHEN (B.media_PTS) IN (., 0) and NmGrupoCargo = 'Gerente Relacionamento CN (Virtualizado)' THEN 495 /*90% DA PONTUAÇÃO TOTAL COM 100% DE ATINGIMENTO - MICHELLY*/
	WHEN (B.media_PTS) IN (., 0) and NmGrupoCargo = 'Gerente Relacionamento Motos' THEN 450 /*90% DA PONTUAÇÃO TOTAL COM 100% DE ATINGIMENTO - MICHELLY*/
	WHEN (B.media_PTS) IN (., 0) and NmGrupoCargo = 'Gerente Relacionamento Pesados' THEN 540 /*90% DA PONTUAÇÃO TOTAL COM 100% DE ATINGIMENTO - MICHELLY*/
ELSE (B.media_PTS) END as media_PTS,

CASE
/*	WHEN CALCULATED media_PTS = 50 THEN 50*/

/*TETO PONTUAÇÃO META DEFINIDO PELO ANDERSON - 02/09/2025 - GFM*/
	WHEN NmGrupoCargo = 'Gerente Relacionamento CN (Virtualizado)' and (B.media_PTS) NOT IN (., 0) and (CALCULATED media_PTS + 55) > 687 THEN 687
	WHEN NmGrupoCargo = 'Gerente Relacionamento Motos' and (B.media_PTS) NOT IN (., 0) and (CALCULATED media_PTS + 50 > 612) THEN 612 /*17/09/2025 - ANDERSON PASSOU ERRADO A PONTUAÇÃO E ALTERAMOS, INVERTENDO CANAL MOTOS E CANAL PESADOS*/
	WHEN NmGrupoCargo = 'Gerente Relacionamento Pesados' and (B.media_PTS) NOT IN (., 0) and (CALCULATED media_PTS + 60 > 687) THEN 687 /*17/09/2025 - ANDERSON PASSOU ERRADO A PONTUAÇÃO E ALTERAMOS, INVERTENDO CANAL MOTOS E CANAL PESADOS*/

	WHEN NmGrupoCargo = 'Gerente Relacionamento CN (Virtualizado)' and (B.media_PTS) NOT IN (., 0) THEN CALCULATED media_PTS + 55 /*META DE PONTOS PASSADA PELA MICHELLY*/
	WHEN NmGrupoCargo = 'Gerente Relacionamento Motos' and (B.media_PTS) NOT IN (., 0) THEN CALCULATED media_PTS + 50 /*META DE PONTOS PASSADA PELA MICHELLY*/
	WHEN NmGrupoCargo = 'Gerente Relacionamento Pesados' and (B.media_PTS) NOT IN (., 0) THEN CALCULATED media_PTS + 60 /*META DE PONTOS PASSADA PELA MICHELLY*/
ELSE CALCULATED media_PTS END AS PTS_Meta

FROM WORK.TbPts_Crescimento_2 A /*BASE ICM DO MÊS*/

LEFT JOIN CSBV.BASE_MEDIA_PTS_CN_CV B ON B.DSLOGINN1 = A.DSLOGINN1 /*BASE FIXA COM O CALCULA DA MÉDIA DO TRIMESTRE*/
LEFT JOIN WORK.ESTRUTURA_COMERCIAL_01 C ON C.DsLoginN1 = A.DsLoginN1 AND C.NUANOMES = &Base_Mes. /*ESTRUTURA COMERCIAL PARA TRAZER O CARGO*/

;RUN;

PROC SQL;
CREATE TABLE BASE_CALCULO_PTS_2 AS
SELECT

NuAnoMes,
A.DsLoginN1,
a.NmGrupoCargoColaborador,
a.CdColaborador,
A.PTS,
ROUND(A.VrPTS, 1) AS VrPTS,
ROUND(A.media_PTS, 1) AS media_PTS,

ROUND(A.PTS_Meta, 1) AS PTS_Meta,

CASE WHEN ROUND(VrPTS, 1) >= ROUND(PTS_Meta, 1) THEN 1 ELSE 0 END AS Fl_Ating_PTS,

CASE WHEN CALCULATED Fl_Ating_PTS = 1 THEN 20 ELSE 0 END AS VrPontosExtra

FROM WORK.BASE_CALCULO_PTS_1 A

WHERE a.NmGrupoCargoColaborador in ('Gerente Relacionamento CN (Virtualizado)', 'Gerente Relacionamento Motos', 'Gerente Relacionamento Pesados')

order by 3, 2

;RUN;
