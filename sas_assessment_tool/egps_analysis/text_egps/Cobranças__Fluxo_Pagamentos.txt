PROC SQL;
CREATE TABLE PAG_MKTPLACES AS
SELECT
	A.*,
	B.*
FROM Boletos_Gerados_SORT A
LEFT JOIN PAGAMENTOS_CL_TOTAL B
ON A.CdAcordo = B.CdAcordo
AND (A.NuParcela+1)=B.PARCELA_PAGA
WHERE VL_PAGO IS NOT MISSING
;QUIT;

DATA PAG_MKTPLACES;
SET PAG_MKTPLACES;
dataInclusao_date=datepart(DtInclusao);
FORMAT dataInclusao_date DDMMYYS10.;
RUN;
PROC SQL;
CREATE TABLE REMUNERACAO_MKTPLACES AS
SELECT
	CPF_CNPJ AS NuCpf,
	NuContrato AS numeroContrato,
	dataInclusao_date AS dataInclusao,
	PARCELA_PAGA AS numeroParcela,
	CdAcordo AS nossoNumero,
	ATR_PGT_ACORDO,
	DT_VENCIMENTO_PARC AS dataVencimentoBoleto,
	VrDocumento AS valorDocumento,
	DATA_PAGTO AS dataPagamento,
	CANAL_MACRO AS TpOrigemAcordo
FROM PAG_MKTPLACES
WHERE CANAL_MACRO IN ('QUERO_QUITAR', 'BLU365')
;QUIT;

PROC SQL;
CREATE TABLE COLCHAO_MKTPLACES AS
SELECT
	CPF_CNPJ AS NuCpf,
	NuContrato AS numeroContrato,
	dataInclusao_date AS dataInclusao,
	PARCELA_PAGA AS numeroParcela,
	CdAcordo AS nossoNumero,
	DT_VENCIMENTO_PARC AS dataVencimentoBoleto,
	VrDocumento AS valorDocumento,
	DATA_PAGTO AS dataPagamento,
	CANAL_MACRO AS TpOrigemAcordo
FROM PAG_MKTPLACES
WHERE CANAL_MACRO IN ('QUERO_QUITAR', 'BLU365', 'SERASA')
;QUIT;

DATA PAGAMENTOS_QUEROQUITAR;
	SET COLCHAO_MKTPLACES;

	FORMAT dataVencimentoBoleto DDMMYYS10.;
	FORMAT dataPagamento DDMMYYS10.;

	WHERE TpOrigemAcordo = 'QUERO_QUITAR';
RUN;
