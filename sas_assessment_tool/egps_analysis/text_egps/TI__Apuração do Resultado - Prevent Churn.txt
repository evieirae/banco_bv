libname FINAL "/riscosbvf/financas/varejo/dre_rateio_final";

libname CARTAO "/financas/cartoes";

proc sql;
    create table ponderadores as
    select
        safra,
        Grupo_Inatividade_Ajustada,
        sum(case when Fl_Weclever = 1 then 1 else 0 end) as qtde_teste,
        sum(case when Fl_Weclever = 0 then 1 else 0 end) as qtde_controle,
        case
            when calculated qtde_controle > 0 then calculated qtde_teste / calculated qtde_controle
            else .
        end as ponderador
    from input_resultado
    where Fl_Ativou_MOB3 in ("Ativou_CPC", "Ativou_Organico", "Nao_Ativou")
    group by safra, Grupo_Inatividade_Ajustada;
quit;

%macro churn(safra_ini, safra_fim_max);

    %local safra_fim carteira ano mes lista_safras i;

    %let safra_fim = &safra_ini.;
    %let lista_safras = ;

    %do %while(&safra_fim. <= &safra_fim_max.);

        %let carteira = &safra_fim.;

        

        data base_ini;
        set input_resultado;

            keep cddimcontacartao Grupo_Inatividade_Ajustada Fl_Weclever Safra
                 Cod_Conta Fl_Ativou_MOB3 MOB_Ativação Recorrente_Ativação;

            rename MOB_Ativação = mob_ativacao Recorrente_Ativação = Recorrente_Ativacao;

            where safra >= &safra_ini. and safra <= &safra_fim.
                  and Fl_Ativou_MOB3 in ("Ativou_CPC", "Ativou_Organico", "Nao_Ativou")
                  and MOB_Ativação = "MOB 0"
                  and Recorrente_Ativação = 1;
        run;

        

        proc sql;
            create table resumo_cartao as
            select CdContaCartao,
                   max(vrmargemliquida) as margem_liquida_cartoes
            from cartao.TBL_RAC_RESUMO_&carteira.
            group by CdContaCartao;
        quit;

        

        proc sql;
            create table base_ini as
            select a.*,
                   b.margem_liquida_cartoes
            from base_ini as a
            left join resumo_cartao as b
            on a.cddimcontacartao = b.CdContaCartao;
        quit;

        

        proc sql;
            create table agrup_&carteira. as
            select
                case Fl_Weclever
                    when 0 then 'Controle'
                    when 1 then 'Teste'
                    else 'Desconhecido'
                end as grupo,
                &carteira. as carteira,
                Grupo_Inatividade_Ajustada,
                safra,
                count(*) as qtde,
                sum(coalesce(margem_liquida_cartoes, 0)) as margem_liquida
            from base_ini
            group by grupo, carteira, Grupo_Inatividade_Ajustada, safra;
        quit;

        

        %let lista_safras = &lista_safras. &carteira.;

        

        %let ano = %substr(&safra_fim., 1, 4);
        %let mes = %substr(&safra_fim., 5, 2);

        %if &mes. = 12 %then %do;
            %let ano = %eval(&ano. + 1);
            %let mes = 01;
        %end;
        %else %do;
            %let mes = %eval(&mes. + 1);
        %end;

        %let safra_fim = %sysfunc(putn(&ano.*100 + &mes., z6.));

    %end;

    

    data agrup_final;
        set _null_;
    run;

    %do i = 1 %to %sysfunc(countw(&lista_safras.));
        %let safra = %scan(&lista_safras., &i.);
        data agrup_final;
            set agrup_final agrup_&safra.;
        run;
    %end;

    
    proc sql;
        create table agrup_final_ajustada as
        select
            a.*,
            b.ponderador,
            case
                when a.grupo = 'Controle' and b.ponderador > 0 then a.margem_liquida * b.ponderador
                else a.margem_liquida
            end as margem_ponderada
        from agrup_final as a
        left join ponderadores as b
        on a.safra = b.safra and a.Grupo_Inatividade_Ajustada = b.Grupo_Inatividade_Ajustada;
    quit;

    

    proc sql;
        create table agrup_final_ponderada as
        select
            safra,
            carteira,
            Grupo_Inatividade_Ajustada,
            grupo,
            sum(qtde) as qtde,
            sum(margem_liquida) as margem_liquida,
            sum(margem_ponderada) as margem_ponderada,
            cats('R$ ', put(sum(margem_ponderada), comma12.2)) as margem_ponderada_formatado
        from agrup_final_ajustada
        group by safra, carteira, Grupo_Inatividade_Ajustada, grupo;
    quit;

    
    proc sql;
        create table Apuracao_Churn as
        select
            grupo,
            carteira,
            safra,
            Grupo_Inatividade_Ajustada as faixa_etaria,
            margem_ponderada
        from agrup_final_ponderada;
    quit;

%mend;
