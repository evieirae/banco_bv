,'Canal Solar' as Dimensao1

,'Mês Fechado' as Semana
,Porte.Porte
,Porte.TipoParceiro
,Seg.QtCorridosDiaUtilMes
,Sum(Seg.VrBVMais) as VrBVMais
,Sum(Seg.VrCAP) as VrCAP
,Sum(Seg.VrPlacaSolar) as VrPlacaSolar
,Sum(Case When Seg.TpPessoa = 'F' Then Seg.VrProducao Else 0 End) as VrProducao_PF_BVMais
,Sum(Case When Seg.FlCapEleg = 1 Then Seg.VrProducao Else 0 End) as VrProducao_PF_AP_Placa
,(Sum(Seg.VrBVMais)/Sum(Case When Seg.TpPessoa = 'F' Then Seg.VrProducao Else 0 End)) as Penet_BvMais
,(Sum(Seg.VrCAP)/Sum(Case When Seg.FlCapEleg = 1 Then Seg.VrProducao Else 0 End)) As Penet_AP
,Max(Seg.DtAtualizacao) as DataBase Format DDMMYY10.

From tmp.TbProducao as Seg
Left Join Base_Porte_Cobans_CNPJ as Porte On Seg.NuCnpjLojaFavorecido = Porte.NuCnpjLojaFavorecido and Seg.NuAnomes = Porte.NuAnomes
Where Seg.NuANomes >= &Base_m5. and Seg.NmGrupoProdutometa = 'Solar' and NmCanalEstruturaComercial = 'Canal Solar' and NmColaborador NE 'OPERADOR ESPECIAL CANAL DIGITAL' and
		Seg.NmAreaEstruturaComercial in
		(
			'Canal Solar CO e NO',
			'Canal Solar MG, BA, AL e SE',
			'Canal Solar NE',
			'Canal Solar SP Interior',
			'Canal Solar SP, RJ e ES',

		)

;Quit;
