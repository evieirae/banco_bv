PROC SQL;
   CREATE TABLE WORK.QUERY_FOR_TBEXTCONTRATO AS
   SELECT t1.NuAnoMes,
          t2.NmGrupoProdutoMeta,
          t1.CdGrupoSubProduto,
          t1.FlZeroKm,
          
            (COUNT(t1.NuContrato)) AS COUNT_of_NuContrato,
          
            (SUM(t1.VrContrato)) FORMAT=20.2 AS SUM_of_VrContrato,
          
            (SUM(t1.VrEntrada)) FORMAT=20.2 AS SUM_of_VrEntrada,
          
            (SUM(t1.VrProducao)) FORMAT=21.2 AS SUM_of_VrProducao
      FROM BASES.TBEXTCONTRATO t1
           INNER JOIN WORK.QUERY_FOR_TBGRUPOPRODUTOMETA t2 ON (t1.CdGrupoSubProduto = t2.CdGrupoSubProduto)

           (
           'Leves',
		   'Leves-BVO',
           'Motos',
           'Pesados',
		   'CVG Leves',

           ) AND StProducao = 'S'
      GROUP BY t1.NuAnoMes,
               t2.NmGrupoProdutoMeta,
               t1.CdGrupoSubProduto,
               t1.FlZeroKm;
QUIT;
