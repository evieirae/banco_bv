    create table resumo_safra_mes as
    select SafraAjuizamento,

           sum(saldo_num) as saldo_total,

%macro resumo_por(var);
    proc sql;
        create table resumo_&var as

               &var,

               sum(saldo_num) as saldo_total,
               sum(case when pago_flag = 1 then saldo_num else 0 end) as saldo_pago,
               calculated saldo_pago / calculated saldo_total as taxa_pagamento format=percent8.2

        group by SafraAjuizamento, &var
