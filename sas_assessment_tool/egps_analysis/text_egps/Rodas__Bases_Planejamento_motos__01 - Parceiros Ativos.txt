PROC SQL;
CREATE TABLE WORK.PARCEIROSATIVOS_M_SIM AS
SELECT

A.CdParceiroComercial,
A.NuCnpjLoja,
A.DsAtividadeFinanciamento,
A.NmFilialC,
A.NmTipoParceiroComercial,
A.NmAutoShopping,
A.NmGrupoConcessionaria,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN A.NmGrupoConcessionaria
	WHEN A.NmAutoShopping <> '' THEN A.NmAutoShopping
	WHEN A.NmTipoParceiroComercial <> '' THEN A.NmTipoParceiroComercial
	ELSE 'Outros'
END) AS NmClassificacaoGrupoParceiro,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN 'GRANDES GRUPOS'
	WHEN A.NmAutoShopping <> '' THEN 'AUTO SHOPPING'
	WHEN A.NmTipoParceiroComercial <> '' THEN UPCASE(A.NmTipoParceiroComercial)
	ELSE 'OUTROS'
END) AS NmClassificacaoTipoGrupoParceiro,

A.NmParceiroComercial,
A.CdUf,
A.CdColaboradorC,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.DtCadastro,
A.FlAtivo
FROM BASES.TBPARCEIROCOMERCIAL A

WHERE
A.FlAtivo = 'S'
AND A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.DsAtividadeFinanciamento = 'VEÍCULO'

AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV BALCAO%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'CA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJAS RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV MESA%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV REMANESCENTE%'
/*AND YEAR(DATEPART(DTCADASTRO))*100 + MONTH(DATEPART(DTCADASTRO)) < &SAFRAM0.*/

ORDER BY 1

;RUN;

PROC SQL;
CREATE TABLE WORK.PARCEIROSATIVOS_M_NAO AS
SELECT

A.CdParceiroComercial,
A.NuCnpjLoja,
A.DsAtividadeFinanciamento,
A.NmFilialC,
A.NmTipoParceiroComercial,
A.NmAutoShopping,
A.NmGrupoConcessionaria,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN A.NmGrupoConcessionaria
	WHEN A.NmAutoShopping <> '' THEN A.NmAutoShopping
	WHEN A.NmTipoParceiroComercial <> '' THEN A.NmTipoParceiroComercial
	ELSE 'Outros'
END) AS NmClassificacaoGrupoParceiro,

PROPCASE(CASE
	WHEN A.NmGrupoConcessionaria <> '' THEN 'GRANDES GRUPOS'
	WHEN A.NmAutoShopping <> '' THEN 'AUTO SHOPPING'
	WHEN A.NmTipoParceiroComercial <> '' THEN UPCASE(A.NmTipoParceiroComercial)
	ELSE 'OUTROS'
END) AS NmClassificacaoTipoGrupoParceiro,

A.NmParceiroComercial,
A.CdUf,
A.CdColaboradorC,
A.CdFilialC,
A.NmAreaEstruturaComercialC,
A.NmSubEstruturaComercialC,
A.NmCanalEstruturaComercialC,
A.DtCadastro,
A.FlAtivo,

datepart(A.DtUltimaAtualizacao) as DtUltimaAtualizacao format date9.,
(YEAR(CALCULATED DtUltimaAtualizacao) * 100) + MONTH(CALCULATED DtUltimaAtualizacao) as NuAnomes_DtUltAtu,
&SAFRAM2. as SAFRA_M2,
&SAFRAM0. as SAFRA_M0,

/*PDVS COM DATA DE INATIVAÇÃO NOS ULTIMOS 90 DIAS ENTRAM PARA O CALCULO DE PONTECIAL - ANDERSON - 28/07/2025*/
CASE
	WHEN CALCULATED NuAnomes_DtUltAtu BETWEEN &SAFRAM2. AND &SAFRAM0.
THEN 'MANTER' ELSE 'EXCLUIR' END AS FlPDVs /**/

FROM BASES.TBPARCEIROCOMERCIAL A

WHERE
A.FlAtivo = 'N'
AND A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.DsAtividadeFinanciamento = 'VEÍCULO'

AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV BALCAO%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'CA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJA RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'LOJAS RENEG%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV MESA%'
AND UPCASE(NMPARCEIROCOMERCIAL) NOT LIKE 'BV REMANESCENTE%'
/*AND YEAR(DATEPART(DTCADASTRO))*100 + MONTH(DATEPART(DTCADASTRO)) < &SAFRAM0.*/

ORDER BY 1

;RUN;

DATA PARCEIROSATIVOS_M_NAO_2;
SET WORK.PARCEIROSATIVOS_M_NAO;
WHERE FlPDVs = 'MANTER';
drop DtUltimaAtualizacao NuAnomes_DtUltAtu safra_m2 safra_m0 FlPDVs;
;RUN;

DATA WORK.PARCEIROSATIVOS_M;
SET WORK.PARCEIROSATIVOS_M_SIM
	WORK.PARCEIROSATIVOS_M_NAO_2;
;RUN;
