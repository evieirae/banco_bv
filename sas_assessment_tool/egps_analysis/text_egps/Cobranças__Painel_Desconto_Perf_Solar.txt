PROC SQL;
CREATE TABLE BASE_ALCADA3 AS
SELECT DISTINCT
A.*
,CASE WHEN B.acmaxdisc1 IS NOT MISSING THEN B.acmaxdisc1/100 ELSE 0 END AS DESCONTO_PERC_ELEITO
,ROUND(((TotalDevido-TotalBaixado)/TotalDevido), 0.01) AS DESCONTO_PERC_REAL
,CALCULATED DESCONTO_PERC_ELEITO * TotalDevido AS DESCONTO_ELEITO

,CASE WHEN A.TIPO_PAGAMENTO IS NOT NULL
	  	   AND (A.Atraso_Pagamento BETWEEN B.QTDIAINICIOFAIXAATRASO AND B.QTDIAFIMFAIXAATRASO)
	  THEN CALCULATED DESCONTO_PERC_ELEITO * TotalDevido
	  ELSE 0 END AS DESCONTO_ELEITO

FROM BASE_ALCADA2 A
LEFT JOIN BASE_CYBER B ON A.LETRA = B.AQQUE
						  AND (A.Atraso_Pagamento BETWEEN B.QTDIAINICIOFAIXAATRASO AND B.QTDIAFIMFAIXAATRASO)
						  AND A.TIPO_PAGAMENTO=B.TPACORDO
;QUIT;
