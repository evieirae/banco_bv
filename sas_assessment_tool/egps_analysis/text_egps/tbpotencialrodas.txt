proc sql;
%CONECTADB(server = BI);
create table TbExtCetipAnalitco as
select * from connection to dbcon
(
  WITH
	 TbDataUltimoArquivo AS (
		SELECT MAX(substr(NmArquivo,26,8)) AS DtUltimoArquivo
		FROM DWBI_DTWH.TbAnalitico

	 )

	,TbCetipAnaliticoMesAtual AS (
		SELECT
			year(CAST(substr(Analitico.NmArquivo,26,8) AS DATE))*100+month(CAST(substr(Analitico.NmArquivo,26,8) AS DATE)) AS NuAnoMes
			,Analitico.NuCnpjLoja
			,Analitico.DsCategoriaVeiculo
			,Analitico.DsSegmento
			,Analitico.QtFinanciamentos
			,Analitico.QtVolumeFinanciamentos

		FROM DWBI_DTWH.TbAnalitico Analitico
			INNER JOIN TbDataUltimoArquivo DataUltimoArquivo
				ON substr(Analitico.NmArquivo,26,8) = DataUltimoArquivo.DtUltimoArquivo
				AND substr(DataUltimoArquivo.DtUltimoArquivo,7,2) <> '01'

		WHERE Analitico.CdConsorcio = 'N'
			  AND Analitico.CdTipoRegistro = 'D'
			  AND Analitico.DsCategoriaVeiculo in
					('AUTOMOVEL'
					,'UTILITARIO'
					,'CAMINHONETE'
					,'CAMIONETA'
					,'MICROONIBUS'
					,'CICLOMOTOR'
					,'MOTONETA'
					,'MOTOCICLETA'
					,'TRICICLO'
					,'QUADRICICLO'
					,'REBOQUE'
					,'SEMI-REBOQUE'
					,'CAMINHAO'
					,'CAMINHAO TRATOR'
					,'ONIBUS'
					,'MOTOR-CASA'
					,'CHASSI/PLATAFORMA'
					,'TRATOR DE RODAS'
				)
		)

		,TbCetipAnaliticoMesFechado AS (
			SELECT
				year(CAST(substr(Analitico.NmArquivo,26,8) AS DATE)-1)*100+month(CAST(substr(Analitico.NmArquivo,26,8) AS DATE)-1) AS NuAnoMes
				,Analitico.NuCnpjLoja
				,Analitico.DsCategoriaVeiculo
				,Analitico.DsSegmento
				,Analitico.QtFinanciamentos
				,Analitico.QtVolumeFinanciamentos

			FROM DWBI_DTWH.TbAnalitico Analitico
			WHERE substr(Analitico.NmArquivo,32,2) = '01'
			  AND Analitico.CdConsorcio = 'N'
			  AND Analitico.CdTipoRegistro = 'D'
			  AND Analitico.DsCategoriaVeiculo in
					('AUTOMOVEL'
					,'UTILITARIO'
					,'CAMINHONETE'
					,'CAMIONETA'
					,'MICROONIBUS'
					,'CICLOMOTOR'
					,'MOTONETA'
					,'MOTOCICLETA'
					,'TRICICLO'
					,'QUADRICICLO'
					,'REBOQUE'
					,'SEMI-REBOQUE'
					,'CAMINHAO'
					,'CAMINHAO TRATOR'
					,'ONIBUS'
					,'MOTOR-CASA'
					,'CHASSI/PLATAFORMA'
					,'TRATOR DE RODAS')
		)

		,TbCetipAnaliticoFinal AS (
			SELECT * FROM TbCetipAnaliticoMesAtual UNION ALL
			SELECT * FROM TbCetipAnaliticoMesFechado
		)

	SELECT
		 NuAnoMes
		,NuCnpjLoja
		,DsCategoriaVeiculo
		,DsSegmento
		,SUM(QtFinanciamentos) AS QtFinanciamentos
		,SUM(QtVolumeFinanciamentos) AS VrFinanciamentos

	FROM TbCetipAnaliticoFinal
	WHERE NuAnoMes >= 201901
	GROUP BY
		 NuAnoMes
		,NuCnpjLoja
		,DsCategoriaVeiculo
		,DsSegmento
	)
;run;

PROC SQL;
CREATE TABLE TbCetipMercado AS
SELECT
	 CetipAnalitico.NuAnoMes
	,CASE
		WHEN BVO.CdParceiroComercial <> . and CetipAnalitico.DsSegmento = 	'NOVO' THEN 'Leves-BVO'
		WHEN CetipAnalitico.DsCategoriaVeiculo in ('AUTOMOVEL','UTILITARIO','CAMINHONETE','CAMIONETA','MICROONIBUS') 	THEN 'Leves'
		WHEN CetipAnalitico.DsCategoriaVeiculo in ('CICLOMOTOR','MOTONETA','MOTOCICLETA','TRICICLO','QUADRICICLO') 		THEN 'Motos'
		WHEN CetipAnalitico.DsCategoriaVeiculo in ('REBOQUE','SEMI-REBOQUE','CAMINHAO','CAMINHAO TRATOR','ONIBUS','MOTOR-CASA','CHASSI/PLATAFORMA','TRATOR DE RODAS') 		THEN 'Pesados'
	 END AS NmGrupoProdutoMeta
	,0 as FlRefin
	,CASE
		WHEN CetipAnalitico.DsSegmento = 	'NOVO'    		THEN 'ZeroKM'
		WHEN CetipAnalitico.DsSegmento =  '> 15'    		THEN 'Acima de 10 anos'
		WHEN CetipAnalitico.DsSegmento =  '11 a 15' 		THEN 'Acima de 10 anos'
		WHEN CetipAnalitico.DsSegmento =  '8 a 10'  		THEN '8 a 10 anos'
		WHEN CetipAnalitico.DsSegmento =  '5 a 7'  			THEN '3 a 7 anos'
		WHEN CetipAnalitico.DsSegmento =  '3 a 4'   		THEN '3 a 7 anos'
		WHEN CetipAnalitico.DsSegmento =  'ate 2 anos' 		THEN '0 a 2 anos'
	 END AS FxIdade
	,put(CetipAnalitico.NuCnpjLoja,z14.) AS NuCnpjLoja

	,PotencialPDVs.CdParceiroComercial
	,PotencialPDVs.FlAtivo
	,case when CetipAnalitico.DsCategoriaVeiculo in ('AUTOMOVEL','UTILITARIO','CAMINHONETE','CAMIONETA','MICROONIBUS') then PotencialPDVs.FlPorteA end as FlPorteA
	,case when CetipAnalitico.DsCategoriaVeiculo in ('CICLOMOTOR','MOTONETA','MOTOCICLETA','TRICICLO','QUADRICICLO') then PotencialPDVs.FlPorteMotos end as FlPorteMotos
	,case when CetipAnalitico.DsCategoriaVeiculo in ('AUTOMOVEL','UTILITARIO','CAMINHONETE','CAMIONETA','MICROONIBUS') then PotencialPDVs.DsClassificacaoA end as DsClassificacaoA

	,CetipAnalitico.DsCategoriaVeiculo

	,PotencialPDVs.NmLoja
	,PotencialPDVs.NmTipoParceiroComercial
	,PotencialPDVs.FlIntegracaoParceiroFeI
	,PotencialPDVs.NmLogradouro

(
WITH
	 TbDataUltimoArquivo AS (
		SELECT MAX(substr(NmArquivo,13,8)) AS DtUltimoArquivo
		FROM DWBI_DTWH.TbRanking
		where NmArquivo not in('00002350432..20250519001739.txt','C0004.050568.PRK.20250201162232.txt.txt','00002350432.PRK.20250224002219.txt')

	 )

	,TbCetipRankingMesAtual AS (
		SELECT
			year(CAST(substr(Ranking.NmArquivo,13,8) AS DATE))*100+month(CAST(substr(Ranking.NmArquivo,13,8) AS DATE)) AS NuAnoMes
			,Ranking.NuCnpjLoja

			,Ranking.NuRankingCliente

			INNER JOIN TbDataUltimoArquivo DataUltimoArquivo
				ON substr(Ranking.NmArquivo,13,8) = DataUltimoArquivo.DtUltimoArquivo
				AND substr(DataUltimoArquivo.DtUltimoArquivo,7,2) <> '01'

		AND  Ranking.NmArquivo not in('00002350432..20250519001739.txt','C0004.050568.PRK.20250201162232.txt.txt','00002350432.PRK.20250224002219.txt')

		SELECT
			year(CAST(substr(Ranking.NmArquivo,13,8) AS DATE)-1)*100+month(CAST(substr(Ranking.NmArquivo,13,8) AS DATE)-1) AS NuAnoMes

PROC SQL;
CREATE TABLE TbCnpjRaiz01 AS
  SELECT
	SUBSTR(NuCnpjLoja,1,8) AS RaizCnpjLoja,
	NuCnpjLoja,
	NmLoja,
    SUM(QtContrato) AS QtdContratos,
    SUM(QtFinanciamentos) AS Mercado
  FROM tmp.TbCetipConsolidado
  GROUP BY 1,2,3
  ORDER BY
    calculated QtdContratos DESC,
    calculated Mercado DESC;
RUN;

PROC SQL;
CREATE TABLE GruposCnpj AS
  SELECT
    RaizCnpjLoja,
	COUNT(DISTINCT NuCnpjLoja) AS QtdLojas
  FROM TbCnpjRaiz01
  GROUP BY 1;
RUN;

PROC SORT DATA=TbCnpjRaiz01 OUT=TbCnpjRaiz02 NODUPKEY;
BY RaizCnpjLoja;
RUN;

PROC SQL;
CREATE TABLE TbCnpjRaiz AS
  SELECT
    A.RaizCnpjLoja,
	STRIP(CASE WHEN B.QtdLojas > 1 THEN 'GRUPO ' END || A.NmLoja) AS NmGrupo,
	CASE WHEN B.QtdLojas <= 1 THEN '0. Individual'
	     WHEN B.QtdLojas BETWEEN 2 AND 5 THEN '1. De 2 a 5 CNPJs'
		 WHEN B.QtdLojas BETWEEN 6 AND 10 THEN '2. De 6 a 10 CNPJs'
		 WHEN B.QtdLojas > 10 THEN '3. Mais de 10 lojas'
	     END AS QtdLojas
  FROM
    TbCnpjRaiz02 A
  LEFT JOIN
    GruposCnpj B
	ON A.RaizCnpjLoja = B.RaizCnpjLoja;
RUN;

PROC SQL;
CREATE TABLE BASE_PDV_DEDICADO AS
SELECT

a.NuCnpjLoja,a.FlAtivo,

sum(CASE WHEN A.NmGrupoProduto = 'Leves' then A.QtGravamesMercado ELSE 0 END) AS QtGravames_Leves format commax20.2,
sum(CASE WHEN A.NmGrupoProduto = 'Motos' then A.QtGravamesMercado ELSE 0 END) AS QtGravames_Motos format commax20.2,
sum(CASE WHEN A.NmGrupoProduto = 'Pesados' then A.QtGravamesMercado ELSE 0 END) AS QtGravames_Pesados format commax20.2,

sum(QtGravamesMercado) as QtGravames_Total format commax20.2,

calculated QtGravames_Leves / calculated QtGravames_Total as Perc_Leves format percent9.2,
calculated QtGravames_Motos / calculated QtGravames_Total as Perc_Motos format percent9.2,
calculated QtGravames_Pesados / calculated QtGravames_Total as Perc_Pesados format percent9.2,

CASE WHEN FlAtivo = 'S' THEN CASE WHEN CALCULATED Perc_Motos >= 0.8 THEN 1 ELSE 0 END /*PDVs ativos*/
	 WHEN FlAtivo <> 'S' THEN CASE WHEN CALCULATED Perc_Motos >= 0.5 THEN 1 ELSE 0 END /*PDVs inativos e nao cadastrados*/
	 END AS PDV_MOTOS,

CASE WHEN FlAtivo = 'S' THEN CASE WHEN CALCULATED Perc_Pesados >= 0.7 THEN 1 ELSE 0 END /*PDVs ativos*/
	 WHEN FlAtivo <> 'S' THEN CASE WHEN CALCULATED Perc_Pesados >= 0.5 THEN 1 ELSE 0 END /*PDVs inativos e nao cadastrados*/
	 END AS PDV_PESADOS

FROM TMP.TBPOTENCIALRODAS A

WHERE NUANOMES BETWEEN &Base_m6. AND &Base_m0.
/*and NuCnpjLoja = '39611478000193'*/
GROUP BY a.NuCnpjLoja,a.FlAtivo
order by 2 desc
;RUN;

by NuCnpjLoja
