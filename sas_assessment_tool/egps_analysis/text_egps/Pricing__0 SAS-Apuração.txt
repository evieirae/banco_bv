PROC SQL;
CREATE TABLE AuxShare AS
SELECT
NuAnoMes,
CdParceiroComercial,
SUM(QtContrato) as QtContrato,
SUM(QtGravamesMercado) AS QtGravamesMercado
FROM PROJETOS.TBPOTENCIALRODAS AS A
WHERE NuAnoMes >= &MesAnteriorApuração AND NuAnoMes <= &DataApuracao AND NmGrupoProduto = &Produto
AND A.FlAtivo = 'S' AND FlAtualizacaoMercado  = 1 AND A.NmCanalEstruturaComercialC = 'Canal Veiculos'
AND A.NmSubEstruturaComercialC in ('Canal Rodas e EGV - Regional I', 'Canal Rodas e EGV - Regional II',
'Canal Rodas e EGV - Regional I', 'Canal Rodas e EGV - Regional II',
'Canal Rodas e EGV - Regional I', 'Canal Rodas e EGV - Regional II',
'Canal Motos','Canal Pesados',
'Canal Rodas e EGV - Regional III', 'Canal Rodas e EGV - Regional IV',
'Canal Rodas e EGV - Regional V', 'Canal Rodas e EGV - Regional VI')
GROUP BY 1,2;
QUIT;
PROC SQL;
CREATE TABLE AuxShareTranposta AS
SELECT
CdParceiroComercial,
SUM (CASE WHEN NuAnoMes = &MesAnteriorApuração THEN QtContrato ELSE 0 END) AS QtGravamesBV_&MesAnteriorApuração,
SUM (CASE WHEN NuAnoMes = &DataApuracao THEN QtContrato ELSE 0 END) AS QtGravamesBV_&DataApuracao,
SUM (CASE WHEN NuAnoMes = &MesAnteriorApuração THEN QtGravamesMercado ELSE 0 END) AS QtGravamesMercado_&MesAnteriorApuração,
SUM (CASE WHEN NuAnoMes = &DataApuracao THEN QtGravamesMercado ELSE 0 END) AS QtGravamesMercado_&DataApuracao
FROM AuxShare AS A
GROUP BY 1;RUN;

PROC SQL;
CREATE TABLE AuxProd AS
SELECT
CdParceiroComercial,
Input (cats(year(DtContrato),put(month(DtContrato),z2.)),8.) as Safra,
SUM(VrProducao) AS VrProducao,
COUNT (NuProposta) AS QntdContratos,
SUM(VrMargemContribuicao) AS Margem,
SUM(VpSpreadContabil) AS VpSpreadContabil,
SUM(VrComissaoTotal) AS VrComissaoTotal,
SUM(VrBVMais) AS VrSPF,
SUM(VrCap) AS VrAP,
SUM(VrCompleto) AS VrCompleto,
(SUM(VrGap)+SUM(VrGarantiaMecanica)+SUM(VrRCF)+SUM(VrCasco)) AS VrDemais,
SUM(FlBVMais) AS FlBVMais,
SUM(FlBVMaisEleg) AS FlBVMaisEleg,
SUM(FlCap) AS FlCap,
SUM(FlCapEleg) AS FlCapEleg,
SUM(FlCompleto) AS FlCompleto,
SUM(FlCompletoEleg) AS FlCompletoEleg,
SUM(FlDemaissemCompleto) AS FlDemaissemCompleto,
SUM(FlDemaisEleg) AS FlDemaisEleg

FROM PROJETOS.TBPRODUCAO AS A
WHERE DtContrato >= &Datade AND DtContrato <= &Dataate AND NmGrupoProdutoMeta = &Produto
GROUP BY 1,2;RUN;

SUM (CASE WHEN Safra = &MesAnteriorApuração THEN VpSpreadContabil ELSE 0 END) AS SpreadPnd_&MesAnteriorApuração,
SUM (CASE WHEN Safra = &DataApuracao THEN VpSpreadContabil ELSE 0 END) AS SpreadPnd_&DataApuracao,
SUM (CASE WHEN Safra = &DataApuracao THEN VrComissaoTotal ELSE 0 END) AS ComissaoRetornoPlus_&DataApuracao,
SUM (CASE WHEN Safra = &DataApuracao THEN VrSPF ELSE 0 END) AS VrSPF_&DataApuracao,
SUM (CASE WHEN Safra = &DataApuracao THEN VrAP ELSE 0 END) AS VrAP_&DataApuracao,
SUM (CASE WHEN Safra = &DataApuracao THEN VrCompleto ELSE 0 END) AS VrCompleto_&DataApuracao,
SUM (CASE WHEN Safra = &DataApuracao THEN VrDemais ELSE 0 END) AS VrDemais_&DataApuracao,

PROC SQL;
CREATE TABLE BaseParceiros AS
SELECT
A.CdParceiroComercial,
CASE WHEN B.CdParceiroComercial IS NOT NULL THEN B.CdParceiroComercial ELSE 0 END AS CdParceiroBVCompensa,
B.TipoCampanha,

;RUN;

SUM (B.FlBVMais_&DataApuracao) AS FlBVMais_&DataApuracao,
