PROC SQL;
CREATE TABLE BASE_CYBER AS
SELECT
	CY.SAFRA,
	CY.DT_ARQUIVO,
	CY.DTENVIO AS DTENVIO_CYBER,
	CY.DTENVIO_REAL,
	CY.DT_NOVA_DISPONIB,
	CY.DT_NOVA_DISPONIB_REAL,
	MA.DTENVIO_REAL AS DTENVIO_REAL_MA,
	CY.TPCLASSE,
	CY.CHAVE,
	CY.NUCONTA,
	CY.PRODUTO,
	BA.NUCPFCNPJ,
	CY.FAIXA_CS_ATU,
	CY.DIG_SCORE,
	CY.DIG_RAND,
	CY.PILOTO_DIGITAL,
	CY.CONTA_DIGITAL,
	CY.COD_ASSESS,
	CY.ASSESSORIA,
	MA.ASSESSORIA AS ASSESSORIA_MA,
	CASE
		WHEN MA.DIAS_AT = 0 THEN MA.VENCIMENTO
		ELSE MA.DATA_INICIO_ATRASO
	END AS DT_INICIO_ATRASO FORMAT DDMMYY10.,
	CY.DIAS_AT,
	CY.ATRASO_ATU,
	CY.ATRASO_INI_MES,
	BA.DIAS_AT 		AS DIAS_AT_BA,
	BA.DIAS_AT_ANT 	AS DIAS_AT_ANT_BA,
	MA.DIAS_AT 		AS DIAS_AT_MA,
	MA.DIAS_AT_ANT	AS DIAS_AT_ANT_MA,
	CY.SALDO_DEVEDOR_VENC,
	CY.DMCURBAL,
	CY.VL_NOVA_DIVIDA,
	MA.VALOR AS VALOR_MA,
	BA.VALOR AS VALOR_BA,
	DA.VALOR AS VALOR_BA_ANT,
	CASE
		WHEN CY.TPCLASSE = 'F'  THEN CY.SALDO_DEVEDOR_VENC
		WHEN MA.VALOR <= 0 		THEN CY.DMCURBAL
		ELSE MA.VALOR
	END AS VLR_COBRANCA,
	CASE
		WHEN ENQ.DATA_ENQUAD <= &DT_ARQ.  THEN ENQ.DATA_ENQUAD
	END AS DATA_ENQUAD FORMAT DDMMYY10.,
	CASE
		WHEN ENQ.DATA_ENQUAD = . 			THEN .
		WHEN ENQ.DATA_ENQUAD <= &DT_ARQ.  THEN VALOR_BA_ANT
		ELSE .
	END AS VLR_ENQUADRAMENTO,
	
	CASE
		WHEN CY.U3SCORE4 >= 2000 AND CY.U3SCORE4 <= 2998 AND CY.COD_ASSESS IN (4001,4006) AND CY.SAFRA IN ('01JUL2023'D,'01AUG2023'D) THEN 'S' 
		WHEN CY.U3SCORE4 >= 2000 AND CY.U3SCORE4 <= 2998 AND CY.COD_ASSESS IN (4001) AND CY.SAFRA IN ('01SEP2023'D,'01OCT2023'D) THEN 'S' 
		WHEN CY.U3SCORE4 IN (2003,2030) AND CY.COD_ASSESS IN (4001) AND CY.SAFRA IN ('01NOV2023'D) THEN 'S' 
		WHEN CY.U3SCORE4 IN (2003,2030) AND CY.COD_ASSESS IN (4001,4006) AND CY.SAFRA >= '01DEC2023'D THEN 'S' 
		WHEN CY.U3SCORE4 IN (3003) AND CY.COD_ASSESS IN (4001,4006) AND CY.SAFRA >= '01DEC2023'D THEN 'C' 
		ELSE 'N'
	END AS PILOTO_MCKINSEY

FROM BASE_CYBER_&DATA_1. AS CY


LEFT JOIN MOVTO.BASILEIA_&DATA_1. AS BA

ON
CY.NUCONTA = BA.NUCONTA
AND
CY.PRODUTO = BA.PRODUTO


LEFT JOIN MOVTO.BASILEIA_&DATA_2. AS DA

ON
CY.NUCONTA = DA.NUCONTA
AND
CY.PRODUTO = DA.PRODUTO


LEFT JOIN MOVTO.BASILEIA_&DATA_3. AS MA
ON
CY.NUCONTA = MA.NUCONTA
AND
CY.PRODUTO = MA.PRODUTO


LEFT JOIN MOVTO.BASE_HIST_ENQUAD AS ENQ
ON
CY.NUCONTA = ENQ.NUCONTA
AND
CY.PRODUTO = ENQ.PRODUTO

;RUN;

	CY.DIAS_AT_ANT_MA,
	CASE
		WHEN CY.TPCLASSE = 'F'		THEN 'FLUXO'
		WHEN CY.DIAS_AT_MA <= 30	THEN 'ATE 30 DIAS DE ATRASO'
		WHEN CY.DIAS_AT_MA > 30		THEN 'ACIMA DE 30 DIAS DE ATRASO'
	END AS FX_ATRASO_2,
	CASE
		WHEN CY.TPCLASSE = 'F'		THEN 'FLUXO'
		WHEN CY.DIAS_AT_MA <= 30	THEN 'ATE 30 DIAS DE ATRASO'
		WHEN CY.DIAS_AT_MA <= 60	THEN 'DE 31 A 60 DIAS DE ATRASO'
		WHEN CY.DIAS_AT_MA > 60		THEN 'ACIMA DE 60 DIAS DE ATRASO'
	END AS FX_ATRASO_NEW,

	CY.DMCURBAL,
	CY.VL_NOVA_DIVIDA,
	CY.VALOR_MA,
	CY.VALOR_BA,
	CY.VALOR_BA_ANT,
	CY.VLR_COBRANCA,
	CY.DATA_ENQUAD,
	CY.VLR_ENQUADRAMENTO,
	CASE
		WHEN CY.ATRASO_INI_MES >= 96																						THEN 'RETIRAR'  
		WHEN CY.TPCLASSE = 'E'	AND CY.DIAS_AT_MA IN(0,.)	AND CY.DIAS_AT_ANT_MA IN (0,.)									THEN 'RETIRAR'  
		WHEN CY.TPCLASSE = 'E'	AND CY.DIAS_AT_MA = 0 	AND CY.DIAS_AT_ANT_MA >= 0											THEN 'RETIRAR'

		WHEN CY.DATA_ENQUAD IS NOT MISSING AND  CY.DATA_ENQUAD <= CY.DTENVIO_REAL	THEN 'RETIRAR'  

		WHEN CY.DTENVIO_REAL = CY.DT_NOVA_DISPONIB_REAL THEN 'RETIRAR'

ELSE 'EM_COBRANCA'
	END AS TPLANCAMENTO,
	CASE
		WHEN CY.ATRASO_INI_MES >= 76																						THEN 'ATRASO_ACIMA_DE_75_DIAS'  
		WHEN CY.TPCLASSE = 'E'	AND CY.DIAS_AT_MA IN(0,.)	AND CY.DIAS_AT_ANT_MA IN (0,.)									THEN 'ESTOQUE_SEM_ATRASO'  
		WHEN CY.TPCLASSE = 'E'	AND CY.DIAS_AT_MA = 0 	AND CY.DIAS_AT_ANT_MA >= 0											THEN 'ESTOQUE_PAGO_NO_ULTIMO_DIA'  

		WHEN CY.TPCLASSE = 'F'	AND CY.DIAS_AT >= 75																		THEN 'ENVIO_ACIMA_DE_75_DIAS'  

		WHEN CY.TPCLASSE = 'F' AND CY.DT_ARQUIVO = CY.DTENVIO_REAL AND CY.DIAS_AT_BA IN(0,.) AND CY.DIAS_AT_ANT_BA IN (0,.) THEN 'ENVIO_PAGO'	
		WHEN CY.TPCLASSE = 'F' AND CY.DT_ARQUIVO = CY.DTENVIO_REAL AND CY.DIAS_AT_BA = 0 AND CY.DIAS_AT_ANT_BA >= 0 		THEN 'ENVIO_PAGO'	
		WHEN CY.DATA_ENQUAD IS NOT MISSING AND CY.DATA_ENQUAD <= CY.DTENVIO_REAL	THEN 'ENVIO_DEPOIS_DO_ENQUADRAMENTO'  
		WHEN CY.DTENVIO_REAL = CY.DT_NOVA_DISPONIB_REAL THEN 'ENVIO E RETIRADA DA ASSESSORIA NO MESMO DIA '
		END AS MOTIVO_RETIRADA,
	CY.PILOTO_MCKINSEY

FROM BASE_CYBER AS CY
LEFT JOIN MOVTO.PERF_AMIGAVEL AS B
ON CY.NUCONTA = B.NUCONTA
AND CY.PRODUTO = B.PRODUTO
AND CY.DT_INICIO_ATRASO = B.DT_INICIO_ATRASO
AND B.PILOTO_DIGITAL = 'S'
AND CY.COD_ASSESS NOT IN (4010)

QUIT;
