%macro Importar;
	%let id = %sysfunc(open(Arquivos));
		%syscall set(id);
		%do %while(%sysfunc(fetch(&id.)) eq 0);
			/* Identifica o delimitador do arquivo */
			data _null_;
				infile "&Diretorio." dsd missover obs=1;
				input Cabecalho	$255.;
				Cabecalho = translate(Cabecalho, '_', ' ');
				call symputx('Delimitador', put(substr(Cabecalho, prxmatch('/[^\w]/',Cabecalho), 1), $hex.));
			run;

			/* Importa o arquivo em uma base temporária */
			data tmp;
				infile "&Diretorio." dsd missover firstobs=2 dlm="&Delimitador."x;
				format  NmArquivo $100.
						DtInicioFeirao date9.
						DtFimFeirao	date9.;
				NmArquivo = "&Arquivo";
				input
					IdFeirao					??:     	best12.
					NmFeirao					??:     	$255.
					DtInicioFeirao				??:			ddmmyy10.
					DtFimFeirao					??:			ddmmyy10.
					NmObjetivoFeirao			??:     	$100.
					NmTipoEvento				??:     	$100.
					CdParceiroComercial			??:     	best6.
					VrProducaoHistoricaPDV		??:     	commax18.2
					NmModeloEvento				??:     	$100.
					VrProducaoMetaFeirao		??:     	commax18.2
					CdFilial					??:     	best12.
					VrInvestimentoPDV			??:     	commax18.2
					VrProducaoIncrementalPDV	??:     	commax18.2
					VrInvestimentoBVFeirao		??:     	commax18.2
					VrProducaoHistoricaFeirao	??:     	commax18.2
					VrProducaoMetaPDV			??:     	commax18.2
				;
			run;

			/* Consolida a base temporária em uma base final */
			proc append base=BaseFinal data=tmp force;run;

			/* Remove base temporária */
			proc datasets lib=work nolist;
				delete tmp;
			run;
		%end;
	%let id = %sysfunc(close(&id.));
%mend;

data  _null_;
