%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbColaborador as
select * from connection to dbcon
(
	WITH
		TbColaboradorAjuste AS
		(
			SELECT

				 CdPessoa
				,DtInicioVigencia
				,CdSistemaOrigem
				,CdFilialComercial
				,CdCargoColaborador
				,CdAreaEstruturaComercial
				,CdEstadoColaborador
				,CdEmpresa
				,CdVinculoEmpregaticio
				,CdHierarquiaColaborador
				,CdFilialCadastro
			/*	,CdColaboradorRH*/
				,CASE
					WHEN CdColaboradorRH = 80767
					AND DsLogin = 'caroliny.guarize'
					AND CdSistemaOrigem = 209
					THEN 80720 ELSE CdColaboradorRH
				END  AS CdColaboradorRH
			/*	CdColaboradorRH = 80767 AND CdSistemaOrigem = 209 AND DsLogin = 'caroliny.guarize'*/
				,CdColaboradorSistemaOrigem
				,CdNivelCargoColaborador
				,CdOperadorRH
				,DtAdmissaoColaborador
				,DtDesligamentoColaborador
				,DtFimVigencia
				,DsEmail
				,DsLogin
				,NuExecucaoProcessoInclusao
				,NuExecucaoProcessoAtualizacao
				,CdEstruturaComercial
				,CdSubEstruturaComercial

				FROM DWBI_DTWH.TbColaborador Colaborador

			)
	SELECT
		 Colaborador.CdColaboradorSistemaOrigem CdColaborador
		,Colaborador.CdColaboradorRH IdMatricula
		,HierarquiaComercial.CdColaborador CdColaboradorBI
		,lower(HierarquiaComercial.DsLogin) DsLogin
   		,upper(coalesce(HierarquiaComercial.NmColaborador,Pessoa.NmPessoa)) NmColaborador
		,DeParaHierarquia.CdGrupoCargo CdGrupoCargoColaborador
		,DeParaHierarquia.NmGrupoCargo NmGrupoCargoColaborador

	FROM TbColaboradorAjuste Colaborador

		LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
			ON  Colaborador.CdPessoa = Pessoa.CdPessoa

		LEFT JOIN TbColaboradorAjuste ColaboradorRH
			ON  Colaborador.CdColaboradorRH = ColaboradorRH.CdColaboradorRH
			AND ColaboradorRH.CdSistemaOrigem = 202
			AND today() BETWEEN ColaboradorRH.DtInicioVigencia AND ColaboradorRH.DtFimVigencia

			LEFT JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
				ON  ColaboradorRH.CdPessoa = HierarquiaComercial.CdColaborador
				AND today() BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia

				LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
					ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

				LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
					ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

					LEFT JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
						ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
						AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador

	WHERE 1=1
	  AND today() BETWEEN Colaborador.DtInicioVigencia AND Colaborador.DtFimVigencia
	  AND Colaborador.CdSistemaOrigem = 209
	  ORDER BY ColaboradorRH.CdPessoa
)
;quit;

proc sort data=TbColaborador nodupkey;by CdColaborador;run;

%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
create table TbOperador as
select * from connection to dbcon
(
WITH TbColaborador as
(
	SELECT DISTINCT
		 Colaborador.CdOperador
		,first_value(Colaborador.CdColaborador)
			over (partition BY Colaborador.CdOperador ORDER BY
				CASE
					WHEN (Colaborador.NmColaborador LIKE 'DUP%'
						OR Colaborador.DsLogin LIKE 'neo.%'
						OR Colaborador.DsLogin LIKE 'parla.%'
						OR Colaborador.DsLogin LIKE 'tivit%'
						OR Colaborador.FlAtivo = 'N') THEN 1 ELSE 0 END) AS CdColaborador
		,first_value(Colaborador.CdFilial)
			over (partition BY Colaborador.CdOperador ORDER BY
				CASE
					WHEN (Colaborador.NmColaborador LIKE 'DUP%'
						OR Colaborador.DsLogin LIKE 'neo.%'
						OR Colaborador.DsLogin LIKE 'parla.%'
						OR Colaborador.DsLogin LIKE 'tivit%'
						OR Colaborador.FlAtivo = 'N') THEN 1 ELSE 0 END) AS CdFilial

	FROM DBA.TbColaborador Colaborador
	WHERE Colaborador.CdColaborador > 0
	AND Colaborador.CdOperador > 0
)

	SELECT
	 	 Colaborador.CdColaborador
		,Colaborador.CdOperador
		,Colaborador.CdFilial
	FROM DBA.TbColaborador Colaborador

		INNER JOIN TbColaborador ColaboradorAtivo
			ON  Colaborador.CdColaborador = ColaboradorAtivo.CdColaborador
)
;quit;

proc sql;
create table bases.TbColaborador as
select
	 Colaborador.CdColaborador
	,Operador.CdOperador
	,Colaborador.IdMatricula
	,Colaborador.CdColaboradorBI
	,Colaborador.DsLogin
	,Colaborador.NmColaborador
	,Colaborador.CdGrupoCargoColaborador
	,Colaborador.NmGrupoCargoColaborador
	,Operador.CdFilial
from TbColaborador Colaborador

	left join TbOperador Operador
		on Colaborador.CdColaborador = Operador.CdColaborador
;run;

proc sql;update bases.TbColaborador set NmGrupoCargoColaborador = 'Gerente Relacionamento Privado' where NmColaborador like 'GR VIRTUAL%';run;
proc sql;update bases.TbColaborador set NmGrupoCargoColaborador = 'Gerente Relacionamento Coban/B2B' where NmColaborador like 'GR VIRTUAL%';run;
proc sql;update bases.TbColaborador set CdGrupoCargoColaborador = 15 where NmColaborador like 'GR VIRTUAL%';run;
proc sql;update bases.TbColaborador set CdGrupoCargoColaborador = 5 where NmColaborador like 'GR VIRTUAL%';run;

data bases.TbColaborador;
length CdColaborador 5.;
length CdOperador 5.;
length IdMatricula 5.;
length CdColaboradorBI 8.;
length DsLogin $25.;
length NmColaborador $70.;
length CdGrupoCargoColaborador 5.;
length NmGrupoCargoColaborador $50.;
length CdFilial 5.;
set bases.TbColaborador
;run;

proc sql;create index CdColaborador on bases.TbColaborador;quit;
