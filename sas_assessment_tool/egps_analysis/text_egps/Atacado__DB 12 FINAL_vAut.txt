CREATE TABLE TB_CONCESSAOLIMITE AS SELECT * FROM CONNECTION TO SYBASE (
    SELECT DISTINCT
	  t1.CdPropostaLimiteCredito,
	  t3.CdPessoa,
	  t7.CdGrupoEconomico,
      t1.DtConclusaoAtividade,
      t1.CdEtapaProposta,
      t1.CdMotivoRecusaProposta,
      t1.NmAtividade,
      t1.NmAcao,
      t1.DsNota,

	  t6.CdUsuarioIsg AS NmGerente,
      t2.CdEtapaProposta,
      t2.NmEtapaProposta,
      t2.FlAtivo,
      t2.DtInclusao,
      t2.DtAlteracao,
      t2.DsLogin
    FROM (SELECT * FROM DBCONCESSAOLIMITECREDITO..TbRoteiroProposta WHERE NmAtividade = 'Validar Rating' AND NmAcao = 'VALIDARRATING')     AS t1
	LEFT JOIN DBCONCESSAOLIMITECREDITO..TbEtapaProposta       AS t2 ON t1.CdEtapaProposta = t2.CdEtapaProposta
	LEFT JOIN DBCONCESSAOLIMITECREDITO..TbEmpresaProposta     AS t3 ON t1.CdPropostaLimiteCredito = t3.CdPropostaLimiteCredito
	LEFT JOIN DBGLOBAL..TbPessoa                              AS t4 ON t1.DsLogin = t4.CdUsuarioIsg
	LEFT JOIN DBGLOBAL..TbFuncionario                         AS t5 ON t4.CdPessoa = t5.CdPessoaFuncionario
	LEFT JOIN DBGLOBAL..TbPessoa                              AS t6 ON t5.CdPessoaFuncSuperior = t6.CdPessoa

PROC SQL;

  CREATE TABLE BACEN_SCR_2 AS

	SELECT

		 DATEPART(t1.DTBASEDOCUMENTO_BACEN) format=date9. AS DTBASEDOCUMENTO_BACEN,
         t1.*,
		 INTNX('month', DATEPART(t1.DTBASEDOCUMENTO_BACEN), 1, 'B') AS DTBASEDOCUMENTO_BACEN_M1 FORMAT=date9.,
		 t2.VLVENCIMENTOOPERACAOCLIENTE_BACE AS VL_M2,
		 (VL_M2 / t1.VLVENCIMENTOOPERACAOCLIENTE_BACE) - 1 AS VAR_PERC

  FROM BACEN_SCR      AS t1

  LEFT JOIN BACEN_SCR AS t2 ON

	DATEPART(t1.DTBASEDOCUMENTO_BACEN) = INTNX('month', DATEPART(t2.DTBASEDOCUMENTO_BACEN), 1, 'B') AND

	t1.RAIZ_CNPJ = t2.RAIZ_CNPJ AND
	t1.NMVENCIMENTOCREDITO_BACEN = t2.NMVENCIMENTOCREDITO_BACEN AND
	t1.NMPRODUTOBACEN_BACEN = t2.NMPRODUTOBACEN_BACEN AND
	t1.NMSUBPRODUTOBACEN_BACEN = t2.NMSUBPRODUTOBACEN_BACEN

  ORDER BY t1.DTBASEDOCUMENTO_BACEN, t1.RAIZ_CNPJ;

QUIT;

PROC SQL;

CREATE TABLE BACEN_SCR_4 AS

    SELECT DISTINCT

		RAIZ_CNPJ                             AS NR_CNPJ_RAIZ,
		NMPESSOA_GLOBAL                       AS NM_PESSOA,
		NMGRUPOECONOMICO_GLOBAL               AS NM_GRUPO_ECONOMICO,

		DTBASEDOCUMENTO_BACEN                 AS DT_BASE_BACEN,

		NMVENCIMENTOCREDITO_BACEN,

		(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a vencer%' OR NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a vender%' THEN 'CREDITOS A VENCER' 
		      WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos vencidos%'  THEN 'CREDITOS VENCIDOS'
		      WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Limite de crédito%'  THEN 'LIMITE DE CREDITO'
			  WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a liberar%' THEN 'CREDITOS A LIBERAR'
		      WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos baixados%'  THEN 'CREDITOS BAIXADOS' ELSE '' END) AS TP_CREDITO_BACEN,

		QTOPERACAOCLIENTE_BACEN               AS QT_OPERACAO_CLIENTE,
		QTINSTUICAOFINANCEIRACLIENTE_BAC      AS QT_INSTITUICAO,
		QTOPERACAOSUBJUDICE_BACEN             AS QT_OPERACAO_SUBJUDICE,

        SUM(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a vencer%' OR NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a vender%' THEN VLVENCIMENTOOPERACAOCLIENTE_BACE ELSE 0 END) AS VL_CREDITOS_A_VENCER,
        SUM(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos vencidos%'  THEN VLVENCIMENTOOPERACAOCLIENTE_BACE ELSE 0 END) AS VL_CREDITOS_VENCIDOS,
        SUM(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Limite de crédito%'  THEN VLVENCIMENTOOPERACAOCLIENTE_BACE ELSE 0 END) AS VL_LIMITE_DE_CREDITO,
		SUM(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos a liberar%' THEN VLVENCIMENTOOPERACAOCLIENTE_BACE ELSE 0 END) AS VL_CREDITOS_A_LIBERAR,
        SUM(CASE WHEN NMVENCIMENTOCREDITO_BACEN LIKE '%Créditos baixados%'  THEN VLVENCIMENTOOPERACAOCLIENTE_BACE ELSE 0 END) AS VL_CREDITOS_BAIXADOS

    FROM BACEN_SCR_3

	GROUP BY RAIZ_CNPJ

    ORDER BY RAIZ_CNPJ;

PROC SQL;
CREATE TABLE TB_SERASA_VAR_ATUAL_X_ANTERIOR AS
    SELECT DISTINCT
        t1.NuCnpjRaiz,
        
        t2.DT_SERASA_ATUAL,
		t3.DT_SERASA_ANTERIOR,
		
        SUM(t2.VrProtesto + t2.VrPefin + t2.VrDivVencida) AS VR_ATUAL_OCORRENCIAS_COMERC,
		SUM(t3.VrProtesto + t3.VrPefin + t3.VrDivVencida) AS VR_ANTERIOR_OCORRENCIAS_COMERC,
		(CASE
			WHEN CALCULATED VR_ATUAL_OCORRENCIAS_COMERC > 0 AND CALCULATED VR_ANTERIOR_OCORRENCIAS_COMERC = 0 THEN 'NOVO APONTAMENTO'
			WHEN CALCULATED VR_ATUAL_OCORRENCIAS_COMERC = 0 AND CALCULATED VR_ANTERIOR_OCORRENCIAS_COMERC > 0 THEN 'BAIXA APONTAMENTO'
			WHEN CALCULATED VR_ATUAL_OCORRENCIAS_COMERC > CALCULATED VR_ANTERIOR_OCORRENCIAS_COMERC           THEN 'VARIAÇÃO POSITIVA'
			WHEN CALCULATED VR_ATUAL_OCORRENCIAS_COMERC < CALCULATED VR_ANTERIOR_OCORRENCIAS_COMERC           THEN 'VARIAÇÃO NEGATIVA'
			WHEN CALCULATED VR_ATUAL_OCORRENCIAS_COMERC = CALCULATED VR_ANTERIOR_OCORRENCIAS_COMERC           THEN 'SEM VARIAÇÃO'
		END) AS DESC_COMPORTAMENTO_OCOR_COMERC,
		ROUND(((SUM(t2.VrProtesto + t2.VrPefin + t2.VrDivVencida) / SUM(t3.VrProtesto + t3.VrPefin + t3.VrDivVencida)) - 1), 0.0001) FORMAT=percent8.2 AS VAR_PERC_OCORRENCIAS_COMERC,
		
        SUM(t2.VrRefin) AS VR_ATUAL_REFIN,
		SUM(t3.VrRefin) AS VR_ANTERIOR_REFIN,
		(CASE
			WHEN COALESCE(t2.VrRefin, 0) > 0 AND COALESCE(t3.VrRefin, 0) = 0 THEN 'NOVO APONTAMENTO'
			WHEN COALESCE(t2.VrRefin, 0) = 0 AND COALESCE(t3.VrRefin, 0) > 0 THEN 'BAIXA APONTAMENTO'
			WHEN COALESCE(t2.VrRefin, 0) > COALESCE(t3.VrRefin, 0)           THEN 'VARIAÇÃO POSITIVA'
			WHEN COALESCE(t2.VrRefin, 0) < COALESCE(t3.VrRefin, 0)           THEN 'VARIAÇÃO NEGATIVA'
			WHEN COALESCE(t2.VrRefin, 0) = COALESCE(t3.VrRefin, 0)           THEN 'SEM VARIAÇÃO'
		END) AS DESC_COMPORTAMENTO_REFIN,
		ROUND(((t2.VrRefin / t3.VrRefin) - 1), 0.0001) FORMAT=percent8.2 AS VAR_PERC_REFIN,
		
        SUM(t2.VrAcoes) AS VR_ATUAL_ACAO_JUDICIAL,
        SUM(t3.VrAcoes) AS VR_ANTERIOR_ACAO_JUDICIAL,
		(CASE
			WHEN t2.VrAcoes > 0 AND t3.VrAcoes = 0 THEN 'NOVO APONTAMENTO'
			WHEN t2.VrAcoes = 0 AND t3.VrAcoes > 0 THEN 'BAIXA APONTAMENTO'
			WHEN t2.VrAcoes > t3.VrAcoes           THEN 'VARIAÇÃO POSITIVA'
			WHEN t2.VrAcoes < t3.VrAcoes           THEN 'VARIAÇÃO NEGATIVA'
			WHEN t2.VrAcoes = t3.VrAcoes           THEN 'SEM VARIAÇÃO'
		END) AS DESC_COMPORTAMENTO_ACAO_JUD,
		ROUND(((t2.VrAcoes / t3.VrAcoes) - 1), 0.0001) FORMAT=percent8.2 AS VAR_PERC_ACAO_JUDICIAL,
		
        SUM(t2.VrProtesto + t2.VrPefin + t2.VrDivVencida + t2.VrRefin + t2.VrAcoes) AS VR_ATUAL_TOTAL_RESTRICOES,
        SUM(t3.VrProtesto + t3.VrPefin + t3.VrDivVencida + t3.VrRefin + t3.VrAcoes) AS VR_ANTERIOR_TOTAL_RESTRICOES,
		(CASE
			WHEN CALCULATED VR_ATUAL_TOTAL_RESTRICOES > 0 AND CALCULATED VR_ANTERIOR_TOTAL_RESTRICOES = 0 THEN 'NOVO APONTAMENTO'
			WHEN CALCULATED VR_ATUAL_TOTAL_RESTRICOES = 0 AND CALCULATED VR_ANTERIOR_TOTAL_RESTRICOES > 0 THEN 'BAIXA APONTAMENTO'
			WHEN CALCULATED VR_ATUAL_TOTAL_RESTRICOES > CALCULATED VR_ANTERIOR_TOTAL_RESTRICOES           THEN 'VARIAÇÃO POSITIVA'
			WHEN CALCULATED VR_ATUAL_TOTAL_RESTRICOES < CALCULATED VR_ANTERIOR_TOTAL_RESTRICOES           THEN 'VARIAÇÃO NEGATIVA'
			WHEN CALCULATED VR_ATUAL_TOTAL_RESTRICOES = CALCULATED VR_ANTERIOR_TOTAL_RESTRICOES           THEN 'SEM VARIAÇÃO'
		END) AS DESC_COMPORTAMENTO_RESTRICOES,
		ROUND(((SUM(t2.VrProtesto + t2.VrPefin + t2.VrDivVencida + t2.VrRefin + t2.VrAcoes) / SUM(t3.VrProtesto + t3.VrPefin + t3.VrDivVencida + t3.VrRefin + t3.VrAcoes)) - 1), 0.0001) FORMAT=percent8.2 AS VAR_PERC_TOTAL_RESTRITIVOS
    FROM TB_SERASA_DIMENSAO_DTCONSULTA    AS t1
    FULL JOIN TB_SERASA_DTCONSULTA_TEMPO1 AS t2 ON t1.NuCnpjRaiz = t2.NuCnpjRaiz AND t1.DT_SERASA_ATUAL = t2.DT_SERASA_ATUAL
    FULL JOIN TB_SERASA_DTCONSULTA_TEMPO2 AS t3 ON t1.NuCnpjRaiz = t3.NuCnpjRaiz AND t1.DT_SERASA_ANTERIOR = t3.DT_SERASA_ANTERIOR
    GROUP BY t1.NuCnpjRaiz, t2.DT_SERASA_ATUAL, t3.DT_SERASA_ANTERIOR
    ORDER BY t2.DT_SERASA_ATUAL DESC;

CREATE TABLE PME.BATIMENTO_TBBRUTA AS

	SELECT DISTINCT

		t1.CD_PESSOA,

		t2.CdPessoa AS CdPessoaTBBRUTA,

		t2.CdGpEconomico,

		t2.FlAtivoGP,

		t1.NU_CPFCNPJ,

		t1.NM_PESSOA,

		DATEPART(t1.DT_VENCIMENTO_MAX_LIMITE) FORMAT=DDMMYY.9 AS DT_VENCIMENTO_MAX_LIMITE,
		t2.DtVencLimiteCliente,

		(DATEPART(t1.DT_VENCIMENTO_MAX_LIMITE) = t2.DtVencLimiteCliente) AS FL_DIF_DATE,

		t1.GL_VL_LIMITE_TOTAL,
		t1.PLCA_VL_LIMITE_TOTAL,

		t1.GL_VL_RISCO_TOTAL,
		t1.PLCA_VL_RISCO_TOTAL,

		(t1.GL_VL_LIMITE_TOTAL + t1.PLCA_VL_LIMITE_TOTAL) AS VALOR_TOTAL,
		(t1.GL_VL_RISCO_TOTAL + t1.PLCA_VL_RISCO_TOTAL) AS VALOR_TOTAL_COMPROMETIDO,

		t2.VrTotalAprovadoCliente AS TBBRUTA_VrTotalAprovadoCliente,
		t2.VrTotalRiscoCliente    AS TBBRUTA_VrTotalRiscoCliente,

		(CALCULATED VALOR_TOTAL - t2.VrTotalAprovadoCliente)           AS DIF_LIMITE,
		(CALCULATED VALOR_TOTAL_COMPROMETIDO - t2.VrTotalRiscoCliente) AS DIF_COMPROMETIDO,

		t2.FlAtivoP,
		t1.FL_ATIVO_GL

		

	FROM WORK.BASE_PUBLICO_TOTAL t1

	FULL JOIN INSIGHT.TBBRUTA   t2 ON (t1.CD_PESSOA = t2.CdPessoa)

	WHERE t2.TpLancamento = 'Pessoa' AND t2.CdPessoa IS NOT NULL

	ORDER BY t2.VrTotalAprovadoCliente DESC;

QUIT;
