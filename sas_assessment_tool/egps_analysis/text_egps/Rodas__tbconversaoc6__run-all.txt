PROC SQL;
%CONECTADB(server=BI);
SELECT NuDiaUtilMes, Mes INTO :DUtil,:Mes
     FROM connection to dbcon (
  SELECT TOP 1 MONTH(DtTempo) AS Mes, NuDiaUtilMes FROM DWBI_DIME.TbDimTempoData
  WHERE DtTempo <= TODAY() AND NuDiaUtilMes > 0
  ORDER BY DtTempo DESC
);
RUN;

PROC SQL OUTOBS=10;
CREATE TABLE TOP10 AS
  SELECT Banco, COUNT(DISTINCT NuCpfCnpj) AS Qtd FROM SOLITMP.TbConversaoC6
  WHERE NuAnoMes = (CASE WHEN MONTH(TODAY()) <> &Mes. OR &DUtil < 2 THEN &safraM1. ELSE &safraM0. END)
  GROUP BY Banco
  ORDER BY calculated Qtd DESC;
RUN;

PROC SQL;
CREATE TABLE TOTAIS AS
  SELECT
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM5. THEN NuCpfCnpj END)) AS "&safraM5."n,
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM4. THEN NuCpfCnpj END)) AS "&safraM4."n,
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM3. THEN NuCpfCnpj END)) AS "&safraM3."n,
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM2. THEN NuCpfCnpj END)) AS "&safraM2."n,
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM1. THEN NuCpfCnpj END)) AS "&safraM1."n,
    COUNT(DISTINCT (CASE WHEN NuAnoMes = &safraM0. THEN NuCpfCnpj END)) AS "&safraM0."n
  FROM
    SOLITMP.TbConversaoC6;
RUN;

PROC SQL;
CREATE TABLE DATA_ATUALIZA AS
  SELECT
    CASE WHEN B.Banco IS NULL THEN 'OUTROS' ELSE UPPER(A.Banco) END AS Banco,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM5. THEN A.NuCpfCnpj END)) / MIN(C."&safraM5."n) AS "&safraM5."n FORMAT=PERCENT10.1,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM4. THEN A.NuCpfCnpj END)) / MIN(C."&safraM4."n) AS "&safraM4."n FORMAT=PERCENT10.1,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM3. THEN A.NuCpfCnpj END)) / MIN(C."&safraM3."n) AS "&safraM3."n FORMAT=PERCENT10.1,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM2. THEN A.NuCpfCnpj END)) / MIN(C."&safraM2."n) AS "&safraM2."n FORMAT=PERCENT10.1,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM1. THEN A.NuCpfCnpj END)) / MIN(C."&safraM1."n) AS "&safraM1."n FORMAT=PERCENT10.1,
    COUNT(DISTINCT (CASE WHEN A.NuAnoMes = &safraM0. THEN A.NuCpfCnpj END)) / MIN(C."&safraM0."n) AS "&safraM0."n FORMAT=PERCENT10.1
  FROM
    SOLITMP.TbConversaoC6 A
  LEFT JOIN
    TOP10 B ON A.Banco = B.Banco
  LEFT JOIN
    TOTAIS C ON 1=1
  GROUP BY
    1
  ORDER BY
    (CASE WHEN MONTH(TODAY()) <> &Mes. OR &DUtil < 2 THEN calculated "&safraM1."n ELSE calculated "&safraM0."n END) DESC;
RUN;
