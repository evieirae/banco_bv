%CRIAVARDB;
proc sql;
%CONECTADB(server=CTR);
create table BASE_00 as
select * from connection to dbcon
(

SELECT

p.CdParceiroComercial,
p.DtUltimaAtualizacao,
p.FlAtivo,
p.NmFantasia,
q.DsLog,
q.CdLog,
q.CdOcorrencia,
q.DtOcorrencia,
w.TpPessoa

FROM dbcor..TbParceiroComercial AS p

INNER JOIN DBCOR..TbLogParceiroComercial AS q
ON q.CdParceiroComercial = p.CdParceiroComercial

INNER JOIN dbcor..TbPessoa AS w
ON w.CdPessoa = p.CdPessoa

WHERE p.DtUltimaAtualizacao BETWEEN '20230101 00:00:00' AND '20241231 23:59:59'
AND p.FlAtivo = 'N'
AND q.DtOcorrencia >= '20230101'

);
;run;

DATA BASE_01;
SET WORK.BASE_00;
DSLOG_ABREV = UPCASE(scan(DsLog,1,"-"));
;RUN;

PROC SORT DATA=WORK.BASE_01 OUT=WORK.BASE_02;BY CdParceiroComercial DtOcorrencia;RUN;

DATA BASE_03;
SET BASE_02;
RETAIN QTD_PC;
IF FIRST.CdParceiroComercial THEN QTD_PC = 1;
ELSE QTD_PC = QTD_PC + 1;
BY CdParceiroComercial;
RUN;

PROC SQL;
CREATE TABLE MAX_REG_PC AS
SELECT
CdParceiroComercial,
MAX(QTD_PC) AS MAX_REG_PC

FROM WORK.BASE_03 A
GROUP BY 1
;RUN;

PROC SQL;
CREATE TABLE Base_Log_ParceiroComercial AS
SELECT

A.*

FROM WORK.BASE_03 A

INNER JOIN WORK.MAX_REG_PC B ON
B.CDPARCEIROCOMERCIAL = A.CDPARCEIROCOMERCIAL
AND B.MAX_REG_PC = A.QTD_PC

;RUN;

PROC SQL;
CREATE TABLE VALIDACAO_LOG AS
SELECT
CDPARCEIROCOMERCIAL,
COUNT(CDPARCEIROCOMERCIAL) AS QTD
FROM WORK.Base_Log_ParceiroComercial
GROUP BY 1
;RUN;
