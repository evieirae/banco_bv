LIBNAME RELEXC '/infogrc/relatorioexecutivo';

PROC SQL;
  CREATE TABLE orc AS
    SELECT
      NuAnoMes AS Safra,
	  NmGrupoProdutoMeta AS Produto,
	  COALESCE(DsTipoCanalProducao,'DESCONHECIDO') AS Canal,
	  '' AS EstadoDoBem,
      SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS OrcamentoVrProducao FORMAT=NUMX18.2,
	  SUM(CASE WHEN Tipo = 'Meta' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS MetaVrProducao FORMAT=NUMX18.2,
	  SUM(CASE WHEN Tipo = 'BN' AND TpValor = 'VrProducao' THEN VrMedida ELSE 0 END) AS BNVrProducao FORMAT=NUMX18.2,
	  0 AS OrcamentoQtdContratos,
	  0 AS OrcamentoRetorno,
	  SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = 'VpTxTir' THEN VrMedida ELSE 0 END) AS OrcamentoTir FORMAT=NUMX18.2,
	  0 AS OrcamentoQtdTc,
	  0 AS OrcamentoQtdTab,
	  SUM(CASE WHEN Tipo = 'Orçamento' AND TpValor = 'VpTxSpread' THEN VrMedida ELSE 0 END) AS OrcamentoSpreadVp FORMAT=NUMX18.2,
	  0 AS BNQtdContratos,
	  0 AS BNRetorno,
	  0 AS BNTIR,
	  0 AS BNQtdTc,
	  0 AS BNQtdTab,
	  0 AS BNSpreadNet
    FROM RELEXC.TbMeta
	WHERE NmGrupoProdutoMeta IN ('CVG Leves','CVG Pesados','Privado','CP','CIG')
	  AND NmGrupoProdutoMeta IS NOT NULL
	  AND NuAnoMes >= 202007
	GROUP BY 1,2,3,4
	HAVING (calculated OrcamentoVrProducao +
           calculated MetaVrProducao +
           calculated BNVrProducao +
           calculated OrcamentoTir +
           calculated OrcamentoSpreadVp) > 0;
RUN;
