proc sql;
	connect to bigquery as perm (project='bv-saem-dtm' credfile='/projetos/acessos/bq/em/saem-dtm.json'
		schema='bv-pdbd-prd' READ_MODE=storage ALLOW_LARGE_RESULTS=must MAX_CHAR_LEN=100);
	Create table CCBENVIODOCS as

	select distinct b.DtEntradaProposta,
		b.DtContrato,
		a.NuCpfCnpj_ori,
		b.nucontrato as nucontrato2,
		a.NuProposta,
		a.NuContrato,
		a.DsStatusEnvioDocumento,
		a.DtInclusao,
		a.DtAlteracao,
		b.NmGrupoProdutoMeta

from connection to perm
/*cdenvioemail*/
	(SELECT
		tced.NuProposta,
		tced.NuContrato,
		tced.DsComentarioCliente AS descricao_cliente,
		tced.CdDocumento,
		tced.FlEnvioDocumento,
		tced.DtInclusao,
		tced.DtAlteracao,
		tced.CdEnvioEmail,
		tced.DsEmailCliente AS email_cliente,
		tced.NuCpfCnpj as NuCpfCnpj_ori,
		(substr(CAST(tced.NuCpfCnpj AS STRING),1,5)) AS NuCpfCnpj,
		tsed.CdStatusEnvioDocumento,
		tsed.DsStatusEnvioDocumento
	FROM `bv-pdbd-prd.trusted_aprocontroleenviocontrato.tbcontroleenviodocumento` tced
		INNER JOIN `bv-pdbd-prd.trusted_aprocontroleenviocontrato.tbstatusenviodocumento` tsed
			ON tsed.CdStatusEnvioDocumento = tced.CdStatusEnvioDocumento
			) a
		right JOIN TMPPRD B
			ON a.NUPROPOSTA = b.NUPROPOSTA

			/*WHERE a.NUPROPOSTA IN (SELECT NUPROPOSTA FROM TMPPRD)*/
	;
	disconnect from perm;
quit;

PROC SORT DATA=CCBENVIODOCS
	OUT=CCBENVIODOCS1;
	BY DESCENDING NuCpfCnpj_ori DESCENDING NUPROPOSTA DESCENDING DtInclusao;
RUN;

data work.CCBENVIODOCS1_2;
	set work.CCBENVIODOCS1;
	BY DESCENDING NuCpfCnpj_ori DESCENDING NUPROPOSTA DESCENDING DtInclusao;

	if first.NUPROPOSTA then
		n=1;
	else n+1;
run;
PROC SQL;
	CREATE TABLE MXENVIODOCS AS
		SELECT   a.DtEntradaProposta,
          a.DtContrato,
          a.NuCpfCnpj_ori,
          a.NuProposta,
          a.NuContrato,
          a.DsStatusEnvioDocumento,
          a.DtInclusao as DtPriInclusaoDoc,
          a.DtAlteracao as DtPriAlteracaoDoc,
          a.NmGrupoProdutoMeta,
          a.n AS TtlInclusoes
		FROM CCBENVIODOCS1_2 A
			JOIN
				(SELECT NUPROPOSTA, MAX(N) AS MXN FROM CCBENVIODOCS1_2 GROUP BY NUPROPOSTA) B
					ON A.NUPROPOSTA = B.NUPROPOSTA AND A.N = B.MXN;
QUIT;
proc sql;

create table CCBENVIODOCS2 AS
SELECT a.DtEntradaProposta,
          a.DtContrato,
          a.NuCpfCnpj_ori,
          a.NuProposta,
          a.NuContrato,
          a.DsStatusEnvioDocumento,
		  b.DtPriInclusaoDoc,
		  b.DtPriAlteracaoDoc,
		  a.DtInclusao as DtUltInclusaoDoc,
          a.DtAlteracao as DtUltAlteracaoDoc,
          a.NmGrupoProdutoMeta,
       	  b.TtlInclusoes as TtlInclusoesDocs

from CCBENVIODOCS1_2 a

join
MXENVIODOCS b
on a.Nuproposta = b.nuproposta
where a.n = 1 ;quit;

PROC SQL;
	CREATE TABLE CCBENVIODOCSCPF AS
		SELECT DISTINCT NuCpfCnpj_ori
			FROM CCBENVIODOCS2 where NuCpfCnpj_ori is not null;
QUIT;
