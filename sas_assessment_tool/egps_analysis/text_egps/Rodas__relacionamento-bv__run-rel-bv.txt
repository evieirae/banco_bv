PROC SQL;
CREATE TABLE Base_Prod_Liq AS
SELECT

a.NuCpfCnpj,
A.NuProposta,
A.nuanomes,
A.DtEntradaProposta,
A.NuContratoGestao AS NuContrato,
C.DtContrato,

A.FlItau,

A.FlMontadoras,
A.FlOutros,
A.FlConcorrente,

case when B.NUANOMES=. then a.nuanomes else B.NUANOMES end AS NuAnoMes_Liquidado,
B.NUCONTRATO AS NUCONTRATO_LIQ,
case when B.NUANOMES=. then a.nuanomes else B.NUANOMES+3 end as NUANOMES_LIQUIDADO_3M,
B.TpLiquidacao AS TpLiquidacao,
B.DtLiquidacaoEfetiva,

CASE WHEN A.NuProposta <> . THEN 1 ELSE 0 END AS FL_PROPOSTA,
CASE WHEN A.NuContratoGestao <> . THEN 1 ELSE 0 END AS FL_CONTRATO,
CASE WHEN B.NUCONTRATO = A.NuContratoGestao AND B.DtLiquidacaoEfetiva <> . THEN 1 ELSE 0 END AS FL_LIQUIDADO

FROM WORK.Base_TbOriginacao A

LEFT JOIN bases.TBEXTLIQUIDACAO B ON B.NUCONTRATO = A.NuContratoGestao
LEFT JOIN WORK.BASE_TBPRODUCAO C ON C.NUCONTRATO = A.NuContratoGestao

/*where a.nucpfcnpj = '33099184837'*/

order by 1,3

;RUN;

proc sql;
create table Base_Final as
select distinct
Pliq.*
,case when Ori.nuanomes<>. and Ori.NuContratoGestao <> . then 'Sim' else 'Não' end as FlaRelacionamentoBV
,case when Ori.nuanomes<>. and Ori.NuContratoGestao <> . then 1 else 0 end as FlaRelacionamentoBV_2

,case when Ori.nuanomes<>. and Ori.FlConcorrente= 1 and calculated FlaRelacionamentoBV_2 = 0 then 'Sim' else 'Não' end as FlaRelacionamentoMercado
,case when Ori.nuanomes<>. and Ori.FlConcorrente= 1 and calculated FlaRelacionamentoBV_2 = 0 then 1 else 0 end as FlaRelacionamentoMercado_2

,case when Ori.nuanomes<>. and Ori.FlConcorrente= 0 and calculated FlaRelacionamentoBV_2 = 0 and calculated FlaRelacionamentoMercado_2 = 0 then 'Sim' else 'Não' end as FlaRelacionamentoSemBvMerado
,case when Ori.nuanomes<>. and Ori.FlConcorrente= 0 and calculated FlaRelacionamentoBV_2 = 0 and calculated FlaRelacionamentoMercado_2 = 0 then 1 else 0 end as FlaRelacionamentoSemBvMerado_2

from Base_Prod_Liq Pliq

left join Base_TbOriginacao Ori on Ori.nucpfcnpj = Pliq.nucpfcnpj
and Ori.nuanomes between Pliq.NuAnoMes_Liquidado and Pliq.NuAnoMes_Liquidado+3

where FL_LIQUIDADO=1 /*and Pliq.nucpfcnpj = '00008993483'*/

order by Pliq.NuAnoMes_Liquidado, Pliq.NuCpfCnpj, CALCULATED FlaRelacionamentoBV_2 DESC, CALCULATED FlaRelacionamentoMercado_2 DESC, CALCULATED FlaRelacionamentoSemBvMerado_2 DESC
;run;
