%import_meliuz(202512);

%macro class_base_meliuz(ref);
	proc sort data=meliuz.tb_meliuz_&ref. out=matriz_&ref. (keep=cpf);
	by cpf;
	run;

	proc sql;
		create table meliuz.tb_meliuz_&ref. (where = (cpf > 0 )) as
			select distinct
				   a.cpf
				  ,case when b.cpf ne . and c.cpf ne . then 'Conta + Cartão'
				  		when b.cpf =  . and c.cpf ne . then 'Cartão'
						else 								'Conta'
				   end as Base
				  ,"&ref." as ref

			from matriz_&ref. as a
				left join (select distinct cpf from meliuz.tb_meliuz_&ref. where Base = 'Conta')  as b
			on a.cpf = b.cpf
				left join (select distinct cpf from meliuz.tb_meliuz_&ref. where Base = 'Cartão') as c
			on a.cpf = c.cpf;
	quit;
%mend;

%macro base_meliuz_pura(ref, data_ref);
libname bk2 '/riscosbvf/fraude/PESSOAL/VITOR/temp';

PROC SQL;
CREATE TABLE tb_clientes_DRE_&ref. (WHERE = (RECEITA > 0 )) AS
	SELECT   cliente
			,DATA_REF
			,COALESCE(RESULTADO_CARTEIRA,0)+COALESCE(RESULTADO_PRODUCAO,0) AS RECEITA
	FROM clientes.DRE_GERAL_&ref._NEW
	WHERE  FLAG_POSSE_PRODUTO not in ('Apenas Seguro','Apenas Conta Atacado')
;QUIT;

proc sql;
	create table tb_clientes_nao_cgr as
		select distinct CLIENTE
		from bk2.BASE_GERAL_CLIENTE_PRODUTO
		where data_ref = &data_ref.
			and input(CLIENTE,best32.) not in (select CLIENTE from tb_clientes_DRE_&ref.)
			and meliuz <= 0;
quit;

PROC SQL;
create table tb_base_bv_mais_meliuz as
	select 'CGR BV' as Base,
		   "&ref." as Ref,
		   'Base BV - Gera Receita' as Origem,
		    count(distinct cliente) as qtde
	from tb_clientes_DRE_&ref.
	group by 1,2,3

		union all

	select 'Base de clientes BV sem CGR' as Base,
		   "&ref." as Ref,
		   'Base Clientes - Sem Receita' as Origem,
		    count(1) as qtde
	from tb_clientes_nao_cgr
	group by 1,2,3

		union all

	select 	a.Base,
		    "&ref." as Ref,
			case when b.cliente =  . and input(c.cliente,14.) = . then 'Base Meliuz Sem BV'
				 when b.cliente ne . then 'Base Meliuz Com BV - CGR'
				 when input(c.cliente,14.) ne . then 'Base Meliuz Com BV - Sem Receita'
			 end as Origem,
			count(distinct cpf) as qtde
	from meliuz.tb_meliuz_&ref. as a
		left join (select cliente from tb_clientes_DRE_&ref.) as b
	on a.cpf = b.cliente
		left join tb_clientes_nao_cgr as c
	on a.cpf = input(c.cliente,14.)

	group by 1,2,3;
quit;

proc datasets lib=work nolist;
	append base=analise_base_meliuz_3
		   data=tb_base_bv_mais_meliuz;
run;
%mend;

PROC SQL;
CREATE TABLE VAREJO_INICIO AS
	SELECT
		CASE WHEN UPPER(PRODUTO_AUTOMACAO) IN ('RODAS','MCN','NA PISTA') THEN 'RODAS'
			 WHEN UPPER(PRODUTO_AUTOMACAO) = 'VCS' AND UPPER(GRUPO_PRODUTO) IN ('MOTOS','PESADOS','VANS','VEICULOS','LEVES') THEN 'RODAS'
			 WHEN UPPER(PRODUTO_AUTOMACAO) IN ('CDC','CIG','CP','CVG','COMISSIONADO','ECOMMERCE','ESTUDANTIL'
				,'FGTS - SAQUE ANIVERSARIO','INSS','PRIVADO','PUBLICO','SAUDE','SOLAR','TURISMO') THEN 'EMPRESTIMOS'
			 WHEN UPPER(PRODUTO_AUTOMACAO) = 'VCS' AND UPPER(GRUPO_PRODUTO) IN ('PRIVADO','FGTS','CIG','CP','CP CONSIGNADO PRIVADO','CP COM GARANTIA - LEVES'
				,'CP COM GARANTIA - PESADOS','CVG PF','CVG PJ','REFINANCIAMENTO - LEVES','REFINANCIAMENTO - PESADOS'
				,'SOLAR') THEN 'EMPRESTIMOS'
			 WHEN UPPER(PRODUTO_AUTOMACAO) = 'PROMOTIVA' THEN 'PROMOTIVA'
			 WHEN UPPER(PRODUTO_AUTOMACAO) = 'VCS' AND UPPER(GRUPO_PRODUTO) = 'VCS' THEN 'VCS'
			 ELSE 'OUTROS'
		 END AS NEGOCIOS
		,CASE WHEN UPPER(PRODUTO_AUTOMACAO) = 'VCS' AND UPPER(GRUPO_PRODUTO) IN ('MOTOS','PESADOS','VANS','VEICULOS','LEVES')     THEN 'SEGUROS RODAS'
			  WHEN UPPER(PRODUTO_AUTOMACAO) = 'VCS' AND UPPER(GRUPO_PRODUTO) NOT IN ('MOTOS','PESADOS','VANS','VEICULOS','LEVES') THEN 'SEGUROS EMPRESTIMOS'
			  ELSE GRUPO_PRODUTO
		 END AS PRODUTO_DESC

		,CASE WHEN MONTH((Mes_Contrato)) = MONTH(&DATA_REF.) AND  YEAR((Mes_Contrato)) = YEAR(&DATA_REF.)
				  THEN  'CONTRATO_MES_CAC' ELSE ''
		 END AS FLAG_MES
		,*
	from FINAL.TABELAO_&MES_ANO._AUTOMACAO

;quit;

PROC SQL;
CREATE TABLE VAREJO_01 AS
		SELECT
			 NUCONTRATOGESTAO
			,NuCpfCnpj
			,NEGOCIOS
			,FLAG_MES
			,put(grupo_produto, $50.) as GRUPO_PRODUTO
			,PRODUTO_AUTOMACAO
			,PRODUTO_DESC
			,MES_CONTRATO
			,ScoreCredito
			,SUM(COALESCE(VLR_ACCRUAL_PROPRIA,0)) + SUM(COALESCE(VLR_CUSTOCAPT_ATE60,0))
			 + SUM(COALESCE(VLR_RCO,0)) as RESULTADO_CARTEIRA










			,SUM(COALESCE(RECEITA_SERVICO,0)) + SUM(COALESCE(VLR_TARIFA,0)) AS RESULTADO_PRODUCAO

			,SUM(COALESCE(VLR_RETORNO,0)) + SUM(COALESCE(VLR_CUSTO_TOTAL,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_DIR,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_INDIR,0))
				+ SUM(COALESCE(VLR_CCOMIS_SUP_INDIR,0)) AS CUSTOS_DIRETO

			,SUM(COALESCE(VLR_CUSTOCAPT_ACIMA60,0)) +  SUM(COALESCE(VLR_MORA_MULTA,0)) + SUM(COALESCE(VLR_COBRANCA,0)) AS RESULTADO_INADIMPLENCIA

			,SUM(COALESCE(VLR_DEMAISRES_DIVIDA_SUB,0)) + SUM(COALESCE(VLR_DEMAISRES_HEDGES,0))
				+ SUM(COALESCE(VLR_DEMAISRES_NFF,0)) + SUM(COALESCE(VLR_DEMAISRES_OUTROS,0)) + SUM(COALESCE(VLR_DEMAISRES_REMCAPITAL,0))
				+ SUM(COALESCE(VLR_FRAUDE,0)) AS OUTROS_FRAUDES

			,SUM(COALESCE(VLR_DESC_PRINCIPAL,0)) + SUM(COALESCE(VLR_PDD_BRUTA,0)) + SUM(COALESCE(VLR_RECUPERACAO,0)) AS CUSTO_CREDITO

			,SUM(COALESCE(VLR_DP_ORIG_DIR,0)) + SUM(COALESCE(VLR_DP_ORIG_IND,0)) AS DP_ORIGINICAO

			,SUM(COALESCE(VLR_DA_ORIG_DIR,0)) + SUM(COALESCE(VLR_DA_ORIG_IND,0)) AS DA_ORIGINICAO

			,SUM(VLR_TRIBUTOS) AS DESPESAS_TRIBUTARIAS

			,SUM(COALESCE(VLR_CONTG_CIVEIS,0)) AS CIVEIS

			,CALCULATED RESULTADO_CARTEIRA + CALCULATED RESULTADO_PRODUCAO + CALCULATED CUSTOS_DIRETO + CALCULATED RESULTADO_INADIMPLENCIA
				+ CALCULATED OUTROS_FRAUDES + CALCULATED CUSTO_CREDITO + CALCULATED DP_ORIGINICAO + CALCULATED DA_ORIGINICAO
				+ CALCULATED DESPESAS_TRIBUTARIAS + CALCULATED CIVEIS AS MARGEM_CONTRIBUICAO

			,SUM(COALESCE(VLR_DP_SUP_IND,0)) AS DP_SUPORTE

			,SUM(COALESCE(VLR_DA_SUP_IND,0)) AS DA_SUPORTE

			,SUM(COALESCE(VLR_DTRAB_DIR,0)) + SUM(COALESCE(VLR_DTRAB_IND,0)) AS TRABALHISTA

			,SUM(COALESCE(VLR_OUTRAS_CONT_FISC,0))  AS FISCAIS

			,SUM(COALESCE(VLR_BNDU,0)) +SUM(COALESCE(VLR_OUTRAS_GCUT,0)) + SUM(COALESCE(VLR_OUTRAS_REC_DESP,0)) AS OUTROS_RESULTADOS

			,SUM(COALESCE(VLR_PLR_DIR,0)) + SUM(COALESCE(VLR_PLR_IND,0)) AS RV

			,CALCULATED MARGEM_CONTRIBUICAO + CALCULATED FISCAIS +  CALCULATED TRABALHISTA + CALCULATED RV

				+ CALCULATED OUTROS_RESULTADOS + CALCULATED DP_SUPORTE + CALCULATED DA_SUPORTE AS LAIR

			,SUM(COALESCE(VLR_IR_CS_FIM,0)) AS IR

			,CALCULATED LAIR + CALCULATED IR AS LUCRO_LIQUIDO

			,SUM(vlr_ll) AS LL_TABELAO

			,((SUM(COALESCE(VLR_CUSTO_TOTAL,0))
				+SUM(COALESCE(VLR_CCOMIS_ORIG_DIR,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_INDIR,0))
				+ SUM(COALESCE(VLR_CCOMIS_SUP_INDIR,0))) * MAX(B.PERC_CAC))
				+ SUM(COALESCE(VLR_DA_ORIG_DIR,0)) + SUM(COALESCE(VLR_DA_ORIG_IND,0))
				 AS CAC

			,((SUM(COALESCE(VLR_CUSTO_TOTAL,0))
				+SUM(COALESCE(VLR_CCOMIS_ORIG_DIR,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_INDIR,0))
				+ SUM(COALESCE(VLR_CCOMIS_SUP_INDIR,0))) * MAX(B.PERC_CAC)) as perc

			,SUM(COALESCE(VLR_DA_ORIG_DIR,0)) + SUM(COALESCE(VLR_DA_ORIG_IND,0))
				+ SUM(COALESCE(VLR_VCS_DESPADM_ORIG,0)) as loja_fisica

			,CALCULATED RESULTADO_CARTEIRA + CALCULATED RESULTADO_PRODUCAO + (calculated CUSTOS_DIRETO - CALCULATED CAC)
				+ CALCULATED RESULTADO_INADIMPLENCIA +  CALCULATED CUSTO_CREDITO + SUM(VLR_TRIBUTOS)
				 AS MARGEM

			,SUM(COALESCE(VLR_RETORNO,0)) + SUM(COALESCE(VLR_CUSTO_TOTAL,0)) - CALCULATED CAC + SUM(COALESCE(VLR_DA_ORIG_DIR,0)) + SUM(COALESCE(VLR_DA_ORIG_IND,0))  AS RETORNO_CUSTOS

			,SUM(COALESCE(VLR_DA_ORIG_DIR,0)) + SUM(COALESCE(VLR_DA_ORIG_IND,0)) as despesas

			,SUM(COALESCE(VLR_RETORNO,0))  AS RETORNO

			,SUM(COALESCE(VLR_CUSTO_TOTAL,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_DIR,0)) + SUM(COALESCE(VLR_CCOMIS_ORIG_INDIR,0))
				+ SUM(COALESCE(VLR_CCOMIS_SUP_INDIR,0)) AS TOTAL_CUSTOS_DIRETO

			,SUM(COALESCE(VLR_PDD_BRUTA,0)) AS PDD
			,SUM(COALESCE(VLR_CUSTO_TOTAL,0)) AS custos

			,SUM(COALESCE(VLR_CUSTO_TOTAL,0)) * MAX(B.PERC_CAC) as vezez

			,sum(COALESCE(VRSALDO,0)) 			as VRSALDO
			,sum(COALESCE(PL_RAR,0))  			as PL_RAR
			,sum(COALESCE(PL_ROE,0))  			as PL_ROE
			,sum(COALESCE(VLR_CART_ATE_60,0))   as VLR_CART_ATE_60
			,sum(COALESCE(VLR_CART_ACIMA_60,0)) as VLR_CART_ACIMA_60
			,sum(COALESCE(VLR_CART_ACIMA_90,0)) as VLR_CART_ACIMA_90

	FROM VAREJO_INICIO AS A
		LEFT JOIN (SELECT *
					FROM PERCENTUAL_CAC_VAREJO
					WHERE UPPER(GRUPOACIONISTA) = 'CAC - AQUISIÇÃO DE CLIENTES')
					AS B
	ON A.NEGOCIOS = B.NEGOCIO






	GROUP BY
		 NUCONTRATOGESTAO
		,NuCpfCnpj
		,NEGOCIOS
		,FLAG_MES
		,GRUPO_PRODUTO
		,PRODUTO_AUTOMACAO
		,PRODUTO_DESC
		,MES_CONTRATO
		,ScoreCredito
;QUIT;

CREATE TABLE VCS_01 AS
	SELECT
		 NUCONTRATO AS NUCONTRATOGESTAO
		,NuCpfCnpj
		,'RODAS' AS NEGOCIOS
		,CASE WHEN UPCASE(GRUPO_PRODUTO) LIKE '%CVG%'  OR UPCASE(GRUPO_PRODUTO) LIKE '%EVG%' THEN 'EGV'
			  WHEN UPCASE(GRUPO_PRODUTO) LIKE '%FGTS%' THEN 'FGTS'
			  ELSE 	GRUPO_PRODUTO
		 END AS GRUPO_PRODUTO
		,'' AS FLAG_MES
		,MES_CONTRATO
		,0 as SCORECREDITO

		,0 AS RESULTADO_CARTEIRA

		,SUM(COALESCE(RECEITA,0)) AS RESULTADO_PRODUCAO

		,SUM(COALESCE(RETORNO,0)) + SUM(COALESCE(CUSTOS_DIRETO,0)) AS CUSTOS_DIRETO

		,0 AS RESULTADO_INADIMPLENCIA

		,0 AS OUTROS_FRAUDES

		,0 AS CUSTO_CREDITO

		,0 AS DP_ORIGINICAO

		,0 AS DA_ORIGINICAO

		,SUM(DESPESAS_TRIBUTARIAS) AS DESPESAS_TRIBUTARIAS

		,SUM(COALESCE(CONTINGENCIAS_CIVEIS,0)) AS CIVEIS

		,CALCULATED RESULTADO_CARTEIRA + CALCULATED RESULTADO_PRODUCAO + CALCULATED CUSTOS_DIRETO + CALCULATED RESULTADO_INADIMPLENCIA
			+ CALCULATED OUTROS_FRAUDES + CALCULATED CUSTO_CREDITO + CALCULATED DP_ORIGINICAO + CALCULATED DA_ORIGINICAO
			+ CALCULATED DESPESAS_TRIBUTARIAS + CALCULATED CIVEIS AS MARGEM_CONTRIBUICAO

		,SUM(COALESCE(DESPESAS_PESSOAL,0)) AS DP_SUPORTE

		,SUM(COALESCE(DESPESAS_ADM,0)) AS DA_SUPORTE

		,SUM(CONTINGENCIAS_TRABALHISTAS) AS TRABALHISTA

		,SUM(COALESCE(CONTINGENCIAS_FISCAIS,0)) AS FISCAIS

		,0 AS OUTROS_RESULTADOS

		,SUM(COALESCE(REMUNERACAO_VARIAVEL,0))  AS RV

		,CALCULATED MARGEM_CONTRIBUICAO + CALCULATED FISCAIS +  CALCULATED TRABALHISTA + CALCULATED RV

			+ CALCULATED OUTROS_RESULTADOS + CALCULATED DP_SUPORTE + CALCULATED DA_SUPORTE AS LAIR

		,SUM(COALESCE(IR,0)) AS IR

		,CALCULATED LAIR + CALCULATED IR AS LUCRO_LIQUIDO

		,SUM(LUCRO_LIQUIDO) AS LL_TABELAO

		,SUM(COALESCE(LAIR,0)) AS LAIR_TABELAO
		,SUM(COALESCE(IR,0)) AS IR_TABELAO

		,(SUM(COALESCE(CUSTOS_DIRETO,0)) * MAX(B.PERC_CAC))  AS CAC

		,CALCULATED RESULTADO_CARTEIRA + CALCULATED RESULTADO_PRODUCAO + (SUM(COALESCE(CUSTOS_DIRETO,0)) - CALCULATED CAC)
			+ CALCULATED RESULTADO_INADIMPLENCIA +  CALCULATED CUSTO_CREDITO + SUM(DESPESAS_TRIBUTARIAS) AS MARGEM

		,SUM(COALESCE(RETORNO,0)) + SUM(COALESCE(CUSTOS_DIRETO,0)) - CALCULATED CAC AS RETORNO_CUSTOS

		,SUM(COALESCE(RETORNO,0))  AS RETORNO

		,SUM(COALESCE(CUSTOS_DIRETO,0))  AS TOTAL_CUSTOS_DIRETO
		,0 AS PDD
		,0 as VRSALDO
		,0 as PL_RAR
		,0 as PL_ROE
		,0 as VLR_CART_ATE_60
		,0 as VLR_CART_ACIMA_60
		,0 as VLR_CART_ACIMA_90

	FROM TABELAO_VCS AS A
		LEFT JOIN PERCENTUAL_CAC_VAREJO (WHERE = (UPPER(GRUPOACIONISTA) = 'CAC - AQUISIÇÃO DE CLIENTES')) AS B
	ON  B.NEGOCIO = 'RODAS'
	GROUP BY
		 NUCONTRATOGESTAO
		,NuCpfCnpj
;QUIT;

PROC SQL;
CREATE TABLE POSSE_PRODUTO AS
	SELECT
		CLIENTE
		,CASE WHEN CARTOES <> 0 AND CONTA <> 0 AND FINANCIAMENTO_SEGURO <> 0 THEN 'Financeira - Conta + Cartao'
		      WHEN CARTOES = 0 AND CONTA <> 0 AND FINANCIAMENTO_SEGURO <> 0 THEN 'Financeira - Conta'
			  WHEN CARTOES <> 0 AND CONTA = 0 AND FINANCIAMENTO_SEGURO <> 0 THEN 'Financeira - Cartao'
			  WHEN CARTOES = 0 AND CONTA = 0 AND FINANCIAMENTO <> 0 THEN 'Financeira - Puro'

			  WHEN CARTOES = 0 AND CONTA <> 0 AND FINANCIAMENTO_SEGURO = 0 THEN   'Não Financeira - Conta'
			  WHEN CARTOES <> 0 AND CONTA = 0 AND FINANCIAMENTO_SEGURO = 0 THEN   'Não Financeira - Cartao'
			  WHEN CARTOES <> 0 AND CONTA <> 0 AND FINANCIAMENTO_SEGURO = 0 THEN   'Não Financeira - Conta + Cartao'
			  WHEN CARTOES = 0 AND CONTA = 0 AND FINANCIAMENTO = 0  AND SEGURO <> 0 THEN 'Apenas Seguro'
			  WHEN CARTOES = 0 AND CONTA = 0 AND FINANCIAMENTO = 0  AND SEGURO = 0 AND CONTA_ATACADO <> 0 THEN 'Apenas Conta Atacado'
			  WHEN MELIUZ <> 0	THEN 'Conta Meliuz'
	  		  ELSE 'OUTROS'
		 END AS FLAG_POSSE_PRODUTO
		,CASE WHEN COALESCE(RODAS,0) <> 0 AND (COALESCE(CARTOES,0) + COALESCE(EMPRESTIMOS,0)) = 0
			  THEN 'Veiculo Puro'
			  WHEN COALESCE(RODAS,0) <> 0 AND COALESCE(CARTOES,0) <> 0 AND  COALESCE(EMPRESTIMOS,0)  = 0
			  THEN 'Veiculo + Cartoes'
			  WHEN COALESCE(RODAS,0) <> 0 AND COALESCE(CARTOES,0) = 0 AND COALESCE(EMPRESTIMOS,0) <> 0
	  		  THEN 'Veiculo + Emprestimos'
			  WHEN COALESCE(EMPRESTIMOS,0) <> 0 AND COALESCE(CARTOES,0) = 0 AND  COALESCE(RODAS,0)   = 0
			  THEN 'Emprestimos Puro'
			  WHEN COALESCE(EMPRESTIMOS,0) <> 0 AND COALESCE(CARTOES,0) <> 0 AND   COALESCE(RODAS,0)  = 0
			  THEN 'Emprestimos + Cartao'
			  WHEN COALESCE(CARTOES,0) <> 0 AND (COALESCE(RODAS,0) + COALESCE(EMPRESTIMOS,0) )  = 0
			  THEN 'Cartao Puro'
			  WHEN COALESCE(CARTOES,0) <> 0 AND COALESCE(RODAS,0) <> 0 AND  COALESCE(EMPRESTIMOS,0)  <> 0
			  THEN 'Veiculo + Emprestimos + Cartao'
			  WHEN  COALESCE(CONTA,0)<> 0 AND ( COALESCE(CARTOES,0) +  COALESCE(RODAS,0) + COALESCE(EMPRESTIMOS,0) )  = 0
			  THEN 'Puro CONTA'
			  WHEN COALESCE(CONTA,0) <> 0
			  THEN 'Puro Meliuz'
			  ELSE 'OUTROS'
		 END AS FLAG_POSSE_CLIENTE



	FROM PREPARA_POSSE_PRODUTO
;QUIT;

libname bk2 '/riscosbvf/fraude/PESSOAL/VITOR/temp';

PROC SQL;
CREATE TABLE tb_clientes_DRE_&ref. (WHERE = (RECEITA > 0 )) AS
	SELECT   cliente
			,DATA_REF
			,COALESCE(RESULTADO_CARTEIRA,0)+COALESCE(RESULTADO_PRODUCAO,0) AS RECEITA
	FROM clientes.DRE_GERAL_&ref._NEW
	WHERE  FLAG_POSSE_PRODUTO not in ('Apenas Seguro','Apenas Conta Atacado')
;QUIT;

proc sql;
	create table tb_clientes_nao_cgr as
		select distinct CLIENTE
		from bk2.BASE_GERAL_CLIENTE_PRODUTO
		where data_ref = &data_ref.
			and input(CLIENTE,best32.) not in (select CLIENTE from tb_clientes_DRE_&ref.)
			and meliuz <= 0;
quit;

PROC SQL;
create table tb_base_bv_mais_meliuz as
	select 'CGR BV' as Base,
		   "&ref." as Ref,
		   'Base BV - Gera Receita' as Origem,
		    count(distinct cliente) as qtde
	from tb_clientes_DRE_&ref.
	group by 1,2,3

		union all

	select 'Base de clientes BV sem CGR' as Base,
		   "&ref." as Ref,
		   'Base Clientes - Sem Receita' as Origem,
		    count(1) as qtde
	from tb_clientes_nao_cgr
	group by 1,2,3

		union all

	select 	a.Base,
		    "&ref." as Ref,
			case when b.cliente =  . and input(c.cliente,14.) = . then 'Base Meliuz Sem BV'
				 when b.cliente ne . then 'Base Meliuz Com BV - CGR'
				 when input(c.cliente,14.) ne . then 'Base Meliuz Com BV - Sem Receita'
			 end as Origem,
			count(distinct cpf) as qtde
	from meliuz.tb_meliuz_&ref. as a
		left join (select cliente from tb_clientes_DRE_&ref.) as b
	on a.cpf = b.cliente
		left join tb_clientes_nao_cgr as c
	on a.cpf = input(c.cliente,14.)

	group by 1,2,3;
quit;

proc datasets lib=work nolist;
	append base=analise_base_meliuz_3
		   data=tb_base_bv_mais_meliuz;
run;
%mend;
