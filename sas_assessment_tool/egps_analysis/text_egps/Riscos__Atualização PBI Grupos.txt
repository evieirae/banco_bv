PROC SQL;

CREATE TABLE AUX_MONITORAMENTO AS

SELECT DISTINCT
		DtProcessamento
		,CdGpEconomico
		,NmGpEconomico
		,CASE WHEN FlEsteiraPME = 1 AND DsClassificacao = "" THEN DtClassificacaoSME
			 	WHEN NmGerenteCredito IN ('BERACA.WEN', 'FABIO.TADEU', 'KARINA.FULANETTI', 'MILENA.CAMPANHA', 'RAFAEL.MARTINEZ') THEN DtClassificacaoGP
			 	ELSE DtClassificacaoSME END AS DT_CALCULO_MONITORAMENTO format date9.
		,CASE WHEN FlEsteiraPME = 1 AND DsClassificacao = "" THEN DsClassificacaoSME
			 	WHEN NmGerenteCredito IN ('BERACA.WEN', 'FABIO.TADEU', 'KARINA.FULANETTI', 'MILENA.CAMPANHA', 'RAFAEL.MARTINEZ') THEN DsClassificacaoGP
			 	ELSE DsClassificacaoSME END AS status
FROM INSIGHT.TBBRUTA
WHERE TpLancamento = 'Pessoa'
;QUIT;

proc sql;
create table base_final as
select
    A.*,
    A.DtVenc format MMYY7. as DATA_VENC,
    put(A.estagio, 2.) as ESTAGIO_TEXTO,
    B.CD_RATING_FINAL as RATING_ATUAL,
    datepart(B.DT_CALCULO_RATING) format=date9. as DT_CALCULO_RATING,
    C.STATUS as MONITORAMENTO_ATUAL,
    C.DT_CALCULO_MONITORAMENTO format=date9.
from infos_grupos_&anomes. A
left join MEMORIA_CALCULO_GERENCIAL_2 B
    on A.CdGrupo_Economico = B.CD_GRUPO_ECONOMICO
left join (
    select CdGpEconomico,
           max(DT_CALCULO_MONITORAMENTO) as DT_CALCULO_MONITORAMENTO,
           min(STATUS) as STATUS
    from AUX_MONITORAMENTO
    where status ne ''
    group by CdGpEconomico
) C
    on A.CdGrupo_Economico = C.CdGpEconomico
order by GRUPO, SALDO;
quit;
