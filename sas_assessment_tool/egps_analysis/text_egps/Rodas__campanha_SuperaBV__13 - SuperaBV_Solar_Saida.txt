,Supera.Eleg_Comissao_Prod
,Supera.Eleg_Acelerador_Prod
,Coalesce(Supera.PenFin_Meta,0) as PenFin_Meta Format COMMAX12.5
,Coalesce(Supera.PenFin_Real,0) as PenFin_Real Format COMMAX12.5
,Coalesce(Supera.Perc_Ating_Seguros,0) as Perc_Ating_Seguros Format COMMAX12.5
