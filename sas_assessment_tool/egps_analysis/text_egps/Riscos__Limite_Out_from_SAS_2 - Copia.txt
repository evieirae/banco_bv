PROC SQL;
   CREATE TABLE tb_limite AS
   SELECT t1.cdgrupoeconomico AS cdgrupoeconomico,
   		  '01JAN1960'D AS dtabertura FORMAT DATE9.,
          t1.NomeGrupo AS nmgrupoeconomico,
          
            (MEAN(t1.Aprovado_Grupo_LMR)) FORMAT=COMMAX22.2 AS  lmr ,
          
            (MAX(INPUT(t1.VencimentoGrupoLMR,DDMMYY10.))) FORMAT=DATE9. AS venclmr,
          
            (MEAN(t1.Aprovado_Grupo_Exposicao)) FORMAT=COMMAX22.2 AS  exposicao,
			CALCULATED venclmr FORMAT=DATE9. AS vencexposicao
      FROM SIC.DMRELATIOLIMITESHIST t1
      WHERE t1.DtReferencia = &ref_limite AND t1.cdgrupoeconomico NOT IS MISSING
      GROUP BY t1.cdgrupoeconomico,
               t1.NomeGrupo
      ORDER BY t1.cdgrupoeconomico;
QUIT;
