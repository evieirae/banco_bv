/* Buscar dados de mensageria */
PROC SQL;
CREATE TABLE MsgProposta AS
  SELECT DISTINCT
    Num_Proposta AS NuProposta,
    CASE WHEN COALESCE(msg_v7,Msg_Dash) = '9. Sem CdSimuPrazofnnco' THEN '9. Não houve mensageria' ELSE COALESCE(msg_v7,Msg_Dash) END AS Mensageria,
    Msg_2 AS Mensageria2,
    Msg_ent_min AS MsgEntradaMinima,
    fx_ent_min AS EntradaMinima,
	DsPzFaixaMotivoRecusaPDEC AS MotivoRecusa
  FROM CRM.SIMU_PROP_AQUISICAO
  WHERE
    AnoMes >= PUT(&safraXM.,6.)
    AND Num_Proposta <> .
    AND (calculated Mensageria || Msg_2) IS NOT NULL;
RUN;

PROC SORT DATA=MsgProposta NODUPKEY;
BY NuProposta;
RUN;

PROC SQL; CREATE INDEX NuProposta ON MsgProposta; QUIT;

/* Buscar informações de leads  */
PROC SQL;
CREATE TABLE Leads AS
  SELECT DISTINCT
    num_proposta as NuProposta,
    Sistema_Origem as CompraLeads,
	CASE WHEN flag_comissao = 1 AND Sistema_Origem IN ('DRYVE','Dryve sem Simul.','Escale sem Simul.','KARVI','ICARROS','MERCADO_LIVRE','MOBIAUTO','USADOSBR','SO_CARRAO') THEN 1
		 WHEN Sistema_Origem IN ('SITE_BV','MEU_CARRO_NOVO') THEN 1
		 ELSE 0
    END AS FlagProducao,
	campanha_agrup as Origem
  FROM
    CRM.SIMU_PROP_LEADS
  WHERE
    num_proposta <> .
    AND Sistema_Origem IS NOT NULL
    AND AnoMes >= PUT(&safraXM.,6.);
RUN;

PROC SORT DATA=Leads NODUPKEY;
BY NuProposta;
RUN;

PROC SQL; CREATE INDEX NuProposta ON Leads; QUIT;

/* Busca dados de rating de simulação */
DATA TbRatingSimulacao;
SET AUXL.TbRatingSimulacao;
  WHERE NuAnoMes >= &safraM1.;
RUN;

options DLDMGACTION=REPAIR;

PROC SQL;
CREATE TABLE Convsimu AS
  SELECT
    A.NuProposta,
	A.VrProposta AS VrSimulacao,
    CASE WHEN COALESCE(A.RatingCliente, B.RatingCliente) <> . THEN 1 ELSE 0 END AS FlRatingC,
	CASE WHEN COALESCE(A.RatingSimulacao, B.RatingSimulacao) <> . THEN 1 ELSE 0 END AS FlRatingS,
    COALESCE(A.RatingCliente, B.RatingCliente) * A.VrProposta AS VpRatingCliente,
	COALESCE(A.RatingSimulacao, B.RatingSimulacao) * A.VrProposta AS VpRatingSimulacao
  FROM
    CONVSIMU.tbsimulacoes A
  LEFT JOIN TbRatingSimulacao B
    ON A.DtSimulacaoInicial = B.DtSimulacao
    AND A.NuCpfCnpj = B.NuCpfCnpj
  WHERE
    A.NuProposta <> .
    AND STRIP(TRANSTRN(A.Produto,"CVG ",TRIMN(''))) IN ('Leves','Motos','Pesados');
RUN;
