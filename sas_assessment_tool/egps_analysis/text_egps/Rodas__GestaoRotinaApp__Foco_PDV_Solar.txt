/*Gerar base da Carta PDV*/
Proc Sql;
Create table BaseFocoPDVSolar as
Select
	 &SafraM0. as NuAnoMEs
	,TbParc.CdParceiroComercial
	,put(TbParc.CdParceiroComercial, 8.) || '-' || TbParc.NmParceiroComercial as NmParceiroComercial
	,TbParc.DsLoginN1C
	,TbParc.DsLoginN2C
	,'ewerton.furtado' as DsLoginN3C
	,0 as Indicado
	,0 as Confirmado
From ritosol.Base_Cobans_Solar_Saida as TbParc
Where TbParc.FlAtivo = 'S'
;Quit;

proc export
data=BaseFocoPDVSolar
file="/riscosbvf/BA/processos/output/tbvisitapdvfocoinput_solar.csv" replace
dbms=dlm;
delimiter=';';
run;
