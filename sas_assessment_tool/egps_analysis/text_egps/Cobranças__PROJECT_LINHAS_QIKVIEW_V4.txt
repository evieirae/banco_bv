PROC SQL;
CREATE TABLE PREVIA_RAZAO_GERAL AS
SELECT
	MESCOMPETENCIA,
	CDORIGEM,
	C.CDLINHA,
	A.DTLANCAMENTO,
	UPCASE(D.NMLINHA) AS NMLINHA,
	A.DESC_HIST_LANCTO,
	A.COD_CONTA_CONTABIL AS NUCONTACONTABIL,
	A.NOME_CONTA_CONTABIL AS NMCONTACONTABIL,
	A.NR_INT AS CDVEICULOLEGAL,
	B.OBSERVACAO AS GRUPOFINANCAS,
	(VL_CRED-VL_DEB) AS VRLANCAMENTO
FROM
PREVIA_RAZAO_GERAL A LEFT JOIN RESULT.GPCONTA B
ON
A.COD_CONTA_CONTABIL = B.NUCONTACONTABIL

LEFT JOIN RESULT.GPFINANC C
ON
B.OBSERVACAO = C.GRUPOFINANCAS

LEFT JOIN RESULT.GPLINHA D
ON
C.CDLINHA = D.CDLINHA

WHERE
C.CDLINHA IS NOT NULL
AND
A.DTLANCAMENTO NOT IS MISSING
AND
DESC_HIST_LANCTO NOT IN ('0043-2-APURAÇÃO RESULTADO 2º SEMESTRE','0051-8-APURAÇÃO DO RESULTADO DO 2º SEM');

SELECT

	'CARTAO' AS PRODUTO,
	ORIGEM, 
	MESCOMPETENCIA,
	DTLANCAMENTO,
	NMLINHA,
	SUM(VRLANCAMENTO) AS VRLANCAMENTO
FROM
	PREVIA_RAZAO_CARTAO_BNDU

GROUP BY
	ORIGEM,
	MESCOMPETENCIA,
	DTLANCAMENTO,
	NMLINHA
ORDER BY
	DTLANCAMENTO ASC;
QUIT;
RUN;

DATA RESULTADO_LINHAS_CARTAO;
LENGTH PRODUTO $50.;
SET
	PREVIA_RAZAO_CARTAO_BNDU
;

IF NMLINHA = 'BNDU' THEN PRODUTO = 'RODAS'; ELSE PRODUTO = PRODUTO;

IF PRODUTO = 'OUTROS' THEN DELETE;

PROC SQL;
CREATE TABLE RESULTADO_LINHAS_CARTAO AS
SELECT
	ORIGEM, 
	PRODUTO,
	MESCOMPETENCIA,
	DTLANCAMENTO,
	NMLINHA,
	VRLANCAMENTO
FROM
	RESULTADO_LINHAS_CARTAO
ORDER BY
	ORIGEM,
	MESCOMPETENCIA,
	DTLANCAMENTO,
	PRODUTO ASC;
QUIT;
RUN;

PROC SQL;
CREATE TABLE RESULTADO_LINHAS_CARTAO AS
SELECT
	A.ORIGEM, 
	A.PRODUTO,
	A.MESCOMPETENCIA,
	A.DTLANCAMENTO,
	B.DU,
	A.NMLINHA,
	A.VRLANCAMENTO
FROM
	RESULTADO_LINHAS_CARTAO A LEFT JOIN RESULT.DIAS_UTEIS B
	ON
	A.DTLANCAMENTO = B.DATA
ORDER BY
	ORIGEM,
	MESCOMPETENCIA,
	DTLANCAMENTO,
	PRODUTO ASC;
QUIT;
RUN;

PROC SQL;
    CREATE TABLE AJUSTADO_FINAL AS
    SELECT
        A.*,
        CASE
            WHEN A.VALIDACAO = 'FALSE' THEN COALESCE(
                
                (SELECT MIN(B.DTPROCESSAMENTO)
                 FROM LINHAS_CARTAO_PROXY AS B
                 WHERE B.SAFRA = A.SAFRA
                   AND B.LINHA = A.LINHA
                   AND B.VALIDACAO = 'TRUE'
                   AND B.DTPROCESSAMENTO > A.DTPROCESSAMENTO),
                
                (SELECT MAX(B.DTPROCESSAMENTO)
                 FROM LINHAS_CARTAO_PROXY AS B
                 WHERE B.SAFRA = A.SAFRA
                   AND B.LINHA = A.LINHA
                   AND B.VALIDACAO = 'TRUE')
            )
            ELSE A.DTPROCESSAMENTO
        END AS DTPROCESSAMENTO_AJUSTADO
    FROM LINHAS_CARTAO_PROXY AS A;
