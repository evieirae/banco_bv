PROC SQL;

CREATE TABLE TBL_BASECONTAS_CARTOES AS

SELECT * FROM CONNECTION TO DBCON
(
	SELECT DISTINCT c.SafraRef,
					a.CdDimContaCartao,
                    year(b.DtAberturaContaCartao)*100+month(b.DtAberturaContaCartao) AS SafraConta,

	                CASE WHEN b.NmTipoCartao IN ('VISA LIVRE','BV MASTERCARD','BV LIVRE','MASTERCARD BV LIVRE','VISA BV LIVRE')  THEN 'LIVRE'
						 WHEN b.NmTipoCartao IN ('VISA MAIS','MASTERCARD MAIS','MASTERCARD BV MAIS','VISA BV MAIS')              THEN 'MAIS'
						 WHEN b.NmTipoCartao IN ('VISA UNICO','MASTERCARD BV UNICO','VISA BV UNICO','MASTERCARD BLACK MULTIPLO') THEN 'UNICO'
						 ELSE 'LEGADO'
						 END AS NmVarianteAgrupado,
					CASE WHEN d.NmTipoCartao IN ('VISA LIVRE','BV MASTERCARD','BV LIVRE','MASTERCARD BV LIVRE','VISA BV LIVRE')  THEN 'LIVRE'
						 WHEN d.NmTipoCartao IN ('VISA MAIS','MASTERCARD MAIS','MASTERCARD BV MAIS','VISA BV MAIS')              THEN 'MAIS'
						 WHEN d.NmTipoCartao IN ('VISA UNICO','MASTERCARD BV UNICO','VISA BV UNICO','MASTERCARD BLACK MULTIPLO') THEN 'UNICO'
						 ELSE 'LEGADO'
						 END AS NmVarianteAgrupadoOrigem,

	                CASE WHEN b.NmTipoCartao IN ('VISA LIVRE','BV MASTERCARD','BV LIVRE','MASTERCARD BV LIVRE','VISA BV LIVRE')  THEN 'LIVRE'
						 WHEN b.NmTipoCartao IN ('VISA MAIS','MASTERCARD MAIS','MASTERCARD BV MAIS','VISA BV MAIS')              THEN 'MAIS'
						 WHEN b.NmTipoCartao IN ('VISA UNICO','MASTERCARD BV UNICO','VISA BV UNICO','MASTERCARD BLACK MULTIPLO') THEN 'UNICO'
						 ELSE b.NmVarianteConta
						 END AS NmVariante,
					CASE WHEN d.NmTipoCartao IN ('VISA LIVRE','BV MASTERCARD','BV LIVRE','MASTERCARD BV LIVRE','VISA BV LIVRE')  THEN 'LIVRE'
						 WHEN d.NmTipoCartao IN ('VISA MAIS','MASTERCARD MAIS','MASTERCARD BV MAIS','VISA BV MAIS')              THEN 'MAIS'
						 WHEN d.NmTipoCartao IN ('VISA UNICO','MASTERCARD BV UNICO','VISA BV UNICO','MASTERCARD BLACK MULTIPLO') THEN 'UNICO'
						 ELSE d.NmVarianteConta
						 END AS NmVarianteOrigem,

					CASE WHEN b.NmBandeiraCartao = 'NAO IDENTIFICADO' THEN 'MASTERCARD'
					     ELSE b.NmBandeiraCartao
						 END AS NmBandeiraCartao,
					CASE WHEN d.NmBandeiraCartao = 'NAO IDENTIFICADO' THEN 'MASTERCARD'
					     ELSE d.NmBandeiraCartao
						 END AS NmBandeiraCartaoOrigem

	FROM DWBI_DMOC.TbFatoContaCartao a

	INNER JOIN DWBI_DIME.TbDimContaCartao b
	ON a.cddimcontacartaoversao = b.cddimcontacartaoversao

	INNER JOIN
    (
	     SELECT year(a.dtreferencia)*100+month(a.dtreferencia) AS SafraRef,
		        max(a.dtreferencia)                            AS MaxSafraRef
		 FROM DWBI_DMOC.TbFatoContaCartao a
		 GROUP BY SafraRef
     ) AS c
	ON a.dtreferencia= c.MaxSafraRef

	INNER JOIN
    (
    	SELECT DISTINCT CdDimContaCartao,
		                NmTipoCartao,
						NmBandeiraCartao,
						NmVarianteConta
        FROM DWBI_DIME.TbDimContaCartao AS a
		WHERE a.DtAberturaContaCartao = a.DtInicioVersaoContaCartao
    ) AS d
	ON a.CdDimContaCartao = d.CdDimContaCartao

	WHERE a.DtReferencia BETWEEN &Dt_Ref_Inicial. AND &Dt_Ref_Final. 
);
QUIT;

PROC SORT DATA=TBL_BASECONTAS_CARTOES; BY SafraRef CdDimContaCartao; RUN;
PROC SORT DATA=TBL_BASECONTAS_CARTOES NODUPKEY DUPOUT=DUPS_1; BY SafraRef CdDimContaCartao; RUN;

CREATE TABLE TBL_CONTASCRIADAS_CANAL AS

SELECT DISTINCT a.Safra_Conta AS SafraConta,
                a.CdDimContaCartao,
				a.DtAberturaContaCartao,
				a.CPF,
				a.DsMeioVenda,
				a.DECISAO_VENDA,
				a.DsTipoCartao,
				a.TIPO_PRODUTO,
				a.VrTotalReceitas,
				a.Idade,
				a.SUBSEGMENTO,

				CASE WHEN c.CdDimContaCartao IS NOT MISSING THEN
                     (CASE WHEN SEGMENTO_N2 = '1. PAP'       THEN 'PAP'
				           WHEN SEGMENTO_N2 = '2. 2ª Camada' THEN 'PAP'
				           WHEN SEGMENTO_N2 = '3. D1 - Core' THEN 'D1 Core'
					       WHEN SEGMENTO_N2 = '3. D1 - Lead' THEN 'D1 Lead'
				           WHEN SEGMENTO_N2 = '4.1 MA - BD Cliente'      THEN 'BD Mar Aberto'
					       WHEN SEGMENTO_N2 = '4.2 MA - BD Nao Cliente'  THEN 'BD Mar Aberto'
						   WHEN SEGMENTO_N2 = '5. MA Site BV'            THEN 'BD Mar Aberto'
					       WHEN SEGMENTO_N2 = '6. Dotz'                  THEN 'Dotz'
					       WHEN SEGMENTO_N2 = '6.1 eCred'                THEN 'Serasa eCred'
					       WHEN SEGMENTO_N2 = '6.2 IQ'                   THEN 'IQ'
					       WHEN SEGMENTO_N2 = '7.Core - Correntista BAU' THEN 'PAP'
					       WHEN SEGMENTO_N2 = '8. MA - Correntista BAU'  THEN 'BD Mar Aberto'
					       WHEN SEGMENTO_N2 = '8.1 Correntista - Testes' THEN 'PAP'
					       WHEN SEGMENTO_N2 = '9. Oferta Integrada'      THEN 'D1 Core'
						   END
					 )
                     WHEN c.CdDimContaCartao IS MISSING AND a.SUBSEGMENTO = 'D1' AND d.NmPublico_2 IS NOT MISSING THEN d.NmPublico_2
					 ELSE b.NmPublico
                     END AS NmPublico,

				CASE WHEN c.CdDimContaCartao IS NOT MISSING AND c.Canal_v2 IS NOT MISSING THEN c.Canal_v2
					 ELSE e.NmCanal
                     END AS NmCanal

FROM MIS2.BASEUNICACARTOES a

LEFT JOIN TBL_DEPARA_PUBLICO b
ON COMPRESS(a.DECISAO_VENDA) = COMPRESS(b.DecisaoVenda) AND
   COMPRESS(a.TIPO_PRODUTO) = COMPRESS(b.TipoProduto)

LEFT JOIN CNLCART.ANALITICO_CUBO_GERENCIAL_V5 c
ON a.CdDimContaCartao = c.CdDimContaCartao

LEFT JOIN WORK.TBL_DEPARA_CANAL_D1 d
ON a.CdDimContaCartao = d.CdDimContaCartao

LEFT JOIN TBL_DEPARA_CANAL e
ON COMPRESS(a.DECISAO_VENDA) = COMPRESS(e.DecisaoVenda)

WHERE a.CdDimContaCartao IS NOT MISSING AND a.VrLimiteContaCartaoGeral <> 0
;
QUIT;

PROC SORT DATA=TBL_CONTASCRIADAS_CANAL; BY CPF CdDimContaCartao; RUN;
PROC SORT DATA=TBL_CONTASCRIADAS_CANAL NODUPKEY DUPOUT=DUPS_6; BY CdDimContaCartao; RUN;

CREATE TABLE TBL_CUBO_RESULTADO_FINAL AS

SELECT
		a.SafraRef,

	    CASE WHEN (INT(a.SafraConta/100)) < 2017 THEN '< 2017'
             ELSE PUT((INT(a.SafraConta/100)),6.)
             END AS AnoConta,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'NOVO PORTFOLIO'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'NOVO PORTFOLIO'
			      ELSE 'LEGADO' END
		     )
			 WHEN b.NmVarianteAgrupadoOrigem = 'LEGADO' THEN 'LEGADO'
			 ELSE 'NOVO PORTFOLIO'
             END AS NmPortfolio,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'LIVRE'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'MAIS'
			      ELSE 'LEGADO' END
		     )
			 ELSE b.NmVarianteAgrupado
             END AS NmVarianteAgrup,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'LIVRE'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'MAIS'
			      ELSE 'LEGADO' END
		     )
			 ELSE b.NmVarianteAgrupadoOrigem
             END AS NmVarianteOrigemAgrup,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'LIVRE'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'MAIS'
			      ELSE a.NmVarianteConta END
		     )
			 ELSE b.NmVariante
             END AS NmVarianteCartao,

		CASE WHEN b.CdDimContaCartao IS MISSING THEN
             (
			 CASE WHEN a.NmVarianteConta = 'GOLD'     AND a.SafraConta >= 202109 THEN 'LIVRE'
		          WHEN a.NmVarianteConta = 'PLATINUM' AND a.SafraConta >= 202109 THEN 'MAIS'

             END AS NmVarianteCartaoOrigem,

		      (
              CASE WHEN a.NmFamiliaProdutoCartao IN ('DOTZ BV','NETPOINTS') THEN 'Dotz'
			       ELSE 'PAP'
				   END
			  )
			  ELSE e.NmPublico
			  END AS NmPublicoCartao,

DATA CARTEIRA_PARC_FXATRASO;

INFILE DATALINES DSD DELIMITER="09"x;

INPUT
	SafraRef :BEST12.
	ChaveUnica :$20.
	CdContaCartao_Cadoc :BEST12.
	DiasAtraso :BEST12.
	SaldoCarteira :BEST12.
;
DATALINES;
RUN;

options symbolgen;
options compress=yes reuse=yes replace;

%MACRO DO_LABEL(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;
		CREATE TABLE CARTEIRA_PARC_FXATRASO_&I. AS

	         SELECT &SafraRef. AS SafraRef,
                    COMPRESS((COMPRESS(a.CdProduto)||COMPRESS(a.Cd)||COMPRESS(PUT(c.ChaveSafra,6.)))) AS ChaveUnica,
			        ((INPUT(a.CdProduto, 8.) * 1000000000) + INPUT(a.NuConta, 9.)) AS CdContaCartao_Cadoc,
				    COALESCE(b.DiaAtraso,0) AS DiasAtraso,
			        SUM(COALESCE(a.Saldo,0)) AS SaldoCarteira

			FROM ESTUDOS2.CARTAO3040_&SafraRef. AS a

			LEFT JOIN
			(
			     SELECT Cd,
			            max(DiaAtraso)as DiaAtraso
			     FROM ESTUDOS2.CARTAO3040_&SafraRef.
			     WHERE Mod NOT IN ('1901','1904') AND ClassOp <> 'HH'
			     GROUP BY 1
			) as b
			ON a.Cd = b.Cd

			LEFT JOIN
				(
					SELECT Cd,
					       CdProduto,
                           MAX(input(SUBSTRN(COMPRESS(IniRelactCli),1,4),best12.)*100 + input(SUBSTRN(COMPRESS(IniRelactCli),6,2),best12.)) AS ChaveSafra
					FROM ESTUDOS2.CARTAO3040_&SafraRef.
					GROUP BY 1,2
				) AS c
			ON a.Cd = c.Cd AND
			   a.CdProduto = c.CdProduto
			WHERE a.Mod NOT IN ('1901','1904') and a.ClassOp <> 'HH'
		    GROUP BY SafraRef,
			         ChaveUnica,
					 CdContaCartao_Cadoc,
					 DiasAtraso
		;
		QUIT;

		DATA CARTEIRA_PARC_FXATRASO;

		SET CARTEIRA_PARC_FXATRASO
		    CARTEIRA_PARC_FXATRASO_&I.
		;
		RUN;

%END;
%MEND DO_LABEL;

%DO_LABEL(&SafraReferencia.,1);

CREATE TABLE TBL_CDCONTACARTAO AS

	       b.CdDimContaCartao

		FROM CARTEIRA_PARC_FXATRASO

	LEFT JOIN
	%IQ (
         SELECT B.*
		 FROM (
				SELECT a.*,
				       ROW_NUMBER() OVER(PARTITION BY a.ChaveUnica ORDER BY A.DtInicioVersaoContaCartao DESC) AS RN
				FROM
				(
						SELECT DISTINCT A.CdDimContaCartao,
		                                RIGHT(('000'+CAST(A.CdTipoProdutoCartaoStmaOrgm AS VARCHAR(2))),2) AS CdTipoProduto,

CREATE TABLE TBL_CARTEIRA_2 AS
	SELECT  A.DtProcessamento,
			COMPRESS(SUBSTRN(A.NuContrato,1,2)) AS Cd_Produto,
			(COMPRESS(SUBSTRN(A.NuContrato,1,2))||COMPRESS(A.NuCpfCnpj)) AS ChaveUnica,
			A.NuCpfCnpj,
			CASE WHEN COALESCE(b.Qt_DiasAtraso,0) = 0                 THEN 'Cart_EmDia'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 1   AND 30  THEN 'Cart_1a30'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 31  AND 60  THEN 'Cart_31a60'

			SUM(A.VlSaldoCarteira) AS Vr_Carteira

			 A.CdProduto,
			 ChaveUnica,
			 A.NuCpfCnpj,
			 FaixaAtraso
	HAVING Vr_Carteira NE 0
	ORDER BY 2,1;

%MACRO DO_ACORDOCL(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;
		CREATE TABLE CARTEIRA_PARC_ACORDOCL_&I. AS

	         SELECT &SafraRef. AS SafraRef,
                    COMPRESS((COMPRESS(a.CdProduto)||COMPRESS(a.Cd)||COMPRESS(PUT(c.ChaveSafra,6.)))) AS ChaveUnica,
			        ((INPUT(a.CdProduto, 8.) * 1000000000) + INPUT(a.NuConta, 9.)) AS CdContaCartao_Cadoc,
			        SUM(COALESCE(a.Saldo,0)) AS SaldoCarteira

			FROM ESTUDOS2.CARTAO3040_&SafraRef. AS a

			LEFT JOIN
				(
					SELECT Cd,
					       CdProduto,
                           MAX(input(SUBSTRN(COMPRESS(IniRelactCli),1,4),best12.)*100 + input(SUBSTRN(COMPRESS(IniRelactCli),6,2),best12.)) AS ChaveSafra
					FROM ESTUDOS2.CARTAO3040_&SafraRef.
					GROUP BY 1,2
				) AS c
			ON a.Cd = c.Cd AND
			   a.CdProduto = c.CdProduto
			WHERE SUBSTRN(COMPRESS(a.Contrt),3,2) = '0D'
		    GROUP BY SafraRef,
			         ChaveUnica,
					 CdContaCartao_Cadoc
		;
		QUIT;

		DATA CARTEIRA_PARC_ACORDOCL;

		SET CARTEIRA_PARC_ACORDOCL
		    CARTEIRA_PARC_ACORDOCL_&I.;
		RUN;

%MACRO DO_CARTWO(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;
		CREATE TABLE CARTEIRA_WO_&I. AS

	         SELECT &SafraRef. AS SafraRef,
                    COMPRESS((COMPRESS(a.CdProduto)||COMPRESS(a.Cd)||COMPRESS(PUT(c.ChaveSafra,6.)))) AS ChaveUnica,
			        ((INPUT(a.CdProduto, 8.) * 1000000000) + INPUT(a.NuConta, 9.)) AS CdContaCartao_Cadoc,
			        SUM(COALESCE(a.Saldo,0)) AS SaldoCarteira

			FROM ESTUDOS2.CARTAO3040_&SafraRef. AS a

			LEFT JOIN
				(
					SELECT Cd,
					       CdProduto,
                           MAX(input(SUBSTRN(COMPRESS(IniRelactCli),1,4),best12.)*100 + input(SUBSTRN(COMPRESS(IniRelactCli),6,2),best12.)) AS ChaveSafra
					FROM ESTUDOS2.CARTAO3040_&SafraRef.
					GROUP BY 1,2
				) AS c
			ON a.Cd = c.Cd AND
			   a.CdProduto = c.CdProduto
			WHERE a.ClassOp = 'HH'
		    GROUP BY SafraRef,
			         ChaveUnica,
					 CdContaCartao_Cadoc
			HAVING SaldoCarteira <> 0
		;
		QUIT;

		DATA CARTEIRA_WO;

		SET CARTEIRA_WO
		    CARTEIRA_WO_&I.;
		RUN;

%END;

	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));

		PROC SQL;

	         SELECT
				&SafraRef. AS SafraRef,
				COMPRESS((COMPRESS(a.CdProduto)||COMPRESS(a.Cd)||COMPRESS(PUT(c.ChaveSafra,6.)))) AS ChaveUnica,
				((INPUT(a.CdProduto, 8.) * 1000000000) + INPUT(a.NuConta, 9.)) AS CdContaCartao_Cadoc,

				SELECT
					Cd,
					CdProduto,
					MAX(input(SUBSTRN(COMPRESS(IniRelactCli),1,4),best12.)*100 + input(SUBSTRN(COMPRESS(IniRelactCli),6,2),best12.)) AS ChaveSafra

%MACRO DO_CARTWO(SafraInicio,COUNT);

%DO I = 1 %TO (&COUNT);

	%let year = %sysfunc(int(&SafraInicio/100));
	%let month = %sysfunc(mod(&SafraInicio,100));
	%let Dt_Ref_Inicial = %sysfunc(mdy(&month,1,&year));
	%let Dt_Ref = %sysfunc(intnx(month.1,&Dt_Ref_Inicial,%sysfunc(sum(&I,-1)),S));
	%let SafraRef = %sysfunc(sum(%eval(%sysfunc(year(&Dt_Ref))*100),%eval(%sysfunc(month(&Dt_Ref)))));

		PROC SQL;
		CREATE TABLE CARTEIRA_WO_&I. AS

	         SELECT
				&SafraRef. AS SafraRef,
				COMPRESS((COMPRESS(a.CdProduto)||COMPRESS(a.Cd)||COMPRESS(PUT(c.ChaveSafra,6.)))) AS ChaveUnica,
				((INPUT(a.CdProduto, 8.) * 1000000000) + INPUT(a.NuConta, 9.)) AS CdContaCartao_Cadoc,
				SUM(COALESCE(a.Saldo,0)) AS SaldoCarteira
			FROM ESTUDOS2.CARTAO3040_&SafraRef. AS a

			LEFT JOIN (
				SELECT
					Cd,
					CdProduto,
					MAX(input(SUBSTRN(COMPRESS(IniRelactCli),1,4),best12.)*100 + input(SUBSTRN(COMPRESS(IniRelactCli),6,2),best12.)) AS ChaveSafra
				FROM ESTUDOS2.CARTAO3040_&SafraRef.
				GROUP BY 1,2
			) AS c
			ON a.Cd = c.Cd
				AND a.CdProduto = c.CdProduto
			WHERE a.ClassOp = 'HH'
		    GROUP BY SafraRef,
			         ChaveUnica,
					 CdContaCartao_Cadoc
			HAVING SaldoCarteira <> 0
		;
		QUIT;

		DATA CARTEIRA_WO;

		SET CARTEIRA_WO
		    CARTEIRA_WO_&I.;
		RUN;

%END;
%MEND DO_CARTWO;

				 WHEN SUBSTRN(A.NuContrato,1,2) IN ('16')           THEN 'ELO'
				 END AS NmBandeiraCartao,

			CASE WHEN COALESCE(b.Qt_DiasAtraso,0) = 0                 THEN 'Cart_EmDia'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 1   AND 14  THEN 'Cart_1a14'
                 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 15  AND 30  THEN 'Cart_15a30'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 31  AND 60  THEN 'Cart_31a60'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 61  AND 90  THEN 'Cart_61a90'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 91  AND 120 THEN 'Cart_91a120'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 121 AND 150 THEN 'Cart_121a150'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) BETWEEN 151 AND 180 THEN 'Cart_151a180'
				 WHEN COALESCE(b.Qt_DiasAtraso,0) >= 181              THEN 'Cart_Acima180'
				 END AS FaixaAtraso,
			A.VlSaldoCarteira
	FROM TBL_CARTEIRA_1 AS a

	LEFT JOIN
	(
		SELECT DISTINCT t.DtProcessamento,
	                    t.NuCpfCnpj,
		                MAX(t.QtDiaAtraso) AS Qt_DiasAtraso
		FROM TBL_CARTEIRA_1 AS t
	    WHERE t.CdClassificacaoOperacao <> 'HH'
		GROUP BY 1,2
	) AS b
	ON a.DtProcessamento = b.DtProcessamento AND
	   a.NuCpfCnpj = b.NuCpfCnpj

	WHERE a.CdClassificacaoOperacao <> 'HH'
) AS t
GROUP BY 1,2,3
;
QUIT;
