%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
CREATE TABLE extracao_ptaiqcont_flex AS
SELECT * FROM connection to dbcon
(
WITH TbContratoValido AS
(
	SELECT
		 ContratoFinanceiro.NuContrato
		,max(ContratoFinanceiro.SqContratoFinanceiro) AS SqContratoFinanceiro
	FROM STGTXT_FLEX.TbContratoFinanceiro ContratoFinanceiro

		LEFT JOIN STGTXT_FLEX.TbContrato Contrato
			ON  ContratoFinanceiro.NuContrato = Contrato.NuContrato

		LEFT JOIN DBA.TbProposta Proposta
			ON  Contrato.NuContrato = Proposta.NuContratoGestao

	WHERE 1=1
		AND Contrato.NuContrato > 0
		AND DATEDIFF(month,Contrato.DtContrato,ContratoFinanceiro.DtContratoFinanceiro) = 0

	GROUP BY ContratoFinanceiro.NuContrato
)
,TbPropostaFavorecido AS
(
	SELECT
		 Proposta.NuContratoGestao
		,SUM(CASE WHEN PropostaFavorecido.CdTipoPropostaFavorecido	= 1						THEN PropostaFavorecido.VrFavorecido	ELSE	0	END)	VrCliente
		,SUM(CASE WHEN PropostaFavorecido.CdTipoPropostaFavorecido	=	2					THEN PropostaFavorecido.VrFavorecido	ELSE	0	END)	VrRefin
		,SUM(CASE WHEN PropostaFavorecido.CdTipoPropostaFavorecido	=	3					THEN PropostaFavorecido.VrFavorecido	ELSE	0	END)	VrCompra
		,SUM(CASE WHEN PropostaFavorecido.CdTipoPropostaFavorecido	= 5						THEN PropostaFavorecido.VrFavorecido	ELSE	0	END)	VrOutroDesdobro
	FROM DBA.TbProposta Proposta

		INNER JOIN TbContratoValido ContratoValido
			ON  Proposta.NuContratoGestao = ContratoValido.NuContrato

		LEFT JOIN DBCRED.TbPropostaFavorecido PropostaFavorecido
			ON Proposta.NuProposta = PropostaFavorecido.NuProposta

	GROUP BY
		 Proposta.NuContratoGestao
)

	SELECT
		 year(DATE(Contrato.DtContrato))*100+month(DATE(Contrato.DtContrato)) NuAnoMes
		,Contrato.DtContrato
		,ContratoFinanceiro.DtContratoFinanceiro
		,Contrato.DtLiquidacaoEfetiva
		,Contrato.DtCancelamento
		,ContratoFinanceiro.DtProcessamento
		,ContratoFinanceiro.DtInicioVigencia
		,ContratoFinanceiro.DtFimVigencia
		,ContratoFinanceiro.DtPrimeiroVencimento
		,ContratoFinanceiro.DtUltimoVencimento

		,Proposta.NuProposta
		,Proposta.DtEntradaProposta
		,ContratoFinanceiro.NuContrato
		,ContratoFinanceiro.SqContratoFinanceiro
		,ContratoFinanceiro.FlAtivo
		,ContratoFinanceiro.CdMotivoContratoFinanceiro
		,ContratoFinanceiro.CdTipoCobranca
		,TipoCobranca.NmTipoCobranca
		,ContratoFinanceiro.NuContratoLegado

		,Colaborador.CdColaborador
		,CASE WHEN Contrato.CdOperador <> Contrato.CdPromotor then Contrato.CdPromotor end CdColaboradorPromotor
		,CASE WHEN Contrato.CdParceiroComercial = 418254 THEN Colaborador.CdFilial ELSE Contrato.CdFilial END AS CdFilial
		,CASE WHEN Contrato.CdParceiroComercial = 418254 THEN FilialColaborador.DsFilial ELSE Filial.DsFilial END AS DsFilial

		,Contrato.CdParceiroComercial
		,Proposta.CdPessoa CdParceiroComercialCoban

		,Contrato.CdPessoa
		,Pessoa.NmPessoa
		,Pessoa.TpPessoa
		,Pessoa.NuCpfCnpj
		,EnderecoTbPessoaSobAnalise.CdUf as CdUfPessoa
		,EnderecoTbPessoaSobAnalise.NmCidade as NmCidadePessoa
		,PessoaFisicaSobAnalise.QtAnosIdade as QtAnosIdadePessoa

		,CASE
			WHEN ContratoFinanceiro.QtParcelas >  60 THEN '60+'
			WHEN ContratoFinanceiro.QtParcelas >  48 THEN '60'
			WHEN ContratoFinanceiro.QtParcelas >  36 THEN '48'
			WHEN ContratoFinanceiro.QtParcelas >  24 THEN '36'
			WHEN ContratoFinanceiro.QtParcelas >  18 THEN '24'
			WHEN ContratoFinanceiro.QtParcelas >  12 THEN '18'
			WHEN ContratoFinanceiro.QtParcelas >   0 THEN '12'
			END FxParcela

		,coalesce(Contrato.PcEntradaMolicar,0) PcEntradaMolicar
		,CASE
            WHEN Contrato.PcEntradaMolicar > 50 THEN '07 - Acima 50%'
            WHEN Contrato.PcEntradaMolicar > 40 THEN '06 - Ate 50%'
            WHEN Contrato.PcEntradaMolicar > 30 THEN '05 - Ate 40%'
            WHEN Contrato.PcEntradaMolicar > 20 THEN '04 - Ate 30%'
            WHEN Contrato.PcEntradaMolicar > 10 THEN '03 - Ate 20%'
            WHEN Contrato.PcEntradaMolicar > 0 THEN  '02 - Ate 10%'
            ELSE '01 - Sem entrada'
            END FxEntrada

		,case when PropostaFinanceiro.TpOrigemTaxa='A' then 1 else 0 end FlTxAlterada
		,Contrato.CdProduto*1000000 + Contrato.CdModalidadeProduto*1000 CdGrupoSubProduto
		,Proposta.CdSubProduto
		,SubProduto.DsSubProduto
		,CASE WHEN (
							Contrato.CdOperador = 2826
						OR  ContratoFinanceiro.SqContratoFinanceiro > 1
						OR DATEDIFF(month,DtContrato,DtCancelamento) = 0
					) THEN 'N' ELSE 'S' END StProducao
		,Contrato.CdSituacaoContrato
		,1 CdTipoMercado
		,ContratoFinanceiro.CdTabelaFinanciamento
		,ContratoFinanceiro.VrIof
		,ContratoFinanceiro.VrRetencao
		,ContratoFinanceiro.VrTarifaFinanciadaTotal
		,ContratoFinanceiro.VrPrestacao

		,ContratoFinanceiro.VrContrato
		,ContratoFinanceiro.VrEntrada
		,(ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada) * (ContratoFinanceiro.VrEntrada/nullif(ContratoFinanceiro.VrContrato,0)) AS VpEntrada

		,Proposta.FlRefinConsignado
		,Proposta.FlCompraDivida
		,CASE WHEN Proposta.FlRefinConsignado = 'S' THEN PropostaFavorecido.VrCliente ELSE ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada END AS VrProducaoLiquida
		,PropostaFavorecido.VrCliente
		,PropostaFavorecido.VrRefin
		,PropostaFavorecido.VrCompra
		,PropostaFavorecido.VrOutroDesdobro

		,ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada VrProducao
		,ContratoFinanceiro.VrComissaoParcelada
		,ContratoFinanceiro.VrComissaoTotal

		,ContratoFinanceiro.VrFinanciamento
		,ContratoCusto.VrTarifaCadastro
		,ContratoCusto.VrTarifaAvaliacaoBem

		,ContratoFinanceiro.TxGerencial
		,(ContratoFinanceiro.TxGerencial/100) * (ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada) VpTxGerencial

		,ContratoFinanceiro.TxJuros
		,(ContratoFinanceiro.TxJuros/100) * (ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada) VpTxJuros

		,case when PessoaSobAnalise.NuScoreFim = 99 then 1 else PessoaSobAnalise.NuScoreFim end as NuScoreFim
		,(case when PessoaSobAnalise.NuScoreFim = 99 then 1 else PessoaSobAnalise.NuScoreFim end) * (ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada) as VpScore

		,ContratoFinanceiro.QtParcelas
		,ContratoFinanceiro.QtParcelas*(ContratoFinanceiro.VrContrato-ContratoFinanceiro.VrEntrada) VpPrazo

	FROM STGTXT_FLEX.TbContratoFinanceiro ContratoFinanceiro

		INNER JOIN TbContratoValido ContratoValido
			ON  ContratoFinanceiro.NuContrato = ContratoValido.NuContrato

		LEFT JOIN TbContratoValido SqContratoValido
			ON  ContratoFinanceiro.NuContrato = SqContratoValido.NuContrato
			AND ContratoFinanceiro.SqContratoFinanceiro = SqContratoValido.SqContratoFinanceiro

		LEFT JOIN
		(
			SELECT
				 NuContrato
				,SqContratoFinanceiro
				,sum(CASE WHEN CdGrupoCusto =  5 THEN VrCusto END) VrTarifaCadastro
				,sum(CASE WHEN CdGrupoCusto = 10 THEN VrCusto END) VrTarifaAvaliacaoBem
			FROM CDCSG.TbContratoCusto
			WHERE CdGrupoCusto IN (5,10)
			GROUP BY
				 NuContrato
				,SqContratoFinanceiro
		) ContratoCusto

			ON  ContratoFinanceiro.NuContrato = ContratoCusto.NuContrato
			AND	ContratoFinanceiro.SqContratoFinanceiro = ContratoCusto.SqContratoFinanceiro

		LEFT JOIN STGTXT_FLEX.TbContrato Contrato
			ON  ContratoFinanceiro.NuContrato = Contrato.NuContrato

			LEFT JOIN DBA.TbFilial Filial
				on  Contrato.CdFilial = Filial.CdFilial

			LEFT JOIN STGTXT_FLEX.TbPessoa Pessoa
				ON  Contrato.CdPessoa = Pessoa.CdPessoa

			LEFT JOIN DBA.TbColaborador Colaborador
				ON  Contrato.CdOperador = Colaborador.CdOperador
				AND Colaborador.CdColaborador NOT IN (503,9777,11046,11210,18512,57936)
				AND Colaborador.CdOperador <> 0

				LEFT JOIN DBA.TbFilial FilialColaborador
					ON  Colaborador.CdFilial = FilialColaborador.CdFilial

			LEFT JOIN DBA.TbParceiroComercial ParceiroComercial
				ON  Contrato.CdParceiroComercial = ParceiroComercial.CdParceiroComercial

		LEFT JOIN DBA.TbProposta Proposta
			ON  Contrato.NuContrato = Proposta.NuContratoGestao

			LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
				ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
				AND PessoaSobAnalise.CdPapel = 1

			LEFT JOIN DBA.TbPessoaFisicaSobAnalise PessoaFisicaSobAnalise
				ON  PessoaSobAnalise.CdPessoaSobAnalise = PessoaFisicaSobAnalise.CdPessoaSobAnalise

			LEFT JOIN DBCRED.TbEnderecoTbPessoaSobAnalise EnderecoTbPessoaSobAnalise
				ON  PessoaSobAnalise.CdPessoaSobAnalise = EnderecoTbPessoaSobAnalise.CdPessoaSobAnalise
				AND EnderecoTbPessoaSobAnalise.CdTipoEndereco = 1

			LEFT JOIN DBA.TbSubProduto SubProduto
				ON  Proposta.CdSubProduto = SubProduto.CdSubProduto

				LEFT JOIN DBCRED.TbPropostaFinanceiro PropostaFinanceiro
					on Proposta.NuProposta = PropostaFinanceiro.NuProposta

			LEFT JOIN DBCOR.TbTipoCobranca TipoCobranca
					ON  PropostaFinanceiro.CdTipoCobranca = TipoCobranca.CdTipoCobranca

		LEFT JOIN TbPropostaFavorecido PropostaFavorecido
			ON  ContratoFinanceiro.NuContrato = PropostaFavorecido.NuContratoGestao

)
;RUN;

create table TbFilial as
select * from connection to dbcon

	SELECT
		 CdFilial
		,DsFilial

	FROM DBA.TbFilial
)

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbDimHierarquiaComercial as
select * from connection to dbcon
(
	with TbDeParaFilial as
	(

		SELECT
			 DeParaFilial.CdAreaEstruturaComercial
			,DeParaFilial.CdFilial
			,Filial.NmFilial
			,CASE
				WHEN Filial.NmFilial LIKE 'CP SOL%' THEN 5
				WHEN Filial.NmFilial LIKE 'CP CONS%' THEN 4
				WHEN DeParaFilial.CdFilial IN (275,276) THEN 3
				WHEN Filial.NmFilial LIKE 'CP %' THEN 2
				WHEN Filial.NmFilial NOT LIKE 'CP ' THEN 1
			 ELSE 0 END AS CdCanalEstruturaComercial

		FROM DWBI_DTWH.TbDeParaFilial DeParaFilial

			LEFT JOIN DWBI_DTWH.TbFilial Filial
				ON  DeParaFilial.CdFilial = Filial.CdFilialSistemaOrigem
				AND Filial.CdSistemaOrigem = 209

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial Area
				ON  DeParaFilial.CdAreaEstruturaComercial = Area.CdAreaEstruturaCmrclStmaOrgm

	)

	SELECT
		 CAST (Pessoa.CdPessoaOrigem AS INTEGER) CdOperador
		,TempoData.DtTempo
		,DeParaFilial.CdFilial

	FROM DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoa

		LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
			ON  PessoaRelacaoPessoa.CdPessoa = Pessoa.CdPessoa

		INNER JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
			ON  PessoaRelacaoPessoa.CdPessoaRelacao = HierarquiaComercial.CdColaborador

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
				ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

			LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
				ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

				INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
					ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
					AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador
					AND DeParaHierarquia.CdCanalEstruturaComercial in (2,4)

				INNER JOIN TbDeParaFilial DeParaFilial
					ON  DeParaHierarquia.CdAreaEstruturaComercial = DeParaFilial.CdAreaEstruturaComercial
					AND DeParaHierarquia.CdCanalEstruturaComercial = DeParaFilial.CdCanalEstruturaComercial

			INNER JOIN DWBI_DIME.TbDimTempoData TempoData
				ON  TempoData.DtTempo BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
				AND TempoData.DtTempo > '20000101'
				AND TempoData.DtTempo < today()

	WHERE PessoaRelacaoPessoa.CdTipoRelacaoPessoa = 7
