PROC SQL;
   CREATE TABLE WORK.BASE_CUBO_SOLAR_CONSO AS
   SELECT t1.Referencia,
          
            (MAX(t2.Du)) FORMAT=BEST12. AS MAX_of_Du,
          t1.INCONSISTENCIA,
          
            (case
            when T1.CD_ASSESSORIA in (1592, 1593) then "Paschoalotto"
            when T1.CD_ASSESSORIA in (1648, 1649) then "Toledo"
            when T1.CD_ASSESSORIA in (2031, 2033) then "Bellinati"
            else ""
            end) AS ASSESSORIA,
          t1.FxAtraso,
          
            (COUNT(t1.NUCONTRATO)) AS QTD_CONTRATOS,
          
            (SUM(t1.RISCO_ENVIO)) FORMAT=BEST9. AS RISCO_ENVIO,
          
            (SUM(t1.RISCO_PERF)) FORMAT=BEST9. AS RISCO_PERF,
          
            ((SUM(t1.RISCO_PERF)) / (SUM(t1.RISCO_ENVIO))) AS RESULTADO,
          
            (CASE
            WHEN t1.FxAtraso = '00. Até 5 dias' THEN 1
            WHEN t1.FxAtraso = '01. 5 a 15' THEN 1
            WHEN t1.FxAtraso = '02. 16 a 30' THEN 0.25
            WHEN t1.FxAtraso = '03. 31 a 60' THEN 0.30
            WHEN t1.FxAtraso = '04. 61 a 120' THEN 0.45

            WHEN t1.FxAtraso = '05. 121 a 180' THEN 0.25
            WHEN t1.FxAtraso = '06. 181 a 360' THEN 0.30
            WHEN t1.FxAtraso = '07. Acima de 360' THEN 0.45
            ELSE 1
            END) AS PESO_FAIXAS,
          
            (((SUM(t1.RISCO_PERF)) / (SUM(t1.RISCO_ENVIO))) * (CASE
            WHEN t1.FxAtraso = '00. Até 5 dias' THEN 1
            WHEN t1.FxAtraso = '01. 5 a 15' THEN 1
            WHEN t1.FxAtraso = '02. 16 a 30' THEN 0.25
            WHEN t1.FxAtraso = '03. 31 a 60' THEN 0.30
            WHEN t1.FxAtraso = '04. 61 a 120' THEN 0.45

            WHEN t1.FxAtraso = '05. 121 a 180' THEN 0.25
            WHEN t1.FxAtraso = '06. 181 a 360' THEN 0.30
            WHEN t1.FxAtraso = '07. Acima de 360' THEN 0.45
            ELSE 1
            END)) AS RESULTADO_PESO,
          
            (CASE
            WHEN t1.FxAtraso IN ( '02. 16 a 30','03. 31 a 60','04. 61 a 120') THEN 'SOLAR1'
            WHEN t1.FxAtraso IN ( '05. 121 a 180','06. 181 a 360','05. Acima de 360','07. Acima de 360') THEN  'SOLAR2'
            ELSE t1.FxAtraso
            END) AS FAIXA_ACUM
      FROM WORK.BASE_CUBO_SOLAR_BI t1
           LEFT JOIN WORK.BASE_DU t2 ON (t1.DT_RECEBIMENTO = t2.Data)
      WHERE t1.INCONSISTENCIA = 'N'
      GROUP BY t1.Referencia,
               t1.INCONSISTENCIA,
               (CALCULATED ASSESSORIA),
               t1.FxAtraso,
               (CALCULATED PESO_FAIXAS),
               (CALCULATED FAIXA_ACUM);
QUIT;

   CREATE TABLE WORK.BASE_CART_PERF_CL AS
   SELECT t2.DATA_BASE,
          t2.FX_ATRASO,

          t1.ULT_DU,
          t2.ASSESSORIA,
          
            (SUM(t2.SALDO_ENVIADO)) FORMAT=BESTX15.2 AS SUM_of_SALDO_ENVIADO,
          
            (SUM(t3.RECEBIDO_PARCELADO)) FORMAT=BESTX15.2 AS SUM_of_RECEBIDO_PARCELADO,

            (case
             when t1.FX_ATRASO ='75 A 120' then 0.5
             when t1.FX_ATRASO ='121 A 180' then 0.3
             when t1.FX_ATRASO ='181 A 360'  then 0.2
             when t1.FX_ATRASO ='361 A 720' then 0.4
             when t1.FX_ATRASO ='ACIMA DE 720'  then 0.6
            end) AS PESOS
      FROM WORK.BASE_ULTI_DU t1
           LEFT JOIN WORK.PERF_CL t2 ON (t1.DATA_BASE = t2.DATA_BASE) AND (t1.ULT_DU = t2.DU_DT_REF) AND (t1.FX_ATRASO
          = t2.FX_ATRASO) AND (t1.ASSESSORIA = t2.ASSESSORIA)
           LEFT JOIN WORK.PERF_CL t3 ON (t1.DATA_BASE = t3.DATA_BASE) AND (t1.ULT_DU = t3.DU_DT_REF) AND (t1.FX_ATRASO
          = t3.FX_ATRASO) AND (t1.ASSESSORIA = t3.ASSESSORIA)
      GROUP BY t2.DATA_BASE,
               t2.FX_ATRASO,
               t3.CL_STATUS,

PROC SQL;
   CREATE TABLE WORK.TAB_CP_RESULTADOS AS
   SELECT t1.SAFRA,
          t1.Assessoria,
          t1.Du,
          t1.FASE,
          t1.FX_ATRASO_NOVA,
          
            (CASE
            WHEN t1.FX_ATRASO_NOVA = '005 a 015' THEN '005 a 015'
            WHEN t1.FX_ATRASO_NOVA IN('016 a 030','031 a 060','061 a 120') THEN 'CP1'
            WHEN t1.FX_ATRASO_NOVA IN('121 a 180','181 a 360') THEN 'CP2'
            WHEN t1.FX_ATRASO_NOVA = '361 a 999' THEN 'WO'
            ELSE 'VALIDAR' END
            ) AS FAIXA_CONS,
          
            (SUM(t1.SALDO_RISCO)) AS SALDO_RISCO_T,
          
            (SUM(t1.RISCO_PAGO)) AS RISCO_PAGO_T,
          t2.MAIOR_DU,
          
            ((SUM(t1.RISCO_PAGO))/(SUM(t1.SALDO_RISCO))) AS RESULTADO,
          
            (CASE
            WHEN t1.FX_ATRASO_NOVA = '005 a 015' THEN 1

            WHEN t1.FX_ATRASO_NOVA = '016 a 030' THEN 0.25
            WHEN t1.FX_ATRASO_NOVA = '031 a 060' THEN 0.35
            WHEN t1.FX_ATRASO_NOVA = '061 a 120' THEN 0.40

            WHEN t1.FX_ATRASO_NOVA = '121 a 180' THEN 0.45
            WHEN t1.FX_ATRASO_NOVA = '181 a 360' THEN 0.55

            WHEN t1.FX_ATRASO_NOVA = '361 a 999' THEN 1

            END) AS PESOS,
          
            (((SUM(t1.RISCO_PAGO))/(SUM(t1.SALDO_RISCO))) * (CASE
            WHEN t1.FX_ATRASO_NOVA = '005 a 015' THEN 1

            WHEN t1.FX_ATRASO_NOVA = '016 a 030' THEN 0.25
            WHEN t1.FX_ATRASO_NOVA = '031 a 060' THEN 0.35
            WHEN t1.FX_ATRASO_NOVA = '061 a 120' THEN 0.40

            WHEN t1.FX_ATRASO_NOVA = '121 a 180' THEN 0.45
            WHEN t1.FX_ATRASO_NOVA = '181 a 360' THEN 0.55

            WHEN t1.FX_ATRASO_NOVA = '361 a 999' THEN 1

            END)) AS RESULADO_C_PESO
      FROM WORK.BASE_MIS_CP t1
           LEFT JOIN WORK.TABELA_MAX_DU_CP t2 ON (t1.SAFRA = t2.SAFRA) AND (t1.FASE = t2.FASE) AND (t1.Du =
          t2.MAIOR_DU) AND (t1.FX_ATRASO_NOVA = t2.FX_ATRASO_NOVA)
      WHERE t2.MAIOR_DU NOT IS MISSING
      GROUP BY t1.SAFRA,
               t1.Assessoria,
               t1.Du,
               t1.FASE,
               t1.FX_ATRASO_NOVA,
               (CALCULATED FAIXA_CONS),
               t2.MAIOR_DU,
               (CALCULATED PESOS);
QUIT;

      FROM WORK.BASE_CUBO_SOLAR_CONSO t1
