PROC SQL;
connect to bigquery as perm (project='bv-sadi-dtm' credfile='/projetos/acessos/bq/di/sadi-dtm.json'
      schema='bv-pdbd-prd' READ_MODE=storage);
CREATE TABLE BASES_NOTIFIC_RETORNO AS
select * from connection to perm(

SELECT DISTINCT
cc.CdContratoAjuizamento
, cc.NuContrato
, cc.SqContratoFinanceiro
, cc.NuParcela
, Ev.CdStatusRetornoAjuizamento
, Ev.dteventoajuizamento
, Ev.cdtipoeventoajuizamento
, St.DsStatusRetornoAjuizamento
, St.Dtinclusao
, St.Dtalteracaoanalitico
, St.dtinclusaoanalitico
, St.dtalteracao
, aa.dstipoeventoajuizamento
,case
            when EV.cdempresaprestadoraservico = 1632 then 'MRS'
            when EV.cdempresaprestadoraservico = 1633 then 'LEAF'
          end as tp_empresa
FROM bv-pdbd-prd.trusted_dbsrcgmurc.tbcontratoajuizamento cc
INNER JOIN  bv-pdbd-prd.trusted_dbsrcgmurc.tbeventoajuizamento Ev
ON  cc.cdcontratoajuizamento = Ev.cdcontratoajuizamento
INNER JOIN bv-pdbd-prd.trusted_dbsrcgmurc.tbstatusretornoajuizamento St ON Ev.CdStatusRetornoAjuizamento=St.CdStatusRetornoAjuizamento
INNER JOIN bv-pdbd-prd.trusted_dbsrcgmurc.tbtipoeventoajuizamento aa  ON Ev.cdtipoeventoajuizamento=aa.cdtipoeventoajuizamento
WHERE Ev.QtTentativaEnvioNotificacao=1
AND  Ev.CdStatusRetornoAjuizamento NOT IN (8)
AND  qttentativaenvionotificacao = (SELECT MAX(qttentativaenvionotificacao)
                                FROM bv-pdbd-prd.trusted_dbsrcgmurc.tbeventoajuizamento e2
                                WHERE e2.CdStatusRetornoAjuizamento NOT IN (8)
                                AND e2.QtTentativaEnvioNotificacao=1

                                AND   e2.cdcontratoajuizamento  = cc.cdcontratoajuizamento

                                GROUP BY e2.cdcontratoajuizamento )
);
disconnect from perm;
QUIT;

SELECT A.*
, B.num_processo as NUM_PROCESSO_AJU

AND A.PARCELA = B.Parcela

CREATE TABLE BASE_2_&SAFRA. AS
SELECT datepart(A.DTPROCESSAMENTO) as dt_referencia FORMAT = date9.
, A.*
, B.dtretomada AS dtretomada FORMAT = date9.
, B.Tipo_apreensao as tp_retomada
, P.DtRecebimento AS dtpagamento FORMAT = date9.
, P.VR_PRINCIAPL_DEVIDO AS VR_PRINCIPAL_DEVIDO
, P.VR_PRINCIPAL_BAIXADO AS VR_PRINCIPAL_BAIXADO
, datepart(Q.Dt_Liquidacao) AS dtquitacao FORMAT = date9.
, datepart(R.dt_baixa) AS dtreneg FORMAT = date9.
FROM BASE_1_&SAFRA. A
LEFT JOIN BASE_BNDU_2025 B ON A.NUCONTRATO = B.NUCONTRATO AND A.SQCONTRATOFINANCEIRO = B.SQCONTRATOFINANCEIRO AND A.PARCELA = B.PARCELA AND B.dtretomada >= datepart(A.DTPROCESSAMENTO)
LEFT JOIN BASE_PAGAMENTOS_2025 P ON A.NUCONTRATO = P.NUCONTRATO AND A.SQCONTRATOFINANCEIRO = P.SQCONTRATOFINANCEIRO AND A.PARCELA = P.NU_PARCELA AND P.DtRecebimento >= datepart(A.DTPROCESSAMENTO)  
LEFT JOIN BASE_LIQUIDACAO_2025 Q ON A.NUCONTRATO = Q.NUCONTRATO AND A.SQCONTRATOFINANCEIRO = Q.SQCONTRATOFINANCEIRO AND A.PARCELA = Q.PARCELA AND datepart(Q.Dt_Liquidacao) >= datepart(A.DTPROCESSAMENTO)  
LEFT JOIN BASE_RENEG_2025 R ON A.NUCONTRATO = R.NUCONTRATO AND A.SQCONTRATOFINANCEIRO = R.SQCONTRATOFINANCEIRO AND A.PARCELA = R.PARCELA AND datepart(R.dt_baixa) >= datepart(A.DTPROCESSAMENTO)
