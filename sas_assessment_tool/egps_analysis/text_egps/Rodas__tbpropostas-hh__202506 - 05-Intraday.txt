%CRIAVARDB;
proc sql;
%CONECTADB(server=PTA);
create table TbFilial as
select * from connection to dbcon
(
	SELECT
		 CdFilial
		,DsFilial
	FROM DBA.TbFilial
)
;run;

%CRIAVARDB;
proc sql;
%CONECTADB(server=BI);
create table TbDimHierarquiaComercial as
select * from connection to dbcon
(
	with TbDeParaFilial as
	(
		SELECT
			 DeParaFilial.CdAreaEstruturaComercial
			,DeParaFilial.CdFilial
			,Filial.NmFilial
			,CASE
				WHEN Filial.NmFilial LIKE 'CP SOL%' THEN 5
				WHEN Filial.NmFilial LIKE 'CP CONS%' THEN 4
				WHEN DeParaFilial.CdFilial IN (275,276) THEN 3
				WHEN Filial.NmFilial LIKE 'CP %' THEN 2
				WHEN Filial.NmFilial NOT LIKE 'CP ' THEN 1
			 ELSE 0 END AS CdCanalEstruturaComercial

		FROM DWBI_DTWH.TbDeParaFilial DeParaFilial

			LEFT JOIN DWBI_DTWH.TbFilial Filial
				ON  DeParaFilial.CdFilial = Filial.CdFilialSistemaOrigem
				AND Filial.CdSistemaOrigem = 209

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial Area
				ON  DeParaFilial.CdAreaEstruturaComercial = Area.CdAreaEstruturaCmrclStmaOrgm
	)
	SELECT
		 CAST (Pessoa.CdPessoaOrigem AS INTEGER) CdOperador
		,TempoData.DtTempo
		,DeParaFilial.CdFilial
		,NmFilial
	FROM DWBI_DTWH.TbPessoaRelacaoPessoa PessoaRelacaoPessoa

		LEFT JOIN DWBI_DTWH.TbPessoa Pessoa
			ON  PessoaRelacaoPessoa.CdPessoa = Pessoa.CdPessoa

		INNER JOIN DWBI_DIME.TbDimHierarquiaComercial HierarquiaComercial
			ON  PessoaRelacaoPessoa.CdPessoaRelacao = HierarquiaComercial.CdColaborador

			LEFT JOIN DWBI_DTWH.TbAreaEstruturaComercial AreaEstruturaComercial
				ON  HierarquiaComercial.CdAreaEstruturaComercial = AreaEstruturaComercial.CdAreaEstruturaComercial

			LEFT JOIN DWBI_DTWH.TbCargoColaborador CargoColaborador
				ON  HierarquiaComercial.CdCargoColaborador = CargoColaborador.CdCargoColaborador

				INNER JOIN DWBI_DTWH.TbDeParaHierarquia DeParaHierarquia
					ON  AreaEstruturaComercial.CdAreaEstruturaCmrclStmaOrgm = DeParaHierarquia.CdAreaEstruturaComercial
					AND CargoColaborador.CdCargoColaboradorStmaOrgm = DeParaHierarquia.CdCargoColaborador
					AND DeParaHierarquia.CdCanalEstruturaComercial in (2,4,5)

				INNER JOIN TbDeParaFilial DeParaFilial
					ON  DeParaHierarquia.CdAreaEstruturaComercial = DeParaFilial.CdAreaEstruturaComercial
					AND DeParaHierarquia.CdCanalEstruturaComercial = DeParaFilial.CdCanalEstruturaComercial

			INNER JOIN DWBI_DIME.TbDimTempoData TempoData
				ON  TempoData.DtTempo BETWEEN HierarquiaComercial.DtInicioVigencia AND HierarquiaComercial.DtFimVigencia
				AND TempoData.DtTempo > '20000101'
				AND TempoData.DtTempo < today()

	WHERE PessoaRelacaoPessoa.CdTipoRelacaoPessoa = 7
	  AND today() BETWEEN PessoaRelacaoPessoa.DtInicioVigencia AND PessoaRelacaoPessoa.DtFimVigencia
)
;run;

proc sql;
create table TbGrVirtual as
select
	 Colaborador.CdColaborador as CdOperador
	,TempoData.DtTempo
	,GrVirtualVigencia.CdFilial

from bases.TbColaborador Colaborador

	left join bases.TbTempoData TempoData
		on 1=1

	inner join basegrc.TbGrVirtualVigencia GrVirtualVigencia
		on  Colaborador.CdColaborador = GrVirtualVigencia.CdColaborador
		and TempoData.DtTempo between GrVirtualVigencia.DtInicioVigencia and GrVirtualVigencia.DtFimVigencia

order by
	 Colaborador.CdColaborador
	,TempoData.DtTempo
;run;

data TbDimHierarquiaComercial;
set
	TbDimHierarquiaComercial
	TbGrVirtual
;run;

proc sql;
create table TbDimHierarquiaComercial as
select
	 DimHierarquiaComercial.CdOperador
	,DimHierarquiaComercial.DtTempo
	,DimHierarquiaComercial.CdFilial
	,Filial.DsFilial
from TbDimHierarquiaComercial DimHierarquiaComercial

	left join TbFilial Filial
		on  DimHierarquiaComercial.CdFilial = Filial.CdFilial

order by 1,2
;run;

proc sort data=TbDimHierarquiaComercial nodupkey;by DtTempo CdOperador;run;

%CRIAVARDB;
PROC SQL;
%CONECTADB(server=PTA);
CREATE TABLE TbProposta AS
SELECT * FROM connection to dbcon
(
WITH

TbColaborador AS
(
	SELECT DISTINCT
		 Colaborador.CdOperador
		,first_value(Colaborador.CdColaborador) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdColaborador
		,first_value(Colaborador.CdFilial) over (partition BY Colaborador.CdOperador ORDER BY CASE WHEN (Colaborador.NmColaborador LIKE 'DUP%' OR Colaborador.DsLogin LIKE 'DUP%') THEN 1 ELSE 0 END) AS CdFilial

	FROM DBA.TbColaborador Colaborador
	WHERE Colaborador.CdColaborador > 0
	AND Colaborador.CdOperador > 0
)

,TbProposta AS
(
			SELECT
				year(DATE(Proposta.DtEntradaProposta))*100+month(DATE(Proposta.DtEntradaProposta)) NuAnoMes
				,DATE(Proposta.DtEntradaProposta) AS DtEntradaProposta
				,DATE(Proposta.DtStatusProposta) AS DtStatusProposta
				,Proposta.StProposta
                ,SubProduto.CdProduto*1000000 + SubProduto.CdModalidadeProduto*1000 + coalesce(SubProduto.CdTipoBem,0) CdGrupoSubProduto
				,Colaborador.CdColaborador
				,Proposta.NuProposta
				,Proposta.NuContratoGestao
				,CASE WHEN Colaborador.CdColaborador <> ColaboradorPromotor.CdColaborador then ColaboradorPromotor.CdColaborador end as CdColaboradorPromotor
				,CASE WHEN SubProduto.CdProduto = 11 THEN Colaborador.CdFilial ELSE Regiao.CdFilial END AS CdFilial
				,Proposta.CdLoja CdParceiroComercial
				,Proposta.CdPessoa CdParceiroComercialCoban

				,CASE
					WHEN Garantia.FlZeroKm = 'S' then 'ZeroKM'
					WHEN Garantia.QtAnosIdade >  10 THEN 'Acima de 10 anos'
					WHEN Garantia.QtAnosIdade >   7 THEN '8 a 10 anos'
					WHEN Garantia.QtAnosIdade >   2 THEN '3 a 7 anos'
					WHEN Garantia.QtAnosIdade >  -1 THEN '0 a 2 anos'
					END FxIdade

				,CASE
					WHEN Proposta.QtParcelas >  60 THEN '60+'
					WHEN Proposta.QtParcelas >  48 THEN '60'
					WHEN Proposta.QtParcelas >  36 THEN '48'
					WHEN Proposta.QtParcelas >  24 THEN '36'
					WHEN Proposta.QtParcelas >  18 THEN '24'
					WHEN Proposta.QtParcelas >  12 THEN '18'
					WHEN Proposta.QtParcelas >   0 THEN '12'
					END AS FxParcela

		   		,CASE
		            WHEN Contrato.PcEntradaMolicar > 50 THEN '07 - Acima 50%'
		            WHEN Contrato.PcEntradaMolicar > 40 THEN '06 - Ate 50%'
		            WHEN Contrato.PcEntradaMolicar > 30 THEN '05 - Ate 40%'
		            WHEN Contrato.PcEntradaMolicar > 20 THEN '04 - Ate 30%'
		            WHEN Contrato.PcEntradaMolicar > 10 THEN '03 - Ate 20%'
		            WHEN Contrato.PcEntradaMolicar > 0 THEN  '02 - Ate 10%'
		            ELSE '01 - Sem entrada'
		         END FxEntrada
				,PessoaSobAnalise.TpPessoa
				,Proposta.VrProposta-Proposta.VrEntrada VrLiberado
				,case when PropostaCorrespondenteBncro.NuProposta is not null then 1 else 0 end as FlPropostaBipartida
				,Proposta.CdSubProduto

			FROM DBA.TbProposta Proposta
				/*Pegando as informaçõe de Percentual de entrada Molicar*/
				LEFT JOIN CDCCG.TbContrato Contrato
					ON  Proposta.NuContratoGestao = Contrato.NuContrato

				LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
					ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
					AND PessoaSobAnalise.CdPapel = 1

				LEFT JOIN DBA.TbPropGarantiaMaqEquip Garantia
					ON  Proposta.NuProposta = Garantia.NuProposta
					AND Garantia.CdPropostaGarantia = 1

				LEFT JOIN DBA.TbSubProduto SubProduto
					ON Proposta.CdSubProduto = SubProduto.CdSubProduto

				LEFT JOIN DBA.TbLoja Loja
					ON Proposta.CdLoja = Loja.CdLoja

				LEFT JOIN DBA.TbRegiao Regiao
					ON  Loja.CdRegiao = Regiao.CdRegiao

				LEFT JOIN DBA.TbParceiroComercial ParceiroComercial
					ON Proposta.CdLoja = ParceiroComercial.CdParceiroComercial

				LEFT JOIN TbColaborador Colaborador
					ON  Proposta.CdOperador = Colaborador.CdOperador

				LEFT JOIN TbColaborador ColaboradorPromotor
					ON  Proposta.CdPromotor = ColaboradorPromotor.CdOperador

				LEFT JOIN DBCRED.TbPropostaCorrespondenteBncro PropostaCorrespondenteBncro
					ON  Proposta.NuProposta = PropostaCorrespondenteBncro.NuProposta

				WHERE Proposta.TpProposta <> '3'
					AND DATE(Proposta.DtEntradaProposta) >= DATEADD (MONTH, -4, TODAY())
					AND Proposta.NuProposta <> 0
					AND Proposta.StProposta IN ('A','AP','PA','CA','N')
)
SELECT

	 Proposta.NuAnoMes
	,Proposta.DtEntradaProposta
	,Proposta.DtStatusProposta
	,Proposta.StProposta
	,Proposta.CdGrupoSubProduto
	,Proposta.CdColaborador
	,Proposta.NuProposta
	,Proposta.NuContratoGestao
	,Proposta.FlPropostaBipartida
	,Proposta.CdColaboradorPromotor
	,Proposta.CdFilial
	,Proposta.CdSubProduto
	,Proposta.CdParceiroComercial
	,Proposta.CdParceiroComercialCoban

create table TbContratosEGI as
select * from connection to dbcon
(
	SELECT
		'EGI' as TbOrigem
		,DATE(Contrato.DtContrato) DtReferencia
		,year(DATE(Contrato.DtContrato))*100+month(DATE(Contrato.DtContrato)) as NuAnoMes
		,Contrato.CdProduto*1000000 + Contrato.CdModalidadeProduto*1000 + 0 CdGrupoSubProduto
		,CASE WHEN Contrato.CdOperador <> Contrato.CdPromotor then Contrato.CdPromotor end CdColaboradorPromotor
		,Contrato.CdOperador as CdColaborador
		,Contrato.CdFilial
		,Contrato.CdParceiroComercial
		,Proposta.CdPessoa as CdParceiroComercialCoban
		,PessoaSobAnalise.TpPessoa
		,Contrato.NuContrato
		,Proposta.NuProposta
		,CASE
			WHEN ContratoFinanceiro.QtParcelas >  60 THEN '60+'
			WHEN ContratoFinanceiro.QtParcelas >  48 THEN '60'
			WHEN ContratoFinanceiro.QtParcelas >  36 THEN '48'
			WHEN ContratoFinanceiro.QtParcelas >  24 THEN '36'
			WHEN ContratoFinanceiro.QtParcelas >  18 THEN '24'
			WHEN ContratoFinanceiro.QtParcelas >  12 THEN '18'
			WHEN ContratoFinanceiro.QtParcelas >   0 THEN '12'
		 END FxParcela
		,ContratoLiberacao.VrLiberado VrProducao
		,ContratoFinanceiro.VrEntrada VrEntrada
	FROM EGI.TbContrato Contrato

		INNER JOIN EGI.TbContratoLiberacao ContratoLiberacao
			ON  Contrato.NuContrato = ContratoLiberacao.NuContrato
			AND Contrato.SqContratoFinanceiroAtivo = ContratoLiberacao.SqContratoFinanceiro

		INNER JOIN EGI.TbContratoFinanceiro ContratoFinanceiro
			ON  Contrato.NuContrato = ContratoFinanceiro.NuContrato
			AND Contrato.SqContratoFinanceiroAtivo = ContratoFinanceiro.SqContratoFinanceiro

		LEFT JOIN DBA.TbProposta Proposta
			ON  Contrato.NuContrato = Proposta.NuContratoGestao

		LEFT JOIN DBA.TbPessoaSobAnalise PessoaSobAnalise
			ON  Proposta.NuProposta = PessoaSobAnalise.NuProposta
			AND PessoaSobAnalise.CdPapel = 1

	WHERE Contrato.DtContrato between today()-31 and today()
	  AND Contrato.CdSituacaoContrato = 1

);quit;

create table TbIntraday as

		else GrupoProdutoMeta.NmGrupoProdutoMeta
	end as NmGrupoProdutoMeta

	/*		CASE WHEN GrupoProdutoMeta.NmGrupoProdutoMeta LIKE 'CVG%' THEN */
	,CASE    WHEN compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%GRCONSUL%' OR ParceiroComercial.NmParceiroComercial LIKE '%COCKPIT%'
					AND TipoCanalProducao.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS')
						THEN 'GR CONSULTOR'

					WHEN EstruturaComercial.NmAreaEstruturaComercial like '% Solar %' then 'CANAL SOLAR'
					WHEN Consolidado.CdParceiroComercial = 418254 then 'MEU FINANCIAMENTO SOLAR'

					WHEN TbFunilSimulacao_Solar.nuproposta is not null THEN upper(TbFunilSimulacao_Solar.DsTipoCanalProducao)   /*ORIGENS DO SOLAR*/

				    WHEN compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%CPGARANTIA%' OR compress(ParceiroComercial.NmParceiroComercial,' ')  LIKE '%BVPARTICULAR%'
					AND TipoCanalProducao.DsTipoCanalProducao IN ('LOJA CANAL NEGOCIOS')
						THEN 'BALCÃO'

				    WHEN ParceiroComercial.NmParceiroComercial  LIKE '%FINANZERO%' OR compress(ParceiroComercial.NmParceiroComercial,' ') LIKE '%QUEROFINANCIAR%'
					OR ParceiroComercial.CdParceiroComercial in(449109,452671,455884,460298,448756,454428,462320,460903,462486,419715,467954,468906,468957,469033,469213,468126,469532,
														412970,412971,412972,413732,414401,414420,433211,443941,444528,448742,448744,450421,453649,461517,461648,461879,462438,
														463084,465690,465799,470511,469570,470698,471098,471160,471162,471164,471181,471422,471505,472133,467403,472245,472263,
														472246,472247,472259,472260,472262,472244,472676,472229,472234,472239,472241,472242 )
					OR (ParceiroComercial.CdParceiroComercial = 441823 AND Consolidado.DtReferencia  >= '01OCT2024'd)
						THEN 'PARCEIRO DIGITAL'

					WHEN (ParceiroComercial.CdParceiroComercial = 407516 AND Consolidado.DtReferencia  BETWEEN '01JUL2020'd AND '31AUG2020'd)
                         OR (ParceiroComercial.CdParceiroComercial = 416382 AND Consolidado.DtReferencia  >= '01AUG2020'd)
                         THEN 'JUST'

					WHEN compress(ParceiroComercial.NmParceiroComercial,' ') LIKE '%CPGARANTMLKT%'
						OR ParceiroComercial.CdParceiroComercial IN(466807,418607,468271,472272) THEN 'TELEMARKETING'

						THEN 'MINHA BV'

						THEN 'CVG ÁREA ABERTA SITE'

                         AND (ParceiroComercial.CdParceiroComercial not in (407516) AND Consolidado.DtReferencia  BETWEEN '01JUL2020'd AND '31AUG2020'd)
