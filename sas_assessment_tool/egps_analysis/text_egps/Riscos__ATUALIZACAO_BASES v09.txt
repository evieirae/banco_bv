	create table WORK.FASE_INICIAL (compress=yes index=(BVT_PASTA/unique)) as
		select	BVT_PASTA ,

PROC SQL;
	
	create table TRAB.SEM_AVALIACAO_&DSRefModel. (compress=yes index=(BVT_PASTA/unique) pw="&DS_PASSWORD.") as
		select A.BVT_PASTA ,
		       A.BVT_PASTA as Causa,

			   A.FASE_ATUAL as FaseProv,

						case when A.FASE_ATUAL eq 'Inicial' then 'PU MÉDIO'  else 'Sem Avaliação' end AS Observacao,
			   0 format=commax22.2 as VlAvaliacaoPortalTerc,
			   0 format=commax22.2 as VlAvaliacaoPortal
		from TRAB.HISTORICA AS A WHERE A.BVT_PASTA NOT IN (SELECT BVT_PASTA FROM TRAB.REL_TRA_PROVISAO_&DSRefModel. (pw="&DS_PASSWORD."))
		AND A.STATUS = 'ATIVA'
		group by A.BVT_PASTA, A.FASE_ATUAL
	;

     (TFIM_IPCAe.VALOR / TCAD_IPCAe.VALOR) as TX_IPCA,

				
    CASE WHEN GRNT.METODO_ATUALIZACAO = 'IPCA-E + 1%'
			       THEN ((GRNT.VRGARPGTO) * (TFIM_IPCAe.VALOR / TCAD_IPCAe.VALOR))*(1+((INTCK('MONTH',GRNT.DTCADASTRO,&DtRefModel.)) * 0.01))
										ELSE 0 END FORMAT=COMMAX22.2 AS VRGARPGTO_IPCAe1,

				CASE WHEN GRNT.METODO_ATUALIZACAO = 'IPCA-E + 1%'
			       THEN ((GRNT.VRGARPGTO) / (TCAD_IPCAe.VALOR / TINI_IPCAe.VALOR))/(1+((INTCK('MONTH',CAUS.DTINICAUSA, GRNT.DTCADASTRO)) * 0.01))
          ELSE 0 END FORMAT=COMMAX22.2 AS VRGARPGTOINI_IPCAe1,

      (CALCULATED VRGARPGTO_IPCAe1)/GRNT.VRGARPGTO as TX_IPCAe1,

				
    CASE WHEN GRNT.METODO_ATUALIZACAO  in ('TR + 1%', 'TR', "")
										THEN (GRNT.VRGARPGTO) * ((TFIM_TRBAC.VALOR / TCAD_TRBAC.VALOR)+(INTCK('MONTH',GRNT.DTCADASTRO,&DtRefModel.) * 0.01))
										ELSE 0 END FORMAT=COMMAX22.2 AS VRGARPGTO_TR,

    CASE WHEN GRNT.METODO_ATUALIZACAO  in ('TR + 1%', 'TR', "")
										THEN (GRNT.VRGARPGTO) / ((TFIM_TRBAC.VALOR / TINI_TRBAC.VALOR)+(INTCK('MONTH',CAUS.DTINICAUSA, GRNT.DTCADASTRO) * 0.01))
										ELSE 0 END FORMAT=COMMAX22.2 AS VRGARPGTOINI_TR,

  ((TFIM_TRBAC.VALOR / TCAD_TRBAC.VALOR)+(INTCK('MONTH',GRNT.DTCADASTRO,&DtRefModel.) * 0.01))  as TX_TR,

				
    CASE WHEN GRNT.METODO_ATUALIZACAO = 'INPC + 1%'
										THEN ((GRNT.VRGARPGTO) * (TFIM_INPC.VALOR / TCAD_INPC.VALOR))*(1+((INTCK('MONTH',GRNT.DTCADASTRO,&DtRefModel.)) * 0.01))
										ELSE 0 END  FORMAT=COMMAX22.2 AS VRGARPGTO_INPC,

				CASE WHEN GRNT.METODO_ATUALIZACAO = 'INPC + 1%'
										THEN ((GRNT.VRGARPGTO) / (TCAD_INPC.VALOR / TINI_INPC.VALOR))/(1+((INTCK('MONTH',CAUS.DTINICAUSA, GRNT.DTCADASTRO)) * 0.01))
										ELSE 0 END  FORMAT=COMMAX22.2 AS VRGARPGTOINI_INPC,

				(CALCULATED VRGARPGTO_INPC)/GRNT.VRGARPGTO as TX_INPC,

			

				
				GRNT.VRTRANSFJUD,
				(0)  FORMAT=COMMAX22.2 AS VRTRANSFJUDPERDA,
				(0)  FORMAT=COMMAX22.2 AS VRTRANSFJUDPERDATR1,
				(0)  FORMAT=COMMAX22.2 AS VRTRANSFJUDPERDAINI,

				
				(GRNT.VRGARALVARA + GRNT.VRGARPGTO) FORMAT=COMMAX22.2 AS VRPERDATOTAL,

    (CALCULATED VRGARALVARA_IPCAe1  + CALCULATED VRGARALVARA_SELIC + CALCULATED VRGARALVARA_TR + CALCULATED VRGARALVARA_TSTIP + CALCULATED VRGARALVARA_IPCAe + CALCULATED VRGARALVARA_INPC)
				FORMAT=COMMAX22.2  as VrGarAlvaraTR1,
     (CALCULATED VRGARPGTO_IPCAe1    + CALCULATED VRGARPGTO_SELIC   + CALCULATED VRGARPGTO_TR   + CALCULATED VRGARPGTO_TSTIP   + CALCULATED VRGARPGTO_IPCAe   + CALCULATED VRGARPGTO_INPC)
    FORMAT=COMMAX22.2  AS VrGarPgtoTR1,

				(CALCULATED VrGarAlvaraTR1 + CALCULATED VrGarPgtoTR1) FORMAT=COMMAX22.2  AS VRPERDATOTALTR1,

				


				CASE WHEN GRNT.ASSUNTO not in ('Acordo Extrajudicial', 'Funcionário')
										THEN (CALCULATED VrGarPgtoTR1)
										ELSE 0 END  FORMAT=COMMAX22.2  AS VRPERDATOTALTR1_TERC,

    CASE WHEN GRNT.ASSUNTO not in ('Acordo Extrajudicial', 'Funcionário')
										THEN (CALCULATED VrGarAlvaraTR1)
										ELSE 0 END  FORMAT=COMMAX22.2  AS VR_PAG_GAR_RECURTR1_TERC,

				
				

				(CALCULATED VRGARALVARAINI_IPCAe1  + CALCULATED VRGARALVARAINI_SELIC + CALCULATED VRGARALVARAINI_TR + CALCULATED VRGARALVARAINI_TSTIP + CALCULATED VRGARALVARAINI_IPCAe + CALCULATED VRGARALVARAINI_INPC)
    FORMAT=COMMAX22.2  AS VrGarAlvaraIni,
    (CALCULATED VRGARPGTOINI_IPCAe1    + CALCULATED VRGARPGTOINI_SELIC   + CALCULATED VRGARPGTOINI_TR   + CALCULATED VRGARPGTOINI_TSTIP   + CALCULATED VRGARPGTOINI_IPCAe   + CALCULATED VRGARPGTOINI_INPC)
    FORMAT=COMMAX22.2  AS VrGarPgtoIni,

   (CALCULATED VrGarAlvaraIni		+ CALCULATED	VrGarPgtoIni	) FORMAT=COMMAX22.2  AS VRPERDATOTALINI,

				GRNT.VRDJNAORECURS,
				GRNT.VrTsfJud,
				GRNT.VRFIANCAATIVA,
				GRNT.VRSENTENCA,
				GRNT.VRACORDAO,
				GRNT.VREXECUCAO,

   LEFT JOIN TAB_SELIR 			AS TINI_SELIR  ON (INTNX('MONTH',CAUS.DTINICAUSA,0,'B') = TINI_SELIR.DTFIM)
			LEFT JOIN TAB_SELIR 			AS TCAD_SELIR  ON (INTNX('MONTH',GRNT.DTCADASTRO,1,'B') = TCAD_SELIR.DTFIM)
			LEFT JOIN TAB_SELIR  		AS TFIM_SELIR  ON (INTNX('MONTH',&DtRefModel.,0,'B') 		 = TFIM_SELIR.DTFIM)

			LEFT JOIN TAB_IPCAe  		AS TFIM_IPCAe  ON (INTNX('MONTH',&DtRefModel.,-1,'B') 		 = TFIM_IPCAe.DTINI)

   
   LEFT JOIN TAB_TRBAC 			AS TINI_TRBAC  ON (INTNX('MONTH',CAUS.DTINICAUSA,0,'B') = TINI_TRBAC.DTFIM)

	create table TRAB.PJ_PERDATRANSFJUD_&DSRefModel. (compress=yes index=(BVT_PASTA)) as
		select	tjud.BVT_PASTA,
				caus.DtIniCausa,
				tjud.DtCadastro,
				tjud.DtBaixa,

				tjud.VlPerda / ((tcad.IndiceTR / tini.IndiceTR) + (intck('month',caus.DtIniCausa,tjud.DtCadastro) * 0.01)) format=commax22.2 as VlPerdaTransfJudIni,
				tjud.VlPerda * ((tfim.IndiceTR / tcad.IndiceTR) + (intck('month',tjud.DtCadastro,&DtRefModel.) * 0.01)) format=commax22.2 as VlPerdaTransfJudTR1

		from TRAB.PJ_BAIXATRANSFJUD_&DSRefModel. (pw="&DS_PASSWORD.") as tjud
			left join TRAB.PJ_CAUSAS_&DSRefModel. (pw="&DS_PASSWORD.")				as caus on (tjud.Causa = caus.Causa)

		where tjud.DtBaixa le &DtRefModel.
