CREATE TABLE WORK.BASE_ACORDO_00 AS
SELECT A.*
			FROM
			CONNECTION TO DBCON(SELECT DISTINCT
					              CON.DTGERACAO
					            , CON.NUCONTRATOGESTAO
					            , CON.DTVENCIMENTO
					            , CON.DTPAGAMENTO
					            , CON.STNEGOCIACAO
					            , CON.TPENTRADA
					            , CASE
					                WHEN CON.STNEGOCIACAO = 'AN' THEN 'A NEGOCIAR'
					                WHEN CON.STNEGOCIACAO = 'N'  THEN 'NEGOCIADO'
					                WHEN CON.STNEGOCIACAO = 'PA' THEN 'PENDENTE DE APROVACAO'
					                WHEN CON.STNEGOCIACAO = 'A'  THEN 'APROVADO PELA BV'
					                WHEN CON.STNEGOCIACAO = 'R'  THEN 'RECUSADO PELA BV'
					                WHEN CON.STNEGOCIACAO = 'NA' THEN 'NEGOCIACAO EM ANDAMENTO'
					                WHEN CON.STNEGOCIACAO = 'C'  THEN 'NEGOCIACAO CANCELADA'
					                WHEN CON.STNEGOCIACAO = 'LP' THEN 'LIBERADO PAGAMENTO'
					                WHEN CON.STNEGOCIACAO = 'PG' THEN 'PAGO'
					                WHEN CON.STNEGOCIACAO = 'PL' THEN 'PAGAMENTO REALIZADO'
					                WHEN CON.STNEGOCIACAO = 'DA' THEN 'DEVOLVIDO ASSESSORIA'
					                WHEN CON.STNEGOCIACAO = 'CD' THEN 'CHEQUE DEVOLVIDO'
					                WHEN CON.STNEGOCIACAO = 'AC' THEN 'CHEQUE AGUARDANDO COMPENSACAO'
					             ELSE '99.VERIFICAR' END STATUS_BOLETO
						FROM DBCONTENCIOSO..TBNEGOCIACAOCONTRATO CON
						WHERE CON.DTGERACAO > DATEADD(DD, -1, GETDATE())
						AND CON.DTVENCIMENTO > DATEADD(DD, -1, GETDATE())
						AND CON.DTVENCIMENTO <= DATEADD(DD, 20, GETDATE())
						ORDER BY CON.DTVENCIMENTO DESC)
AS A
;QUIT;




PROC SQL;
CREATE TABLE WORK.BASE_ACORDO AS
SELECT DISTINCT A.*
, D.NUCPFCNPJ
, intck('DAY', DATEPART(A.DtGeracao), DATEPART(A.DtVencimento), 'C')  AS Intervalo_GerVenc_Bol
, DATEPART(A.DtGeracao) Format yymmdd10. AS Dt_Geracao_Boleto
, DATEPART(A.DtVencimento) Format yymmdd10. AS Dt_Vencimento_Boleto
, HOUR(A.DtGeracao) AS Hr_Geracao
, Case
	WHEN B.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) <=  7 THEN 1
	WHEN B.Canal_Macro = 'ESCOB' AND HOUR(A.DtGeracao) >= 20 THEN 1
	ELSE 0 END AS Flag_ForaExped
, B.Canal As Canal_Boleto
, B.Canal_Macro
, C.UltimoDUSafra
, C.DU
, C.Periodo
, CASE WHEN D.NUCPFCNPJ = E.NUCPFCNPJ THEN 'S' ELSE 'N' END AS Flag_Mult_Contr
, D.Collection
, D.Fx_Idade
, D.Fx_Atraso
, D.Flag_HU
FROM WORK.BASE_ACORDO_00 AS A
LEFT JOIN TCONFIG.CONFIG_TPENTRADA AS B ON A.TpEntrada = B.TpEntrada
LEFT JOIN TCONFIG.AUX_DU AS C ON DATEPART(A.DtGeracao) = C.DATA
LEFT JOIN CONTATOS.AUX_BASE_CYBER AS D ON A.NUCONTRATOGESTAO = D.NUCONTRATOGESTAO
LEFT JOIN (SELECT DISTINCT A.NUCPFCNPJ, COUNT(DISTINCT A.NuContratoGestao) AS Qtd_Contr FROM CONTATOS.AUX_BASE_CYBER AS A GROUP BY A.NUCPFCNPJ Having Qtd_Contr > 1) AS E ON D.NUCPFCNPJ = E.NUCPFCNPJ
LEFT JOIN (SELECT DISTINCT A.NUCONTRATOGESTAO, MAX(A.DtGeracao) AS ULT_BOLETO FROM WORK.BASE_ACORDO_00 AS A GROUP BY A.NUCONTRATOGESTAO) AS F ON A.NUCONTRATOGESTAO = F.NUCONTRATOGESTAO AND A.DtGeracao = F.ULT_BOLETO
WHERE D.ATRASO <= 70
;QUIT;
