CREATE TABLE CONSULTA_AJUIZAMENTO AS SELECT
* from
CONNECTION TO DBCON	(

SELECT NuContratoGestao,
				SqContratoFinanceiro,
				DATE(Dt_DESP) AS Dt_DESP
		INTO	#BASE1
		FROM	DBA.SAJ_DESP_CONTRATO
		WHERE	dt_desp > '20210101' and cod_desp in (14,80)
		GROUP BY NuContratoGestao, SqContratoFinanceiro, Dt_DESP

		SELECT	A.NuContratoGestao,
				A.SqContratoFinanceiro,
				DATE(A.Dt_DESP) AS Dt_DESP,
				DATEDIFF(DAY, MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END), Dt_DESP) AS Atraso,
				SUM(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
				COUNT(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
				SUM(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
				COUNT(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN Nuparcela ELSE NULL END) as PARCELA

		FROM	#BASE1 AS A
				LEFT JOIN DBA.VwTbParcela AS B ON	A.NuContratoGestao = B.NuContrato AND
													A.SqContratoFinanceiro = B.SqContratoFinanceiro
		GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP
)
;QUIT;

PROC SORT DATA=CONSULTA_AJUIZAMENTO NODUPKEY; BY NuContratoGestao SqContratoFinanceiro PARCELA;RUN;

CREATE TABLE CONSULTA_AJUIZAMENTO AS SELECT
* from
CONNECTION TO DBCON	(

SELECT NuContratoGestao,
				SqContratoFinanceiro,
				DATE(Dt_DESP) AS Dt_DESP
		INTO	#BASE1
		FROM	DBA.SAJ_DESP_CONTRATO
		WHERE	dt_desp > '20210101' and cod_desp in (14,80)
		GROUP BY NuContratoGestao, SqContratoFinanceiro, Dt_DESP

		SELECT	A.NuContratoGestao,
				A.SqContratoFinanceiro,
				DATE(A.Dt_DESP) AS Dt_DESP,
				DATEDIFF(DAY, MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END), Dt_DESP) AS Atraso,
				SUM(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
				COUNT(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
				SUM(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
				COUNT(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN Nuparcela ELSE NULL END) as PARCELA

		FROM	#BASE1 AS A
				LEFT JOIN DBA.VwTbParcela AS B ON	A.NuContratoGestao = B.NuContrato AND
													A.SqContratoFinanceiro = B.SqContratoFinanceiro
		GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP

)

;QUIT;

PROC SORT DATA=CONSULTA_AJUIZAMENTO NODUPKEY; BY NuContratoGestao SqContratoFinanceiro PARCELA;RUN;

CREATE TABLE CONSULTA_AJUIZAMENTO AS SELECT
* from
CONNECTION TO DBCON	(

SELECT NuContratoGestao,
				SqContratoFinanceiro,
				DATE(Dt_DESP) AS Dt_DESP
		INTO	#BASE1
		FROM	DBA.SAJ_DESP_CONTRATO
		WHERE	dt_desp > '20210101' and cod_desp in (14,80)
		GROUP BY NuContratoGestao, SqContratoFinanceiro, Dt_DESP

		SELECT	A.NuContratoGestao,
				A.SqContratoFinanceiro,
				DATE(A.Dt_DESP) AS Dt_DESP,
				DATEDIFF(DAY, MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END), Dt_DESP) AS Atraso,
				SUM(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE 0 END) AS Risco_Aberto,
				COUNT(CASE WHEN	B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								B.DtUltimoRecebimento >= Dt_DESP)  THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_Abertas,
				SUM(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE 0 END) AS Risco_Aberto_em_Atraso,
				COUNT(CASE WHEN	(B.StParcela <> 'Q' OR
								(B.StParcela = 'Q' AND
								DATE(B.DtUltimoRecebimento) >= Dt_DESP)) AND
								DATE(B.DtVencimento) < A.Dt_DESP THEN VrParcela ELSE NULL END) AS Qtd_Parcelas_em_Atraso,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN DtVencimento ELSE NULL END) as DT_VENCIMENTO_NOVO,
				MIN(CASE WHEN	B.StParcela <> 'Q' OR
											(B.StParcela = 'Q' AND
											B.DtUltimoRecebimento >= Dt_DESP)  THEN Nuparcela ELSE NULL END) as PARCELA

		FROM	#BASE1 AS A
				LEFT JOIN DBA.VwTbParcela AS B ON	A.NuContratoGestao = B.NuContrato AND
													A.SqContratoFinanceiro = B.SqContratoFinanceiro
		GROUP BY A.NuContratoGestao, A.SqContratoFinanceiro, A.Dt_DESP

)

;QUIT;

PROC SORT DATA=CONSULTA_AJUIZAMENTO NODUPKEY; BY NuContratoGestao SqContratoFinanceiro PARCELA;RUN;
