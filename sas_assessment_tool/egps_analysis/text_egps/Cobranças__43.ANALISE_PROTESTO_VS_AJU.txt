PROC SQL;
%CONNECT_GCP;
CREATE TABLE BASE_PAGAMENTOS_&anomes. AS
SELECT *
FROM CONNECTION TO perm
(
  WITH base AS (
    SELECT
      CAST(z.CD_CONTRATO AS INT64)  AS NUCONTRATO,
      z.SQ_CONTRATO_FINANCEIRO      AS SQCONTRATOFINANCEIRO,
      z.Dt_Recebimento              AS DtRecebimento,
      z.Vr_Principal_Devido         AS VR_PRINCIPAL_DEVIDO,
      z.Vr_Principal_Baixado        AS VR_PRINCIPAL_BAIXADO,
	  z.vr_recebido					AS VR_RECEBIDO,
	  z.vr_devido_total				AS VR_DEVIDO_TOTAL,
	  z.Nu_Parcela,
      EXTRACT(YEAR FROM z.DT_RECEBIMENTO) * 100
        + EXTRACT(MONTH FROM z.DT_RECEBIMENTO) AS SafraAjuste,
      B.NuDiasAtraso                AS NuDiasAtraso_raw,
      DATE(B.DTPROCESSAMENTO)       AS DtProcessamento,
      DATE_DIFF(DATE(B.DTPROCESSAMENTO), DATE(z.DT_RECEBIMENTO), DAY) AS diff,
      z.CD_CONTRATO,
      z.SQ_CONTRATO_FINANCEIRO
    FROM  `bv-pdbd-prd.dvry_core.tbparcelarecebimento` AS z
    LEFT JOIN `bv-pdbd-prd.trusted_gestaocontabil.tbcontratocontabil` AS B
      ON z.CD_CONTRATO = B.NUCONTRATO
     AND z.SQ_CONTRATO_FINANCEIRO = B.SQCONTRATOFINANCEIRO
     AND DATE(B.DTPROCESSAMENTO) BETWEEN DATE(z.DT_RECEBIMENTO) - 5 AND DATE(z.DT_RECEBIMENTO) + 5
    WHERE DATE(z.Dt_Recebimento) >= DATE(&dt_inicio.)
      AND DATE(z.Dt_Recebimento) <= DATE(&dt_fim.)
      AND CAST(z.Vr_Recebido AS FLOAT64) NOT IN (0, 0.01)
      AND CAST(z.Fl_Estornado AS STRING) <> 'S'
  ),
  ranked AS (
    SELECT
      NUCONTRATO,
      SQCONTRATOFINANCEIRO,
      DtRecebimento,
      VR_PRINCIPAL_DEVIDO,
      VR_PRINCIPAL_BAIXADO,
	  VR_RECEBIDO,
	  VR_DEVIDO_TOTAL,
	  Nu_Parcela,
      SafraAjuste,
      NuDiasAtraso_raw,
      diff,
      ROW_NUMBER() OVER (
        PARTITION BY CD_CONTRATO, SQ_CONTRATO_FINANCEIRO, DtRecebimento, Nu_Parcela, diff
        ORDER BY ABS(diff) ASC, DtProcessamento ASC
      ) AS rn
    FROM base
  ),
  best_per_diff AS (
    SELECT *
    FROM ranked
    WHERE rn = 1
  )

  SELECT
    NUCONTRATO,
    SQCONTRATOFINANCEIRO,
	MAX(CASE WHEN diff between -5 AND -1 THEN NuDiasAtraso_raw END) AS NuDiasAtraso_PRE_Acao,
    MAX(CASE WHEN diff =  0 THEN NuDiasAtraso_raw END) AS NuDiasAtraso_Dia_Acao,
	MIN(CASE WHEN diff between 1 AND 5 THEN NuDiasAtraso_raw END) AS NuDiasAtraso_POS_Acao,
    DtRecebimento,
    VR_PRINCIPAL_DEVIDO,
    VR_PRINCIPAL_BAIXADO,
	VR_RECEBIDO,
	VR_DEVIDO_TOTAL,
	Nu_Parcela,
    SafraAjuste
  FROM best_per_diff
  GROUP BY
    NUCONTRATO,
    SQCONTRATOFINANCEIRO,
    DtRecebimento,
    VR_PRINCIPAL_DEVIDO,
    VR_PRINCIPAL_BAIXADO,
	VR_RECEBIDO,
	VR_DEVIDO_TOTAL,
	Nu_Parcela,
    SafraAjuste
);
QUIT;

FROM CONNECTION TO perm
(
    SELECT CAST(a.NucontratoGestao AS INT64) AS NUCONTRATO,
        a.SqContratoFinanceiro,

        a.NuProcesso,
        MAX(k.VrBNDU) AS VrBNDU,

			 when tp.TpApreensao = 4 then 'EA_S_QUITACAO'
			 when tp.TpApreensao = 5 then 'EA_CONVER'

        MIN(CASE WHEN b.St_Parcela <> 'Q'
                 OR (b.St_Parcela = 'Q' AND b.dt_processamento_ult_recebimento > DATE(k.DtInclusao))
                 THEN b.Nu_parcela ELSE NULL END) AS Parcela

        ON a.NucontratoGestao = b.cd_contrato
        AND a.SqContratoFinanceiro = b.Sq_Contrato_Financeiro

		ON a.NucontratoGestao = tp.nucontrato
    WHERE DATE(k.DtInclusao) >= &dt_inicio. AND DATE(k.DtInclusao) <= &dt_fim.
    GROUP BY a.NucontratoGestao, a.SqContratoFinanceiro, a.CdBem, a.NuProcesso, k.DtInclusao, tp.TpApreensao

PROC SQL;
CREATE TABLE REGULARIZACAO AS
SELECT DISTINCT A.*,
B.VR_PRINCIPAL_DEVIDO  AS VR_PRINCIPAL_DEVIDO,
B.VR_PRINCIPAL_BAIXADO AS VR_PRINCIPAL_BAIXADO,
B.VR_RECEBIDO		   AS VR_RECEBIDO,
B.VR_DEVIDO_TOTAL	   AS VR_DEVIDO_TOTAL,
B.Nu_Parcela 		   AS NU_PARCELA_PAGA,
                CASE WHEN E.NuContrato NE . THEN 1
					 WHEN B.NuContrato NE . THEN 1
				     WHEN C.NuContrato NE . THEN 1
				     WHEN D.NuContrato NE . THEN 1
                ELSE 0 END AS Fl_Regularizacao,
				CASE WHEN E.NuContrato NE .            THEN 'BNDU'
					 WHEN C.NuContrato NE .            THEN 'Reneg'
				     WHEN D.NuContrato NE .            THEN 'Quitacao'
					 WHEN B.NuContrato NE . AND B.NuDiasAtraso_POS_Acao > NuDiasAtraso_PRE_Acao THEN 'Pagamento Parcial (Pulada)'
					 WHEN B.NuContrato NE . AND B.NuDiasAtraso_POS_Acao > 0 THEN 'Pagamento Parcial'
					 WHEN B.NuContrato NE .            THEN 'Atualizacao'

                ELSE 'Sem Acao' END AS Tipo_Recebimento,
				CASE WHEN E.NuContrato NE .            THEN E.dtretomada
					 WHEN C.NuContrato NE .       	   THEN datepart(C.dt_baixa)
				     WHEN D.NuContrato NE .            THEN datepart(D.Dt_Liquidacao)
					 WHEN B.NuContrato NE .            THEN B.DtRecebimento

                ELSE . END AS DT_ACAO FORMAT = date9.

FROM HIST_PROTESTO_VS_AJU AS A

LEFT JOIN BASE_PAGAMENTOS_EMP   AS B
ON A.nucontratogestao = B.NuContrato       AND A.sqcontratofinanceiro = B.SqContratoFinanceiro AND B.DtRecebimento >= A.dtgravacaoscore


LEFT JOIN BASE_RENEG_EMP       AS C
ON A.nucontratogestao = C.NuContrato       AND A.sqcontratofinanceiro = C.SqContratoFinanceiro AND datepart(C.dt_baixa) >= A.dtgravacaoscore


LEFT JOIN BASE_LIQUIDACAO_EMP   AS D
ON A.nucontratogestao = D.NuContrato       AND A.sqcontratofinanceiro = D.SqContratoFinanceiro AND datepart(D.Dt_Liquidacao) >= A.dtgravacaoscore


LEFT JOIN BASE_BNDU_EMP         AS E
ON A.nucontratogestao = E.NuContrato       AND A.sqcontratofinanceiro = E.SqContratoFinanceiro AND E.dtretomada >= A.dtgravacaoscore


;QUIT;

PROC SORT DATA=REGULARIZACAO;BY nucontratogestao sqcontratofinanceiro dtgravacaoscore ds_estrategia DT_ACAO NU_PARCELA_PAGA;RUN;
PROC SORT DATA=REGULARIZACAO_NDP (KEEP=nucontratogestao sqcontratofinanceiro dtgravacaoscore ds_estrategia DT_ACAO) NODUPKEY OUT=REGULARIZACAO_NDP; BY nucontratogestao sqcontratofinanceiro dtgravacaoscore ds_estrategia ; RUN;

                CASE WHEN E.NuContrato NE . THEN 1
					 WHEN B.NuContrato NE . THEN 1
				     WHEN C.NuContrato NE . THEN 1
				     WHEN D.NuContrato NE . THEN 1
                ELSE 0 END AS Fl_Regularizacao,
				CASE WHEN E.NuContrato NE .            THEN 'BNDU'
					 WHEN C.NuContrato NE .            THEN 'Reneg'
				     WHEN D.NuContrato NE .            THEN 'Quitacao'
					 WHEN B.NuContrato NE . AND B.NuDiasAtraso_POS_Acao > NuDiasAtraso_PRE_Acao THEN 'Pagamento Parcial (Pulada)'
					 WHEN B.NuContrato NE . AND B.NuDiasAtraso_POS_Acao > 0 THEN 'Pagamento Parcial'
					 WHEN B.NuContrato NE .            THEN 'Atualizacao'

                ELSE 'Sem Acao' END AS Tipo_Recebimento,
				CASE WHEN E.NuContrato NE .            THEN E.dtretomada
					 WHEN C.NuContrato NE .       	   THEN datepart(C.dt_baixa)
				     WHEN D.NuContrato NE .            THEN datepart(D.Dt_Liquidacao)

ON A.nucontratogestao = C.NuContrato       AND A.sqcontratofinanceiro = C.SqContratoFinanceiro AND datepart(C.dt_baixa) >= A.dtgravacaoscore


LEFT JOIN BASE_LIQUIDACAO_EMP   AS D
ON A.nucontratogestao = D.NuContrato       AND A.sqcontratofinanceiro = D.SqContratoFinanceiro AND datepart(D.Dt_Liquidacao) >= A.dtgravacaoscore


LEFT JOIN BASE_BNDU_EMP         AS E
