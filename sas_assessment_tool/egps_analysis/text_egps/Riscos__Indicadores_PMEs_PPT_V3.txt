		,SUM(CASE WHEN Fx_Atraso IN ("Em dia",  "1 - 5d") THEN Saldo END) AS 'Em dia / Atraso <6d'n FORMAT COMMAX26.
		,SUM(CASE WHEN Fx_Atraso NOT IN ("Em dia", "1 - 5d") THEN Saldo END) AS 'Em Atraso >6d'n FORMAT COMMAX26.
		,CALCULATED 'Em Atraso >6d'n / SUM(Saldo) AS '%Atraso >6d'n FORMAT PERCENT10.2

FROM BASE_5 GROUP BY 1;

SELECT AnoMes_Referencia
		,(SUM(Provisao) / SUM(Saldo)) AS '%PDD'n FORMAT PERCENT10.2
		,(SUM(CASE WHEN Fx_Atraso IN ("Em dia", "1 - 5d") THEN Provisao END) / SUM(CASE WHEN Fx_Atraso IN ("Em dia", "1 - 5d") THEN Saldo END)) AS '%PDD Em Dia'n FORMAT PERCENT10.2

			WHEN Tipo_Carteira_Financas = 'CORPORATE' THEN 'CORPORATE'
			WHEN Tipo_Carteira_Financas = 'LARGE' THEN 'LARGE'
			WHEN Tipo_Carteira_Financas = 'FI' THEN 'FI'
			WHEN Tipo_Carteira_Financas = 'ATIVOS ESPECIAIS' THEN 'ATIVOS ESPECIAIS'
			ELSE 'PM/UM' END AS SEGMENTO_CARTEIRA
		,CASE 	WHEN (CDMODALIDADEPRODUTO = 211 AND DSGRUPOECONOMICORISCO NE 'FIDC TRADEMASTER') OR (CDMODALIDADEPRODUTO = 214 AND DSGRUPOECONOMICORISCO CONTAINS ('DEB')) THEN 'FIDC'
				WHEN DsGrupoProduto = 'Empréstimos' THEN 'Credito'
				WHEN DsGrupoProduto = 'Trade Finance' THEN 'Credito'
				WHEN DsGrupoProduto = 'External Trade Finance' THEN 'Credito'
				WHEN DsGrupoProduto = 'BNDES' THEN 'Credito'
				WHEN DsGrupoProduto = 'External Loans' THEN 'Credito'
				WHEN DsGrupoProduto = 'Crédito Rural' THEN 'Credito'
				WHEN DsGrupoProduto = 'Leasing' THEN 'Credito'
				WHEN DsGrupoProduto = 'Créditos do Agronegócio' THEN 'Credito'
				WHEN DsGrupoProduto = 'Operações de Garantias' THEN 'Fianca'
				WHEN DsGrupoProduto = 'TVM' THEN 'TVM'
				ELSE 'Derivativos' END AS SEGMENTO_PRODUTO

		,NuEstagioFinal AS ESTAGIO
		,SUM(VrSaldo) AS Saldo format commax26.2
		,SUM(VrProvisao) 	AS PROVISAOt FORMAT COMMAX26.0

FROM WORK.BASE_UNICA
GROUP BY 1,2,3,4,5,6
;QUIT;

PROC SQL;
CREATE TABLE ROLAGEM_&ANOMES. AS
SELECT DISTINCT
		A.AnoMes_Referencia
		,A.NuObjetoNegociacao
		,A.DsGrupoEconomicoRisco
		,A.Saldo
		,CASE WHEN A.VNP GT 0 AND A.VNP LE 30 THEN A.Saldo ELSE 0 END AS OVER1
		,CASE WHEN A.VNP GT 30 AND A.VNP LE 60 THEN A.Saldo ELSE 0 END AS OVER30
		,CASE WHEN A.VNP GT 60 AND A.VNP LE 90 THEN A.Saldo ELSE 0 END AS OVER60
		,CASE WHEN A.VNP GT 90 THEN A.Saldo ELSE 0 END AS OVER90
		,CASE WHEN B.VNP GT 0  AND B.VNP LE 30 THEN B.Saldo ELSE 0 END AS OVER1_ANT
		,CASE WHEN B.VNP GT 30 AND B.VNP LE 60 THEN B.Saldo ELSE 0 END AS OVER30_ANT
		,CASE WHEN B.VNP GT 60 AND B.VNP LE 90 THEN B.Saldo ELSE 0 END AS OVER60_ANT
		,CASE WHEN B.VNP GT 90 				   THEN B.Saldo ELSE 0 END AS OVER90_ANT

FROM BASE_5_CIB AS A
LEFT JOIN (SELECT NuObjetoNegociacao, Saldo, Provisao, VNP, Dt_Venc_Atraso FROM BASE_5_CIB WHERE VNP > 0 AND AnoMes_Referencia = %REFM2(&ANOMES., -1)) AS B ON A.NuObjetoNegociacao = B.NuObjetoNegociacao
WHERE A.AnoMes_Referencia = &ANOMES. AND A.DsProduto CONTAINS "PEAC"
;QUIT;
