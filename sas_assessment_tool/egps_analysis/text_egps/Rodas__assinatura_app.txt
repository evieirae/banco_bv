proc sql;
create table duplicidade_2a as
	select	  t1.sessionId
			, sum(t1.fxIncorreto) 		as fxIncorreto
			, sum(t1.fxIntroducao) 		as fxIntroducao
			, sum(t1.fxenviarCodigo) 	as fxEnviarCodigo
			, sum(t1.fxOutras) 			as fxOutras
	from (
			select	  sessionId
					, case when screen = 'fluxoIncorreto' or
								screen = 'algoDeuErrado' or
								screen = 'finalizandoDetalhesFinanciamento'
						   then 1 else 0 end as fxIncorreto
					, case when screen = 'introducao'
						   then 1 else 0 end as fxIntroducao
					, case when screen = 'enviarCodigo'
						   then 1 else 0 end as fxEnviarCodigo
					, case when screen ne 'fluxoIncorreto' and
								screen ne 'algoDeuErrado' and
								screen ne 'finalizandoDetalhesFinanciamento' and
								screen ne 'introducao' and
								screen ne 'enviarCodigo'
						   then 1 else 0 end as fxOutras
			from app_tag_3
/*			where eventType = 'Screen'*/
	) t1
	group by 1
;quit;

/*identificar as sessões cujos disparos das telas introducao e enviarCodigo ocorreram mas não eram esperadas*/
data duplicidade_21a; set duplicidade_2a;
	if 	 fxIncorreto  > 0 and							/*esteve no fluxo incorreto*/
		(fxIntroducao > 0 or fxEnviarCodigo > 0) and	/*esteve na telaInicial, conforme flag cliente ativo*/
		 fxOutras     = 0 								/*não esteve em outras telas além do fluxo incorreto e tela inicial*/
		 				  then flAtivo2 = 0;			/*as condições indicam que a tela introdução e enviarCodigo não eram esperadas*/
						  else delete;

	drop fxEnviarCodigo fxIncorreto fxIntroducao fxOutras;
run;
