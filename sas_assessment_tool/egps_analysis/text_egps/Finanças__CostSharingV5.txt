    Create table CS_ALOCACAO as
    select * from connection to perm
    (
        with tbclassevalor as (
            select *
            from `bv-pdbd-prd.dvry_corp_financas.analise_classevalor_dime`
            where cd_centro_custo in ('400100','400101','400102','400103','400104','400105','400106','400107','400108','400109','400140','400150','400160','400170','400180','400190')
        ),

                nm_veiculo_legal as nome_veiculo,

            from `bv-pdbd-prd.dvry_corp_financas.myabc_custosdespesas_fato` fct
            inner join tbclassevalor tcv
            on CAST(CAST(SPLIT(fct.nm_classe_valor, '|')[OFFSET(0)] AS INT) AS STRING) = tcv.cd_classe_valor

            on upper(trim(nm_produto)) = upper(trim(Produto))
            where ds_modelo like "R'%"
            and dt_periodo = &DATA_ALOCACAO.
			and cc <> '120499612' 
            group by 1, 2, 3, 4, 5, 6, 7
        ),
        tbalocacaoRateio as (
            select
                CAST(CAST(SPLIT(nm_centro_custo, '|')[OFFSET(0)] AS INT) AS STRING) as centro_custo,
                dt_periodo,
                date_add(dt_periodo, INTERVAL +1 MONTH) as dt_periodo_gcut,
                nm_veiculo_legal as nome_veiculo,
                cd_veiculo_legal as codigo_veiculo,
                nm_negocio as nome_negocio,
                cc as centro_custo_veiculo,
                sum(vr_gasto) * -1 as gasto
            from `bv-pdbd-prd.dvry_corp_financas.myabc_custosdespesas_fato`
            inner join (select * from `bv-pdbd-prd.trusted_corp_financas.tbprodutocc` where PARSE_DATE('%d/%m/%Y', Data) = (select max(PARSE_DATE('%d/%m/%Y', Data)) from `bv-pdbd-prd.trusted_corp_financas.tbprodutocc` ))
            on upper(trim(nm_produto)) = upper(trim(Produto))
            where ds_modelo like "R'%"
            and dt_periodo = &DATA_ALOCACAO.
            group by 1, 2, 3, 4, 5, 6, 7
            union all
            select * from tbalocacaoClasse
        )
        select
            *,
            SAFE_DIVIDE(cast(gasto as numeric), sum(cast(gasto as numeric)) over(partition by centro_custo)) as vlr
        from tbalocacaoRateio

Proc Sql;
create table Real_filtro_bankly as
 	select * from tb_costsharing_banco
WHERE nome_veiculo_costsharing = 'Bankly' AND NOT (UPCASE(nome_fornecedor) contains 'DIMENSA ')
AND nome_veiculo_costsharing = 'Bankly' AND NOT (UPCASE(nome_fornecedor) contains 'ACESSO ')
AND  nome_veiculo_costsharing = 'Bankly' AND NOT (UPCASE(Historico) contains 'FIN-')
 ;Quit;

proc sql;
    create table tb_pivot_banco_FiltroBankly as
    select
        conta_contabil,
		nome_conta_contabil,
        nm_agrupamento,
        sum(case when nome_veiculo_costsharing = 'Banco Votorantim' then vlr else 0 end) as Banco_Votorantim,
        sum(case when nome_veiculo_costsharing = 'Bankly' then vlr else 0 end) as Bankly,
        sum(case when nome_veiculo_costsharing = 'VCS' then vlr else 0 end) as VCS,
        sum(case when nome_veiculo_costsharing = 'BV S.A.' then vlr else 0 end) as BVSA,
        sum(case when nome_veiculo_costsharing = 'BVIP' then vlr else 0 end) as BVIP,
        sum(case when nome_veiculo_costsharing = 'BVEP' then vlr else 0 end) as BVEP,
        sum(case when nome_veiculo_costsharing = 'BVIA' then vlr else 0 end) as BVIA
    from
        (select
            conta_contabil,
			nome_conta_contabil,
            nm_agrupamento,
            nome_veiculo_costsharing,
            sum(vlr_final) as vlr
         from
            Real_filtro_bankly
         group by
            conta_contabil,
			nome_conta_contabil,
            nm_agrupamento,
            nome_veiculo_costsharing)
    group by
        conta_contabil,
		nome_conta_contabil,
        nm_agrupamento;
quit;

proc sql;
	create table tb_banco_carga as
	select
		tbb.conta_contabil,
		tbb.nome_conta_contabil,
		coalesce(tbb.BVSA,0) - coalesce(tbv.Banco_Votorantim,0) as BVSA ,
		0 as Asset,
		tbb.Bankly,
		tbb.VCS,
		tbb.BVEP,
		tbb.BVIP,
		tbb.BVIA,
		case
         when contas_provisao is null then 'Despesa'
         else 'Provisao'
      end as tipo_conta,
		tbb.nm_agrupamento as tipo_despesa,
		tbb.BVSA as BVSA_antigo,
		tbv.Banco_Votorantim as Banco_bvsa
	from tb_pivot_banco as tbb
	left join tb_pivot_bvsa as tbv
	on tbb.conta_contabil = tbv.conta_contabil
	left join work.cs_depara_prov
	on contas_provisao = tbb.conta_contabil
;quit;

    create table tb_export_esvaziamento_bvsa_t1 as
    select
        LEFT(sub_lote,1) as sub_lote FORMAT $char9.,
        CASE WHEN Banco_votorantim > 0  THEN debito_bvsa ELSE conta_contabil END  AS cta_debito,
        CASE WHEN Banco_votorantim > 0  THEN  conta_contabil ELSE  debito_bvsa END as cta_credito,
		hist_lanc,
        cc_debito,
        cc_credito,
        item_cont,

        CASE WHEN Banco_votorantim > 0 THEN Banco_votorantim ELSE Banco_votorantim * (-1) END as Valor
    from
        tb_pivot_esvaziamento_bd as tbe
    inner join tb_template_esvaziamento_bd as tbt
        on tbe.id = tbt.id
    inner join CS_DE_PARA_CONTA_D_C as tbc
        on tbe.conta_contabil = tbc.credito_bvsa
    where Banco_Votorantim <> 0 and sub_lote = '010001000'
    order by sub_lote asc;
quit;

proc sql;
    create table tb_export_esvaziamento_bvsa_t1 as

        LEFT(sub_lote,1) as sub_lote FORMAT $char9.,
        CASE WHEN Banco_votorantim > 0  THEN debito_bvsa ELSE conta_contabil END  AS cta_debito,

    inner join tb_template_esvaziamento_bd as tbt
