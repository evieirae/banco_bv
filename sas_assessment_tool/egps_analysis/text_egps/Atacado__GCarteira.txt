,A.R_Serasa

PROC SQL; CREATE TABLE TbNuCnpj AS
SELECT
 TODAY()				AS DtInclusao	FORMAT DDMMYY8.
,A.NuCnpjMatriz
,A.NuCnpjRaiz
,MIN(A.FlEsteiraPME)	AS FlPortfolio
,MAX(A.FlAtivo)			AS FlAtivo
,MAX(A.R_Serasa)		AS R_Serasa
,MAX(A.FlGPAtivoCIB) 	AS FlGPAtivoCIB

FROM TbRFCnpj AS A

GROUP BY  A.DtProcessamento,A.NuCnpjMatriz, A.NuCnpjRaiz

;QUIT;RUN;


DATA TbNuCnpj ; SET TbNuCnpj;

	 IF R_Serasa    = 1 AND FlGPAtivoCIB = 1 THEN CdStatus = 7;	
ELSE IF R_Serasa    = 1 AND FlAtivo = 1		 THEN CdStatus = 5;	

ELSE IF FlPortfolio = 1 AND FlAtivo = 1 	 THEN CdStatus = 1; 
ELSE IF FlPortfolio = 1 AND FlAtivo = 0 	 THEN CdStatus = 3; 

ELSE IF R_Serasa    = 1 AND FlAtivo = 0		 THEN CdStatus = 2;

SELECT A.*


,INPUT(B.CdStCadastral		,BEST02.)	AS CdStCadastral			FORMAT BEST02.

,E.Email		 						AS Email					LENGHT = 70

    FROM WORK.RESUMOOCORRENCIAS;

        C.NuCnpjRaiz,
        D.DtConsulta
    FROM DIM_CNPJ C

    ORDER BY C.NuCnpjRaiz, D.DtConsulta;

        RT_QdAcoes

        RT_QdProtesto   = QdProtesto;
