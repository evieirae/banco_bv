	CREATE TABLE WORK.OpAtacado AS

	FROM connection to moriq
	(


	SELECT DISTINCT

		AND A.CdVeiculoLegal = B.CdVeiculoLegal
	JOIN POSICAOCONSOLIDADA.TbEstruturaCliente E
		ON A.CdCnsldrEstruturaClienteContraparte = E.CdCnsldrEstruturaCliente
	INNER JOIN POSICAOCONSOLIDADA.TbGrupo G
		ON B.CdGrupoConciliacao = G.CdGrupo
	WHERE B.CdGrupoConciliacao IN (1,2,3,4,8,9,10,11,12,13,15,16)
	AND B.DtReferencia = &UltimoDiaMes.
	AND B.CdVeiculoLegal = 1
	AND B.CdSituacaoPosicao <> 8
	AND ROUND(B.VrSaldoContaContabil,2) <> 0
	AND (B.NuContaContabil LIKE '1%' OR B.NuContaContabil LIKE '4%')
	AND A.NuCpfCnpjContraparte IN (&CpfCnpjPartesRelacionadasAtacado.)

	UNION


	SELECT DISTINCT
		B.NuContaContabil,
		CONVERT(CHAR(20),B.NuObjetoNegociacao) AS NuObjetoNegociacao,
		E.NmCliente,
		A.NuCpfCnpjContraparte,
		B.DtReferencia,
		B.VrSaldoContaContabil,
		B.DtVencimentoOperacao,
		G.NmGrupo
	FROM POSICAOCONSOLIDADA.TbFatoResultadoPosicao A
	INNER JOIN POSICAOCONSOLIDADA.TbFatoResultadoContabil B
		ON A.CdChaveNegocio = B.CdChaveNegocio
		AND A.NuObjetoNegociacao = B.NuObjetoNegociacao
		AND A.CdCnsldrEstruturaCliente = B.CdCnsldrEstruturaCliente
		AND A.DtReferencia = B.DtReferencia
		AND A.CdCnsldrEstruturaProduto = B.CdCnsldrEstruturaProduto
		AND A.CdVeiculoLegal = B.CdVeiculoLegal
	JOIN POSICAOCONSOLIDADA.TbEstruturaCliente E
		ON A.CdCnsldrEstruturaClienteContraparte = E.CdCnsldrEstruturaCliente
	INNER JOIN POSICAOCONSOLIDADA.TbGrupo G
		ON B.CdGrupoConciliacao = G.CdGrupo
	WHERE B.CdGrupoConciliacao IN (1,2,3,4,8,9,10,11,12,13,15,16)
	AND B.DtReferencia BETWEEN %STR(%'&DtRefMin.%') AND %STR(%'&DtRefMax.%')
	AND B.CdVeiculoLegal = 1
	AND B.CdSituacaoPosicao <> 8
	AND (B.NuContaContabil LIKE '7%' OR B.NuContaContabil LIKE '8%')
	AND B.VrSaldoContaContabil <> 0
	AND A.NuCpfCnpjContraparte IN (&CpfCnpjPartesRelacionadasAtacado.)
 	GROUP BY
 		B.NuContaContabil,
		CONVERT(CHAR(20),B.NuObjetoNegociacao),
		E.NmCliente,
		A.NuCpfCnpjContraparte,
		B.DtVencimentoOperacao,
		G.NmGrupo,
		B.VrSaldoContaContabil,
		B.DtReferencia

	UNION


	SELECT DISTINCT
		B.NuContaContabil,
		CONVERT(CHAR(20),B.NuObjetoNegociacao) AS NuObjetoNegociacao,
		E.NmCliente,
		A.NuCpfCnpjContraparte,
		B.DtReferencia,
		ROUND(B.VrSaldoContaContabil,2) AS VrSaldoContaContabil,
		B.DtVencimentoOperacao,
		G.NmGrupo
	FROM POSICAOCONSOLIDADA.TbFatoResultadoPosicao A
	INNER JOIN POSICAOCONSOLIDADA.TbFatoResultadoContabil B
		ON A.CdChaveNegocio = B.CdChaveNegocio
		AND A.NuObjetoNegociacao = B.NuObjetoNegociacao
		AND A.CdCnsldrEstruturaCliente = B.CdCnsldrEstruturaCliente
		AND A.DtReferencia = B.DtReferencia
		AND A.CdCnsldrEstruturaProduto = B.CdCnsldrEstruturaProduto
		AND A.CdVeiculoLegal = B.CdVeiculoLegal
	JOIN POSICAOCONSOLIDADA.TbEstruturaCliente E
		ON A.CdCnsldrEstruturaClienteContraparte = E.CdCnsldrEstruturaCliente
	INNER JOIN POSICAOCONSOLIDADA.TbGrupo G
		ON B.CdGrupoConciliacao = G.CdGrupo
	WHERE B.CdGrupoConciliacao IN(1,4,12)
	AND B.DtReferencia IN (&UltimoDiaMes.)
	AND B.CdVeiculoLegal = 1
	AND B.CdSituacaoPosicao <> 8
	AND (B.NuContaContabil LIKE '3%' OR B.NuContaContabil LIKE '5%' OR B.NuContaContabil LIKE '9%')
	AND ROUND(B.VrSaldoContaContabil,2)  <> 0
	AND A.NuCpfCnpjContraparte IN (&CpfCnpjPartesRelacionadasAtacado.)
	)
		WHERE NuCpfCnpjContraparte IN (&CpfCnpjPartesRelacionadasAtacado.);;
